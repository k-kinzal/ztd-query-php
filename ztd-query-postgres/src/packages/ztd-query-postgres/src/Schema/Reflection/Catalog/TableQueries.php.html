29        $escapedTableName = str_replace("'", "''", $tableName);
30        return $this->connection->query(
31            'SELECT column_name, data_type, character_maximum_length, '
32            . 'numeric_precision, numeric_scale, is_nullable, column_default, '
33            . 'udt_name, domain_schema, domain_name, is_identity, identity_generation, '
34            . 'is_generated, generation_expression '
35            . 'FROM information_schema.columns '
36            . "WHERE table_schema = current_schema() AND table_name = '" . $escapedTableName . "' "
37            . 'ORDER BY ordinal_position'

44    public function primaryKey(string $tableName): StatementInterface|false
45    {
46        $escapedTableName = str_replace("'", "''", $tableName);
47        return $this->connection->query(
48            'SELECT kcu.column_name '
49            . 'FROM information_schema.table_constraints tc '
50            . 'JOIN information_schema.key_column_usage kcu '
51            . '  ON tc.constraint_name = kcu.constraint_name '
52            . '  AND tc.table_schema = kcu.table_schema '
53            . 'WHERE tc.table_schema = current_schema() '
54            . "  AND tc.table_name = '" . $escapedTableName . "' "
55            . "  AND tc.constraint_type = 'PRIMARY KEY' "
56            . 'ORDER BY kcu.ordinal_position'
57        );
58    }

60    /**
61     * Fetches unique indexes metadata in its catalog order.
62     */
63    public function uniqueIndexes(string $tableName): StatementInterface|false
64    {
65        $escapedTableName = str_replace("'", "''", $tableName);
66        return $this->connection->query(
67            'SELECT index_relation.relname AS constraint_name, attribute.attname AS column_name, '
68            . 'pg_get_expr(index_metadata.indpred, index_metadata.indrelid) AS predicate '
69            . 'FROM pg_catalog.pg_class table_relation '
70            . 'JOIN pg_catalog.pg_namespace namespace ON namespace.oid = table_relation.relnamespace '
71            . 'JOIN pg_catalog.pg_index index_metadata ON index_metadata.indrelid = table_relation.oid '
72            . 'JOIN pg_catalog.pg_class index_relation ON index_relation.oid = index_metadata.indexrelid '
73            . 'JOIN LATERAL unnest(index_metadata.indkey) WITH ORDINALITY key_column(attnum, ordinality) '
74            . '  ON key_column.ordinality <= index_metadata.indnkeyatts '
75            . 'LEFT JOIN pg_catalog.pg_attribute attribute '
76            . '  ON attribute.attrelid = table_relation.oid AND attribute.attnum = key_column.attnum '
77            . 'WHERE namespace.nspname = current_schema() '
78            . "  AND table_relation.relname = '" . $escapedTableName . "' "
79            . '  AND index_metadata.indisunique '
80            . '  AND index_metadata.indisvalid '
81            . '  AND NOT index_metadata.indisprimary '
82            . 'ORDER BY index_relation.relname, key_column.ordinality'
83        );
84    }

92        return $this->connection->query(
93            'SELECT fk.constraint_name, fk.column_name, '
94            . 'pk.table_name AS foreign_table_name, pk.column_name AS foreign_column_name, '
95            . 'rc.update_rule, rc.delete_rule '
96            . 'FROM information_schema.referential_constraints rc '
97            . 'JOIN information_schema.key_column_usage fk '
98            . '  ON fk.constraint_catalog = rc.constraint_catalog '
99            . '  AND fk.constraint_schema = rc.constraint_schema '
100            . '  AND fk.constraint_name = rc.constraint_name '
101            . 'JOIN information_schema.key_column_usage pk '
102            . '  ON pk.constraint_catalog = rc.unique_constraint_catalog '
103            . '  AND pk.constraint_schema = rc.unique_constraint_schema '
104            . '  AND pk.constraint_name = rc.unique_constraint_name '
105            . '  AND pk.ordinal_position = fk.position_in_unique_constraint '
106            . 'WHERE fk.table_schema = current_schema() '
107            . "  AND fk.table_name = '" . $escapedTableName . "' "
108            . 'ORDER BY fk.constraint_name, fk.ordinal_position'
109        );
110    }
