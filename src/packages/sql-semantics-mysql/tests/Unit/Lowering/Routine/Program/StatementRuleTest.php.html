56    #[DataProvider('providerStatementRoutesEveryKindOfProgramStatement')]
57    public function testStatementRoutesEveryKindOfProgramStatement(string $release): void
58    {
59        $operation = (new Semantics(Dialect::MySql, $release))->analyze('create function f(x int) returns int begin declare a int; declare k cursor for select 1; set a = 1; if x then set a = 1; end if; case x when 1 then set a = 2; end case; b: begin leave b; end b; begin end; l: loop iterate l; end loop; while 0 do set a = 0; end while; open k; fetch k into a; close k; return a; end');
60        $statement = $operation->statement;
61        self::assertInstanceOf(CreateFunction::class, $statement);
62        self::assertInstanceOf(Block::class, $statement->body);
63
64        self::assertSame([
65            SetVariables::class, IfStatement::class, SimpleCase::class, Block::class, Block::class, Loop::class, WhileLoop::class,
66            OpenCursor::class, FetchCursor::class, CloseCursor::class, ReturnStatement::class,
67        ], array_map(static fn (StatementNode $node): string => $node::class, $statement->body->statements));
68        self::assertSame('CREATE FUNCTION f(x INT) RETURNS INT BEGIN DECLARE a INT; DECLARE k CURSOR FOR SELECT 1; SET a = 1; IF x THEN SET a = 1; END IF; CASE x WHEN 1 THEN SET a = 2; END CASE; b: BEGIN LEAVE b; END b; BEGIN END; l: LOOP ITERATE l; END LOOP; WHILE 0 DO SET a = 0; END WHILE; OPEN k; FETCH k INTO a; CLOSE k; RETURN a; END', $operation->toString());
69        self::assertSame([], $operation->facts->diagnostics);
70    }

176    #[DataProvider('providerBodyLowersAnExternalBody')]
177    public function testBodyLowersAnExternalBody(string $release): void
178    {
179        $semantics = new Semantics(Dialect::MySql, $release);
180        $function = $semantics->analyze("create function f() returns int language javascript as 'return 1'");
181        self::assertInstanceOf(CreateFunction::class, $function->statement);
182        self::assertEquals(new ExternalBody(new Text('return 1')), $function->statement->body);
183        self::assertSame("CREATE FUNCTION f() RETURNS INT LANGUAGE javascript AS 'return 1'", $function->toString());
184
185        $procedure = $semantics->analyze('create procedure p() language javascript as $$ return 1 $$');
186        self::assertInstanceOf(CreateProcedure::class, $procedure->statement);
187        self::assertEquals(new ExternalBody(new DollarQuotedText(' return 1 ')), $procedure->statement->body);
188        self::assertSame('CREATE PROCEDURE p() LANGUAGE javascript AS $$ return 1 $$', $procedure->toString());
