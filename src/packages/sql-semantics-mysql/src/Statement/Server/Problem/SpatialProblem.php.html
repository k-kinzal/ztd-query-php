40            SpatialRule::IdentifierOutOfRange => $subject . ' is out of range: it is at most 4294967295',
41            SpatialRule::IdentifierZero => 'SRID 0 cannot be modified',
42            SpatialRule::RepeatedAttribute => 'Multiple definitions of attribute ' . $subject,
43            SpatialRule::MissingAttribute => 'Missing mandatory attribute ' . $subject,
44            SpatialRule::BlankName => 'The spatial reference system name cannot be an empty string or start or end with whitespace',
45            SpatialRule::BlankOrganization => 'The organization name cannot be an empty string or start or end with whitespace',
46            SpatialRule::ControlCharacter => 'Invalid character in attribute ' . $subject,
47            SpatialRule::TooLong => 'Attribute ' . $subject . ' is too long',
