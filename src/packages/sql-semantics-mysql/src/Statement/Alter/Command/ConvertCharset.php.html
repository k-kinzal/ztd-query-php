17 *
18 * Mirrors PT_alter_table_convert_to_charset. DEFAULT means the character
19 * set of the database. CHARACTER SET and CHARSET are synonyms; CHARACTER
20 * SET is written.
21 * Source: https://dev.mysql.com/doc/refman/8.4/en/alter-table.html#alter-table-character-set.
22 *
23 * @visibility public
24 * @example Converting a table to utf8mb4
25 *     $alter = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\MySql\Dialect::MySql))->analyze('ALTER TABLE t CONVERT TO CHARSET utf8mb4');
26 *     [$alter->statement->commands[0]->charset->name?->value, $alter->toString()] // => ['utf8mb4', 'ALTER TABLE t CONVERT TO CHARACTER SET utf8mb4']
27 */
