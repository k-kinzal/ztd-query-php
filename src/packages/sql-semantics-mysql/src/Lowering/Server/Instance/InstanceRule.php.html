28/**
29 * Lowers KILL, SHUTDOWN, RESTART, CLONE and ALTER INSTANCE.
30 *
31 * Rule: MYSQL-INSTANCE-001. Scope: kill, kill_option, shutdown_stmt,
32 * restart_server_stmt, clone_stmt, opt_datadir_ssl, opt_ssl,
33 * alter_instance_stmt, alter_instance_action. The KILL identifier goes
34 * through the expression family. The equals sign of DATA DIRECTORY is
35 * optional (LeafNoise). The server raises a syntax error in the grammar
36 * actions, rejected as such here, when a space surrounds the colon between
37 * the CLONE account and port, when ROTATE names another key than INNODB
38 * (or, from MySQL 8.0, BINLOG), and when ENABLE or DISABLE names other than
39 * INNODB REDO_LOG. Constructs: Kill, Shutdown, Restart, CloneLocal,
40 * CloneInstance, AlterInstance with RotateMasterKey, ReloadTls,
41 * RedoLogSwitch, ReloadKeyring. Terminates: every child is a strict
42 * subtree.
43 * Source: https://dev.mysql.com/doc/refman/8.4/en/kill.html,
44 * https://dev.mysql.com/doc/refman/8.4/en/clone.html,
45 * https://dev.mysql.com/doc/refman/8.4/en/alter-instance.html.
46 * Status: Implemented.
47 *
48 * @visibility SqlSemantics\Platform\MySql\Lowering\Server
49 */

69     * Lowers an instance statement.
70     *
71     * @throws ImplementationGap When a production has no rule
72     * @throws AnalysisException When the server rejects the statement while it parses it
73     */
74    public function statement(Form $form): Statement
75    {
76        return match ($form->signature) {
77            'kill: KILL_SYM kill_option expr' => new Kill($this->scope($form->node(1)), $this->lowering->expressions->expression($form->node(2))),
78            'shutdown_stmt: SHUTDOWN' => new Shutdown(),
79            'restart_server_stmt: RESTART_SYM' => new Restart(),
80            'clone_stmt: CLONE_SYM LOCAL_SYM DATA_SYM DIRECTORY_SYM opt_equal TEXT_STRING_filesystem' => $this->local($form),
81            'clone_stmt: CLONE_SYM INSTANCE_SYM FROM user : ulong_num IDENTIFIED_SYM BY TEXT_STRING_sys opt_datadir_ssl' => $this->remote($form),
82            'alter_instance_stmt: ALTER INSTANCE_SYM alter_instance_action' => new AlterInstance($this->action($form->node(2))),
83            default => throw ImplementationGap::production($form),
84        };
85    }

117     * Lowers CLONE INSTANCE FROM.
118     *
119     * @throws ImplementationGap When a production has no rule
120     * @throws AnalysisException When a space surrounds the colon before the port
121     */
122    public function remote(Form $form): CloneInstance
123    {
124        $port = $form->node(5)->tokens();
125        if ($form->token(4)->leading !== '' || ($port[0] ?? null)?->leading !== '') {
126            throw new AnalysisException('Syntax error: CLONE INSTANCE writes the account, the colon and the port without spaces.');
127        }
128        $donor = $this->lowering->users->account($form->node(3));
129        $number = $this->lowering->numbers->numeral($form->node(5));
130        $password = $this->lowering->literals->text($form->node(8));
131        $options = $this->lowering->form($form->node(9));
132        [$directory, $ssl] = match ($options->signature) {
133            'opt_datadir_ssl: opt_ssl' => [null, $this->ssl($options->node(0))],
134            'opt_datadir_ssl: DATA_SYM DIRECTORY_SYM opt_equal TEXT_STRING_filesystem opt_ssl' => $this->directory($options),
135            default => throw ImplementationGap::production($options),
136        };
137
138        return new CloneInstance($donor, $number, $password, $directory, $ssl);
139    }

171    /**
172     * Lowers the action of ALTER INSTANCE: a node of `alter_instance_action`.
173     *
174     * @throws ImplementationGap When a production has no rule
175     * @throws AnalysisException When a name is not the one the server expects
176     */
177    public function action(Node $action): InstanceAction
178    {
179        $form = $this->lowering->form($action);
180        if (isset(self::RELOADS[$form->signature])) {
181            [$channel, $noRollback] = self::RELOADS[$form->signature];
182
183            return new ReloadTls($channel === null ? null : $this->lowering->names->identifier($form->node($channel)), $noRollback);
184        }
185
186        return match ($form->signature) {
187            'alter_instance_action: ROTATE_SYM ident_or_text MASTER_SYM KEY_SYM' => $this->rotate($form->node(1)),
188            'alter_instance_action: ENABLE_SYM ident ident' => $this->redo(true, $form),
189            'alter_instance_action: DISABLE_SYM ident ident' => $this->redo(false, $form),
190            'alter_instance_action: RELOAD KEYRING_SYM' => new ReloadKeyring(),
191            default => throw ImplementationGap::production($form),
192        };
193    }

195    /**
196     * Lowers ROTATE … MASTER KEY from its name: a node of `ident_or_text`.
197     *
198     * @throws ImplementationGap When a production has no rule
199     * @throws AnalysisException When the name is not one the release accepts
200     */
201    public function rotate(Node $name): RotateMasterKey
202    {
203        $keyring = $this->lowering->names->identifier($name);
204        $accepted = $this->lowering->profile->grammar === GrammarRelease::MySql5744 ? ['INNODB'] : ['INNODB', 'BINLOG'];
205        if (!in_array(strtoupper($keyring->value), $accepted, true)) {
206            throw new AnalysisException('Syntax error: ALTER INSTANCE ROTATE names ' . implode(' or ', $accepted) . ', not ' . $keyring->value . '.');
207        }
208
209        return new RotateMasterKey($keyring);
210    }

212    /**
213     * Lowers ENABLE or DISABLE INNODB REDO_LOG.
214     *
215     * @throws ImplementationGap When a production has no rule
216     * @throws AnalysisException When the names are not INNODB REDO_LOG
217     */
218    public function redo(bool $enable, Form $form): RedoLogSwitch
219    {
220        $engine = $this->lowering->names->identifier($form->node(1));
221        $log = $this->lowering->names->identifier($form->node(2));
222        if (strtoupper($engine->value) !== 'INNODB' || strtoupper($log->value) !== 'REDO_LOG') {
223            throw new AnalysisException('Syntax error: ALTER INSTANCE ' . ($enable ? 'ENABLE' : 'DISABLE') . ' names INNODB REDO_LOG.');
224        }
225
226        return new RedoLogSwitch($enable, $engine, $log);
227    }
