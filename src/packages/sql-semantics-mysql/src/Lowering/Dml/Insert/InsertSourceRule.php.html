15/**
16 * Lowers what INSERT and REPLACE write: the column list and the rows of VALUES or the query.
17 *
18 * Rule: MYSQL-INSERT-SOURCE-LOWERING-001. Scope: insert_field_spec,
19 * insert_from_constructor, insert_from_subquery, insert_query_expression,
20 * insert_values, values_list, no_braces, row_value, value_or_values, fields,
21 * insert_ident, insert_columns, insert_column. A column list is
22 * ColumnList, also when it is empty; rows are InsertedRow; a query source
23 * is lowered by the query family, the 5.x forms through legacyQuery and
24 * legacyParenthesizedQuery. The answer is the column list with either the
25 * rows or the query. Terminates: lists are flattened by MYSQL-DML-LIST-001;
26 * the parts are strict subtrees. Source:
27 * https://dev.mysql.com/doc/refman/8.4/en/insert.html,
28 * https://dev.mysql.com/doc/refman/5.7/en/insert.html. Status: Implemented.
29 *
30 * @visibility SqlSemantics\Platform\MySql\Lowering\Dml
31 */
32final class InsertSourceRule
33{
34    /**
35     * The productions that write a column list before their source, by the position of the list.
36     */
37    private const LISTED = [
38        'insert_field_spec: insert_values' => null, 'insert_field_spec: ( ) insert_values' => -1, 'insert_field_spec: ( fields ) insert_values' => 1,
39        'insert_from_constructor: insert_values' => null, 'insert_from_constructor: ( ) insert_values' => -1,
40        'insert_from_constructor: ( fields ) insert_values' => 1, 'insert_from_constructor: ( insert_columns ) insert_values' => 1,
41        'insert_from_subquery: insert_query_expression' => null, 'insert_from_subquery: ( ) insert_query_expression' => -1,
42        'insert_from_subquery: ( fields ) insert_query_expression' => 1,
43        'insert_query_expression: query_expression_with_opt_locking_clauses' => null,
44        'insert_query_expression: ( ) query_expression_with_opt_locking_clauses' => -1,
45        'insert_query_expression: ( insert_columns ) query_expression_with_opt_locking_clauses' => 1,
46    ];
47
48    /**
49     * The productions of the column lists.
50     */
51    private const COLUMN_LISTS = ['fields: fields , insert_ident', 'fields: insert_ident', 'insert_columns: insert_columns , insert_column', 'insert_columns: insert_column'];
52
53    /**
54     * The productions of the row lists.
55     */
56    private const ROW_LISTS = ['values_list: values_list , no_braces', 'values_list: no_braces', 'values_list: values_list , row_value', 'values_list: row_value'];
57
58    /**
59     * @param Lowering $lowering The lowering this rule belongs to
60     */
61    public function __construct(private readonly Lowering $lowering)
62    {
63    }
64
65    /**
66     * Lowers a source with its optional column list: a node of `insert_field_spec`, `insert_from_constructor`, `insert_from_subquery` or `insert_query_expression`.
67     *
68     * @return array{ColumnList|null, list<InsertedRow>|Query}
69     * @throws ImplementationGap When a production has no rule
70     */
71    public function source(Node $source): array
72    {
73        $form = $this->lowering->form($source);
74        if (!array_key_exists($form->signature, self::LISTED)) {
75            return [null, $this->body($source)];
76        }
77        $position = self::LISTED[$form->signature];
78        $columns = $position === null ? null : new ColumnList($position === -1 ? [] : $this->columns($form->node($position)));
79        $body = $form->node->children[count($form->node->children) - 1];
80        if (!$body instanceof Node) {
81            throw ImplementationGap::production($form);
82        }
83        if ($body->name === 'query_expression_with_opt_locking_clauses') {
84            return [$columns, $this->lowering->queries->query($body)];
85        }
86        $inner = $this->source($body);
87        if ($inner[0] !== null) {
88            throw ImplementationGap::production($form);
89        }
90
91        return [$columns, $inner[1]];
92    }
93
94    /**
95     * Lowers rows or a query without a column list: a node of `insert_values` or `insert_query_expression` (5.7).
96     *
97     * @return list<InsertedRow>|Query
98     * @throws ImplementationGap When a production has no rule
99     */
100    public function body(Node $body): array|Query
101    {
102        $form = $this->lowering->form($body);
103
104        return match ($form->signature) {
105            'insert_values: VALUES values_list', 'insert_values: VALUE_SYM values_list' => $this->rows($form->node(1)),
106            'insert_values: value_or_values values_list' => $this->valued($form->node(0), $form->node(1)),
107            'insert_values: create_select union_clause', 'insert_query_expression: create_select opt_union_clause' => $this->lowering->queries->legacyQuery($form->node(0), $form->node(1)),
108            'insert_values: ( create_select ) union_opt', 'insert_query_expression: ( create_select ) union_opt' => $this->lowering->queries->legacyParenthesizedQuery($form->node(1), $form->node(3)),
109            default => throw ImplementationGap::production($form),
110        };
111    }
112
113    /**
114     * Lowers the rows after the keyword VALUE or VALUES: a node of `value_or_values` and a node of `values_list`.
115     *
116     * @return list<InsertedRow>
117     * @throws ImplementationGap When a production has no rule
118     */
119    public function valued(Node $keyword, Node $list): array
120    {
121        $form = $this->lowering->form($keyword);
122        if ($form->signature !== 'value_or_values: VALUE_SYM' && $form->signature !== 'value_or_values: VALUES') {
123            throw ImplementationGap::production($form);
124        }
125
126        return $this->rows($list);
127    }
128
129    /**
130     * Lowers the rows of a node of `values_list`.
131     *
132     * @return list<InsertedRow>
133     * @throws ImplementationGap When a production has no rule
134     */
135    public function rows(Node $list): array
136    {
137        $rows = [];
138        foreach ((new ListRule($this->lowering))->items($list, self::ROW_LISTS) as $item) {
139            $row = $this->lowering->form($item);
140            if ($row->signature !== 'no_braces: ( opt_values )' && $row->signature !== 'row_value: ( opt_values )') {
141                throw ImplementationGap::production($row);
142            }
143            $rows[] = new InsertedRow($this->lowering->dml->rowValues($row->node(1)));
144        }
145
146        return $rows;
147    }
148
149    /**
150     * Lowers the columns of a node of `fields` or `insert_columns`.
151     *
152     * @return list<\SqlSemantics\Platform\MySql\Statement\Name\ColumnUse|\SqlSemantics\Platform\MySql\Statement\Name\TableWildcard>
153     * @throws ImplementationGap When a production has no rule
154     */
155    public function columns(Node $list): array
156    {
157        $columns = [];
158        foreach ((new ListRule($this->lowering))->items($list, self::COLUMN_LISTS) as $item) {
159            $column = $this->lowering->form($item);
160            $columns[] = match ($column->signature) {
161                'insert_ident: simple_ident_nospvar', 'insert_column: simple_ident_nospvar' => $this->lowering->names->column($column->node(0)),
162                'insert_ident: table_wild' => $this->lowering->names->wildcard($column->node(0)),
163                default => throw ImplementationGap::production($column),
164            };
165        }
166
167        return $columns;
168    }
