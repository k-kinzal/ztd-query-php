22/**
23 * Lowers the bracketed, subquery and column-function forms of simple_expr.
24 *
25 * Rule: MYSQL-SIMPLE-FORM-001. Scope: the simple_expr productions for a
26 * grouping `( expr )`, a row `( expr , expr_list )` and `ROW ( … )` (the
27 * keyword ROW is optional), a subquery `( subselect )` or row_subquery,
28 * EXISTS, the ODBC escape `{ ident expr }`, `DEFAULT ( col )`,
29 * `VALUES ( col )`, the leading `INTERVAL expr unit + expr` and the JSON
30 * operators `->` and `->>`; every other production goes to
31 * MYSQL-CONSTRUCT-001. Constructs: Grouped, Row, ScalarSubquery, Exists,
32 * OdbcEscape, DefaultOfColumn, InsertedColumn, IntervalAddition,
33 * JsonExtraction. Terminates: every child is a strict subtree.
34 * Source: https://dev.mysql.com/doc/refman/8.4/en/expressions.html.
35 * Status: Implemented.
36 *
37 * @visibility SqlSemantics\Platform\MySql
38 */
39final class FormRule
40{
41    /**
42     * @param Lowering $lowering The lowering this rule belongs to
43     */
44    public function __construct(private readonly Lowering $lowering)
45    {
46    }
47
48    /**
49     * Lowers one of the forms.
50     *
51     * @throws ImplementationGap When a production has no rule
52     */
53    public function lower(Form $form): Scalar
54    {
55        $expressions = $this->lowering->expressions;
56        $names = $this->lowering->names;
57        $queries = $this->lowering->queries;
58
59        return match ($form->signature) {
60            'simple_expr: ( expr )' => new Grouped($expressions->expression($form->node(1))),
61            'simple_expr: ( expr , expr_list )' => new Row([$expressions->expression($form->node(1)), ...$expressions->expressions($form->node(3))]),
62            'simple_expr: ROW_SYM ( expr , expr_list )' => new Row([$expressions->expression($form->node(2)), ...$expressions->expressions($form->node(4))], OptionalWords::Written),
63            'simple_expr: ( subselect )' => new ScalarSubquery($queries->query($form->node(1))),
64            'simple_expr: row_subquery' => new ScalarSubquery($queries->query($form->node(0))),
65            'simple_expr: EXISTS ( subselect )' => new Exists($queries->query($form->node(2))),
66            'simple_expr: EXISTS table_subquery' => new Exists($queries->query($form->node(1))),
67            'simple_expr: { ident expr }' => new OdbcEscape($names->identifier($form->node(1)), $expressions->expression($form->node(2))),
68            'simple_expr: DEFAULT ( simple_ident )', 'simple_expr: DEFAULT_SYM ( simple_ident )' => new DefaultOfColumn($names->column($form->node(2))),
69            'simple_expr: VALUES ( simple_ident_nospvar )' => new InsertedColumn($names->column($form->node(2))),
70            'simple_expr: INTERVAL_SYM expr interval + expr' => new IntervalAddition($expressions->interval($form->node(1), $form->node(2)), $expressions->expression($form->node(4))),
71            'simple_expr: simple_ident JSON_SEPARATOR_SYM TEXT_STRING_literal' => new JsonExtraction($names->column($form->node(0)), $this->lowering->literals->text($form->node(2))),
72            'simple_expr: simple_ident JSON_UNQUOTED_SEPARATOR_SYM TEXT_STRING_literal' => new JsonExtraction($names->column($form->node(0)), $this->lowering->literals->text($form->node(2)), true),
73            default => (new ConstructRule($this->lowering))->lower($form),
74        };
75    }
76}
