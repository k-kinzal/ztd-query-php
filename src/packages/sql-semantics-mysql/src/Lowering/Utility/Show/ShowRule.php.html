26/**
27 * Lowers the SHOW statements of MySQL 8.0 and later: one rule per statement.
28 *
29 * Rule: MYSQL-SHOW-LOWERING-001. Scope: every `show_*_stmt` rule. A rule
30 * lowers into the class of its command; the statements about schema
31 * objects, stored programs and the server are lowered by
32 * MYSQL-SHOW-OBJECT-LOWERING-001, the clauses by
33 * MYSQL-SHOW-CLAUSE-LOWERING-001. SLAVE and REPLICA, and MASTER STATUS and
34 * BINARY LOG STATUS, are kept as written (Terminology). SHOW PARSE_TREE
35 * lowers the statement it wraps through the statement dispatcher.
36 * Terminates: every part is a strict subtree.
37 * Source: https://dev.mysql.com/doc/refman/8.4/en/show.html. Status: Implemented.
38 *
39 * @visibility SqlSemantics\Platform\MySql\Lowering\Utility
40 */
41final class ShowRule
42{
43    /**
44     * @param Lowering $lowering The lowering this rule belongs to
45     */
46    public function __construct(private readonly Lowering $lowering)
47    {
48    }
49
50    /**
51     * Lowers a node of one of the `show_*_stmt` rules.
52     *
53     * @throws ImplementationGap When a production has no rule
54     */
55    public function statement(Form $form): Statement
56    {
57        $clauses = new ClauseRule($this->lowering);
58        switch ($form->signature) {
59            case 'show_warnings_stmt: SHOW WARNINGS opt_limit_clause':
60                return new ShowWarnings($this->lowering->queries->limit($form->node(2)));
61            case 'show_errors_stmt: SHOW ERRORS opt_limit_clause':
62                return new ShowErrors($this->lowering->queries->limit($form->node(2)));
63            case 'show_count_warnings_stmt: SHOW COUNT_SYM ( * ) WARNINGS':
64                return new ShowWarningCount();
65            case 'show_count_errors_stmt: SHOW COUNT_SYM ( * ) ERRORS':
66                return new ShowErrorCount();
67            case 'show_profiles_stmt: SHOW PROFILES_SYM':
68                return new ShowProfiles();
69            case 'show_profile_stmt: SHOW PROFILE_SYM opt_profile_defs opt_for_query opt_limit_clause':
70                return new ShowProfile($clauses->sections($form->node(2)), $clauses->query($form->node(3)), $this->lowering->queries->limit($form->node(4)));
71            case 'show_parse_tree_stmt: SHOW PARSE_TREE_SYM simple_statement':
72                return new ShowParseTree($this->lowering->statement($form->node(2)));
73            default:
74                return $this->replication($form);
75        }
76    }
77
78    /**
79     * Lowers the SHOW statements about binary logs and replication.
80     *
81     * @throws ImplementationGap When a production has no rule
82     */
83    public function replication(Form $form): Statement
84    {
85        $clauses = new ClauseRule($this->lowering);
86        switch ($form->signature) {
87            case 'show_binary_logs_stmt: SHOW master_or_binary LOGS_SYM':
88                $this->lowering->utility->binaryLogsWord($form->node(1));
89
90                return new ShowBinaryLogs();
91            case 'show_binary_logs_stmt: SHOW BINARY_SYM LOGS_SYM':
92                return new ShowBinaryLogs();
93            case 'show_replicas_stmt: SHOW SLAVE HOSTS_SYM':
94                return new ShowReplicas(Terminology::Legacy);
95            case 'show_replicas_stmt: SHOW REPLICAS_SYM':
96                return new ShowReplicas(Terminology::Current);
97            case 'show_binlog_events_stmt: SHOW BINLOG_SYM EVENTS_SYM opt_binlog_in binlog_from opt_limit_clause':
98                return new ShowBinlogEvents($clauses->file($form->node(3)), $clauses->position($form->node(4)), $this->lowering->queries->limit($form->node(5)));
99            case 'show_relaylog_events_stmt: SHOW RELAYLOG_SYM EVENTS_SYM opt_binlog_in binlog_from opt_limit_clause opt_channel':
100                return new ShowRelaylogEvents($clauses->file($form->node(3)), $clauses->position($form->node(4)), $this->lowering->queries->limit($form->node(5)), $this->lowering->replication->channel($form->node(6)));
101            case 'show_replica_status_stmt: SHOW replica STATUS_SYM opt_channel':
102                return new ShowReplicaStatus($this->lowering->replication->terminology($form->node(1)), $this->lowering->replication->channel($form->node(3)));
103            case 'show_replica_status_stmt: SHOW REPLICA_SYM STATUS_SYM opt_channel':
104                return new ShowReplicaStatus(Terminology::Current, $this->lowering->replication->channel($form->node(3)));
105            case 'show_master_status_stmt: SHOW MASTER_SYM STATUS_SYM':
106                return new ShowBinaryLogStatus(Terminology::Legacy);
107            case 'show_binary_log_status_stmt: SHOW BINARY_SYM LOG_SYM STATUS_SYM':
108                return new ShowBinaryLogStatus(Terminology::Current);
109            default:
110                return (new ObjectRule($this->lowering))->statement($form);
111        }
112    }
