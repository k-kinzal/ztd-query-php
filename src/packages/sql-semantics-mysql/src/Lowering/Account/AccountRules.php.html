14/**
15 * The entry rules of the account family: the methods other families and the statement dispatcher call.
16 *
17 * Rule: MYSQL-ACCOUNT-ENTRY-001. Scope: users, roles, privileges, passwords
18 * and resource groups. A statement node is handed to the rule of its
19 * nonterminal: GRANT and REVOKE by generation (MYSQL-ACCOUNT-GRANT-001,
20 * MYSQL-ACCOUNT-GRANT-LEGACY-001), the user statements
21 * (MYSQL-ACCOUNT-USER-001), the role statements (MYSQL-ACCOUNT-ROLE-001) and
22 * the resource group statements (MYSQL-ACCOUNT-RESOURCE-GROUP-001); SET
23 * PASSWORD comes from the SET rules (MYSQL-ACCOUNT-PASSWORD-001).
24 * Terminates: each method delegates once to a rule over a strict subtree.
25 * Source: https://dev.mysql.com/doc/refman/8.4/en/account-management-statements.html.
26 * Status: Implemented.
27 *
28 * @visibility SqlSemantics\Platform\MySql
29 */
30final class AccountRules
31{
32    /**
33     * The 5.6 and 5.7 GRANT and REVOKE productions, which the legacy rule lowers.
34     */
35    private const LEGACY = ['grant: GRANT clear_privileges grant_command' => true, 'revoke: REVOKE clear_privileges revoke_command' => true];
36
37    /**
38     * @param Lowering $lowering The lowering this rule belongs to
39     */
40    public function __construct(public readonly Lowering $lowering)
41    {
42    }

44    /**
45     * Lowers an account management statement: a node of one of the statement rules this family owns.
46     *
47     * @throws ImplementationGap When the production has no rule
48     */
49    public function statement(Node $statement): Statement
50    {
51        $form = $this->lowering->form($statement);
52        if (isset(self::LEGACY[$form->signature])) {
53            $legacy = new LegacyGrantRule($this->lowering);
54
55            return $statement->name === 'grant' ? $legacy->grant($form) : $legacy->revoke($form);
56        }
57
58        return match ($statement->name) {
59            'grant' => (new GrantRule($this->lowering))->grant($form),
60            'revoke' => (new GrantRule($this->lowering))->revoke($form),
61            'alter_user_stmt' => (new UserStatementRule($this->lowering))->alter($form),
62            'drop_user_stmt' => (new UserStatementRule($this->lowering))->drop($form),
63            'create_role_stmt', 'drop_role_stmt', 'set_role_stmt' => (new RoleRule($this->lowering))->statement($form),
64            'create_resource_group_stmt', 'alter_resource_group_stmt', 'drop_resource_group_stmt', 'set_resource_group_stmt' => (new ResourceGroupRule($this->lowering))->statement($form),
65            default => throw ImplementationGap::production($form),
66        };
67    }
68
69    /**
70     * Lowers a production of `create`, `alter` or `drop` that MYSQL-DEFINITION-ROUTES-001 routes to this
71     * family.
72     *
73     * @throws ImplementationGap When the production has no rule
74     */
75    public function definition(Form $form): Statement
76    {
77        return (new UserStatementRule($this->lowering))->definition($form);
78    }

82     *
83     * @throws ImplementationGap When a production has no rule
84     */

92     * whose first symbol is PASSWORD.
93     *
94     * @throws ImplementationGap When the production has no rule
95     */
96    public function setPassword(Form $form): Statement

104     * @throws ImplementationGap When the production has no rule
