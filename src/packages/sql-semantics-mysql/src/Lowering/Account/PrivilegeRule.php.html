28/**
29 * Lowers the privilege lists, column lists and privilege levels of GRANT and REVOKE.
30 *
31 * Rule: MYSQL-ACCOUNT-PRIVILEGE-001. Scope: role_or_privilege_list,
32 * role_or_privilege (8.0+), grant_privileges, object_privilege_list,
33 * object_privilege (5.x), opt_privileges, opt_column_list, column_list,
34 * column_list_id, grant_ident, opt_acl_type. A static privilege maps to its
35 * PrivilegeKind by the words written; a name alone is a role in a role list
36 * and a dynamic privilege in a privilege list, as the server reads
37 * PT_role_or_dynamic_privilege; a name with a column list is a dynamic
38 * privilege (PT_dynamic_privilege); a name with a host is a role
39 * (PT_role_at_host). ALL [PRIVILEGES] is AllPrivileges. The level is `*.*`,
40 * `*`, `db.*` or an object name; an absent or TABLE kind is a table.
41 * Constructs: StaticPrivilege, DynamicPrivilege, GrantedRole, AllPrivileges,
42 * GlobalLevel, CurrentDatabaseLevel, DatabaseLevel, ObjectLevel.
43 * Terminates: lists are flattened iteratively; every other child is a strict
44 * subtree. Source: https://dev.mysql.com/doc/refman/8.4/en/grant.html.
45 * Status: Implemented.
46 *
47 * @visibility SqlSemantics\Platform\MySql
48 */

136    /**
137     * Lowers the items of an 8.0 role_or_privilege_list, reading a name alone as a role or as a dynamic privilege.
138     *
139     * @return non-empty-list<Grantable>
140     * @throws ImplementationGap When a production has no rule
141     */
142    public function items(Node $list, bool $roles): array
143    {
144        $items = [];
145        foreach ($this->spine($list) as $item) {
146            $items[] = $this->item($item, $roles);
147        }
148        Check::invariant($items !== [], 'A privilege list holds at least one item.');
149
150        return $items;
151    }

153    /**
154     * Lowers one role_or_privilege or object_privilege.
155     *
156     * @throws ImplementationGap When the production has no rule
157     */
158    public function item(Node $item, bool $roles): Grantable
159    {
160        $form = $this->lowering->form($item);
161        if ($item->name !== 'role_or_privilege' && $item->name !== 'object_privilege') {
162            throw ImplementationGap::production($form);
163        }
164        $kind = self::STATIC[$form->signature] ?? null;
165        if ($kind !== null) {
166            return new StaticPrivilege($kind, $kind->columns() ? $this->columns($form->node(1)) : []);
167        }
168        $names = $this->lowering->names;
169
170        return match ($form->signature) {
171            'role_or_privilege: role_ident_or_text @ ident_or_text' => new GrantedRole($this->lowering->leaves->record(new AccountName($names->identifier($form->node(0)), $names->identifier($form->node(2))))),
172            'role_or_privilege: role_ident_or_text opt_column_list' => $this->named($names->identifier($form->node(0)), $this->columns($form->node(1)), $roles),
173            default => throw ImplementationGap::production($form),
174        };
175    }

177    /**
178     * Reads a name with its column list: a role in a role list when no columns are written, a dynamic privilege otherwise.
179     *
180     * @param list<Name> $columns
181     */
182    public function named(Name $name, array $columns, bool $roles): Grantable
183    {
184        return $roles && $columns === [] ? new GrantedRole($this->lowering->leaves->record(new AccountName($name))) : new DynamicPrivilege($name, $columns);
185    }

187    /**
188     * Lowers a MySQL 5.x grant_privileges: ALL [PRIVILEGES] or a list of privileges.
189     *
190     * @return non-empty-list<Grantable>
191     * @throws ImplementationGap When a production has no rule
192     */
193    public function privileges(Node $privileges): array
194    {
195        $form = $this->lowering->form($privileges);
196        if ($form->signature === 'grant_privileges: ALL opt_privileges') {
197            $this->words($form->node(1));
198
199            return [new AllPrivileges()];
200        }
201        if ($form->signature !== 'grant_privileges: object_privilege_list') {
202            throw ImplementationGap::production($form);
203        }
204
205        return $this->items($form->node(0), false);
206    }

221    /**
222     * Lowers an opt_column_list; an absent list is empty.
223     *
224     * @return list<Name>
225     * @throws ImplementationGap When a production has no rule
226     */
227    public function columns(Node $optional): array
228    {
229        $form = $this->lowering->form($optional);
230        if ($form->signature === 'opt_column_list:') {
231            return [];
232        }
233        if ($form->signature !== 'opt_column_list: ( column_list )') {
234            throw ImplementationGap::production($form);
235        }
236        $columns = [];
237        foreach ($this->spine($form->node(1)) as $column) {
238            $columns[] = $this->column($column);
239        }
240
241        return $columns;
242    }

244    /**
245     * Lowers one column of a column list.
246     *
247     * @throws ImplementationGap When the production has no rule
248     */
249    public function column(Node $column): Name
250    {
251        if ($column->name !== 'column_list_id') {
252            return $this->lowering->names->identifier($column);
253        }
254        $form = $this->lowering->form($column);
255        if ($form->signature !== 'column_list_id: ident') {
256            throw ImplementationGap::production($form);
257        }
258
259        return $this->lowering->names->identifier($form->node(0));
260    }

262    /**
263     * Lowers a grant_ident.
264     *
265     * @throws ImplementationGap When the production has no rule
266     */
267    public function level(Node $level): PrivilegeLevel
268    {
269        $form = $this->lowering->form($level);
270        $names = $this->lowering->names;
271
272        return match ($form->signature) {
273            'grant_ident: * . *' => new GlobalLevel(),
274            'grant_ident: *' => new CurrentDatabaseLevel(),
275            'grant_ident: ident . *', 'grant_ident: schema . *' => new DatabaseLevel($names->identifier($form->node(0))),
276            'grant_ident: ident' => new ObjectLevel(new QualifiedName($names->identifier($form->node(0)))),
277            'grant_ident: schema . ident' => new ObjectLevel(new QualifiedName($names->identifier($form->node(2)), $names->identifier($form->node(0)))),
278            'grant_ident: table_ident' => new ObjectLevel($names->qualified($form->node(0))),
279            default => throw ImplementationGap::production($form),
280        };
281    }

303     * @return list<Node>
304     * @throws ImplementationGap When the production has no rule

306    public function spine(Node $list): array
307    {
308        $form = $this->lowering->form($list);

310            throw ImplementationGap::production($form);

312
313        return (new Lists())->items($list);
