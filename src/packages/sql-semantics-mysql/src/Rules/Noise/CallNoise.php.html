8 * The noise and synonym token positions of the productions of function-like expressions.
9 *
10 * Only a token with no influence on meaning in its production may be listed
11 * as noise, and only terminals the manual defines as synonyms may share a
12 * key. Every entry is listed in the method documentation with its reason and
13 * the manual page that states it. Nothing else is skipped or merged by the
14 * token correspondence check.
15 *
16 * @visibility SqlSemantics\Platform\MySql
17 */

19{
20    /**
21     * Answers the noise positions by production signature.
22     *
23     * None: the empty parentheses after a clock function change nothing, but
24     * they are part of the text MySQL names an unaliased select list
25     * expression after, so the model keeps them (OptionalWords).
26     *
27     * @return array<string, list<int>>
28     */
29    public static function positions(): array
30    {
31        return [];
32    }
33
34    /**
35     * Answers the key of each synonym position by production signature.
36     *
37     * - `SUBSTRING(str FROM pos FOR len)` is `SUBSTRING(str, pos, len)`: FROM
38     *   and FOR take the keys of the commas
39     *   (https://dev.mysql.com/doc/refman/8.4/en/string-functions.html#function_substring:
40     *   "The forms with a FROM keyword are standard SQL syntax";
41     *   the server builds the same Item_func_substr for both).
42     * - `ADDDATE(date, INTERVAL expr unit)` is `DATE_ADD(date, INTERVAL expr
43     *   unit)` and `SUBDATE(...)` with INTERVAL is `DATE_SUB(...)`
44     *   (https://dev.mysql.com/doc/refman/8.4/en/date-and-time-functions.html#function_adddate:
45     *   "When invoked with the INTERVAL form of the second argument, ADDDATE()
46     *   is a synonym for DATE_ADD()"; likewise SUBDATE() for DATE_SUB()).
47     *
48     * @return array<string, array<int, string>>
49     */
50    public static function synonyms(): array
51    {
52        return [
53            'function_call_nonkeyword: SUBSTRING ( expr FROM expr FOR_SYM expr )' => [3 => ',', 5 => ','],
54            'function_call_nonkeyword: SUBSTRING ( expr FROM expr )' => [3 => ','],
55            'function_call_nonkeyword: ADDDATE_SYM ( expr , INTERVAL_SYM expr interval )' => [0 => 'DATE_ADD_INTERVAL'],
56            'function_call_nonkeyword: SUBDATE_SYM ( expr , INTERVAL_SYM expr interval )' => [0 => 'DATE_SUB_INTERVAL'],
57        ];
58    }
59}
