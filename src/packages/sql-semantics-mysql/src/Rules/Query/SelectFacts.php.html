29/**
30 * Derives the facts of a query block.
31 *
32 * Rule: MYSQL-SELECT-FACTS-001. The FROM clause is derived in the enclosing
33 * environment (MYSQL-FROM-SCOPE-001). The select list sees the FROM tables
34 * and, beyond them, the enclosing queries; it does not see its own aliases.
35 * WHERE sees the FROM tables. GROUP BY, QUALIFY and ORDER BY also see
36 * the aliases of the select list (MYSQL-SORT-SCOPE-001); HAVING sees the
37 * GROUP BY columns and the select list first (MYSQL-HAVING-SCOPE-001). The named windows
38 * see the FROM tables. In a block that aggregates without GROUP BY
39 * (MYSQL-AGGREGATE-QUERY-001; HAVING alone filters rows like WHERE) or groups WITH ROLLUP, ROLLUP or CUBE, every
40 * column of the FROM tables read by the select list, HAVING, the windows,
41 * QUALIFY and ORDER BY can be NULL. LIMIT, PROCEDURE ANALYSE, INTO and the
42 * locking clauses follow MYSQL-TAIL-FACTS-001; a late ordering is derived
43 * like the ORDER BY and LIMIT of the block. A window name defined twice
44 * is reported, and so is a window name the block does not define
45 * (MYSQL-WINDOW-NAME-001). The output fields follow MYSQL-STAR-001. Terminates: every
46 * clause is a strict part of the block. Source:
47 * https://dev.mysql.com/doc/refman/8.4/en/select.html,
48 * https://dev.mysql.com/doc/refman/8.4/en/group-by-modifiers.html ("the
49 * super-aggregate rows ... NULL"). Status: Implemented.
50 *
51 * @visibility SqlSemantics\Platform\MySql
52 */
53final class SelectFacts

55    /**
56     * Derives every part of the block and answers its output.
57     */
58    public function derive(Select $select, Derivation $derivation, Environment $outer): QueryFact
59    {
60        $context = $derivation->context;
61        $from = $select->from === null ? new JoinedInput(new RelationFact(new RowShape([])), [], []) : (new FromScope())->open($select->from, $derivation, $outer, []);
62        $visible = $from->visible;
63        $ordering = array_map(static fn (OrderItem $item): object => $item->expression, [...$select->orderBy, ...($select->late === null ? [] : $select->late->orderBy)]);
64        $expressions = array_map(static fn (object $item): object => $item instanceof SelectExpression ? $item->expression : $item, $select->items);
65        $aggregate = $select->groupBy === null && (new Aggregation())->aggregates([...$expressions, ...$ordering, ...array_values(array_filter([$select->having, $select->qualify]))]);
66        $output = $aggregate || $select->groupBy?->modifier !== null ? array_map(static fn ($relation) => (new Joining())->extend($relation), $visible) : $visible;
67        $items = (new Projection())->items($select->items, $derivation, new Environment($context, $outer, $output), new JoinedInput($from->fact, $output, $from->star));
68        $aliases = $this->aliases($select, $items, $context->profile);
69        if ($select->where !== null) {
70            (new Operands())->single($derivation->scalar($select->where, new Environment($context, $outer, $visible)), $derivation);
71        }
72        $grouping = $select->groupBy === null ? [] : (new SortScopes())->derive($select->groupBy->items, $derivation, new Environment($context, $outer, $visible, [], $aliases), $items, false);
73        $results = new Environment($context, $outer, $output, [], $aliases);
74        if ($select->having !== null) {
75            $scope = new HavingScope();
76            $row = new GroupedRow($items, $scope->grouping($grouping, $visible, $output), $select->groupBy !== null || $aggregate || in_array(SelectOption::Distinct, $select->options, true), $scope->undecided($grouping, $items));
77            (new Operands())->single($derivation->scalar($select->having, $scope->enter($results, $row)), $derivation);
78        }
79        $this->windows($select, $derivation, new Environment($context, $outer, $output));
80        (new WindowReferences())->check($select, $derivation);
81        if ($select->qualify !== null) {
82            $derivation->scalar($select->qualify, $results);
83        }
84        (new SortScopes())->derive([...$select->orderBy, ...($select->late === null ? [] : $select->late->orderBy)], $derivation, $results, $items, true);
85        (new TailFacts())->limit($select->limit, $derivation, $outer);
86        (new TailFacts())->limit($select->late?->limit, $derivation, $outer);
87        foreach ($select->procedure === null ? [] : $select->procedure->arguments as $argument) {
88            $derivation->scalar($argument, new Environment($context, $outer));
89        }
90        $fact = new QueryFact($items, $context->columnNames);
91        (new TailFacts())->derive($derivation, $outer, $fact, $select->into, $select->locking, $visible);
92
93        return $fact;
94    }

96    /**
97     * Derives the named windows and reports a name defined twice.
98     */
99    public function windows(Select $select, Derivation $derivation, Environment $environment): void
100    {
101        $seen = [];
102        foreach ($select->windows as $window) {
103            $window->specification->deriveWindow($derivation, $environment);
104            $key = $derivation->context->columnNames->fold($window->name->value);
105            if (isset($seen[$key])) {
106                $derivation->report(new Misuse(MisuseRule::DuplicateWindow));
107            }
108            $seen[$key] = true;
109        }
110    }

112    /**
113     * Answers the output fields a name can refer to by their item name, in output order: aliased items and the items not named after a column.
114     *
115     * The server finds an item of the select list by its name, alias or
116     * name given after its text alike, unless the item is a column
117     * reference, which is found as the column it reads.
118     *
119     * @param list<Field|OpenStar> $items
120     * @return list<Field>
121     */
122    public function aliases(Select $select, array $items, LanguageProfile $profile): array
123    {
124        $aliased = [];
125        $naming = new ItemNaming($profile);
126        foreach ($select->items as $item) {
127            if ($item instanceof SelectExpression && ($item->alias !== null || !$naming->own($item->expression) instanceof ColumnUse)) {
128                $aliased[spl_object_id($item->expression)] = true;
129            }
130        }
131        $fields = [];
132        foreach ($items as $field) {
133            if ($field instanceof Field && $field->expression !== null && isset($aliased[spl_object_id($field->expression)])) {
134                $fields[] = $field;
135            }
136        }
137
138        return $fields;
139    }
