123    public function testStatementsCompilesReceiverAndBindingsDerivedFromTheSameBranch(): void
124    {
125        $file = (new SourceParser())->parse('query.php', '<?php use Illuminate\\Support\\Facades\\DB; $q = DB::table("users"); $q->where("id", 7)->get();');
126        $index = (new ProgramIndexBuilder())->build([$file]);
127        $calls = (new \PhpParser\NodeFinder())->findInstanceOf($file->statements, \PhpParser\Node\Expr\MethodCall::class);
128        $deriver = (new Interpreter($index, (new LaravelExtension())->sinks(), dialect: 'sqlite', modelProviders: [new LaravelExtension()]))->deriverFor([$file]);
129        $solutions = (new \SqlCatalog\Analysis\Model\ModelQueries())->solve($calls[0], new BuilderQueries($index), $deriver);
130        self::assertCount(1, $solutions);
131        self::assertSame('select * from "users" where "id" = ?', $solutions[0]->values[0]->soleLiteral()?->value);
132        self::assertSame(7, $solutions[0]->values[1]->soleArray()?->positional()[0]->soleLiteral()?->value);
133    }

135    public function testCompileAppliesSoftDeletesOnlyAtExecution(): void
136    {
137        $state = new QueryState(['dialect' => Domain::literal('sqlite'), 'table' => Domain::literal('users'), 'softDeletes' => Domain::literal(true), 'deletedColumn' => Domain::literal('deleted_at')]);
138        $queries = new BuilderQueries(new ProgramIndex());
139        self::assertSame('select * from "users" where "users"."deleted_at" is null', $queries->compile($state, 'get', [])[0]->soleLiteral()?->value);
140        self::assertSame('select * from "users"', $queries->compile($state->with('trashed', Domain::literal('withtrashed')), 'get', [])[0]->soleLiteral()?->value);
141        self::assertSame('select * from "users" where "users"."deleted_at" is not null', $queries->compile($state->with('trashed', Domain::literal('onlytrashed')), 'get', [])[0]->soleLiteral()?->value);
142        self::assertFalse($queries->compile($state, 'delete', [])[0]->isExact());
143        self::assertFalse($queries->compile($state->with('timestamps', Domain::literal(true)), 'update', [])[0]->isExact());
144    }
