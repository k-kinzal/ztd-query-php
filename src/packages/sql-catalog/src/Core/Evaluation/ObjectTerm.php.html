13/**
14 * An object the analyzer tracked by class.
15 *
16 * An enum case carries the case name,
17 * which resolves `->value` to a literal. A prepared statement carries the
18 * identifier of the catalog entry it came from, which is how a later
19 * `execute()` finds the query it binds values to. Modelled builders also carry
20 * an allocation identity and immutable state, so aliases observe mutations.
21 *
22 * @visibility root
23 */
24final class ObjectTerm implements Term
25{
26    /**
27     * @param string $className The fully qualified class name, without a leading backslash
28     * @param string|null $enumCase The case name when the object is an enum case
29     * @param string|null $statementId The catalog entry a prepared statement belongs to
30     * @param string|null $identity The allocation followed across aliases
31     * @param ArrayTerm|null $state The immutable properties of a modelled object
32     */
33    public function __construct(
34        public readonly string $className,
35        public readonly ?string $enumCase = null,
36        public readonly ?string $statementId = null,
37        public readonly ?string $identity = null,
38        public readonly ?ArrayTerm $state = null,
39    ) {
40    }
41
42    /**
43     * An object never resolves to text, so it becomes a gap.
44     */
45    #[Override]
46    public function toPattern(): TextPattern
47    {
48        return TextPattern::fromHole(new TextHole(Origin::Unresolved, $this->type()));
49    }
50
51    /**
52     * The class of the object.
53     */
54    #[Override]
55    public function type(): TypeShape
56    {
57        return TypeShape::of([$this->className]);
58    }
59
60    /**
61     * The class, and the case or statement it stands for.
62     */
63    #[Override]
64    public function signature(): string
65    {
66        $base = 'object:' . $this->className . ':' . ($this->enumCase ?? '') . ':' . ($this->statementId ?? '');
67
68        return $this->identity === null ? $base : $base . ':' . $this->identity . ':' . ($this->state?->signature() ?? '');
69    }
70}
