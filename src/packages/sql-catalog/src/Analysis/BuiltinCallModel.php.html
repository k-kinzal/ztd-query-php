17/**
18 * What the PHP functions that shape query text do to the values they are given.
19 *
20 * Queries are assembled with a small, stable set of string functions. Modelling
21 * those keeps a statement resolved that would otherwise become a gap the moment
22 * it passed through `sprintf()` or `implode()`.
23 *
24 * @visibility root
25 */
26final class BuiltinCallModel
27{

29        'json_encode' => true,
30        'addslashes' => true,

33        'substr' => true,
34        'number_format' => true,
35        'date' => true,
36        'ucwords' => true,
37        'nl2br' => true,
38        'serialize' => true,

45    {

47            'sprintf',

49            'implode',

59            'str_replace',
60            'strval',
61            'intval',
62            'count',
63            'strlen',
64        ], true) || isset(self::STRING_RESULTS[$this->normalize($name)]);

67    /**
68     * The result of calling the function with the given argument domains.
69     *
70     * @param list<Domain> $arguments
71     */
72    public function evaluate(string $name, array $arguments): Domain
73    {
74        $function = $this->normalize($name);
75
76        return match ($function) {
77            'sprintf' => $this->sprintf(array_slice($arguments, 1), $arguments[0] ?? Domain::unknown()),
78            'vsprintf' => $this->vsprintf($arguments),
79            'implode', 'join' => $this->implode($arguments),
80            'str_repeat' => $this->repeat($arguments),
81            'strtolower', 'strtoupper', 'ucfirst', 'lcfirst', 'trim', 'ltrim', 'rtrim'
82                => $this->transform($function, $arguments),
83            'str_replace' => $this->replace($arguments),
84            'strval' => $arguments[0] ?? Domain::unknown(),
85            'intval', 'count', 'strlen' => Domain::opaque(TypeShape::of(['int']), Origin::Call, $function),
86            default => Domain::opaque(TypeShape::of(['string']), Origin::Call, $function),
87        };
88    }

90    /**
91     * The function name without its namespace, in lower case.
92     */
93    public function normalize(string $name): string
94    {
95        $parts = explode('\\', ltrim($name, '\\'));
96
97        return strtolower($parts[count($parts) - 1]);
98    }

100    /**
101     * The result of `sprintf()`, resolved as far as the format and the arguments allow.
102     *
103     * A format the analyzer knows only in part — `"SELECT * FROM $table WHERE
104     * id IN (%s)"` — still has conversions in the parts it knows, so those are
105     * filled in and the gaps are carried through, rather than the whole result
106     * becoming one gap because one piece of the format was not fixed.
107     *
108     * @param list<Domain> $arguments
109     */
110    public function sprintf(array $arguments, Domain $format): Domain
111    {
112        $result = null;
113        foreach ($format->terms as $term) {
114            $value = $term instanceof LiteralTerm || $term instanceof PatternTerm
115                ? $this->formatPattern($term->toPattern(), $arguments)
116                : Domain::opaque(TypeShape::of(['string']), $this->originOf($format, Origin::Call), 'sprintf');
117            $result = $result === null ? $value : $result->union($value);
118        }
119
120        return $result ?? Domain::opaque(TypeShape::of(['string']), Origin::Call, 'sprintf');
121    }

123    /**
124     * A known or partly known format with its conversions filled in by the arguments.
125     *
126     * @param list<Domain> $arguments
127     */
128    public function formatPattern(TextPattern $format, array $arguments): Domain
129    {
130        $result = Domain::literal('');
131        $index = 0;
132        foreach ($format->segments as $segment) {
133            if (!$segment instanceof LiteralText) {
134                $result = $result->concat(Domain::of(Domain::asTerm(TextPattern::fromSegments([$segment]))));
135                continue;
136            }
137            foreach ($this->splitFormat($segment->text) as $piece) {
138                if (!str_starts_with($piece, '%') || $piece === '%%') {
139                    $result = $result->concat(Domain::literal($piece === '%%' ? '%' : $piece));
140                    continue;
141                }
142                $result = $result->concat(
143                    $arguments[$index] ?? Domain::opaque(TypeShape::unknown(), Origin::Call, 'sprintf'),
144                );
145                $index++;
146            }
147        }
148
149        return $result;
150    }

152    /**
153     * Where the unresolved part of a value comes from, or the fallback when nothing in it says.
154     */
155    public function originOf(Domain $domain, Origin $fallback): Origin
156    {
157        foreach ($domain->terms as $term) {
158            if ($term instanceof OpaqueTerm) {
159                return $term->origin;
160            }
161        }
162
163        return $fallback;
164    }

166    /**
167     * The result of `vsprintf()`, which takes its arguments as one array.
168     *
169     * @param list<Domain> $arguments
170     */
171    public function vsprintf(array $arguments): Domain
172    {
173        $values = ($arguments[1] ?? Domain::unknown())->soleArray();
174        if ($values === null) {
175            return Domain::opaque(TypeShape::of(['string']), Origin::Call, 'vsprintf');
176        }
177
178        return $this->sprintf($values->positional(), $arguments[0] ?? Domain::unknown());
179    }

181    /**
182     * A format string split into literal runs and conversion specifications.
183     *
184     * @return list<string>
185     */
186    public function splitFormat(string $format): array
187    {
188        $pieces = preg_split('/(%%|%[-+ 0\']*\d*(?:\.\d+)?[bcdeEfFgGosuxX])/', $format, -1, PREG_SPLIT_DELIM_CAPTURE);
189        if ($pieces === false) {
190            return [$format];
191        }
192
193        return array_values(array_filter($pieces, static fn (string $piece): bool => $piece !== ''));
194    }

196    /**
197     * The result of `implode()`, resolved when the array is known element by element.
198     *
199     * @param list<Domain> $arguments
200     */
201    public function implode(array $arguments): Domain
202    {
203        $glue = count($arguments) > 1 ? $arguments[0] : Domain::literal('');
204        $array = $this->arrayArgument($arguments);
205        if ($array === null) {
206            $pieces = $arguments[count($arguments) - 1] ?? Domain::unknown();
207
208            return Domain::opaque(TypeShape::of(['string']), $this->originOf($pieces, Origin::Call), 'implode');
209        }
210
211        $result = Domain::literal('');
212        $first = true;
213        foreach ($array->positional() as $element) {
214            $result = $first ? $element : $result->concat($glue)->concat($element);
215            $first = false;
216        }
217
218        return $array->complete ? $result : $result->concat(Domain::opaque(TypeShape::of(['string']), Origin::Call, 'implode'));
219    }

221    /**
222     * The array argument of an implode call, whichever position it is written in.
223     *
224     * @param list<Domain> $arguments
225     */
226    public function arrayArgument(array $arguments): ?ArrayTerm
227    {
228        foreach ($arguments as $argument) {
229            $array = $argument->soleArray();
230            if ($array !== null) {
231                return $array;
232            }
233        }
234
235        return null;
236    }

238    /**
239     * The result of `str_repeat()`, resolved when both arguments are known.
240     *
241     * @param list<Domain> $arguments
242     */
243    public function repeat(array $arguments): Domain
244    {
245        $subject = ($arguments[0] ?? Domain::unknown())->soleLiteral();
246        $times = ($arguments[1] ?? Domain::unknown())->soleLiteral();
247        if ($subject === null || $times === null || !is_int($times->value) || $times->value < 0 || $times->value > 1000) {
248            return Domain::opaque(TypeShape::of(['string']), Origin::Call, 'str_repeat');
249        }
250
251        return Domain::literal(str_repeat($subject->toText(), $times->value));
252    }

254    /**
255     * The result of a one-argument string transformation.
256     *
257     * @param list<Domain> $arguments
258     */
259    public function transform(string $function, array $arguments): Domain
260    {
261        $subject = ($arguments[0] ?? Domain::unknown())->soleLiteral();
262        if ($subject === null) {
263            return Domain::opaque(TypeShape::of(['string']), Origin::Call, $function);
264        }
265        $text = $subject->toText();
266
267        return Domain::literal(match ($function) {
268            'strtolower' => strtolower($text),
269            'strtoupper' => strtoupper($text),
270            'ucfirst' => ucfirst($text),
271            'lcfirst' => lcfirst($text),
272            'ltrim' => ltrim($text),
273            'rtrim' => rtrim($text),
274            default => trim($text),
275        });
276    }
277
278    /**
279     * The result of `str_replace()`, resolved when every argument is a known string.
280     *
281     * @param list<Domain> $arguments
282     */
283    public function replace(array $arguments): Domain
284    {
285        $search = ($arguments[0] ?? Domain::unknown())->soleLiteral();
286        $replace = ($arguments[1] ?? Domain::unknown())->soleLiteral();
287        $subject = ($arguments[2] ?? Domain::unknown())->soleLiteral();
288        if ($search === null || $replace === null || $subject === null) {
289            return Domain::opaque(TypeShape::of(['string']), Origin::Call, 'str_replace');
290        }
291
292        return Domain::literal(str_replace($search->toText(), $replace->toText(), $subject->toText()));
293    }
