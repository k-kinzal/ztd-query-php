16/**
17 * The calls in the source that reach a given function or method.
18 *
19 * A function is reached by the calls written with its name. A method is
20 * reached by the calls written with its name on something that can be an
21 * instance of its class; what they are written on is worked out the same way
22 * any other value is, so `$repository->find(…)` is a caller of
23 * `Repository::find` when `$repository` can be a `Repository`. A call on
24 * something whose class could not be worked out is not taken as a caller, since
25 * a method of the same name on a built-in class or on a class outside the
26 * analyzed files may be the one it reaches; the callers found are then marked
27 * as partial instead.
28 *
29 * A call written on a class name reaches the method that class has, declared or
30 * inherited: `parent::` looks from the parent of the class it is written in and
31 * `self::` from that class itself. Only `static::` and a call on an instance
32 * can also land in a subclass that overrides the method.
33 *
34 * A method of a class an extension models has no callers here. The extension
35 * already says what calling it does, and every call of it is read where it is
36 * written; reading it again from inside the class, once per caller, would list
37 * each of those statements a second time under the class's own line.
38 *
39 * @visibility root
40 */
41final class Callers
42{

53    {
54        $this->calls = $calls;
55        $this->index = $index;
56        $this->sinks = $sinks;
57    }

59    /**
60     * The calls that reach a body, in the order they are written.
61     */
62    public function of(FunctionLike $body, int $depth, Deriver $deriver): CallerSet
63    {
64        if ($body instanceof Stmt\Function_) {
65            return new CallerSet($this->ofFunction($body));
66        }
67        if (!$body instanceof Stmt\ClassMethod) {
68            return new CallerSet([]);
69        }
70        $className = $deriver->classOf($body);
71        if ($className === null || $this->sinks?->models(Domain::of(new ObjectTerm($className))) === true) {
72            return new CallerSet([]);
73        }
74        $method = $body->name->toString();
75        $callers = strtolower($method) === '__construct' ? $this->instantiations($className, $deriver) : [];
76        $partial = false;
77        foreach ($this->calls->named($method) as $call) {
78            $reaches = $this->reaches($call, $className, $depth, $deriver);
79            if ($reaches === true) {
80                $callers[] = $call;
81            }
82            $partial = $partial || $reaches === null;
83        }
84
85        return new CallerSet($callers, $partial);
86    }

88    /**
89     * The calls written with a function's fully qualified name.
90     *
91     * @return list<Expr\CallLike>
92     */
93    public function ofFunction(Stmt\Function_ $function): array
94    {
95        $wanted = strtolower($function->namespacedName?->toString() ?? $function->name->toString());
96        $callers = [];
97        foreach ($this->calls->named($function->name->toString()) as $call) {
98            if (!$call instanceof Expr\FuncCall || !$call->name instanceof Node\Name) {
99                continue;
100            }
101            $namespaced = $call->name->getAttribute('namespacedName');
102            $candidates = [strtolower(ltrim($call->name->toString(), '\\'))];
103            if ($namespaced instanceof Node\Name) {
104                $candidates[] = strtolower($namespaced->toString());
105            }
106            if (in_array($wanted, $candidates, true)) {
107                $callers[] = $call;
108            }
109        }
110
111        return $callers;
112    }

114    /**
115     * The instantiations that run a class's constructor.
116     *
117     * @return list<Expr\CallLike>
118     */
119    public function instantiations(string $className, Deriver $deriver): array
120    {
121        $callers = [];
122        foreach ($this->inheritorsOf($className) as $constructed) {
123            foreach ($this->calls->instantiating($constructed) as $new) {
124                $written = $new->class instanceof Node\Name ? $new->class->toString() : null;
125                $resolved = $written !== null && in_array(strtolower($written), ['self', 'static'], true)
126                    ? $deriver->classOf($new)
127                    : $written;
128                if ($resolved !== null && strcasecmp(ltrim($resolved, '\\'), ltrim($constructed, '\\')) === 0) {
129                    $callers[] = $new;
130                }
131            }
132        }
133
134        return $callers;
135    }

137    /**
138     * The classes whose instantiation runs a class's constructor: the class, and every subclass that inherits the constructor rather than declaring its own.
139     *
140     * @return list<string>
141     */
142    public function inheritorsOf(string $className): array
143    {
144        $wanted = strtolower(ltrim($className, '\\'));
145        $classes = [$className];
146        foreach ($this->index->classes as $key => $shape) {
147            $current = $key === $wanted ? null : $shape;
148            $seen = [];
149            while ($current !== null && !isset($current->methods['__construct']) && !isset($seen[$current->name])) {
150                $seen[$current->name] = true;
151                $current = $this->index->findClass($current->parent);
152            }
153            if ($current !== null && strtolower(ltrim($current->name, '\\')) === $wanted) {
154                $classes[] = $shape->name;
155            }
156        }
157
158        return $classes;
159    }

161    /**
162     * Whether a call written with the method's name can reach the method of that class, or null when that cannot be told.
163     */
164    public function reaches(Expr\CallLike $call, string $className, int $depth, Deriver $deriver): ?bool
165    {
166        $method = $this->calls->nameOf($call) ?? '';
167        if ($call instanceof Expr\StaticCall) {
168            if (!$call->class instanceof Node\Name) {
169                return null;
170            }
171            $target = $this->staticTarget($call, $deriver);
172            if ($target === null) {
173                return false;
174            }
175            $late = strtolower($call->class->toString()) === 'static';
176
177            return $late ? $this->related($target, $className, $method) : $this->index->isInstanceOf($target, $className);
178        }
179        if (!$call instanceof Expr\MethodCall && !$call instanceof Expr\NullsafeMethodCall) {
180            return false;
181        }
182        if ($call->var instanceof Expr\Variable && $call->var->name === FreeNames::THIS) {
183            $enclosing = $deriver->classOf($call);
184
185            return $enclosing !== null && $this->related($enclosing, $className, $method);
186        }
187        $classes = [];
188        foreach ($deriver->solveUnlessSpent($call, [$call->var], $depth + 1) as $solution) {
189            foreach ($solution->values[0]->type()->classNames() as $candidate) {
190                $classes[] = $candidate;
191            }
192        }
193        if ($classes === []) {
194            return null;
195        }
196        foreach ($classes as $candidate) {
197            if ($this->related($candidate, $className, $method)) {
198                return true;
199            }
200        }
201
202        return false;
203    }

205    /**
206     * The class a call written on a class name looks its method up from, or null when it cannot be named.
207     */
208    public function staticTarget(Expr\StaticCall $call, Deriver $deriver): ?string
209    {
210        if (!$call->class instanceof Node\Name) {
211            return null;
212        }
213        $written = $call->class->toString();
214
215        return match (strtolower($written)) {
216            'self', 'static' => $deriver->classOf($call),
217            'parent' => $this->index->findClass($deriver->classOf($call) ?? '')?->parent,
218            default => $written,
219        };
220    }

222    /**
223     * Whether a call of the method on an instance of one class can land in the method as the other class declares it.
224     *
225     * The method is looked up from the class of the instance upwards, and the
226     * first class that declares it is the one that runs. So a call on an
227     * instance of a subclass lands in the declaring class's method only when
228     * nothing between them overrides it, and a call on an instance of a parent
229     * class can land in the declaring class's method when the instance is of
230     * that subclass.
231     */
232    public function related(string $receiver, string $declaring, string $method): bool
233    {
234        if ($this->index->isInstanceOf($declaring, $receiver)) {
235            return true;
236        }
237        if (!$this->index->isInstanceOf($receiver, $declaring)) {
238            return false;
239        }
240        $key = strtolower($method);
241        $wanted = strtolower(ltrim($declaring, '\\'));
242        $current = $this->index->findClass($receiver);
243        $seen = [];
244        while ($current !== null && strtolower($current->name) !== $wanted && !isset($seen[$current->name])) {
245            if (isset($current->methods[$key])) {
246                return false;
247            }
248            $seen[$current->name] = true;
249            $current = $this->index->findClass($current->parent);
250        }
251
252        return true;
253    }
