21/**
22 * What a call into the analyzed source returns, read back from its `return` statements.
23 *
24 * Each `return` is walked back like any other point, with the callee's
25 * parameters standing for the arguments the call passes, so only what the
26 * returned value depends on is run. A call reached again with the same
27 * arguments is answered from memory.
28 *
29 * @visibility root
30 */
31final class CalleeReturns
32{
33    private BackwardSlicer $slicer;
34
35    private SliceExecutor $executor;
36
37    private FreeNames $names;
38
39    private EvaluationBudget $budget;
40
41    private CallResults $remembered;

43    /**
44     * Wires the reader to the slicer and executor it reads returns with.
45     */
46    public function __construct(
47        BackwardSlicer $slicer,
48        SliceExecutor $executor,
49        EvaluationBudget $budget,
50        ?FreeNames $names = null,
51        ?CallResults $remembered = null,
52    ) {
53        $this->slicer = $slicer;
54        $this->executor = $executor;
55        $this->budget = $budget;
56        $this->names = $names ?? new FreeNames();
57        $this->remembered = $remembered ?? new CallResults();
58    }

60    /**
61     * Every value the callee can return for these arguments.
62     *
63     * @param list<Domain> $arguments
64     */
65    public function valueOf(
66        MethodShape|FunctionShape $callee,
67        array $arguments,
68        FunctionScope $scope,
69        ExpressionEvaluator $expressions,
70    ): Domain {
71        $name = $callee instanceof MethodShape ? $callee->qualifiedName() : $callee->name;
72        $body = $callee->node?->getStmts();
73        if ($body === null) {
74            return Domain::opaque($callee->returnType, Origin::Call, $name . '()');
75        }
76        if ($scope->depth() >= $this->budget->maxDepth || $scope->isFollowing($name) || $this->budget->isExhausted()) {
77            return Domain::opaque($callee->returnType, Origin::Budget, $name . '()');
78        }
79        $key = $this->remembered->keyFor($name, $arguments);
80
81        return $this->remembered->recall($key) ?? $this->read($callee, $name, $key, $arguments, $scope, $expressions);
82    }
83
84    /**
85     * What the callee returns, read from its body and remembered unless the budget cut the reading short.
86     *
87     * @param list<Domain> $arguments
88     */
89    public function read(
90        MethodShape|FunctionShape $callee,
91        string $name,
92        string $key,
93        array $arguments,
94        FunctionScope $scope,
95        ExpressionEvaluator $expressions,
96    ): Domain {
97        $inner = $scope->enter($name, $callee instanceof MethodShape ? $callee->className : $scope->className, $callee->file);
98        $start = $this->parameters($callee->parameters, $arguments, $inner, $expressions);
99        $result = null;
100        foreach ($this->returnsIn($callee->node?->getStmts() ?? []) as $return) {
101            $value = $this->returned($return, $start, $inner, $expressions);
102            $result = $result === null ? $value : $result->union($value);
103        }
104        $result ??= Domain::literal(null);
105        if (!$this->budget->isExhausted()) {
106            $this->remembered->remember($key, $result);
107        }
108
109        return $result;
110    }
111
112    /**
113     * What one `return` statement can return, starting from the given parameters.
114     */
115    public function returned(Stmt\Return_ $return, Environment $start, FunctionScope $scope, ExpressionEvaluator $expressions): Domain
116    {
117        if ($return->expr === null) {
118            return Domain::literal(null);
119        }
120        $result = null;
121        foreach ($this->slicer->sliceFrom($return, $this->names->of([$return->expr])) as $arrival) {
122            foreach ($this->executor->run($arrival->path->forward(), $start->copy(), $scope, $expressions) as $environment) {
123                $value = $expressions->evaluate($return->expr, $environment, $scope);
124                $result = $result === null ? $value : $result->union($value);
125            }
126        }
127
128        return $result ?? Domain::literal(null);
129    }

131    /**
132     * The parameters bound to the arguments, with defaults for the ones not passed.
133     *
134     * @param list<ParameterShape> $parameters
135     * @param list<Domain> $arguments
136     */
137    public function parameters(array $parameters, array $arguments, FunctionScope $scope, ExpressionEvaluator $expressions): Environment
138    {
139        $environment = new Environment();
140        foreach ($parameters as $position => $parameter) {
141            $value = $parameter->variadic
142                ? Domain::opaque($parameter->type, Origin::Parameter, '...$' . $parameter->name)
143                : ($arguments[$position] ?? null);
144            if ($value === null && $parameter->default !== null) {
145                $value = $expressions->evaluate($parameter->default, new Environment(), $scope);
146            }
147            $environment->write($parameter->name, $value ?? Domain::opaque($parameter->type, Origin::Parameter, '$' . $parameter->name));
148        }
149
150        return $environment;
151    }
152
153    /**
154     * The `return` statements of a body, leaving out those of closures and classes declared in it.
155     *
156     * @param array<array-key, Node> $nodes
157     * @return list<Stmt\Return_>
158     */
159    public function returnsIn(array $nodes): array
160    {
161        $found = [];
162        foreach ($nodes as $node) {
163            if ($node instanceof Node\FunctionLike || $node instanceof Stmt\ClassLike) {
164                continue;
165            }
166            if ($node instanceof Stmt\Return_) {
167                $found[] = $node;
168            }
169            foreach (get_object_vars($node) as $sub) {
170                $children = [];
171                foreach (is_array($sub) ? $sub : [$sub] as $child) {
172                    if ($child instanceof Stmt) {
173                        $children[] = $child;
174                    }
175                }
176                $found = array_merge($found, $this->returnsIn($children));
177            }
178        }
179
180        return $found;
181    }
