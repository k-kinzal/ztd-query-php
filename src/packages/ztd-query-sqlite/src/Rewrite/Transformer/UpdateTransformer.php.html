14/**
15 * Transforms UPDATE statements into SELECT projections with CTE shadowing for SQLite.
16 *
17 * SQLite does not support multi-table UPDATE, so this is simpler than MySQL.
18 */
19final class UpdateTransformer implements SqlTransformer
20{
21    private SqliteParser $parser;
22    private SelectTransformer $selectTransformer;
23    private SqliteCteShadowComposer $cteComposer;
24
25    /**
26     * Binds the dependencies used by this operation.
27     */
28    public function __construct(
29        SqliteParser $parser,
30        SelectTransformer $selectTransformer,
31    ) {
32        $this->parser = $parser;
33        $this->selectTransformer = $selectTransformer;
34        $this->cteComposer = new SqliteCteShadowComposer();
35    }
36
37    /**
38     * {@inheritDoc}
39     * @throws UnsupportedSqlException
40     */
41    public function transform(string $sql, array $tables): string
42    {
43        $type = $this->parser->classifyStatement($sql);
44        if ($type !== 'UPDATE') {
45            throw new UnsupportedSqlException($sql, 'Expected UPDATE statement');
46        }
47
48        $targetTable = $this->parser->extractTargetTable($sql);
49        if ($targetTable === null) {
50            throw new UnsupportedSqlException($sql, 'Cannot resolve UPDATE target');
51        }
52
53        $columns = $tables[$targetTable]['columns'] ?? [];
54
55        $projection = $this->buildProjection(
56            $sql,
57            $targetTable,
58            $columns,
59            $tables[$targetTable]['primaryKeys'] ?? [],
60        );
61
62        return $this->selectTransformer->transform(
63            $this->cteComposer->carryPrefix($sql, $projection),
64            $tables,
65        );
66    }
67
68    /**
69     * Build a result-select SQL from an UPDATE statement.
70     *
71     * @param string $sql
72     * @param string $targetTable
73     * @param array<int, string> $columns
74     * @param array<int, string> $primaryKeys
75     * @return string
76     */
77    public function buildProjection(
78        string $sql,
79        string $targetTable,
80        array $columns,
81        array $primaryKeys = [],
82    ): string {
83        $assignments = $this->parser->extractUpdateAssignments($sql);
84        $alias = $this->parser->extractUpdateAlias($sql);
85        $qualifier = $alias ?? $targetTable;
86
87        $selectCols = [];
88        $coveredCols = [];
89
90        foreach ($assignments as $colName => $value) {
91            $selectCols[] = "$value AS \"$colName\"";
92            $coveredCols[$colName] = true;
93        }
94
95        foreach ($columns as $col) {
96            if (!isset($coveredCols[$col])) {
97                $selectCols[] = "\"$qualifier\".\"$col\"";
98            }
99        }
100
101        $identity = new MutationRowIdentity();
102        foreach ($primaryKeys as $primaryKey) {
103            $selectCols[] = '"' . $qualifier . '"."' . $primaryKey . '" AS "' . $identity->column($primaryKey) . '"';
104        }
105
106        if ($selectCols === []) {
107            $selectCols[] = '*';
108        }
109
110        $selectList = implode(', ', $selectCols);
111
112        $aliasClause = $alias === null ? '' : ' AS "' . $alias . '"';
113        $fromClause = $this->parser->extractUpdateFromClause($sql);
114        $additionalFrom = $fromClause === null ? '' : ', ' . $fromClause;
115
116        $where = $this->parser->extractWhereClause($sql);
117        $whereClause = $where === null ? '' : " WHERE $where";
118        $orderBy = $this->parser->extractOrderByClause($sql);
119        $orderByClause = $orderBy === null ? '' : " ORDER BY $orderBy";
120        $limit = $this->parser->extractLimitClause($sql);
121        $limitClause = $limit === null ? '' : " LIMIT $limit";
122
123        return "SELECT $selectList FROM \"$targetTable\"$aliasClause$additionalFrom$whereClause$orderByClause$limitClause";
124    }
125
126    /**
127     * Build projection metadata for mutation resolver.
128     *
129     * @param string $sql
130     * @param array<int, string> $columns
131     * @return array{sql: string, table: string}
132     * @throws RuntimeException
133     */
134    public function buildProjectionMeta(string $sql, array $columns): array
135    {
136        $targetTable = $this->parser->extractTargetTable($sql);
137        if ($targetTable === null) {
138            throw new RuntimeException('Cannot resolve UPDATE target');
139        }
140
141        return [
142            'sql' => $this->buildProjection($sql, $targetTable, $columns),
143            'table' => $targetTable,
144        ];
145    }
