23    /**
24     * Binds the collaborators used by this operation.

26    public function __construct(

32    /**
33     * Resolves INSERT conflict behavior against candidate keys.
34     * @throws UnsupportedSqlException
35     */
36    public function resolveInsert(string $sql): ShadowMutation
37    {
38        $tableName = $this->parser->extractTargetTable($sql);
39        if ($tableName === null) {
40            throw new UnsupportedSqlException($sql, 'Cannot resolve INSERT target');
41        }
42        (new MutationTableLookup($this->registry))->assertTableWasNotRemoved($sql, $tableName);
43
44        if ($this->parser->isReplace($sql)) {
45            $definition = $this->registry->get($tableName);
46            $primaryKeys = $definition !== null ? $definition->primaryKeys : [];
47
48            return new ReplaceMutation($tableName, $primaryKeys, $definition?->candidateKeys());
49        }
50
51        if ($this->parser->hasOnConflict($sql)) {
52            return $this->resolveOnConflict($sql, $tableName);
53        }
54
55        $isIgnore = $this->parser->isInsertIgnore($sql);
56
57        $definition = $this->registry->get($tableName);
58        $primaryKeys = $isIgnore ? ($definition !== null ? $definition->primaryKeys : []) : [];
59
60        return new InsertMutation(
61            $tableName,
62            $primaryKeys,
63            $isIgnore,
64            candidateKeys: $definition?->candidateKeys(),
65        );
66    }
67
68    /**
69     * Builds an upsert or do-nothing mutation using candidate-key semantics.
70     */
71    public function resolveOnConflict(string $sql, string $tableName): ShadowMutation
72    {
73        $updateColumns = [];
74        $definition = $this->registry->get($tableName);
75        $databaseEvaluated = $definition !== null && $definition->candidateKeys()->keys() !== [];
76        /**
77         * @var array<string, \ZtdQuery\Shadow\Mutation\UpsertExpression|null> $updateValues
78         */
79        $updateValues = [];
80        $expressionParser = new SqliteUpsertExpressionParser();
81        $onConflictUpdates = $this->parser->extractOnConflictUpdates($sql);
82        foreach ($onConflictUpdates as $colName => $value) {
83            $updateColumns[] = $colName;
84            $updateValues[$colName] = $databaseEvaluated
85                ? $expressionParser->parseIfSupported($value, $tableName)
86                : $expressionParser->parse($value, $tableName);
87        }
88
89        if ($updateColumns !== []) {
90            $primaryKeys = $definition !== null ? $definition->primaryKeys : [];
91            $predicate = $this->parser->extractOnConflictUpdateWhere($sql);
92
93            return new UpsertMutation(
94                $tableName,
95                $primaryKeys,
96                $updateColumns,
97                $updateValues,
98                $definition?->candidateKeys(),
99                $predicate !== null
100                    ? ($databaseEvaluated
101                        ? $expressionParser->parseIfSupported($predicate, $tableName)
102                        : $expressionParser->parse($predicate, $tableName))
103                    : null,
104                databaseEvaluated: $databaseEvaluated,
105                updateSqlValues: $onConflictUpdates,
106                updateSqlPredicate: $predicate,
107            );
108        }
109
110        $definition = $this->registry->get($tableName);
111        $primaryKeys = $definition !== null ? $definition->primaryKeys : [];
112
113        return new InsertMutation(
114            $tableName,
115            $primaryKeys,
116            true,
117            candidateKeys: $definition?->candidateKeys(),
118        );
119    }
