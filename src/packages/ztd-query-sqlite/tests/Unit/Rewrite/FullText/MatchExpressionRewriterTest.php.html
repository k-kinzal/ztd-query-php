40    public function testExpressionEditProjectsAllDocumentColumns(): void
41    {
42        $stream = SqlTokenStream::tokenize("docs MATCH 'hello'", SqliteLexerProfile::create());
43        $rewriter = new MatchExpressionRewriter(new \ZtdQuery\Platform\Sqlite\Sql\SqliteParser(), new \ZtdQuery\Platform\Sqlite\Sql\SqliteIdentifierQuoter());
44        $edit = $rewriter->expressionEdit($stream, $stream->significantTokens()[1], ['docs' => ['columns' => ['title', 'body']]]);
45        self::assertNotNull($edit);
46        self::assertSame(0, $edit['start']);
47        self::assertSame(18, $edit['end']);
48        self::assertStringContainsString('COALESCE(CAST("title" AS TEXT)', $edit['replacement']);
49        self::assertStringContainsString('COALESCE(CAST("body" AS TEXT)', $edit['replacement']);
50        self::assertStringContainsString("CAST(('hello') AS TEXT)", $edit['replacement']);
51    }
