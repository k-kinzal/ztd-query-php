9/**
10 * Realizes parser terminals as SQL text and verifies the resulting token stream.
11 *
12 * Every dialect writes its terminals differently, but all of them owe the same
13 * guarantee: text written from a terminal sequence must read back as that same
14 * sequence through the dialect's own lexer. An implementation that only wrote
15 * text would let a generator drift away from the server it claims to target.
16 *
17 * @visibility root
18 */
19interface LexicalGrammar
20{
21    /**
22     * Writes the one lexeme a lexical generation plan asks for.
23     *
24     * @param GenerationPlan<bool> $plan Plan naming the lexeme kind and its bounds
25     *
26     * @return non-empty-string The lexeme
27     */
28    public function generate(GenerationPlan $plan): string;
29
30    /**
31     * Names the server version this grammar generates for.
32     *
33     * @return string Profile version, e.g. "mysql-8.4.7"
34     */
35    public function version(): string;
36
37    /**
38     * Reports whether a parser terminal can be written as SQL.
39     *
40     * @param string $terminal Terminal to look for
41     *
42     * @return bool True when the terminal can be realized
43     */
44    public function supports(string $terminal): bool;
45
46    /**
47     * Writes a terminal sequence as SQL and checks that it reads back as itself.
48     *
49     * @param list<string> $terminals Terminals to write, in order
50     * @param GenerationPlan<bool>|null $plan Plan that may pin exact lexemes for some terminals
51     *
52     * @return string SQL that tokenizes back to the terminals it was written from
53     * @throws LexicalException When realization cannot round-trip through the lexer
54     */
55    public function realize(array $terminals, ?GenerationPlan $plan = null): string;
56}
