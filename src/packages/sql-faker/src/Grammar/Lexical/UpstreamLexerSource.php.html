11/**
12 * Reads a server's own lexer source over the network.
13 *
14 * Profiles are generated from the exact source of the release they name, so the
15 * fetch is part of the record: a URL that cannot be read means the profile
16 * cannot be built rather than that it should be built from something else.
17 * PHP reports a failed stream as a warning and a false return, so warnings are
18 * turned into the failure they describe.
19 *
20 * @visibility root
21 */
22final class UpstreamLexerSource implements LexerSource
23{
24    /**
25     * Reads one upstream file over HTTP.
26     *
27     * @param string $url Location of the file
28     *
29     * @return string The file's contents
30     *
31     * @throws RuntimeException When the file cannot be read
32     */
33    #[Override]
34    public function fetch(string $url): string
35    {
36        $context = stream_context_create([
37            'http' => [
38                'timeout' => 120,
39                'user_agent' => 'sql-faker/1.0',
40            ],
41        ]);
42
43        set_error_handler(static function (int $severity, string $message): never {
44            throw new RuntimeException($message);
45        });
46        try {
47            $contents = file_get_contents($url, false, $context);
48        } finally {
49            restore_error_handler();
50        }
51
52        if ($contents === false) {
53            throw new RuntimeException("Failed to fetch {$url}");
54        }
55
56        return $contents;
57    }
58}
