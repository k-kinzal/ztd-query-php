12 * Decides which rule a derivation begins from.
13 *
14 * Bison takes the rule named by `%start`, and the first rule in the file when
15 * nothing named one. The fallback is not a guess: it is what Bison itself does,
16 * so a grammar without `%start` still has one answer rather than none.
17 *
18 * @visibility root
19 */
20final class BisonStartSymbol
21{
22    /**
23     * Reports the start rule of a grammar that has been read.
24     *
25     * @param list<BisonDeclaration> $declarations Declarations from the grammar's preamble
26     * @param non-empty-list<BisonRuleNode> $rules Rules in the order they were declared
27     *
28     * @return string Name of the rule to derive from
29     */
30    public function from(array $declarations, array $rules): string
31    {
32        foreach ($declarations as $declaration) {
33            if ($declaration instanceof BisonStartDeclaration) {
34                return $declaration->symbol;
35            }
36        }
37
38        return $rules[0]->name;
39    }
