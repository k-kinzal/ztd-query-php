7/**
8 * Removes from a Lemon file everything that is not grammar.
9 *
10 * A Lemon file carries the C the generated parser is built from as well as the
11 * rules it recognises: comments, include blocks, destructors, and declarations
12 * about types and names. None of that says what the parser accepts, and all of
13 * it can contain text that reads like a rule, so it is taken out before the
14 * rules are looked for.
15 *
16 * @visibility root
17 */
18final class LemonText
19{
20    /**
21     * Removes C comments.
22     *
23     * @param string $input Contents of the grammar file
24     *
25     * @return string The same text with its comments gone
26     */
27    public function withoutComments(string $input): string
28    {
29        return preg_replace(['/\/\*.*?\*\//s', '/\/\/.*$/m'], '', $input) ?? $input;
30    }
31
32    /**
33     * Removes the directives that carry C or configure the generated parser.
34     *
35     * @param string $input Contents of the grammar file
36     *
37     * @return string The same text with those directives gone
38     */
39    public function withoutDirectiveBlocks(string $input): string
40    {
41        return preg_replace([
42            '/%(?:include|destructor|syntax_error|parse_accept|parse_failure|stack_overflow|code)\s*\{[^}]*\}/s',
43            '/^%(?:token_type|default_type|extra_context|name|token_prefix|stack_size|ifdef|ifndef|endif)\b.*$/m',
44        ], '', $input) ?? $input;
45    }
46}
