21/**
22 * SQLite lexical realization for one exact release.
23 *
24 * Writing a terminal sequence as SQL is only half of what this does. The text
25 * is read straight back by the dialect's own tokenizer and the two token
26 * sequences are compared, so a generator that believed it was writing one
27 * statement and a server that would have read another is caught here rather
28 * than by the server. That round trip is the contract; realizing and tokenizing
29 * are the two collaborators it holds.
30 * @phpstan-import-type Catalog from LexicalCatalog
31 *
32 * @visibility root
33 */
34final class LexicalGrammar implements LexicalGrammarContract
35{

53    /**
54     * @param FakerGenerator $faker Source of the choices realization makes
55     * @param string $profileVersion Exact release to generate for, e.g. "sqlite-3.47.2"
56     * @param bool $allowSyntheticTerminals Whether terminals may be written without a catalogued witness
57     * @param LexicalProfileSource|null $profiles Loads the checked-in profile for the version
58     * @param LexicalKeywordIndex|null $index Inverts the profile's terminal-to-spelling map
59     *
60     * @throws RuntimeException When the profile is missing or describes another release
61     */
62    public function __construct(
63        private readonly FakerGenerator $faker,
64        private readonly string $profileVersion,
65        private readonly bool $allowSyntheticTerminals = false,
66        ?LexicalProfileSource $profiles = null,
67        ?LexicalKeywordIndex $index = null,
68    ) {
69        /**
70         * @var array{keywords: array<string, list<string>>, catalog: Catalog} $profile
71         */
72        $profile = ($profiles ?? new LexicalProfileSource())->load('sqlite', $profileVersion);
73        $index ??= new LexicalKeywordIndex();
74
75        $this->strings = new RandomStringGenerator($faker);
76        $this->catalog = new LexicalCatalog($profile['catalog']);
77        $this->tokenizer = new SqliteTokenizer($index->reversed($profile['keywords']));
78        $this->realizer = new SqliteTerminalRealizer(
79            $faker,
80            $this->catalog,
81            $this->tokenizer,
82            $profile['keywords'],
83            $profileVersion,
84            $allowSyntheticTerminals,
85            $this->strings,
86        );
87    }

113    /**
114     * Checks that every terminal a grammar declares can be accounted for.
115     *
116     * The strict-table option is spelled as an ordinary identifier, so the
117     * catalog has no witness under that name and it is not asked about.
118     *
119     * @param list<string> $terminals Terminals the grammar declares

124    {
125        $this->catalog->assertTerminalsCovered(array_values(array_filter(

127            static fn (string $terminal): bool => $terminal !== self::STRICT_TABLE_OPTION,
128        )));

131    /**
132     * Writes a terminal sequence as SQL and checks that it reads back as itself.
133     *
134     * @param list<string> $terminals Terminals to write, in order
135     * @param GenerationPlan<bool>|null $plan Plan that may pin exact lexemes for some terminals
136     *
137     * @return string SQL that tokenizes back to the terminals it was written from
138     *
139     * @throws LexicalException When a terminal cannot be written, or the text does not read back
140     */
141    #[Override]
142    public function realize(array $terminals, ?GenerationPlan $plan = null): string
143    {
144        $lexemes = [];
145        $expected = [];
146        /** @var array<string, int> $occurrences */
147        $occurrences = [];
148
149        foreach ($terminals as $terminal) {
150            $occurrence = $occurrences[$terminal] ?? 0;
151            $occurrences[$terminal] = $occurrence + 1;
152
153            [$lexeme, $tokens] = $this->realizer->realize($terminal, $plan?->lexemeAt($terminal, $occurrence));
154            $lexemes[] = $lexeme;
155            array_push($expected, ...$tokens);
156        }
157
158        $sql = TokenJoiner::join(
159            $lexemes,
160            [['->', '*'], ['*', '->']],
161            fn (): string => $this->realizer->trivia(),
162            fn (): string => $this->realizer->optionalTrivia(),
163        );
164
165        $actual = $this->tokenize($sql);
166        if ($this->allowSyntheticTerminals) {
167            $expected = $actual;
168        }
169        if ($actual !== $expected) {
170            throw LexicalException::roundTripMismatch('SQLite', $this->profileVersion, $expected, $actual, $sql);
171        }
172
173        return $sql;
174    }

244     *
245     * @param GenerationPlan<bool> $plan Plan naming the lexeme kind and its bounds

252    public function generate(GenerationPlan $plan): string
253    {
254        $target = $plan->lexicalTarget();
255        $parameters = $plan->parameters();
256
257        return match ($target) {
258            'quoted_identifier' => $this->generateQuotedIdentifier($parameters['minLength'], $parameters['maxLength']),
259            'string_literal' => $this->generateStringLiteral($parameters['minLength'], $parameters['maxLength']),
260            'integer_literal' => $this->generateIntegerLiteral($parameters['min'], $parameters['max']),
261            'decimal_literal' => $this->generateDecimalLiteral($parameters['precision'], $parameters['scale']),
262            default => throw new InvalidArgumentException("Unknown SQLite lexical generation target: {$target}"),
263        };
264    }
