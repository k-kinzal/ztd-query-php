7/**
8 * Turns raw fuzzer input into a deterministic RNG seed.
9 *
10 * php-fuzzer mutates arbitrary byte strings, including strings shorter than the
11 * four bytes crc32() needs to spread its output over the whole range, so short
12 * input is padded before it is hashed. The same bytes always produce the same
13 * seed, which is what makes a reported finding reproducible from its corpus
14 * entry alone.
15 */
16final class CorpusSeed
17{
18    /**
19     * Derives the RNG seed that a generator run should use for the given input.
20     *
21     * @param string $input Raw fuzzer input (mutated bytes)
22     *
23     * @return int Seed to hand to the Faker generator
24     */
25    public function forInput(string $input): int
26    {
27        if (strlen($input) < 4) {
28            $input = str_pad($input, 4, "\0");
29        }
30
31        return crc32($input);
32    }
