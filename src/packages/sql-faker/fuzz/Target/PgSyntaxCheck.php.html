11/**
12 * Executes generated SQL against PostgreSQL and reports unexpected rejections.
13 *
14 * PostgreSQL defers preparation, so PDO::prepare() accepts syntax the server
15 * would still refuse. The statement therefore has to be executed, and each run
16 * is wrapped in a savepoint so a rejected statement does not poison the
17 * surrounding transaction. A fuzz run has no schema, so the grammar
18 * legitimately produces statements that reference missing objects; those
19 * SQLSTATEs are tolerated. Any other rejection means the grammar emitted
20 * something PostgreSQL cannot parse, which is a finding and surfaces as an
21 * Error for php-fuzzer to record.
22 */
23final class PgSyntaxCheck
24{
25    /**
26     * @param PDO $pdo Connection to the PostgreSQL instance under test, with a transaction already open
27     * @param PgBracketIndirection $bracketIndirection Recognises the dialect's bracketed-indirection syntax error
28     */
29    public function __construct(
30        private readonly PDO $pdo,
31        private readonly PgBracketIndirection $bracketIndirection,
32    ) {
33    }
34
35    /**
36     * Verifies that PostgreSQL accepts the generated statement.
37     *
38     * @param string $sql Statement produced by the grammar
39     * @param int $seed Seed that produced the statement, so a finding can be replayed
40     *
41     * @throws Error When PostgreSQL rejects the statement for a reason the grammar should not produce
42     */
43    public function verify(string $sql, int $seed): void
44    {
45        try {
46            $this->pdo->exec('SAVEPOINT fuzz_check');
47            $this->pdo->exec($sql);
48            $this->pdo->exec('RELEASE SAVEPOINT fuzz_check');
49        } catch (PDOException $rejection) {
50            $this->rollBack();
51
52            $sqlState = is_string($rejection->errorInfo[0] ?? null) ? $rejection->errorInfo[0] : '';
53
54            // A schema-less fuzz run cannot satisfy name lookups, so those SQLSTATEs are expected.
55            $acceptable = match ($sqlState) {
56                // SQLSTATE 42704: Undefined object
57                '42704' => true,
58                // SQLSTATE 42P01: Undefined table
59                '42P01' => true,
60                // SQLSTATE 42703: Undefined column
61                '42703' => true,
62                // SQLSTATE 3F000: Invalid schema name
63                '3F000' => true,
64                // SQLSTATE 0A000: Feature not supported
65                '0A000' => true,
66                // SQLSTATE 42809: Wrong object type
67                '42809' => true,
68                // SQLSTATE 25001: Active sql transaction
69                '25001' => true,
70                // SQLSTATE 22023: Invalid parameter value
71                '22023' => true,
72                // SQLSTATE 26000: Invalid sql statement name
73                '26000' => true,
74                // SQLSTATE 2BP01: Dependent objects still exist
75                '2BP01' => true,
76                // SQLSTATE 42602: Invalid name
77                '42602' => true,
78                // SQLSTATE 42883: Undefined function
79                '42883' => true,
80                // SQLSTATE 42939: Reserved name
81                '42939' => true,
82                // SQLSTATE 42P07: Duplicate table
83                '42P07' => true,
84                // SQLSTATE 42P10: Invalid column reference
85                '42P10' => true,
86                // SQLSTATE 58P01: Undefined file
87                '58P01' => true,
88                // SQLSTATE 42P13: Invalid function definition
89                '42P13' => true,
90                // SQLSTATE 3D000: Invalid catalog name
91                '3D000' => true,
92                // SQLSTATE 42P03: Duplicate cursor
93                '42P03' => true,
94                // SQLSTATE 22P02: Invalid text representation
95                '22P02' => true,
96                // SQLSTATE 25P01: No active sql transaction
97                '25P01' => true,
98                // SQLSTATE 42601: Syntax error (bracket indirection)
99                '42601' => $this->bracketIndirection->explains($sql, $rejection->getMessage()),
100                default => false,
101            };
102
103            if ($acceptable) {
104                return;
105            }
106
107            throw new Error(
108                "Unexpected error in generated SQL\n" .
109                "Seed: $seed\n" .
110                "SQL: $sql\n" .
111                "SQLSTATE: $sqlState\n" .
112                "Error: {$rejection->getMessage()}",
113                0,
114                $rejection
115            );
116        }
117    }
118
119    /**
120     * Returns the connection to a state where the next statement can be checked.
121     *
122     * Rolling back to the savepoint is enough for an ordinary rejection. When
123     * the savepoint itself is gone the whole transaction is unusable, so it is
124     * discarded and a fresh one is opened instead.
125     */
126    public function rollBack(): void
127    {
128        try {
129            $this->pdo->exec('ROLLBACK TO SAVEPOINT fuzz_check');
130
131            return;
132        } catch (PDOException $savepointLoss) {
133            // The savepoint did not survive the rejection, so the transaction is replaced below.
134            fwrite(STDERR, "Savepoint rollback failed: {$savepointLoss->getMessage()}\n");
135        }
136
137        try {
138            $this->pdo->exec('ROLLBACK');
139        } catch (PDOException $rollbackFailure) {
140            // No transaction was open, which the BEGIN below settles on its own.
141            fwrite(STDERR, "Transaction rollback failed: {$rollbackFailure->getMessage()}\n");
142        }
143
144        $this->pdo->exec('BEGIN');
145    }
146}
