28     * Expands the requested dependency and retains unresolved children.
29     */
30    public function value(Frame $frame, string $address, int $header, int $depth): Term
31    {
32        if ((new Invariant())->check($this->engine, $frame, $address, $header)) {
33            return $this->version($frame->iteration($header, 0), $address, $header, 0, $depth);
34        }
35        if (isset($frame->iterations[$header])) {
36            return $this->version($frame, $address, $header, $frame->iterations[$header], $depth);
37        }
38        $condition = $frame->graph->body->blocks[$header]->terminator->operand;
39        $iteration = 0;
40        for (; $iteration < $this->engine->context->budget->iterations; $iteration++) {
41            $version = $frame->iteration($header, $iteration);
42            $test = $this->engine->value($version, $condition, $depth);
43            $truths = array_map(static fn (array $alternative): ?bool => (new Operations())->truth($alternative[0]), iterator_to_array((new Choices())->alternatives($test), false));
44            if ($truths !== [] && array_unique($truths, SORT_REGULAR) === [false]) {
45                return $this->version($version, $address, $header, $iteration, $depth);
46            }
47            if ($truths === [] || in_array(null, $truths, true) || in_array(false, $truths, true) || $this->engine->context->boundary($depth) !== null) {
48                break;
49            }
50        }
51        $initial = $this->version($frame->iteration($header, 0), $address, $header, 0, $depth);
52        $reason = $this->engine->context->stopReason ?? ($iteration >= $this->engine->context->budget->iterations ? 'ITERATION_LIMIT' : 'UNKNOWN_ITERATION_COUNT');
53        $reference = $this->engine->context->reference($frame, (new Storage($this->engine))->key($frame, $address), $frame->graph->body->source, reason: $reason, kind: 'recursive');
54        $update = $this->version($frame->iteration($header, 1), $address, $header, 1, $depth);
55        return new Term('recurrence', operands: [$initial, $update, $reference], attributes: ['reason' => $reason, 'header' => $header]);
56    }

58    /**
59     * Resolves one demanded recurrence version from its predecessor.
60     */
61    public function version(Frame $frame, string $address, int $header, int $iteration, int $depth): Term
62    {
63        $key = $frame->identity . ':loop-value:' . $address . ':' . $header . ':' . $iteration . ':' . $depth;
64        if (isset($this->engine->context->values[$key])) {
65            return $this->engine->context->values[$key];
66        }
67        $alternatives = [];
68        foreach ($frame->graph->predecessors[$header] ?? [] as $parent) {
69            $backedge = $parent >= $header;
70            $post = $this->postCondition($frame, $header);
71            if (!$post && ($iteration === 0) === $backedge) {
72                continue;
73            }
74            $source = $backedge && !$post ? $frame->iteration($header, $iteration - 1) : $frame;
75            $value = (new Storage($this->engine))->search($source, $address, $parent, count($source->graph->body->blocks[$parent]->instructions), $depth);
76            $alternatives[] = [$value, []];
77        }
78        return $this->engine->context->values[$key] = (new Choices())->make($alternatives);
79    }
80    /**
81     * Recognizes a loop whose first body precedes its first condition.
82     */
83    public function postCondition(Frame $frame, int $header): bool
84    {
85        foreach ($frame->graph->predecessors[$header] ?? [] as $parent) {
86            if ($parent < $header) {
87                return false;
88            }
89        }
90        return true;
91    }

96     */
97    public function parents(Frame $frame, int $block): array
98    {
99        $parents = $frame->graph->predecessors[$block] ?? [];
100        foreach ($frame->iterations as $header => $version) {
101            if ($frame->graph->body->blocks[$header]->terminator->targets[0] === $block && $this->postCondition($frame, $header)) {
102                return array_values(array_filter($parents, static fn (int $parent): bool => $version === 0 ? $parent !== $header : $parent === $header));
103            }
104        }
105        return $parents;
106    }
