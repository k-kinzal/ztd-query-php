28    /**
29     * Widens an unbounded loop instead of reporting its first iterations as exhaustive.
30     * @param CallableGraph $callable Current callable
31     * @param State $state Loop header entry
32     * @return bool Whether the loop was replaced by a residual exit
33     */
34    public function widen(CallableGraph $callable, State $state): bool
35    {
36        $header = $state->block;
37        $state->visits[$header] = ($state->visits[$header] ?? 0) + 1;
38        $state->loopEntries[$header] ??= $state->memory->cells;
39        $state->loopGuards[$header] ??= $state->guard;
40        if ($state->visits[$header] <= $this->context->query->budget()->iterations) {
41            return false;
42        }
43        $previous = $state->approximations[$header] ?? $state->loopEntries[$header];
44        $lattice = new Lattice($this->context->models->extensions->domains);
45        $structure = $this->structure($state);
46        $stable = isset($state->approximations[$header]) && ($state->loopStructures[$header] ?? null) === $structure;
47        $state->loopStructures[$header] = $structure;
48        $next = [];
49        foreach (array_unique([...array_keys($previous), ...array_keys($state->memory->cells)]) as $root) {
50            $a = $previous[$root] ?? new Term('uninitialized');
51            $b = $state->memory->cells[$root] ?? new Term('uninitialized');
52            $stable = $stable && $lattice->contains($a, $b);
53            $next[$root] = $lattice->widen($a, $b);
54        }
55        $state->memory->cells = $next;
56        $state->approximations[$header] = $next;
57        $state->guard = $state->loopGuards[$header];
58        $state->constraints = [];
59        $this->context->frontier('WIDENED', $callable->source, 'loop-fixed-point');
60        if ($stable) {
61            $state->stableHeader = $header;
62        }
63        if ($state->visits[$header] > $this->context->query->budget()->iterations + 64) {
64            $this->context->frontier('BUDGET_EXCEEDED', $callable->source, 'loop-fixed-point');
65            (new Havoc())->symbols($state, 'BUDGET_EXCEEDED');
66            return true;
67        }
68        return false;
69    }

71    /**
72     * Hashes loop alias and cursor structure without expanding shared value subgraphs.
73     * @param State $state Loop header state
74     * @return string Alias and iteration identity
75     */
76    public function structure(State $state): string
77    {
78        $identity = $this->context->identity;
79        $iterators = [];
80        foreach ($state->iterators as $id => $iterator) {
81            $iterators[$id] = [$identity->key($iterator->array), $iterator->location, $this->position($iterator)];
82        }
83        return hash('sha256', serialize([$state->locals, $iterators, $state->memory->liveArrays, $state->memory->unknownShared, $state->observed, $state->observedQueries]));
84    }

86    /**
87     * Distinguishes cursor positions only where they select a known entry.
88     * Positions past the known entries of an unknown iterable all read unknown entries, so they share one structure.
89     * @param IteratorCursor $iterator Loop cursor
90     * @return int Position, capped after the known entries
91     */
92    public function position(IteratorCursor $iterator): int
93    {
94        $array = $iterator->array;
95        if ($array->kind === 'array' && ($array->attributes['open'] ?? false) !== true) {
96            return $iterator->position;
97        }
98        $head = (new Arrays())->head($array);
99        return min($iterator->position, $head === null ? 0 : count($head->operands) + 1);
100    }
