26/**
27 * Derives candidate values repeatedly from one captured set of PHP sources.
28 * @visibility public
29 * @example Preparing source once for independent queries
30 *     $path = tempnam(sys_get_temp_dir(), 'deriver-example-');
31 *     file_put_contents($path, '<?php function answer() { return 42; } function other() { return 7; }');
32 *     $deriver = \Deriver\Deriver::prepare([$path]);
33 *     unlink($path);
34 *     $deriver->derive('answer#return')[0]->result // => 42
35 *     $deriver->derive('other#return')[0]->result // => 7
36 */
37final class Deriver
38{

41     */
42    private function __construct(private readonly SelectorResolution $selection, private readonly CandidateQuery $evaluation)

44    }

46    /**
47     * Captures and parses the selected files without executing application code.
48     * Builds reverse indexes for calls, property writes, closure owners and script globals.
49     * Function, method and closure bodies are compiled on demand and reused by later queries.
50     * The returned instance keeps these contents even if the files later change or disappear.
51     * @param list<string> $files Local paths, for example the result of Finder::files()
52     * @param DeriverConfig|null $configuration Independent language and evaluation settings; null selects the defaults
53     * @throws Exception\CollectorException If a selected file cannot be collected
54     * @throws Exception\ParserException If a selected source cannot be parsed
55     * @throws Exception\ResolutionException If a selected source cannot be resolved
56     * @throws Exception\InvalidInputException If declarations are incomplete or declarations or evaluator registrations conflict
57     */
58    public static function prepare(array $files, ?DeriverConfig $configuration = null): self
59    {
60        $settings = $configuration ?? DeriverConfig::default();
61        $sources = (new Collector())->collect($files);
62        $target = new TargetProfile($settings->phpVersion, $settings->integerBits, $settings->floatPrecision);
63        $environment = array_map(Term::fromNative(...), $settings->environment);
64        $evaluationConfig = new EvaluatorConfig($target, $environment, $settings->standardEvaluators, array_map((new MicroEvaluatorConversion())->convert(...), $settings->evaluators));
65        $evaluators = new Evaluators($target, standardEvaluators: $settings->standardEvaluators);
66        $identity = new SnapshotIdentity($sources, $evaluationConfig, $evaluators);
67        $parser = new Parser(new ParserConfig($settings->phpVersion));
68        $resolver = new Resolver();
69        $trees = [];
70        foreach ($sources as $source) {
71            $trees[] = $resolver->resolve($parser->parse($source));
72        }
73        $program = new Index($identity->id, $trees);
74        $diagnostics = $program->diagnostics();
75        foreach ($diagnostics as $diagnostic) {
76            if ($diagnostic->code === 'INCOMPLETE_SOURCE') {
77                throw new Exception\InvalidInputException('Malformed captured PHP in ' . $diagnostic->at->path . ': ' . $diagnostic->operation);
78            }
79        }
80        $snapshot = $identity->snapshot(array_map(Frontier::fromDiagnostic(...), $diagnostics));
81        return new self(new SelectorResolution($program), new CandidateQuery(new DefinitionIndex($program), $evaluationConfig, $evaluators, $snapshot));
82    }
83
84    /**
85     * Answers a selector against the prepared sources with fresh evaluation state.
86     * Each result owns its evidence; the deriver does not retain query results.
87     * @param string $query Callable or source-position selector
88     * @return CandidateCollection Candidates and their self-contained evidence
89     * @throws Exception\InvalidInputException If the selector cannot identify a supported target
90     */
91    public function derive(string $query): CandidateCollection
92    {
93        return $this->evaluation->derive($this->selection->resolve(Query::parse($query)));
94    }
