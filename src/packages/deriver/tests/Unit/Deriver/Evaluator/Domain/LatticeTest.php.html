249    {
250        $source = '<?php function target(int $n){$s="";for($i=0;$i<$n;$i++){$s.="x";}return $s;}';
251        $symbol = 'target';
252        $input = new Sources([new Source('fixture.php', $source)]);
253        $configuration = new DeriverConfig();
254        $contributions = new Contributions();
255        $target = new TargetProfile($configuration->phpVersion, $configuration->integerBits, $configuration->floatPrecision);
256        $providers = new ProviderExpansion($input, array_map(Term::fromNative(...), $configuration->environment), $contributions);
257        $contributions1 = $providers->contributions;
258        $evaluators = new Evaluators(new TargetProfile($configuration->phpVersion, $configuration->integerBits, $configuration->floatPrecision), new Registrations($contributions1->evaluators, $contributions1->providers, $contributions1->domains, $contributions1->intrinsics, $contributions1->stateSlots), $configuration->standardEvaluators);
259        $settings = new EvaluatorConfig($target, $providers->environment, $configuration->standardEvaluators, array_map((new MicroEvaluatorConversion())->convert(...), $configuration->evaluators));
260        $identity = new SnapshotIdentity($providers->input, $settings, $evaluators, $providers->manifest);
261        $snapshotId = $identity->id;
262        $input1 = $providers->input;
263        $parser = new Parser(new ParserConfig($target->version));
264        $trees = array_map(static fn (Source $source1): File => (new Resolver())->resolve($parser->parse(new Source(Path::normalize($source1->path), $source1->contents, $source1->declarationsOnly))), iterator_to_array($input1, false));
265        $program = new Index($snapshotId, $trees);
266        $result = (new QueryExecution($program, new ExecutionEvaluatorConfig($target, $providers->environment, $configuration->standardEvaluators), $evaluators, $identity->snapshot(array_map(Frontier::fromDiagnostic(...), $program->diagnostics()))))->derive(new ReturnQuery($symbol));
267        self::assertSame(DependencyClosure::Closed, $result->assessment->closure);
268        self::assertSame(ValuePrecision::Abstract, $result->assessment->precision);
269        self::assertSame('WIDENED', $result->frontiers[0]->code);
270        self::assertSame(Enumeration::NotEnumerated, $result->assessment->enumeration);
271    }

313    {
314        $domain = new class () implements AbstractDomain {
315            /**
316             * @return string Domain identity
317             */
318            #[Override]
319            public function id(): string
320            {
321                return 'example.policy';
322            }
323            /**
324             * @return string Semantic version
325             */
326            #[Override]
327            public function version(): string
328            {
329                return '1';
330            }
331            /**
332             * @return DomainFact Empty set
333             */
334            #[Override]
335            public function bottom(): DomainFact
336            {
337                return new DomainFact($this->id(), new Term(TermKind::Bottom));
338            }
339            /**
340             * @return DomainFact All policy records
341             */
342            #[Override]
343            public function top(): DomainFact
344            {
345                return new DomainFact($this->id(), new Term(TermKind::Top));
346            }
347            /**
348             * @param DomainFact $a Contained fact
349             * @param DomainFact $b Containing fact
350             * @return bool Flat-lattice inclusion
351             */
352            #[Override]
353            public function lessOrEqual(DomainFact $a, DomainFact $b): bool
354            {
355                return $a->representation->kind === TermKind::Bottom || $b->representation->kind === TermKind::Top || $a->representation->isConcrete() && $b->representation->isConcrete() && $a->representation->native() === $b->representation->native();
356            }
357            /**
358             * @param DomainFact $a First possibility
359             * @param DomainFact $b Second possibility
360             * @return DomainFact Inclusive alternative
361             */
362            #[Override]
363            public function join(DomainFact $a, DomainFact $b): DomainFact
364            {
365                if ($this->lessOrEqual($a, $b)) {
366                    return $b;
367                }
368                return $this->lessOrEqual($b, $a) ? $a : $this->top();
369            }
370            /**
371             * @param DomainFact $previous Previous approximation
372             * @param DomainFact $next New approximation
373             * @return DomainFact Finite widening
374             */
375            #[Override]
376            public function widen(DomainFact $previous, DomainFact $next): DomainFact
377            {
378                return $this->join($previous, $next);
379            }
380            /**
381             * @param DomainFact $value Policy record
382             * @param Projection $projection Selected fields
383             * @return DomainFact Projected fact
384             */
385            #[Override]
386            public function project(DomainFact $value, Projection $projection): DomainFact
387            {
388                $term = $value->representation;
389                foreach ($projection->path as $key) {
390                    $term = $term->operands[$key] ?? new Term(TermKind::Top);
391                }
392                return new DomainFact($this->id(), $term);
393            }
394        };
395        $top = $domain->top()->term();
396        $lattice = new Lattice([$domain->id() => $domain]);
397        self::assertFalse($lattice->contains($top, Term::constant($domain->id())));
398        self::assertTrue($lattice->contains($lattice->widen($top, Term::constant($domain->id())), Term::constant($domain->id())));
399    }

506            #[Override]
507            public function lessOrEqual(DomainFact $a, DomainFact $b): bool
508            {
509                return $a->representation->kind === TermKind::Bottom || $b->representation->kind === TermKind::Top || $a->representation->isConcrete() && $b->representation->isConcrete() && $a->representation->native() === $b->representation->native();
510            }

512             * @param DomainFact $a First possibility
513             * @param DomainFact $b Second possibility
514             * @return DomainFact Inclusive alternative
515             */
516            #[Override]
517            public function join(DomainFact $a, DomainFact $b): DomainFact
518            {
519                if ($this->lessOrEqual($a, $b)) {
520                    return $b;
521                }
522                return $this->lessOrEqual($b, $a) ? $a : $this->top();
523            }

527             * @return DomainFact Finite widening

529            #[Override]
