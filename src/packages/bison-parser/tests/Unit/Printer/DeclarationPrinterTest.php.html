69    public function testPrint(): void
70    {
71        $printer = new DeclarationPrinter();
72        $at = new Location(1, 1);
73        $num = new Symbol(SymbolKind::Identifier, 'NUM', $at);
74
75        self::assertSame('%{ int x; %}', $printer->print(new Prologue(' int x; ', $at)));
76        self::assertSame('%pure_parser', $printer->print(new Flag('pure-parser', '%pure_parser', $at)));
77        self::assertSame('%defines', $printer->print(new Option('header', '%defines', null, $at)));
78        self::assertSame('%require "3.8"', $printer->print(new Option('require', '%require', '3.8', $at)));
79        self::assertSame('%define api.pure full', $printer->print(new Define('api.pure', 'full', DefineForm::Keyword, $at)));
80        self::assertSame('%expect 2', $printer->print(new Expect(2, false, $at)));
81        self::assertSame('%expect-rr 1', $printer->print(new Expect(1, true, $at)));
82        self::assertSame('%initial-action { @$ = 1; }', $printer->print(new InitialAction(' @$ = 1; ', $at)));
83        self::assertSame('%lex-param {int a} {int b}', $printer->print(new Param(ParamKind::Lex, ['int a', 'int b'], $at)));
84        self::assertSame('%union { int i; }', $printer->print(new UnionDeclaration(null, ' int i; ', $at)));
85        self::assertSame('%union YYSTYPE { int i; }', $printer->print(new UnionDeclaration('YYSTYPE', ' int i; ', $at)));
86        self::assertSame('%start program', $printer->print(new Start([new Symbol(SymbolKind::Identifier, 'program', $at)], $at)));
87        self::assertSame('%printer { p($$); } <*> NUM', $printer->print(new CodeProps(true, ' p($$); ', [new Tag(Tag::ANY, $at), $num], $at)));
88        self::assertSame('%destructor { free($$); } <>', $printer->print(new CodeProps(false, ' free($$); ', [new Tag(Tag::NONE, $at)], $at)));
89        self::assertSame('%code { x }', $printer->print(new Code(null, ' x ', $at)));
90        self::assertSame('%code requires { x }', $printer->print(new Code('requires', ' x ', $at)));
91        self::assertSame('%token <int> NUM 258 "number"', $printer->print(new SymbolDeclaration(SymbolClass::Token, [new SymbolEntry($num, 'int', 258, new Alias('number', false, $at))], $at)));
92        self::assertSame("%left '+' '-'", $printer->print(new PrecedenceDeclaration(Associativity::Left, [new SymbolEntry(new Symbol(SymbolKind::CharLiteral, '+', $at), null, null, null), new SymbolEntry(new Symbol(SymbolKind::CharLiteral, '-', $at), null, null, null)], $at)));
93    }
