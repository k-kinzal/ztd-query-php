21    public function testProjectsMySqlFunctionsThroughExistingAndIncomingRows(): void
22    {
23        $projector = new MySqlNativeUpsertProjector();
24
25        $result = $projector->project(
26            'SELECT 1 AS `id`, 15 AS `price`, 2 AS `version`, \'{}\' AS `meta`',
27            'products',
28            ['id', 'price', 'version', 'meta'],
29            ['empty_key' => [], 'PRIMARY' => ['id'], 'version_key' => ['price', 'version']],
30            [
31                'price' => 'IF(VALUES(version) > version, VALUES(price), price)',
32                'meta' => "JSON_SET(meta, '$.price', VALUES(price))",
33            ],
34            'VALUES(version) > version',
35        );
36
37        self::assertSame(
38            'SELECT `__ztd_incoming`.`id` AS `id`, `__ztd_incoming`.`price` AS `price`, '
39            . '`__ztd_incoming`.`version` AS `version`, `__ztd_incoming`.`meta` AS `meta`, '
40            . '(SELECT IF(`__ztd_incoming`.`version` > `__ztd_existing`.`version`, '
41            . '`__ztd_incoming`.`price`, `__ztd_existing`.`price`) FROM `products` AS `__ztd_existing` '
42            . 'WHERE ((`__ztd_existing`.`id` = `__ztd_incoming`.`id`) OR '
43            . '(`__ztd_existing`.`price` = `__ztd_incoming`.`price` AND '
44            . '`__ztd_existing`.`version` = `__ztd_incoming`.`version`)) LIMIT 1) AS `__ztd_upsert_value_0`, '
45            . "(SELECT JSON_SET(`__ztd_existing`.`meta`, '$.price', `__ztd_incoming`.`price`) "
46            . 'FROM `products` AS `__ztd_existing` WHERE ((`__ztd_existing`.`id` = `__ztd_incoming`.`id`) OR '
47            . '(`__ztd_existing`.`price` = `__ztd_incoming`.`price` AND '
48            . '`__ztd_existing`.`version` = `__ztd_incoming`.`version`)) LIMIT 1) AS `__ztd_upsert_value_1`, '
49            . '(SELECT `__ztd_incoming`.`version` > `__ztd_existing`.`version` '
50            . 'FROM `products` AS `__ztd_existing` WHERE ((`__ztd_existing`.`id` = `__ztd_incoming`.`id`) OR '
51            . '(`__ztd_existing`.`price` = `__ztd_incoming`.`price` AND '
52            . '`__ztd_existing`.`version` = `__ztd_incoming`.`version`)) LIMIT 1) AS `__ztd_upsert_predicate` '
53            . "FROM (SELECT 1 AS `id`, 15 AS `price`, 2 AS `version`, '{}' AS `meta`) AS `__ztd_incoming`",
54            $result,
