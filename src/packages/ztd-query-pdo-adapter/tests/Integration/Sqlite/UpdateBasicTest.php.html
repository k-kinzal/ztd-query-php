98        self::assertSame(2, $ztdPdo->exec('DELETE FROM pairs'));
99        $statement = $ztdPdo->query('SELECT * FROM pairs');
100        self::assertNotFalse($statement);

104    public function testUpdateWithGroupedInSubquery(): void
105    {
106        $rawPdo = new PDO('sqlite::memory:', null, null, [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION, PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC]);
107        $rawPdo->exec('CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT, tier TEXT)');
108        $rawPdo->exec('CREATE TABLE orders (id INTEGER PRIMARY KEY, user_id INTEGER, total REAL, status TEXT)');
109        $rawPdo->exec("INSERT INTO users VALUES (1, 'Alice', 'standard'), (2, 'Bob', 'standard')");
110        $rawPdo->exec("INSERT INTO orders VALUES (1, 1, 500, 'completed'), (2, 1, 300, 'completed'), (3, 2, 100, 'completed')");
111        $ztdPdo = ZtdPdo::fromPdo($rawPdo);
112        $ztdPdo->exec("INSERT INTO users VALUES (1, 'Alice', 'standard'), (2, 'Bob', 'standard')");
113        $ztdPdo->exec("INSERT INTO orders VALUES (1, 1, 500, 'completed'), (2, 1, 300, 'completed'), (3, 2, 100, 'completed')");
114
115        $sql = "UPDATE users SET tier = 'premium' WHERE id IN (SELECT user_id FROM orders WHERE status = 'completed' GROUP BY user_id HAVING SUM(total) > 400)";
116        self::assertSame($rawPdo->exec($sql), $ztdPdo->exec($sql));
117
118        $raw = $rawPdo->query('SELECT * FROM users ORDER BY id');
119        $shadow = $ztdPdo->query('SELECT * FROM users ORDER BY id');
120        self::assertNotFalse($raw);
121        self::assertNotFalse($shadow);
122        self::assertSame($raw->fetchAll(), $shadow->fetchAll());
123    }
