21/**
22 * Reports the attributes a definition lacks or combines in a way its command rejects.
23 *
24 * Rule: PG-DEFINE-CHECK-001. Attribute names are compared respecting case
25 * (an unquoted name is folded to lower case). An operator needs a function
26 * (`function` or `procedure`) and, as postfix operators no longer exist, a
27 * right argument type. An aggregate needs a state type and a state
28 * transition function; the moving-aggregate attributes come with `mstype`,
29 * and `mstype` with `msfunc` and `minvfunc`; serialization needs both
30 * functions; only an ordered-set aggregate is hypothetical; an old-style
31 * aggregate names its input type with `basetype`, which is redundant in the
32 * new style. A base type needs input and output functions and a modifier
33 * input function for a modifier output function. A range type needs a
34 * subtype. A collation copied `from` another takes no other attribute and
35 * `locale` excludes `lc_collate` and `lc_ctype`; otherwise its provider
36 * (libc when not given) needs `lc_collate` and `lc_ctype` or `locale`, or for
37 * ICU and builtin `locale`, and only ICU takes rules or is nondeterministic.
38 * A text search parser needs its start, gettoken, end and lextypes methods, a
39 * template its lexize method, a dictionary its template, and a configuration
40 * a parser or a configuration to copy, but not both. Attribute values and
41 * unrecognized names are PG-DEFINE-ATTRIBUTE-001. Terminates: one pass over
42 * the attributes.
43 * Source: https://www.postgresql.org/docs/17/sql-createaggregate.html, https://www.postgresql.org/docs/17/sql-createoperator.html,
44 * https://www.postgresql.org/docs/17/sql-createtype.html, https://www.postgresql.org/docs/17/sql-createcollation.html,
45 * `DefineOperator`, `DefineAggregate`, `DefineType`, `DefineRange`, `DefineCollation`, `DefineTSParser`, `DefineTSTemplate`,
46 * `DefineTSDictionary` and `DefineTSConfiguration` in `src/backend/commands` of PostgreSQL 17.
47 * Status: Implemented.
48 *
49 * @visibility SqlSemantics\Platform\PostgreSql
50 */
51final class DefineChecks

58    /**
59     * Reports the problems of the attributes and of their combination.
60     */
61    public function derive(Define $define, Derivation $derivation): void
62    {
63        $attributes = $define->definition ?? [];
64        $command = (new AttributeReader())->command($define->kind);
65        (new AttributeChecks())->derive($derivation, $command, $attributes, $define->kind === ObjectKind::Type || $define->kind === ObjectKind::Collation);
66        $present = $this->present($attributes);
67        $problems = match (true) {
68            $define->kind === ObjectKind::Operator => $this->operator($present),
69            $define->kind === ObjectKind::Aggregate => $this->aggregate($present, $define->arguments),
70            $define->kind === ObjectKind::Type => $define->definition === null ? [] : $this->baseType($present),
71            $define->kind === ObjectKind::Collation => $this->collation($present, count($attributes), $derivation->context->profile->grammar),
72            default => $this->search($define->kind, $present),
73        };
74        foreach ($problems as $problem) {
75            $derivation->report($problem);
76        }
77    }

85    {

100    {
101        $present = [];
102        foreach ($attributes as $attribute) {
103            if ($attribute->known !== null) {

131    /**
132     * Answers the missing and contradictory attributes of an aggregate.
133     *
134     * @param array<string, Attribute> $present
135     *
136     * @return list<AttributeProblem>
137     */
138    public function aggregate(array $present, ?AggregateArguments $arguments): array
139    {
140        $problems = [];
141        foreach (['stype' => 'stype1', 'sfunc' => 'sfunc1'] as $name => $obsolete) {
142            if (!isset($present[$name]) && !isset($present[$obsolete])) {
143                $problems[] = new AttributeProblem(AttributeProblemKind::MissingAggregateAttribute, [$name]);
144            }
145        }
146        $moving = isset($present['mstype']);
147        $space = $present['msspace'] ?? null;
148        $given = ['msfunc' => isset($present['msfunc']), 'minvfunc' => isset($present['minvfunc']), 'mfinalfunc' => isset($present['mfinalfunc']), 'msspace' => $space !== null && (!$space->value instanceof IntegerArgument || $space->value->value() !== 0), 'minitcond' => isset($present['minitcond'])];
149        foreach ($given as $name => $set) {
150            if ($moving && !$set && in_array($name, ['msfunc', 'minvfunc'], true)) {
151                $problems[] = new AttributeProblem(AttributeProblemKind::MissingMovingAttribute, [$name]);
152            }
153            if (!$moving && $set) {
154                $problems[] = new AttributeProblem(AttributeProblemKind::MovingWithoutState, [$name]);
155            }
156        }
157        $hypothetical = ($present['hypothetical'] ?? null)?->value;
158        if ($hypothetical instanceof BooleanArgument && $hypothetical->value() && $arguments?->ordered === null) {
159            $problems[] = new AttributeProblem(AttributeProblemKind::HypotheticalPlain);
160        }
161        if ($arguments === null xor isset($present['basetype'])) {
162            $problems[] = new AttributeProblem($arguments === null ? AttributeProblemKind::MissingInputType : AttributeProblemKind::RedundantBaseType);
163        }
164        if (isset($present['serialfunc']) xor isset($present['deserialfunc'])) {
165            $problems[] = new AttributeProblem(AttributeProblemKind::HalfSerialization);
166        }
167
168        return $problems;
169    }

194    /**
195     * Answers the contradictory and missing attributes of a collation; a provider the server does not know stops the checks, as it does.
196     *
197     * @param array<string, Attribute> $present
198     *
199     * @return list<AttributeProblem>
200     */
201    public function collation(array $present, int $count, GrammarRelease $release): array
202    {
203        $locale = isset($present['locale']);
204        $problems = [];
205        if (($locale && (isset($present['lc_collate']) || isset($present['lc_ctype']))) || (isset($present['from']) && $count !== 1)) {
206            $problems[] = new AttributeProblem(AttributeProblemKind::Conflicting);
207        }
208        $written = ($present['provider'] ?? null)?->value;
209        $chosen = $written instanceof ChoiceArgument ? $written->choice : null;
210        $provider = isset($present['provider']) ? ($chosen instanceof CollationProvider ? $chosen : null) : CollationProvider::Libc;
211        if (isset($present['from']) || $provider === null || ($provider === CollationProvider::Builtin && $release === GrammarRelease::PostgreSql166)) {
212            return $problems;
213        }
214
215        return [...$problems, ...$this->provided($provider, $present)];
216    }

218    /**
219     * Answers the attributes a collation of a provider lacks or may not take.
220     *
221     * @param array<string, Attribute> $present
222     *
223     * @return list<AttributeProblem>
224     */
225    public function provided(CollationProvider $provider, array $present): array
226    {
227        $problems = [];
228        foreach ($provider === CollationProvider::Libc ? ['lc_collate', 'lc_ctype'] : ['locale'] as $name) {
229            if (!isset($present['locale']) && !isset($present[$name])) {
230                $problems[] = new AttributeProblem(AttributeProblemKind::MissingCollationParameter, [$name]);
231            }
232        }
233        $deterministic = ($present['deterministic'] ?? null)?->value;
234        if ($provider !== CollationProvider::Icu && $deterministic instanceof BooleanArgument && !$deterministic->value()) {
235            $problems[] = new AttributeProblem(AttributeProblemKind::Nondeterministic);
236        }
237        if ($provider !== CollationProvider::Icu && isset($present['rules'])) {
238            $problems[] = new AttributeProblem(AttributeProblemKind::RulesWithoutIcu);
239        }
240
241        return $problems;
242    }

244    /**
245     * Answers the missing or contradictory attributes of a text search object.
246     *
247     * @param array<string, Attribute> $present
248     *
249     * @return list<AttributeProblem>
250     */
251    public function search(ObjectKind $kind, array $present): array
252    {
253        $problems = [];
254        foreach ($kind === ObjectKind::TextSearchParser ? self::PARSER_METHODS : [] as $method) {
255            if (!isset($present[$method])) {
256                $problems[] = new AttributeProblem(AttributeProblemKind::MissingParserMethod, [$method]);
257            }
258        }
259        $missing = match (true) {
260            $kind === ObjectKind::TextSearchTemplate && !isset($present['lexize']) => AttributeProblemKind::MissingLexize,
261            $kind === ObjectKind::TextSearchDictionary && !isset($present['template']) => AttributeProblemKind::MissingTemplate,
262            $kind === ObjectKind::TextSearchConfiguration && isset($present['parser'], $present['copy']) => AttributeProblemKind::ParserAndCopy,
263            $kind === ObjectKind::TextSearchConfiguration && !isset($present['parser']) && !isset($present['copy']) => AttributeProblemKind::MissingParser,
264            default => null,
265        };
266        if ($missing !== null) {
267            $problems[] = new AttributeProblem($missing);
268        }
269
270        return $problems;
271    }
