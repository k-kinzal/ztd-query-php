17/**
18 * Answers the type a column definition declares.
19 *
20 * Rule: PG-COLUMN-TYPE-001. A column has the type its type name denotes
21 * (PG-TYPE-LOOKUP-001). The serial pseudo-types are expanded first, as
22 * `transformColumnDefinition` does for a one-part type name without array
23 * bounds: `smallserial`/`serial2` is `int2`, `serial`/`serial4` is `int4`,
24 * `bigserial`/`serial8` is `int8`, and the column is NOT NULL with a
25 * sequence default. An array of a serial type is an error. A type name
26 * the context cannot identify, a user type or a `pg_catalog` name an
27 * undeclared type of an earlier schema may hide, declares the column with
28 * the type that name denotes on the search path (`NamedOnPath`). A type
29 * name that is an error has no descriptor, and the declaration stops there.
30 * Source: https://www.postgresql.org/docs/17/datatype-numeric.html#DATATYPE-SERIAL.
31 * Status: Implemented.
32 *
33 * @visibility SqlSemantics\Platform\PostgreSql
34 */
35final class ColumnTyping
36{
37    /**
38     * The serial pseudo-types and the integer types they stand for.
39     */
40    private const SERIAL = [
41        'smallserial' => Builtin::Int2, 'serial2' => Builtin::Int2,
42        'serial' => Builtin::Int4, 'serial4' => Builtin::Int4,
43        'bigserial' => Builtin::Int8, 'serial8' => Builtin::Int8,
44    ];
45
46    /**
47     * Answers the integer type a serial type name stands for, or null when the name is not a serial type.
48     */
49    public function serial(TypeName $type): ?Builtin
50    {
51        $designation = $type->designation;
52        if (!$designation instanceof NamedDesignation || count($designation->name->parts) !== 1 || $type->setOf) {
53            return null;
54        }
55
56        return self::SERIAL[$designation->name->parts[0]->value] ?? null;
57    }
58
59    /**
60     * Answers the declared type of a column, or null when the type name is an error.
61     */
62    public function descriptor(TypeName $type, AnalysisContext $context): ?TypeDescriptor
63    {
64        $serial = $this->serial($type);
65        if ($serial !== null) {
66            return $type->array === null ? $serial : null;
67        }
68        $fact = $type->typeFact($context);
69        if ($fact instanceof Known) {
70            return $fact->descriptor;
71        }
72        $name = $type->designation instanceof NamedDesignation ? $type->designation->name->qualified() : null;
73        if (!$fact instanceof Dependent || $name === null) {
74            return null;
75        }
76        $named = new NamedOnPath($name, $fact->missing);
77
78        return $type->array === null ? $named : new ArrayOf($named);
79    }
80}
