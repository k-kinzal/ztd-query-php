21/**
22 * Lowers the domain commands.
23 *
24 * Rule: PG-DOMAIN-LOWER-001. Scope: `CreateDomainStmt`, `AlterDomainStmt`,
25 * `DomainConstraint`, `DomainConstraintElem`. Constructors: `CreateDomain`,
26 * `DomainDefaultChange`, `AlterDomainNotNull`, `AlterDomainConstraint`,
27 * `DomainCheck`, `DomainNotNull`, `DropDomainConstraint`,
28 * `ValidateDomainConstraint`. Column qualifiers, table constraints, constraint
29 * attributes and defaults are lowered by the table family. The optional AS
30 * is noise (LeafNoise). Source: https://www.postgresql.org/docs/17/sql-createdomain.html,
31 * https://www.postgresql.org/docs/17/sql-alterdomain.html. Status: Implemented.
32 *
33 * @visibility SqlSemantics\Platform\PostgreSql\Lowering
34 */
35final class DomainRule

39     */

44    /**
45     * Lowers a domain command.
46     *
47     * @throws ImplementationGap When the production has no rule
48     */
49    public function statement(Node $statement): Statement
50    {
51        $form = $this->lowering->productions->form($statement);
52        $name = $this->lowering->names->dotted($form->node(2));
53        $tables = $this->lowering->tables;
54        $names = $this->lowering->names;
55
56        return match ($form->signature) {
57            'CreateDomainStmt: CREATE DOMAIN_P any_name opt_as Typename ColQualList' => new CreateDomain($name, $this->lowering->types->typeName($form->node(4)), $tables->columnQualifiers($form->node(5))),
58            'AlterDomainStmt: ALTER DOMAIN_P any_name alter_column_default' => new DomainDefaultChange($name, $tables->columnDefault($form->node(3))),
59            'AlterDomainStmt: ALTER DOMAIN_P any_name DROP NOT NULL_P' => new AlterDomainNotNull($name, false),
60            'AlterDomainStmt: ALTER DOMAIN_P any_name SET NOT NULL_P' => new AlterDomainNotNull($name, true),
61            'AlterDomainStmt: ALTER DOMAIN_P any_name ADD_P TableConstraint' => new AlterDomainConstraint($name, $tables->tableConstraint($form->node(4))),
62            'AlterDomainStmt: ALTER DOMAIN_P any_name ADD_P DomainConstraint' => new AlterDomainConstraint($name, $this->constraint($form->node(4))),
63            'AlterDomainStmt: ALTER DOMAIN_P any_name DROP CONSTRAINT name opt_drop_behavior' => new DropDomainConstraint($name, $names->name($form->node(5)), false, $this->lowering->flags->dropBehavior($form->node(6))),
64            'AlterDomainStmt: ALTER DOMAIN_P any_name DROP CONSTRAINT IF_P EXISTS name opt_drop_behavior' => new DropDomainConstraint($name, $names->name($form->node(7)), true, $this->lowering->flags->dropBehavior($form->node(8))),
65            'AlterDomainStmt: ALTER DOMAIN_P any_name VALIDATE CONSTRAINT name' => new ValidateDomainConstraint($name, $names->name($form->node(5))),
66            default => throw ImplementationGap::production($form),
67        };
68    }

75    public function constraint(Node $constraint): DomainCheck|DomainNotNull
76    {
77        $form = $this->lowering->productions->form($constraint);
78
79        return match ($form->signature) {
80            'DomainConstraint: CONSTRAINT name DomainConstraintElem' => $this->element($form->node(2), $this->lowering->names->name($form->node(1))),
81            'DomainConstraint: DomainConstraintElem' => $this->element($form->node(0), null),
82            default => throw ImplementationGap::production($form),
83        };
84    }

86    /**
87     * Lowers `DomainConstraintElem` with the constraint name written before it.
88     *
89     * @throws ImplementationGap When the production has no rule
90     */
91    public function element(Node $element, ?Name $name): DomainCheck|DomainNotNull
92    {
93        $form = $this->lowering->productions->form($element);
94        $tables = $this->lowering->tables;
95
96        return match ($form->signature) {
97            'DomainConstraintElem: CHECK ( a_expr ) ConstraintAttributeSpec' => new DomainCheck($name, $this->lowering->expressions->expression($form->node(2)), $tables->constraintAttributes($form->node(4))),
98            'DomainConstraintElem: NOT NULL_P ConstraintAttributeSpec' => new DomainNotNull($name, $tables->constraintAttributes($form->node(2))),
99            default => throw ImplementationGap::production($form),
100        };
101    }
