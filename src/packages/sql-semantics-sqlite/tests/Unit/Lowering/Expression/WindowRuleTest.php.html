29    {
30        $operation = (new Semantics(Dialect::Sqlite))->analyze('SELECT sum(a) OVER (), sum(a) OVER (w), sum(a) OVER (PARTITION BY b), sum(a) OVER (w PARTITION BY b, c ORDER BY d), sum(a) OVER (ORDER BY d, e), sum(a) OVER (w ORDER BY d) FROM t WINDOW w AS ()');
31
32        self::assertInstanceOf(Select::class, $operation->statement);

35            self::assertInstanceOf(FunctionCall::class, $column->expression);
36            self::assertInstanceOf(WindowSpec::class, $column->expression->over);
37            self::assertNull($column->expression->over->frame);
