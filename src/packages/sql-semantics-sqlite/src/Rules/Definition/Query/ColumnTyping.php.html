25/**
26 * Derives the declared type SQLite records for a column of a view or of a table made from a query.
27 *
28 * Rule: SQLITE-QUERY-COLUMN-TYPE-001. A result expression carries a
29 * declared type text when it is a column reference, in parentheses or not:
30 * the declared type of a table column (none when the column declares no
31 * type), `INTEGER` for the row identifier, the type a view records for its
32 * column, or, for a column of a derived table or a common table, the text
33 * carried by the expression at that position in the rightmost arm of its
34 * query; a scalar subquery carries the text of the first result of its
35 * rightmost arm; COLLATE and every other expression carry none. A column
36 * of the view or table being defined carries the text of the expression at
37 * its position in the leftmost arm of the defining query.
38 *
39 * A view records the carried text when the affinity SQLite derives from that
40 * text (the five substring rules, NUMERIC for an empty text) is the affinity
41 * of the column (SQLITE-QUERY-COLUMN-AFFINITY-001) and that affinity is not
42 * the flexible numeric one; otherwise it records `NUM` for a numeric
43 * affinity, the standard name of any other affinity, and no type (NoAffinity)
44 * for a column without affinity. A table made from a query is written from
45 * the affinity alone: `INT`, `TEXT`, `REAL`, `NUM`, and no type for BLOB or
46 * no affinity. Precision: exact. Terminates: each step enters a strictly
47 * nested query or a strict sub-expression.
48 * Source: https://sqlite.org/lang_createtable.html#create_table_as_select_statements,
49 * https://sqlite.org/lang_createview.html, https://sqlite.org/datatype3.html#determination_of_column_affinity
50 * (and `columnType()`, `sqlite3SubqueryColumnTypes()` and `createTableStmt()`
51 * in select.c and build.c of the release). Status: Implemented.
52 *
53 * @visibility SqlSemantics\Platform\Sqlite
54 */
55final class ColumnTyping
56{
57    /**
58     * The standard type name a view records for each affinity the carried text does not give.
59     */
60    private const VIEW = ['Integer' => 'INT', 'Text' => 'TEXT', 'Blob' => 'BLOB', 'Real' => 'REAL', 'Numeric' => 'NUM'];
61
62    /**
63     * The declared type `CREATE TABLE ... AS` writes for each affinity.
64     */
65    private const TABLE = ['Integer' => 'INT', 'Text' => 'TEXT', 'Blob' => '', 'Real' => 'REAL', 'Numeric' => 'NUM'];

67    /**
68     * Answers the affinity SQLite derives from a type text by the ordered substring rules; an empty text gives NUMERIC.
69     */
70    public function textAffinity(string $text): Affinity
71    {
72        $text = strtr($text, 'abcdefghijklmnopqrstuvwxyz', 'ABCDEFGHIJKLMNOPQRSTUVWXYZ');
73
74        return match (true) {
75            str_contains($text, 'INT') => Affinity::Integer,
76            str_contains($text, 'CHAR'), str_contains($text, 'CLOB'), str_contains($text, 'TEXT') => Affinity::Text,
77            str_contains($text, 'BLOB') => Affinity::Blob,
78            str_contains($text, 'REAL'), str_contains($text, 'FLOA'), str_contains($text, 'DOUB') => Affinity::Real,
79            default => Affinity::Numeric,
80        };
81    }

83    /**
84     * Answers the declared type text an expression carries, or null when it carries none.
85     */
86    public function text(Scalar $expression, Facts $facts): ?string
87    {
88        while ($expression instanceof Grouped) {
89            $expression = $expression->operand;
90        }
91        if ($expression instanceof ScalarSubquery) {
92            $arms = (new ExpressionAffinity())->arms($expression->query);
93
94            return $this->arm($arms[count($arms) - 1], 0, $facts);
95        }
96        if ($expression instanceof ColumnUse || $expression instanceof DoubleQuotedWord || $expression instanceof LiteralColumn) {
97            $resolution = $facts->covers($expression) ? $facts->scalar($expression)->resolution : null;
98
99            return $resolution instanceof ResolvedColumn ? $this->resolved($resolution, $facts) : null;
100        }
101
102        return null;
103    }
104
105    /**
106     * Answers the declared type text a resolved column carries.
107     */
108    public function resolved(ResolvedColumn $resolution, Facts $facts): ?string
109    {
110        if ($resolution->slot->column !== null) {
111            return $this->column($resolution->slot->column);
112        }
113        $affinities = new ExpressionAffinity();
114        $query = $affinities->computing($resolution->relation, $facts);
115        if ($query !== null) {
116            $position = $affinities->position($query, $resolution->slot, $facts);
117            $arms = $affinities->arms($query);
118
119            return $position === null ? null : $this->arm($arms[count($arms) - 1], $position, $facts);
120        }
121        $column = $resolution->slot->declaration();
122
123        return $column === null ? null : $this->column($column);
124    }
125
126    /**
127     * Answers the declared type text of a declared column, or null when it declares none.
128     */
129    public function column(Column $column): ?string
130    {
131        $type = $column->type;
132        if ($type instanceof ColumnDomain) {
133            return $type->typed() ? $type->declared : null;
134        }
135
136        return $type instanceof NoAffinity ? null : $type->name();
137    }
138
139    /**
140     * Answers the declared type text the result of an arm at a position carries; an expanded star is a column reference.
141     */
142    public function arm(Select|ValueRow $arm, int $position, Facts $facts): ?string
143    {
144        if ($arm instanceof ValueRow) {
145            return isset($arm->values[$position]) ? $this->text($arm->values[$position], $facts) : null;
146        }
147        $item = $facts->covers($arm) ? ($facts->query($arm)->projection[$position] ?? null) : null;
148
149        return $item instanceof Field ? $this->field($item, $facts) : null;
150    }
151
152    /**
153     * Answers the declared type text a column of a defining query carries: that of the leftmost arm at its position.
154     */
155    public function leftmost(Query $query, int $position, Facts $facts): ?string
156    {
157        return $this->arm((new ExpressionAffinity())->arms($query)[0], $position, $facts);
158    }
159
160    /**
161     * Answers the declared type text an output field carries: that of its expression, or of the column an expanded star exposes.
162     */
163    public function field(Field $field, Facts $facts): ?string
164    {
165        if ($field->expression !== null) {
166            return $this->text($field->expression, $facts);
167        }
168
169        return $field->resolution instanceof ResolvedColumn ? $this->resolved($field->resolution, $facts) : null;
170    }
171
172    /**
173     * Answers the type a view records for a column with the given affinity and carried text.
174     */
175    public function viewType(DerivedAffinity $affinity, ?string $text): TypeDescriptor
176    {
177        if ($affinity->affinity === null) {
178            return new NoAffinity();
179        }
180        if ($text !== null && !$affinity->flexible && $this->textAffinity($text) === $affinity->affinity) {
181            return new ColumnDomain($text, false, $text !== '');
182        }
183
184        return new ColumnDomain(self::VIEW[$affinity->affinity->name]);
185    }
186
187    /**
188     * Answers the type `CREATE TABLE ... AS` writes for a column with the given affinity.
189     */
190    public function tableType(DerivedAffinity $affinity): ColumnDomain
191    {
192        return new ColumnDomain($affinity->affinity === null ? '' : self::TABLE[$affinity->affinity->name]);
193    }
