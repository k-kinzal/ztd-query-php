14/**
15 * Renders the spec report: one row per item with its statement, result and passing / linked
16 * targets, followed by the failure messages.
17 */
18final class SpecificationTable
19{
20    /**
21     * Renders the specification table and the messages of failed items.
22     *
23     * @param SymfonyStyle $io The console style
24     * @param OutputInterface $output Receives the table
25     * @param array<string, mixed> $report The spec report
26     *
27     * @throws InvalidInputException When a row does not have the shape of a specification record
28     */
29    public function render(SymfonyStyle $io, OutputInterface $output, array $report): void
30    {
31        $rows = [];
32        foreach (Fields::mapping($report['specifications'], 'specifications') as $id => $value) {
33            $spec = Fields::mapping($value, 'specification');
34            $statement = Text::plain($spec['statement']);
35            $statement .= "\n" . Text::plain($spec['kind']) . ' · ' . Text::plain($spec['source'] ?? $spec['origin']);
36            if ($spec['category'] !== '') {
37                $statement .= ' · ' . Text::plain($spec['category']);
38            }
39            $labels = Fields::strings($spec['labels'], 'labels');
40            if ($labels !== []) {
41                $statement .= "\nLabels: " . Text::plain(implode(', ', $labels));
42            }
43            if ($spec['reason'] !== '') {
44                $statement .= "\nReason: " . Text::plain($spec['reason']);
45            }
46            $ratio = Text::plain($spec['passed_targets'] ?? '-') . '/' . Text::plain($spec['total_targets']);
47            $rows[] = [Text::plain($id), $statement, Text::plain($spec['status']), $ratio];
48        }
49        $this->table($output, ['ID', 'Statement', 'Result', 'Tests'], $rows);
50        $io->text(count($rows) . ' item(s). Tests: passed / linked targets. Use --json for execution counts and complete traceability records.');
51        foreach (Fields::mapping($report['specifications'], 'specifications') as $id => $value) {
52            $spec = Fields::mapping($value, 'specification');
53            if ($spec['message'] !== '') {
54                $io->text(Text::plain($id) . ': ' . Text::plain($spec['message']));
55            }
56        }
57    }
58
59    /**
60     * Renders rows with the statement column wrapped to the terminal width.
61     *
62     * @param OutputInterface $output Receives the table
63     * @param list<string> $headers The ID, statement, result and tests headers
64     * @param list<list<string>> $rows The rows
65     */
66    public function table(OutputInterface $output, array $headers, array $rows): void
67    {
68        $widths = [];
69        foreach ([0 => 24, 2 => 14, 3 => 12] as $column => $maximum) {
70            $width = mb_strwidth($headers[$column]);
71            foreach ($rows as $row) {
72                $width = max($width, mb_strwidth($row[$column]));
73            }
74            $widths[$column] = min($maximum, $width);
75        }
76        (new Table($output))->setHeaders($headers)->setRows($rows)
77            ->setColumnMaxWidth(0, $widths[0])->setColumnMaxWidth(1, max(24, min(72, (new Terminal())->getWidth() - array_sum($widths) - 13)))
78            ->setColumnMaxWidth(2, $widths[2])->setColumnMaxWidth(3, $widths[3])->render();
79    }
80}
