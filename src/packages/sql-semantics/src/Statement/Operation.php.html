19/**
20 * One analyzed statement: its structure, the facts derived against a fixed context, and its SQL.
21 *
22 * The constructor is the only way to obtain an operation and it is a checking
23 * boundary. It audits the structure, derives every fact, renders the SQL, and
24 * confirms that the rendered SQL has the same structure. Nothing is filled in
25 * afterwards and nothing can be changed. A different statement is a new
26 * operation built from explicit inputs or from newly analyzed SQL; it has no
27 * guaranteed relation to this one.
28 *
29 * @visibility public
30 * @example Analyzing against the declaration another statement provides
31 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite);
32 *     $bar = $semantics->analyze('CREATE TABLE bar (foo INTEGER, baz TEXT)');
33 *     $query = $semantics->analyze('SELECT foo FROM bar', [$bar]);
34 *     [$query->field('foo')->column() === $bar->declarations()[0]->columns[0], $query->toString()] // => [true, 'SELECT foo FROM bar']
35 * @example Building a new root from explicit structure
36 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite);
37 *     $query = $semantics->analyze('SELECT foo FROM bar WHERE foo > 1');
38 *     $other = new \SqlSemantics\Statement\Operation($query->context, $semantics->analyze('SELECT baz FROM bar')->statement);
39 *     $other->toString() // => 'SELECT baz FROM bar'
40 */
41final class Operation
42{
43    use Snapshot;
44
45    /**
46     * @var Facts Every fact derived for the statement against the context
47     */
48    public readonly Facts $facts;
49
50    private readonly string $sql;

52    /**
53     * Derives and checks a new root from a context and an explicit statement structure.
54     *
55     * A structure outside the closed value domain, a node shared between two
56     * positions, or a context of another profile is refused as an invalid
57     * construction. A missing rule or a failed correspondence check is a defect
58     * of the library and discards the candidate.
59     *
60     * @param AnalysisContext $context The fixed profile and declaration snapshot
61     * @param Statement $statement The requested statement
62     */
63    public function __construct(public readonly AnalysisContext $context, public readonly Statement $statement)
64    {
65        [$facts, $sql] = (new Publication())->establish($context, $statement);
66        $this->facts = $facts;
67        $this->sql = $sql;
68    }
69
70    /**
71     * Answers the SQL rendered from the structure and confirmed to correspond to it; never the analyzed text.
72     */
73    public function toString(): string
74    {
75        return $this->sql;
76    }
77
78    /**
79     * Answers the fixed language profile the operation was built under.
80     */
81    public function profile(): LanguageProfile
82    {
83        return $this->context->profile;
84    }
85
86    /**
87     * Answers the relation declarations the statement provides to a context.
88     *
89     * @return list<Table>
90     */
91    public function declarations(): array
92    {
93        return $this->facts->declarations;
94    }
95
96    /**
97     * Answers the shape of the rows the statement returns, or null when it returns none.
98     */
99    public function shape(): ?RowShape
100    {
101        return $this->facts->output?->shape;
102    }
103
104    /**
105     * Answers the complete ordered output fields, or null when the statement returns no rows or its shape is open.
106     */
107    public function fields(): ?Fields
108    {
109        return $this->facts->output?->fields();
110    }
111
112    /**
113     * Answers the one field at a position or with a name.
114     *
115     * This is a convenience for the unique case; use lookupField() to tell
116     * absent, ambiguous and dependent lookups apart.
117     *
118     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When no field or more than one field answers the key
119     */
120    public function field(string|int $key): Field
121    {
122        if (is_int($key)) {
123            $fields = $this->fields();
124            Check::input($fields !== null, 'The statement has no complete field list.');
125
126            return $fields->at($key);
127        }
128        $lookup = $this->lookupField($key);
129        Check::input($lookup instanceof UniqueField, 'The name does not denote exactly one output field.');
130
131        return $lookup->field;
132    }
133
134    /**
135     * Looks an output field up by name, distinguishing unique, absent, ambiguous and dependent results.
136     *
137     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When the statement returns no rows
138     */
139    public function lookupField(string $name): FieldLookup
140    {
141        Check::input($this->facts->output !== null, 'The statement returns no rows.');
142
143        return $this->facts->output->lookup($name);
144    }
145
146    /**
147     * Answers the input relation structure of a selection, or null when there is none.
148     */
149    public function inputRelation(): ?Relation
150    {
151        return $this->statement instanceof Selection ? $this->statement->input() : null;
152    }
153
154    /**
155     * Answers the input when it is exactly one plainly named relation.
156     *
157     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When the input is a join, a derived relation, several relations, or absent
158     */
159    public function singleNamedInput(): NamedRelation
160    {
161        $input = $this->inputRelation();
162        Check::input($input instanceof NamedRelation, 'The statement does not read from exactly one named relation.');
163
164        return $input;
165    }
166}
