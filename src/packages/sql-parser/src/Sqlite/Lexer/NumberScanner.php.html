10/**
11 * Reads SQLite's numeric literals.
12 *
13 * A number with underscores between its digits is `QNUMBER`; otherwise a
14 * hexadecimal or plain integer is `INTEGER` and anything with a fraction or
15 * exponent is `FLOAT`. An identifier character right after a number makes
16 * the whole run illegal.
17 *
18 * @visibility root
19 */
20final class NumberScanner
21{
22    /**
23     * Reads the number at the cursor, if one starts there.
24     *
25     * @param Scan $scan The tokenization in progress
26     *
27     * @return Lexeme|null The lexeme, or null when no number starts here
28     *
29     * @throws LexicalException When an identifier character follows the number
30     */
31    public function scan(Scan $scan): ?Lexeme
32    {
33        $cursor = $scan->cursor;
34        $start = $cursor->offset();
35        $text = $cursor->match('0[xX][0-9A-Fa-f](?:_?[0-9A-Fa-f])*');
36        if ($text !== null) {
37            $name = str_contains($text, '_') ? 'QNUMBER' : 'INTEGER';
38        } else {
39            $digits = '[0-9](?:_?[0-9])*';
40            $text = $cursor->match("(?:{$digits}(?:\\.(?:{$digits})?)?|\\.{$digits})(?:[eE][+-]?{$digits})?");
41            if ($text === null) {
42                return null;
43            }
44            $name = str_contains($text, '_') ? 'QNUMBER' : (preg_match('/[.eE]/', $text) === 1 ? 'FLOAT' : 'INTEGER');
45        }
46        if (Scan::isIdentifierByte($cursor->peek())) {
47            throw LexicalException::unexpectedCharacter($cursor->source, $start);
48        }
49
50        return $scan->lexeme($name, $start);
51    }
52}
