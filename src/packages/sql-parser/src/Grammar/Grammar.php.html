7/**
8 * A context-free grammar with the precedence declarations that disambiguate it.
9 *
10 * The grammar is augmented: rule zero derives the start symbol followed by
11 * the end marker. Token classes, fallbacks and the wildcard are the lexical
12 * hints a Lemon grammar carries; a Bison grammar declares none of them.
13 *
14 * @visibility root
15 */
16final class Grammar
17{
18    /**
19     * @var array<int, list<int>>
20     */
21    private readonly array $rulesByLhs;
22
23    /**
24     * @param SymbolTable $symbols Every symbol, numbered
25     * @param list<Rule> $rules Every rule, the augmented start rule first
26     * @param array<int, Precedence> $precedences Declared precedence by terminal
27     * @param PrecedencePolicy $policy Which terminal lends an unranked rule its precedence
28     * @param int|null $expectedConflicts Shift/reduce conflicts the grammar declares as expected
29     * @param array<int, list<int>> $tokenClasses Member terminals by token-class terminal
30     * @param array<int, int> $fallbacks Terminal to retry with by the terminal that failed
31     * @param int|null $wildcard Terminal that stands for any other, if any
32     *
33     * @throws GrammarException When the rules are not numbered in order
34     */
35    public function __construct(
36        public readonly SymbolTable $symbols,
37        public readonly array $rules,
38        public readonly array $precedences = [],
39        public readonly PrecedencePolicy $policy = PrecedencePolicy::LastTerminal,
40        public readonly ?int $expectedConflicts = null,
41        public readonly array $tokenClasses = [],
42        public readonly array $fallbacks = [],
43        public readonly ?int $wildcard = null,
44    ) {
45        $rulesByLhs = [];
46        foreach ($rules as $index => $rule) {
47            if ($rule->index !== $index) {
48                throw new GrammarException("Rule {$rule->index} is stored at position {$index}");
49            }
50            $rulesByLhs[$rule->lhs][] = $index;
51        }
52        $this->rulesByLhs = $rulesByLhs;
53    }
54
55    /**
56     * Answers the rules that derive one nonterminal.
57     *
58     * @param int $nonterminal Nonterminal number
59     *
60     * @return list<int> Rule indexes in declaration order
61     */
62    public function rulesOf(int $nonterminal): array
63    {
64        return $this->rulesByLhs[$nonterminal] ?? [];
65    }
66
67    /**
68     * Answers the precedence declared for a terminal, if any.
69     *
70     * @param int $terminal Terminal number
71     *
72     * @return Precedence|null The declaration, or null when the terminal is unranked
73     */
74    public function precedenceOf(int $terminal): ?Precedence
75    {
76        return $this->precedences[$terminal] ?? null;
77    }
78
79    /**
80     * Answers the precedence a rule carries into conflict resolution.
81     *
82     * A rule that names a terminal takes that terminal's declaration. Otherwise
83     * the policy picks a terminal from the right-hand side, and a token class in
84     * that position lends the declaration of its first ranked member.
85     *
86     * @param Rule $rule Rule to rank
87     *
88     * @return Precedence|null The declaration, or null when nothing ranks the rule
89     */
90    public function rulePrecedence(Rule $rule): ?Precedence
91    {
92        if ($rule->precedenceSymbol !== null) {
93            return $this->precedenceOf($rule->precedenceSymbol);
94        }
95        $chosen = null;
96        foreach ($rule->rhs as $symbol) {
97            if (!$this->symbols->isTerminal($symbol)) {
98                continue;
99            }
100            $precedence = $this->memberPrecedence($symbol);
101            if ($this->policy === PrecedencePolicy::LastTerminal) {
102                $chosen = $precedence;
103            } elseif ($precedence !== null) {
104                return $precedence;
105            }
106        }
107
108        return $chosen;
109    }
110
111    /**
112     * Answers the precedence of a terminal, looking through a token class.
113     *
114     * @param int $terminal Terminal or token-class number
115     *
116     * @return Precedence|null The first declaration found, or null when none is ranked
117     */
118    public function memberPrecedence(int $terminal): ?Precedence
119    {
120        foreach ($this->tokenClasses[$terminal] ?? [$terminal] as $member) {
121            $precedence = $this->precedenceOf($member);
122            if ($precedence !== null) {
123                return $precedence;
124            }
125        }
126
127        return null;
128    }
129
130    /**
131     * Answers the nonterminal the augmented start rule derives from.
132     *
133     * @return int Number of the grammar's own start symbol
134     */
135    public function startSymbol(): int
136    {
137        return $this->rules[0]->rhs[0];
138    }
139}
