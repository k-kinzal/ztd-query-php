11/**
12 * Prepares generated SQL against SQLite and reports unexpected rejections.
13 *
14 * SQLite validates syntax when a statement is prepared, so a statement that
15 * survives PDO::prepare() is syntactically accepted. SQLite reports its
16 * failures as message text rather than as distinct error codes, so the
17 * tolerated cases — name lookups a schema-less fuzz run cannot satisfy, plus a
18 * handful of documented restrictions — are matched on the message. Any other
19 * rejection means the grammar emitted something SQLite cannot parse, which is a
20 * finding and surfaces as an Error for php-fuzzer to record.
21 */
22final class SqliteSyntaxCheck
23{
24    /**
25     * @param PDO $pdo Connection to the in-memory SQLite database under test
26     */
27    public function __construct(private readonly PDO $pdo)
28    {
29    }
30
31    /**
32     * Verifies that SQLite parses the generated statement.
33     *
34     * @param string $sql Statement produced by the grammar
35     * @param int $seed Seed that produced the statement, so a finding can be replayed
36     *
37     * @throws Error When SQLite rejects the statement for a reason the grammar should not produce
38     */
39    public function verify(string $sql, int $seed): void
40    {
41        if ($sql === '') {
42            return;
43        }
44
45        try {
46            $statement = $this->pdo->prepare($sql);
47            if ($statement === false) {
48                throw new Error(
49                    "PDO::prepare returned false\n" .
50                    "Seed: $seed\n" .
51                    "SQL: $sql"
52                );
53            }
54        } catch (PDOException $rejection) {
55            // A schema-less fuzz run cannot satisfy name lookups, so those messages are expected.
56            $message = $rejection->getMessage();
57
58            $acceptable = match (true) {
59                str_contains($message, 'General error: 1 no such table:') => true,
60                str_contains($message, 'General error: 1 incomplete input') => true,
61                str_contains($message, 'General error: 1 unknown database') => true,
62                str_contains($message, 'General error: 1 no such view:') => true,
63                str_contains($message, 'temporary trigger may not have qualified name') => true,
64                str_contains($message, 'ORDER BY may not be used with non-aggregate') => true,
65                str_contains($message, 'General error: 1 no such index:') => true,
66                str_contains($message, 'General error: 1 no tables specified') => true,
67                str_contains($message, 'General error: 1 no such column:') => true,
68                str_contains($message, 'all VALUES must have the same number of terms') => true,
69                str_contains($message, 'General error: 1 no such function:') => true,
70                str_contains($message, 'SELECTs to the left and right of UNION do not have the same number of result columns') => true,
71                str_contains($message, 'General error: 1 no such trigger:') => true,
72                str_contains($message, 'unable to identify the object to be reindexed') => true,
73                str_contains($message, 'RAISE() may only be used within a trigger-program') => true,
74                str_contains($message, 'General error: 1 row value misused') => true,
75                str_contains($message, 'General error: 1 no such collation sequence:') => true,
76                str_contains($message, 'DISTINCT is not supported for window functions') => true,
77                str_contains($message, 'wrong number of arguments to function GLOB()') => true,
78                str_contains($message, 'duplicate WITH table name:') => true,
79                default => false,
80            };
81
82            if ($acceptable) {
83                return;
84            }
85
86            throw new Error(
87                "Unexpected error in generated SQL\n" .
88                "Seed: $seed\n" .
89                "SQL: $sql\n" .
90                "Error: $message",
91                0,
92                $rejection
93            );
94        }
95    }
