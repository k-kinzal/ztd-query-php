7/**
8 * Joins SQL tokens into a string with correct spacing.
9 *
10 * Shared across all SQL generators. Handles common spacing rules
11 * (parentheses, dots, brackets, commas) while supporting dialect-specific
12 * no-space pairs.
13 *
14 * @visibility root
15 */
16final class TokenJoiner
17{
18    /**
19     * @param list<string> $tokens SQL tokens to join
20     * @param list<list<string>> $noSpacePairs Additional [prev, token] pairs that need no space.
21     *                                          Use '*' as wildcard for either position.
22     * @param (callable(): string)|null $separator Generates trivia for boundaries that require separation.
23     * @param (callable(): string)|null $optionalTrivia Generates optional trivia at the outer and compact boundaries.
24     */
25    public static function join(
26        array $tokens,
27        array $noSpacePairs = [],
28        ?callable $separator = null,
29        ?callable $optionalTrivia = null,
30    ): string {
31        if ($tokens === []) {
32            return '';
33        }
34
35        $out = '';
36        $prev = null;
37        $count = count($tokens);
38
39        if ($optionalTrivia !== null) {
40            $out .= $optionalTrivia();
41        }
42
43        for ($i = 0; $i < $count; $i++) {
44            $token = $tokens[$i];
45
46            if ($prev === null) {
47                $out .= $token;
48                $prev = $token;
49                continue;
50            }
51
52            $needsSpace = self::needsSpace($prev, $token, $noSpacePairs);
53
54            if ($needsSpace) {
55                $out .= $separator !== null ? $separator() : ' ';
56            } elseif ($optionalTrivia !== null) {
57                $out .= $optionalTrivia();
58            }
59
60            $out .= $token;
61            $prev = $token;
62        }
63
64        if ($optionalTrivia !== null) {
65            $out .= $optionalTrivia();
66        }
67
68        return $out;
69    }

71    /**
72     * Reports whether two adjacent tokens are named by a pair that forbids a space.
73     *
74     * A pair may name a wildcard on either side, so a dialect can say "never
75     * after an opening parenthesis" without listing what may follow it.
76     *
77     * @param list<list<string>> $noSpacePairs Pairs of tokens that must be written together
78     * @param string|null $prev Token already written, or null at the start
79     * @param string $token Token about to be written
80     *
81     * @return bool True when nothing may be written between them
82     */
83    public static function matchesNoSpacePair(array $noSpacePairs, ?string $prev, string $token): bool
84    {
85        foreach ($noSpacePairs as $pair) {
86            if (($pair[0] === '*' || $pair[0] === $prev)
87                && ($pair[1] === '*' || $pair[1] === $token)) {
88                return true;
89            }
90        }
91
92        return false;
93    }
94
95    /**
96     * Check if a token looks like an SQL identifier (word or quoted).
97     */
98    public static function isIdentifier(string $token): bool
99    {
100        return preg_match('/^[A-Za-z_][A-Za-z0-9_]*$/', $token) === 1
101            || self::isQuotedIdentifier($token);
102    }
103
104    /**
105     * Reports whether a token is an identifier written inside quotes.
106     *
107     * A quoted identifier carries its own boundaries, so it never needs a space
108     * between itself and what comes next.
109     *
110     * @param string $token Token to judge
111     *
112     * @return bool True when the token opens and closes with the same quote
113     */
114    public static function isQuotedIdentifier(string $token): bool
115    {
116        $len = strlen($token);
117        if ($len < 2) {
118            return false;
119        }
120        $first = $token[0];
121
122        return ($first === '"' || $first === '`') && $token[$len - 1] === $first;
123    }
124
125    /**
126     * Decides whether a token boundary requires separating trivia.
127     *
128     * @param list<list<string>> $noSpacePairs Dialect-specific compact boundaries
129     */
130    public static function needsSpace(string $prev, string $token, array $noSpacePairs): bool
131    {
132        $needsSpace = true;
133
134        if ($token === '(' && self::isIdentifier($prev)) {
135            $needsSpace = false;
136        } elseif ($token === ')' || $prev === '(' || $token === ',' || $token === ';') {
137            $needsSpace = false;
138        } elseif ($prev === '.' || $token === '.') {
139            $needsSpace = false;
140        } elseif ($prev === '[' || $token === ']') {
141            $needsSpace = false;
142        } elseif (self::matchesNoSpacePair($noSpacePairs, $prev, $token)) {
143            $needsSpace = false;
144        }
145
146        return $needsSpace;
147    }
