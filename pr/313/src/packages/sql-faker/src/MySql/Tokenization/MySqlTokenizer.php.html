36    /**
37     * Reads SQL text into the tokens MySQL's own lexer would produce.
38     *
39     * @param string $sql Text to read
40     *
41     * @return list<string> Parser token names, in order
42     *
43     * @throws LexicalException When the text holds something the lexer cannot read
44     */
45    public function tokenize(string $sql): array
46    {
47        $tokens = [];
48        $length = strlen($sql);
49        $offset = 0;
50
51        while ($offset < $length) {
52            $startedAt = $offset;
53            if ($this->skipTrivia($sql, $offset)) {
54                if ($offset <= $startedAt) {
55                    throw LexicalException::noProgress('MySQL', $startedAt, $sql);
56                }
57
58                continue;
59            }
60
61            $token = $this->tokenAt($sql, $offset, $tokens);
62            if ($token === null) {
63                throw LexicalException::unsupportedInput('MySQL', $offset, $sql);
64            }
65
66            if ($offset <= $startedAt) {
67                throw LexicalException::noProgress('MySQL', $startedAt, $sql);
68            }
69
70            $tokens[] = $token;
71        }
72
73        return $this->merged($tokens);
74    }

77     * Reads the one token that starts at the offset, advancing past it.
78     *
79     * @param string $sql Text being read
80     * @param int $offset Where to read from; moved past what was consumed
81     * @param list<string> $emitted Tokens read so far, for the rules that depend on them
82     *
83     * @return string|null The token name, or null when nothing here is a token
84     *
85     * @throws LexicalException When a quoted or dollar-quoted run never closes
86     */
87    public function tokenAt(string $sql, int &$offset, array $emitted): ?string
88    {
89        return $this->quotedTokenAt($sql, $offset)
90            ?? $this->markerTokenAt($sql, $offset, $emitted)
91            ?? $this->numericTokenAt($sql, $offset)
92            ?? $this->wordTokenAt($sql, $offset)
93            ?? $this->operatorTokenAt($sql, $offset);

96    /**
97     * Reads a backtick, single-quoted or dollar-quoted run.
98     *
99     * @param string $sql Text being read
100     * @param int $offset Where to read from; moved past what was consumed
101     *
102     * @return string|null The token name, or null when no quote opens here
103     *
104     * @throws LexicalException When the run never closes
105     */
106    public function quotedTokenAt(string $sql, int &$offset): ?string
107    {
108        $character = $sql[$offset];
109
110        if ($character === '`') {
111            $this->skipQuoted($sql, $offset, '`');
112
113            return 'IDENT_QUOTED';
114        }
115
116        if ($character === "'") {
117            $this->skipQuoted($sql, $offset, "'");
118
119            return 'TEXT_STRING';
120        }
121
122        if ($character !== '$' || !$this->dollarQuotedStrings || substr($sql, $offset, 2) !== '$$') {
123            return null;
124        }
125
126        $end = strpos($sql, '$$', $offset + 2);
127        if ($end === false) {
128            throw LexicalException::unterminatedDollarQuotedString('MySQL');
129        }
130        $offset = $end + 2;
131
132        return 'DOLLAR_QUOTED_STRING_SYM';
133    }

135    /**
136     * Reads a parameter marker, a user-variable sigil, or the hostname after one.
137     *
138     * A hostname is only a hostname because an `@` came before it; the same
139     * characters anywhere else are an ordinary identifier, which is why the
140     * tokens read so far decide this.
141     *
142     * @param string $sql Text being read
143     * @param int $offset Where to read from; moved past what was consumed
144     * @param list<string> $emitted Tokens read so far
145     *
146     * @return string|null The token name, or null when none of these start here
147     */
148    public function markerTokenAt(string $sql, int &$offset, array $emitted): ?string
149    {
150        if ($sql[$offset] === '?') {
151            ++$offset;
152
153            return 'PARAM_MARKER';
154        }
155
156        if ($sql[$offset] === '@') {
157            ++$offset;
158
159            return '@';
160        }
161
162        if (($emitted[count($emitted) - 1] ?? null) !== '@'
163            || preg_match('/\G[A-Za-z0-9_.%-]+/A', $sql, $match, 0, $offset) !== 1
164        ) {
165            return null;
166        }
167
168        $offset += strlen($match[0]);
169
170        return 'LEX_HOSTNAME';
171    }

173    /**
174     * Reads a numeric literal in any of the spellings MySQL accepts.
175     *
176     * @param string $sql Text being read
177     * @param int $offset Where to read from; moved past what was consumed
178     *
179     * @return string|null The token name, or null when no number starts here
180     *
181     * @throws LexicalException When a prefixed quoted literal never closes
182     */
183    public function numericTokenAt(string $sql, int &$offset): ?string
184    {
185        if (preg_match('/\G(?:0[xX][0-9A-Fa-f]+|0[bB][01]+)/A', $sql, $match, 0, $offset) === 1) {
186            $offset += strlen($match[0]);
187
188            return strtolower(substr($match[0], 0, 2)) === '0x' ? 'HEX_NUM' : 'BIN_NUM';
189        }
190
191        if (preg_match('/\G(?:[nN]|[xX]|[bB])\'/A', $sql, $match, 0, $offset) === 1) {
192            $prefix = strtoupper($match[0][0]);
193            ++$offset;
194            $this->skipQuoted($sql, $offset, "'");
195
196            return match ($prefix) {
197                'N' => 'NCHAR_STRING',
198                'X' => 'HEX_NUM',
199                default => 'BIN_NUM',
200            };
201        }
202
203        if (preg_match('/\G(?:\d+\.\d*|\.\d+)(?:[eE][+-]?\d+)?|\G\d+[eE][+-]?\d+/A', $sql, $match, 0, $offset) === 1) {
204            $offset += strlen($match[0]);
205
206            return str_contains(strtolower($match[0]), 'e') ? 'FLOAT_NUM' : 'DECIMAL_NUM';
207        }
208
209        if (preg_match('/\G\d+/A', $sql, $match, 0, $offset) !== 1) {
210            return null;
211        }
212
213        $offset += strlen($match[0]);
214
215        return $this->integerToken($match[0]);
216    }

218    /**
219     * Reads a keyword, function name, charset introducer or identifier.
220     *
221     * @param string $sql Text being read
222     * @param int $offset Where to read from; moved past what was consumed
223     *
224     * @return string|null The token name, or null when no word starts here
225     */
226    public function wordTokenAt(string $sql, int &$offset): ?string
227    {
228        if (preg_match('/\G_[A-Za-z0-9_]*/A', $sql, $match, 0, $offset) === 1) {
229            $offset += strlen($match[0]);
230
231            return in_array(strtolower($match[0]), ['_utf8mb4', '_utf8mb3', '_latin1', '_ascii', '_binary'], true) ? 'UNDERSCORE_CHARSET' : 'IDENT';
232        }
233
234        if (preg_match('/\G[A-Za-z][A-Za-z0-9_$]*/A', $sql, $match, 0, $offset) !== 1) {
235            return null;
236        }
237
238        $offset += strlen($match[0]);
239        $word = strtoupper($match[0]);
240        $asFunction = ($sql[$offset] ?? null) === '(' ? ($this->functionTokens[$word] ?? null) : null;
241
242        return $asFunction ?? $this->symbolTokens[$word] ?? 'IDENT';
243    }

247     *
248     * @param string $sql Text being read

254    {

256        if ($operator === null) {

258        }

266    /**
267     * Consumes whitespace or one comment.
268     *
269     * @param string $sql Text being read
270     * @param int $offset Where to read from; moved past what was consumed
271     *
272     * @return bool True when something was skipped
273     *
274     * @throws LexicalException When a block comment never closes
275     */
276    public function skipTrivia(string $sql, int &$offset): bool
277    {
278        if (preg_match('/\G\s+/A', $sql, $match, 0, $offset) === 1) {
279            $offset += strlen($match[0]);
280
281            return true;
282        }
283
284        if (substr($sql, $offset, 2) === '/*') {
285            $end = strpos($sql, '*/', $offset + 2);
286            if ($end === false) {
287                throw LexicalException::unterminatedBlockComment('MySQL');
288            }
289            $offset = $end + 2;
290
291            return true;
292        }
293
294        $opensLineComment = $sql[$offset] === '#'
295            || (substr($sql, $offset, 2) === '--' && preg_match('/\s/', $sql[$offset + 2] ?? '') === 1);
296        if (!$opensLineComment) {
297            return false;
298        }
299
300        $end = strpos($sql, "\n", $offset + 1);
301        $offset = $end === false ? strlen($sql) : $end + 1;
302
303        return true;
304    }

306    /**
307     * Consumes a quoted run, doubling being the way a quote escapes itself.
308     *
309     * @param string $sql Text being read
310     * @param int $offset Offset of the opening quote; moved past the closing one
311     * @param string $quote The quote character
312     *
313     * @throws LexicalException When the run never closes
314     */
315    public function skipQuoted(string $sql, int &$offset, string $quote): void
316    {
317        $length = strlen($sql);
318        ++$offset;
319
320        while ($offset < $length) {
321            if ($sql[$offset] !== $quote) {
322                ++$offset;
323                continue;
324            }
325            if (($sql[$offset + 1] ?? null) === $quote) {
326                $offset += 2;
327                continue;
328            }
329            ++$offset;
330
331            return;
332        }
333
334        throw LexicalException::unterminatedQuotedToken('MySQL', $sql);
335    }

337    /**
338     * Reports which integer token a run of digits widens to.
339     *
340     * MySQL picks the narrowest of three tokens that can hold the value, so the
341     * boundaries are compared as text: the digits may be wider than an int.
342     *
343     * @param string $integer The digits as written
344     *
345     * @return string One of NUM, LONG_NUM or ULONGLONG_NUM
346     */
347    public function integerToken(string $integer): string
348    {
349        $normalized = ltrim($integer, '0');
350        $normalized = $normalized === '' ? '0' : $normalized;
351
352        foreach ([['2147483647', 'NUM'], ['9223372036854775807', 'LONG_NUM'], ['18446744073709551615', 'ULONGLONG_NUM']] as [$maximum, $token]) {
353            if (strlen($normalized) < strlen($maximum) || (strlen($normalized) === strlen($maximum) && strcmp($normalized, $maximum) <= 0)) {
354                return $token;
355            }
356        }
357
358        return 'DECIMAL_NUM';
359    }

362     * Reports the operator written at the offset, longest first.
363     *
364     * @param string $sql Text being read
365     * @param int $offset Where to look
366     *
367     * @return string|null The operator as written, or null when there is none
368     */
369    public function operatorAt(string $sql, int $offset): ?string
370    {
371        foreach (self::OPERATORS as $operator) {
372            if (substr($sql, $offset, strlen($operator)) === $operator) {
373                return $operator;
374            }
375        }
376
377        return $this->isPunctuation($sql[$offset]) ? $sql[$offset] : null;
378    }

392    /**
393     * Joins the token pairs MySQL's parser sees as one.
394     *
395     * @param list<string> $tokens Tokens as read
396     *
397     * @return list<string> Tokens with `WITH ROLLUP` collapsed
398     */
399    public function merged(array $tokens): array
400    {
401        $merged = [];
402        for ($index = 0; isset($tokens[$index]); $index++) {
403            $token = $tokens[$index];
404            if ($token === 'WITH' && ($tokens[$index + 1] ?? null) === 'ROLLUP_SYM') {
405                $merged[] = 'WITH_ROLLUP_SYM';
406                $index++;
407                continue;
408            }
409            $merged[] = $token;
410        }
411
412        return $merged;
413    }
