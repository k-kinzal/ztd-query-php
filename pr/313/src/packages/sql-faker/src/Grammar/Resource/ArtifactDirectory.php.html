9/**
10 * Answers which directory a generated artifact is written into.
11 *
12 * A build names the file it wants to produce, not the directory that has to
13 * exist first, and a checkout may not carry an empty directory at all. Asking
14 * the filesystem to create the directory is therefore part of naming it, and a
15 * path that can never become a writable directory is a failure the build has
16 * to hear about before it has staged anything.
17 *
18 * @visibility parent
19 */
20final class ArtifactDirectory
21{
22    /**
23     * Names the directory an artifact is written into, creating it when the filesystem allows.
24     *
25     * The nearest ancestor that already exists decides the answer: if it is not
26     * a writable directory then nothing below it can become one, and reporting
27     * that ancestor says why rather than only that the attempt failed.
28     *
29     * @param string $artifactPath Path of the file that is about to be written
30     *
31     * @return string Directory that now exists and can be written to
32     *
33     * @throws RuntimeException When the path cannot become a writable directory
34     */
35    public function prepared(string $artifactPath): string
36    {
37        $directory = dirname($artifactPath);
38
39        $existing = $directory;
40        while (!file_exists($existing) && dirname($existing) !== $existing) {
41            $existing = dirname($existing);
42        }
43        if (!is_dir($existing) || !is_writable($existing)) {
44            throw new RuntimeException("Failed to create {$directory}: {$existing} is not a writable directory");
45        }
46        if (!is_dir($directory) && !mkdir($directory, 0755, true) && !is_dir($directory)) {
47            throw new RuntimeException("Failed to create {$directory}");
48        }
49
50        return $directory;
51    }
52}
