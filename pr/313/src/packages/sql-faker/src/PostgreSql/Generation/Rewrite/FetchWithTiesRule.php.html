12 * Completes FETCH WITH TIES ordering in gram.y/insertSelectOptions and PLpgSQL_Expr.
13 * @see https://github.com/postgres/postgres/blob/REL_17_2/src/backend/parser/gram.y
14 * @see https://github.com/postgres/postgres/blob/REL_17_2/src/backend/parser/gram.y#L16398-L16427
15 */
16final class FetchWithTiesRule implements RewriteRule
17{
18    /**
19     * Supplies an absent ORDER BY and removes the incompatible SKIP LOCKED wait policy.
20     */
21    #[Override]
22    public function rewrite(TerminalSequence $sequence): TerminalSequence
23    {
24        foreach ($sequence->occurrences('limit_clause') as $id) {
25            $range = $sequence->range($id);
26            if ($range === null || $sequence->nameAt($range[1] - 1) !== 'TIES') {
27                continue;
28            }
29            $query = $sequence->terminals[$range[0]]->ancestor('select_no_parens')
30                ?? $sequence->terminals[$range[0]]->ancestor('PLpgSQL_Expr');
31            if ($query === null) {
32                continue;
33            }
34            $sequence = $this->ordered($sequence, $query, $range[0]);
35            foreach ($sequence->occurrences('opt_nowait_or_skip') as $wait) {
36                $waitRange = $sequence->range($wait);
37                if ($waitRange !== null && $sequence->nameAt($waitRange[0]) === 'SKIP'
38                    && $sequence->terminals[$waitRange[0]]->ancestor('select_no_parens') === $query) {
39                    $sequence = $sequence->replace($waitRange[0], $waitRange[1] - $waitRange[0], [], 'gram.y:WITH-TIES-SKIP-LOCKED');
40                }
41            }
42        }
43        return $sequence;
44    }
45
46    /**
47     * Adds a positional ordering only when the same query has no sort clause.
48     */
49    public function ordered(TerminalSequence $sequence, int $query, int $offset): TerminalSequence
50    {
51        foreach ($sequence->terminals as $terminal) {
52            if (($terminal->within('sort_clause') || $terminal->within('opt_sort_clause'))
53                && ($terminal->ancestor('select_no_parens') ?? $terminal->ancestor('PLpgSQL_Expr')) === $query) {
54                return $sequence;
55            }
56        }
57        foreach (['for_locking_clause', 'select_limit', 'opt_select_limit'] as $following) {
58            $clause = $sequence->child($query, $following);
59            $range = $clause === null ? null : $sequence->range($clause->id);
60            if ($range !== null) {
61                $offset = min($offset, $range[0]);
62            }
63        }
64        $sort = $sequence->child($query, 'opt_sort_clause');
65        $scope = $sort->id ?? $query;
66        $source = 'gram.y:WITH-TIES-ORDER-BY';
67        return $sequence->replace($offset, 0, [
68            $sequence->insertedFor('ORDER', $scope, $source),
69            $sequence->insertedFor('BY', $scope, $source, 1),
70            $sequence->insertedFor('ICONST', $scope, $source, 2),
71        ], $source);
72    }
