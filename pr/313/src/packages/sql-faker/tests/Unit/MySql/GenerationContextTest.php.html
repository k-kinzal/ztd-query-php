148    public function testBindsTheRequestedReleaseAndPreservesTheGrammarStart(): void
149    {
150        $grammar = new Grammar('entry', ['entry' => new ProductionRule('entry', [new Production([new Terminal('SELECT_SYM')])])]);
151        $context = new GenerationContext($grammar, Factory::create(), 'mysql-8.4.7');
152        self::assertSame('mysql-8.4.7', $context->lexicalGrammar->version());
153        self::assertSame('entry', $context->grammar->startSymbol);
154        self::assertSame($grammar->ruleMap['entry'], $context->grammar->ruleMap['entry']);
155    }

157    public function testUsesTheGrammarStartWhenThePlanDoesNotSpecifyOne(): void
158    {
159        $grammar = new Grammar('entry', [
160            'entry' => new ProductionRule('entry', [new Production([new Terminal('SELECT_SYM')])]),
161            'fragment' => new ProductionRule('fragment', [new Production([])]),
162        ]);
163        $faker = Factory::create();
164        $context = new GenerationContext($grammar, $faker, 'mysql-8.4.7');
165        $generator = new SqlGenerator($context->grammar, $faker, $context->lexicalGrammar, $context->rewriter, $context->startSymbol);
166        self::assertSame('SELECT', $generator->generate(GenerationPlan::all()));
167        self::assertSame('', $generator->generate(GenerationPlan::fromRule('fragment')));
168        self::assertSame('SELECT', $generator->generate(GenerationPlan::all()));
169    }

171    public function testLeavesThePlanInControlOfNullableProductionChoices(): void
172    {
173        $grammar = new Grammar('entry', ['entry' => new ProductionRule('entry', [
174            new Production([]), new Production([new Terminal('SELECT_SYM')]),
175        ])]);
176        $faker = Factory::create();
177        $context = new GenerationContext($grammar, $faker, 'mysql-8.4.7');
178        $generator = new SqlGenerator($context->grammar, $faker, $context->lexicalGrammar, $context->rewriter, $context->startSymbol);
179        self::assertSame('SELECT', $generator->generate(GenerationPlan::all()->requiringNonEmpty()->withMaxDepth(1)));
180        self::assertSame('', $generator->generate(GenerationPlan::constrained('entry', ['entry' => [ProductionPattern::exactly()]])));
181    }

183    public function testReportsMissingLexicalImplementationWhenTheTerminalIsActuallyGenerated(): void
184    {
