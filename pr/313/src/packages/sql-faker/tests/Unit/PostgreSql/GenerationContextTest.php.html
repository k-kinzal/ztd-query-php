137    public function testBindsTheRequestedReleaseAndPreservesTheGrammarStart(): void
138    {
139        $grammar = new Grammar('entry', ['entry' => new ProductionRule('entry', [new Production([new Terminal('SELECT')])])]);
140        $context = new GenerationContext($grammar, Factory::create(), 'pg-17.2');
141        self::assertSame('pg-17.2', $context->lexicalGrammar->version());
142        self::assertSame('entry', $context->grammar->startSymbol);
143        self::assertSame($grammar->ruleMap['entry'], $context->grammar->ruleMap['entry']);
144    }

146    public function testUsesTheGrammarStartWhenThePlanDoesNotSpecifyOne(): void
147    {
148        $grammar = new Grammar('entry', [
149            'entry' => new ProductionRule('entry', [new Production([new Terminal('SELECT')])]),
150            'fragment' => new ProductionRule('fragment', [new Production([])]),
151        ]);
152        $faker = Factory::create();
153        $context = new GenerationContext($grammar, $faker, 'pg-17.2');
154        $generator = new SqlGenerator($context->grammar, $faker, $context->lexicalGrammar, $context->rewriter, $context->startSymbol);
155        self::assertSame('SELECT', $generator->generate(GenerationPlan::all()));
156        self::assertSame('', $generator->generate(GenerationPlan::fromRule('fragment')));
157        self::assertSame('SELECT', $generator->generate(GenerationPlan::all()));
158    }
159
160    public function testLeavesThePlanInControlOfNullableProductionChoices(): void
161    {
162        $grammar = new Grammar('entry', ['entry' => new ProductionRule('entry', [
163            new Production([]), new Production([new Terminal('SELECT')]),
164        ])]);
165        $faker = Factory::create();
166        $context = new GenerationContext($grammar, $faker, 'pg-17.2');
167        $generator = new SqlGenerator($context->grammar, $faker, $context->lexicalGrammar, $context->rewriter, $context->startSymbol);
168        self::assertSame('SELECT', $generator->generate(GenerationPlan::all()->requiringNonEmpty()->withMaxDepth(1)));
169        self::assertSame('', $generator->generate(GenerationPlan::constrained('entry', ['entry' => [ProductionPattern::exactly()]])));
170    }
