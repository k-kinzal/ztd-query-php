148    /**
149     * Verifies that MySQL parses the generated statement.
150     *
151     * @param string $sql Statement produced by the grammar
152     * @param string $input Fuzzer input that produced the statement, so a finding can be replayed
153     *
154     * @throws Error When MySQL rejects the statement for a reason the grammar should not produce
155     */
156    public function verify(string $sql, string $input): void
157    {
158        if ($sql === '') {
159            throw new Error("Statement generation returned an empty string\nInput (hex): " . bin2hex($input));
160        }
161        try {
162            $quoted = $this->pdo->quote($sql);
163            if ($quoted === false) {
164                fwrite(STDERR, "Cannot quote generated SQL for server PREPARE.\n");
165                exit(2);
166            }
167            $this->pdo->exec('SET @sql_faker_input = ' . $quoted);
168            $this->pdo->exec('PREPARE sql_faker_check FROM @sql_faker_input');
169            $this->pdo->exec('DEALLOCATE PREPARE sql_faker_check');
170        } catch (PDOException $rejection) {
171            $info = $rejection->errorInfo ?? [];
172            $code = $info[1] ?? 0;
173            $code = is_int($code) ? $code : 0;
174            $detail = $info[2] ?? '';
175            $detail = is_string($detail) ? $detail : '';
176            if (in_array($code, self::CONNECTION_ERRORS, true)) {
177                fwrite(STDERR, "MySQL connection failed: {$rejection->getMessage()}\n");
178                exit(2);
179            }
180            if (isset(self::IGNORED_CODES[$code])) {
181                return;
182            }
183            foreach (self::IGNORED_MESSAGES as [$ignoredCode, $pattern, $reason]) {
184                if ($code === $ignoredCode && preg_match($pattern, $detail) === 1) {
185                    return;
186                }
187            }
188            throw new Error(
189                "Unexpected error in generated SQL\n" .
190                "Grammar: {$this->grammarVersion}\n" .
191                'Input (hex): ' . bin2hex($input) . "\n" .
192                "SQL: $sql\n" .
193                'SQLSTATE: ' . (is_scalar($info[0] ?? null) ? (string) $info[0] : 'unknown') . "\n" .
194                "Error Code: $code\n" .
195                "Error: {$rejection->getMessage()}",
196                0,
197                $rejection
198            );
199        }
200    }
