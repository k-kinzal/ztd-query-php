333    /**
334     * Process an already-executed statement based on the rewrite plan.
335     *
336     * This method handles post-execution logic like shadow application for write queries.
337     * Use this when you need to control statement preparation and execution externally.
338     *
339     * @param RewritePlan $plan The rewrite plan from rewrite().
340     * @param StatementInterface $statement The already-executed statement.
341     * @return ExecuteResult The execution result.
342     *
343     * @throws DatabaseException When the shadow refuses the statement
344     */
345    public function processExecutedStatement(RewritePlan $plan, StatementInterface $statement): ExecuteResult
346    {
347        if ($plan->kind() === QueryKind::READ) {
348            return GenericExecuteResult::fromStatement($statement, QueryKind::READ);
349        }
350
351        $resultSet = $this->resultSelectRunner->readResultSet($statement, $this->resultColumnTypeResolver);
352        $rows = $resultSet->rows;
353
354        $mutation = $plan->requireMutation();
355        $impact = $this->applyShadow($mutation, $resultSet, $plan->sql());
356        $returningProjection = $plan->returningProjection();
357        $resultRows = $returningProjection !== null
358            ? $returningProjection->project($impact->returningRows())
359            : $rows;
360
361        return GenericExecuteResult::fromBufferedRows(
362            $resultRows,
363            QueryKind::WRITE_SIMULATED,
364            $impact->affectedRowCount($plan->affectedRowsMode()),
365            $returningProjection !== null,
366        );
367    }

396    /**
397     * Execute an exec-style statement with ZTD rewriting and shadow application.
398     *
399     * @param string $sql The original SQL statement.
400     * @return int|false The number of affected rows, or false on failure.
401     * @throws DatabaseException When config is Exception mode and rewrite fails.
402     */
403    public function execStatement(string $sql): int|false
404    {
405        $plan = $this->rewrite($sql);
406
407        if ($plan->kind() === QueryKind::SKIPPED) {
408            return 0;
409        }
410
411        if ($plan->kind() === QueryKind::READ) {
412            $stmt = $this->connection->query($plan->sql());
413            if ($stmt === false) {
414                return false;
415            }
416            return $stmt->rowCount();
417        }
418
419        $mutation = $plan->requireMutation();
420
421        $resultSet = $this->resultSelectRunner->runResultSet(
422            $plan->sql(),
423            fn (string $s) => $this->connection->query($s),
424            $this->resultColumnTypeResolver,
425        );
426        $impact = $this->applyShadow($mutation, $resultSet, $sql);
427
428        return $impact->affectedRowCount($plan->affectedRowsMode());
