144    public function testCreateRegistersReflectedViews(): void
145    {
146        $empty = self::createStub(StatementInterface::class);
147        $empty->method('fetchAll')->willReturn([]);
148        $views = self::createStub(StatementInterface::class);
149        $views->method('fetchAll')->willReturn([['name' => 'active_users']]);
150        $create = self::createStub(StatementInterface::class);
151        $create->method('fetchAll')->willReturn([
152            ['Create View' => 'CREATE VIEW active_users AS SELECT 1 AS id'],
153        ]);
154        $connection = self::createStub(ConnectionInterface::class);
155        $connection->method('query')->willReturnCallback(
156            static fn (string $sql): StatementInterface => match ($sql) {
157                'SHOW TABLES' => $empty,
158                "SHOW FULL TABLES WHERE Table_type = 'VIEW'" => $views,
159                'SHOW CREATE VIEW `active_users`' => $create,
160                default => $empty,
161            },
162        );
163
164        $session = (new MySqlSessionFactory())->create($connection, ZtdConfig::default());
165
166        self::assertSame(
167            "WITH `active_users` AS (SELECT 1 AS id)\nSELECT * FROM active_users",
168            $session->rewrite('SELECT * FROM active_users')->sql(),
169        );
170    }
171
172    public function testCreateReturnsSession(): void

185    }

187    public function testCreateAppliesReflectedAnsiQuotesModeToProductionLexing(): void
188    {
189        $empty = self::createStub(StatementInterface::class);
190        $empty->method('fetchAll')->willReturn([]);
191        $sqlMode = self::createStub(StatementInterface::class);
192        $sqlMode->method('fetchAll')->willReturn([['ztd_sql_mode' => 'STRICT_TRANS_TABLES,ANSI_QUOTES']]);
193        $connection = self::createStub(ConnectionInterface::class);
194        $connection->method('query')->willReturnCallback(
195            static fn (string $sql): StatementInterface => $sql === 'SELECT @@SESSION.sql_mode AS ztd_sql_mode'
196                ? $sqlMode
197                : $empty,
198        );
199        $previousMode = Context::getMode();
200
201        try {
202            (new MySqlSessionFactory())->create($connection, ZtdConfig::default());
203            $tokens = SqlTokenStream::tokenize('"column"', MySqlLexerProfile::create())->significantTokens();
204
205            self::assertSame(SqlTokenKind::QuotedIdentifier, $tokens[0]->kind);
206        } finally {
207            Context::setMode($previousMode);
208        }
209    }
