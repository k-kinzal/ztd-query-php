172        $columns->method('fetchAll')->willReturn([['column_name' => 'id', 'data_type' => 'integer', 'character_maximum_length' => null, 'numeric_precision' => 32, 'numeric_scale' => 0, 'is_nullable' => 'NO', 'column_default' => null, 'udt_name' => 'int4'], ['column_name' => 'log_date', 'data_type' => 'date', 'character_maximum_length' => null, 'numeric_precision' => null, 'numeric_scale' => null, 'is_nullable' => 'NO', 'column_default' => null, 'udt_name' => 'date']]);
173        $primaryKey = self::createStub(StatementInterface::class);
174        $primaryKey->method('fetchAll')->willReturn([['column_name' => 'id'], ['column_name' => 'log_date']]);
175        $empty = self::createStub(StatementInterface::class);
176        $empty->method('fetchAll')->willReturn([]);
177        $partitionKeys = self::createStub(StatementInterface::class);
178        $partitionKeys->method('fetchAll')->willReturn([['table_name' => 'logs', 'partition_key' => 'RANGE (log_date)']]);
179        $relations = self::createStub(StatementInterface::class);
180        $relations->method('fetchAll')->willReturn([['child_table' => 'logs_2024', 'parent_table' => 'logs', 'predicate' => "log_date >= '2024-01-01'::date AND log_date < '2025-01-01'::date"]]);
181        $connection = self::createStub(ConnectionInterface::class);
182        $connection->method('query')->willReturnCallback(static function (string $sql) use ($tables, $columns, $primaryKey, $empty, $partitionKeys, $relations) {
183            return match (true) {
184                str_contains($sql, 'information_schema.tables') => $tables,
185                str_contains($sql, 'information_schema.columns') => $columns,
186                str_contains($sql, "constraint_type = 'PRIMARY KEY'") => $primaryKey,
187                str_contains($sql, 'SELECT c.relname AS table_name') => $partitionKeys,
188                str_contains($sql, 'SELECT child.relname AS child_table') => $relations,
189                default => $empty,
190            };
191        });
192        $session = (new PgSqlSessionFactory())->create($connection, ZtdConfig::default());
193        $sql = $session->rewrite('SELECT * FROM logs_2024')->sql();
194        self::assertStringContainsString('"logs" AS MATERIALIZED', $sql);
195        self::assertStringContainsString('"logs_2024" AS MATERIALIZED (SELECT * FROM "logs" WHERE log_date >=', $sql);
196        $create = $session->rewrite("CREATE TABLE logs_2025 PARTITION OF logs FOR VALUES FROM ('2025-01-01') TO ('2026-01-01')");
