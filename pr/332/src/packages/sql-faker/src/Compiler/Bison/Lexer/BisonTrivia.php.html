7use SqlFaker\Compiler\GrammarParseException;

9/**
10 * The text between two lexemes that carries no token of its own.
11 *
12 * Whitespace and comments may appear anywhere a lexeme may, so treating them as
13 * lexemes would mean every scanner returning "nothing this time" and the lexer
14 * looping until one of them returned something. Skipping them first keeps every
15 * scanner's contract honest: it is asked to read a lexeme, and it always
16 * produces one.
17 *
18 * Inside a semantic action the same comment forms appear but a bare slash is
19 * division rather than a mistake, so recognising a comment and requiring one
20 * are offered separately.
21 *
22 * @visibility root
23 */
24final class BisonTrivia
25{
26    /**
27     * Skips whitespace and comments until a lexeme starts or the source ends.
28     *
29     * @param SourceCursor $cursor Cursor to advance
30     *
31     * @throws GrammarParseException When a slash opens neither comment form
32     */
33    public function skipFrom(SourceCursor $cursor): void
34    {
35        while (true) {
36            $cursor->skipWhitespace();
37
38            if ($this->skipCommentAt($cursor)) {
39                continue;
40            }
41
42            if ($cursor->current() === '/') {
43                throw GrammarParseException::danglingSlash($cursor->offset());
44            }
45
46            return;
47        }
48    }
49
50    /**
51     * Skips one comment when the cursor sits on the start of one.
52     *
53     * @param SourceCursor $cursor Cursor to advance
54     *
55     * @return bool True when a comment was consumed, false when the cursor did not move
56     */
57    public function skipCommentAt(SourceCursor $cursor): bool
58    {
59        if ($cursor->current() !== '/') {
60            return false;
61        }
62
63        return match ($cursor->peek()) {
64            '/' => $this->skipToEndOfLine($cursor),
65            '*' => $this->skipToCommentClose($cursor),
66            default => false,
67        };
68    }
69
70    /**
71     * Consumes a line comment, stopping before the newline that ends it.
72     *
73     * The newline is left behind because it is whitespace, and whitespace is
74     * skipped by the caller that owns the loop.
75     *
76     * @param SourceCursor $cursor Cursor positioned on the opening slash
77     *
78     * @return bool Always true
79     */
80    public function skipToEndOfLine(SourceCursor $cursor): bool
81    {
82        $cursor->advance(2);
83        $cursor->takeWhile(static fn (string $character): bool => $character !== "\n");
84
85        return true;
86    }
87
88    /**
89     * Consumes a block comment together with its terminator.
90     *
91     * A comment left open by a truncated grammar ends at the end of the file.
92     *
93     * @param SourceCursor $cursor Cursor positioned on the opening slash
94     *
95     * @return bool Always true
96     */
97    public function skipToCommentClose(SourceCursor $cursor): bool
98    {
99        $cursor->advance(2);
100        $cursor->takeUntil('*/');
101
102        return true;
103    }
