14/**
15 * Reads `%token`, which names the terminals the scanner may hand the parser.
16 *
17 * Each terminal is a name that may be followed by an explicit token code and by
18 * a quoted alias, in that order and both optional, so the entries have to be
19 * read one at a time rather than collected as a flat list of names.
20 *
21 * @visibility root
22 */
23final class TokenDirectiveReader implements BisonDirectiveReader
24{
25    /** @readonly */
26    private BisonDeclarationBoundary $boundary;
27
28    /**
29     * @param BisonDeclarationBoundary|null $boundary Decides where the declaration ends
30     */
31    public function __construct(?BisonDeclarationBoundary $boundary = null)
32    {
33        $this->boundary = $boundary ?? new BisonDeclarationBoundary();
34    }
35
36    /**
37     * Reports whether the directive is `%token`.
38     *
39     * @param string $directive Directive name including its percent sign
40     *
41     * @return bool True for "%token"
42     */
43    #[Override]
44    public function handles(string $directive): bool
45    {
46        return $directive === '%token';
47    }
48
49    /**
50     * Consumes the optional type tag and every terminal that follows it.
51     *
52     * @param BisonTokenStream $stream Stream positioned just after the directive name
53     * @param string $directive Directive name including its percent sign
54     *
55     * @return BisonDeclaration The declaration, with no terminals when none followed
56     */
57    #[Override]
58    public function read(BisonTokenStream $stream, string $directive): BisonDeclaration
59    {
60        unset($directive);
61
62        $typeTag = $stream->nextIf(BisonLexeme::TypeTag)?->asString();
63
64        /** @var list<BisonTokenDefinition> $declared */
65        $declared = [];
66        while ($this->boundary->continuesWith($stream->peek()->type)) {
67            if ($stream->peek()->type !== BisonLexeme::Identifier) {
68                $stream->next();
69                continue;
70            }
71
72            $declared[] = $this->readTerminal($stream);
73        }
74
75        return new BisonTokenDeclaration($typeTag, $declared);
76    }
77
78    /**
79     * Consumes one terminal together with the code and alias it may carry.
80     *
81     * @param BisonTokenStream $stream Stream positioned on the terminal's name
82     *
83     * @return BisonTokenDefinition The terminal, with null for whatever it omitted
84     */
85    public function readTerminal(BisonTokenStream $stream): BisonTokenDefinition
86    {
87        return new BisonTokenDefinition(
88            $stream->nextString(),
89            $stream->nextIf(BisonLexeme::Number)?->asInt(),
90            $stream->nextIf(BisonLexeme::StringLiteral)?->asString(),
91        );
92    }
93}
