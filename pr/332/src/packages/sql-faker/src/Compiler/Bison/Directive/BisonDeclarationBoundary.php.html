9/**
10 * Decides where one declaration stops and the next thing starts.
11 *
12 * Bison declarations are not terminated: `%token A B C` runs until something
13 * that can only begin something else appears. Every reader that consumes a list
14 * of symbols needs that judgement, and it has to be the same judgement in all
15 * of them, or two readers would disagree about which declaration a symbol
16 * belongs to.
17 *
18 * @visibility root
19 */
20final class BisonDeclarationBoundary
21{
22    /**
23     * Reports whether a lexeme still belongs to the declaration being read.
24     *
25     * @param BisonLexeme $lexeme Lexeme waiting at the front of the stream
26     *
27     * @return bool True while the declaration continues, false where it ends
28     */
29    public function continuesWith(BisonLexeme $lexeme): bool
30    {
31        return match ($lexeme) {
32            BisonLexeme::Directive,
33            BisonLexeme::Prologue,
34            BisonLexeme::PercentPercent,
35            BisonLexeme::Eof => false,
36            BisonLexeme::Identifier,
37            BisonLexeme::Number,
38            BisonLexeme::CharLiteral,
39            BisonLexeme::StringLiteral,
40            BisonLexeme::TypeTag,
41            BisonLexeme::Colon,
42            BisonLexeme::Semicolon,
43            BisonLexeme::Pipe,
44            BisonLexeme::Action => true,
45        };
46    }
47}
