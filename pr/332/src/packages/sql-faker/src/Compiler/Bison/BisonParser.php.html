12/**
13 * Reads a GNU Bison or Yacc grammar file into an AST.
14 *
15 * A grammar file has three sections in a fixed order — declarations, rules,
16 * epilogue — so parsing is a sequence of three reads rather than a decision.
17 * Each section knows its own end, which is what lets this class say what a
18 * grammar file is without also saying how any part of it is spelled.
19 *
20 * A file with no rules is refused. Unknown declarations are retained as
21 * source text so later passes can preserve constructs outside the AST model.
22 *
23 * @visibility root
24 */
25final class BisonParser
26{
27    /** @readonly */
28    private BisonPreambleReader $preamble;
29
30    /** @readonly */
31    private BisonRuleReader $rules;
32
33    /** @readonly */
34    private BisonStartSymbol $startSymbol;
35
36    /**
37     * @param BisonPreambleReader|null $preamble Reads the prologue and declarations
38     * @param BisonRuleReader|null $rules Reads the production rules
39     * @param BisonStartSymbol|null $startSymbol Decides which rule a derivation begins from
40     */
41    public function __construct(
42        ?BisonPreambleReader $preamble = null,
43        ?BisonRuleReader $rules = null,
44        ?BisonStartSymbol $startSymbol = null,
45    ) {
46        $this->preamble = $preamble ?? new BisonPreambleReader();
47        $this->rules = $rules ?? new BisonRuleReader();
48        $this->startSymbol = $startSymbol ?? new BisonStartSymbol();
49    }
50
51    /**
52     * Reads grammar source text into an AST.
53     *
54     * @param string $input Bison grammar text
55     *
56     * @return BisonAst The grammar the text describes
57     *
58     * @throws GrammarParseException When the text yields no production rules
59     */
60    public function parse(string $input): BisonAst
61    {
62        $stream = BisonTokenStream::over($input);
63
64        $preamble = $this->preamble->read($stream);
65        $rules = $this->rules->readAll($stream);
66        $epilogue = trim($stream->consumeRemaining());
67
68        if ($rules === []) {
69            throw GrammarParseException::noRulesParsed('Bison');
70        }
71
72        return new BisonAst(
73            $this->startSymbol->from($preamble->declarations, $rules),
74            $preamble->prologue,
75            $preamble->declarations,
76            $rules,
77            $epilogue !== '' ? $epilogue : null,
78        );
79    }
80
81    /**
82     * Reads a grammar file from disk.
83     *
84     * @param string $path Path to a Bison grammar file
85     *
86     * @return BisonAst The grammar the file describes
87     *
88     * @throws GrammarParseException When the file cannot be read, or yields no production rules
89     */
90    public function parseFile(string $path): BisonAst
91    {
92        if (!is_file($path)) {
93            throw GrammarParseException::unreadableSource($path);
94        }
95
96        $contents = file_get_contents($path);
97        if ($contents === false) {
98            throw GrammarParseException::unreadableSource($path);
99        }
100
101        return $this->parse($contents);
102    }
103}
