13/**
14 * Reads the four directives that rank operators against each other.
15 *
16 * `%left`, `%right`, `%nonassoc` and `%precedence` differ only in the
17 * associativity they declare, and the directive name carries that, so one
18 * reader covers all four. Operators may be written as names or as quoted
19 * characters, and both spellings name the same terminal.
20 *
21 * @visibility root
22 */
23final class PrecedenceDirectiveReader implements BisonDirectiveReader
24{
25    /** @readonly */
26    private BisonDeclarationBoundary $boundary;
27
28    /**
29     * @param BisonDeclarationBoundary|null $boundary Decides where the declaration ends
30     */
31    public function __construct(?BisonDeclarationBoundary $boundary = null)
32    {
33        $this->boundary = $boundary ?? new BisonDeclarationBoundary();
34    }
35
36    /**
37     * Reports whether the directive declares an associativity.
38     *
39     * @param string $directive Directive name including its percent sign
40     *
41     * @return bool True for "%left", "%right", "%nonassoc" and "%precedence"
42     */
43    #[Override]
44    public function handles(string $directive): bool
45    {
46        return match ($directive) {
47            '%left', '%right', '%nonassoc', '%precedence' => true,
48            default => false,
49        };
50    }
51
52    /**
53     * Consumes the optional type tag and the operators being ranked.
54     *
55     * @param BisonTokenStream $stream Stream positioned just after the directive name
56     * @param string $directive Directive name including its percent sign
57     *
58     * @return BisonDeclaration The declaration, with no operators when none followed
59     */
60    #[Override]
61    public function read(BisonTokenStream $stream, string $directive): BisonDeclaration
62    {
63        /** @var 'left'|'right'|'nonassoc'|'precedence' $associativity */
64        $associativity = substr($directive, 1);
65
66        $typeTag = $stream->nextIf(BisonLexeme::TypeTag)?->asString();
67
68        /** @var list<string> $symbols */
69        $symbols = [];
70        while ($this->boundary->continuesWith($stream->peek()->type)) {
71            $symbol = $stream->nextIf(BisonLexeme::Identifier, BisonLexeme::CharLiteral)?->asString();
72            if ($symbol === null) {
73                $stream->next();
74                continue;
75            }
76
77            $symbols[] = $symbol;
78        }
79
80        return new BisonPrecedenceDeclaration($associativity, $typeTag, $symbols);
81    }
82}
