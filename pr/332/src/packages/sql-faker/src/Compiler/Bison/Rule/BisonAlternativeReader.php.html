13/**
14 * Reads the alternatives on the right-hand side of one rule.
15 *
16 * Bison lets the final semicolon be left out, so an alternative list also ends
17 * where the next rule begins — an identifier followed by a colon — or at the
18 * end of the section. All three endings complete the alternative in hand, which
19 * is why the reader looks two tokens ahead before it consumes anything.
20 *
21 * @visibility root
22 */
23final class BisonAlternativeReader
24{
25    /**
26     * Reads every alternative of the current rule.
27     *
28     * @param BisonTokenStream $stream Stream positioned on the first symbol after the colon
29     *
30     * @return list<BisonAlternativeNode> The alternatives, always at least one
31     */
32    public function readAll(BisonTokenStream $stream): array
33    {
34        $draft = new BisonAlternativeDraft();
35        $alternatives = [];
36
37        while (true) {
38            if ($this->endsTheRule($stream)) {
39                $alternatives[] = $draft->complete();
40
41                return $alternatives;
42            }
43
44            $lexeme = $stream->peek()->type;
45            if ($lexeme === BisonLexeme::Semicolon) {
46                $stream->next();
47                $alternatives[] = $draft->complete();
48
49                return $alternatives;
50            }
51
52            if ($lexeme === BisonLexeme::Pipe) {
53                $stream->next();
54                $alternatives[] = $draft->complete();
55                continue;
56            }
57
58            $this->readPart($stream, $draft);
59        }
60    }
61
62    /**
63     * Reports whether the alternative list is over without consuming anything.
64     *
65     * @param BisonTokenStream $stream Stream to look ahead in
66     *
67     * @return bool True at the end of the section or at the start of the next rule
68     */
69    public function endsTheRule(BisonTokenStream $stream): bool
70    {
71        $lexeme = $stream->peek()->type;
72
73        if ($lexeme === BisonLexeme::Eof || $lexeme === BisonLexeme::PercentPercent) {
74            return true;
75        }
76
77        return $lexeme === BisonLexeme::Identifier && $stream->peekN(2)->type === BisonLexeme::Colon;
78    }
79
80    /**
81     * Reads one symbol, action or inline directive into the alternative.
82     *
83     * @param BisonTokenStream $stream Stream positioned on the part to read
84     * @param BisonAlternativeDraft $draft Alternative being assembled
85     */
86    public function readPart(BisonTokenStream $stream, BisonAlternativeDraft $draft): void
87    {
88        match ($stream->peek()->type) {
89            BisonLexeme::Action => $draft->setAction($stream->nextString()),
90            BisonLexeme::Identifier => $draft->addSymbol(
91                new BisonSymbolNode(BisonSymbolForm::Identifier, $stream->nextString()),
92            ),
93            BisonLexeme::CharLiteral => $draft->addSymbol(
94                new BisonSymbolNode(BisonSymbolForm::CharLiteral, $stream->nextString()),
95            ),
96            BisonLexeme::Directive => $this->readInlineDirective($stream, $draft),
97            BisonLexeme::Number,
98            BisonLexeme::StringLiteral,
99            BisonLexeme::TypeTag,
100            BisonLexeme::Colon,
101            BisonLexeme::Semicolon,
102            BisonLexeme::Pipe,
103            BisonLexeme::PercentPercent,
104            BisonLexeme::Prologue,
105            BisonLexeme::Eof => $stream->next(),
106        };
107    }
108
109    /**
110     * Applies `%prec`, `%dprec` or `%merge` to the alternative being read.
111     *
112     * @param BisonTokenStream $stream Stream positioned on the directive
113     * @param BisonAlternativeDraft $draft Alternative being assembled
114     */
115    public function readInlineDirective(BisonTokenStream $stream, BisonAlternativeDraft $draft): void
116    {
117        match ($stream->nextString()) {
118            '%prec' => $draft->setPrecedenceSymbol($this->readPrecedenceSymbol($stream)),
119            '%dprec' => $draft->setDynamicPrecedence($this->readDynamicPrecedence($stream)),
120            '%merge' => $draft->setMergeFunction($this->readMergeFunction($stream)),
121            default => null,
122        };
123    }
124
125    /**
126     * Reads the terminal whose precedence `%prec` borrows.
127     *
128     * @param BisonTokenStream $stream Stream positioned just after `%prec`
129     *
130     * @return string|null The terminal, or null when the directive named none
131     */
132    public function readPrecedenceSymbol(BisonTokenStream $stream): ?string
133    {
134        return $stream->nextIf(BisonLexeme::Identifier, BisonLexeme::CharLiteral)?->asString();
135    }
136
137    /**
138     * Reads the rank `%dprec` assigns to this alternative.
139     *
140     * @param BisonTokenStream $stream Stream positioned just after `%dprec`
141     *
142     * @return int|null The rank, or null when the directive named none
143     */
144    public function readDynamicPrecedence(BisonTokenStream $stream): ?int
145    {
146        return $stream->nextIf(BisonLexeme::Number)?->asInt();
147    }
148
149    /**
150     * Reads the merge function `%merge` names for this alternative.
151     *
152     * @param BisonTokenStream $stream Stream positioned just after `%merge`
153     *
154     * @return string|null The function name, or null when the directive named none
155     */
156    public function readMergeFunction(BisonTokenStream $stream): ?string
157    {
158        return $stream->nextIf(BisonLexeme::TypeTag)?->asString();
159    }
