10/**
11 * Adds lookahead to a Bison lexer.
12 *
13 * A recursive-descent parser has to decide what it is reading before it commits
14 * to reading it, which means looking at tokens it does not want to consume yet.
15 * Buffering those here keeps the lexer a plain forward scan: nothing in it has
16 * to know that a token may be read twice.
17 *
18 * @visibility root
19 */
20final class BisonTokenStream
21{
22    /** @readonly */
23    private BisonLexer $lexer;
24
25    /**
26     * Tokens already scanned but not yet consumed, in reading order.
27     *
28     * @var list<BisonToken>
29     */
30    private array $lookahead = [];

38    }

42     *
43     * @param string $source Bison grammar text
44     *
45     * @return self A stream reading that source with the default lexeme rules
46     */

52    /**
53     * Consumes the next token.
54     *
55     * @return BisonToken The token, taken from the lookahead buffer when one is waiting
56     *
57     * @throws GrammarParseException When the source cannot be tokenized
58     */
59    public function next(): BisonToken
60    {
61        $buffered = array_shift($this->lookahead);
62
63        return $buffered ?? $this->lexer->scan();
64    }

66    /**
67     * Reads the next token without consuming it.
68     *
69     * @return BisonToken The token that the next call to next() will return
70     *
71     * @throws GrammarParseException When the source cannot be tokenized
72     */
73    public function peek(): BisonToken
74    {
75        return $this->peekN(1);
76    }
77
78    /**
79     * Reads a token further ahead without consuming anything.
80     *
81     * @param int $distance How many tokens ahead to look, counting from one
82     *
83     * @return BisonToken The token at that distance
84     *
85     * @throws InvalidArgumentException When the distance is less than one
86     * @throws GrammarParseException When the source cannot be tokenized
87     */
88    public function peekN(int $distance): BisonToken
89    {
90        if ($distance < 1) {
91            throw new InvalidArgumentException('peekN($n) requires $n >= 1');
92        }
93
94        while (count($this->lookahead) < $distance) {
95            $this->lookahead[] = $this->lexer->scan();
96        }
97
98        return $this->lookahead[$distance - 1];
99    }
100
101    /**
102     * Consumes the next token only when it is one of the given kinds.
103     *
104     * Reading a grammar is mostly "take this if it is there": a type tag, an
105     * explicit token code, an alias. Asking the stream keeps that one step, so
106     * a reader cannot look at one token and then consume another.
107     *
108     * @param BisonLexeme ...$accepted Kinds the caller is willing to take
109     *
110     * @return BisonToken|null The consumed token, or null with the stream unmoved
111     *
112     * @throws GrammarParseException When the source cannot be tokenized
113     */
114    public function nextIf(BisonLexeme ...$accepted): ?BisonToken
115    {
116        if (!in_array($this->peek()->type, $accepted, true)) {
117            return null;
118        }
119
120        return $this->next();
121    }

126     * @return string The token value, with a numeric value rendered as digits

145    }
146
147    /**
148     * Consumes the rest of the source as raw text.
149     *
150     * Buffered lookahead is discarded rather than prepended: the cursor has
151     * already moved past those tokens, so the text they came from is behind the
152     * point the epilogue starts at.
153     *
154     * @return string Everything left after the tokens already scanned
155     */
156    public function consumeRemaining(): string
157    {
158        $this->lookahead = [];
159
160        return $this->lexer->consumeRemaining();
161    }
