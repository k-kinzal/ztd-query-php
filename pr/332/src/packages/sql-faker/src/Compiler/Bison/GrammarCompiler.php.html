20/**
21 * Compiles a Grammar from a BisonAst.
22 *
23 * Transforms the supplied Bison AST into productions and symbols without
24 * selecting a database dialect or interpreting its scanner actions.
25 *
26 * @visibility root
27 */
28final class GrammarCompiler

30    /**
31     * Compile a BisonAst into a Grammar.
32     *
33     * @throws UnknownSymbolException When a production names a symbol the grammar never declares
34     */
35    public function compile(BisonAst $ast): Grammar
36    {
37        /** @var array<string, BisonRuleNode> $ruleTable */
38        $ruleTable = [];
39        foreach ($ast->rules as $rule) {
40            $ruleTable[$rule->name] = $rule;
41        }
42
43        $declarationTable = $this->declarations($ast);
44
45        /** @var array<string, ProductionRule> $ruleMap */
46        $ruleMap = [];
47
48        foreach ($ast->rules as $ruleNode) {
49            /** @var list<Production> $productions */
50            $productions = [];
51
52            foreach ($ruleNode->alternatives as $altNode) {
53                /** @var list<Symbol> $symbols */
54                $symbols = [];
55
56                foreach ($altNode->symbols as $symNode) {
57                    if ($symNode->type === BisonSymbolForm::CharLiteral) {
58                        $symbols[] = new Terminal($symNode->value);
59                    } elseif (isset($ruleTable[$symNode->value])) {
60                        $symbols[] = new NonTerminal($symNode->value);
61                    } elseif (isset($declarationTable[$symNode->value])) {
62                        $symbols[] = new Terminal($symNode->value);
63                    } else {
64                        throw new UnknownSymbolException($symNode->value);
65                    }
66                }
67
68                $productions[] = new Production($symbols);
69            }
70
71            if (isset($ruleMap[$ruleNode->name])) {
72                $merged = array_merge(
73                    $ruleMap[$ruleNode->name]->alternatives,
74                    $productions
75                );
76                $ruleMap[$ruleNode->name] = new ProductionRule($ruleNode->name, $merged);
77            } else {
78                $ruleMap[$ruleNode->name] = new ProductionRule($ruleNode->name, $productions);
79            }
80        }
81
82        return new Grammar($ast->startSymbol, $ruleMap);
83    }

86     * Indexes declared terminal tokens by name before compiling productions.

89     */

91    {

93        $declarationTable = [];

99            }
100        }
