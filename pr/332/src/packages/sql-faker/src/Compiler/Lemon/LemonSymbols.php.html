7/**
8 * Remembers which names a Lemon grammar uses as tokens and which as rules.
9 *
10 * Lemon does not declare every symbol. It declares its tokens, and everything
11 * else is a rule if some rule is written for it, so what a name means is only
12 * known once the whole file has been read. A reader therefore says what it has
13 * learned as it goes, and the question "is this a terminal" is answered here
14 * rather than by whichever reader happens to be looking at the name.
15 *
16 * @visibility root
17 */
18final class LemonSymbols
19{
20    /** @var array<string, true> Names the grammar declares or uses as tokens */
21    private array $tokens = [];
22
23    /** @var array<string, true> Names the grammar writes a rule for or expands to */
24    private array $rules = [];
25
26    /**
27     * Records that a name is a token.
28     *
29     * @param string $name Token name as the grammar spells it
30     */
31    public function declareToken(string $name): void
32    {
33        $this->tokens[$name] = true;
34    }
35
36    /**
37     * Records that a name is a rule.
38     *
39     * @param string $name Rule name as the grammar spells it
40     */
41    public function declareRule(string $name): void
42    {
43        $this->rules[$name] = true;
44    }
45
46    /**
47     * Records every token named on one line of a directive.
48     *
49     * A directive lists its tokens separated by whitespace or, for a token
50     * class, by an alternation bar, and the line may end in the period Lemon
51     * uses to close a declaration. Anything on such a line that is not spelled
52     * like a token is not one.
53     *
54     * @param string $line Everything the directive named
55     * @param string $splitPattern Pattern separating one name from the next
56     */
57    public function declareTokensOn(string $line, string $splitPattern): void
58    {
59        $names = preg_split($splitPattern, trim($line));
60        if ($names === false) {
61            return;
62        }
63        foreach ($names as $name) {
64            $name = trim($name, '.');
65            if ($name !== '' && self::isTokenName($name)) {
66                $this->declareToken($name);
67            }
68        }
69    }
70
71    /**
72     * Reports whether a name stands for a token rather than a rule.
73     *
74     * What the grammar said outright wins. A name nothing said anything about
75     * falls back to Lemon's spelling convention.
76     *
77     * @param string $name Symbol name to judge
78     *
79     * @return bool True when the name stands for a token
80     */
81    public function isTerminal(string $name): bool
82    {
83        if (isset($this->tokens[$name])) {
84            return true;
85        }
86        if (isset($this->rules[$name])) {
87            return false;
88        }
89
90        return self::isTokenName($name);
91    }
92
93    /**
94     * Reports whether a name is spelled the way Lemon spells a token.
95     *
96     * @param string $name Symbol name to judge
97     *
98     * @return bool True when the name is written in capitals
99     */
100    public static function isTokenName(string $name): bool
101    {
102        return preg_match('/^[A-Z][A-Z0-9_]*$/', $name) === 1;
103    }
104}
