7/**
8 * Reads the rules a Lemon grammar writes.
9 *
10 * Lemon writes one alternative per line rather than grouping them, so the
11 * alternatives of a rule are gathered by name as the file is read. Within a
12 * line it allows an alternation between tokens in a single position, which
13 * stands for as many alternatives as the positions multiply out to, and it
14 * lets each symbol carry an alias the parser uses to name its value.
15 *
16 * @visibility root
17 */
18final class LemonRules
19{
20    /**
21     * @param LemonText $text Removes what is not grammar before the rules are looked for
22     */
23    public function __construct(private readonly LemonText $text = new LemonText())
24    {
25    }

27    /**
28     * Reads every rule, telling the symbol table what each name turned out to be.
29     *
30     * @param string $input Contents of the grammar file
31     * @param LemonSymbols $symbols Symbol table to record names in
32     *
33     * @return array<string, list<list<string>>> Rule name => its alternatives, each a list of symbol names
34     */
35    public function readFrom(string $input, LemonSymbols $symbols): array
36    {
37        /** @var array<string, list<list<string>>> $rules */
38        $rules = [];
39
40        $input = $this->text->withoutDirectiveBlocks($input);
41        $pattern = '/^(\w+)(?:\([^)]*\))?\s*::=\s*(.*?)\.\s*(?:\{[^}]*\})?/ms';
42        if (preg_match_all($pattern, $input, $matches, PREG_SET_ORDER) === false) {
43            return [];
44        }
45
46        foreach ($matches as $match) {
47            $lhs = $match[1];
48            $rhs = trim($match[2]);
49            if (str_starts_with($lhs, '%')) {
50                continue;
51            }
52
53            $symbols->declareRule($lhs);
54            if (!isset($rules[$lhs])) {
55                $rules[$lhs] = [];
56            }
57            if ($rhs === '') {
58                $rules[$lhs][] = [];
59                continue;
60            }
61            array_push($rules[$lhs], ...$this->alternatives($rhs, $symbols));
62        }
63
64        return $rules;
65    }

67    /**
68     * Multiplies out one right-hand side into the alternatives it stands for.
69     *
70     * A position written as `A|B` means either, so a right-hand side with two
71     * such positions stands for four alternatives. Positions Lemon marks with
72     * a leading percent sign configure the parser rather than name a symbol.
73     *
74     * @param string $rhs Right-hand side as the grammar writes it
75     * @param LemonSymbols $symbols Symbol table to record names in
76     *
77     * @return non-empty-list<list<string>> Every alternative the right-hand side stands for
78     */
79    public function alternatives(string $rhs, LemonSymbols $symbols): array
80    {
81        /** @var non-empty-list<list<string>> $alternatives */
82        $alternatives = [[]];
83
84        $parts = preg_split('/\s+/', $rhs);
85        if ($parts === false) {
86            return [[]];
87        }
88
89        foreach ($parts as $part) {
90            $part = trim($part);
91            if ($part === '') {
92                continue;
93            }
94
95            $names = array_values(array_filter(array_map(
96                $this->withoutAlias(...),
97                explode('|', $part),
98            ), static fn (string $name): bool => $name !== '' && !str_starts_with($name, '%')));
99            if ($names === []) {
100                continue;
101            }
102
103            foreach ($names as $name) {
104                if (LemonSymbols::isTokenName($name)) {
105                    $symbols->declareToken($name);
106                } else {
107                    $symbols->declareRule($name);
108                }
109            }
110
111            $expanded = [];
112            foreach ($alternatives as $alternative) {
113                foreach ($names as $name) {
114                    $expanded[] = [...$alternative, $name];
115                }
116            }
117            $alternatives = $expanded;
118        }
119
120        return $alternatives;
121    }
122
123    /**
124     * Drops the alias Lemon lets a symbol carry, as in `expr(A)`.
125     *
126     * @param string $symbol Symbol as the grammar writes it
127     *
128     * @return string The symbol's name alone
129     */
130    public function withoutAlias(string $symbol): string
131    {
132        $open = strpos($symbol, '(');
133
134        return $open !== false ? substr($symbol, 0, $open) : $symbol;
135    }
