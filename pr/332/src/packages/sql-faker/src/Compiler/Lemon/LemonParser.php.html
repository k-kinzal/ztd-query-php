17/**
18 * Reads Lemon source declarations and production rules into a grammar.
19 *
20 * Lemon writes one alternative per line — `lhs(alias) ::= symbol ... .` — so
21 * the alternatives of a rule are gathered by name rather than grouped in the
22 * file. It declares its tokens in directives, spells them in capitals, and
23 * carries the C its parser is built from alongside the rules. Reading the
24 * declarations, reading the rules, and remembering what each name turned out
25 * to be are three subjects, and this brings their answers together.
26 *
27 * @visibility root
28 */
29final class LemonParser
30{
31    /**
32     * @param LemonText $text Removes what is not grammar
33     * @param LemonDirectives $directives Reads the tokens the grammar declares
34     * @param LemonRules $rules Reads the rules the grammar writes
35     */
36    public function __construct(
37        private readonly LemonText $text = new LemonText(),
38        private readonly LemonDirectives $directives = new LemonDirectives(),
39        private readonly LemonRules $rules = new LemonRules(),
40    ) {
41    }
42
43    /**
44     * Reads a Lemon grammar as a grammar.
45     *
46     * The first rule written is the start symbol, which is what Lemon itself
47     * assumes when no other is named.
48     *
49     * @param string $input Contents of the grammar file
50     * @param list<string> $defines Named parser-build definitions
51     *
52     * @return Grammar Grammar the file describes
53     *
54     * @throws RuntimeException When source conditionals are malformed
55     * @throws GrammarParseException When the file writes no rules
56     * @throws InvalidArgumentException When a rule is filed under a name other than its own left-hand side
57     */
58    public function parse(string $input, array $defines = []): Grammar
59    {
60        $symbols = new LemonSymbols();
61        $input = (new LemonPreprocessor(new LemonCondition($defines)))->process($this->text->withoutComments($input));
62        $this->directives->declareInto($input, $symbols);
63        $rules = $this->rules->readFrom($input, $symbols);
64        $classes = $this->directives->tokenClasses($input);
65        if ($rules === []) {
66            throw GrammarParseException::noRulesParsed('Lemon');
67        }
68
69        /** @var string $startSymbol */
70        $startSymbol = array_key_first($rules);
71
72        /** @var array<string, ProductionRule> $ruleMap */
73        $ruleMap = [];
74        foreach ($rules as $lhs => $alternatives) {
75            /** @var list<Production> $productions */
76            $productions = [];
77            foreach ($alternatives as $symbolNames) {
78                /** @var list<Symbol> $written */
79                $written = [];
80                foreach ($symbolNames as $name) {
81                    $written[] = $symbols->isTerminal($name) ? new Terminal($name) : new NonTerminal($name);
82                }
83                $productions[] = new Production($written);
84            }
85            $ruleMap[$lhs] = new ProductionRule($lhs, $productions);
86        }
87
88        foreach ($classes as $name => $tokens) {
89            $ruleMap[$name] = new ProductionRule($name, array_map(
90                static fn (string $token): Production => new Production([new Terminal($token)]),
91                $tokens,
92            ));
93        }
94        return new Grammar($startSymbol, $ruleMap);
95    }
96
97    /**
98     * Reads a Lemon grammar file as a grammar.
99     *
100     * @param string $path Path of the grammar file
101     *
102     * @return Grammar Grammar the file describes
103     *
104     * @throws RuntimeException When the grammar file cannot be read
105     * @throws GrammarParseException When the file writes no rules
106     * @throws InvalidArgumentException When a rule is filed under a name other than its own left-hand side
107     */
108    public function parseFile(string $path): Grammar
109    {
110        if (!is_file($path) || !is_readable($path)) {
111            throw new RuntimeException("Failed to read: {$path}");
112        }
113        $contents = file_get_contents($path);
114        if ($contents === false) {
115            throw new RuntimeException("Failed to read: {$path}");
116        }
117
118        return $this->parse($contents);
119    }
