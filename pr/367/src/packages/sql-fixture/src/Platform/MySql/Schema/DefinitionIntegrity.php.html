14 * @visibility root

16final class DefinitionIntegrity

18    /**
19     * Stop if the parser abandoned part of the definition list.
20     *
21     * On a syntax error the parser gives up on the rest of the definitions,
22     * so a stray keyword takes every column after it. The result is a schema
23     * that looks complete and is not, and the damage only shows up much later
24     * as a column that generates no value.
25     *
26     * Not every error loses something. DEC and FIXED are valid synonyms the
27     * parser reports as unrecognised while still reading the column, so the
28     * test is what came out rather than whether anything was said.
29     * @throws \SqlFixture\Schema\Exception\InvalidSqlException
30     */
31    public function assertNothingWasLost(Parser $parser, CreateStatement $stmt, string $sql): void
32    {
33        if ($parser->errors === []) {
34            return;
35        }
36
37        $fields = $stmt->fields;
38        $declared = $this->countDeclaredDefinitions($parser);
39
40        if ($declared !== null && is_array($fields) && count($fields) >= $declared) {
41            return;
42        }
43
44        throw new \SqlFixture\Schema\Exception\InvalidSqlException($sql, $parser->errors[0]->getMessage());
45    }
46
47    /**
48     * How many definitions the parenthesised body separates, or null when it
49     * is never closed.
50     *
51     * Counting uses the parser's own tokens, which already know that a comma
52     * inside a string or an ENUM is not a separator.
53     */
54    public function countDeclaredDefinitions(Parser $parser): ?int
55    {
56        $list = $parser->list;
57        if ($list === null) {
58            return null;
59        }
60
61        $depth = 0;
62        $definitions = 1;
63
64        foreach ($list->tokens as $token) {
65            if ($token->type !== Token::TYPE_OPERATOR) {
66                continue;
67            }
68
69            if ($token->value === '(') {
70                $depth++;
71                continue;
72            }
73
74            if ($token->value === ')') {
75                $depth--;
76
77                if ($depth === 0) {
78                    return $definitions;
79                }
80
81                continue;
82            }
83
84            if ($depth === 1 && $token->value === ',') {
85                $definitions++;
86            }
87        }
88
89        return null;
90    }
