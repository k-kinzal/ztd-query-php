103    public function testRead(): void
104    {
105        $grammar = (new LemonGrammarReader())->read("%token_prefix TK_\n%token SEMI.\n%left PLUS MINUS.\n%fallback ID ABORT AFTER.\n%wildcard ANY.\n%token_class id ID|INDEXED.\n%start_symbol input\nexpr(A) ::= expr(B) PLUS|MINUS expr(C). [PLUS] { A = B + C; }\nexpr ::= NUM.\ninput ::= expr SEMI.\n");
106
107        self::assertSame('input', $grammar->symbols->name($grammar->startSymbol()));
108        self::assertSame(PrecedencePolicy::FirstRankedTerminal, $grammar->policy);
109        self::assertSame(['$accept', 'expr', 'expr', 'input'], array_map(static fn (Rule $rule): string => $grammar->symbols->name($rule->lhs), $grammar->rules));
110        self::assertSame(['expr', 'PLUS|MINUS', 'expr'], array_map(static fn (int $id): string => $grammar->symbols->name($id), $grammar->rules[1]->rhs));
111        self::assertSame('PLUS', $grammar->symbols->name($grammar->rules[1]->precedenceSymbol ?? -1));
112        self::assertSame('ANY', $grammar->symbols->name($grammar->wildcard ?? -1));
113        self::assertCount(2, $grammar->tokenClasses);
114        self::assertCount(2, $grammar->fallbacks);
115        self::assertSame(Associativity::Left, $grammar->precedenceOf($grammar->symbols->id('MINUS') ?? -1)?->associativity);
116    }
