132    #[DataProvider('providerSqliteExecution')]
133    public function testRenderPreservesSqliteResultsAndOperatorAssociation(string $sql): void
134    {
135        $database = new PDO('sqlite::memory:');
136        $database->exec('CREATE TABLE t(a INT,b INT,c INT); INSERT INTO t VALUES(1,2,3),(2,NULL,4),(5,3,2)');
137        $parser = new SqliteParser();
138        $renderer = new Renderer($parser);
139        $original = $database->query($sql);
140        $formatted = $database->query($renderer->render($parser->parse($sql)));
141        self::assertNotFalse($original);
142        self::assertNotFalse($formatted);
