9/**
10 * Consumes a symbol name.
11 *
12 * Rule names and token names share one spelling in Bison: which of the two a
13 * name refers to is decided by the parser from where it appears, not by the
14 * lexer. Digits and dots may follow the first character but may not open the
15 * name, which is what keeps a number from being read as an identifier.
16 *
17 * @visibility root
18 */
19final class IdentifierScanner implements BisonScanner
20{
21    /**
22     * Reports whether a symbol name can start here.
23     *
24     * @param string $character A single character; never the empty string
25     *
26     * @return bool True for a letter or an underscore
27     */
28    #[Override]
29    public function handles(string $character): bool
30    {
31        return preg_match('/[A-Za-z_]/', $character) === 1;
32    }
33
34    /**
35     * Consumes the name.
36     *
37     * @param SourceCursor $cursor Cursor positioned on the first character
38     *
39     * @return BisonToken The symbol name
40     */
41    #[Override]
42    public function scan(SourceCursor $cursor): BisonToken
43    {
44        $offset = $cursor->offset();
45        $name = $cursor->takeWhile(
46            static fn (string $character): bool => preg_match('/[A-Za-z0-9_.]/', $character) === 1,
47        );
48
49        return new BisonToken(BisonLexeme::Identifier, $name, $offset);
50    }
51}
