7/**
8 * Applies what MySQL's parser enforces outside its grammar.
9 *
10 * A Bison grammar only says which token sequences the parser will shift and
11 * reduce; the C actions attached to those rules reject a good deal more. A
12 * derivation that satisfied only the grammar would write SQL the parser
13 * refuses, so every such constraint is applied here before the terminals are
14 * handed to the lexer.
15 *
16 * @visibility root
17 */
18final class ParserSemantics
19{
20    /**
21     * Applies the constraints MySQL enforces in semantic actions rather than in its grammar.
22     *
23     * The grammar accepts a qualified name in front of a user variable, a
24     * CURRENT_USER() call where the parser wants the bare keyword, an ALTER
25     * EVENT with no clause after the name, and a handful of tokens whose
26     * spelling depends on what follows them. Each of those is rejected or
27     * rewritten by the C actions, so a derivation that satisfied only the
28     * grammar would write SQL MySQL refuses to parse.
29     *
30     * @param list<string> $terminals Terminals a derivation produced
31     *
32     * @return list<string> Terminals with every semantic constraint satisfied
33     */
34    public function applied(array $terminals): array
35    {
36        $terminals = $this->normalizeStatement($terminals);
37
38        $result = [];
39        foreach ($terminals as $index => $terminal) {
40            $previous = $result[count($result) - 1] ?? null;
41            if ($terminal === 'EQUAL_SYM'
42                && in_array($terminals[$index + 1] ?? null, ['ALL', 'ALL_SYM', 'ANY', 'ANY_SYM', 'SOME', 'SOME_SYM'], true)
43            ) {
44                $terminal = 'EQ';
45            }
46            if (in_array($terminal, ['RELEASE', 'RELEASE_SYM'], true)
47                && in_array($previous, ['CHAIN', 'CHAIN_SYM'], true)
48                && !in_array($result[count($result) - 2] ?? null, ['NO', 'NO_SYM'], true)
49            ) {
50                continue;
51            }
52            if (in_array($terminal, ['DECIMAL_NUM', 'FLOAT_NUM'], true)
53                && ($previous === ':' || in_array($previous, ['SYSTEM', 'SYSTEM_SYM'], true))
54            ) {
55                $terminal = 'NUM';
56            }
57            $result[] = $terminal;
58        }
59
60        return $result;
61    }

63    /**
64     * Normalizes user names and ensures ALTER EVENT has an action.
65     *
66     * @param list<string> $terminals
67     * @return list<string>
68     */
69    public function normalizeStatement(array $terminals): array
70    {
71        $remove = [];
72        foreach ($terminals as $index => $terminal) {
73            if ($terminal !== '@') {
74                continue;
75            }
76            for ($dot = $index - 2; $dot >= 1 && $terminals[$dot] === '.'; $dot -= 2) {
77                $remove[$dot] = true;
78                $remove[$dot - 1] = true;
79            }
80        }
81        if ($remove !== []) {
82            $terminals = array_values(array_diff_key($terminals, $remove));
83        }
84
85        foreach ($terminals as $index => $terminal) {
86            if (in_array($terminal, ['CURRENT_USER', 'CURRENT_USER_SYM'], true)
87                && ($terminals[$index + 1] ?? null) === '('
88                && ($terminals[$index + 2] ?? null) === ')'
89                && ($terminals[$index + 3] ?? null) === ':'
90            ) {
91                array_splice($terminals, $index + 1, 2);
92            }
93        }
94
95        $event = array_search('EVENT_SYM', $terminals, true);
96        $alter = array_search('ALTER_SYM', $terminals, true);
97        if ($event !== false && $alter !== false) {
98            $afterName = $event + 2;
99            if (($terminals[$afterName] ?? null) === '.' && isset($terminals[$afterName + 1])) {
100                $afterName += 2;
101            }
102            if ($afterName >= count($terminals)) {
103                $terminals[] = 'ENABLE_SYM';
104            }
105        }
106
107        return $terminals;
108    }
