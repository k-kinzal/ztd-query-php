12/**
13 * Writes one MySQL parser terminal as the SQL text that produces it.
14 *
15 * There are two ways to write a terminal and they are not interchangeable. The
16 * catalogued way replays SQL that the server's own lexer was observed to turn
17 * into the terminal, which is what makes generated statements trustworthy. The
18 * synthetic way constructs text from the terminal's name, which reaches
19 * terminals no catalog witnesses but only promises to be plausible — so it is
20 * off unless a caller asks for it.
21 *
22 * @phpstan-type Realization array{string|null, list<string>}
23 *
24 * @visibility root
25 */
26final class MySqlTerminalRealizer
27{

35     * @param array<string, list<string>> $symbols Spellings by keyword or operator terminal
36     * @param array<string, list<string>> $functions Spellings by function terminal
37     * @param string $profileVersion Lexical profile version in use
38     * @param bool $allowSyntheticTerminals Whether terminals may be written without a witness

54    /**
55     * Writes one terminal, reporting the tokens the text should read back as.
56     *
57     * @param string $terminal Terminal to write
58     * @param non-empty-string|null $requestedLexeme Exact text the caller wants, when it has one
59     *
60     * @return Realization The text, or null where the terminal writes nothing, and its tokens
61     *
62     * @throws LexicalException When the terminal or the requested text cannot be realized
63     */
64    public function realize(string $terminal, ?string $requestedLexeme = null): array
65    {
66        if (!$this->supports($terminal)) {
67            throw LexicalException::unsupportedTerminal('MySQL', $this->profileVersion, $terminal);
68        }
69
70        if ($requestedLexeme !== null) {
71            return $this->realizeRequested($terminal, $requestedLexeme);
72        }
73
74        if (!$this->allowSyntheticTerminals) {
75            return $this->realizeWitnessed($terminal);
76        }
77
78        if (str_starts_with($terminal, 'GRAMMAR_SELECTOR_') || $terminal === 'END_OF_INPUT') {
79            return [null, []];
80        }
81
82        return $this->realizeSynthetic($terminal);
83    }

90     * @return bool True when the catalog witnesses it, or synthetic writing is allowed

92    public function supports(string $terminal): bool
93    {
94        return $this->allowSyntheticTerminals || $this->catalog->supports($terminal);

104    public function realizeWitnessed(string $terminal): array
105    {
106        $witnesses = $this->catalog->witnesses($terminal);
107        $witness = $witnesses[$this->faker->numberBetween(0, count($witnesses) - 1)];

112    /**
113     * Writes a terminal from its name rather than from an observed example.
114     *
115     * @param string $terminal Terminal to write
116     *
117     * @return Realization The constructed text and its tokens
118     *
119     * @throws LexicalException When the constructed text cannot be read back
120     */
121    public function realizeSynthetic(string $terminal): array
122    {
123        return match ($terminal) {
124            'IDENT' => [$this->identifier(), ['IDENT']],
125            'IDENT_QUOTED' => [$this->quotedIdentifier(), ['IDENT_QUOTED']],
126            'TEXT_STRING' => [$this->stringLiteral(), ['TEXT_STRING']],
127            'NCHAR_STRING' => ['N' . $this->stringLiteral(), ['NCHAR_STRING']],
128            'DOLLAR_QUOTED_STRING_SYM' => [$this->dollarQuotedString(), [$terminal]],
129            'NUM' => [$this->strings->integerString(0, 2147483647), ['NUM']],
130            'LONG_NUM' => [(string) $this->faker->numberBetween(2147483648, PHP_INT_MAX), ['LONG_NUM']],
131            'ULONGLONG_NUM' => ['18446744073709551615', ['ULONGLONG_NUM']],
132            'DECIMAL_NUM' => [$this->strings->decimalString(), ['DECIMAL_NUM']],
133            'FLOAT_NUM' => [$this->strings->floatString($this->strings->integerString()), ['FLOAT_NUM']],
134            'HEX_NUM' => [$this->hexadecimalLiteral(), ['HEX_NUM']],
135            'BIN_NUM' => [$this->binaryLiteral(), ['BIN_NUM']],
136            'LEX_HOSTNAME' => ['localhost', ['IDENT']],
137            'PARAM_MARKER' => ['?', ['PARAM_MARKER']],
138            'OR2_SYM' => ['||', ['OR2_SYM']],
139            'WITH_ROLLUP_SYM' => ['WITH ROLLUP', ['WITH_ROLLUP_SYM']],
140            'UNDERSCORE_CHARSET' => ['_utf8mb4', ['UNDERSCORE_CHARSET']],
141            default => $this->realizeFixed($terminal),
142        };
143    }

145    /**
146     * Writes the exact text a caller asked for, once it is shown to realize the terminal.
147     *
148     * @param string $terminal Terminal the text is meant to realize
149     * @param non-empty-string $requestedLexeme Exact text the caller wants
150     *
151     * @return array{non-empty-string, list<string>} The text and its tokens
152     *
153     * @throws LexicalException When the text does not realize the terminal
154     */
155    public function realizeRequested(string $terminal, string $requestedLexeme): array
156    {
157        if ($this->allowSyntheticTerminals) {
158            $tokens = $this->tokenizer->tokenize($requestedLexeme);
159            if ($tokens !== [$terminal]) {
160                throw LexicalException::lexemeDoesNotRealizeTerminal('MySQL', $terminal, $requestedLexeme);
161            }
162
163            return [$requestedLexeme, $tokens];
164        }
165
166        foreach ($this->catalog->witnesses($terminal) as $witness) {
167            if ($witness['sql'] === $requestedLexeme) {
168                return [$requestedLexeme, $witness['tokens']];
169            }
170        }
171
172        throw LexicalException::noWitnessForLexeme('MySQL', $terminal, $requestedLexeme);
173    }

175    /**
176     * Writes a terminal that stands for a fixed keyword, operator or function name.
177     *
178     * @param string $terminal Terminal to write
179     *
180     * @return array{string, list<string>} The spelling and its tokens
181     *
182     * @throws LexicalException When the spelling cannot be read back
183     */
184    public function realizeFixed(string $terminal): array
185    {
186        if ($this->allowSyntheticTerminals) {
187            $lexeme = $this->syntheticSpelling($terminal);
188
189            return [$lexeme, $this->tokenizer->tokenize($lexeme)];
190        }
191
192        $spellings = $this->symbols[$terminal] ?? $this->functions[$terminal] ?? null;
193        $lexeme = $spellings !== null
194            ? $spellings[$this->faker->numberBetween(0, count($spellings) - 1)]
195            : $terminal;
196
197        return [
198            $lexeme,
199            isset($this->symbols[$terminal]) || isset($this->functions[$terminal])
200                ? [$terminal]
201                : $this->tokenizer->tokenize($lexeme),
202        ];
203    }

205    /**
206     * Guesses how a terminal is spelled from its name alone.
207     *
208     * Operator terminals are named rather than spelled, so those are listed;
209     * everything else drops the `_SYM` suffix MySQL's grammar uses to keep
210     * keyword terminals apart from the words they stand for.
211     *
212     * @param string $terminal Terminal to spell
213     *
214     * @return string The spelling
215     */
216    public function syntheticSpelling(string $terminal): string
217    {
218        return match ($terminal) {
219            'EQ' => '=',
220            'EQUAL_SYM' => '<=>',
221            'LT' => '<',
222            'GT_SYM' => '>',
223            'LE' => '<=',
224            'GE' => '>=',
225            'NE' => '<>',
226            'SHIFT_LEFT' => '<<',
227            'SHIFT_RIGHT' => '>>',
228            'AND_AND_SYM' => '&&',
229            'OR_OR_SYM', 'OR2_SYM' => '||',
230            'NOT2_SYM' => 'NOT',
231            'SET_VAR' => ':=',
232            'JSON_SEPARATOR_SYM' => '->',
233            'JSON_UNQUOTED_SEPARATOR_SYM' => '->>',
234            'NEG' => '-',
235            default => str_ends_with($terminal, '_SYM') ? substr($terminal, 0, -4) : $terminal,
236        };
237    }

273    {
