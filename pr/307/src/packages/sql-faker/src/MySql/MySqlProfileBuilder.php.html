14/**
15 * Builds a MySQL lexical profile from the server's own lexer source.
16 *
17 * The three files it reads are the keyword table, the scanner and the character
18 * state header, and where the last of those lives moved twice across the

20 * the hash of every file it was built from.

22 * @phpstan-type Witness array{id: string, sql: string, tokens: list<string>, units: list<string>, context_sql?: string}

38    /**
39     * Reports the upstream files a release keeps its lexer in.
40     *
41     * The character-state header moved twice across the supported releases, so
42     * where it lives is decided by the version rather than fixed.
43     *
44     * @param string $version Release tag, e.g. "mysql-8.4.7"
45     *
46     * @return array{table: string, scanner: string, state: string} URLs of the three files
47     */
48    public function sourceUrls(string $version): array
49    {
50        $base = "https://raw.githubusercontent.com/mysql/mysql-server/refs/tags/{$version}";
51        $statePath = match (true) {
52            str_starts_with($version, 'mysql-5.6.') => '/include/m_ctype.h',
53            version_compare(substr($version, strlen('mysql-')), '8.1.0', '<') => '/include/sql_chars.h',
54            default => '/strings/sql_chars.h',
55        };
56
57        return [
58            'table' => $base . '/sql/lex.h',
59            'scanner' => $base . '/sql/sql_lex.cc',
60            'state' => $base . $statePath,
61        ];
62    }

65     * @return array<string, mixed>
66     */
67    public function build(string $version, Grammar $grammar): array
68    {
69        ['table' => $tableUrl, 'scanner' => $scannerUrl, 'state' => $stateUrl] = $this->sourceUrls($version);
70        $table = $this->source->fetch($tableUrl);
71        $scanner = $this->source->fetch($scannerUrl);
72        $stateHeader = $this->source->fetch($stateUrl);
73        $compiled = (new MySqlCompiler())->compile($table);
74
75        $profile = [
76            'dialect' => 'mysql',
77            'version' => $version,
78            'sources' => [
79                $tableUrl => hash('sha256', $table),
80                $scannerUrl => hash('sha256', $scanner),
81                $stateUrl => hash('sha256', $stateHeader),
82            ],
83            'symbols' => $compiled['symbols'],
84            'functions' => $compiled['functions'],
85            'features' => [
86                'dollar_quoted_strings' => str_contains($scanner, 'DOLLAR_QUOTED_STRING_SYM'),
87            ],
88        ];
89
90        $profile['catalog'] = $this->catalog(
91            $version,
92            $profile,
93            (new MySqlSourceParser())->parseStates($stateHeader, $scanner),
94            $grammar,
95        );
96
97        return $profile;
98    }

108        string $version,
109        array $profile,
110        array $states,
111        Grammar $grammar,
112    ): array {
113        /** @var array<string, list<string>> $symbols */
114        $symbols = $profile['symbols'];
115        /** @var array<string, list<string>> $functions */
116        $functions = $profile['functions'];
117        /** @var array{dollar_quoted_strings: bool} $features */
118        $features = $profile['features'];
119        $terminals = $this->keywordWitnesses($symbols, $functions);
120        $samples = $this->lexicalSamples($states, $features['dollar_quoted_strings']);
121        $this->addLexicalWitnesses($terminals, $samples);
122
123        $versionedParserTokens = $this->addParserWitnesses($terminals, $version, $grammar);
124
125        $coverage = $this->coverage($terminals, $states, $versionedParserTokens);
126
127        $terminalExclusions = [
128            'NOT2_SYM' => 'Default sql_mode emits NOT_SYM; NOT2_SYM requires HIGH_NOT_PRECEDENCE.',
129            'OR_OR_SYM' => 'Default sql_mode normalizes double-pipe to OR2_SYM; OR_OR_SYM requires PIPES_AS_CONCAT.',
130            'UDF_RETURNS_SYM' => 'The token is declared by legacy grammars but has no mapping in the official lexer table.',
131        ];
132        ksort($terminals);
133
134        return [
135            'source' => [
136                'engine' => 'mysql',
137                'entrypoint' => 'my_sql_parser_lex',
138            ],
139            'terminals' => $terminals,
140            'terminal_exclusions' => $terminalExclusions,
141            'coverage' => $coverage,
142        ];

165    /**
166     * Builds witnesses for the keyword and function tables.
167     *
168     * @param array<string, list<string>> $symbols
169     * @param array<string, list<string>> $functions
170     * @return Terminals
171     */
172    public function keywordWitnesses(array $symbols, array $functions): array
173    {
174        $terminals = [];
175        foreach ($symbols as $terminal => $lexemes) {
176            if (in_array($terminal, ['NOT2_SYM', 'OR_OR_SYM'], true)) {
177                continue;
178            }
179            foreach ($lexemes as $index => $lexeme) {
180                $terminals[$terminal][] = $this->witness(
181                    "mysql.symbol.{$terminal}.{$index}",
182                    $lexeme,
183                    [$terminal],
184                    ['MY_LEX_START', 'MY_LEX_IDENT'],
185                );
186            }
187        }
188        foreach ($functions as $terminal => $lexemes) {
189            foreach ($lexemes as $index => $lexeme) {
190                $terminals[$terminal][] = $this->witness(
191                    "mysql.function.{$terminal}.{$index}",
192                    $lexeme,
193                    [$terminal],
194                    ['MY_LEX_START', 'MY_LEX_IDENT'],
195                    $lexeme . '(',
196                );
197            }
198        }
199
200        return $terminals;
201    }

204     * Adds witnesses for lexical states available in this server version.
205     *
206     * @param list<string> $states
207     * @return array<string, list<array{string, list<string>, list<string>, 3?: string}>>

209     * @throws RuntimeException When dollar quoting has no source state

211    public function lexicalSamples(array $states, bool $dollarQuotedStrings): array
212    {
213        $samples = (new MySqlLexicalSamples())->all();
214        $dollarState = current(array_values(array_filter(
215            $states,
216            static fn (string $state): bool => str_starts_with($state, 'MY_LEX_IDENT_OR_DOLLAR'),
217        )));
218        if ($dollarQuotedStrings) {
219            if (!is_string($dollarState)) {
220                throw new RuntimeException('MySQL dollar-quoted string state was not found.');
221            }
222            $samples['DOLLAR_QUOTED_STRING_SYM'] = [
223                ['$$text$$', ['DOLLAR_QUOTED_STRING_SYM'], ['MY_LEX_START', $dollarState]],
224            ];
225        } elseif (is_string($dollarState)) {
226            $samples['@COVERAGE'][] = ['$identifier', ['IDENT'], ['MY_LEX_START', $dollarState]];
227        }
228        if (in_array('MY_LEX_ESCAPE', $states, true)) {
229            $samples['@COVERAGE'][] = ['\\N', ['NULL_SYM'], ['MY_LEX_START', 'MY_LEX_ESCAPE']];
230        }
231        if (in_array('MY_LEX_STRING_OR_DELIMITER', $states, true)) {
232            $samples['@COVERAGE'][] = ['"text"', ['TEXT_STRING'], ['MY_LEX_START', 'MY_LEX_STRING_OR_DELIMITER']];
233        }
234        $samples['@COVERAGE'][] = ['name.column', ['IDENT', '.', 'IDENT'], ['MY_LEX_IDENT_SEP', 'MY_LEX_IDENT_START']];
235        $samples['@COVERAGE'][] = ['.5', ['DECIMAL_NUM'], ['MY_LEX_REAL_OR_POINT', 'MY_LEX_REAL']];
236        $samples['@COVERAGE'][] = ['<=', ['LE'], ['MY_LEX_CMP_OP']];
237        $samples['@COVERAGE'][] = ['<=>', ['EQUAL_SYM'], ['MY_LEX_CMP_OP', 'MY_LEX_LONG_CMP_OP']];
238        $samples['@COVERAGE'][] = ['*/', ['*', '/'], ['MY_LEX_END_LONG_COMMENT']];
239        $samples['@COVERAGE'][] = [';', [';'], ['MY_LEX_SEMICOLON']];
240        $samples['@COVERAGE'][] = ['@name', ['@', 'LEX_HOSTNAME'], ['MY_LEX_USER_END', 'MY_LEX_HOSTNAME']];
241        $samples['@COVERAGE'][] = ["@'name'", ['@', 'IDENT_QUOTED'], ['MY_LEX_USER_END', 'MY_LEX_USER_VARIABLE_DELIMITER']];
242        $samples['@COVERAGE'][] = ['@@name', ['@', '@', 'IDENT'], ['MY_LEX_USER_END', 'MY_LEX_SYSTEM_VAR', 'MY_LEX_IDENT_OR_KEYWORD']];
243
244        return $samples;
245    }

247    /**
248     * Adds lexical families and punctuation absent from the keyword table.
249     *
250     * @param Terminals $terminals
251     * @param array<string, list<array{string, list<string>, list<string>, 3?: string}>> $samples
252     */
253    public function addLexicalWitnesses(array &$terminals, array $samples): void
254    {
255        foreach ($samples as $terminal => $witnesses) {
256            foreach ($witnesses as $index => $sample) {
257                [$sql, $tokens, $units] = $sample;
258                $terminals[$terminal][] = $this->witness(
259                    "mysql.family.{$terminal}.{$index}",
260                    $sql,
261                    $tokens,
262                    $units,
263                    $sample[3] ?? null,
264                );
265            }
266        }
267
268        $punctuation = str_split('!%&()*+,-./:;<=>?@[]^{}|~');
269        foreach ($punctuation as $terminal) {
270            if (!isset($terminals[$terminal])) {
271                $terminals[$terminal][] = $this->witness(
272                    'mysql.punctuation.' . bin2hex($terminal),
273                    $terminal,
274                    [$terminal],
275                    ['MY_LEX_START', 'MY_LEX_CHAR'],
276                );
277            }
278        }
279    }

282     * Adds grammar entry tokens and version-specific parser spellings.
283     *
284     * @param Terminals $terminals
285     * @return list<string>
286     */
287    public function addParserWitnesses(array &$terminals, string $version, Grammar $grammar): array
288    {
289        $versionedParserTokens = ['END_OF_INPUT'];
290        foreach (\SqlFaker\Grammar\Lexical\TerminalInventory::fromGrammar($grammar) as $terminal) {
291            if (str_starts_with($terminal, 'GRAMMAR_SELECTOR_')) {
292                $versionedParserTokens[] = $terminal;
293            }
294        }
295        foreach ($versionedParserTokens as $terminal) {
296            $unit = 'parser-entry:' . $terminal;
297            $terminals[$terminal][] = $this->witness(
298                'mysql.parser.' . $terminal,
299                '',
300                [],
301                [$unit],
302            );
303        }
304        if (version_compare(substr($version, strlen('mysql-')), '8.0.0', '<')) {
305            $terminals['WITH_CUBE_SYM'][] = $this->witness(
306                'mysql.parser.WITH_CUBE_SYM',
307                'WITH CUBE',
308                ['WITH_CUBE_SYM'],
309                ['MY_LEX_START', 'MY_LEX_IDENT'],
310            );
311        }
312
313        return $versionedParserTokens;
314    }

316    /**
317     * Verifies every lexer state and parser entry has a witness.
318     *
319     * @param Terminals $terminals
320     * @param list<string> $states
321     * @param list<string> $versionedParserTokens
322     * @return array{units: list<string>, witnessed: array<string, string>, excluded: array<string, string>}
323     *
324     * @throws RuntimeException When a source state lacks coverage
325     */
326    public function coverage(array &$terminals, array $states, array $versionedParserTokens): array
327    {
328        $coverageUnits = $states;
329        foreach ($versionedParserTokens as $terminal) {
330            $coverageUnits[] = 'parser-entry:' . $terminal;
331        }
332        $coverageWitnessed = [];
333        foreach ($terminals as $witnesses) {
334            foreach ($witnesses as $witness) {
335                foreach ($witness['units'] as $unit) {
336                    if (in_array($unit, $coverageUnits, true)) {
337                        $coverageWitnessed[$unit] ??= $witness['id'];
338                    }
339                }
340            }
341        }
342        foreach (['MY_LEX_END', 'MY_LEX_EOL'] as $endState) {
343            if (in_array($endState, $states, true)) {
344                $terminals['@COVERAGE'][] = $this->witness(
345                    'mysql.coverage.' . $endState,
346                    '',
347                    [],
348                    [$endState],
349                );
350                $coverageWitnessed[$endState] = 'mysql.coverage.' . $endState;
351            }
352        }
353        if (in_array('MY_LEX_OPERATOR_OR_IDENT', $states, true)) {
354            $terminals['@COVERAGE'][] = $this->witness(
355                'mysql.coverage.MY_LEX_OPERATOR_OR_IDENT',
356                'a + b',
357                ['IDENT', '+', 'IDENT'],
358                ['MY_LEX_OPERATOR_OR_IDENT'],
359            );
360            $coverageWitnessed['MY_LEX_OPERATOR_OR_IDENT'] = 'mysql.coverage.MY_LEX_OPERATOR_OR_IDENT';
361        }
362        $coverageExcluded = [];
363        $missingCoverage = array_values(array_diff($coverageUnits, array_keys($coverageWitnessed)));
364        if ($missingCoverage !== []) {
365            throw new RuntimeException('MySQL source model misses lexical states: ' . implode(', ', $missingCoverage));
366        }
367
368        ksort($coverageWitnessed);
369
370        return ['units' => $coverageUnits, 'witnessed' => $coverageWitnessed, 'excluded' => $coverageExcluded];
371    }
