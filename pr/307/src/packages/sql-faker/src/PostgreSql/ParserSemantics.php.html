7/**
8 * Applies what PostgreSQL's parser enforces outside its grammar.
9 *
10 * A Bison grammar only says which token sequences the parser will shift and
11 * reduce; the C actions attached to those rules reject a good deal more. A
12 * derivation that satisfied only the grammar would write SQL the parser
13 * refuses, so every such constraint is applied here before the terminals are
14 * handed to the lexer.
15 *
16 * @visibility root
17 */
18final class ParserSemantics

27    /**
28     * Applies the constraints PostgreSQL enforces in semantic actions rather than in its grammar.
29     *
30     * The grammar accepts SET (name) and OPERATOR (op) with shapes the parser
31     * then rejects in C, and it accepts qualified names of any depth where the
32     * parser accepts at most three parts. A derivation that only satisfied the
33     * grammar would therefore write SQL PostgreSQL refuses to parse.
34     *
35     * @param list<string> $terminals Terminals a derivation produced
36     *
37     * @return list<string> Terminals with every semantic constraint satisfied
38     */
39    public function applied(array $terminals): array
40    {
41        $terminals = $this->truncateQualifiedNames($terminals);
42
43        $terminals = $this->completeSetOptions($terminals);
44
45        foreach ($terminals as $index => $terminal) {
46            if ($terminal !== 'OPERATOR' || ($terminals[$index + 1] ?? null) === '(') {
47                continue;
48            }
49            $start = array_search('(', array_slice($terminals, $index + 1), true);
50            if ($start === false) {
51                continue;
52            }
53            $start += $index + 1;
54            $end = $this->matchingParen($terminals, $start);
55            if ($end !== null && $end - $start === 2 && $terminals[$start + 1] !== ',') {
56                array_splice($terminals, $start + 1, 0, ['NONE', ',']);
57            }
58        }
59
60        return $this->lexicalGrammar->normalizeLookahead($terminals);
61    }

63    /**
64     * Cuts a qualified name down to the depth the parser accepts.
65     *
66     * @param list<string> $terminals Terminals a derivation produced
67     *
68     * @return list<string> Terminals whose qualified names are at most three parts deep
69     */
70    public function truncateQualifiedNames(array $terminals): array
71    {
72        $result = [];
73        $count = count($terminals);
74        for ($index = 0; $index < $count; $index++) {
75            $terminal = $terminals[$index];
76            if (!$this->isIdentifierTerminal($terminal) || ($terminals[$index + 1] ?? null) !== '.') {
77                $result[] = $terminal;
78                continue;
79            }
80
81            $chain = [$terminal];
82            while ($index + 2 < $count && $terminals[$index + 1] === '.') {
83                $following = $terminals[$index + 2];
84                if ($following === '*') {
85                    $index += 2;
86                    break;
87                }
88                if (!$this->isIdentifierTerminal($following)) {
89                    break;
90                }
91                $chain[] = '.';
92                $chain[] = $following;
93                $index += 2;
94            }
95            array_push($result, ...array_slice($chain, 0, 5));
96        }
97
98        return $result;
99    }

101    /**
102     * Answers where the parenthesis opened at one position is closed.
103     *
104     * @param list<string> $terminals Terminals a derivation produced
105     * @param int $open Position of the opening parenthesis
106     *
107     * @return int|null Position of the matching close, or null when there is none
108     */
109    public function matchingParen(array $terminals, int $open): ?int
110    {
111        $depth = 0;
112        for ($index = $open; $index < count($terminals); $index++) {
113            if ($terminals[$index] === '(') {
114                $depth++;
115            } elseif ($terminals[$index] === ')' && --$depth === 0) {
116                return $index;
117            }
118        }
119
120        return null;
121    }

130    public function isIdentifierTerminal(string $terminal): bool
131    {
132        return in_array($terminal, ['IDENT', 'UIDENT'], true)
133            || preg_match('/^[a-z_][a-z0-9_]*$/', $terminal) === 1;
134    }

136    /**
137     * Supplies NONE for SET options without an explicit value.
138     *
139     * @param list<string> $terminals
140     * @return list<string>
141     */
142    public function completeSetOptions(array $terminals): array
143    {
144        foreach ($terminals as $index => $terminal) {
145            if ($terminal !== 'SET' || ($terminals[$index + 1] ?? null) !== '(') {
146                continue;
147            }
148            $end = $this->matchingParen($terminals, $index + 1);
149            if ($end === null) {
150                continue;
151            }
152            $depth = 1;
153            for ($cursor = $index + 2; $cursor < $end; $cursor++) {
154                if ($terminals[$cursor] === '(') {
155                    $depth++;
156                } elseif ($terminals[$cursor] === ')') {
157                    $depth--;
158                }
159                if ($depth === 1
160                    && $this->isIdentifierTerminal($terminals[$cursor])
161                    && in_array($terminals[$cursor + 1] ?? null, [',', ')'], true)
162                    && ($terminals[$cursor - 1] ?? null) !== '='
163                ) {
164                    array_splice($terminals, $cursor + 1, 0, ['=', 'NONE']);
165                    $end += 2;
166                    $cursor += 2;
167                }
168            }
169        }
170
171        return $terminals;
172    }
