166    public function attachUnit(array &$terminals, string $witnessId, string $unit): void
167    {
168        foreach ($terminals as &$witnesses) {
169            foreach ($witnesses as &$witness) {
170                if ($witness['id'] === $witnessId) {
171                    $witness['units'][] = $unit;

173                    return;
174                }
175            }
176            unset($witness);
177        }
178        unset($witnesses);

180        throw new RuntimeException("PostgreSQL scanner rule references an unknown witness: {$witnessId}");

183    /**
184     * Builds keyword witnesses including the lookahead context required by the parser.
185     *
186     * @param array<string, list<string>> $keywords
187     * @param array<string, array{token: string, followed_by: list<string>}> $lookahead
188     * @return Terminals
189     */
190    public function keywordWitnesses(array $keywords, array $lookahead): array
191    {
192        $terminals = [];
193        foreach ($keywords as $terminal => $lexemes) {
194            foreach ($lexemes as $index => $lexeme) {
195                $terminals[$terminal][] = $this->witness(
196                    "postgresql.keyword.{$terminal}.{$index}",
197                    $lexeme,
198                    [$terminal],
199                );
200            }
201        }
202
203        foreach ($lookahead as $baseTerminal => $rule) {
204            $followingTerminal = $rule['followed_by'][0];
205            $baseLexeme = $keywords[$baseTerminal][0];
206            $followingLexeme = $keywords[$followingTerminal][0];
207            $contextSql = $baseLexeme . ' ' . $followingLexeme;
208            $terminals[$rule['token']][] = [
209                'id' => "postgresql.lookahead.{$rule['token']}",
210                'sql' => $baseLexeme,
211                'context_sql' => $contextSql,
212                'tokens' => [$rule['token']],
213                'units' => [],
214            ];
215        }
216
217        return $terminals;
218    }

262    /**
263     * Verifies that scanner rules and parser modes are witnessed or explicitly excluded.
264     *
265     * @param Terminals $terminals
266     * @param list<string> $rules
267     * @return array{units: list<string>, witnessed: array<string, string>, excluded: array<string, string>}
268     *
269     * @throws RuntimeException When the inventory and its witness mapping disagree
270     */
271    public function coverage(array &$terminals, array $rules): array
272    {
273        $ruleCount = count($rules) + 1;
274        $coverageUnits = $this->addParserModes($terminals, $ruleCount);
275
276        $ruleWitnesses = $this->ruleWitnesses();
277        if ($ruleWitnesses === [] || max(array_keys($ruleWitnesses)) >= $ruleCount) {
278            throw new RuntimeException('PostgreSQL source rule mapping exceeds the parsed rule inventory.');
279        }
280        foreach ($ruleWitnesses as $rule => $witnessId) {
281            $this->attachUnit($terminals, $witnessId, 'rule:' . $rule);
282        }
283
284        $coverageWitnessed = [];
285        foreach ($terminals as $witnesses) {
286            foreach ($witnesses as $witness) {
287                foreach ($witness['units'] as $unit) {
288                    $coverageWitnessed[$unit] ??= $witness['id'];
289                }
290            }
291        }
292        $coverageExcluded = $this->excludedRules($ruleCount);
293        $missing = array_values(array_diff(
294            $coverageUnits,
295            array_keys($coverageWitnessed),
296            array_keys($coverageExcluded),
297        ));
298        if ($missing !== []) {
299            throw new RuntimeException('PostgreSQL source model misses scanner rules: ' . implode(', ', $missing));
300        }
301        ksort($coverageWitnessed);
302
303        return ['units' => $coverageUnits, 'witnessed' => $coverageWitnessed, 'excluded' => $coverageExcluded];
304    }
