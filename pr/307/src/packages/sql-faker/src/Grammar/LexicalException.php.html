9/**
10 * Reports that concrete SQL did not preserve the derived parser-token sequence.
11 *
12 * Tokenizing is applied to SQL text, so a mismatch describes the input rather
13 * than a defect in the lexer, and callers are expected to handle it.
14 *
15 * Each dialect hits the same failures with its own name attached, so the
16 * wording lives here rather than being spelled out three times. What the
17 * generator writes and what the tokenizer reads back are the two halves of one
18 * contract, and every way they can disagree is named below.
19 *
20 * @visibility public
21 * @example Catch this failure as a runtime exception
22 *     $failure = new \SqlFaker\Grammar\LexicalException('SQL generation failed');
23 *     $failure instanceof \RuntimeException // => true
24 *     $failure->getMessage() // => 'SQL generation failed'
25 */
26final class LexicalException extends RuntimeException

160    /**
161     * Reports SQL that did not tokenize back to the terminals it was generated from.
162     *
163     * This is the failure the whole realization path exists to catch: the
164     * generator believed it was writing one token sequence and the dialect's own
165     * rules read another, so both sequences and the text are reported together.
166     *
167     * @param string $dialect Dialect name as it appears in messages
168     * @param string $version Lexical profile version in use
169     * @param list<string> $expected Tokens the terminals were meant to produce
170     * @param list<string> $actual Tokens the text actually produced
171     * @param string $sql The generated text
172     *
173     * @return self Exception carrying both sequences and the text
174     * @visibility root
175     */
176    public static function roundTripMismatch(
177        string $dialect,
178        string $version,
179        array $expected,
180        array $actual,
181        string $sql,
182    ): self {
183        return new self(sprintf(
184            "%s lexical round-trip failed for %s.\nExpected: %s\nActual: %s\nSQL: %s",
185            $dialect,
186            $version,
187            self::rendered($expected),
188            self::rendered($actual),
189            $sql,
190        ));
191    }

193    /**
194     * Renders a token sequence for a failure message.
195     *
196     * The sequences being compared came out of arbitrary generated bytes, so
197     * rendering them can meet input JSON has no encoding for. Substituting
198     * those bytes keeps the round-trip failure reportable: a message that could
199     * itself fail would hide the thing worth reporting behind an encoding
200     * error.
201     *
202     * @param list<string> $tokens Tokens to render
203     *
204     * @return string The sequence as JSON, or a placeholder when it cannot be rendered
205     * @visibility root
206     */
207    public static function rendered(array $tokens): string
208    {
209        $json = json_encode($tokens, JSON_INVALID_UTF8_SUBSTITUTE);
210
211        return is_string($json) ? $json : '(unrenderable)';
212    }
