38    /**
39     * Walks from one symbol until nothing but terminals is left.
40     *
41     * @param string $startSymbol Symbol the walk begins at
42     * @param GenerationPlan<bool> $plan Plan directing the walk
43     *
44     * @return list<Terminal> Terminals the walk arrived at
45     *
46     * @throws GenerationException When the grammar, the plan, or the step budget leaves no production to take
47     */
48    public function of(string $startSymbol, GenerationPlan $plan): array
49    {
50        /** @var list<Symbol> $form */
51        $form = [new NonTerminal($startSymbol)];
52        /** @var array<string, int> $occurrences */
53        $occurrences = [];
54
55        while (true) {
56            $index = $this->firstNonTerminal($form);
57            if ($index === null) {
58                break;
59            }
60
61            $this->steps++;
62            if ($this->steps > self::STEP_LIMIT) {
63                throw GenerationException::derivationLimitExceeded();
64            }
65
66            /** @var NonTerminal $nonTerminal */
67            $nonTerminal = $form[$index];
68            $occurrence = $occurrences[$nonTerminal->value] ?? 0;
69            $occurrences[$nonTerminal->value] = $occurrence + 1;
70            $alternatives = $this->alternatives($nonTerminal, $plan, $occurrence);
71
72            if ($plan->usesStepBudget()) {
73                $alternatives = $this->affordable($alternatives, new Production(array_slice($form, $index + 1)));
74            }
75            $production = $this->selectProduction($alternatives, $plan);
76
77            $form = [
78                ...array_slice($form, 0, $index),
79                ...$production->symbols,
80                ...array_slice($form, $index + 1),
81            ];
82        }
83
84        /** @var list<Terminal> $form */
85        return $form;
86    }

97        foreach ($form as $index => $symbol) {

106    /**
107     * Keeps only the alternatives the walk can still afford to finish.
108     *
109     * What is left of the form behind the symbol being rewritten has to be
110     * derived too, so the budget an alternative may spend is the step limit
111     * less what has been spent and less what the remainder will cost.
112     *
113     * @param non-empty-list<Production> $alternatives Alternatives the grammar and the plan both allow
114     * @param Production $remainder What the walk still has to derive behind this symbol
115     *
116     * @return non-empty-list<Production> Alternatives that still leave room to finish
117     *
118     * @throws GenerationException When none of them fits in the remaining budget
119     */
120    public function affordable(array $alternatives, Production $remainder): array
121    {
122        $budget = self::STEP_LIMIT - $this->steps - $this->analyzer->estimateProductionSteps($remainder);
123        if ($budget < 0) {
124            throw GenerationException::derivationLimitExceeded();
125        }
126
127        $affordable = array_values(array_filter(
128            $alternatives,
129            fn (Production $alternative): bool => $this->analyzer->estimateProductionSteps($alternative) <= $budget,
130        ));
131        if ($affordable === []) {
132            throw GenerationException::derivationLimitExceeded();
133        }
134
135        return $affordable;
136    }

138    /**
139     * Answers which of the alternatives the walk takes.
140     *
141     * Up to the plan's depth the choice is free. Past it the walk is trying to
142     * finish, so it takes whichever alternative gets there in the fewest steps,
143     * and among equals whichever writes the least.
144     *
145     * @param non-empty-list<Production> $alternatives Alternatives the walk may still take
146     * @param GenerationPlan<bool> $plan Plan directing the walk
147     *
148     * @return Production Alternative to rewrite with
149     */
150    public function selectProduction(array $alternatives, GenerationPlan $plan): Production
151    {
152        if ($this->steps < $plan->maxDepth()) {
153            return $alternatives[$this->faker->numberBetween(0, count($alternatives) - 1)];
154        }
155
156        $selected = 0;
157        $bestSteps = PHP_INT_MAX;
158        $bestLength = PHP_INT_MAX;
159        foreach ($alternatives as $index => $alternative) {
160            $steps = $plan->usesStepBudget() ? $this->analyzer->estimateProductionSteps($alternative) : 0;
161            $length = $this->analyzer->estimateProductionLength($alternative);
162            if ($steps < $bestSteps || ($steps === $bestSteps && $length < $bestLength)) {
163                $bestSteps = $steps;
164                $bestLength = $length;
165                $selected = $index;
166            }
167        }
168
169        return $alternatives[$selected];
170    }

172    /**
173     * Finds realizable alternatives satisfying the plan at this occurrence.
174     *
175     * @param GenerationPlan<bool> $plan Plan directing the walk
176     * @return non-empty-list<Production>
177     *
178     * @throws GenerationException When no alternative satisfies the grammar and plan
179     */
180    public function alternatives(NonTerminal $nonTerminal, GenerationPlan $plan, int $occurrence): array
181    {
182        $rule = $this->grammar->ruleMap[$nonTerminal->value]
183            ?? throw GenerationException::unknownRule($nonTerminal->value);
184        if ($rule->alternatives === []) {
185            throw GenerationException::ruleHasNoAlternatives($nonTerminal->value);
186        }
187        $alternatives = array_values(array_filter(
188            $rule->alternatives,
189            $this->analyzer->isProductionViable(...),
190        ));
191        if ($alternatives === []) {
192            throw GenerationException::noRealizableAlternative($nonTerminal->value);
193        }
194        $pattern = $plan->patternAt($nonTerminal->value, $occurrence);
195        if ($pattern !== null) {
196            $alternatives = array_values(array_filter(
197                $alternatives,
198                static fn (Production $production): bool => $pattern->matches(array_map(
199                    static fn (Symbol $symbol): string => $symbol->value(),
200                    $production->symbols,
201                )),
202            ));
203            if ($alternatives === []) {
204                throw GenerationException::noAlternativeMatchingPlan($nonTerminal->value);
205            }
206        }
207        if ($this->steps === 1 && $plan->requiresNonEmpty()) {
208            $alternatives = array_values(array_filter(
209                $alternatives,
210                fn (Production $production): bool => $this->analyzer->estimateProductionLength($production) > 0,
211            ));
212            if ($alternatives === []) {
213                throw GenerationException::startRuleCannotProduceOutput($nonTerminal->value);
214            }
215        }
216
217        return $alternatives;
218    }
