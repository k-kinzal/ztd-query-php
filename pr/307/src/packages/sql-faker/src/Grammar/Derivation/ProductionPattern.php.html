9 *
10 * A grammar rule offers every alternative its parser recognises, and a caller
11 * generating a specific statement usually wants one of them. A pattern says
12 * which, either by naming symbols the alternative must contain, by naming the
13 * alternative outright, or by asking only that it write something.
14 *
15 * @visibility root
16 */

20    private const EXACT = 'exact';
21    private const NON_EMPTY = 'non-empty';

23    /**
24     * @param array<array-key, string> $symbols Symbols the pattern is written in terms of
25     * @param self::CONTAINING|self::EXACT|self::NON_EMPTY $mode How those symbols are compared against an alternative
26     */
27    public function __construct(
28        private readonly array $symbols,
29        private readonly string $mode,
30    ) {
31    }
32
33    /**

35     *
36     * @param string ...$symbols Symbols the alternative must contain
37     *
38     * @return self Pattern matching on containment
39     */
40    public static function containing(string ...$symbols): self
41    {
42        return new self($symbols, self::CONTAINING);
43    }
44
45    /**
46     * Matches only the alternative written exactly as the named symbols.
47     *
48     * @param string ...$symbols Symbols the alternative is made of, in order
49     *
50     * @return self Pattern matching one alternative
51     */
52    public static function exactly(string ...$symbols): self
53    {
54        return new self(array_values($symbols), self::EXACT);
55    }
56
57    /**
58     * Matches any alternative that writes at least one symbol.
59     *
60     * @return self Pattern refusing only the empty alternative
61     */
62    public static function nonEmpty(): self
63    {
64        return new self([], self::NON_EMPTY);
65    }
66
67    /**
68     * @param list<string> $symbols
69     */
70    public function matches(array $symbols): bool
71    {
72        if ($this->mode === self::EXACT) {
73            return $symbols === $this->symbols;
74        }
75        if ($this->mode === self::NON_EMPTY) {
76            return $symbols !== [];
77        }
78        foreach ($this->symbols as $required) {
79            if (!in_array($required, $symbols, true)) {
80                return false;
81            }
82        }
83
84        return true;
85    }
