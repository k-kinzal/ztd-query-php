174    /**
175     * @param array<string, list<string>> $lexemesByTerminal
176     * @return non-empty-string
177     *
178     * @throws InvalidArgumentException When the term bounds are not a positive ordered range, or no terminal carries a lexeme
179     * @throws LogicException When the chosen lexemes join to nothing
180     */
181    public function lexicalSequence(array $lexemesByTerminal, int $minTerms = 2, int $maxTerms = 4): string
182    {
183        if ($minTerms < 1 || $maxTerms < $minTerms) {
184            throw new InvalidArgumentException('Lexical sequence bounds must be positive and ordered.');
185        }
186        $lexemeSets = array_values(array_filter(
187            $lexemesByTerminal,
188            static fn (array $lexemes): bool => $lexemes !== [],
189        ));
190        if ($lexemeSets === []) {
191            throw new InvalidArgumentException('A lexical sequence requires at least one terminal lexeme.');
192        }
193
194        $terms = $this->faker->numberBetween($minTerms, $maxTerms);
195        $lexemes = [];
196        for ($index = 0; $index < $terms; $index++) {
197            $set = $lexemeSets[$this->faker->numberBetween(0, count($lexemeSets) - 1)];
198            $lexemes[] = $set[$this->faker->numberBetween(0, count($set) - 1)];
199        }
200
201        $sequence = implode(' ', $lexemes);
202        if ($sequence === '') {
203            throw new LogicException('The lexical catalog produced an empty sequence.');
204        }
205
206        return $sequence;
207    }
