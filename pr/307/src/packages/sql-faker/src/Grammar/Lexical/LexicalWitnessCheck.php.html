7use SqlFaker\Grammar\LexicalCatalogException;

9/**
10 * Checks that the catalog's witnesses and its coverage agree with each other.
11 *
12 * The two halves of the catalog are written by the same generator but read
13 * independently, so nothing in the file itself guarantees they still describe
14 * the same lexer. These checks are what turn "both halves parsed" into "both
15 * halves refer to each other": every witness identifier is unique, every unit a
16 * witness claims exists, and every witnessed unit is claimed back by the
17 * witness the coverage names.
18 *
19 * @phpstan-type Witness array{id: string, sql: string, tokens: list<string>, units: list<string>, context_sql?: string}
20 * @phpstan-type Coverage array{units: list<string>, witnessed: array<string, string>, excluded: array<string, string>}
21 *
22 * @visibility parent
23 */
24final class LexicalWitnessCheck
25{
26    /**
27     * Checks the witnesses of a catalog against its coverage.
28     *
29     * @param array<string, list<Witness>> $terminals Witnesses by terminal name
30     * @param array<string, string> $exclusions Reasons by excluded terminal name
31     * @param Coverage $coverage Coverage section of the catalog
32     *
33     * @throws LexicalCatalogException When the two halves disagree
34     */
35    public function verify(array $terminals, array $exclusions, array $coverage): void
36    {
37        $identifiers = $this->identifiersOf($terminals, $exclusions, $coverage['units']);
38        $this->verifyCoverageNamesRealWitnesses($terminals, $coverage['witnessed'], $identifiers);
39    }
40
41    /**
42     * Collects every witness identifier, checking each witness as it goes.
43     *
44     * @param array<string, list<Witness>> $terminals Witnesses by terminal name
45     * @param array<string, string> $exclusions Reasons by excluded terminal name
46     * @param list<string> $units Units the upstream lexer is made of
47     *
48     * @return array<string, true> The identifiers, as a set
49     *
50     * @throws LexicalCatalogException When a terminal is empty or excluded, or an identifier or unit is wrong
51     */
52    public function identifiersOf(array $terminals, array $exclusions, array $units): array
53    {
54        $known = array_fill_keys($units, true);
55        $identifiers = [];
56
57        foreach ($terminals as $terminal => $witnesses) {
58            if (isset($exclusions[$terminal])) {
59                throw LexicalCatalogException::terminalIsAlsoExcluded();
60            }
61            if ($witnesses === []) {
62                throw LexicalCatalogException::emptyTerminal($terminal);
63            }
64
65            foreach ($witnesses as $witness) {
66                if (isset($identifiers[$witness['id']])) {
67                    throw LexicalCatalogException::duplicateWitnessId($witness['id']);
68                }
69                $identifiers[$witness['id']] = true;
70
71                foreach ($witness['units'] as $unit) {
72                    if (!isset($known[$unit])) {
73                        throw LexicalCatalogException::unknownCoverageUnit($unit);
74                    }
75                }
76            }
77        }
78
79        return $identifiers;
80    }
81
82    /**
83     * Checks that each witnessed unit is claimed back by the witness naming it.
84     *
85     * @param array<string, list<Witness>> $terminals Witnesses by terminal name
86     * @param array<string, string> $witnessed Witness identifiers by unit name
87     * @param array<string, true> $identifiers Every identifier the witnesses carry
88     *
89     * @throws LexicalCatalogException When a named witness is missing or does not cover the unit
90     */
91    public function verifyCoverageNamesRealWitnesses(array $terminals, array $witnessed, array $identifiers): void
92    {
93        foreach ($witnessed as $unit => $id) {
94            if (!isset($identifiers[$id])) {
95                throw LexicalCatalogException::unknownWitness($unit);
96            }
97            if (!$this->covers($terminals, $id, $unit)) {
98                throw LexicalCatalogException::witnessDoesNotCoverItsUnit($unit);
99            }
100        }
101    }
102
103    /**
104     * Reports whether the named witness lists the unit among the ones it covers.
105     *
106     * @param array<string, list<Witness>> $terminals Witnesses by terminal name
107     * @param string $id Identifier of the witness to look for
108     * @param string $unit Unit the witness is expected to cover
109     *
110     * @return bool True when the witness exists and claims the unit
111     */
112    public function covers(array $terminals, string $id, string $unit): bool
113    {
114        foreach ($terminals as $witnesses) {
115            foreach ($witnesses as $witness) {
116                if ($witness['id'] === $id && in_array($unit, $witness['units'], true)) {
117                    return true;
118                }
119            }
120        }
121
122        return false;
123    }
124}
