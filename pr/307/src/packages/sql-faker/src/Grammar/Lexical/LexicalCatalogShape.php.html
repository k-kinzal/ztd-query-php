9/**
10 * Reads the checked-in terminal catalog into the shape the rest of the code assumes.
11 *
12 * The catalog arrives as a decoded resource file, so nothing about its contents
13 * is known before it is inspected. Each section is read on its own and reports
14 * which field was wrong, because a catalog is regenerated from an upstream
15 * lexer and "the shape is invalid" is not enough to find what changed.
16 *
17 * @phpstan-type Witness array{id: string, sql: string, tokens: list<string>, units: list<string>, context_sql?: string}
18 * @phpstan-type Catalog array{
19 *     source: array{engine: string, entrypoint: string},
20 *     terminals: array<string, list<Witness>>,
21 *     terminal_exclusions: array<string, string>,
22 *     coverage: array{units: list<string>, witnessed: array<string, string>, excluded: array<string, string>}
23 * }
24 */
25final class LexicalCatalogShape
26{
27    /** @readonly */
28    private LexicalWitnessShape $witnesses;

31     * @param LexicalWitnessShape|null $witnesses Reads the individual lexing examples
32     */
33    public function __construct(?LexicalWitnessShape $witnesses = null)
34    {
35        $this->witnesses = $witnesses ?? new LexicalWitnessShape();
36    }

38    /**
39     * Reads a decoded catalog file.
40     *
41     * @param array<string, mixed> $catalog Catalog as decoded from its resource file
42     *
43     * @return Catalog The catalog with every section checked
44     *
45     * @throws LexicalCatalogException When any section is not what it must be
46     */
47    public function of(array $catalog): array
48    {
49        return [
50            'source' => $this->sourceOf($catalog),
51            'terminals' => $this->terminalsOf($catalog),
52            'terminal_exclusions' => $this->exclusionsOf($catalog),
53            'coverage' => $this->coverageOf($catalog),
54        ];
55    }

57    /**
58     * Reads the record of which lexer the catalog was generated from.
59     *
60     * @param array<string, mixed> $catalog Catalog as decoded from its resource file
61     *
62     * @return array{engine: string, entrypoint: string} The upstream lexer's identity
63     *
64     * @throws LexicalCatalogException When the record is absent or mistyped
65     */
66    public function sourceOf(array $catalog): array
67    {
68        $source = $catalog['source'] ?? throw LexicalCatalogException::malformedShape('source');
69        if (!is_array($source)) {
70            throw LexicalCatalogException::malformedShape('source');
71        }
72
73        $engine = $source['engine'] ?? throw LexicalCatalogException::malformedShape('source.engine');
74        if (!is_string($engine)) {
75            throw LexicalCatalogException::malformedShape('source.engine');
76        }
77
78        $entrypoint = $source['entrypoint'] ?? throw LexicalCatalogException::malformedShape('source.entrypoint');
79        if (!is_string($entrypoint)) {
80            throw LexicalCatalogException::malformedShape('source.entrypoint');
81        }
82
83        return ['engine' => $engine, 'entrypoint' => $entrypoint];
84    }

86    /**
87     * Reads the witnesses catalogued for each terminal.
88     *
89     * @param array<string, mixed> $catalog Catalog as decoded from its resource file
90     *
91     * @return array<string, list<Witness>> Witnesses by terminal name
92     *
93     * @throws LexicalCatalogException When the section or any witness is mistyped
94     */
95    public function terminalsOf(array $catalog): array
96    {
97        $terminals = $catalog['terminals'] ?? throw LexicalCatalogException::malformedShape('terminals');
98        if (!is_array($terminals)) {
99            throw LexicalCatalogException::malformedShape('terminals');
100        }
101
102        $read = [];
103        foreach ($terminals as $terminal => $witnesses) {
104            if (!is_string($terminal) || !is_array($witnesses)) {
105                throw LexicalCatalogException::malformedTerminalCatalog();
106            }
107
108            $readWitnesses = [];
109            foreach ($witnesses as $witness) {
110                $readWitnesses[] = $this->witnesses->of($terminal, $witness);
111            }
112            $read[$terminal] = $readWitnesses;
113        }
114
115        return $read;
116    }

120     *
121     * @param array<string, mixed> $catalog Catalog as decoded from its resource file
122     *
123     * @return array<string, string> Reasons by terminal name
124     *
125     * @throws LexicalCatalogException When the section is mistyped or a reason is empty
126     */
127    public function exclusionsOf(array $catalog): array
128    {
129        $exclusions = $catalog['terminal_exclusions']

131        if (!is_array($exclusions)) {
132            throw LexicalCatalogException::malformedShape('terminal_exclusions');
133        }
134
135        $read = [];
136        foreach ($exclusions as $terminal => $reason) {
137            if (!is_string($terminal) || !is_string($reason) || $reason === '') {
138                throw LexicalCatalogException::malformedExclusion();
139            }
140            $read[$terminal] = $reason;
141        }
142
143        return $read;
144    }

170    /**
171     * Reads the list of units the upstream lexer is made of.
172     *
173     * @param array<mixed> $coverage Coverage section of the catalog
174     *
175     * @return list<string> Unit names
176     *
177     * @throws LexicalCatalogException When the list is absent or holds anything but names
178     */
179    public function unitsOf(array $coverage): array
180    {
181        $units = $coverage['units'] ?? throw LexicalCatalogException::malformedShape('coverage.units');
182        if (!is_array($units)) {
183            throw LexicalCatalogException::malformedShape('coverage.units');
184        }
185        if (!array_is_list($units)) {
186            throw LexicalCatalogException::malformedCoverageUnits();
187        }
188        foreach ($units as $unit) {
189            if (!is_string($unit)) {
190                throw LexicalCatalogException::malformedCoverageUnits();
191            }
192        }
193
194        /** @var list<string> $units */
195        return $units;
196    }

198    /**
199     * Reads which witness accounts for each covered unit.

204     *

207    public function witnessedOf(array $coverage): array
208    {
209        $witnessed = $coverage['witnessed'] ?? throw LexicalCatalogException::malformedShape('coverage.witnessed');
210        if (!is_array($witnessed)) {
211            throw LexicalCatalogException::malformedShape('coverage.witnessed');
212        }
213
214        $read = [];
215        foreach ($witnessed as $unit => $id) {
216            if (!is_string($unit) || !is_string($id)) {
217                throw LexicalCatalogException::malformedCoverageWitness();
218            }
219            $read[$unit] = $id;
220        }
