7/**
8 * A moving read position over grammar source text.
9 *
10 * Scanners share one cursor and advance it as they consume characters, so the
11 * position is a collaborator they are handed rather than a field each of them
12 * reaches into. That is what lets a single lexeme rule — a quoted literal, a
13 * block comment, a brace-delimited action — be exercised on its own: give it a
14 * cursor over the input it should read, and inspect where the cursor stopped.
15 *
16 * @visibility root
17 */

104    /**
105     * Consumes characters for as long as the predicate accepts them.
106     *
107     * @param callable(string): bool $accepts Decides whether a character belongs to the run
108     *
109     * @return string The consumed run, empty when the first character is rejected
110     */
111    public function takeWhile(callable $accepts): string
112    {
113        $start = $this->offset;
114        while ($this->offset < $this->length && $accepts($this->source[$this->offset])) {
115            ++$this->offset;
116        }
117
118        return substr($this->source, $start, $this->offset - $start);
119    }

121    /**
122     * Consumes text up to the terminator, and the terminator with it.
123     *
124     * An absent terminator consumes the rest of the source, which is how an
125     * unterminated prologue or comment is reported as its content instead of
126     * as a failure.
127     *
128     * @param string $terminator Text that closes the run
129     *
130     * @return string The text before the terminator
131     */
132    public function takeUntil(string $terminator): string
133    {
134        $start = $this->offset;
135        $end = strpos($this->source, $terminator, $this->offset);
136        if ($end === false) {
137            $this->offset = $this->length;
138
139            return substr($this->source, $start);
140        }
141
142        $this->offset = $end + strlen($terminator);
143
144        return substr($this->source, $start, $end - $start);
145    }

147    /**
148     * Consumes a quoted run, starting at its opening quote.
149     *
150     * A backslash escapes the character after it, so an escaped quote does not
151     * close the run. An unterminated run consumes the rest of the source.
152     *
153     * @param string $quote The quote character that opens and closes the run
154     *
155     * @return string The unescaped text between the quotes
156     */
157    public function takeQuoted(string $quote): string
158    {
159        $this->advance();
160        $taken = '';
161
162        while ($this->offset < $this->length) {
163            $character = $this->source[$this->offset];
164
165            if ($character === '\\') {
166                ++$this->offset;
167                if ($this->offset < $this->length) {
168                    $taken .= $this->source[$this->offset];
169                    ++$this->offset;
170                }
171                continue;
172            }
173
174            ++$this->offset;
175            if ($character === $quote) {
176                break;
177            }
178
179            $taken .= $character;
180        }
181
182        return $taken;
183    }

185    /**
186     * Reads back text the cursor has already passed over.
187     *
188     * A lexeme whose body is defined by delimiters — a brace-delimited action,
189     * for instance — is recognised by walking to its closing delimiter and then
190     * asking for the text in between, rather than by accumulating characters it
191     * may still turn out not to own.
192     *
193     * @param int $start Offset the text begins at
194     * @param int $end Offset the text ends before
195     *
196     * @return string The text between the two offsets, empty when they cross
197     */
198    public function textBetween(int $start, int $end): string
199    {
200        return $end <= $start ? '' : substr($this->source, $start, $end - $start);
201    }

203    /**
204     * Consumes everything that is left.
205     *
206     * @return string The remaining source, empty when the cursor is at the end
207     */
208    public function takeRest(): string
209    {
210        $rest = substr($this->source, $this->offset);
211        $this->offset = $this->length;
212
213        return $rest;
214    }
