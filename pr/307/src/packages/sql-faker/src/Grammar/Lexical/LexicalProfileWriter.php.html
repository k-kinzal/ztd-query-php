11/**
12 * Writes a generated profile and its grammar to the files they are read from.
13 *
14 * A profile and the AST it was built against describe one release together, so
15 * publishing one without the other would leave the package generating against
16 * a grammar its lexer does not match. Both are staged beside their destination
17 * and renamed into place, which is the only way to replace two files without a
18 * window where one has moved and the other has not.
19 */

30     * Publishes the grammar and lexical profile only after both have been generated and validated.
31     *
32     * @param array<string, mixed> $profile
33     *
34     * @throws RuntimeException When either artifact cannot be written
35     */
36    public function publishVersion(SqlVersion $version, string $ast, array $profile): void
37    {
38        $artifacts = [
39            $version->astPath => $ast,
40            $version->lexicalPath => $this->rendered($profile),
41        ];
42        $temporaryPaths = [];
43        try {
44            foreach ($artifacts as $path => $contents) {
45                $directory = $this->directory->prepared($path);
46                $temporaryPath = tempnam($directory, '.sql-faker-');
47                if ($temporaryPath === false || file_put_contents($temporaryPath, $contents) === false) {
48                    throw new RuntimeException("Failed to stage {$path}");
49                }
50                $temporaryPaths[$path] = $temporaryPath;
51            }
52            foreach ($temporaryPaths as $path => $temporaryPath) {
53                if (!rename($temporaryPath, $path)) {
54                    throw new RuntimeException("Failed to publish {$path}");
55                }
56                unset($temporaryPaths[$path]);
57                fwrite(STDOUT, "Generated: {$path}\n");
58            }
59        } finally {
60            foreach ($temporaryPaths as $temporaryPath) {
61                if (file_exists($temporaryPath)) {
62                    unlink($temporaryPath);
63                }
64            }
65        }
66    }

79     * @param array<string, mixed> $profile
80     * @return array<string, mixed>
81     *
82     * @throws RuntimeException When a witness is not shaped like one
83     */
84    public function compacted(array $profile): array
85    {
86        if (!isset($profile['catalog']) || !is_array($profile['catalog'])
87            || !isset($profile['catalog']['terminals']) || !is_array($profile['catalog']['terminals'])
88        ) {
89            return $profile;
90        }
91
92        $terminals = [];
93        foreach ($profile['catalog']['terminals'] as $terminal => $witnesses) {
94            if (!is_array($witnesses)) {
95                throw new RuntimeException('Invalid lexical terminal witnesses while compacting.');
96            }
97            foreach ($witnesses as $witness) {
98                if (!is_array($witness)
99                    || !isset($witness['id'], $witness['sql'], $witness['tokens'], $witness['units'])
100                ) {
101                    throw new RuntimeException('Invalid lexical witness while compacting.');
102                }
103                $compact = [$witness['id'], $witness['sql'], $witness['tokens'], $witness['units']];
104                if (isset($witness['context_sql'])) {
105                    $compact[] = $witness['context_sql'];
106                }
107                $terminals[$terminal][] = $compact;
108            }
109        }
110        $profile['catalog']['terminals'] = $terminals;
111
112        return $profile;
113    }

115    /**
116     * Renders one value as the PHP literal a profile file is written from.
117     *
118     * A profile carries SQL that the server's lexer accepts, including control
119     * characters that var_export() would write literally into the file. Those
120     * are escaped by hand so the generated file stays readable and reloadable.
121     *
122     * @param mixed $value Value to render
123     * @param int $indent Current indentation, in spaces
124     *
125     * @return string The value as PHP source
126     *
127     * @throws RuntimeException When a string cannot be escaped
128     */
129    public function exported(mixed $value, int $indent = 0): string
130    {
131        if (is_string($value) && preg_match('/[\x00-\x1F\x7F]/', $value) === 1) {
132            $escaped = str_replace(['\\', '"', '$'], ['\\\\', '\\"', '\\$'], $value);
133            $escaped = preg_replace_callback(
134                '/[\x00-\x1F\x7F]/',
135                static fn (array $match): string => sprintf('\\x%02X', ord($match[0])),
136                $escaped,
137            );
138            if ($escaped === null) {
139                throw new RuntimeException('Failed to escape a lexical profile string.');
140            }
141
142            return '"' . $escaped . '"';
143        }
144        if (!is_array($value)) {
145            return var_export($value, true);
146        }
147        if ($value === []) {
148            return '[]';
149        }
150        if (array_is_list($value)) {
151            $items = [];
152            foreach ($value as $item) {
153                $items[] = $this->exported($item, $indent);
154            }
155
156            return '[' . implode(', ', $items) . ']';
157        }
158
159        $padding = str_repeat(' ', $indent);
160        $childPadding = str_repeat(' ', $indent + 4);
161        $lines = [];
162        foreach ($value as $key => $item) {
163            $lines[] = $childPadding . var_export($key, true) . ' => ' . $this->exported($item, $indent + 4) . ',';
164        }
165
166        return "[\n" . implode("\n", $lines) . "\n{$padding}]";
167    }
