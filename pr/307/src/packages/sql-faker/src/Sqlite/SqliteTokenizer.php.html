9/**
10 * Reads SQLite text back into the parser tokens the server would produce.
11 *
12 * This is the inverse of realization and the reason realization can be trusted:
13 * SQL written from a terminal sequence is read back here, and the two sequences
14 * must agree. SQLite's rules are its own — an identifier may be quoted four
15 * ways, a digit-separated number is a token of its own, and a parameter may be
16 * written with any of five sigils — so a tokenizer that merely looked plausible
17 * would agree with a generator that was equally wrong.
18 *
19 * @visibility root
20 */
21final class SqliteTokenizer

62    /**
63     * Reads SQL text into the tokens SQLite's own lexer would produce.
64     *
65     * @param string $sql Text to read
66     *
67     * @return list<string> Parser token names, in order
68     *
69     * @throws LexicalException When the text holds something the lexer cannot read
70     */
71    public function tokenize(string $sql): array
72    {
73        $tokens = [];
74        $length = strlen($sql);
75        $offset = 0;
76
77        while ($offset < $length) {
78            $startedAt = $offset;
79            if ($this->skipTrivia($sql, $offset)) {
80                if ($offset <= $startedAt) {
81                    throw LexicalException::noProgress('SQLite', $startedAt, $sql);
82                }
83
84                continue;
85            }
86
87            $token = $this->tokenAt($sql, $offset);
88            if ($token === null) {
89                throw LexicalException::unsupportedInput('SQLite', $offset, $sql);
90            }
91
92            if ($offset <= $startedAt) {
93                throw LexicalException::noProgress('SQLite', $startedAt, $sql);
94            }
95
96            $tokens[] = $token;
97        }
98
99        return $tokens;
100    }

103     * Reads the one token that starts at the offset, advancing past it.
104     *
105     * @param string $sql Text being read
106     * @param int $offset Where to read from; moved past what was consumed
107     *
108     * @return string|null The token name, or null when nothing here is a token
109     *
110     * @throws LexicalException When a quoted or bracketed run never closes
111     */
112    public function tokenAt(string $sql, int &$offset): ?string
113    {
114        return $this->quotedTokenAt($sql, $offset)
115            ?? $this->literalTokenAt($sql, $offset)
116            ?? $this->wordTokenAt($sql, $offset)
117            ?? $this->operatorTokenAt($sql, $offset);

120    /**
121     * Reads a quoted string or a quoted identifier in any of SQLite's spellings.
122     *
123     * @param string $sql Text being read
124     * @param int $offset Where to read from; moved past what was consumed
125     *
126     * @return string|null The token name, or null when no quoted run opens here
127     *
128     * @throws LexicalException When the run never closes
129     */
130    public function quotedTokenAt(string $sql, int &$offset): ?string
131    {
132        $character = $sql[$offset];
133
134        if ($character === "'") {
135            $this->skipQuoted($sql, $offset, "'");
136
137            return 'STRING';
138        }
139
140        if ($character === '"' || $character === '`') {
141            $this->skipQuoted($sql, $offset, $character);
142
143            return 'ID';
144        }
145
146        if ($character !== '[') {
147            return null;
148        }
149
150        $end = strpos($sql, ']', $offset + 1);
151        if ($end === false) {
152            throw LexicalException::unterminatedBracketIdentifier('SQLite');
153        }
154        $offset = $end + 1;
155
156        return 'ID';
157    }

159    /**
160     * Reads a blob, a bound parameter, or a numeric literal.
161     *
162     * @param string $sql Text being read
163     * @param int $offset Where to read from; moved past what was consumed
164     *
165     * @return string|null The token name, or null when none of these start here
166     */
167    public function literalTokenAt(string $sql, int &$offset): ?string
168    {
169        $patterns = [
170            "/\G[Xx]'(?:[0-9A-Fa-f]{2})*'/" => 'BLOB',
171            '/\G(?:\?[0-9]*|[:@$#][A-Za-z_][A-Za-z0-9_]*)/' => 'VARIABLE',
172            '/\G(?:(?:\d+\.\d*|\.\d+)(?:[eE][+-]?\d+)?|\d+[eE][+-]?\d+)/' => 'FLOAT',
173        ];
174
175        foreach ($patterns as $pattern => $token) {
176            if (preg_match($pattern, $sql, $match, 0, $offset) !== 1) {
177                continue;
178            }
179
180            $offset += strlen($match[0]);
181
182            return $token;
183        }
184
185        if (preg_match('/\G(?:0[xX][0-9A-Fa-f](?:_?[0-9A-Fa-f])*|\d(?:_?\d)*)/', $sql, $match, 0, $offset) !== 1) {
186            return null;
187        }
188
189        $offset += strlen($match[0]);
190
191        return str_contains($match[0], '_') ? 'QNUMBER' : 'INTEGER';
192    }

194    /**
195     * Reads a keyword or an unquoted identifier.
196     *
197     * @param string $sql Text being read
198     * @param int $offset Where to read from; moved past what was consumed
199     *
200     * @return string|null The token name, or null when no word starts here
201     */
202    public function wordTokenAt(string $sql, int &$offset): ?string
203    {
204        if (preg_match('/\G[A-Za-z_][A-Za-z0-9_$]*/', $sql, $match, 0, $offset) !== 1) {
205            return null;
206        }
207
208        $offset += strlen($match[0]);
209
210        return $this->keywordTokens[strtoupper($match[0])] ?? 'ID';
211    }

233    /**
234     * Consumes whitespace or one comment.
235     *
236     * @param string $sql Text being read
237     * @param int $offset Where to read from; moved past what was consumed
238     *
239     * @return bool True when something was skipped
240     *
241     * @throws LexicalException When a block comment never closes
242     */
243    public function skipTrivia(string $sql, int &$offset): bool
244    {
245        if (preg_match('/\G\s+/A', $sql, $match, 0, $offset) === 1) {
246            $offset += strlen($match[0]);
247
248            return true;
249        }
250
251        if (substr($sql, $offset, 2) === '--') {
252            $end = strpos($sql, "\n", $offset + 2);
253            $offset = $end === false ? strlen($sql) : $end + 1;
254
255            return true;
256        }
257
258        if (substr($sql, $offset, 2) !== '/*') {
259            return false;
260        }
261
262        $end = strpos($sql, '*/', $offset + 2);
263        if ($end === false) {
264            throw LexicalException::unterminatedBlockComment('SQLite');
265        }
266        $offset = $end + 2;
267
268        return true;
269    }

271    /**
272     * Consumes a quoted run, doubling being the way a quote escapes itself.
273     *
274     * @param string $sql Text being read
275     * @param int $offset Offset of the opening quote; moved past the closing one
276     * @param string $quote The quote character
277     *
278     * @throws LexicalException When the run never closes
279     */
280    public function skipQuoted(string $sql, int &$offset, string $quote): void
281    {
282        $length = strlen($sql);
283        ++$offset;
284
285        while ($offset < $length) {
286            if ($sql[$offset] !== $quote) {
287                ++$offset;
288                continue;
289            }
290            if (($sql[$offset + 1] ?? null) === $quote) {
291                $offset += 2;
292                continue;
293            }
294            ++$offset;
295
296            return;
297        }
298
299        throw LexicalException::unterminatedQuotedToken('SQLite', $sql);
300    }

302    /**
303     * Reports the operator written at the offset and the token it stands for.
304     *
305     * @param string $sql Text being read
306     * @param int $offset Where to look
307     *
308     * @return array{string, string}|null The operator and its token, or null when there is none
309     */
310    public function operatorAt(string $sql, int $offset): ?array
311    {
312        foreach (self::OPERATORS as $operator => $token) {
313            if (str_starts_with(substr($sql, $offset), $operator)) {
314                return [$operator, $token];
315            }
316        }
317
318        $token = self::PUNCTUATION[$sql[$offset]] ?? null;
319
320        return $token !== null ? [$sql[$offset], $token] : null;
321    }

323    /**
324     * Strips the `TK_` prefix and the spacing tokens SQLite's own lexer emits.
325     *
326     * Witnesses record what SQLite's lexer produced, which names tokens with a
327     * prefix and reports whitespace as a token of its own. Neither belongs in
328     * the parser-token sequence this tokenizer is compared against.
329     *
330     * @param list<string> $tokens Tokens as the upstream lexer named them
331     *
332     * @return list<string> The same tokens as the parser names them
333     */
334    public function normalizedSourceTokens(array $tokens): array
335    {
336        $normalized = [];
337        foreach ($tokens as $token) {
338            if ($token === 'TK_SPACE') {
339                continue;
340            }
341            $normalized[] = str_starts_with($token, 'TK_') ? substr($token, 3) : $token;
342        }
343
344        return $normalized;
345    }
