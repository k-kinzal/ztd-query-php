12/**
13 * Writes one SQLite parser terminal as the SQL text that produces it.
14 *
15 * There are two ways to write a terminal and they are not interchangeable. The
16 * catalogued way replays SQL that SQLite's own lexer was observed to turn into
17 * the terminal, which is what makes generated statements trustworthy. The
18 * synthetic way constructs text from the terminal's name, which reaches
19 * terminals no catalog witnesses but only promises to be plausible — so it is
20 * off unless a caller asks for it.
21 *
22 * @phpstan-type Realization array{string, list<string>}
23 *
24 * @visibility root
25 */
26final class SqliteTerminalRealizer
27{
28    /**
29     * The table option that makes a table reject values of the wrong type.
30     *
31     * SQLite's grammar spells this as an ordinary identifier, so the generator
32     * needs a terminal of its own to ask for it deliberately rather than hoping
33     * an identifier happens to come out as the word STRICT.
34     */
35    public const STRICT_TABLE_OPTION = 'STRICT_TABLE_OPTION';

38        'LP' => '(', 'RP' => ')', 'SEMI' => ';', 'COMMA' => ',', 'DOT' => '.',
39        'EQ' => '=', 'LT' => '<', 'LE' => '<=', 'GT' => '>', 'GE' => '>=', 'NE' => '<>',
40        'PLUS' => '+', 'MINUS' => '-', 'STAR' => '*', 'SLASH' => '/', 'REM' => '%',
41        'BITAND' => '&', 'BITOR' => '|', 'BITNOT' => '~', 'LSHIFT' => '<<', 'RSHIFT' => '>>',
42        'CONCAT' => '||', 'PTR' => '->',

69    /**
70     * Writes one terminal, reporting the tokens the text should read back as.
71     *
72     * @param string $terminal Terminal to write
73     * @param non-empty-string|null $requestedLexeme Exact text the caller wants, when it has one
74     *
75     * @return Realization The text and the tokens it should read back as
76     *
77     * @throws LexicalException When the terminal or the requested text cannot be realized
78     */
79    public function realize(string $terminal, ?string $requestedLexeme = null): array
80    {
81        if (!$this->supports($terminal)) {
82            throw LexicalException::unsupportedTerminal('SQLite', $this->profileVersion, $terminal);
83        }
84
85        if ($terminal === self::STRICT_TABLE_OPTION) {
86            return ['STRICT', ['ID']];
87        }
88
89        if ($requestedLexeme !== null) {
90            return $this->realizeRequested($terminal, $requestedLexeme);
91        }
92
93        if (!$this->allowSyntheticTerminals) {
94            return $this->realizeWitnessed($terminal);
95        }
96
97        return $this->realizeSynthetic($terminal);
98    }

100    /**
101     * Reports whether this realizer can write a terminal at all.
102     *
103     * @param string $terminal Terminal to look for
104     *
105     * @return bool True when the catalog witnesses it, or synthetic writing is allowed
106     */
107    public function supports(string $terminal): bool
108    {
109        return $terminal === self::STRICT_TABLE_OPTION
110            || $this->allowSyntheticTerminals
111            || $this->catalog->supports($terminal);
112    }

114    /**
115     * Writes a terminal by replaying one of its catalogued examples.
116     *
117     * @param string $terminal Terminal to write
118     *
119     * @return Realization The example's text and the tokens it was observed to produce
120     */
121    public function realizeWitnessed(string $terminal): array
122    {
123        $witnesses = $this->catalog->witnesses($terminal);
124        $witness = $witnesses[$this->faker->numberBetween(0, count($witnesses) - 1)];
125
126        return [$witness['sql'], $this->tokenizer->normalizedSourceTokens($witness['tokens'])];
127    }

129    /**
130     * Writes a terminal from its name rather than from an observed example.
131     *
132     * @param string $terminal Terminal to write
133     *
134     * @return Realization The constructed text and its tokens
135     *
136     * @throws LexicalException When the constructed text cannot be read back
137     */
138    public function realizeSynthetic(string $terminal): array
139    {
140        $operator = self::OPERATOR_TERMINALS[$terminal] ?? null;
141        if ($operator !== null) {
142            return [$operator, [$terminal]];
143        }
144
145        return match ($terminal) {
146            'ID', 'id', 'idj' => [$this->identifier(), ['ID']],
147            'ids', 'STRING' => [$this->stringLiteral(), ['STRING']],
148            'BLOB' => [$this->blobLiteral(), ['BLOB']],
149            'number', 'INTEGER' => [$this->strings->integerString(0, PHP_INT_MAX), ['INTEGER']],
150            'FLOAT' => [$this->strings->decimalString(), ['FLOAT']],
151            'QNUMBER' => ['1_0', ['QNUMBER']],
152            'VARIABLE' => [$this->parameter(), ['VARIABLE']],
153            'ANY' => ['_any', ['ID']],
154            default => $this->realizeFixed($terminal),
155        };
156    }

158    /**
159     * Writes the exact text a caller asked for, once it is shown to realize the terminal.
160     *
161     * @param string $terminal Terminal the text is meant to realize
162     * @param non-empty-string $requestedLexeme Exact text the caller wants
163     *
164     * @return array{non-empty-string, list<string>} The text and its tokens
165     *
166     * @throws LexicalException When the text does not realize the terminal
167     */
168    public function realizeRequested(string $terminal, string $requestedLexeme): array
169    {
170        if ($this->allowSyntheticTerminals) {
171            $tokens = $this->tokenizer->tokenize($requestedLexeme);
172            if ($tokens !== [$terminal]) {
173                throw LexicalException::lexemeDoesNotRealizeTerminal('SQLite', $terminal, $requestedLexeme);
174            }
175
176            return [$requestedLexeme, $tokens];
177        }
178
179        foreach ($this->catalog->witnesses($terminal) as $witness) {
180            if ($witness['sql'] === $requestedLexeme) {
181                return [$requestedLexeme, $this->tokenizer->normalizedSourceTokens($witness['tokens'])];
182            }
183        }
184
185        throw LexicalException::noWitnessForLexeme('SQLite', $terminal, $requestedLexeme);
186    }

189     * Writes a terminal that stands for a fixed keyword.
190     *
191     * @param string $terminal Terminal to write
192     *
193     * @return Realization The spelling and its tokens
194     *
195     * @throws LexicalException When the spelling cannot be read back
196     */
197    public function realizeFixed(string $terminal): array
198    {
199        $spellings = $this->keywords[$terminal] ?? null;
200        $lexeme = $spellings !== null
201            ? $spellings[$this->faker->numberBetween(0, count($spellings) - 1)]
202            : $terminal;
203
204        return [$lexeme, $spellings !== null ? [$terminal] : $this->tokenizer->tokenize($lexeme)];
205    }

210     * @return string An identifier

212    public function identifier(): string
213    {
214        $body = $this->faker->numberBetween(0, 3) === 0 ? 'select' : '_' . $this->strings->rawIdentifier();
215
216        return match ($this->faker->numberBetween(0, 7)) {
217            0 => SqliteQuoting::identifier($body . '"quoted'),
218            1 => SqliteQuoting::backtickIdentifier($body . '`quoted'),
219            2 => SqliteQuoting::bracketIdentifier($body . ']quoted'),
220            default => $body,
221        };
222    }

230    {

258    public function parameter(): string
259    {
260        return match ($this->faker->numberBetween(0, 4)) {
261            0 => '?',
262            1 => '?' . $this->faker->numberBetween(1, 10),
263            2 => ':' . $this->strings->rawIdentifier(),
264            3 => '@' . $this->strings->rawIdentifier(),
