52    {
53        $keywords = $this->keywords($version);
54        return new ChoiceLexemeGenerator(

56            new VersionedLexemeGenerator($version, new VersionCase(['sqlite-3.47.2'], new ChoiceLexemeGenerator(
57                $this->values(),
58                $this->symbols(),
59                $this->strictTypes(),
60                new ValueLexemeGenerator('GENERATED_STORAGE', new WordDomain(['VIRTUAL', 'STORED'], true), ['VIRTUAL', 'STORED'], 'identifier', 'src/build.c:sqlite3AddGenerated'),
61                new JoinLexemeGenerator($keywords['JOIN_KW'] ?? []),
62            ), 'sqlite-3.47.2-scanner')),
63        );
64    }

67     * Declares multi-character scanner operators as single lexemes.
68     */
69    public function symbols(): LexemeGenerator
70    {
71        $fixed = ['LP' => '(', 'RP' => ')', 'SEMI' => ';', 'COMMA' => ',', 'DOT' => '.',
72            'EQ' => '=', 'LT' => '<', 'LE' => '<=', 'GT' => '>', 'GE' => '>=', 'NE' => '<>',
73            'PLUS' => '+', 'MINUS' => '-', 'STAR' => '*', 'SLASH' => '/', 'REM' => '%',
74            'BITAND' => '&', 'BITOR' => '|', 'BITNOT' => '~', 'LSHIFT' => '<<', 'RSHIFT' => '>>',
75            'CONCAT' => '||', 'PTR' => '->', 'STRICT_TABLE_OPTION' => 'STRICT', 'ROWID_TABLE_OPTION' => 'ROWID'];
76        $generators = [];
77        foreach ($fixed as $terminal => $text) {
78            $generators[] = new MatchingLexemeGenerator($terminal, new FixedLexemeGenerator($text, 'symbol', 'src/tokenize.c:' . $terminal));
79        }
80        return new ChoiceLexemeGenerator(...$generators);
81    }

102    /**
103     * CC_ID / CC_QUOTE / CC_VARNUM / CC_VARALPHA.
104     * @see https://github.com/sqlite/sqlite/blob/version-3.47.2/src/tokenize.c
105     */
106    public function names(): LexemeGenerator
107    {
108        $name = new ChoiceDomain(
109            new QuotedDomain('"', minimum: 1, maximum: 64),
110            new IdentifierDomain('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ_', 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_$'),
111            new QuotedDomain('`', minimum: 1, maximum: 64),
112            new QuotedDomain('[', minimum: 1, maximum: 64, closing: ']'),
113        );
114        $names = [];
115        foreach (['ID', 'id', 'idj', 'ANY'] as $terminal) {
116            $names[] = new ValueLexemeGenerator($terminal, $name, ['name'], 'identifier', 'src/tokenize.c:CC_ID/CC_QUOTE');
117        }
118        $names[] = new ValueLexemeGenerator('VARIABLE', new ChoiceDomain(
119            new WordDomain(['?']),
120            new SequenceDomain(new WordDomain(['?']), new IntegerDomain('1', '32766', 0)),
121            new SequenceDomain(new WordDomain([':', '@', '$']), new IdentifierDomain('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ_', 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789_$', 'v', 64)),
122        ), ['?1', ':value', '@value', '$value'], 'parameter', 'src/tokenize.c:CC_VARNUM/CC_VARALPHA');
123        return new ChoiceLexemeGenerator(...$names);
124    }

126    /**
127     * CC_DIGIT: decimal, hexadecimal, real and TK_QNUMBER digit separators.
128     * @see https://github.com/sqlite/sqlite/blob/version-3.47.2/src/tokenize.c
129     */
130    public function numbers(): LexemeGenerator
131    {
132        $digits = new CharacterDomain(str_split('0123456789'), 1, 19);
133        $decimal = new ChoiceDomain(new SequenceDomain($digits, new WordDomain(['.']), new CharacterDomain(str_split('0123456789'), 0, 30)), new SequenceDomain(new WordDomain(['.']), $digits));
134        $exponent = new SequenceDomain(new WordDomain(['e', 'E']), new WordDomain(['', '+', '-']), new CharacterDomain(str_split('0123456789'), 1, 3));
135        return new ChoiceLexemeGenerator(
136            new ValueLexemeGenerator('INTEGER', new ChoiceDomain($digits, new SequenceDomain(new WordDomain(['0x', '0X']), new CharacterDomain(str_split('0123456789abcdefABCDEF'), 1, 16))), ['1', '0', '2'], 'number', 'src/tokenize.c:CC_DIGIT'),
137            new ValueLexemeGenerator('number', $digits, ['1'], 'number', 'src/parse.y:number'),
138            new ValueLexemeGenerator('FLOAT', new ChoiceDomain($decimal, new SequenceDomain(new ChoiceDomain($digits, $decimal), $exponent)), ['1.5', '.5', '1e2'], 'number', 'src/tokenize.c:CC_DIGIT:float'),
139            new ValueLexemeGenerator('QNUMBER', new SequenceDomain($digits, new RepeatDomain(new SequenceDomain(new WordDomain(['_']), $digits), 1, 8)), ['1_0'], 'number', 'src/tokenize.c:TK_QNUMBER'),
140        );
141    }

144     * CC_QUOTE and CC_X: doubled quotes and whole hexadecimal bytes.
145     * @see https://github.com/sqlite/sqlite/blob/version-3.47.2/src/tokenize.c
146     */
147    public function strings(): LexemeGenerator
148    {
149        $text = new QuotedDomain("'", alphabet: [...array_map(chr(...), range(1, 127)), 'é', '猫', '😀']);
150        return new ChoiceLexemeGenerator(
151            new ValueLexemeGenerator('STRING', $text, ["'text'", "'a''b'"], 'string', 'src/tokenize.c:CC_QUOTE'),
152            new ValueLexemeGenerator('ids', $text, ["'text'"], 'string', 'src/parse.y:ids'),
153            new ValueLexemeGenerator('BLOB', new SequenceDomain(new WordDomain(['X', 'x']), new CharacterDomain(str_split('0123456789abcdefABCDEF'), 0, 127, "'", "'", 2)), ["X'00'", "X''"], 'string', 'src/tokenize.c:CC_X'),
154        );
155    }
156    /**
157     * Binds the reviewed keyword dispatch to one exact release.
158     * @param array<string, list<string>> $keywords
159     */
160    public function keywordLexemes(string $version, array $keywords): LexemeGenerator
161    {
162        return new VersionedLexemeGenerator($version, new VersionCase(
163            ['sqlite-3.47.2'],
164            new MatchingLexemeGenerator(
165                static fn (LexemeInput $input): bool => isset($keywords[$input->terminal()->name]) && !($input->terminal()->name === 'JOIN_KW' && $input->terminal()->within('joinop')),
166                new WindowNameLexemeGenerator(new RegisteredLexemeGenerator($keywords, 'tool/mkkeywordhash.c', [])),
167            ),
168            'sqlite-3.47.2-keywords',
169        ));
170    }
