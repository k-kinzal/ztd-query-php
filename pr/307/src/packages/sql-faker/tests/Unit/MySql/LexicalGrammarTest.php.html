164    public function testGeneratesEveryStringLiteralStrategy(int $choice, string $expected): void
165    {
166        $faker = new class ($choice) extends \Faker\Generator {
167            private bool $first = true;
168
169            public function __construct(private readonly int $choice)
170            {
171                parent::__construct();
172            }
173
174            /**
175             * @param mixed $min
176             * @param mixed $max
177             *
178             * @throws UnexpectedValueException When the bound is not an integer
179             */
180            #[Override]
181            public function numberBetween($min = 0, $max = 2147483647): int
182            {
183                if ($this->first) {
184                    $this->first = false;
185
186                    return $this->choice;
187                }
188                if (!is_int($min)) {
189                    throw new UnexpectedValueException();
190                }
191
192                return $min;
193            }
194        };
195
196        self::assertSame(
197            $expected,
198            (new LexicalGrammar($faker, 'mysql-8.4.7', true))->realize(['TEXT_STRING']),
199        );
200    }
