100    private static function tokenize(string $value): array
101    {
102        $tokens = [];
103        $length = strlen($value);
104        for ($index = 0; $index < $length;) {
105            $character = $value[$index];
106            $next = $value[$index + 1] ?? '';
107            if (ctype_space($character)) {
108                $index++;
109                continue;
110            }
111            if ($character === '/' && $next === '*') {
112                $index += 2;
113                while ($index < $length && !($value[$index] === '*' && ($value[$index + 1] ?? '') === '/')) {
114                    $index++;
115                }
116                $index = min($length, $index + 2);
117                continue;
118            }
119            if (($character === '-' && $next === '-') || $character === '#') {
120                while ($index < $length && !in_array($value[$index], ["\r", "\n"], true)) {
121                    $index++;
122                }
123                continue;
124            }
125            if (in_array($character, ["'", '"', '`'], true)) {
126                $quote = $character;
127                $index++;
128                while ($index < $length) {
129                    if ($value[$index] === '\\') {
130                        $index += 2;
131                        continue;
132                    }
133                    if ($value[$index] !== $quote) {
134                        $index++;
135                        continue;
136                    }
137                    if (($value[$index + 1] ?? '') === $quote) {
138                        $index += 2;
139                        continue;
140                    }
141                    $index++;
142                    break;
143                }
144                $tokens[] = '__ZTD_VALUE__';
145                continue;
146            }
147            if ($character === '[') {
148                $index++;
149                while ($index < $length && $value[$index] !== ']') {
150                    $index++;
151                }
152                $index = min($length, $index + 1);
153                $tokens[] = '__ZTD_VALUE__';
154                continue;
155            }
156            if (ctype_alnum($character) || in_array($character, ['_', '$', ':', '%', '{'], true)) {
157                $start = $index;
158                do {
159                    $index++;
160                    $current = $value[$index] ?? '';
161                } while ($index < $length
162                    && (ctype_alnum($current) || in_array($current, ['_', '$', ':', '%', '.', '}'], true))
163                );
164                $tokens[] = strtoupper(substr($value, $start, $index - $start));
165                continue;
166            }
167            if (in_array($character . $next, ['<>', '!=', '<=', '>='], true)) {
168                $tokens[] = $character . $next;
169                $index += 2;
170                continue;
171            }
172            $tokens[] = $character;
173            $index++;
174        }
175
176        return $tokens;
177    }
