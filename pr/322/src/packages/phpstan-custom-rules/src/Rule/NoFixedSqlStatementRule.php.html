73            && !$node instanceof InterpolatedString

75            && !$node instanceof Assign

125    private function isApprovedGeneratorReturn(
126        Expr $expr,
127        Scope $scope,
128        array $assignments,
129        array $visited,
130    ): bool {
131        if ($expr instanceof Match_) {
132            foreach ($expr->arms as $arm) {
133                if (!$this->isApprovedGeneratorReturn($arm->body, $scope, $assignments, $visited)) {
134                    return false;
135                }
136            }
137
138            return $expr->arms !== [];
139        }
140        if ($expr instanceof Variable && is_string($expr->name)) {
141            return $this->isGrammarDerivedVariable($expr->name, $assignments, $scope, $visited);
142        }
143        if (!$expr instanceof MethodCall || !$expr->name instanceof Identifier) {
144            return false;
145        }
146        if ($expr->var instanceof Variable && $expr->var->name === 'this') {
147            return str_starts_with($expr->name->name, 'generate');
148        }
149        if (!$expr->var instanceof PropertyFetch
150            || !$expr->var->var instanceof Variable
151            || $expr->var->var->name !== 'this'
152            || !$expr->var->name instanceof Identifier
153            || $expr->var->name->name !== 'lexicalGrammar'
154        ) {
155            return false;
156        }
157
158        if (str_starts_with($expr->name->name, 'generate')) {
159            return true;
160        }
161        $argument = $expr->getArgs()[0]->value ?? null;
162
163        return $expr->name->name === 'realize'
164            && $argument instanceof Expr
165            && $this->isDerivedTerminalExpression($argument, $assignments, $scope, []);
166    }

281    private function collectVariableFlow(Node $node, array &$returns, array &$assignments): void
282    {
283        if ($node instanceof \PhpParser\Node\FunctionLike && !$node instanceof ClassMethod) {
284            return;
285        }
286        if ($node instanceof Return_ && $node->expr !== null) {
287            $returns[] = [$node->expr, $node->getStartLine()];
288        }
289        if (($node instanceof Assign || $node instanceof AssignOp)
290            && $node->var instanceof Variable
291            && is_string($node->var->name)
292        ) {
293            $assignments[$node->var->name][] = $node->expr;
294        }
295        $subNodes = get_object_vars($node);
296        foreach ($node->getSubNodeNames() as $name) {
297            $child = $subNodes[$name] ?? null;
298            if ($child instanceof Node) {
299                $this->collectVariableFlow($child, $returns, $assignments);
300                continue;
301            }
302            if (!is_array($child)) {
303                continue;
304            }
305            foreach ($child as $item) {
306                if ($item instanceof Node) {
307                    $this->collectVariableFlow($item, $returns, $assignments);
308                }
309            }
310        }

328    private function literalValue(Node $node): ?string
329    {
330        if ($node instanceof String_ || $node instanceof InterpolatedStringPart) {
331            return $node->value;
332        }
333        if ($node instanceof InterpolatedString) {
334            return implode('', array_map(
335                static fn (Expr|InterpolatedStringPart $part): string => $part instanceof InterpolatedStringPart
336                    ? $part->value
337                    : '__ztd_dynamic__',
338                $node->parts,
339            ));
340        }
341        if ($node instanceof FuncCall && $node->name instanceof Name) {
342            $function = strtolower($node->name->toString());
343            if (in_array($function, ['sprintf', 'vsprintf'], true)) {
344                return $this->formattedValue($node, $function === 'vsprintf');
345            }
346            if ($function === 'str_replace') {
347                $arguments = $node->getArgs();
348                $search = isset($arguments[0]) ? $this->literalValue($arguments[0]->value) : null;
349                $replace = isset($arguments[1]) ? $this->literalValue($arguments[1]->value) : null;
350                $subject = isset($arguments[2]) ? $this->literalValue($arguments[2]->value) : null;
351
352                return $search === null || $replace === null || $subject === null
353                    ? null
354                    : str_replace($search, $replace, $subject);
355            }
356            if (!in_array($function, ['implode', 'join'], true)) {
357                return null;
358            }
359            $arguments = $node->getArgs();
360            $separatorNode = count($arguments) === 1 ? null : ($arguments[0]->value ?? null);
361            $valuesNode = count($arguments) === 1
362                ? ($arguments[0]->value ?? null)
363                : ($arguments[1]->value ?? null);
364            if (!$valuesNode instanceof Array_) {
365                return null;
366            }
367            $separator = $separatorNode === null ? '' : ($this->literalValue($separatorNode) ?? ' ');
368            $values = [];
369            foreach ($valuesNode->items as $item) {
370                $values[] = $this->literalValue($item->value) ?? '__ztd_dynamic__';
371            }
372
373            return implode($separator, $values);
374        }
375        if ($node instanceof AssignConcat) {
376            $suffix = $this->literalValue($node->expr);
377
378            return $suffix === null ? null : 'SELECT __ztd_dynamic__ ' . $suffix;
379        }
380        if ($node instanceof Assign
381            && $node->var instanceof Variable
382            && is_string($node->var->name)
383            && $node->expr instanceof Concat
384            && $node->expr->left instanceof Variable
385            && $node->expr->left->name === $node->var->name
386        ) {
387            $suffix = $this->literalValue($node->expr->right);
388
389            return $suffix === null ? null : 'SELECT __ztd_dynamic__ ' . $suffix;
390        }
391        if (!$node instanceof Concat) {
392            return null;
393        }
394        $left = $this->literalValue($node->left) ?? '__ztd_dynamic__';
395        $right = $this->literalValue($node->right) ?? '__ztd_dynamic__';
396
397        return $left . $right;
398    }

400    private function formattedValue(FuncCall $call, bool $argumentsInArray): ?string
401    {
402        $arguments = $call->getArgs();
403        $formatNode = $arguments[0]->value ?? null;
404        if (!$formatNode instanceof Node) {
405            return null;
406        }
407        $format = $this->literalValue($formatNode);
408        if ($format === null) {
409            return null;
410        }
411        if ($this->containsSqlTemplate($format)) {
412            return null;
413        }
414        if ($argumentsInArray) {
415            $valuesNode = $arguments[1]->value ?? null;
416            if (!$valuesNode instanceof Array_) {
417                return null;
418            }
419            $valueNodes = array_map(static fn ($item): Node => $item->value, $valuesNode->items);
420        } else {
421            $valueNodes = array_map(static fn ($argument): Node => $argument->value, array_slice($arguments, 1));
422        }
423        $values = array_map(
424            fn (Node $value): string => $this->literalValue($value) ?? '__ztd_dynamic__',
425            $valueNodes,
426        );
427        $index = 0;
428        $formatted = preg_replace_callback(
429            '~%(?:[0-9]+\\$)?[-+0\x20\x27]*(?:[0-9]+|\*)?(?:\.(?:[0-9]+|\*))?[bcdeEfFgGosuxX]~',
430            static function () use (&$index, $values): string {
431                return $values[$index++] ?? '';
432            },
433            $format,
434        );
435
436        return $formatted;
437    }
