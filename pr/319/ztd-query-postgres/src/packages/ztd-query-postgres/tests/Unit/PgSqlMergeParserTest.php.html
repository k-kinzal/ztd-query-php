27    public function testParsesMatchedUpdateAndNotMatchedInsert(): void
28    {
29        $statement = (new PgSqlMergeParser())->parse(
30            'MERGE INTO target AS t USING source AS s ON t.id = s.id '
31            . 'WHEN MATCHED AND s.enabled THEN UPDATE SET name = s.name, value = s.value '
32            . 'WHEN NOT MATCHED THEN INSERT (id, name, value) VALUES (s.id, s.name, $1);',
33        );
34
35        self::assertSame('target', $statement->targetTable);
36        self::assertSame('target', $statement->targetSql);
37        self::assertSame('t', $statement->targetAlias);
38        self::assertSame('source AS s', $statement->sourceSql);
39        self::assertSame('t.id = s.id', $statement->joinConditionSql);
40        self::assertCount(2, $statement->clauses);
41        self::assertSame(PgSqlMergeMatchKind::Matched, $statement->clauses[0]->matchKind);
42        self::assertSame('s.enabled', $statement->clauses[0]->conditionSql);
43        self::assertSame(PgSqlMergeActionKind::Update, $statement->clauses[0]->actionKind);
44        self::assertSame(['name' => 's.name', 'value' => 's.value'], $statement->clauses[0]->assignments);
45        self::assertSame(PgSqlMergeMatchKind::NotMatched, $statement->clauses[1]->matchKind);
46        self::assertSame(PgSqlMergeActionKind::Insert, $statement->clauses[1]->actionKind);
47        self::assertSame(['id', 'name', 'value'], $statement->clauses[1]->insertColumns);
48        self::assertSame(['s.id', 's.name', '$1'], $statement->clauses[1]->insertValues);
49    }
