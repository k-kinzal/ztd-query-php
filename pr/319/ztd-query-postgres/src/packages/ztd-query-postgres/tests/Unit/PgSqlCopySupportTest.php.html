207    public function testDecodeRowSupportsEveryControlUnknownAndVariableLengthByteEscape(): void
208    {
209        $codec = new PgSqlCopySupport();
210
211        self::assertSame(
212            ["\x08\x0C\n\r\t\x0B\\|q"],
213            $codec->decodeRow('\\b\\f\\n\\r\\t\\v\\\\\\|\\q', '|', '\\N'),
214        );
215        self::assertSame(
216            ["\0", "\x07", "\n", 'A4', "\x04", 'O', 'Oa'],
217            $codec->decodeRow('\\0|\\7|\\12|\\1014|\\x4|\\x4F|\\x4Fa', '|', '\\N'),
218        );
219        self::assertSame(['S', 'tail'], $codec->decodeRow('\\1232tail', '2', '\\N'));
220        self::assertSame(
221            ['prefixN', 'octal-A', 'hex-A'],

223        );
224        self::assertSame(

226            $codec->decodeRow('\\377|\\77|\\x0|\\xff', '|', '\\N'),
227        );
228        self::assertSame(["\x01/"], $codec->decodeRow('\\1/', '|', '\\N'));
229        self::assertSame([null], $codec->decodeRow('prefix\\101', '|', 'prefix\\101'));
230        self::assertSame([null], $codec->decodeRow('prefix\\x41', '|', 'prefix\\x41'));
231        self::assertSame(['value', null, 'tail'], $codec->decodeRow('value|NULL|tail', '|', 'NULL'));
