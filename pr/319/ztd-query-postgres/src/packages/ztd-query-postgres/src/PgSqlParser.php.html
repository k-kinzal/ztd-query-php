76    /**
77     * Extract VALUES rows from INSERT statement.
78     *
79     * @return list<list<string>>
80     */
81    public function extractInsertValues(string $sql): array
82    {
83        $sql = $this->maskComments($sql);
84        $rows = [];
85        $source = $this->findInsertSourceClause($sql);
86        if ($source === null) {
87            return [];
88        }
89        if ($source['keyword'] !== 'VALUES') {
90            return [];
91        }
92
93        $rest = substr($sql, $source['offset'] + strlen($source['keyword']));
94        $pos = 0;
95        $len = strlen($rest);
96
97        while ($pos < $len) {
98            while ($pos < $len && ($rest[$pos] === ' ' || $rest[$pos] === "\n" || $rest[$pos] === "\r" || $rest[$pos] === "\t" || $rest[$pos] === ',')) {
99                $pos++;
100            }
101
102            if ($pos >= $len || $rest[$pos] !== '(') {
103                break;
104            }
105
106            $values = $this->extractParenthesizedList($rest, $pos);
107            if ($values === null) {
108                break;
109            }
110            $rows[] = $values['items'];
111            $pos = $values['end'];
112        }
113
114        return $rows;
115    }

126    public function extractOnConflictTarget(string $sql): ?PgSqlConflictTarget
127    {
128        $tokens = SqlTokenStream::tokenize($sql, PgSqlLexerProfile::create())->significantTokens();
129        $conflict = self::findOnConflictTargetStart($tokens);
130        if ($conflict === null) {
131            return null;
132        }
133
134        $next = $tokens[$conflict] ?? null;
135        if ($next === null) {
136            return null;
137        }
138        if ($next->isKeyword('DO')) {
139            return new PgSqlConflictTarget(false);
140        }
141        if ($next->isKeyword('ON')) {
142            $constraintKeyword = $tokens[$conflict + 1] ?? null;
143            if ($constraintKeyword === null) {
144                return null;
145            }
146            if (!$constraintKeyword->isKeyword('CONSTRAINT')) {
147                return null;
148            }
149            $constraintToken = $tokens[$conflict + 2] ?? null;
150            if ($constraintToken === null) {
151                return null;
152            }
153            $identifier = SqlTokenStream::tokenize($constraintToken->text, PgSqlLexerProfile::create())->identifierAt();
154            if ($identifier === null) {
155                return null;
156            }
157
158            return new PgSqlConflictTarget(true, constraint: $identifier['name']);
159        }
160        if ($next->text !== '(') {
161            return null;
162        }
163
164        $closing = self::findTopLevelSymbol($tokens, $conflict, ')');
165        if ($closing === null) {
166            return null;
167        }
168
169        $columnSql = substr(
170            $sql,
171            $next->endOffset(),
172            $tokens[$closing]->offset - $next->endOffset(),
173        );
174        $columns = [];
175        foreach (SqlTokenStream::tokenize($columnSql, PgSqlLexerProfile::create())->splitTopLevel() as $part) {
176            $stream = SqlTokenStream::tokenize($part, PgSqlLexerProfile::create());
177            $identifier = $stream->identifierAt();
178            if ($identifier === null) {
179                return null;
180            }
181            if ($identifier['next'] !== count($stream->significantTokens())) {
182                return null;
183            }
184            $columns[] = $identifier['name'];
185        }
186
187        $do = self::findTopLevelKeyword($tokens, $closing, 'DO');
188        if ($do === null) {
189            return null;
190        }
191
192        $predicate = null;
193        $afterColumns = $tokens[$closing + 1] ?? null;
194        if ($afterColumns !== null) {
195            if ($afterColumns->isKeyword('WHERE')) {
196                $predicate = trim(substr(
197                    $sql,
198                    $afterColumns->endOffset(),
199                    $tokens[$do]->offset - $afterColumns->endOffset(),
200                ));
201                if ($predicate === '') {
202                    return null;
203                }
204            }
205        }
206
207        return new PgSqlConflictTarget(true, $columns, $predicate);
208    }

213    private static function findOnConflictTargetStart(array $tokens): ?int
214    {
215        foreach ($tokens as $index => $token) {
216            if (!$token->isTopLevel()) {
217                continue;
218            }
219            if (!$token->isKeyword('ON')) {
220                continue;
221            }
222            $following = $tokens[$index + 1] ?? $token;
223            if ($following->isKeyword('CONFLICT')) {
224                return $index + 2;
225            }
226        }
227
228        return null;

235    {

462    /** @return list<string> */
463    public function extractTruncateTables(string $sql): array
464    {
465        $stream = SqlTokenStream::tokenize($sql, PgSqlLexerProfile::create());
466        $tokens = $stream->significantTokens();
467        if ($stream->firstTopLevelKeyword() !== 'TRUNCATE') {
468            return [];
469        }
470
471        $index = 1;
472        $tableKeyword = $tokens[$index] ?? null;
473        if ($tableKeyword !== null && $tableKeyword->isKeyword('TABLE')) {
474            $index++;
475        }
476
477        $tableNames = [];
478        while (isset($tokens[$index])) {
479            if ($tokens[$index]->isKeyword('ONLY')) {
480                $index++;
481            }
482
483            $identifier = $this->truncateIdentifierAt($stream, $index);
484            if ($identifier === null) {
485                break;
486            }
487            $tableName = $identifier['name'];
488            $index = $identifier['next'];
489
490            while (($tokens[$index] ?? null)?->text === '.') {
491                $identifier = $this->truncateIdentifierAt($stream, $index + 1);
492                if ($identifier === null) {
493                    break 2;
494                }
495                $tableName = $identifier['name'];
496                $index = $identifier['next'];
497            }
498
499            if (($tokens[$index] ?? null)?->text === '*') {
500                $index++;
501            }
502            $tableNames[] = $tableName;
503
504            if (($tokens[$index] ?? null)?->text !== ',') {
505                break;
506            }
507            $index++;
508        }
509
510        return $tableNames;
511    }

513    /** @return array{name: string, next: int}|null */
514    private function truncateIdentifierAt(SqlTokenStream $stream, int $index): ?array
515    {
516        $tokens = $stream->significantTokens();
517        $prefix = $tokens[$index] ?? null;
518        if ($prefix === null) {
519            return null;
520        }
521        if ($prefix->isKeyword('U')) {
522            $ampersand = $tokens[$index + 1] ?? null;
523            if ($ampersand !== null && $ampersand->text === '&') {
524                $quotedIdentifier = $tokens[$index + 2] ?? null;
525                if ($quotedIdentifier === null || $quotedIdentifier->kind !== SqlTokenKind::QuotedIdentifier) {
526                    return null;
527                }
528
529                return [
530                    'name' => $this->unquoteIdentifier($quotedIdentifier->text),
531                    'next' => $index + 3,
532                ];
533            }
534        }
535
536        return $stream->identifierAt($index);
537    }

680    /** @return 'SELECT'|'INSERT'|'UPDATE'|'DELETE'|'MERGE'|null */
681    private function classifyWithStatement(string $sql): ?string
682    {
683        $stripped = $this->stripStringLiterals($sql);
684        $len = strlen($stripped);
685        $depth = 0;
686        $seenCteBody = false;
687        $inQuote = false;
688
689        for ($i = 0; $i < $len; $i++) {
690            $char = $stripped[$i];
691
692            if ($inQuote) {
693                if ($char === '"') {
694                    $inQuote = false;
695                }
696                continue;
697            }
698
699            if ($char === '"') {
700                $inQuote = true;
701                continue;
702            }
703
704            if ($char === '(') {
705                $depth++;
706                $seenCteBody = true;
707                continue;
708            }
709
710            if ($char === ')') {
711                if ($depth > 0) {
712                    $depth--;
713                }
714                continue;
715            }
716
717            if (!$seenCteBody || $depth !== 0 || !ctype_alpha($char)) {
718                continue;
719            }
720
721            $prev = $i > 0 ? $stripped[$i - 1] : ' ';
722            if (ctype_alpha($prev) || $prev === '_') {
723                continue;
724            }
725
726            $j = $i;
727            while ($j < $len && (ctype_alpha($stripped[$j]) || $stripped[$j] === '_')) {
728                $j++;
729            }
730
731            $keyword = strtoupper(substr($stripped, $i, $j - $i));
732
733            $result = match ($keyword) {
734                'SELECT' => 'SELECT',
735                'INSERT' => 'INSERT',
736                'UPDATE' => 'UPDATE',
737                'DELETE' => 'DELETE',
738                'MERGE' => 'MERGE',
739                default => null,
740            };
741
742            if ($result !== null) {
743                return $result;
744            }
745
746            $i = $j - 1;
747        }
748
749        return null;
750    }

816    private function extractParenthesizedList(string $str, int $start): ?array
817    {
818        if (!isset($str[$start]) || $str[$start] !== '(') {
819            return null;
820        }
821
822        $items = [];
823        $current = '';
824        $depth = 0;
825        $bracketDepth = 0;
826        $len = strlen($str);
827        $inSingleQuote = false;
828
829        for ($pos = $start; $pos < $len; $pos++) {
830            $char = $str[$pos];
831
832            if ($inSingleQuote) {
833                $current .= $char;
834                if ($char === "'" && isset($str[$pos + 1]) && $str[$pos + 1] === "'") {
835                    $current .= "'";
836                    $pos++;
837                } elseif ($char === "'") {
838                    $inSingleQuote = false;
839                }
840                continue;
841            }
842
843            if ($char === "'") {
844                $current .= $char;
845                $inSingleQuote = true;
846                continue;
847            }
848
849            if ($char === '(') {
850                $depth++;
851                if ($depth === 1) {
852                    continue;
853                }
854                $current .= $char;
855                continue;
856            }
857
858            if ($char === ')') {
859                $depth--;
860                if ($depth === 0) {
861                    $val = trim($current);
862                    if ($val !== '') {
863                        $items[] = $val;
864                    }
865
866                    return ['items' => $items, 'end' => $pos + 1];
867                }
868                $current .= $char;
869                continue;
870            }
871
872            if ($char === '[') {
873                $bracketDepth++;
874                $current .= $char;
875                continue;
876            }
877
878            if ($char === ']' && $bracketDepth > 0) {
879                $bracketDepth--;
880                $current .= $char;
881                continue;
882            }
883
884            if ($char === ',' && $depth === 1 && $bracketDepth === 0) {
885                $items[] = trim($current);
886                $current = '';
887                continue;
888            }

890            $current .= $char;
891        }

909    private function findInsertSourceClause(string $sql): ?array
910    {
911        $searchable = $this->stripStringLiterals($sql);
912        $length = strlen($searchable);
913        $depth = 0;
914        $foundInsert = false;
915
916        for ($i = 0; $i < $length; $i++) {
917            if ($searchable[$i] === '"') {
918                for ($i++; $i < $length; $i++) {
919                    if ($searchable[$i] !== '"') {
920                        continue;
921                    }
922                    break;
923                }
924                continue;
925            }
926
927            if ($searchable[$i] === '(') {
928                $depth++;
929                continue;
930            }
931
932            if ($searchable[$i] === ')') {
933                if ($depth > 0) {
934                    $depth--;
935                }
936                continue;
937            }
938
939            $tokenLength = strspn(
940                $searchable,
941                'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789_$',
942                $i,
943            );
944            if ($tokenLength === 0) {
945                continue;
946            }
947
948            if ($depth === 0 && ctype_alpha($searchable[$i])) {
949                $keyword = strtoupper(substr($searchable, $i, $tokenLength));
950                if (!$foundInsert) {
951                    $foundInsert = $keyword === 'INSERT';
952                } elseif ($keyword === 'VALUES' || $keyword === 'SELECT') {
953                    return ['keyword' => $keyword, 'offset' => $i];
