132                $expectSource = true;

137            }

154                    || $innerTokens[0]->isKeyword('WITH')
155                    || $innerTokens[0]->isKeyword('VALUES')
156                ) {
157                    continue;
158                }
159                foreach ($this->referencesFromClause($inner) as $reference) {

198    {
199        $token = $tokens[$index];

201            return null;
202        }

205        if ($component === null) {
206            return null;
207        }

211            && $tokens[$nextIndex]->text === '.'
212        ) {
213            $component = $this->identifierComponentAt($tokens, $nextIndex + 1);
214            if ($component === null) {

260    /** @param list<SqlToken> $tokens */
261    private function findFromEnd(string $sql, array $tokens, SqlToken $fromToken): int
262    {
263        $terminators = [
264            ['WHERE'], ['GROUP', 'BY'], ['HAVING'], ['WINDOW'], ['ORDER', 'BY'],
265            ['LIMIT'], ['OFFSET'], ['FETCH'], ['FOR'], ['RETURNING'],
266            ['UNION'], ['INTERSECT'], ['EXCEPT'],
267        ];
268        $afterFrom = false;
269        foreach ($tokens as $index => $token) {
270            if (!$afterFrom) {
271                $afterFrom = $token === $fromToken;
272                continue;
273            }
274            if ($token->depth < $fromToken->depth || $token->bracketDepth < $fromToken->bracketDepth) {
275                return $token->offset;
276            }
277            if ($token->depth !== $fromToken->depth || $token->bracketDepth !== $fromToken->bracketDepth) {
278                continue;
279            }
280            foreach ($terminators as $sequence) {
281                if ($this->matchesKeywordSequence($tokens, $index, $sequence)) {
282                    return $token->offset;
283                }
284            }
285        }
286
287        return strlen($sql);
288    }

294    private function matchesKeywordSequence(array $tokens, int $index, array $keywords): bool
295    {
296        foreach ($keywords as $relative => $keyword) {
