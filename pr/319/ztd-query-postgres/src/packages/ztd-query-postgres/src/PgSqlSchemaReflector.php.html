42    public function reflectAll(): array
43    {
44        $stmt = $this->connection->query(
45            "SELECT table_name FROM information_schema.tables "
46            . "WHERE table_schema = current_schema() AND table_type = 'BASE TABLE' "
47            . "ORDER BY table_name"
48        );
49        if ($stmt === false) {
50            return [];
51        }
52
53        $tables = $stmt->fetchAll();
54        $result = [];
55
56        foreach ($tables as $row) {
57            $tableName = $row['table_name'] ?? null;
58            if (!is_string($tableName) || $tableName === '') {
59                continue;
60            }
61
62            $createSql = $this->buildCreateTableSql($tableName);
63            if ($createSql !== null) {
64                $result[$tableName] = $createSql;
65            }
66        }
67
68        return $result;
69    }

79    {
80        $stmt = $this->connection->query(
81            'SELECT viewname, definition FROM pg_views WHERE schemaname = current_schema() ORDER BY viewname',
82        );
83        if ($stmt === false) {
84            return [];
85        }
86
87        $definitions = [];
88        foreach ($stmt->fetchAll() as $row) {
89            $viewName = $row['viewname'] ?? null;
90            $query = $row['definition'] ?? null;
91            if (!is_string($viewName) || $viewName === '' || !is_string($query) || trim($query) === '') {
92                continue;
93            }
94            $definitions[$viewName] = (new PgSqlViewDefinitionParser())->fromQuery($query);
95        }
96
97        return $definitions;

100    private function buildCreateTableSql(string $tableName): ?string
101    {
102        $escapedTableName = str_replace("'", "''", $tableName);
103        $stmt = $this->connection->query(
104            "SELECT column_name, data_type, character_maximum_length, "
105            . "numeric_precision, numeric_scale, is_nullable, column_default, "
106            . "udt_name, domain_schema, domain_name, is_identity, identity_generation, "
107            . "is_generated, generation_expression "
108            . "FROM information_schema.columns "
109            . "WHERE table_schema = current_schema() AND table_name = '" . $escapedTableName . "' "
110            . "ORDER BY ordinal_position"
111        );
112        if ($stmt === false) {
113            return null;
114        }
115
116        $columns = $stmt->fetchAll();
117        if ($columns === []) {
118            return null;
119        }
120
121        $columnDefs = [];
122        foreach ($columns as $col) {
123            $columnDefs[] = $this->buildColumnDefinition($col);
124        }
125
126        $pkStmt = $this->connection->query(
127            "SELECT kcu.column_name "
128            . "FROM information_schema.table_constraints tc "
129            . "JOIN information_schema.key_column_usage kcu "
130            . "  ON tc.constraint_name = kcu.constraint_name "
131            . "  AND tc.table_schema = kcu.table_schema "
132            . "WHERE tc.table_schema = current_schema() "
133            . "  AND tc.table_name = '" . $escapedTableName . "' "
134            . "  AND tc.constraint_type = 'PRIMARY KEY' "
135            . "ORDER BY kcu.ordinal_position"
136        );
137
138        $primaryKeyCols = [];
139        if ($pkStmt !== false) {
140            $pkRows = $pkStmt->fetchAll();
141            foreach ($pkRows as $pkRow) {
142                $colName = $pkRow['column_name'] ?? null;
143                if (is_string($colName)) {
144                    $primaryKeyCols[] = '"' . $colName . '"';
145                }
146            }
147        }
148
149        $uniqueStmt = $this->connection->query(
150            'SELECT index_relation.relname AS constraint_name, attribute.attname AS column_name, '
151            . 'pg_get_expr(index_metadata.indpred, index_metadata.indrelid) AS predicate '
152            . 'FROM pg_catalog.pg_class table_relation '
153            . 'JOIN pg_catalog.pg_namespace namespace ON namespace.oid = table_relation.relnamespace '
154            . 'JOIN pg_catalog.pg_index index_metadata ON index_metadata.indrelid = table_relation.oid '
155            . 'JOIN pg_catalog.pg_class index_relation ON index_relation.oid = index_metadata.indexrelid '
156            . 'JOIN LATERAL unnest(index_metadata.indkey) WITH ORDINALITY key_column(attnum, ordinality) '
157            . '  ON key_column.ordinality <= index_metadata.indnkeyatts '
158            . 'LEFT JOIN pg_catalog.pg_attribute attribute '
159            . '  ON attribute.attrelid = table_relation.oid AND attribute.attnum = key_column.attnum '
160            . 'WHERE namespace.nspname = current_schema() '
161            . "  AND table_relation.relname = '" . $escapedTableName . "' "
162            . '  AND index_metadata.indisunique '
163            . '  AND index_metadata.indisvalid '
164            . '  AND NOT index_metadata.indisprimary '
165            . 'ORDER BY index_relation.relname, key_column.ordinality'
166        );
167
168        /** @var array<string, list<string>> $uniqueConstraints */
169        $uniqueConstraints = [];
170        /** @var array<string, array{columns: list<string>, predicate: string}> $partialIndexes */
171        $partialIndexes = [];
172        /** @var array<string, string> $invalidIndexes */
173        $invalidIndexes = [];
174        if ($uniqueStmt !== false) {
175            $uniqueRows = $uniqueStmt->fetchAll();
176            foreach ($uniqueRows as $uRow) {
177                $constraintName = $uRow['constraint_name'] ?? '';
178                $colName = $uRow['column_name'] ?? null;
179                $predicate = $uRow['predicate'] ?? null;
180                if (!is_string($constraintName)) {
181                    continue;
182                }
183                if ($constraintName === '') {
184                    continue;
185                }
186                if (!is_string($colName)) {
187                    $invalidIndexes[$constraintName] = $constraintName;
188                    continue;
189                }
190                if (!is_string($predicate) || trim($predicate) === '') {
191                    $uniqueConstraints[$constraintName][] = '"' . $colName . '"';
192
193                    continue;
194                }
195                if (!isset($partialIndexes[$constraintName])) {
196                    $partialIndexes[$constraintName] = ['columns' => [$colName], 'predicate' => $predicate];
197
198                    continue;
199                }
200                $partialIndexes[$constraintName]['columns'][] = $colName;
201            }
202        }
203        foreach ($invalidIndexes as $invalidIndex) {
204            unset($uniqueConstraints[$invalidIndex], $partialIndexes[$invalidIndex]);
205        }
206        $this->partialUniqueIndexes[$tableName] = [];
207        foreach ($partialIndexes as $name => $index) {
208            if ($index['columns'] === []) {
209                continue;
210            }
211            $this->partialUniqueIndexes[$tableName][$name] = new PartialUniqueIndex(
212                $name,
213                $index['columns'],
214                $index['predicate'],
215            );
216        }
217
218        $foreignKeyStmt = $this->connection->query(
219            "SELECT fk.constraint_name, fk.column_name, "
220            . "pk.table_name AS foreign_table_name, pk.column_name AS foreign_column_name, "
221            . "rc.update_rule, rc.delete_rule "
222            . "FROM information_schema.referential_constraints rc "
223            . "JOIN information_schema.key_column_usage fk "
224            . "  ON fk.constraint_catalog = rc.constraint_catalog "
225            . "  AND fk.constraint_schema = rc.constraint_schema "
226            . "  AND fk.constraint_name = rc.constraint_name "
227            . "JOIN information_schema.key_column_usage pk "
228            . "  ON pk.constraint_catalog = rc.unique_constraint_catalog "
229            . "  AND pk.constraint_schema = rc.unique_constraint_schema "
230            . "  AND pk.constraint_name = rc.unique_constraint_name "
231            . "  AND pk.ordinal_position = fk.position_in_unique_constraint "
232            . "WHERE fk.table_schema = current_schema() "
233            . "  AND fk.table_name = '" . $escapedTableName . "' "
234            . "ORDER BY fk.constraint_name, fk.ordinal_position"
235        );
236
237        /** @var array<string, array{columns: list<string>, table: string, referencedColumns: list<string>, onUpdate: string, onDelete: string}> $foreignKeys */
238        $foreignKeys = [];
239        if ($foreignKeyStmt !== false) {
240            foreach ($foreignKeyStmt->fetchAll() as $foreignKeyRow) {
241                $constraintName = $foreignKeyRow['constraint_name'] ?? null;
242                $columnName = $foreignKeyRow['column_name'] ?? null;
243                $foreignTable = $foreignKeyRow['foreign_table_name'] ?? null;
244                $foreignColumn = $foreignKeyRow['foreign_column_name'] ?? null;
245                $onUpdate = $foreignKeyRow['update_rule'] ?? null;
246                $onDelete = $foreignKeyRow['delete_rule'] ?? null;
247                if (!is_string($constraintName)
248                    || !is_string($columnName)
249                    || !is_string($foreignTable)
250                    || !is_string($foreignColumn)
251                    || !is_string($onUpdate)
252                    || !is_string($onDelete)
253                ) {
254                    continue;
255                }
256                if (!isset($foreignKeys[$constraintName])) {
257                    $foreignKeys[$constraintName] = [
258                        'columns' => ['"' . $columnName . '"'],
259                        'table' => $foreignTable,
260                        'referencedColumns' => ['"' . $foreignColumn . '"'],
261                        'onUpdate' => $onUpdate,
262                        'onDelete' => $onDelete,
263                    ];
264
265                    continue;
266                }
267                $foreignKeys[$constraintName]['columns'][] = '"' . $columnName . '"';
268                $foreignKeys[$constraintName]['referencedColumns'][] = '"' . $foreignColumn . '"';
269            }
270        }
271
272        $parts = $columnDefs;
273
274        if ($primaryKeyCols !== []) {
275            $parts[] = 'PRIMARY KEY (' . implode(', ', $primaryKeyCols) . ')';
276        }
277
278        foreach ($uniqueConstraints as $constraintName => $constraintCols) {
279            $parts[] = 'CONSTRAINT "' . $constraintName . '" UNIQUE (' . implode(', ', $constraintCols) . ')';
280        }
281
282        foreach ($foreignKeys as $constraintName => $foreignKey) {
283            $parts[] = 'CONSTRAINT "' . $constraintName . '" FOREIGN KEY ('
284                . implode(', ', $foreignKey['columns']) . ') REFERENCES "'
285                . $foreignKey['table'] . '" (' . implode(', ', $foreignKey['referencedColumns']) . ')'
286                . ' ON UPDATE ' . $foreignKey['onUpdate']
287                . ' ON DELETE ' . $foreignKey['onDelete'];
288        }
289
290        return 'CREATE TABLE "' . $tableName . '" (' . "\n  " . implode(",\n  ", $parts) . "\n)";
291    }

296    private function buildColumnDefinition(array $col): string
297    {
298        $columnName = isset($col['column_name']) && is_string($col['column_name']) ? $col['column_name'] : '';
299        $name = '"' . $columnName . '"';
300        $dataType = strtoupper(isset($col['data_type']) && is_string($col['data_type']) ? $col['data_type'] : 'TEXT');
301        $udtName = strtoupper(isset($col['udt_name']) && is_string($col['udt_name']) ? $col['udt_name'] : '');
302
303        $typeSql = $this->domainTypeSql($col) ?? $this->buildTypeSql($dataType, $udtName, $col);
304
305        $def = "$name $typeSql";
306
307        $isNullable = $col['is_nullable'] ?? 'YES';
308        if ($isNullable === 'NO') {
309            $def .= ' NOT NULL';
310        }
311
312        $generationExpression = $col['generation_expression'] ?? null;
313        if (($col['is_generated'] ?? 'NEVER') === 'ALWAYS'
314            && is_string($generationExpression)
315            && trim($generationExpression) !== ''
316        ) {
317            $def .= ' GENERATED ALWAYS AS (' . $generationExpression . ') STORED';
318        } elseif (($col['is_identity'] ?? 'NO') === 'YES') {
319            $identityGeneration = ($col['identity_generation'] ?? 'BY DEFAULT') === 'ALWAYS'
320                ? 'ALWAYS'
321                : 'BY DEFAULT';
322            $def .= " GENERATED $identityGeneration AS IDENTITY";
323        } else {
324            $default = $col['column_default'] ?? null;
325            if (is_string($default) && $default !== '') {
326                $def .= ' DEFAULT ' . $default;
327            }
328        }
329
330        return $def;
331    }

336    private function domainTypeSql(array $col): ?string
337    {
338        $domainName = $col['domain_name'] ?? null;
339        if (!is_string($domainName)) {
340            return null;
341        }
342        if ($domainName === '') {
343            return null;
344        }
345
346        $quoter = new PgSqlIdentifierQuoter();
347        $domainSchema = $col['domain_schema'] ?? null;
348        if (!is_string($domainSchema)) {
349            return $quoter->quote($domainName);
350        }
351        if ($domainSchema === '') {
352            return $quoter->quote($domainName);
353        }
354
355        return $quoter->quote($domainSchema) . '.' . $quoter->quote($domainName);
356    }

361    private function buildTypeSql(string $dataType, string $udtName, array $col): string
362    {
363        return match ($dataType) {
364            'CHARACTER VARYING' => $this->buildVarcharType($col),
365            'CHARACTER' => $this->buildCharType($col),
366            'BIT', 'BIT VARYING' => $this->buildBitType($dataType, $col),
367            'NUMERIC' => $this->buildNumericType($col),
368            'TIMESTAMP WITHOUT TIME ZONE' => 'TIMESTAMP',
369            'TIMESTAMP WITH TIME ZONE' => 'TIMESTAMPTZ',
370            'TIME WITHOUT TIME ZONE' => 'TIME',
371            'TIME WITH TIME ZONE' => 'TIMETZ',
372            'USER-DEFINED' => $this->resolveUserDefinedType($udtName),
373            'ARRAY' => $this->resolveArrayType($udtName),
374            default => $dataType,
375        };
376    }

420    private function buildNumericType(array $col): string
421    {
422        $precision = $col['numeric_precision'] ?? null;
423        $scale = $col['numeric_scale'] ?? null;
424
425        if ($precision !== null && $scale !== null) {
426            $p = is_int($precision) ? $precision : (is_string($precision) ? (int) $precision : 0);
427            $s = is_int($scale) ? $scale : (is_string($scale) ? (int) $scale : 0);
428
429            return "NUMERIC($p,$s)";
430        }
431        if ($precision !== null) {
432            $p = is_int($precision) ? $precision : (is_string($precision) ? (int) $precision : 0);
433
434            return "NUMERIC($p)";
435        }
436
437        return 'NUMERIC';
