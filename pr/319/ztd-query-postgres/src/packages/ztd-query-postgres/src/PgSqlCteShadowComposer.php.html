145    private function parseHeader(string $sql): array
146    {
147        $tokens = [];
148        foreach (SqlTokenStream::tokenize($sql, PgSqlLexerProfile::create())->significantTokens() as $token) {
149            if ($token->isTopLevel()) {

154            return ['names' => [], 'statementOffset' => null];
155        }
156
157        $index = 1;
158        if (($tokens[$index] ?? null)?->isKeyword('RECURSIVE') === true) {
159            $index++;
160        }
161
162        $names = [];
163        while (isset($tokens[$index])) {

165            if ($name === null) {

167            }
168            $index++;

171            $index = ($asIndex ?? count($tokens)) + 1;

173            if (($tokens[$index] ?? null)?->isKeyword('NOT') === true) {
174                $index++;
175            }
176            if (($tokens[$index] ?? null)?->isKeyword('MATERIALIZED') === true) {
177                $index++;
178            }
179
180            if (!$this->isSymbol($tokens[$index] ?? null, '(')
181                || !$this->isSymbol($tokens[$index + 1] ?? null, ')')
182            ) {
183                return ['names' => $names, 'statementOffset' => null];
184            }
185            $names[] = strtolower($name);
186            $index += 2;

189            if (!$this->isSymbol($separator, ',')) {

191            }

193        }

197        return [

199            'statementOffset' => $statement?->offset,

252    private function identifierName(SqlToken $token): ?string
253    {
254        if ($token->kind === SqlTokenKind::Word) {
255            return $token->text;
256        }
257        if ($token->kind !== SqlTokenKind::QuotedIdentifier || strlen($token->text) < 2) {
258            return null;
259        }
260

262        $inner = substr($token->text, 1, -1);

264        return str_replace($quote . $quote, $quote, $inner);
