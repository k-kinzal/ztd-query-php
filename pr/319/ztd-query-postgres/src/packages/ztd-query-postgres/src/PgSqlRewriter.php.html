72    {
73        $sql = trim($sql);
74        if ($sql === '') {
75            throw new UnsupportedSqlException($sql, 'Empty or unparseable');
76        }
77
78        $statements = $this->splitStatements($sql);
79        if ($statements === []) {
80            throw new UnsupportedSqlException($sql, 'Empty or unparseable');
81        }
82
83        if (count($statements) > 1) {

85        }
86
87        return $this->rewriteStatement($statements[0]);

127    private function rewriteStatement(string $sql): RewritePlan
128    {
129        if (PgSqlReadOnlyDiagnosticStatement::isSafe($sql)) {
130            return new RewritePlan($sql, QueryKind::READ);
131        }
132        $kind = $this->guard->classify($sql);
133        if ($kind === null) {
134            throw new UnsupportedSqlException($sql, 'Statement type not supported');
135        }
136
137        if ($kind === QueryKind::SKIPPED) {
138            return new RewritePlan($sql, QueryKind::SKIPPED);
139        }
140
141        $statementType = $this->parser->classifyStatement($sql);
142        if ($statementType === 'DO') {
143            return new RewritePlan($sql, QueryKind::READ);
144        }
145        $tableContext = $this->buildTableContext();
146
147        if ($kind === QueryKind::READ) {
148            if ($this->hasSchemaContext()) {
149                $tableNames = $this->parser->extractSelectTableNames($sql);
150                $declaredCtes = array_fill_keys($this->cteComposer->declaredCteNames($sql), true);
151                foreach ($tableNames as $tableName) {
152                    if (isset($declaredCtes[strtolower($tableName)])) {
153                        continue;
154                    }
155                    if (!$this->tableExists($tableName)) {
156                        throw new UnknownSchemaException($sql, $tableName, 'table');
157                    }
158                }
159            }
160
161            $transformedSql = $this->transformer->transform($sql, $tableContext);
162
163            return new RewritePlan($transformedSql, QueryKind::READ);
164        }
165
166        if ($kind === QueryKind::DDL_SIMULATED) {
167            $mutation = $this->mutationResolver->resolve($sql, $statementType ?? '', $kind);
168
169            if ($statementType === 'CREATE_TABLE' && $this->parser->hasCreateTableAsSelect($sql)) {
170                $selectSql = $this->parser->extractCreateTableSelectSql($sql);
171                if ($selectSql !== null) {
172                    $transformedSelectSql = $this->transformer->transform($selectSql, $tableContext);
173
174                    return new RewritePlan($transformedSelectSql, QueryKind::DDL_SIMULATED, $mutation);
175                }
176            }
177
178            return new RewritePlan($this->emptyResultSelect(), QueryKind::DDL_SIMULATED, $mutation);
179        }
180
181        $mutation = $this->mutationResolver->resolve($sql, $statementType ?? '', $kind);
182
183        if ($statementType === 'TRUNCATE') {
184            return new RewritePlan($this->emptyResultSelect(), QueryKind::WRITE_SIMULATED, $mutation);
185        }
186
187        $transformedSql = $this->transformer->transform($sql, $tableContext);
188
189        return new RewritePlan(
190            $transformedSql,
191            QueryKind::WRITE_SIMULATED,
192            $mutation,
193            $this->returningProjectionParser->parse($sql),
194            $statementType === 'MERGE' ? AffectedRowsMode::Changed : AffectedRowsMode::Matched,
195        );
196    }

198    /**
199     * Build the table context map for transformers.
200     *
201     * @return array<string, array{viewSql: string}|array{
202     *     rows: array<int, array<string, mixed>>,
203     *     columns: array<int, string>,
204     *     columnTypes: array<string, \ZtdQuery\Schema\ColumnType>,
205     *     columnDefaults: array<string, string>,
206     *     identityStrategies: array<string, \ZtdQuery\Schema\IdentityGenerationStrategy>,
207     *     generatedExpressions: array<string, string>,
208     *     sourceSql?: string,
209     *     storageTable?: string
210     * }>
211     */
212    private function buildTableContext(): array
213    {
214        $context = [];
215        $allData = $this->shadowStore->getAll();
216
217        foreach ($allData as $tableName => $rows) {
218            $definition = $this->registry->get($tableName);
219            $columns = $definition?->columns;
220            if ($columns === null && $rows !== []) {
221                $columns = array_keys($rows[0]);
222                foreach ($rows as $row) {
223                    foreach (array_keys($row) as $column) {
224                        if (!in_array($column, $columns, true)) {
225                            $columns[] = $column;
226                        }
227                    }
228                }
229            }
230
231            $columnTypes = $definition !== null ? $definition->typedColumns : [];
232            $columnDefaults = $definition !== null ? $definition->columnDefaults : [];
233            $identityStrategies = $definition !== null ? $definition->identityStrategies : [];
234            $generatedExpressions = $definition !== null ? $definition->generatedExpressions : [];
235
236            $context[$tableName] = [
237                'rows' => $rows,
238                'columns' => $columns ?? [],
239                'columnTypes' => $columnTypes,
240                'columnDefaults' => $columnDefaults,
241                'identityStrategies' => $identityStrategies,
242                'generatedExpressions' => $generatedExpressions,
243                'primaryKeys' => $definition !== null ? $definition->primaryKeys : [],
244                'candidateKeys' => $definition !== null ? $definition->candidateKeys()->keys() : [],
245                'partialUniqueIndexes' => $definition !== null ? $definition->partialUniqueIndexes : [],
246            ];
247        }
248
249        $allDefinitions = $this->registry->getAll();
250        foreach ($allDefinitions as $tableName => $definition) {
251            if (isset($context[$tableName])) {
252                continue;
253            }
254
255            $definitionContext = [
256                'rows' => [],
257                'columns' => $definition->columns,
258                'columnTypes' => $definition->typedColumns,
259                'columnDefaults' => $definition->columnDefaults,
260                'identityStrategies' => $definition->identityStrategies,
261                'generatedExpressions' => $definition->generatedExpressions,
262                'primaryKeys' => $definition->primaryKeys,
263                'candidateKeys' => $definition->candidateKeys()->keys(),
264            ];
265            $definitionContext['partialUniqueIndexes'] = $definition->partialUniqueIndexes;
266            $context[$tableName] = $definitionContext;
267        }
268
269        $quoter = new PgSqlIdentifierQuoter();
270        foreach ($allDefinitions as $tableName => $definition) {
271            $relation = $definition->partitionRelation;
272            if ($relation === null) {
273                continue;
274            }
275
276            $siblingPredicates = [];
277            foreach ($allDefinitions as $siblingDefinition) {
278                $sibling = $siblingDefinition->partitionRelation;
279                if ($sibling !== null
280                    && strcasecmp($sibling->parentTable, $relation->parentTable) === 0
281                    && $sibling->predicate !== null
282                ) {
283                    $siblingPredicates[] = $sibling->predicate;
284                }
285            }
286            $predicate = $this->partitionPredicateRenderer->render($relation, $siblingPredicates);
287            $storageTable = $this->storageTable($tableName);
288            $partitionContext = $context[$tableName] ?? null;
289            if ($partitionContext === null) {
290                continue;
291            }
292            $partitionContext['rows'] = $this->shadowStore->get($storageTable);
293            $partitionContext['storageTable'] = $storageTable;
294            $partitionContext['sourceSql'] = 'SELECT * FROM '
295                . $quoter->quote($relation->parentTable)
296                . " WHERE $predicate";
297            $context[$tableName] = $partitionContext;
298        }
299
300        foreach ((new PgSqlViewShadowRenderer())->render($this->views, array_keys($context)) as $viewName => $viewSql) {
301            if (isset($context[$viewName])) {
302                continue;
303            }
304            $context[$viewName] = ['viewSql' => $viewSql];
305        }
306
307        return $context;
308    }

310    private function storageTable(string $tableName): string
311    {
312        $seen = [];
313        while (!in_array($tableName, $seen, true)) {
314            $seen[] = $tableName;
315            $parent = $this->registry->get($tableName)?->partitionRelation?->parentTable;
316            if ($parent === null) {
317                return $tableName;
318            }
319            $tableName = $parent;
320        }
321
322        return $tableName;
323    }
