44    {
45        $statement = $this->parser->parse($sql);

49        }
50        if (isset($table['viewSql'])) {
51            throw new UnsupportedSqlException($sql, 'Cannot resolve MERGE target schema');
52        }
53
54        $columns = $table['columns'];
55        if (!array_is_list($columns)) {
56            throw new UnsupportedSqlException($sql, 'MERGE target columns must preserve declaration order');
57        }
58        if ($columns === []) {
59            throw new UnsupportedSqlException($sql, 'Cannot determine MERGE target columns');
60        }
61        if (($table['storageTable'] ?? $statement->targetTable) !== $statement->targetTable) {
62            throw new UnsupportedSqlException($sql, 'MERGE into a child partition is not supported');
63        }
64
65        $defaults = $table['columnDefaults'] ?? [];
66        $identityStrategies = $table['identityStrategies'] ?? [];
67        $existingRows = $table['rows'];
68        $generatedExpressions = $table['generatedExpressions'] ?? [];
67        $effectiveConditions = $this->effectiveConditions($statement);
69        $effectiveConditions = (new Merge\MatchConditions())->effectiveConditions($statement);
70        $parts = [];
69        $parts[] = $this->unchangedRows($statement, $columns, $effectiveConditions);
71        $parts[] = (new Merge\RowActions($this->insertSelectRenderer, $this->quoter, $this->rowRenderer))->unchangedRows($statement, $columns, $effectiveConditions);
72
71        foreach ($statement->clauses as $index => $clause) {
72            $effective = $effectiveConditions[$index];
73            if ($clause->actionKind === PgSqlMergeActionKind::Update) {
74                $parts[] = $this->updatedRows($sql, $statement, $clause, $columns, $defaults, $effective);

76            if ($clause->actionKind === PgSqlMergeActionKind::Insert) {

81                    $columns,
82                    $defaults,
83                    $identityStrategies,
84                    $existingRows,
85                    $effective,

100    private function effectiveConditions(PgSqlMergeStatement $statement): array
101    {
102        $priorMatched = [];
103        $priorNotMatched = [];
104        $effective = [];
105        foreach ($statement->clauses as $clause) {
106            $prior = $clause->matchKind === PgSqlMergeMatchKind::Matched
107                ? $priorMatched
108                : $priorNotMatched;
109            $condition = $clause->conditionSql ?? 'TRUE';
110            $predicate = '(' . $condition . ')';
111            if ($prior !== []) {
112                $excluded = array_map(
113                    static fn (string $previous): string => 'COALESCE((' . $previous . '), FALSE)',
114                    $prior,
115                );
116                $predicate .= ' AND NOT (' . implode(' OR ', $excluded) . ')';
117            }
118            $effective[] = $predicate;
119            if ($clause->matchKind === PgSqlMergeMatchKind::Matched) {
120                $priorMatched[] = $condition;
121            } else {
122                $priorNotMatched[] = $condition;
123            }
124        }
125
126        return $effective;
127    }

134        PgSqlMergeStatement $statement,
135        array $columns,
136        array $effectiveConditions,
137    ): string {
138        $qualifier = $this->quoter->quote($statement->targetAlias);
139        $selects = [];
140        foreach ($columns as $column) {
141            $quoted = $this->quoter->quote($column);
142            $selects[] = $qualifier . '.' . $quoted . ' AS ' . $quoted;
143        }
144
145        $modifications = [];
146        foreach ($statement->clauses as $index => $clause) {
147            if ($clause->matchKind !== PgSqlMergeMatchKind::Matched
148                || !in_array($clause->actionKind, [PgSqlMergeActionKind::Update, PgSqlMergeActionKind::Delete], true)
149            ) {
150                continue;
151            }
152            $modifications[] = 'EXISTS (SELECT 1 FROM ' . $statement->sourceSql
153                . ' WHERE (' . $statement->joinConditionSql . ') AND (' . $effectiveConditions[$index] . '))';
154        }
155
156        $sql = 'SELECT ' . implode(', ', $selects)
157            . ' FROM ' . $statement->targetSql . ' AS ' . $qualifier;
158        if ($modifications !== []) {
159            $sql .= ' WHERE NOT (' . implode(' OR ', $modifications) . ')';
160        }

172        PgSqlMergeClause $clause,
173        array $columns,
174        array $defaults,
175        string $effectiveCondition,
176    ): string {
177        foreach (array_keys($clause->assignments) as $column) {
178            if (!in_array($column, $columns, true)) {
179                throw new UnsupportedSqlException($sql, 'MERGE UPDATE references an unknown target column');
180            }
181        }
182
183        $qualifier = $this->quoter->quote($statement->targetAlias);
184        $selects = [];
185        foreach ($columns as $column) {
186            $quoted = $this->quoter->quote($column);
187            $expression = $clause->assignments[$column] ?? null;
188            if ($expression === null) {
189                $expression = $qualifier . '.' . $quoted;
190            } elseif (strcasecmp($expression, 'DEFAULT') === 0) {
191                $expression = $defaults[$column] ?? 'NULL';
192            }
193            $selects[] = $expression . ' AS ' . $quoted;
194        }
195
196        return 'SELECT ' . implode(', ', $selects)
197            . ' FROM ' . $statement->targetSql . ' AS ' . $qualifier
198            . ' JOIN ' . $statement->sourceSql
199            . ' ON (' . $statement->joinConditionSql . ')'
200            . ' WHERE ' . $effectiveCondition;

207     * @param array<int, array<string, mixed>> $existingRows

209    private function insertedRows(
210        string $sql,
211        PgSqlMergeStatement $statement,
212        PgSqlMergeClause $clause,
213        array $columns,
214        array $defaults,
215        array $identityStrategies,
216        array $existingRows,
217        string $effectiveCondition,
218    ): string {
219        if ($clause->insertColumns !== []) {
220            $sourceColumns = $clause->insertColumns;
221        } elseif ($clause->insertValues === []) {
222            $sourceColumns = [];
223        } else {
224            $sourceColumns = $columns;
225        }
226        foreach ($sourceColumns as $column) {
227            if (!in_array($column, $columns, true)) {
228                throw new UnsupportedSqlException($sql, 'MERGE INSERT references an unknown target column');
229            }
230        }
231
232        try {
233            $providedExpressions = $this->rowRenderer->providedExpressions($sourceColumns, $clause->insertValues);
234        } catch (\InvalidArgumentException) {
235            throw new UnsupportedSqlException($sql, 'MERGE INSERT values count does not match column count');
236        }
237        $generatedStarts = (new ShadowIdentityAllocator())->allocateSelectStarts(
238            $statement->targetTable,
239            $identityStrategies,
240            array_keys($providedExpressions),
241            $existingRows,
242        );
243        foreach ($generatedStarts as $column => $start) {
244            $providedExpressions[$column] = $this->insertSelectRenderer->renderGeneratedIdentity($start);
245        }
246        $projected = $this->rowRenderer->render($columns, $providedExpressions, $defaults);
247
248        $selects = [];
249        foreach ($projected as $column => $expression) {
250            $selects[] = $expression . ' AS ' . $this->quoter->quote($column);
251        }
252        $targetAlias = $this->quoter->quote($statement->targetAlias);
253
254        return 'SELECT ' . implode(', ', $selects)
255            . ' FROM ' . $statement->sourceSql
256            . ' WHERE NOT EXISTS (SELECT 1 FROM ' . $statement->targetSql . ' AS ' . $targetAlias
257            . ' WHERE ' . $statement->joinConditionSql . ')'
258            . ' AND (' . $effectiveCondition . ')';
259    }
