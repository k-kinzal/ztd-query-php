19/**
20 * Applies CTE shadowing to SELECT statements for PostgreSQL.
21 *
22 * Key differences from MySQL SelectTransformer:
23 * - Uses double-quote identifiers ("table") instead of backticks
24 * - Uses AS MATERIALIZED for CTE definition (PG 12+ inline prevention)
25 * - Uses VALUES clause for multi-row CTEs instead of UNION ALL chains
26 * - Uses WHERE FALSE for empty CTEs instead of FROM DUAL WHERE 0
27 * - Uses PostgreSQL CAST types (INTEGER, TEXT, BOOLEAN, etc.)
28 */
29final class SelectTransformer implements SqlTransformer

54    public function transform(string $sql, array $tables): string
55    {
56        $sql = $this->tableSampleRewriter->rewrite($sql, $tables);
57        $ctes = [];
58        foreach ($tables as $tableName => $tableContext) {
59            if (isset($tableContext['viewSql'])) {
60                $ctes[$tableName] = $this->quoter->quote($tableName) . " AS MATERIALIZED ({$tableContext['viewSql']})";
61                continue;
62            }
63            if (isset($tableContext['sourceSql'])) {
64                $ctes[$tableName] = $this->quoter->quote($tableName) . " AS MATERIALIZED ({$tableContext['sourceSql']})";
65                continue;
66            }
67
68            $rows = $tableContext['rows'];
69            $columns = $tableContext['columns'];
70            /** @var array<string, ColumnType> $columnTypes */
71            $columnTypes = $tableContext['columnTypes'];
72            $generatedExpressions = $tableContext['generatedExpressions'] ?? [];
73
74            if ($columns === [] && $rows !== []) {
75                $columns = array_keys($rows[0]);
76                foreach ($rows as $row) {
77                    foreach (array_keys($row) as $column) {
78                        if (!in_array($column, $columns, true)) {
79                            $columns[] = $column;
80                        }
81                    }
82                }
83            }
84
85            if ($columns === [] && $rows === []) {
86                continue;
87            }
88
89            $ctes[$tableName] = $this->generateCte(
90                $tableName,
91                $rows,
92                $columns,
93                $columnTypes,
94                $generatedExpressions,
95            );
96        }
97
98        return $this->cteComposer->compose($sql, $ctes);
99    }

109        array $rows,
110        array $columns,
111        array $columnTypes,
112        array $generatedExpressions,
113    ): string {
114        $quotedTable = $this->quoter->quote($tableName);

117            if ($rows === []) {

123                        : $this->renderFallbackNullCast();

125                }

133            }
134
135            if (count($rows) === 1) {
136                $selects = [];
137                $row = $rows[0];
138                foreach ($columns as $col) {
139                    $colType = $columnTypes[$col] ?? null;
140                    $valStr = $this->formatValue($row[$col] ?? null, $colType);
141                    $selects[] = "$valStr AS " . $this->quoter->quote($col);
142                }
143
144                return $this->wrapCte(
145                    $quotedTable,
146                    'SELECT ' . implode(', ', $selects),
147                    $columns,
148                    $generatedExpressions,
149                );
150            }
151
152            $baseSql = $this->generateMultiRowSource($rows, $columns, $columnTypes);
153
154            return $this->wrapCte($quotedTable, $baseSql, $columns, $generatedExpressions);
155        }
156
157        if ($rows === []) {
158            throw new \RuntimeException("Cannot shadow table '$tableName' with empty data (columns unknown).");
159        }
160
161        $ctes = [];
162        foreach ($rows as $row) {
163            $selects = [];
164            foreach ($row as $col => $val) {
165                $colName = $col;
166                $colType = $columnTypes[$colName] ?? null;
167                $valStr = $this->formatValue($val, $colType);
168                $selects[] = "$valStr AS " . $this->quoter->quote($colName);
169            }
170            $ctes[] = 'SELECT ' . implode(', ', $selects);
171        }
172
173        $union = implode(' UNION ALL ', $ctes);
174
175        return $this->wrapCte($quotedTable, $union, array_keys($rows[0]), $generatedExpressions);
176    }

178    /**
179     * @param array<int, array<string, mixed>> $rows
180     * @param array<int, string> $columns
181     * @param array<string, ColumnType> $columnTypes
182     */
183    private function generateMultiRowSource(
184        array $rows,
185        array $columns,
186        array $columnTypes
187    ): string {
188        $valueRows = [];
189        foreach ($rows as $row) {
190            $values = [];
191            foreach ($columns as $col) {
192                $colType = $columnTypes[$col] ?? null;
193                $values[] = $this->formatValue($row[$col] ?? null, $colType);
194            }
195            $valueRows[] = '(' . implode(', ', $values) . ')';
196        }
197
198        $quotedColumns = [];
199        foreach ($columns as $col) {
200            $quotedColumns[] = $this->quoter->quote($col);
201        }
202
203        $valuesClause = implode(",\n    ", $valueRows);
204        $columnList = implode(', ', $quotedColumns);
205
206        return "\n  SELECT * FROM (VALUES\n    $valuesClause\n  ) AS t($columnList)\n";
207    }
