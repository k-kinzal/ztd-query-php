56    {
57        $this->identityAllocator->beginProjection();
58        $tableName = $this->parser->extractInsertTable($sql);
59        if ($tableName === null) {
60            throw new UnsupportedSqlException($sql, 'Cannot resolve INSERT target');
61        }
62
63        $insertColumns = $this->parser->extractInsertColumns($sql);
64        $tableColumns = self::orderedValues($tables[$tableName]['columns'] ?? $insertColumns);
65        if ($tableColumns === []) {
66            throw new UnsupportedSqlException($sql, 'Cannot determine columns');
67        }
68
69        $columnDefaults = $tables[$tableName]['columnDefaults'] ?? [];
70        $identityStrategies = $tables[$tableName]['identityStrategies'] ?? [];
71        $existingRows = $tables[$tableName]['rows'] ?? [];
72        $identityTable = $tables[$tableName]['storageTable'] ?? $tableName;
73
74        if ($this->parser->hasInsertSelect($sql)) {
75            $selectSql = $this->parser->extractInsertSelectSql($sql);
76            if ($selectSql === null) {
77                throw new UnsupportedSqlException($sql, 'Cannot extract INSERT ... SELECT subquery');
78            }
79
80            $sourceColumns = $insertColumns !== [] ? $insertColumns : $tableColumns;
81            $generatedIdentityStarts = $this->identityAllocator->allocateSelectStarts(
82                $identityTable,
83                $identityStrategies,
84                $sourceColumns,
85                $existingRows,
86            );
87            $projectedSql = $this->insertSelectRenderer->render(
88                $this->cteComposer->carryPrefix($sql, $selectSql),
89                $tableColumns,
90                $sourceColumns,
91                $columnDefaults,
92                $generatedIdentityStarts,
93            );
94            $projectedSql = $this->projectUpsert($sql, $projectedSql, $tableName, $tableColumns, $tables);
95
96            return $this->selectTransformer->transform($projectedSql, $tables);
97        }
98
99        $valueRows = SqlTokenStream::tokenize($sql, PgSqlLexerProfile::create())->topLevelClause(['DEFAULT', 'VALUES']) !== null
100            ? [[]]
101            : $this->parser->extractInsertValues($sql);
102        if ($valueRows === []) {
103            throw new UnsupportedSqlException($sql, 'Cannot extract INSERT values');
104        }
105
106        $selectParts = [];
107        $columnTypes = $tables[$tableName]['columnTypes'] ?? [];
108        foreach ($valueRows as $values) {
109            $sourceColumns = $insertColumns !== [] || $values === [] ? $insertColumns : $tableColumns;
110            try {
111                $providedExpressions = $this->rowRenderer->providedExpressions($sourceColumns, $values);
112            } catch (\InvalidArgumentException) {
113                throw new UnsupportedSqlException($sql, 'Insert values count does not match column count');
114            }
115            $generatedValues = $this->identityAllocator->allocateMissing(
116                $identityTable,
117                $identityStrategies,
118                array_keys($providedExpressions),
119                $existingRows,
120            );
121            $projected = $this->rowRenderer->render($tableColumns, $providedExpressions, $columnDefaults, $generatedValues);
122
123            $selects = [];
124            foreach ($projected as $column => $expr) {
125                $type = $columnTypes[$column] ?? null;
126                if ($type instanceof ColumnType) {
127                    $expr = $this->castInsertExpression($expr, $type);
128                }
129                $selects[] = $expr . ' AS "' . $column . '"';
130            }
131            $selectParts[] = 'SELECT ' . implode(', ', $selects);
132        }
133
134        $selectSql = implode(' UNION ALL ', $selectParts);
135        $selectSql = $this->projectUpsert($sql, $selectSql, $tableName, $tableColumns, $tables);
136
137        return $this->selectTransformer->transform(
138            $this->cteComposer->carryPrefix($sql, $selectSql),
139            $tables,
140        );
141    }

150    private function projectUpsert(
151        string $sql,
152        string $selectSql,
153        string $tableName,
154        array $tableColumns,
155        array $tables,
156    ): string {
157        $conflict = $this->parser->extractOnConflictUpdateColumns($sql);
158        $candidateKeys = new CandidateKeySet(
159            isset($tables[$tableName]['candidateKeys']) ? $tables[$tableName]['candidateKeys'] : [],
160        );
161        $conflictPredicate = null;
162        $target = $this->parser->extractOnConflictTarget($sql);
163        if ($target !== null) {
164            $resolved = $target->resolve(
165                $candidateKeys,
166                isset($tables[$tableName]['partialUniqueIndexes'])
167                    ? $tables[$tableName]['partialUniqueIndexes']
168                    : [],
169                $sql,
170            );
171            $candidateKeys = $resolved['keys'];
172            $conflictPredicate = $resolved['predicate'];
173        }
174
175        return $this->upsertProjector->project(
176            $selectSql,
177            $tableName,
178            $tableColumns,
179            $candidateKeys->keys(),
180            $conflict['values'],
181            $this->parser->extractOnConflictUpdateWhere($sql),
182            $conflictPredicate,
183        );
184    }

206    private function castInsertExpression(string $expression, ColumnType $type): string
207    {
208        if ($type->family === ColumnTypeFamily::BOOLEAN && $expression === '?') {
209            return "CAST(COALESCE(NULLIF(CAST(? AS TEXT), ''), 'false') AS BOOLEAN)";
