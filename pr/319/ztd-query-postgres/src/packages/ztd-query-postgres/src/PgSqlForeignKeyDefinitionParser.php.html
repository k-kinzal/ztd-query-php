214    private function action(array $tokens, string $event): ReferentialAction
215    {
216        foreach ($tokens as $index => $token) {
217            if (!$token->isTopLevel() || !$token->isKeyword('ON')) {
218                continue;
219            }
220            $eventToken = $tokens[$index + 1] ?? null;
221            if ($eventToken === null || !$eventToken->isKeyword($event)) {
222                continue;
223            }
224
225            $action = $tokens[$index + 2] ?? null;
226            if ($action === null) {
227                return ReferentialAction::NoAction;
228            }
229            if ($action->isKeyword('CASCADE')) {
230                return ReferentialAction::Cascade;
231            }
232            if ($action->isKeyword('RESTRICT')) {
233                return ReferentialAction::Restrict;
234            }
235            if (!$action->isKeyword('SET')) {
236                return ReferentialAction::NoAction;
237            }
238
239            $qualifier = $tokens[$index + 3] ?? null;
240            if ($qualifier === null) {
241                return ReferentialAction::NoAction;
242            }
243            if ($qualifier->isKeyword('NULL')) {
244                return ReferentialAction::SetNull;
245            }
246            if ($qualifier->isKeyword('DEFAULT')) {
247                return ReferentialAction::SetDefault;
248            }
249
250            return ReferentialAction::NoAction;
251        }
252
253        return ReferentialAction::NoAction;
254    }
