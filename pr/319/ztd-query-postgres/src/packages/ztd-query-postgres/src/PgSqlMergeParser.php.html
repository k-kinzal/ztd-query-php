22    {
23        $statementSql = $this->cteComposer->statementSql($sql);
24        $tokens = SqlTokenStream::tokenize($statementSql, PgSqlLexerProfile::create())->significantTokens();
25        $index = 0;
26        $first = $tokens[$index] ?? null;
27        if ($first === null) {
28            throw new UnsupportedSqlException($sql, 'Malformed MERGE statement');
29        }
30        if (!$first->isKeyword('MERGE')) {
31            throw new UnsupportedSqlException($sql, 'Malformed MERGE statement');
32        }
33        $index++;
34        $next = $tokens[$index] ?? null;
35        if ($next !== null && $next->isKeyword('INTO')) {
36            $index++;
37            $next = $tokens[$index] ?? null;
38        }
39        if ($next !== null && $next->isKeyword('ONLY')) {
40            $index++;
41        }
42
43        $target = $this->relationAt($statementSql, $tokens, $index);
44        if ($target === null) {
45            throw new UnsupportedSqlException($sql, 'Cannot resolve MERGE target');
46        }
47        $index = $target['next'];
48        if ($this->isSymbol($tokens[$index] ?? null, '*')) {
49            $target['last'] = $tokens[$index];
50            $index++;
51        }
52
53        $usingIndex = $this->keywordIndexAfter($tokens, 'USING', $target['last']);
54        if ($usingIndex === null) {
55            throw new UnsupportedSqlException($sql, 'MERGE requires USING');
56        }
57        $targetAlias = $this->targetAlias($sql, $tokens, $index, $usingIndex, $target['name']);
58
59        $usingToken = $tokens[$usingIndex];
60        $whenTokens = $this->mergeWhenTokens($tokens);
61        $firstWhen = $whenTokens[0] ?? null;
62        if ($firstWhen === null) {
63            throw new UnsupportedSqlException($sql, 'MERGE requires a WHEN clause');
64        }
65        $onToken = $this->lastKeywordBetween($tokens, 'ON', $usingToken, $firstWhen);
66        if ($onToken === null) {
67            throw new UnsupportedSqlException($sql, 'MERGE requires an ON condition');
68        }
69
70        $sourceSql = trim(substr(
71            $statementSql,
72            $usingToken->endOffset(),
73            $onToken->offset - $usingToken->endOffset(),
74        ));
75        $joinConditionSql = trim(substr(
76            $statementSql,
77            $onToken->endOffset(),
78            $firstWhen->offset - $onToken->endOffset(),
79        ));
80        if ($sourceSql === '') {
81            throw new UnsupportedSqlException($sql, 'MERGE requires a source and join condition');
82        }
83        if ($joinConditionSql === '') {
84            throw new UnsupportedSqlException($sql, 'MERGE requires a source and join condition');
85        }
86
87        $clauses = [];
88        foreach ($whenTokens as $clauseIndex => $whenToken) {
89            $end = isset($whenTokens[$clauseIndex + 1])
90                ? $whenTokens[$clauseIndex + 1]->offset
91                : strlen($statementSql);
92            $clauseSql = substr($statementSql, $whenToken->offset, $end - $whenToken->offset);
93            $clauses[] = $this->parseClause($sql, $clauseSql);
94        }
95        if ($clauses === []) {
96            throw new UnsupportedSqlException($sql, 'MERGE requires a WHEN clause');
97        }
98
99        return new PgSqlMergeStatement(
100            $target['name'],
101            $target['sql'],
102            $targetAlias,
103            $sourceSql,
104            $joinConditionSql,
105            $clauses,
106        );
107    }

113    private function relationAt(string $sql, array $tokens, int $index): ?array
114    {
115        $first = $tokens[$index] ?? null;
116        $name = $first !== null ? $this->identifierName($first) : null;
117        if ($first === null || $name === null) {
118            return null;
119        }
120
121        $start = $first->offset;
122        $end = $first->endOffset();
123        $last = $first;
124        $index++;
125        while ($this->isSymbol($tokens[$index] ?? null, '.')) {
126            $component = $tokens[$index + 1] ?? null;
127            if ($component === null) {
128                return null;
129            }
130            $componentName = $this->identifierName($component);
131            if ($componentName === null) {
132                return null;
133            }
134            $name = $componentName;
135            $end = $component->endOffset();
136            $last = $component;
137            $index += 2;
138        }
139
140        return [
141            'name' => $name,
142            'sql' => substr($sql, $start, $end - $start),
143            'next' => $index,
144            'last' => $last,
145        ];
146    }

173    private function parseClause(string $originalSql, string $clauseSql): PgSqlMergeClause
174    {
175        $clauseSql = rtrim($clauseSql, "; \t\n\r\0\x0B");
176        $tokens = SqlTokenStream::tokenize($clauseSql, PgSqlLexerProfile::create())->significantTokens();
177        $index = 1;
178        $matchKind = PgSqlMergeMatchKind::Matched;
179        $matchToken = $tokens[$index];
180        if ($matchToken->isKeyword('NOT')) {
181            $matchKind = PgSqlMergeMatchKind::NotMatched;
182            $index++;
183            $matchToken = $tokens[$index];
184        }
185        if (!$matchToken->isKeyword('MATCHED')) {
186            throw new UnsupportedSqlException($originalSql, 'Malformed MERGE WHEN clause');
187        }
188        $index++;
189        $modifier = $tokens[$index] ?? null;
190        if ($modifier !== null && $modifier->isKeyword('BY')) {
191            throw new UnsupportedSqlException($originalSql, 'MERGE BY SOURCE and BY TARGET are not supported');
192        }
193
194        $thenIndex = $this->keywordIndexOutsideCase($tokens, 'THEN');
195        if ($thenIndex === null) {
196            throw new UnsupportedSqlException($originalSql, 'MERGE WHEN clause requires THEN');
197        }
198        $conditionSql = null;
199        if ($index !== $thenIndex) {
200            if (!$tokens[$index]->isKeyword('AND')) {
201                throw new UnsupportedSqlException($originalSql, 'Malformed MERGE WHEN condition');
202            }
203            $conditionSql = trim(substr(
204                $clauseSql,
205                $tokens[$index]->endOffset(),
206                $tokens[$thenIndex]->offset - $tokens[$index]->endOffset(),
207            ));
208            if ($conditionSql === '') {
209                throw new UnsupportedSqlException($originalSql, 'MERGE WHEN condition cannot be empty');
210            }
211        }
212
213        $actionSql = substr($clauseSql, $tokens[$thenIndex]->endOffset());
214        $actionTokens = SqlTokenStream::tokenize($actionSql, PgSqlLexerProfile::create())->significantTokens();
215        $first = $actionTokens[0] ?? null;
216        if ($first === null) {
217            throw new UnsupportedSqlException($originalSql, 'MERGE action is not supported');
218        }
219        if ($first->isKeyword('DO')) {
220            $second = $actionTokens[1] ?? null;
221            if ($second !== null && $second->isKeyword('NOTHING')) {
222                if (count($actionTokens) === 2) {
223                    return new PgSqlMergeClause($matchKind, $conditionSql, PgSqlMergeActionKind::DoNothing);
224                }
225            }
226        }
227        if ($matchKind === PgSqlMergeMatchKind::Matched && $first->isKeyword('DELETE') && count($actionTokens) === 1) {
228            return new PgSqlMergeClause($matchKind, $conditionSql, PgSqlMergeActionKind::Delete);
229        }
230        if ($matchKind === PgSqlMergeMatchKind::Matched && $first->isKeyword('UPDATE')) {
231            return new PgSqlMergeClause(
232                $matchKind,
233                $conditionSql,
234                PgSqlMergeActionKind::Update,
235                $this->parseAssignments($originalSql, $actionSql, $actionTokens),
236            );
237        }
238        if ($matchKind === PgSqlMergeMatchKind::NotMatched && $first->isKeyword('INSERT')) {
239            $insert = $this->parseInsert($originalSql, $actionSql, $actionTokens);
240
241            return new PgSqlMergeClause(
242                $matchKind,
243                $conditionSql,
244                PgSqlMergeActionKind::Insert,
245                [],
246                $insert['columns'],
247                $insert['values'],
248            );
249        }
250
251        throw new UnsupportedSqlException($originalSql, 'MERGE action is not supported');
252    }

258    private function parseAssignments(string $originalSql, string $actionSql, array $tokens): array
259    {
260        $set = $tokens[1] ?? null;
261        if ($set === null) {
262            throw new UnsupportedSqlException($originalSql, 'MERGE UPDATE requires SET');
263        }
264        if (!$set->isKeyword('SET')) {
265            throw new UnsupportedSqlException($originalSql, 'MERGE UPDATE requires SET');
266        }
267        $setSql = substr($actionSql, $set->endOffset());
268        $assignments = [];
269        foreach (SqlTokenStream::tokenize($setSql, PgSqlLexerProfile::create())->splitTopLevel() as $assignmentSql) {
270            $assignmentTokens = SqlTokenStream::tokenize($assignmentSql, PgSqlLexerProfile::create())->significantTokens();
271            $equals = [];
272            foreach ($assignmentTokens as $assignmentIndex => $token) {
273                if ($this->isSymbol($token, '=') && $token->isTopLevel()) {
274                    $equals[] = $assignmentIndex;
275                }
276            }
277            if (count($equals) !== 1) {
278                throw new UnsupportedSqlException($originalSql, 'MERGE UPDATE requires simple column assignments');
279            }
280            $equalsIndex = $equals[0];
281            if ($equalsIndex !== 1 || count($assignmentTokens) < 3) {
282                throw new UnsupportedSqlException($originalSql, 'MERGE UPDATE requires simple column assignments');
283            }
284            $column = $this->identifierName($assignmentTokens[0]);
285            if ($column === null) {
286                throw new UnsupportedSqlException($originalSql, 'MERGE UPDATE target must be a column');
287            }
288            $value = trim(substr($assignmentSql, $assignmentTokens[$equalsIndex]->endOffset()));
289            if ($value === '') {
290                throw new UnsupportedSqlException($originalSql, 'MERGE UPDATE value cannot be empty');
291            }
292            if (array_key_exists($column, $assignments)) {
293                throw new UnsupportedSqlException($originalSql, 'MERGE UPDATE cannot assign a column more than once');
294            }
295            $assignments[$column] = $value;
296        }
297        if ($assignments === []) {
298            throw new UnsupportedSqlException($originalSql, 'MERGE UPDATE requires assignments');
299        }
300
301        return $assignments;
302    }

308    private function parseInsert(string $originalSql, string $actionSql, array $tokens): array
309    {
310        $index = 1;
311        $columns = [];
312        if ($this->isSymbol($tokens[$index] ?? null, '(')) {
313            $list = $this->parenthesizedList($originalSql, $actionSql, $tokens, $index);
314            foreach ($list['items'] as $columnSql) {
315                $columnTokens = SqlTokenStream::tokenize($columnSql, PgSqlLexerProfile::create())->significantTokens();
316                $column = count($columnTokens) === 1 ? $this->identifierName($columnTokens[0]) : null;
317                if ($column === null) {
318                    throw new UnsupportedSqlException($originalSql, 'MERGE INSERT columns must be identifiers');
319                }
320                if (in_array($column, $columns, true)) {
321                    throw new UnsupportedSqlException($originalSql, 'MERGE INSERT cannot name a column more than once');
322                }
323                $columns[] = $column;
324            }
325            $index = $list['next'];
326        }
327
328        $insertAction = $tokens[$index] ?? null;
329        $afterAction = $tokens[$index + 1] ?? null;
330        if ($insertAction !== null && $insertAction->isKeyword('DEFAULT')) {
331            if ($afterAction === null || !$afterAction->isKeyword('VALUES') || $index + 2 !== count($tokens)) {
332                throw new UnsupportedSqlException($originalSql, 'MERGE INSERT requires VALUES');
333            }
334            if ($columns !== []) {
335                throw new UnsupportedSqlException($originalSql, 'MERGE INSERT DEFAULT VALUES cannot name columns');
336            }
337
338            return ['columns' => [], 'values' => []];
339        }
340        if ($insertAction === null) {
341            throw new UnsupportedSqlException($originalSql, 'MERGE INSERT requires VALUES');
342        }
343        if (!$insertAction->isKeyword('VALUES')) {
344            throw new UnsupportedSqlException($originalSql, 'MERGE INSERT requires VALUES');
345        }
346        if (!$this->isSymbol($afterAction, '(')) {
347            throw new UnsupportedSqlException($originalSql, 'MERGE INSERT requires VALUES');
348        }
349
350        $values = $this->parenthesizedList($originalSql, $actionSql, $tokens, $index + 1);
351        if ($values['next'] !== count($tokens)) {
352            throw new UnsupportedSqlException($originalSql, 'Malformed MERGE INSERT values');
353        }
354        if ($columns !== [] && count($columns) !== count($values['items'])) {
355            throw new UnsupportedSqlException($originalSql, 'MERGE INSERT values count does not match column count');
356        }
357
358        return ['columns' => $columns, 'values' => $values['items']];
359    }

365    private function parenthesizedList(
366        string $originalSql,
367        string $sql,
368        array $tokens,
369        int $openIndex,
370    ): array {
371        $open = $tokens[$openIndex];
372        $afterOpen = false;
373        foreach ($tokens as $closeIndex => $token) {
374            if ($token === $open) {
375                $afterOpen = true;
376                continue;
377            }
378            if (!$afterOpen) {
379                continue;
380            }
381            if (!$this->isSymbol($token, ')')) {
382                continue;
383            }
384            if ($token->depth !== $open->depth) {
385                continue;
386            }
387            $listSql = substr($sql, $open->endOffset(), $token->offset - $open->endOffset());
388            $items = SqlTokenStream::tokenize($listSql, PgSqlLexerProfile::create())->splitTopLevel();
389
390            return ['items' => $items, 'next' => $closeIndex + 1];
391        }
392
393        throw new UnsupportedSqlException($originalSql, 'Malformed MERGE parenthesized list');
394    }

440     * @return list<SqlToken>
441     */
442    private function mergeWhenTokens(array $tokens): array
443    {
444        $whenTokens = [];
445        $caseDepth = 0;
446        foreach ($tokens as $index => $token) {
447            if (!$token->isTopLevel()) {
448                continue;
449            }
450            if ($token->isKeyword('CASE')) {
451                $caseDepth++;
452                continue;
453            }
454            if ($token->isKeyword('END')) {
455                if ($caseDepth !== 0) {
456                    $caseDepth--;
457                    continue;
458                }
459            }
460            if ($caseDepth !== 0 || !$token->isKeyword('WHEN')) {
461                continue;
462            }
463            $next = $tokens[$index + 1] ?? null;
464            if ($next === null) {
465                continue;
466            }
467            if ($next->isKeyword('MATCHED')) {
468                $whenTokens[] = $token;
469                continue;
470            }
471            if (!$next->isKeyword('NOT')) {
472                continue;
473            }
474            $afterNext = $tokens[$index + 2] ?? null;
475            if ($afterNext === null) {
476                continue;
477            }
478            if ($afterNext->isKeyword('MATCHED')) {
479                $whenTokens[] = $token;
480            }
481        }
482
483        return $whenTokens;
484    }
