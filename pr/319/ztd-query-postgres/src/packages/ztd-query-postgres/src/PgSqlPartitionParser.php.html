283    private function qualifiedIdentifierAt(SqlTokenStream $stream, array $tokens, int $index): ?array
284    {
285        $token = $tokens[$index] ?? null;
286        if (!$token instanceof SqlToken) {
287            return null;
288        }
289        if (!in_array($token->kind, [SqlTokenKind::Word, SqlTokenKind::QuotedIdentifier], true)) {
290            return null;
291        }
292        $identifier = $stream->identifierAt($index);
293        if ($identifier === null) {
294            return null;
295        }
296        if ($token->kind === SqlTokenKind::Word) {
297            $identifier['name'] = strtolower($identifier['name']);
298        }
299
300        $dot = $tokens[$identifier['next']] ?? null;
301        while ($dot instanceof SqlToken && $this->isSymbol($dot, '.')) {
302            $componentIndex = $identifier['next'] + 1;
303            $component = $this->qualifiedIdentifierAt($stream, $tokens, $componentIndex);
304            if ($component === null) {

307            $identifier = $component;
