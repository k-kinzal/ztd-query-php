122    private function tableBody(string $sql): ?string
123    {
124        $stream = SqlTokenStream::tokenize($sql, PgSqlLexerProfile::create());
125        $tokens = $stream->significantTokens();
126        $create = $tokens[0] ?? null;
127        if (!$create instanceof SqlToken || !$create->isKeyword('CREATE')) {
128            return null;
129        }
130
131        $tableIndex = null;
132        foreach ($tokens as $index => $token) {
133            if ($token->isTopLevel() && $token->isKeyword('TABLE')) {
134                $tableIndex = $index;
135            }
136        }
137        if ($tableIndex === null) {
138            return null;
139        }
140
141        $index = $tableIndex + 1;
142        $candidate = $tokens[$index] ?? null;
143        if (!$candidate instanceof SqlToken) {
144            return null;
145        }
146        if ($candidate->isKeyword('IF')) {
147            $not = $tokens[$index + 1] ?? null;
148            $exists = $tokens[$index + 2] ?? null;
149            if (!$not instanceof SqlToken || !$not->isKeyword('NOT')) {
150                return null;
151            }
152            if (!$exists instanceof SqlToken || !$exists->isKeyword('EXISTS')) {
153                return null;
154            }
155            $index += 3;
156        }
157        $identifier = $this->qualifiedIdentifierAt($stream, $tokens, $index);
158        if ($identifier === null) {
159            return null;
160        }
161
162        $open = $tokens[$identifier['next']] ?? null;
163        if (!$open instanceof SqlToken) {
164            return null;
165        }
166        if (!$this->isSymbol($open, '(')) {
167            return null;
168        }
169
170        foreach ($tokens as $token) {
171            if ($this->isSymbol($token, ')') && $token->depth === $open->depth) {
172                return substr($sql, $open->endOffset(), $token->offset - $open->endOffset());
173            }
174        }
175
176        return null;
177    }

180     * @param list<SqlToken> $tokens
181     * @return array{name: string, next: int}|null
182     */
183    private function qualifiedIdentifierAt(SqlTokenStream $stream, array $tokens, int $index): ?array
184    {
185        $token = $tokens[$index] ?? null;
186        if (!$token instanceof SqlToken) {
187            return null;
188        }
189        if (!in_array($token->kind, [SqlTokenKind::Word, SqlTokenKind::QuotedIdentifier], true)) {
190            return null;
191        }
192        $identifier = $stream->identifierAt($index);
193        if ($identifier === null) {
194            return null;
195        }
196
197        $dot = $tokens[$identifier['next']] ?? null;
198        while ($dot instanceof SqlToken && $this->isSymbol($dot, '.')) {
199            $component = $this->qualifiedIdentifierAt($stream, $tokens, $identifier['next'] + 1);
200            if ($component === null) {
201                return null;
202            }
203            $identifier = $component;
204            $dot = $tokens[$identifier['next']] ?? null;
205        }
206
207        return $identifier;
208    }

230     */
231    private function parseColumnDefinition(string $entry): ?array
232    {
233        if (preg_match('/^("[^"]+"|[a-zA-Z_]\w*)\s+(.+)$/is', $entry, $m) !== 1) {
234            return null;
235        }
236
237        $name = $this->unquoteIdentifier($m[1]);
238        $rest = trim($m[2]);
239
240        $typeInfo = $this->extractType($rest);
241        if ($typeInfo === null) {
242            return null;
243        }
244
245        $nativeType = $typeInfo['type'];
246        $afterType = $typeInfo['rest'];
247
248        $notNull = preg_match('/\bNOT\s+NULL\b/i', $afterType) === 1;
249        $primaryKey = preg_match('/\bPRIMARY\s+KEY\b/i', $afterType) === 1;
250        $unique = preg_match('/\bUNIQUE\b/i', $afterType) === 1;
251        $default = SqlTokenStream::tokenize($afterType, PgSqlLexerProfile::create())->topLevelClause(
252            ['DEFAULT'],
253            [
254                ['NOT', 'NULL'], ['PRIMARY', 'KEY'], ['UNIQUE'], ['CHECK'],
255                ['REFERENCES'], ['COLLATE'], ['CONSTRAINT'], ['GENERATED'], ['DEFERRABLE'],
256            ],
257        );
258        $identity = self::isSerialType($nativeType)
259            || ($default !== null && self::isSequenceDefault($default))
260            || self::hasGeneratedIdentity($afterType);
261        $generatedExpression = null;
262        if (!$identity) {
263            $generatedExpression = SqlTokenStream::tokenize($afterType, PgSqlLexerProfile::create())->topLevelClause(
264                ['GENERATED', 'ALWAYS', 'AS'],
265                [['STORED']],
266            );
267            if ($generatedExpression === '') {
268                $generatedExpression = null;
269            }
270        }
271
272        $normalizedType = str_contains($nativeType, '"') ? $nativeType : strtoupper($nativeType);
273        $columnType = (new PgSqlColumnTypeMapper())->map($normalizedType);
274
275        return [
276            'name' => $name,
277            'type' => $normalizedType,
278            'columnType' => $columnType,
279            'notNull' => $notNull,
280            'primaryKey' => $primaryKey,
281            'unique' => $unique,
282            'default' => $default,
283            'identity' => $identity,
284            'generatedExpression' => $generatedExpression,
285        ];
286    }

300    {
301        $tokens = SqlTokenStream::tokenize($constraints, PgSqlLexerProfile::create())->significantTokens();
302        $sequences = [
303            ['GENERATED', 'ALWAYS', 'AS', 'IDENTITY'],
304            ['GENERATED', 'BY', 'DEFAULT', 'AS', 'IDENTITY'],

343    private function extractType(string $str): ?array
344    {
345        $str = ltrim($str);
346
347        $quotedType = $this->extractQuotedType($str);
348        if ($quotedType !== null) {
349            return $quotedType;
350        }
351
352        if (preg_match('/^([a-zA-Z_]\w*)/i', $str, $m) !== 1) {
353            return null;
354        }
355
356        $baseType = $m[1];
357        $pos = strlen($baseType);
358        $rest = substr($str, $pos);
359
360        if (in_array(strtoupper($baseType), self::MULTI_WORD_TYPE_PREFIXES, true)) {
361            $trimmedRest = ltrim($rest);
362            if (preg_match('/^([a-zA-Z_]\w*)/i', $trimmedRest, $m2) === 1) {
363                $secondWord = $m2[1];
364                if (!in_array(strtoupper($secondWord), self::CONSTRAINT_KEYWORDS, true)) {
365                    $baseType .= ' ' . $secondWord;
366                    $pos = strlen($str) - strlen($trimmedRest) + strlen($secondWord);
367                    $rest = substr($str, $pos);
368                }
369            }
370        }
371
372        $params = '';
373        $trimmedRest = ltrim($rest);
374        if (str_starts_with($trimmedRest, '(')) {
375            if (preg_match('/^(\([^)]*\))/', $trimmedRest, $pm) === 1) {
376                $params = $pm[1];
377                $rest = substr($trimmedRest, strlen($params));
378            }
379        } else {
380            $rest = $trimmedRest;
381        }
382
383        $arrayBrackets = '';
384        $trimmedRest = ltrim($rest);
385        while (preg_match('/^\[\s*\]/', $trimmedRest, $ab) === 1) {
386            $arrayBrackets .= $ab[0];
387            $trimmedRest = ltrim(substr($trimmedRest, strlen($ab[0])));
388        }
389
390        $fullType = $baseType . $params . $arrayBrackets;
391
392        return ['type' => $fullType, 'rest' => trim($trimmedRest)];
393    }

396    private function extractQuotedType(string $str): ?array
397    {
398        $tokens = SqlTokenStream::tokenize($str, PgSqlLexerProfile::create())->significantTokens();
399        $first = $tokens[0] ?? null;
400        if ($first === null) {
401            return null;
402        }
403        if ($first->kind !== SqlTokenKind::QuotedIdentifier) {
404            return null;
405        }
406
407        $last = $first;
408        $index = 1;
409        $separator = $tokens[$index] ?? null;
410        if ($separator !== null && $separator->text === '.') {
411            $index++;
412            $qualifiedName = $tokens[$index] ?? null;
413            if ($qualifiedName === null) {
414                return null;
415            }
416            if (!self::isTypeIdentifier($qualifiedName)) {
417                return null;
418            }
419            if ($qualifiedName->kind === SqlTokenKind::Word) {
420                if (in_array(strtoupper($qualifiedName->text), self::CONSTRAINT_KEYWORDS, true)) {
421                    return null;
422                }
423            }
424            $last = $qualifiedName;
425            $index++;
426        }
427
428        while (isset($tokens[$index]) && $tokens[$index]->text === '[') {
429            $index++;
430            $arrayClosing = $tokens[$index] ?? null;
431            if ($arrayClosing === null) {
432                return null;
433            }
434            if ($arrayClosing->text !== ']') {
435                return null;
436            }
437            $last = $arrayClosing;
438            $index++;
439        }
440
441        return [
442            'type' => substr($str, 0, $last->endOffset()),
443            'rest' => substr($str, $last->endOffset()),
444        ];
445    }

456    private function parseConstraint(string $entry, array &$primaryKeys, array &$uniqueConstraints, int &$uniqueIndex): void
457    {
458        if (preg_match('/PRIMARY\s+KEY\s*\(([^)]+)\)/i', $entry, $m) === 1) {
459            $cols = $this->parseColumnRefList($m[1]);
460            foreach ($cols as $col) {
461                if (!in_array($col, $primaryKeys, true)) {
462                    $primaryKeys[] = $col;
463                }
464            }
465
466            return;
467        }
468
469        if (preg_match('/UNIQUE\s*\(([^)]+)\)/i', $entry, $m) === 1) {
470            $cols = $this->parseColumnRefList($m[1]);
471            if ($cols !== []) {
472                $keyName = 'unique_' . $uniqueIndex++;
473                if (preg_match('/CONSTRAINT\s+("[^"]+"|[a-zA-Z_]\w*)/i', $entry, $cNameMatch) === 1) {
474                    $keyName = $this->unquoteIdentifier($cNameMatch[1]);
475                }
476                $uniqueConstraints[$keyName] = $cols;
477            }
478        }
479    }

513    /**
514     * Split table body by top-level commas (respecting parentheses).
515     *
516     * @return list<string>
517     */
518    private function splitTableBody(string $body): array
519    {
520        $entries = [];
521        $current = '';
522        $depth = 0;
523        $inSingleQuote = false;
524        $inDoubleQuote = false;
525        $len = strlen($body);
526
527        for ($i = 0; $i < $len; $i++) {
528            $char = $body[$i];
529
530            if ($inSingleQuote) {
531                $current .= $char;
532                if ($char === "'" && isset($body[$i + 1]) && $body[$i + 1] === "'") {
533                    $current .= "'";
534                    $i++;
535                } elseif ($char === "'") {
536                    $inSingleQuote = false;
537                }
538                continue;
539            }
540
541            if ($inDoubleQuote) {
542                $current .= $char;
543                if ($char === '"' && isset($body[$i + 1]) && $body[$i + 1] === '"') {
544                    $current .= '"';
545                    $i++;
546                } elseif ($char === '"') {
547                    $inDoubleQuote = false;
548                }
549                continue;
550            }
551
552            if ($char === "'") {
553                $current .= $char;
554                $inSingleQuote = true;
555                continue;
556            }
557
558            if ($char === '"') {
559                $current .= $char;
560                $inDoubleQuote = true;
561                continue;
562            }
563
564            if ($char === '(') {
565                $depth++;
566                $current .= $char;
567                continue;
568            }
569
570            if ($char === ')') {
571                $depth--;
572                $current .= $char;
573                continue;
574            }
575
576            if ($char === ',' && $depth === 0) {
577                $entries[] = trim($current);
578                $current = '';
579                continue;
580            }
581
582            $current .= $char;
583        }
584
585        $val = trim($current);
586        if ($val !== '') {
587            $entries[] = $val;
588        }
589
590        return $entries;
591    }
