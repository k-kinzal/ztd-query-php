10/**
11 * Every table the catalog names, with the statements that read and write it.
12 *
13 * Reading a table and writing it are listed side by side because that is the
14 * question a catalog is usually opened with: what puts rows into this table,
15 * and does anything read them back the way they were written.
16 *
17 * @visibility root
18 */

20{

24
25    /**
26     * Wires the page to the rendering it is written with.
27     */
28    public function __construct(?HtmlText $text = null, ?StatementCard $card = null)
29    {
30        $this->text = $text ?? new HtmlText();
31        $this->card = $card ?? new StatementCard($this->text);
32    }
33
34    /**
35     * The body of the tables page.
36     */
37    public function render(ReportSite $site, Catalog $catalog, CatalogStatistics $stats): string
38    {
39        $tables = $stats->tables();
40        if ($tables === []) {
41            return '<h1>Tables</h1><p class="lede">No statement names a table.</p>';
42        }
43        $byTable = $this->group($catalog);
44
45        $sections = '';
46        foreach ($tables as $table) {
47            $sections .= $this->section($site, $table, $byTable[$table['name']] ?? []);
48        }
49
50        return '<h1>Tables <span class="count">' . $this->text->escape($this->text->plural(count($tables), 'table'))
51            . '</span></h1>'
52            . '<p class="lede">Every table the statements name. A table is counted once per statement that names it, '
53            . 'so a join counts for each of its tables.</p>'
54            . $sections;
55    }
56
57    /**
58     * The statements naming each table, in reporting order.
59     *
60     * @return array<string, list<CatalogEntry>>
61     */
62    public function group(Catalog $catalog): array
63    {
64        $grouped = [];
65        foreach ($catalog->sorted() as $entry) {
66            foreach ($entry->tables as $table) {
67                $grouped[$table][] = $entry;
68            }
69        }
70
71        return $grouped;
72    }

74    /**
75     * One table, with the statements that name it.
76     *
77     * @param array{name: string, reads: int, writes: int, statements: int} $table
78     * @param list<CatalogEntry> $entries
79     */
80    public function section(ReportSite $site, array $table, array $entries): string
81    {
82        $rows = '';
83        foreach ($entries as $entry) {
84            $rows .= $this->row($site, $entry);
85        }
86
87        return '<details class="usage" id="table-' . $this->text->escape($this->text->slug($table['name'])) . '">'
88            . '<summary><code>' . $this->text->escape($table['name']) . '</code>'
89            . '<span class="count">' . $this->text->escape($this->text->plural($table['statements'], 'statement'))
90            . ' · ' . $this->text->number($table['reads']) . ' read · '
91            . $this->text->number($table['writes']) . ' written</span></summary>'
92            . '<div class="table-wrap"><table><thead><tr><th class="tight">Kind</th><th>Statement</th>'
93            . '<th class="tight">Where</th></tr></thead><tbody>' . $rows . '</tbody></table></div></details>';
94    }

99    public function row(ReportSite $site, CatalogEntry $entry): string
100    {

104            . $this->text->escape($this->text->truncate($entry->sql(), 110)) . '</code></a></td>'
105            . '<td class="tight"><span class="stmt-site">' . $this->text->escape($entry->site->display())
106            . '</span></td></tr>';
107    }
