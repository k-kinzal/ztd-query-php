53    public function cards(CatalogStatistics $stats): string
54    {
55        $cards = [
56            ['', $stats->statements(), 'statements'],
57            ['card-ok', $stats->resolved(), 'fully resolved'],
58            ['card-warn', $stats->open(), 'search left open'],
59            ['card-danger', $stats->atLeast(Severity::High), 'statements needing attention'],
60        ];
61        $written = '';

63            $written .= '<div class="card ' . $role . '"><p class="card-figure">' . $this->text->number($figure)

73    public function resolutions(CatalogStatistics $stats): string
74    {
75        $total = $stats->statements();
76        $rows = '';
77        foreach ($stats->byResolution() as $value => $count) {
78            $resolution = Resolution::from($value);
79            $rows .= '<tr><td class="tight">'
80                . $this->text->chip($value, $this->card->resolutionRole($resolution)) . '</td>'
81                . '<td class="num">' . $this->text->number($count) . '</td>'
82                . '<td class="num">' . $this->text->escape($this->text->percent($count, $total)) . '</td>'
83                . '<td>' . $this->text->escape($resolution->describe())
84                . $this->text->bar($count, $total, $this->barRole($this->card->resolutionRole($resolution)))
85                . '</td></tr>';
86        }
87
88        return '<h2>Resolution</h2><div class="table-wrap"><table><thead><tr>'
89            . '<th class="tight">Resolution</th><th class="num">Statements</th><th class="num">Share</th>'

126                . '<code>' . $this->text->escape($value) . '</code></a></td>'
127                . '<td class="tight">' . $this->text->chip($rule->severity()->value, $this->card->severityRole($rule->severity())) . '</td>'
128                . '<td class="num">' . $this->text->number($count) . '</td>'
129                . '<td>' . $this->text->escape($rule->describe()) . '</td></tr>';
130        }
131
132        return '<h2>Findings <span class="count">' . $this->text->escape($this->text->plural($stats->findings(), 'finding'))
133            . '</span></h2><div class="table-wrap"><table><thead><tr><th>Rule</th><th class="tight">Severity</th>'
