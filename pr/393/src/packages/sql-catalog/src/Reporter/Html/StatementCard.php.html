12/**
13 * One statement, written the way the report shows it.
14 *
15 * What the statement is and what the analyzer managed to establish about it are
16 * shown as two different things. The kind carries an identity hue because it is
17 * a property of the statement; the resolution and the findings carry state hues
18 * because they are judgements about the reading. A call that no statement was
19 * read from is not dressed up as a statement at all.
20 *
21 * @visibility root
22 */
23final class StatementCard
24{
25    private HtmlText $text;
26
27    private SqlHighlighter $sql;

29    /**
30     * Wires the card to the text and SQL rendering it is written with.
31     */
32    public function __construct(?HtmlText $text = null, ?SqlHighlighter $sql = null)
33    {
34        $this->text = $text ?? new HtmlText();
35        $this->sql = $sql ?? new SqlHighlighter($this->text);
36    }

38    /**
39     * The whole statement as one card.
40     */
41    public function render(CatalogEntry $entry): string
42    {
43        return '<article class="stmt" id="' . $this->text->escape($entry->id) . '" data-severity="'
44            . $this->text->escape($entry->severity()->value) . '">'
45            . $this->head($entry)
46            . $this->body($entry)
47            . $this->tables($entry)
48            . $this->caveats($entry)
49            . '<div class="stmt-blocks">' . $this->values($entry) . $this->findings($entry) . '</div>'
50            . '</article>';
51    }

53    /**
54     * The line naming what the statement is and where it is issued.
55     */
56    public function head(CatalogEntry $entry): string
57    {
58        $resolution = $entry->resolution();
59
60        return '<header class="stmt-head">'
61            . $this->text->chip(strtoupper($entry->kind->value), $this->kindRole($entry->kind->value))
62            . $this->text->chip($resolution->value, $this->resolutionRole($resolution), $resolution->describe())
63            . ($entry->severity()->atLeast(Severity::Medium)
64                ? $this->text->chip($entry->severity()->value, $this->severityRole($entry->severity()))
65                : '')
66            . '<span class="stmt-site">' . $this->text->escape($entry->site->display()) . '</span>'
67            . '<span class="stmt-fn">' . $this->text->escape($entry->site->function) . '</span>'
68            . '<a class="anchor" href="#' . $this->text->escape($entry->id) . '" title="'
69            . $this->text->escape($entry->id) . '">#</a>'
70            . '</header>';
71    }

73    /**
74     * The statement itself, or the call it was not read from.
75     */
76    public function body(CatalogEntry $entry): string
77    {
78        if ($entry->resolution() !== Resolution::NotAnalyzed) {
79            return '<pre class="sql">' . $this->sql->render($entry->parts()) . '</pre>';
80        }
81        $written = $entry->firstGap()?->expression;
82
83        return '<pre class="sql"><span class="tok-com">-- no statement was read from this call</span>'
84            . ($written === null ? '' : "\n" . $this->text->escape($written)) . '</pre>';
85    }

87    /**
88     * The tables the statement names, and the call it goes through.
89     */
90    public function tables(CatalogEntry $entry): string
91    {
92        $chips = '';
93        foreach ($entry->tables as $table) {
94            $chips .= $this->text->chip($table, 'chip-ghost') . ' ';
95        }
96        if ($entry->tables === [] && $entry->resolution() !== Resolution::NotAnalyzed) {
97            $chips = '<span class="none">no table named</span> ';
98        }
99
100        return '<div class="stmt-tables">' . $chips
101            . $this->text->chip($entry->site->sink, 'chip-sm', 'The database call that was matched') . '</div>';
102    }

104    /**
105     * What a reader has to know before trusting the statement.
106     */
107    public function caveats(CatalogEntry $entry): string
108    {
109        $caveats = [];
110        if ($entry->resolution() === Resolution::NotAnalyzed) {
111            $caveats[] = $entry->resolution()->describe();
112        } elseif (!$entry->resolution()->isClosed()) {
113            $caveats[] = $entry->resolution()->describe() . ' The statements listed here may not be all of them.';
114        } elseif ($entry->truncated) {
115            $caveats[] = 'A limit on loop passes or on callers cut the search short. The statements listed here may not be all of them.';
116        }
117        if (!$entry->correlated) {
118            $caveats[] = 'Assembled from parts that vary independently, so some of these may be unreachable.';
119        }
120        if (count($entry->through) > 1) {
121            $caveats[] = 'Read through ' . implode(' → ', $entry->through) . '.';
122        }
123        if ($caveats === []) {
124            return '';
125        }
126        $items = '';
127        foreach ($caveats as $caveat) {
128            $items .= '<li>' . $this->text->escape($caveat) . '</li>';
129        }
130
131        return '<div class="notice notice-warn"><ul>' . $items . '</ul></div>';
132    }

134    /**
135     * The bind parameters of the statement, with what each is bound to.
136     */
137    public function values(CatalogEntry $entry): string
138    {
139        if ($entry->placeholders === []) {
140            return '';
141        }
142        $rows = '';
143        foreach ($entry->placeholders as $placeholder) {
144            $rows .= $this->valueRow($placeholder);
145        }
146
147        return '<section class="stmt-block"><h4>Values</h4><div class="table-wrap"><table>'
148            . '<thead><tr><th class="tight">Parameter</th><th class="tight">Type</th><th>Bound to</th></tr></thead>'
149            . '<tbody>' . $rows . '</tbody></table></div></section>';
150    }

152    /**
153     * One bind parameter as a row.
154     */
155    public function valueRow(Placeholder $placeholder): string
156    {
157        $value = $placeholder->value;
158        $bound = $value === null
159            ? '<span class="none">unbound</span>'
160            : ($value->isResolved()
161                ? '<code>' . $this->text->escape($value->display()) . '</code>'
162                : '<span class="muted">' . $this->text->escape($this->openValue($value->origins)) . '</span>');
163
164        return '<tr><td class="tight"><code>' . $this->text->escape($placeholder->token) . '</code></td>'
165            . '<td class="tight"><code>' . $this->text->escape($value === null ? '?' : $value->type) . '</code></td>'
166            . '<td>' . $bound . '</td></tr>';
167    }

169    /**
170     * How a value that did not resolve to a set of alternatives is described.
171     *
172     * @param list<string> $origins
173     */
174    public function openValue(array $origins): string
175    {
176        return $origins === [] ? 'not pinned down' : 'not pinned down: ' . implode(', ', $origins);
177    }

179    /**
180     * What is worth reporting about the statement.
181     */
182    public function findings(CatalogEntry $entry): string
183    {
184        if ($entry->findings === []) {
185            return '';
186        }
187        $items = '';
188        foreach ($entry->findings as $finding) {
189            $items .= '<li>' . $this->text->chip($finding->severity->value, $this->severityRole($finding->severity))
190                . '<span>' . $this->text->escape($finding->message) . '</span></li>';
191        }
192
193        return '<section class="stmt-block"><h4>Findings</h4><ul class="finding-list">' . $items . '</ul></section>';
194    }

196    /**
197     * The identity hue a statement of that kind is written in.
198     */
199    public function kindRole(string $kind): string
200    {
201        return match ($kind) {
202            'select' => 'k-select',
203            'insert', 'replace', 'merge' => 'k-insert',
204            'update' => 'k-update',
205            'delete' => 'k-delete',
206            'create', 'alter', 'drop', 'truncate' => 'k-schema',
207            default => 'k-other',
208        };
209    }
210
211    /**
212     * The state hue a resolution is written in.
213     */
214    public function resolutionRole(Resolution $resolution): string
215    {
216        return match ($resolution) {
217            Resolution::Resolved => 's-ok',
218            Resolution::ExternalInput => 's-danger',
219            Resolution::IncompleteModel, Resolution::Incomplete => 's-warn',
220            Resolution::NotAnalyzed => 's-neutral',
221        };
222    }
223
224    /**
225     * The state hue a severity is written in.
226     */
227    public function severityRole(Severity $severity): string
228    {
229        return match ($severity) {
230            Severity::High => 's-danger',
231            Severity::Medium => 's-warn',
232            Severity::Low, Severity::Info => 's-neutral',
233        };
234    }
