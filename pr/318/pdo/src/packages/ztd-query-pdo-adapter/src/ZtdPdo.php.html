102     * Detect the appropriate SessionFactory based on the PDO driver name.
103     *
104     * @throws RuntimeException If the driver is unsupported or the required platform package is not installed.

106    private static function detectFactory(PDO $pdo): SessionFactory
107    {
108        /** @var string $driver */
109        $driver = $pdo->getAttribute(PDO::ATTR_DRIVER_NAME);
110
111        if (!isset(self::DRIVER_MAP[$driver])) {
112            throw new RuntimeException(sprintf(
113                'Unsupported PDO driver: "%s". Supported drivers: %s.',
114                $driver,
115                implode(', ', array_keys(self::DRIVER_MAP))
116            ));
117        }
118
119        $mapping = self::DRIVER_MAP[$driver];
120        /** @var class-string<SessionFactory> $className */
121        $className = $mapping['class'];
122        $packageName = $mapping['package'];
123
124        if (!class_exists($className)) {
125            throw new RuntimeException(sprintf(
126                'Platform package for PDO driver "%s" is not installed. Install it with: composer require %s',
127                $driver,
128                $packageName
129            ));
130        }
131
132        return new $className();
133    }

165    public function prepare(string $query, array $options = []): PDOStatement|false
166    {
167        if (!$this->session->isEnabled()) {
168            return $this->pdo->prepare($query, $options);
169        }
170
171        $this->guardRawPostgreSqlCopy($query);
172
173        try {
174            $execution = new PdoPreparedExecution($this->pdo, $this->session, $query, $options);
175            $prepared = $execution->prepare(null);
176        } catch (DatabaseException $e) {
177            throw new ZtdPdoException($e->getMessage(), 0, $e);
178        }
179
180        $defaultFetchMode = $this->pdo->getAttribute(PDO::ATTR_DEFAULT_FETCH_MODE);
181
182        return new ZtdPdoStatement(
183            $prepared['statement'],
184            $this->session,
185            $prepared['plan'],
186            $execution,
187            is_int($defaultFetchMode) ? $defaultFetchMode : PDO::FETCH_BOTH,
188        );
189    }

194     * @throws ZtdPdoException When ZTD-specific exception occurs (wraps DatabaseException).

196    public function query(string $query, ?int $fetchMode = null, mixed ...$fetchModeArgs): PDOStatement|false
197    {
198        if ($this->session->isEnabled()) {
199            $transactionStatement = $this->session->transactionStatement($query);
200            if ($transactionStatement !== null) {
201                $statement = $this->pdo->query($query, $fetchMode, ...$fetchModeArgs);
202                if ($statement !== false) {
203                    $this->session->applyTransactionStatement($transactionStatement);
204                }
205
206                return $statement;
207            }
208        }
209
210        $stmt = $this->prepare($query);
211        if ($stmt === false) {
212            return false;
213        }
214
215        if ($fetchMode !== null) {
216            $stmt->setFetchMode($fetchMode, ...$fetchModeArgs);
217        }
218
219        if (!$stmt->execute()) {
220            return false;
221        }
222
223        return $stmt;
224    }

231    public function exec(string $statement): int|false
232    {
233        if (!$this->session->isEnabled()) {
234            return $this->pdo->exec($statement);
235        }
236
237        $statements = $this->session->splitStatements($statement);
238        if (count($statements) > 1) {
239            return $this->execMultiple($statements);
240        }
241
242        $this->guardRawPostgreSqlCopy($statement);
243
244        $transactionStatement = $this->session->transactionStatement($statement);
245        if ($transactionStatement !== null) {
246            $result = $this->pdo->exec($statement);
247            if ($result !== false) {
248                $this->session->applyTransactionStatement($transactionStatement);
249            }
250
251            return $result;
252        }
253
254        try {
255            return $this->session->execStatement($statement);
256        } catch (DatabaseException $e) {
257            throw new ZtdPdoException($e->getMessage(), 0, $e);
258        }
259    }

264    private function execMultiple(array $statements): int|false
265    {
266        $affectedRows = $this->exec($statements[0]);
267        if ($affectedRows === false) {
268            return false;
269        }
270
271        foreach (array_slice($statements, 1) as $statement) {
272            $result = $this->exec($statement);
273            if ($result === false) {
274                return false;
275            }
276            $affectedRows = $result;
277        }

279        return $affectedRows;

506    /** @return array<int, string>|false */
507    private function copyToArrayThroughZtd(
508        string $tableName,
509        string $separator,
510        string $nullAs,
511        ?string $fields,
512    ): array|false {
513        [$copy, $target] = $this->postgreSqlCopyTarget($tableName, $fields);
514        $statement = $this->query($copy->selectSql($target));
515        if ($statement === false) {
516            return false;
517        }
518
519        $rows = [];
520        while (($values = $statement->fetch(PDO::FETCH_NUM)) !== false) {
521            if (!is_array($values)) {
522                throw new ZtdPdoException('PostgreSQL COPY query returned an invalid row.');
523            }
524            $rows[] = $copy->encodeRow(array_values($values), $separator, $nullAs);
525        }
526
527        return $rows;
528    }

530    /** @param array<mixed>|\Traversable<mixed> $rows */
531    private function copyFromArrayThroughZtd(
532        string $tableName,
533        array|\Traversable $rows,
534        string $separator,
535        string $nullAs,
536        ?string $fields,
537    ): bool {
538        [$copy, $target] = $this->postgreSqlCopyTarget($tableName, $fields);
539        $decodedRows = [];
540        foreach ($rows as $row) {
541            if (!is_string($row)) {
542                throw new \TypeError(sprintf('PostgreSQL COPY rows must be strings, %s given.', get_debug_type($row)));
543            }
544            $decodedRow = $copy->decodeRow($row, $separator, $nullAs);
545            if (count($decodedRow) !== count($target->columns)) {
546                throw new \ValueError(sprintf(
547                    'PostgreSQL COPY row has %d fields, but %d fields are required.',
548                    count($decodedRow),
549                    count($target->columns),
550                ));
551            }
552            $decodedRows[] = $decodedRow;
553        }
554        if ($decodedRows === []) {
555            return true;
556        }
557
558        $parameters = [];
559        foreach ($decodedRows as $parameterValues) {
560            foreach ($parameterValues as $value) {
561                $parameters[] = $value;
562            }
563        }
564
565        $statement = $this->prepare($copy->insertSql(
566            $target,
567            count($decodedRows),
568            !$this->session->isEnabled(),
569        ));
570
571        return $statement !== false && $statement->execute($parameters);
572    }

23 *
24 * Uses delegation pattern: extends PDO for type compatibility,
25 * but delegates all operations to an inner PDO instance when using fromPdo().
26 *
27 * Supports multiple database platforms via SessionFactory injection or auto-detection:
28 * - mysql  -> MySqlSessionFactory  (k-kinzal/ztd-query-mysql)
29 * - pgsql  -> PgSqlSessionFactory  (k-kinzal/ztd-query-postgres)
30 * - sqlite -> SqliteSessionFactory (k-kinzal/ztd-query-sqlite)
31 *
32 * @example Simulate a write without changing the physical table
33 *     $native = new \PDO('sqlite::memory:');
34 *     $native->exec('CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT)');
35 *     $pdo = \ZtdQuery\Adapter\Pdo\ZtdPdo::fromPdo($native);
36 *     $pdo->exec("INSERT INTO users VALUES (1, 'Alice')") // => 1
37 *     $pdo->query('SELECT name FROM users')->fetchColumn() // => 'Alice'

95     * @throws RuntimeException When the driver has no platform package installed
96     *
97     * @visibility public
98     * @example Wrap an existing PDO connection
99     *     $pdo = \ZtdQuery\Adapter\Pdo\ZtdPdo::fromPdo(new \PDO('sqlite::memory:'));
100     *     $pdo->isZtdEnabled() // => true
101     *     $pdo->getAttribute(\PDO::ATTR_DRIVER_NAME) // => 'sqlite'
102     */
103    public static function fromPdo(PDO $pdo, ?ZtdConfig $config = null, ?SessionFactory $factory = null): static
104    {
105        /** @var static $instance */
106        $instance = (new ReflectionClass(static::class))->newInstanceWithoutConstructor();
107        $instance->pdo = $pdo;
108
109        $resolvedFactory = $factory ?? (new DriverSessionFactory())->forConnection($instance->pdo);
110        $connection = new PdoConnection($instance->pdo);
111        $instance->session = $resolvedFactory->create($connection, $config ?? ZtdConfig::default());
112        $instance->copy = new PostgreSqlCopy($instance->session);
113
114        return $instance;
115    }

156    /**
157     * {@inheritDoc}
158     *
159     * @param array<mixed> $options Driver options, as PDO::prepare() takes them
160     *
161     * @return PDOStatement|false The prepared statement, or false where the driver would not prepare one
162     *
163     * @throws ZtdPdoException When ZTD cannot carry the statement out, or an option is one PDO cannot be given
164     *
165     * @example Bind values to a simulated INSERT
166     *     $native = new \PDO('sqlite::memory:');
167     *     $native->exec('CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT)');
168     *     $pdo = \ZtdQuery\Adapter\Pdo\ZtdPdo::fromPdo($native);
169     *     $statement = $pdo->prepare('INSERT INTO users VALUES (:id, :name)');
170     *     $statement->execute(['id' => 1, 'name' => 'Alice']) // => true
171     *     $statement->rowCount() // => 1
172     *     $native->query('SELECT COUNT(*) FROM users')->fetchColumn() // => 0
173     */
174    #[Override]
175    public function prepare(string $query, array $options = []): PDOStatement|false
176    {
177        if (!$this->session->isEnabled()) {
178            return $this->pdo->prepare($query, $options);
179        }
180
181        $this->copy->guardRaw($query);
182
183        $execution = new PdoPreparedExecution($this->pdo, $this->session, $query, $options);
184        $prepared = $execution->prepare(null);
185
186        $defaultFetchMode = $this->pdo->getAttribute(PDO::ATTR_DEFAULT_FETCH_MODE);
187
188        return new ZtdPdoStatement(
189            $prepared['statement'],
190            $this->session,
191            $prepared['plan'],
192            $execution,
193            is_int($defaultFetchMode) ? $defaultFetchMode : PDO::FETCH_BOTH,
194        );
195    }

197    /**
198     * {@inheritDoc}
199     *
200     * @param mixed ...$fetchModeArgs The rest of what the fetch mode reads
201     *
202     * @return PDOStatement|false The executed statement, or false where it did not run
203     *
204     * @throws ZtdPdoException When ZTD cannot carry the statement out
205     */
206    #[Override]
207    public function query(string $query, ?int $fetchMode = null, mixed ...$fetchModeArgs): PDOStatement|false
208    {
209        if ($this->session->isEnabled()) {
210            $transactionStatement = $this->session->transactionStatement($query);
211            if ($transactionStatement !== null) {
212                $statement = $this->pdo->query($query, $fetchMode, ...$fetchModeArgs);
213                if ($statement !== false) {
214                    $this->session->applyTransactionStatement($transactionStatement);
215                }
216
217                return $statement;
218            }
219        }
220
221        $stmt = $this->prepare($query);
222        if ($stmt === false) {
223            return false;
224        }
225
226        if ($fetchMode !== null) {
227            $stmt->setFetchMode($fetchMode, ...$fetchModeArgs);
228        }
229
230        if (!$stmt->execute()) {
231            return false;
232        }
233
234        return $stmt;
235    }

237    /**
238     * {@inheritDoc}
239     *
240     * A batch is carried out one statement at a time, and stops at the first
241     * one that does not run; what it answers is what the last one that ran
242     * affected, which is what PDO answers for a batch.
243     *
244     * @return int|false Rows the statement affected, or false where it did not run
245     *
246     * @throws ZtdPdoException When ZTD cannot carry the statement out
247     */
248    #[Override]
249    public function exec(string $statement): int|false
250    {
251        if (!$this->session->isEnabled()) {
252            return $this->pdo->exec($statement);
253        }
254
255        $statements = $this->session->splitStatements($statement);
256        if (count($statements) > 1) {
257            $affectedRows = 0;
258            foreach ($statements as $one) {
259                $result = $this->exec($one);
260                if ($result === false) {
261                    return false;
262                }
263                $affectedRows = $result;
264            }
265
266            return $affectedRows;
267        }
268
269        $this->copy->guardRaw($statement);
270
271        $transactionStatement = $this->session->transactionStatement($statement);
272        if ($transactionStatement !== null) {
273            $result = $this->pdo->exec($statement);
274            if ($result !== false) {
275                $this->session->applyTransactionStatement($transactionStatement);
276            }
277
278            return $result;
279        }
280
281        try {
282            return $this->session->execStatement($statement);
283        } catch (DatabaseException $e) {
284            throw new ZtdPdoException($e->getMessage(), 0, $e);
285        }
286    }
