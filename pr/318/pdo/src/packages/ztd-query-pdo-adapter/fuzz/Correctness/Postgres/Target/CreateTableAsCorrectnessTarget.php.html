152    /**
153     * Answers every row the connection reads.

155     * @param PDO $pdo The pdo

162    public function fetchAll(PDO $pdo, string $query): array
163    {
164        $statement = $pdo->query($query);
165        if ($statement === false) {
166            return [];
167        }
168
169        $rows = [];
170        foreach ($statement->fetchAll(PDO::FETCH_ASSOC) as $row) {
171            if (!is_array($row)) {
172                throw new Error('CTAS correctness query returned a non-array row');
173            }
174            $normalized = [];
175            foreach ($row as $column => $value) {
176                if (!is_string($column)) {
177                    throw new Error('CTAS correctness query returned a non-string column');
178                }
179                if ($value !== null && !is_scalar($value)) {
180                    throw new Error('CTAS correctness query returned a value no comparison can read');
181                }
182                $normalized[$column] = $value;
183            }
184            $rows[] = $normalized;
185        }
186
187        return $rows;
