450    private function extractTableName(string $createSql): ?string
451    {
452        if (preg_match('/CREATE\s+(?:TEMPORARY\s+)?TABLE\s+(?:IF\s+NOT\s+EXISTS\s+)?(?:"([^"]+)"|`([^`]+)`|\[([^\]]+)\]|(\S+))\s*\(/i', $createSql, $m) !== 1) {
453            return null;
454        }
455
456        return $m[1] !== '' ? $m[1] : ($m[2] !== '' ? $m[2] : ($m[3] !== '' ? $m[3] : $m[4]));
