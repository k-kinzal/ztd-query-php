289    private static function hasValidTableOptions(string $suffix): bool
290    {
291        $tokens = SqlTokenStream::tokenize($suffix, SqliteLexerProfile::create())->significantTokens();
292        $last = $tokens[count($tokens) - 1] ?? null;
293        if ($last !== null
294            && $last->kind === SqlTokenKind::Symbol
295            && $last->text === ';'
296        ) {
297            array_pop($tokens);
298        }
299        if ($tokens === []) {
300            return true;
301        }
302
303        $seen = [];
304        $index = 0;
305        while ($index < count($tokens)) {
306            if ($tokens[$index]->isKeyword('STRICT')) {
307                $option = 'STRICT';
308                $index++;
309            } elseif ($tokens[$index]->isKeyword('WITHOUT')
310                && ($tokens[$index + 1] ?? null)?->isKeyword('ROWID') === true
311            ) {
312                $option = 'WITHOUT ROWID';
313                $index += 2;
314            } else {
315                return false;
316            }
317
318            if (isset($seen[$option])) {
319                return false;
320            }
321            $seen[$option] = $option;
322
323            if ($index === count($tokens)) {
324                return true;
325            }
326            if ($tokens[$index]->kind !== SqlTokenKind::Symbol
327                || $tokens[$index]->text !== ','
328            ) {
329                return false;
330            }
331            $index++;
332        }
333
334        return false;
335    }

349     *
350     * @return array<int, string>

352    private function splitColumnDefinitions(string $body): array
353    {
354        $definitions = [];
355        $current = '';
356        $depth = 0;
357        $len = strlen($body);
358        $inQuote = '';
359
360        for ($i = 0; $i < $len; $i++) {
361            $char = $body[$i];
362
363            if ($inQuote !== '') {
364                $current .= $char;
365                if ($char === $inQuote) {
366                    if ($i + 1 < $len && $body[$i + 1] === $inQuote) {
367                        $current .= $body[$i + 1];
368                        $i++;
369                    } else {
370                        $inQuote = '';
371                    }
372                }
373                continue;
374            }
375
376            if ($char === '\'' || $char === '"') {
377                $inQuote = $char;
378                $current .= $char;
379                continue;
380            }
381
382            if ($char === '(') {
383                $depth++;
384                $current .= $char;
385                continue;
386            }
387
388            if ($char === ')') {
389                if ($depth > 0) {
390                    $depth--;
391                }
392                $current .= $char;
393                continue;
394            }
395
396            if ($char === ',' && $depth === 0) {
397                $definitions[] = trim($current);
398                $current = '';
399                continue;
400            }
401
402            $current .= $char;
403        }
404
405        $last = trim($current);
406        if ($last !== '') {
407            $definitions[] = $last;
408        }
409
410        return $definitions;
