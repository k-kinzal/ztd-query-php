119    private function rewriteStatement(string $stmtSql, string $originalSql): RewritePlan
120    {
121        if (SqliteInMemoryAttachStatement::isSafe($stmtSql)) {
122            return new RewritePlan($stmtSql, QueryKind::READ);
123        }
124        if (SqliteReadOnlyDiagnosticStatement::isSafe($stmtSql)) {
125            return new RewritePlan($stmtSql, QueryKind::READ);
126        }
127        $kind = $this->guard->classify($stmtSql);
128        if ($kind === null) {
129            throw new UnsupportedSqlException($originalSql, 'Statement type not supported');
130        }
131
132        $tableContext = $this->buildTableContext();
133
134        if ($kind === QueryKind::READ) {
135            if ($this->hasSchemaContext()) {
136                $unknownTable = $this->findUnknownTable($stmtSql);
137                if ($unknownTable !== null) {
138                    throw new UnknownSchemaException($originalSql, $unknownTable, 'table');
139                }
140            }
141
142            $transformedSql = $this->transformer->transform($stmtSql, $tableContext);
143
144            return new RewritePlan($transformedSql, QueryKind::READ);
145        }
146
147        if ($kind === QueryKind::DDL_SIMULATED) {
148            $mutation = $this->mutationResolver->resolve($stmtSql, $kind);
149            if ($mutation instanceof AlterTableMutation) {
150                return new RewritePlan(
151                    $this->transformer->transform($mutation->resultSelect(), $tableContext),
152                    QueryKind::DDL_SIMULATED,
153                    $mutation,
154                    affectedRowsMode: AffectedRowsMode::None,
155                );
156            }
157
158            return new RewritePlan($this->emptyResultSelect(), QueryKind::DDL_SIMULATED, $mutation);
159        }
160
161        $mutation = $this->mutationResolver->resolve($stmtSql, $kind);
162
163        $transformedSql = $this->transformer->transform($stmtSql, $tableContext);
164
165        return new RewritePlan(
166            $transformedSql,
167            QueryKind::WRITE_SIMULATED,
168            $mutation,
169            $this->returningProjectionParser->parse($stmtSql),
170            AffectedRowsMode::Matched,
171        );
172    }

122    {
190        $context = [];
191        $allData = $this->shadowStore->getAll();
192
193        foreach ($allData as $tableName => $rows) {
194            $definition = $this->registry->get($tableName);
195            if ($definition !== null) {
196                $context[$tableName] = self::contextFromDefinition($definition, $rows);
197            } else {
198                $columns = [];
199                if ($rows !== []) {
200                    $columns = array_keys($rows[0]);
201                    foreach ($rows as $row) {
202                        foreach (array_keys($row) as $column) {
203                            if (!in_array($column, $columns, true)) {
204                                $columns[] = $column;
205                            }
206                        }
207                    }
208                }
209
210                $context[$tableName] = [
211                    'rows' => $rows,
212                    'columns' => $columns,
213                    'columnTypes' => [],
214                    'columnDefaults' => [],
215                    'identityStrategies' => [],
216                    'generatedExpressions' => [],
217                    'primaryKeys' => [],
218                    'candidateKeys' => [],
219                ];
220            }
221        }
222
223        $allDefinitions = $this->registry->getAll();
224        foreach ($allDefinitions as $tableName => $definition) {
225            if (isset($context[$tableName])) {
226                continue;
227            }
228
229            $context[$tableName] = self::contextFromDefinition($definition, []);
230        }
231        foreach ($this->registry->getAllRemoved() as $tableName => $definition) {
232            $context[$tableName] = self::contextFromDefinition($definition, []);
233        }
234
235        foreach ((new SqliteViewShadowRenderer())->render($this->views, array_keys($context)) as $viewName => $viewSql) {
236            if (isset($context[$viewName])) {
237                continue;
238            }
239            $context[$viewName] = ['viewSql' => $viewSql];
240        }
241
242        return $context;
123        $this->transformer->commitRewriteState();
124    }

272    private function findUnknownTable(string $sql): ?string
273    {
274        $type = $this->parser->classifyStatement($sql);
275        if ($type !== 'SELECT') {
276            return null;
277        }
278
279        $tableNames = $this->parser->extractSelectTables($sql);
280        $declaredCtes = array_fill_keys($this->cteComposer->declaredCteNames($sql), true);
281
282        foreach ($tableNames as $tableName) {
283            if (isset($declaredCtes[strtolower($tableName)])) {
284                continue;
285            }
286            if (!$this->tableExists($tableName)) {
287                return $tableName;
288            }
289        }
290
291        return null;
292    }
