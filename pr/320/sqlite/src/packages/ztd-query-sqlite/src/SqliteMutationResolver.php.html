55            'DELETE' => (new Rewriting\Mutation\Resolution\RowMutationResolver($this->parser, $this->registry, $this->shadowStore))->resolveDelete($sql),

58            'DROP_TABLE' => (new Rewriting\Mutation\Resolution\TableMutationResolver($this->parser, $this->registry, $this->schemaParser))->resolveDropTable($sql),

77    private function resolveUpdate(string $sql): ShadowMutation
78    {
79        $targetTable = $this->parser->extractTargetTable($sql);
80        if ($targetTable === null) {
81            throw new UnsupportedSqlException($sql, 'Cannot resolve UPDATE target');
82        }
83        $this->assertTableWasNotRemoved($sql, $targetTable);
84
85        $definition = $this->registry->get($targetTable);
86        if ($definition === null && $this->shadowStore->state($targetTable) !== ShadowTableState::Initialized) {
87            throw new UnknownSchemaException($sql, $targetTable, 'table');
88        }
89
90        $this->shadowStore->ensure($targetTable);
91        $primaryKeys = $definition !== null ? $definition->primaryKeys : [];
92
93        return new UpdateMutation($targetTable, $primaryKeys);
94    }

121    private function resolveInsert(string $sql): ShadowMutation
122    {
123        $tableName = $this->parser->extractTargetTable($sql);
124        if ($tableName === null) {
125            throw new UnsupportedSqlException($sql, 'Cannot resolve INSERT target');
126        }
127        $this->assertTableWasNotRemoved($sql, $tableName);
128
129        if ($this->parser->isReplace($sql)) {
130            $definition = $this->registry->get($tableName);
131            $primaryKeys = $definition !== null ? $definition->primaryKeys : [];
132
133            return new ReplaceMutation($tableName, $primaryKeys, $definition?->candidateKeys());
134        }
135
136        if ($this->parser->hasOnConflict($sql)) {
137            $updateColumns = [];
138            $definition = $this->registry->get($tableName);
139            $databaseEvaluated = $definition !== null && $definition->candidateKeys()->keys() !== [];
140            /** @var array<string, \ZtdQuery\Shadow\Mutation\UpsertExpression|null> $updateValues */
141            $updateValues = [];
142            $expressionParser = new SqliteUpsertExpressionParser();
143            $onConflictUpdates = $this->parser->extractOnConflictUpdates($sql);
144            foreach ($onConflictUpdates as $colName => $value) {
145                $updateColumns[] = $colName;
146                $updateValues[$colName] = $databaseEvaluated
147                    ? $expressionParser->parseIfSupported($value, $tableName)
148                    : $expressionParser->parse($value, $tableName);
149            }
150
151            if ($updateColumns !== []) {
152                $primaryKeys = $definition !== null ? $definition->primaryKeys : [];
153                $predicate = $this->parser->extractOnConflictUpdateWhere($sql);
154
155                return new UpsertMutation(
156                    $tableName,
157                    $primaryKeys,
158                    $updateColumns,
159                    $updateValues,
160                    $definition?->candidateKeys(),
161                    $predicate !== null
162                        ? ($databaseEvaluated
163                            ? $expressionParser->parseIfSupported($predicate, $tableName)
164                            : $expressionParser->parse($predicate, $tableName))
165                        : null,
166                    databaseEvaluated: $databaseEvaluated,
167                    updateSqlValues: $onConflictUpdates,
168                    updateSqlPredicate: $predicate,
169                );
170            }
171
172            $definition = $this->registry->get($tableName);
173            $primaryKeys = $definition !== null ? $definition->primaryKeys : [];
174
175            return new InsertMutation(
176                $tableName,
177                $primaryKeys,
178                true,
179                candidateKeys: $definition?->candidateKeys(),
180            );
181        }
182
183        $isIgnore = $this->parser->isInsertIgnore($sql);
184
185        $definition = $this->registry->get($tableName);
186        $primaryKeys = $isIgnore ? ($definition !== null ? $definition->primaryKeys : []) : [];
187
188        return new InsertMutation(
189            $tableName,
190            $primaryKeys,
191            $isIgnore,
192            candidateKeys: $definition?->candidateKeys(),
193        );
194    }

196    private function resolveCreateTable(string $sql): ShadowMutation
197    {
198        $tableName = $this->parser->extractTargetTable($sql);
199        if ($tableName === null) {
200            throw new UnsupportedSqlException($sql, 'Cannot resolve table name');
201        }
202
203        $ifNotExists = (bool) preg_match('/\bIF\s+NOT\s+EXISTS\b/i', $sql);
204

215    {
216        $tableName = $this->parser->extractTargetTable($sql);
217        if ($tableName === null) {
218            throw new UnsupportedSqlException($sql, 'Cannot resolve table name');
219        }
220
221        $ifExists = (bool) preg_match('/\bIF\s+EXISTS\b/i', $sql);
222
223        if ($this->registry->isRemoved($tableName)) {
224            if ($ifExists) {

226            }
227            throw new UnsupportedSqlException($sql, 'Table was removed from the virtual schema');
228        }

232        }
233
234        return new DropTableMutation($tableName, $this->registry, $sql, $ifExists);

242    {
243        $tableName = $this->parser->extractTargetTable($sql);
244        if ($tableName === null) {
245            throw new UnsupportedSqlException($sql, 'Cannot resolve table name');
246        }
247
248        $this->assertTableWasNotRemoved($sql, $tableName);
249
250        if (!$this->registry->has($tableName)) {
251            throw new UnknownSchemaException($sql, $tableName, 'table');
252        }
253
254        $operation = $this->alterOperation($sql);
255        if ($operation === null) {
256            throw new UnsupportedSqlException($sql, 'Unsupported ALTER TABLE operation');
257        }
258

260            'add' => $this->resolveAlterAddColumn($sql, $tableName, $operation['clause']),
261            'drop' => $this->resolveAlterDropColumn($sql, $tableName, $operation['clause']),
262            'rename_table' => $this->resolveAlterRenameTable($sql, $tableName, $operation['clause']),
263            'rename_column' => $this->resolveAlterRenameColumn($sql, $tableName, $operation['clause']),
264        };

267    private function resolveAlterAddColumn(string $sql, string $tableName, string $columnSql): ShadowMutation
268    {
269        $existing = $this->definition($sql, $tableName);
270        $added = $this->schemaParser->parse('CREATE TABLE "__ztd_alter" (' . $columnSql . ')');
271        if ($added === null) {
272            throw new UnsupportedSqlException($sql, 'Cannot parse ADD COLUMN');
273        }
274        if (count($added->columns) !== 1) {
275            throw new UnsupportedSqlException($sql, 'Cannot parse ADD COLUMN');
276        }
277
278        $columnName = $added->columns[0];
279        if ($this->existingColumn($existing, $columnName) !== null) {
280            throw new ColumnAlreadyExistsException($sql, $tableName, $columnName);
281        }
282
283        $foreignKeys = $existing->foreignKeys;
284        foreach ($added->foreignKeys as $name => $foreignKey) {
285            while (isset($foreignKeys[$name])) {
286                $name .= '_added';
287            }
288            $foreignKeys[$name] = $foreignKey;
289        }
290
291        $definition = new TableDefinition(
292            [...$existing->columns, $columnName],
293            array_merge($existing->columnTypes, $added->columnTypes),
294            $existing->primaryKeys,
295            [...$existing->notNullColumns, ...$added->notNullColumns],
296            $existing->uniqueConstraints,
297            array_merge($existing->typedColumns, $added->typedColumns),
298            array_merge($existing->columnDefaults, $added->columnDefaults),
299            $existing->identityStrategies,
300            array_merge($existing->generatedExpressions, $added->generatedExpressions),
301            $foreignKeys,
302        );
303        $projection = $this->quotedColumns($existing->columns);
304        $projection[] = ($added->columnDefaults[$columnName] ?? 'NULL')
305            . ' AS ' . $this->quote($columnName);
306
307        return $this->alterMutation($sql, $tableName, $tableName, $definition, $projection);
308    }

310    private function resolveAlterDropColumn(string $sql, string $tableName, string $columnClause): ShadowMutation
311    {
312        $requestedName = $this->singleIdentifier($columnClause);
313        if ($requestedName === null) {
314            throw new UnsupportedSqlException($sql, 'Cannot parse DROP COLUMN');
315        }
316
317        $existing = $this->definition($sql, $tableName);
318        $columnName = $this->existingColumn($existing, $requestedName);
319        if ($columnName === null) {
320            throw new ColumnNotFoundException($sql, $tableName, $requestedName);
321        }
322
323        $newColumns = self::withoutColumn($existing->columns, $columnName);
324        $newColumnTypes = self::withoutMapKey($existing->columnTypes, $columnName);
325        $newTypedColumns = self::withoutMapKey($existing->typedColumns, $columnName);
326        $newDefaults = self::withoutMapKey($existing->columnDefaults, $columnName);
327        $newIdentityStrategies = self::withoutMapKey($existing->identityStrategies, $columnName);
328        $newGeneratedExpressions = self::withoutMapKey($existing->generatedExpressions, $columnName);
329        $newForeignKeys = array_filter(
330            $existing->foreignKeys,
331            static fn (\ZtdQuery\Schema\ForeignKeyDefinition $foreignKey): bool => !in_array(
332                $columnName,
333                $foreignKey->columns,
334                true,
335            ),
336        );
337        $newNotNull = self::withoutColumn($existing->notNullColumns, $columnName);
338        $newPrimaryKeys = self::withoutColumn($existing->primaryKeys, $columnName);
339        $newUniqueConstraints = [];
340        foreach ($existing->uniqueConstraints as $name => $columns) {
341            $filtered = self::withoutColumn($columns, $columnName);
342            if ($filtered !== []) {
343                $newUniqueConstraints[$name] = $filtered;
344            }
345        }
346
347        $definition = new TableDefinition(
348            $newColumns,
349            $newColumnTypes,
350            $newPrimaryKeys,
351            $newNotNull,
352            $newUniqueConstraints,
353            $newTypedColumns,
354            $newDefaults,
355            $newIdentityStrategies,
356            $newGeneratedExpressions,
357            $newForeignKeys,
358        );
359
360        return $this->alterMutation(
361            $sql,
362            $tableName,
363            $tableName,
364            $definition,
365            $this->quotedColumns($newColumns),
366        );
367    }

387    private function resolveAlterRenameColumn(string $sql, string $tableName, string $columnClause): ShadowMutation
388    {
389        $renamed = $this->renamedIdentifiers($columnClause);
390        if ($renamed === null) {
391            throw new UnsupportedSqlException($sql, 'Cannot parse RENAME COLUMN');
392        }
393
394        [$requestedName, $newName] = $renamed;
395        $existing = $this->definition($sql, $tableName);
396        $oldName = $this->existingColumn($existing, $requestedName);
397        if ($oldName === null) {
398            throw new ColumnNotFoundException($sql, $tableName, $requestedName);
399        }
400        $duplicate = $this->existingColumn($existing, $newName);
401        if ($duplicate !== null && strcasecmp($duplicate, $oldName) !== 0) {
402            throw new ColumnAlreadyExistsException($sql, $tableName, $newName);
403        }
404
405        $newColumns = self::renamedColumns($existing->columns, $oldName, $newName);
406        $newUniqueConstraints = [];
407        foreach ($existing->uniqueConstraints as $name => $columns) {
408            $newUniqueConstraints[$name] = self::renamedColumns($columns, $oldName, $newName);
409        }
410
411        $definition = new TableDefinition(
412            $newColumns,
413            self::renamedMapKey($existing->columnTypes, $oldName, $newName),
414            self::renamedColumns($existing->primaryKeys, $oldName, $newName),
415            self::renamedColumns($existing->notNullColumns, $oldName, $newName),
416            $newUniqueConstraints,
417            self::renamedMapKey($existing->typedColumns, $oldName, $newName),
418            self::renamedMapKey($existing->columnDefaults, $oldName, $newName),
419            self::renamedMapKey($existing->identityStrategies, $oldName, $newName),
420            self::renamedMapKey($existing->generatedExpressions, $oldName, $newName),
421            array_map(
422                static fn (ForeignKeyDefinition $foreignKey): ForeignKeyDefinition => self::renamedForeignKey(
423                    $foreignKey,
424                    $oldName,
425                    $newName,
426                ),
427                $existing->foreignKeys,
428            ),
429        );
430        $projection = [];
431        foreach ($existing->columns as $column) {
432            $expression = $this->quote($column);
433            if ($column === $oldName) {
434                $expression .= ' AS ' . $this->quote($newName);
435            }
436            $projection[] = $expression;
437        }
438
439        return $this->alterMutation($sql, $tableName, $tableName, $definition, $projection);
440    }

511    /**
512     * @return array{kind: 'add'|'drop'|'rename_table'|'rename_column', clause: string}|null
513     */
514    private function alterOperation(string $sql): ?array
515    {
516        $stream = SqlTokenStream::tokenize($sql, SqliteLexerProfile::create());
517        $tokens = $stream->significantTokens();
518        $source = $stream->identifierAt(2);
519        if ($source === null) {
520            return null;
521        }
522        $operationIndex = $source['next'];
523        $operation = $tokens[$operationIndex] ?? null;
524        if ($operation === null) {
525            return null;
526        }
527        if ($operation->isKeyword('ADD')) {
528            $clauseIndex = $operationIndex + 1;
529            $clauseStart = $tokens[$clauseIndex] ?? null;
530            if ($clauseStart !== null && $clauseStart->isKeyword('COLUMN')) {
531                ++$clauseIndex;
532            }
533
534            return $this->alterClause($sql, $tokens, 'add', $clauseIndex);
535        }
536        if ($operation->isKeyword('DROP')) {
537            $columnKeyword = $tokens[$operationIndex + 1] ?? null;
538            if ($columnKeyword === null || !$columnKeyword->isKeyword('COLUMN')) {
539                return null;
540            }
541
542            return $this->alterClause($sql, $tokens, 'drop', $operationIndex + 2);
543        }
544        if (!$operation->isKeyword('RENAME')) {
545            return null;
546        }
547
548        $clauseIndex = $operationIndex + 1;
549        $clauseStart = $tokens[$clauseIndex] ?? null;
550        if ($clauseStart === null) {
551            return null;
552        }
553        if ($clauseStart->isKeyword('TO')) {
554            return $this->alterClause($sql, $tokens, 'rename_table', $clauseIndex + 1);
555        }
556        if ($clauseStart->isKeyword('COLUMN')) {
557            ++$clauseIndex;
558        }
559
560        return $this->alterClause($sql, $tokens, 'rename_column', $clauseIndex);
561    }

563    /**
564     * @param list<SqlToken> $tokens
565     * @param 'add'|'drop'|'rename_table'|'rename_column' $kind
566     * @return array{kind: 'add'|'drop'|'rename_table'|'rename_column', clause: string}|null
567     */
568    private function alterClause(string $sql, array $tokens, string $kind, int $startIndex): ?array
569    {
570        $first = $tokens[$startIndex] ?? null;
571        if ($first === null) {
572            return null;
573        }
574        $last = $tokens[count($tokens) - 1];
575        $endOffset = $last->kind === SqlTokenKind::Symbol && $last->text === ';'
576            ? $last->offset
577            : $last->endOffset();
578
579        return [
580            'kind' => $kind,
581            'clause' => substr($sql, $first->offset, $endOffset - $first->offset),
582        ];
583    }

596    /** @return array{string, string}|null */
597    private function renamedIdentifiers(string $sql): ?array
598    {
599        $stream = SqlTokenStream::tokenize($sql, SqliteLexerProfile::create());
600        $tokens = $stream->significantTokens();
601        $old = $stream->identifierAt();
602        if ($old === null) {
603            return null;
604        }
605        $separator = $tokens[$old['next']] ?? null;
606        if ($separator === null || !$separator->isKeyword('TO')) {
607            return null;
608        }
609        $new = $stream->identifierAt($old['next'] + 1);
610        if ($new === null || $new['next'] !== count($tokens)) {
611            return null;
612        }
613
614        return [$old['name'], $new['name']];
615    }

646    {

648        foreach ($columns as $column) {
