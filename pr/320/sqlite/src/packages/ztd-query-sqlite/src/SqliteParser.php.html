401    /**
402     * @return array<int, array{keyword: string, afterGroup: bool, offset: int}>
403     */
404    private function scanTopLevelKeywords(string $sql): array
405    {
406        $keywords = [];
407        $len = strlen($sql);
408        $depth = 0;
409        $completedGroup = false;
410
411        for ($i = 0; $i < $len; $i++) {
412            $char = $sql[$i];
413            $pair = substr($sql, $i, 2);
414
415            if ($pair === '--' || $char === '#') {
416                $commentLength = strcspn($sql, "\r\n", $i);
417                $i += $commentLength;
418                continue;
419            }
420
421            if ($pair === '/*') {
422                $end = strpos($sql, '*/', $i + 2);
423                if ($end === false) {
424                    break;
425                }
426                $i = $end + 1;
427                continue;
428            }
429
430            if ($char === '\'' || $char === '"' || $char === '`') {
431                $quote = $char;
432                while (true) {
433                    $end = strpos($sql, $quote, $i + 1);
434                    if ($end === false) {
435                        $i = $len;
436                        break;
437                    }
438                    $i = $end;
439                    if (str_starts_with(substr($sql, $i), $quote . $quote)) {
440                        $i++;
441                        continue;
442                    }
443                    break;
444                }
445                continue;
446            }
447
448            if ($char === '[') {
449                $end = strpos($sql, ']', $i);
450                $i = $end === false ? $len : $end;
451                continue;
452            }
453
454            if ($char === '(') {
455                $depth++;
456                continue;
457            }
458
459            if ($char === ')') {
460                if ($depth > 0) {
461                    $depth--;
462                    if ($depth === 0) {
463                        $completedGroup = true;
464                    }
465                }
466                continue;
467            }
468
469            $start = $i;
470            $tokenLength = strspn($sql, 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789_$', $i);
471            if ($tokenLength === 0) {
472                continue;
473            }
474
475            $token = substr($sql, $i, $tokenLength);
476            $i += $tokenLength - 1;
477            if ($depth === 0 && strspn($token, 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz') > 0) {
478                $keywords[] = [
479                    'keyword' => strtoupper($token),
480                    'afterGroup' => $completedGroup,
481                    'offset' => $start,
482                ];
483            }
484        }
485
486        return $keywords;
487    }

647        $sets = [];
648        $len = strlen($rest);

650
