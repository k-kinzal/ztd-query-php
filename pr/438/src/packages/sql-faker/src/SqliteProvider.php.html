17/**
18 * Faker Provider for generating syntactically valid SQLite SQL statements.
19 *
20 * This provider uses SQLite's official Lemon grammar (parse.y) to generate
21 * SQL that is syntactically valid. Note that generated SQL may not be semantically
22 * valid (tables/columns may not exist).
23 *
24 * maxDepth selects shortest productions once the depth is reached; it is not a SQL length limit.
25 * The basic pipeline resolves lexical spaces without inserting comments. Optional clauses may be empty.
26 * Seed the supplied Faker generator to reproduce a run within the same package version.
27 *
28 * @visibility public
29 *
30 * @example Register the provider with Faker
31 *     $faker = \Faker\Factory::create();
32 *     $faker->addProvider(new \SqlFaker\SqliteProvider($faker));
33 *     $faker->integerLiteral(min: 42, max: 42) // => '42'
34 */
35final class SqliteProvider extends Base
36{

39    /**
40     * Register SQL formatters on the supplied Faker generator.
41     * @param Generator $generator Faker generator
42     * @param string|null $version SQLite version tag. Null for default.
43     *
44     * @visibility public
45     * @example Choose a supported database version
46     *     $faker = \Faker\Factory::create();

53    {
54        parent::__construct($generator);

56        $generator->addProvider($this);
57        $resolvedVersion = SqliteGrammar::resolveVersion($version);
58        $this->sql = SqlGeneratorFactory::forSqlite($generator, SqliteGrammar::load($resolvedVersion), $resolvedVersion, $coverage);
59    }
