17/**
18 * Faker Provider for generating syntactically valid PostgreSQL SQL statements.
19 *
20 * This provider uses PostgreSQL's official Bison grammar (gram.y) to generate
21 * SQL that is syntactically valid. Note that generated SQL may not be semantically
22 * valid (tables/columns may not exist).
23 *
24 * maxDepth selects shortest productions once the depth is reached; it is not a SQL length limit.
25 * The basic pipeline resolves lexical spaces without inserting comments. Optional clauses may be empty.
26 * Seed the supplied Faker generator to reproduce a run within the same package version.
27 *
28 * @visibility public
29 *
30 * @example Register the provider with Faker
31 *     $faker = \Faker\Factory::create();
32 *     $faker->addProvider(new \SqlFaker\PostgreSqlProvider($faker));
33 *     $faker->parameterMarker(min: 2, max: 2) // => '$2'
34 */
35final class PostgreSqlProvider extends Base
36{
37    private SqlGenerator $sql;

39    /**
40     * Register SQL formatters on the supplied Faker generator.
41     * @param Generator $generator Faker generator
42     * @param string|null $version PostgreSQL version tag. Null for default.
43     *
44     * @visibility public
45     * @example Choose a supported database version
46     *     $faker = \Faker\Factory::create();
47     *     $provider = new \SqlFaker\PostgreSqlProvider($faker, 'pg-17.2');
48     *     $provider->integerLiteral(min: 42, max: 42) // => '42'
49     * @example Reject an unsupported database version
50     *     new \SqlFaker\PostgreSqlProvider(\Faker\Factory::create(), 'missing-version') // throws \RuntimeException: Unsupported
51     */
52    public function __construct(Generator $generator, ?string $version = null, ?GrammarCoverage $coverage = null)
53    {
54        parent::__construct($generator);
55
56        $generator->addProvider($this);
57        $resolvedVersion = PgGrammar::resolveVersion($version);
58        $this->sql = SqlGeneratorFactory::forPostgreSql($generator, PgGrammar::load($resolvedVersion), $resolvedVersion, $coverage);
59    }
