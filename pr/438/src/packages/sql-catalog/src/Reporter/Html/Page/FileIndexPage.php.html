11/**
12 * Every file the statements are written in, under the directory that holds it.
13 *
14 * A codebase without classes is still organised, by directory and file, and
15 * that is the route through it. The listing is the tree of the source as far
16 * as the statements reach, with what was found in each file beside it.
17 *
18 * @visibility root
19 */
20final class FileIndexPage
21{
22    private HtmlText $text;
23
24    /**
25     * Wires the page to the escaping it writes through.
26     */
27    public function __construct(?HtmlText $text = null)
28    {
29        $this->text = $text ?? new HtmlText();
30    }
31
32    /**
33     * The body of the files listing.
34     */
35    public function render(ReportSite $site): string
36    {
37        $directories = $site->index()->byDirectory();
38        if ($directories === []) {
39            return '<h1>Files</h1><p class="lede">No statement was found.</p>';
40        }
41        $sections = '';
42        foreach ($directories as $directory => $files) {
43            $sections .= $this->section($site, $directory, $files);
44        }
45
46        return '<h1>Files' . $this->text->count(count($site->files()), 'file') . '</h1>'
47            . '<p class="lede">Every file a statement is written in, by directory. Open a file to read its statements function by function.</p>'
48            . '<input type="search" name="filter" class="input input-block" data-filter-rows placeholder="Narrow by file name…" aria-label="Narrow by file name" autocomplete="off" spellcheck="false">'
49            . $sections;
50    }
51
52    /**
53     * One directory, with the files in it.
54     *
55     * @param list<string> $files
56     */
57    public function section(ReportSite $site, string $directory, array $files): string
58    {
59        $index = $site->index();
60        $byFile = $index->byFile();
61        $rows = '';
62        foreach ($files as $file) {
63            $entries = $byFile[$file] ?? [];
64            $rows .= $this->row($site, $file, $entries);
65        }
66        $label = $directory === '' ? '(root)' : $directory . '/';
67
68        return '<section class="group"><h2 id="' . $this->text->escape('dir-' . $this->text->slug($label)) . '"><code>' . $this->text->escape($label) . '</code>'
69            . $this->text->count(count($files), 'file') . '</h2>'
70            . '<div class="table-wrap"><table class="sortable filter-target" data-dd-sortable><thead><tr><th scope="col" data-dd-sort="text">File</th>'
71            . '<th scope="col" class="num" data-dd-sort="number">Statements</th><th scope="col" class="num" data-dd-sort="number">Functions</th>'
72            . '<th scope="col" class="num" data-dd-sort="number">Tables</th><th scope="col" class="num" data-dd-sort="number">Attention</th></tr></thead><tbody>'
73            . $rows . '</tbody></table></div></section>';
74    }
75
76    /**
77     * One file as a row.
78     *
79     * @param list<CatalogEntry> $entries
80     */
81    public function row(ReportSite $site, string $file, array $entries): string
82    {
83        $index = $site->index();
84        $usage = $index->usage($entries);
85        $slash = strrpos($file, '/');
86        $cells = [count($entries), count($index->functionsOf($entries)), count($index->tablesOf($entries)), $usage['attention']];
87        $written = '';
88        foreach ($cells as $cell) {
89            $written .= '<td class="num">' . ($cell === 0 ? '<span class="none">0</span>' : $this->text->number($cell)) . '</td>';
90        }
91
92        return '<tr><td>' . $this->text->link($slash === false ? $file : substr($file, $slash + 1), $site->filePage($file), 'mono') . '</td>' . $written . '</tr>';
93    }

95    /**
96     * The sections of the page, for the navigation.

100    public function anchors(ReportSite $site): array
101    {
102        $anchors = [];
103        foreach (array_keys($site->index()->byDirectory()) as $directory) {
104            $label = $directory === '' ? '(root)' : $directory . '/';
105            $anchors[] = [$label, 'dir-' . $this->text->slug($label)];
106        }
107
108        return $anchors;
109    }
