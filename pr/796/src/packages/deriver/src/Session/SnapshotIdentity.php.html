19 * Computes the identity of a project snapshot from its sources, models, environment and target.
20 * Two sessions with equal identities answer every query identically; nothing else enters the identity.
21 * @visibility namespace
22 */
23final class SnapshotIdentity
24{
25    /**
26     * Content identity of the whole snapshot.
27     */
28    public readonly string $id;
29    /**
30     * @var array<string, string> Source hashes by normalized path
31     */

46    /**
47     * @param ProjectInput $input Effective captured sources
48     * @param Configuration $configuration Effective configuration
49     * @param Registry $models Registered models
50     * @param ProviderExpansion $providers Provider contributions already applied to the input
51     * @throws JsonException If the identity cannot be encoded
52     */
53    public function __construct(ProjectInput $input, public readonly Configuration $configuration, Registry $models, ProviderExpansion $providers)
54    {
55        $sources = [];
56        $sourceModes = [];
57        foreach ($input->files as $file) {
58            $path = ProjectInput::normalize($file->path);
59            $sources[$path] = hash('sha256', $file->contents);
60            if ($file->declarationsOnly) {
61                $sourceModes[$path] = 'declarations';
62            }
63        }
64        ksort($sources);
65        ksort($sourceModes);
66        $manifest = [];
67        foreach ($models->manifest as $descriptor) {
68            $manifest['model:' . $descriptor->id] = hash('sha256', serialize([$descriptor->version, $descriptor->symbol, $descriptor->priority, $descriptor->replaces, $descriptor->replaceSource, $descriptor->useSourceSignature, (new SignatureIdentity())->key($descriptor->signature)]));
69        }
70        $manifest = [...$manifest, ...$models->extensions->manifest, ...$models->state->manifest, ...$providers->manifest];
71        foreach ($configuration->expansionRules as $rule) {
72            $manifest['expansion:' . $rule->id] = hash('sha256', serialize([$rule->version, $rule->operation, $rule->name, $rule->priority, $rule->path, $rule->start, $rule->end]));
73        }
74        ksort($manifest);
75        $environment = [];
76        foreach ($configuration->environment as $key => $value) {
77            $environment[$key] = (new Identity())->key($value);
78        }
79        ksort($environment);
80        $dependencies = $configuration->dependencyVersions;
81        ksort($dependencies);
82        $identity = [$sourceModes, $dependencies, $sources, $manifest, $configuration->target->id(), $configuration->closedWorld, $configuration->environmentVersion, $environment, $configuration->standardModels, 'deriver-semantics-2'];
83        $this->id = hash('sha256', json_encode((new JsonText())->tree($identity), JSON_THROW_ON_ERROR));
84        $this->sources = $sources;
85        $this->sourceModes = $sourceModes;
86        $this->models = $manifest;
87        $this->dependencies = $dependencies;
88    }
89
90    /**
91     * Publishes the identity together with the diagnostics the declaration stage produced.
92     * @param list<Frontier> $diagnostics Source diagnostics
93     */
94    public function snapshot(array $diagnostics): ProjectSnapshot
95    {
96        return new ProjectSnapshot($this->id, $this->sources, $this->models, $this->configuration->target, $this->configuration->closedWorld, $this->configuration->environmentVersion, $diagnostics, $this->dependencies, $this->sourceModes);
97    }
