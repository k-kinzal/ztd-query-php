119     * Reads the parsed file and indexes its declarations without executing user PHP.
120     * @param SourceFile $file Captured source
121     * @param TargetProfile $profile Grammar profile
122     * @throws InvalidInputException If parser output violates its top-level contract
123     */
124    public function scan(SourceFile $file, TargetProfile $profile): void
125    {
126        $tree = $this->syntax->read($file, $profile, $this->limits);
127        $this->syntaxNodes += $tree->size;
128        if ($this->syntaxNodes > $this->limits->nodes) {
129            throw new InvalidInputException('SOURCE_LIMIT: project syntax exceeds the admission limit.');
130        }
131        if ($tree->errors !== []) {
132            foreach ($tree->errors as $error) {
133                $at = new SourceRef($this->snapshotId, $file->path, 0, strlen($file->contents), $error['line']);
134                $this->issues[] = new Frontier('INCOMPLETE_SOURCE', $at, $error['message'], missingCapability: 'valid-php-source');
135            }
136            return;
137        }
138        $nodes = $tree->nodes;
139        $script = new Stmt\Namespace_(null, $nodes, ['startFilePos' => 0, 'endFilePos' => strlen($file->contents) - 1, 'startLine' => 1]);

142            $this->register(new CallableSource('script:' . $file->path, $script, $file->path, strict: $strict));

148     * Registers a declaration and reports duplicate names instead of choosing one.
149     * @param CallableSource $source Captured callable
150     */
151    public function register(CallableSource $source): void
152    {
153        $key = (new CallableIdentity())->key($source->symbol);
154        if (isset($this->declarations[$key])) {
155            if ($this->files[$source->path]->declarationsOnly && !$this->files[$this->declarations[$key]->path]->declarationsOnly) {
156                return;
157            }
158            if ($source->node->getAttribute('deriverConditional') === true || $this->declarations[$key]->node->getAttribute('deriverConditional') === true) {
159                $variant = $source->symbol . '@declaration:' . $source->path . ':' . $source->node->getStartFilePos();
160                $this->conditionalDeclarations[$key][] = $variant;
161                $this->declarations[(new CallableIdentity())->key($variant)] = new CallableSource($variant, $source->node, $source->path, $source->className, $source->strict, $source->cacheSalt);
162                return;
163            }
164            $this->issues[] = new Frontier('INVALID_PROGRAM', $this->builder($source->path)->source($source->node), 'duplicate:' . $source->symbol);
165            return;
166        }
167        $this->declarations[$key] = $source;
168    }
