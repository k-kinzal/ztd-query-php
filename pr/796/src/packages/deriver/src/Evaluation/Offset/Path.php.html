32     * Finds the underlying storage and ordered offset registers.
33     * @param State $state Address metadata
34     * @param string $register Final address register
35     * @return array{base: Location, offsets: list<string>} Original storage and chain
36     */
37    public function chain(State $state, string $register): array
38    {
39        $offsets = [];
40        while (isset($state->offsets[$register])) {
41            $offsets[] = $register;
42            $register = $state->offsets[$register]->parent;
43        }
44        return ['base' => $state->addresses[$register] ?? new Location('unknown', unknown: true), 'offsets' => array_reverse($offsets)];
45    }

47    /**
48     * Reads each container without allocating storage for scalar or absent values.
49     * @param State $state Input memory
50     * @param Instruction $instruction Read operation
51     * @return Term|ProtocolAccess Final read result or an implicit offset protocol call
52     */
53    public function read(State $state, Instruction $instruction): Term|ProtocolAccess
54    {
55        $chain = $this->chain($state, $instruction->operands[0]);
56        $value = $state->memory->read($chain['base']);
57        $silent = $instruction->operation === 'read-silent';
58        if ($value->kind === 'uninitialized' && ($value->attributes['type'] ?? 'mixed') !== 'mixed' && !$silent) {
59            return new Term('throwable', 'Error');
60        }
61        foreach ($chain['offsets'] as $position => $register) {
62            if ($value->kind === 'object' && !(new Reader($this->context))->plainObject($value)) {
63                return new ProtocolAccess($value, $state->offsets[$register]->key, array_slice($chain['offsets'], $position + 1));
64            }
65            $value = (new Reader($this->context))->read($value, $state->offsets[$register]->key, $instruction, $state, $silent);
66            if ($value->kind === 'throwable' || $value->kind === 'opaque' && $value->literal === 'OFFSET_OPERATION') {
67                return $value;
68            }
69        }
70        return $value;
71    }

73    /**
74     * Resolves a write, reference, or unset, creating intermediate arrays only when allowed.
75     * @param CallableGraph $caller Executing callable
76     * @param State $state Mutable path memory
77     * @param Instruction $instruction Storage operation
78     * @return Location|StringAccess|ProtocolAccess|Term Resolved storage, overloaded access, or terminal result
79     */
80    public function locate(CallableGraph $caller, State $state, Instruction $instruction): Location|StringAccess|ProtocolAccess|Term
81    {
82        $chain = $this->chain($state, $instruction->operands[0]);
83        $address = $chain['base'];
84        $create = $instruction->operation !== 'unset';
85        foreach ($chain['offsets'] as $position => $register) {
86            $container = $this->prepare($caller, $state, $address, $instruction, $create);
87            $key = $state->offsets[$register]->key;
88            if ($container->kind === 'object') {
89                return new ProtocolAccess($container, $key, array_slice($chain['offsets'], $position + 1));
90            }
91            if ($container->kind === 'constant' && is_string($container->literal)) {
92                if ($position === count($chain['offsets']) - 1) {
93                    return new StringAccess($address, $key);
94                }
95                $index = (new Strings($this->context))->index($key, $instruction);
96                return $index->kind === 'constant' ? new Term('throwable', 'Error') : $index;
97            }
98            if ($container->kind !== 'array') {
99                return $container;
100            }
101            $normalized = $key === null ? (new Arrays())->appendKey($container) : (new Reader($this->context))->key($key, $instruction);
102            if ($normalized->kind === 'throwable') {
103                return $normalized;
104            }
105            if ($normalized->kind !== 'constant' || !is_int($normalized->literal) && !is_string($normalized->literal)) {
106                return Term::opaque('OFFSET_OPERATION', dependencies: [$container, ...($key === null ? [] : [$key])]);
107            }
108            $address = new Location($address->root, [...$address->path, $normalized->literal]);
109            $state->addresses[$register] = $address;
110            if ($key === null) {
111                $state->offsets[$register] = new Address($state->offsets[$register]->parent, $normalized);
112            }
113        }
114        return $address;
115    }

117    /**
118     * Autovivifies legal containers before checking the following key.
119     * @param CallableGraph $caller Calling-file type coercion mode
120     * @param State $state Input memory
121     * @param Location $address Current container storage
122     * @param Instruction $instruction Originating storage operation
123     * @param bool $create Whether absence may create an array
124     * @return Term Container or a terminal result
125     */
126    public function prepare(CallableGraph $caller, State $state, Location $address, Instruction $instruction, bool $create): Term
127    {
128        $before = $state->memory->read($address);
129        if ($before->kind === 'array' || $before->kind === 'constant' && is_string($before->literal)) {
130            return $before;
131        }
132        if ($before->kind === 'uninitialized' || $before->kind === 'constant' && in_array($before->literal, [null, false], true)) {
133            if (!$create) {
134                if ($before->kind === 'constant' && $before->literal === false) {
135                    (new Strings($this->context))->warning($instruction);
136                }
137                return Term::constant(null);
138            }
139            return $this->create($caller, $state, $address, $before, $instruction);
140        }
141        if ($before->kind === 'constant') {
142            return new Term('throwable', 'Error');
143        }
144        if ((new Reader($this->context))->plainObject($before)) {
145            return new Term('throwable', 'Error');
146        }
147        if ($before->kind === 'object') {
148            return $before;
149        }
150        return Term::opaque('OFFSET_OPERATION', dependencies: [$before]);
151    }

153    /**
154     * Creates an intermediate array subject to every live property reference constraint.
155     * @param CallableGraph $caller Calling-file coercion mode
156     * @param State $state Path memory
157     * @param Location $address Container to initialize
158     * @param Term $before Previous null, false, or uninitialized value
159     * @param Instruction $instruction Origin
160     * @return Term Created array or a type error before mutation
161     */
162    public function create(CallableGraph $caller, State $state, Location $address, Term $before, Instruction $instruction): Term
163    {
164        $array = Term::array([]);
165        $types = (new ReferenceConstraint())->find($state->memory, $address);
166        $check = (new ReferenceAssignment($this->context))->check($array, $types, $caller->strict);
167        (new TypeBinding($this->context))->report($check, $instruction->source, $state);
168        if ($check->mustFail) {
169            return new Term('throwable', 'TypeError');
170        }
171        if ($check->mayFail) {
172            return Term::opaque('OFFSET_OPERATION', dependencies: [$before]);
173        }
174        if ($before->kind === 'constant' && $before->literal === false) {
175            (new Strings($this->context))->warning($instruction);
176        }
177        $state->memory->write($address, $array);
178        return $array;
179    }
