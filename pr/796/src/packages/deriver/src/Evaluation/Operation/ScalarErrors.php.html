30     * @param Instruction $instruction Scalar operation

32     * @return list<State> Normal and possible exceptional states
33     */
34    public function paths(Instruction $instruction, State $state): array
35    {
36        $left = $state->value($instruction->operands[0] ?? '');
37        $right = $state->value($instruction->operands[1] ?? '');
38        if ($instruction->operation === 'unary') {
39            return $this->unary($instruction, $state, $left);
40        }
41        if (!$this->eligible($instruction, $state, $left, $right)) {
42            return [$state];
43        }
44        $paths = [$state];
45        if (in_array($instruction->name, ['/', '%', '<<', '>>'], true)) {
46            $shift = in_array($instruction->name, ['<<', '>>'], true);
47            $paths = $this->split($state, $this->predicate($instruction, $right), $shift ? 'ArithmeticError' : 'DivisionByZeroError');
48        }
49        if ($this->mayTypeError($instruction, $left, $right)) {
50            $exception = $state->fork();
51            $exception->completion = new Completion('throw', new Term('throwable', 'TypeError'));
52            $paths[] = $exception;
53            $this->context->frontier('WIDENED', $instruction->source, 'symbolic-numeric-conversion', [$left, $right]);
54        }
55        return $paths;
56    }

59     * Distinguishes the failing predicate from successful execution using ordinary guards.
60     * @param State $state Evaluated expression
61     * @param Term $predicate Condition under which the operation throws
62     * @param string $exception Throwable class
63     * @return list<State> Feasible normal and failing paths
64     */
65    public function split(State $state, Term $predicate, string $exception): array
66    {
67        $paths = [];
68        foreach ([false, true] as $fails) {
69            $path = $state->fork();
70            if ((new Constraints($this->context))->assume($path, $predicate, $fails)) {
71                if ($fails) {
72                    $path->completion = new Completion('throw', new Term('throwable', $exception));
73                }
74                $paths[] = $path;
75            }
76        }

108     * @param Instruction $instruction Division, remainder, or shift

112    public function predicate(Instruction $instruction, Term $right): Term
113    {
114        $integral = $instruction->name !== '/';
115        $divisor = $integral && ($right->attributes['type'] ?? '') !== 'int' ? (new Operations())->cast('int', $right) : $right;
116        $shift = in_array($instruction->name, ['<<', '>>'], true);
117        $operator = $shift ? '<' : (!$integral && ($right->attributes['type'] ?? '') !== 'int' ? '==' : '===');
118        return (new Operations())->binary($operator, $divisor, Term::constant(0));

128    public function mayTypeError(Instruction $instruction, Term $left, Term $right): bool
129    {
130        if ($instruction->name === '+' && $left->kind === 'array' && $right->kind === 'array') {
131            return false;
132        }
133        if (in_array($instruction->name, ['&', '|', '^'], true) && ($left->attributes['type'] ?? '') === 'string' && ($right->attributes['type'] ?? '') === 'string') {

146    public function unary(Instruction $instruction, State $state, Term $value): array
147    {
148        if ($value->kind === 'constant' || $state->value($instruction->result)->kind === 'throwable') {
149            return [$state];
150        }
151        $valid = match ($instruction->name) {
152            'Expr_UnaryPlus', 'Expr_UnaryMinus' => $this->numeric($value),
153            'Expr_BitwiseNot' => array_diff(explode('|', (string) ($value->attributes['type'] ?? 'mixed')), ['int', 'string']) === [],
154            default => true,
155        };
156        if ($valid) {
157            return [$state];
158        }
159        $exception = $state->fork();
160        $exception->completion = new Completion('throw', new Term('throwable', 'TypeError'));
161        $this->context->frontier('WIDENED', $instruction->source, 'symbolic-unary-conversion', [$value]);
162        return [$state, $exception];
