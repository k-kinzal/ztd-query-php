170        if ($source->unknown) {

173        }
174        $cell = $this->memory->reference($source);
175        if ($destination->local !== '' && $destination->path === []) {
176            if (str_starts_with($destination->root, 'global:') && $destination->root !== $cell) {
177                $this->memory->writePath($destination->root, [], new Term('cell', $cell), true);
178            }
179            $this->locals[$destination->local] = new Location($cell);

182        $this->memory->writePath($destination->root, $destination->path, new Term('cell', $cell), true);
