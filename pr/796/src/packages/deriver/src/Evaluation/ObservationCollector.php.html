64                }
65            }
66        }

77    /**
78     * Checks cycles, conservatively retaining loops with implicit exception-region edges.
79     * @param CallableGraph $callable Observed graph
80     * @param int $block Observation block
81     * @return bool Whether the observation lies on a control-flow cycle
82     */
83    public function repeated(CallableGraph $callable, int $block): bool
84    {
85        if ($callable->regions !== []) {
86            foreach ($callable->blocks as $candidate) {
87                if ($candidate->loopHeader) {
88                    return true;
89                }
90            }
91        }
92        $pending = $callable->blocks[$block]->terminator->targets ?? [];
93        $seen = [];
94        while ($pending !== []) {
95            $next = array_pop($pending);
96            if ($next === $block) {
97                return true;
98            }
99            if (!isset($callable->blocks[$next]) || isset($seen[$next])) {
100                continue;
101            }
102            $seen[$next] = true;
103            array_push($pending, ...$callable->blocks[$next]->terminator->targets);
104        }
105        return false;
106    }

109     * Captures requested returns and exceptional completion paths.
110     * @param CallableGraph $callable Completing callable
111     * @param State $state Completed state
112     */
113    public function completion(CallableGraph $callable, State $state): void
114    {
115        if ($this->context->batch !== null) {
116            $this->context->batch->completion($callable, $state);
117            return;
118        }
119        if ($state->completion->kind === 'exit') {
120            return;
121        }
122        $q = $this->context->query;
123        if (!$q instanceof ReturnQuery) {
124            if ($state->completion->kind === 'throw' && !$state->observed && $callable->symbol === $this->context->entrySymbol && array_sum($this->context->active) <= 1) {
125                $this->context->exceptional[] = new Exceptional($state->memory->materialize($state->completion->value ?? new Term('throwable', 'Throwable')), $state->guard, $state->snapshot(), $state->evidence, (new StorageCapture())->capture($state->memory, $state->locals, ['exception' => $state->completion->value ?? new Term('throwable', 'Throwable')]));
126                (new ObservationLimit($this->context))->enforce($callable->source);
127            }
128            return;
129        }
130        if ((new CallableIdentity())->key($q->symbol) !== (new CallableIdentity())->key($callable->symbol)) {
131            return;
132        }
133        if ($q->scope()->mode === 'symbolic' && ($this->context->active[(new CallableIdentity())->key($callable->symbol)] ?? 1) > 1) {
134            return;
135        }
136        $value = $state->memory->materialize($state->completion->value ?? Term::constant(null));
137        if ($state->completion->kind === 'throw') {
138            $this->context->exceptional[] = new Exceptional($value, $state->guard, $state->snapshot(), $state->evidence, (new StorageCapture())->capture($state->memory, $state->locals, ['exception' => $state->completion->value ?? new Term('throwable', 'Throwable')]));
139        } else {
140            $this->context->normal[] = new Alternative(['return' => $value], $state->guard, $state->snapshot(), $state->evidence, (new StorageCapture())->capture($state->memory, $state->locals, ['return' => $value]));
141        }
142        (new ObservationLimit($this->context))->enforce($callable->source);
143    }

160        if ($value->kind === 'domain' && is_string($value->literal) && $path !== []) {
161            $domain = $this->context->models->extensions->domains[$value->literal] ?? null;
162            return $domain === null ? new Term('projection', operands: [$value], attributes: ['type' => 'mixed']) : (new \Deriver\Model\Operations\DomainOperations())->apply($domain, 'project', $value, projection: new Projection($path));
163        }
164        $secret = false;
165        foreach ($path as $key) {
166            $value = $state->memory->dereference($value);
167            $secret = $secret || $value->secret;
168            if ($value->kind === 'object' && is_string($value->literal)) {
