32    /**
33     * Executes one storage instruction.
34     * @param CallableGraph $callable Current callable
35     * @param Instruction $instruction Storage operation
36     * @param State $state Path-specific memory
37     * @return Term Evaluated value
38     */
39    public function evaluate(CallableGraph $callable, Instruction $instruction, State $state): Term
40    {
41        $op = $instruction->operation;
42        $prepared = $this->prepareStorage($callable, $instruction, $state);
43        if ($prepared !== null) {
44            return $prepared;
45        }
46        $address = $state->addresses[$instruction->operands[0] ?? ''] ?? new Location('unknown', unknown: true);
47        if ($address->root === 'symbol-table' && !in_array($op, ['read', 'read-silent'], true)) {
48            (new Havoc())->symbols($state, 'DYNAMIC_VARIABLE_WRITE');
49            $this->context->frontier('DYNAMIC_VARIABLE_WRITE', $instruction->source, $op);
50        }
51        if ($op === 'read' || $op === 'read-silent') {
52            return $this->read($instruction, $address, $state);
53        }
54        if ($op === 'write') {
55            $value = $state->value($instruction->operands[1] ?? '');
56            $state->memory->write($address, $value);
57            return $value;
58        }
59        if ($op === 'alias') {
60            $source = $state->addresses[$instruction->operands[1] ?? ''] ?? new Location('unknown', unknown: true);
61            $state->alias($address, $source);
62            return $state->memory->read($source);
63        }
64        if ($op === 'reference') {
65            return new Term('cell', $state->memory->reference($address));
66        }
67        if ($op === 'increment') {
68            $before = $state->memory->read($address);
69            $after = $this->incremented($before, $instruction);
70            if ($after->kind === 'throwable') {
71                return $after;
72            }
73            $state->memory->write($address, $after);
74            return ($instruction->attributes['post'] ?? false) === true ? $before : $after;
75        }
76        return $this->binding($callable, $instruction, $state, $address);
77    }

79    /**
80     * Reads storage, reporting undefined variables and globals that the configuration does not supply.
81     * @param Instruction $instruction Read, or a silent read for isset-like operations
82     * @param Location $address Read storage
83     * @param State $state Current path
84     * @return Term Stored value, PHP's undefined-read result, or a throwable
85     */
86    public function read(Instruction $instruction, Location $address, State $state): Term
87    {
88        $value = $state->memory->read($address);
89        if ($value->kind === 'uninitialized' && $instruction->operation === 'read') {
90            return $this->uninitialized($instruction, $address, $state);
91        }
92        if ($value->kind === 'external' && $value->literal === $address->root && $address->path === [] && str_starts_with($address->root, 'global:')) {
93            $this->context->frontier('EXTERNAL_INPUT', $instruction->source, 'global-read', knownDependencies: [$address->root]);
94        }
95        return $value;
96    }

109    /**
110     * Resolves local and dynamic names before ordinary storage operations.
111     * @param CallableGraph $callable Current scope
112     * @param Instruction $instruction Address instruction
113     * @param State $state Current frame
114     * @return Term|null Address token, or an ordinary storage operation
115     */
116    public function prepareStorage(CallableGraph $callable, Instruction $instruction, State $state): ?Term
117    {
118        $op = $instruction->operation;
119        if ($op === 'dynamic-local') {
120            $name = (new Operations($this->context->configuration->target->floatPrecision))->cast('string', $state->value($instruction->operands[0]));
121            if ($name->kind === 'constant' && is_string($name->literal)) {
122                return $this->evaluate($callable, new Instruction($instruction->id, 'local', $instruction->source, $instruction->result, name: $name->literal), $state);
123            }
124            $state->addresses[$instruction->result] = new Location('symbol-table', unknown: true);
125            return new Term('location', 'symbol-table');
126        }
127        if ($op === 'local' && (str_starts_with($callable->symbol, 'script:') || in_array($instruction->name, ['_GET', '_POST', '_COOKIE', '_SERVER', '_ENV', '_REQUEST', '_FILES', '_SESSION'], true))) {
128            $root = 'global:' . $instruction->name;
129            $state->memory->cells[$root] ??= $this->context->configuration->environment[$root] ?? $this->unknownShared($state) ?? $state->memory->read($state->local($instruction->name));
130            $state->locals[$instruction->name] = new Location($root);
131        }
132        if (in_array($op, ['local', 'element-address', 'field-address', 'static-address', 'unsupported-address', 'returned-address'], true)) {
133            return $this->prepareAddress($instruction, $state);
134        }
135        return null;
136    }

145    {
146        if ($instruction->operation === 'element-address') {

171    /**
172     * Computes an address, retaining dynamic-key uncertainty.
173     * @param Instruction $instruction Address operation
174     * @param State $state Current memory
175     * @return Location Address
176     */
177    public function address(Instruction $instruction, State $state): Location
178    {
179        if ($instruction->operation === 'returned-address') {
180            return $this->returnedAddress($instruction, $state);
181        }
182        if ($instruction->operation === 'local') {
183            return $state->local($instruction->name);
184        }
185        if ($instruction->operation === 'element-address') {
186            $parent = $state->addresses[$instruction->operands[0] ?? ''] ?? new Location('unknown', unknown: true);
187            $property = $state->properties[$instruction->operands[0] ?? ''] ?? null;
188            if ($property !== null) {
189                $state->properties[$instruction->result] = $property;
190            }
191            $register = $instruction->operands[1] ?? '';
192            $key = $register === '' ? (new Arrays())->appendKey($state->memory->read($parent)) : (new Operations())->arrayKey($state->value($register));
193            if ($key->kind !== 'constant' || (!is_int($key->literal) && !is_string($key->literal))) {
194                return new Location($parent->root, $parent->path, unknown: true);
195            }
196            return new Location($parent->root, [...$parent->path, $key->literal]);
197        }
198        if ($instruction->operation === 'field-address' || $instruction->operation === 'static-address') {
199            return (new ObjectAccess($this->context))->address($instruction, $state);
200        }
201        $this->context->frontier('UNSUPPORTED_LANGUAGE_FEATURE', $instruction->source, $instruction->name);
202        return new Location('unknown', unknown: true);
203    }

212    public function uninitialized(Instruction $instruction, Location $address, State $state): Term
213    {
214        if ($address->local === 'this' || ($state->memory->read($address)->attributes['type'] ?? 'mixed') !== 'mixed') {
215            return new Term('throwable', 'Error');

227    {
228        if ($address->path !== [] || $address->unknown) {
229            return [];

237    /**
238     * Applies unset, global, and function-static bindings.
239     * @param CallableGraph $callable Current callable
240     * @param Instruction $instruction Binding operation
241     * @param State $state Current path
242     * @param Location $address Local address
243     * @return Term Result of the effect
244     */
245    public function binding(CallableGraph $callable, Instruction $instruction, State $state, Location $address): Term
246    {
247        if ($instruction->operation === 'static-initialized') {
248            $root = 'static:' . $callable->symbol . ':' . $address->local;
249            if ($state->memory->unknownShared !== null) {
250                $state->memory->cells[$root] ??= Term::opaque($state->memory->unknownShared);
251            }
252            $present = array_key_exists($root, $state->memory->cells);
253            if ($present) {
254                $state->locals[$address->local] = new Location($root);
255            }
256            return Term::constant($present);
257        }
258        if ($instruction->operation === 'unset') {
259            $this->unset($callable, $state, $address);
260        } elseif ($instruction->operation === 'global') {
261            $root = 'global:' . $address->local;
262            $state->memory->cells[$root] ??= $this->context->configuration->environment[$root] ?? $this->unknownShared($state) ?? new Term('external', $root, attributes: ['type' => 'mixed']);
263            $state->locals[$address->local] = new Location($root);
264        } elseif ($instruction->operation === 'static-local') {
265            $root = 'static:' . $callable->symbol . ':' . $address->local;
266            $state->memory->cells[$root] ??= $state->value($instruction->operands[1] ?? '');
267            $state->locals[$address->local] = new Location($root);
268        }
269        return Term::constant(null);
270    }
271    /**
272     * Removes a variable binding or an element; an unset variable stays undefined until the next symbol-table boundary.
273     * At script scope the global slot is cleared; values shared by reference live in a separate cell, so other names keep them.
274     * @param CallableGraph $callable Current scope
275     * @param State $state Current path
276     * @param Location $address Unset variable or element
277     */
278    public function unset(CallableGraph $callable, State $state, Location $address): void
279    {
280        if ($address->local === '' || $address->path !== []) {
281            $state->memory->remove($address);
282            return;
283        }
284        unset($state->locals[$address->local]);
285        if (str_starts_with($callable->symbol, 'script:')) {
286            $state->memory->cells['global:' . $address->local] = new Term('uninitialized');
287        }
288        if ($state->unknownLocals !== null) {
289            $state->locals[$address->local] = $state->memory->allocate(new Term('uninitialized'));
290        }
291    }

293    /**
294     * Resolves a returned reference or allocates permitted temporary iteration storage.
295     * @param Instruction $instruction Returned-address operation and diagnostic policy
296     * @param State $state Evaluated return register and reference memory
297     * @return Location Shared cell, temporary storage, or an unresolved reference
298     */
299    public function returnedAddress(Instruction $instruction, State $state): Location
300    {
301        $value = $state->registers[$instruction->operands[0] ?? ''] ?? Term::opaque('INVALID_REFERENCE');
302        if ($value->kind === 'cell' && is_string($value->literal)) {
303            return new Location($value->literal);
304        }
305        if (($instruction->attributes['temporary-reference'] ?? false) === true) {
306            if (($instruction->attributes['temporary-warning'] ?? true) === true) {
307                $this->context->frontier('PHP_WARNING', $instruction->source, 'non-referenceable-return');
308            }
309            return $state->memory->allocate($value);
310        }
311        return new Location('invalid-reference', unknown: true);
312    }
