28    /**
29     * Normalizes a query with entry values in the shared lossless graph.
30     * @param Query $query Requested observation
31     * @param ValueGraph $graph Value encoding and redaction policy
32     * @return QueryRecord Versioned query fields
33     * @throws InvalidInputException If the query implementation is unsupported
34     */
35    public function record(Query $query, ValueGraph $graph): array
36    {
37        $record = match (true) {
38            $query instanceof \Deriver\Query\ParameterQuery => ['kind' => 'parameter', 'symbol' => $query->symbol, 'parameter' => $query->parameter],
39            $query instanceof ReturnQuery => ['kind' => 'return', 'symbol' => $query->symbol],
40            $query instanceof ValueQuery => ['kind' => 'value', 'expression' => $query->expression, 'projection' => $query->projection],
41            $query instanceof StateQuery => ['kind' => 'state', 'point' => $query->point, 'variable' => $query->variable, 'projection' => $query->projection],
42            $query instanceof TupleQuery => ['kind' => 'tuple', 'point' => $query->point, 'values' => (object) $query->values],
43            default => throw new InvalidInputException('Unsupported query implementation.'),
44        };
45        $entries = [];
46        foreach ($query->scope()->entries as $entry) {
47            $arguments = [];
48            foreach ($entry->arguments as $name => $value) {
49                $arguments[] = ['name' => $graph->scalar($name), 'value' => $graph->add($value)];
50            }
51            $properties = [];
52            foreach ($entry->properties as $name => $value) {
53                $properties[] = ['name' => $name, 'value' => $graph->add($value)];
54            }
55            $captures = [];
56            foreach ($entry->captures as $name => $value) {
57                $captures[] = ['name' => $name, 'value' => $graph->add($value)];
58            }
59            $entries[] = ['symbol' => $entry->symbol, 'arguments' => $arguments, 'receiver' => $entry->receiver === null ? null : $graph->add($entry->receiver), 'properties' => $properties, 'captures' => $captures, 'symbolicArguments' => $entry->symbolicArguments];
60        }
61        return [...$record, 'scope' => ['mode' => $query->scope()->mode, 'entries' => $entries], 'budget' => $query->budget()];
62    }

64    /**
65     * Identifies an immutable query independently of object serialization.
66     * @param Query $query Requested observation
67     * @return string Semantic content digest

70    public function key(Query $query): string
71    {
72        $graph = new ValueGraph(true);
73        $record = $this->record($query, $graph);
74        $encoded = json_encode((new JsonText())->tree([$record, $graph->records]), JSON_THROW_ON_ERROR);
75        $confidential = false;
76        foreach ($query->scope()->entries as $entry) {
77            $confidential = $confidential || $entry->receiver?->isSecret() === true;
78            foreach ([...array_values($entry->arguments), ...array_values($entry->properties), ...array_values($entry->captures)] as $value) {

80            }
81        }
82        return $confidential ? (new SecretFingerprint())->digest($encoded) : hash('sha256', $encoded);
