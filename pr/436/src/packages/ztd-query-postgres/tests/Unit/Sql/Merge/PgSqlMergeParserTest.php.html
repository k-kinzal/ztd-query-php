38    public function testParsesMatchedUpdateAndNotMatchedInsert(): void
39    {
40        $statement = (new PgSqlMergeParser())->parse(
41            'MERGE INTO target AS t USING source AS s ON t.id = s.id '
42            . 'WHEN MATCHED AND s.enabled THEN UPDATE SET name = s.name, value = s.value '
43            . 'WHEN NOT MATCHED THEN INSERT (id, name, value) VALUES (s.id, s.name, $1);',
44        );
45
46        self::assertSame('target', $statement->targetTable);
47        self::assertSame('target', $statement->targetSql);
48        self::assertSame('t', $statement->targetAlias);
49        self::assertSame('source AS s', $statement->sourceSql);
50        self::assertSame('t.id = s.id', $statement->joinConditionSql);
51        self::assertCount(2, $statement->clauses);
52        self::assertSame(PgSqlMergeMatchKind::Matched, $statement->clauses[0]->matchKind);
53        self::assertSame('s.enabled', $statement->clauses[0]->conditionSql);
54        self::assertSame(PgSqlMergeActionKind::Update, $statement->clauses[0]->actionKind);
55        self::assertSame(['name' => 's.name', 'value' => 's.value'], $statement->clauses[0]->assignments);
56        self::assertSame(PgSqlMergeMatchKind::NotMatched, $statement->clauses[1]->matchKind);
57        self::assertSame(PgSqlMergeActionKind::Insert, $statement->clauses[1]->actionKind);
58        self::assertSame(['id', 'name', 'value'], $statement->clauses[1]->insertColumns);
59        self::assertSame(['s.id', 's.name', '$1'], $statement->clauses[1]->insertValues);
60    }
