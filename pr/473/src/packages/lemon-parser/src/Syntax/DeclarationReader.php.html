39    /**
40     * Reads the declaration whose `%` has been taken.
41     *
42     * @param TokenStream $tokens The rest
43     * @param SymbolRegistry $registry Symbols seen so far
44     * @param Location $at Where the `%` is
45     *
46     * @return Declaration The declaration
47     *
48     * @throws SyntaxException Where Lemon reports an error
49     */
50    public function read(TokenStream $tokens, SymbolRegistry $registry, Location $at): Declaration
51    {
52        $keyword = $tokens->next();
53        if (!$keyword->isAlphaWord()) {
54            throw new SyntaxException("Illegal declaration keyword: \"{$keyword->raw}\".", $keyword->location);
55        }
56        $directive = DirectiveKeyword::tryFrom($keyword->text);
57        if ($directive !== null) {
58            [$value, $form] = $this->argument($tokens, $keyword);
59
60            return new Directive($directive, $value, $form, $at);
61        }
62        $associativity = Associativity::tryFrom($keyword->text);
63        if ($associativity !== null) {
64            return new PrecedenceDeclaration($associativity, $this->lists->ranked($tokens, $registry), $at);
65        }
66
67        return match ($keyword->text) {
68            'destructor' => $this->readDestructor($tokens, $registry, $at),
69            'type' => $this->readType($tokens, $registry, $at),
70            'fallback' => new Fallback($this->lists->fallbacks($tokens, $registry), $at),
71            'token' => new TokenDeclaration($this->lists->tokens($tokens, $registry, 'token'), $at),
72            'wildcard' => new Wildcard($this->lists->wildcard($tokens, $registry), $at),
73            'token_class' => $this->readTokenClass($tokens, $registry, $at),
74            default => throw new SyntaxException("Unknown declaration keyword: \"%{$keyword->text}\".", $keyword->location),
75        };
76    }

78    /**
79     * Reads the one argument of a keyword: braced code, a string or a word.
80     *
81     * @param TokenStream $tokens The rest
82     * @param Token $keyword The keyword, for messages
83     *
84     * @return array{string, ArgumentForm} The argument without delimiters and how it was written
85     *
86     * @throws SyntaxException When what follows is none of the three
87     */
88    public function argument(TokenStream $tokens, Token $keyword): array
89    {
90        $token = $tokens->next();
91        $form = match ($token->kind) {
92            TokenKind::Code => ArgumentForm::Code,
93            TokenKind::String => ArgumentForm::String,
94            TokenKind::Word => ArgumentForm::Word,
95            TokenKind::Arrow, TokenKind::Compound, TokenKind::Punctuation, TokenKind::End => null,
96        };
97        if ($form === null) {
98            throw new SyntaxException("Illegal argument to %{$keyword->text}: {$token->raw}", $token->location);
99        }
100
101        return [$token->text, $form];
102    }

104    /**
105     * Reads `%destructor symbol argument`.
106     *
107     * @param TokenStream $tokens The rest
108     * @param SymbolRegistry $registry Symbols seen so far
109     * @param Location $at Where the `%` is
110     *
111     * @return Destructor The declaration
112     *
113     * @throws SyntaxException When no symbol follows
114     */
115    public function readDestructor(TokenStream $tokens, SymbolRegistry $registry, Location $at): Destructor
116    {
117        $symbol = $tokens->next();
118        if (!$symbol->isAlphaWord()) {
119            throw new SyntaxException('Symbol name missing after %destructor keyword', $symbol->location);
120        }
121        $registry->see($symbol->text);
122        [$value, $form] = $this->argument($tokens, $symbol);
123
124        return new Destructor(new Symbol($symbol->text, $symbol->location), $value, $form, $at);
125    }
126
127    /**
128     * Reads `%type symbol argument`.
129     *
130     * @param TokenStream $tokens The rest
131     * @param SymbolRegistry $registry Symbols seen so far
132     * @param Location $at Where the `%` is
133     *
134     * @return TypeDeclaration The declaration
135     *
136     * @throws SyntaxException When no symbol follows or it is already typed
137     */
138    public function readType(TokenStream $tokens, SymbolRegistry $registry, Location $at): TypeDeclaration
139    {
140        $symbol = $tokens->next();
141        if (!$symbol->isAlphaWord()) {
142            throw new SyntaxException('Symbol name missing after %type keyword', $symbol->location);
143        }
144        $registry->type($symbol->text, $symbol->location);
145        [$value, $form] = $this->argument($tokens, $symbol);
146
147        return new TypeDeclaration(new Symbol($symbol->text, $symbol->location), $value, $form, $at);
148    }
149
150    /**
151     * Reads `%token_class name TOKEN|TOKEN... .`.
152     *
153     * @param TokenStream $tokens The rest
154     * @param SymbolRegistry $registry Symbols seen so far
155     * @param Location $at Where the `%` is
156     *
157     * @return TokenClass The declaration
158     *
159     * @throws SyntaxException When the name is not a fresh nonterminal name
160     */
161    public function readTokenClass(TokenStream $tokens, SymbolRegistry $registry, Location $at): TokenClass
162    {
163        $name = $tokens->next();
164        if (!$name->isLowerWord()) {
165            throw new SyntaxException("%token_class must be followed by an identifier: {$name->raw}", $name->location);
166        }
167        if ($registry->isKnown($name->text)) {
168            throw new SyntaxException("Symbol \"{$name->text}\" already used", $name->location);
169        }
170        $registry->see($name->text);
171
172        return new TokenClass(new Symbol($name->text, $name->location), $this->lists->classTokens($tokens, $registry), $at);
173    }
