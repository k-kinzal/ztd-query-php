101    public function guardDefaults(\SqlCatalog\Core\Php\ClassShape $shape, QueryState $state): QueryState
102    {
103        foreach (['table', 'primaryKey', 'connection', 'timestamps', 'with', 'withCount'] as $property) {
104            $expression = $shape->propertyDefaults[$property] ?? null;
105            if ($expression === null) {
106                continue;
107            }
108            try {
109                $value = (new ConstExprEvaluator())->evaluateDirectly($expression);
110                if (in_array($property, ['with', 'withCount'], true) && $value !== []) {
111                    $state = $state->reject('Implicit Eloquent eager loads are not modelled');
112                } elseif (!in_array($property, ['with', 'withCount'], true) && $value !== null && !is_scalar($value)) {
113                    $state = $state->reject('Invalid Eloquent metadata: ' . $property);
114                }
115            } catch (ConstExprEvaluationException) {
116                $state = $state->reject('Unresolved Eloquent metadata: ' . $property);
117            }
118        }
119        if (isset($shape->constants['DELETED_AT']) && $this->scalar($shape->constants['DELETED_AT']) === null) {
120            $state = $state->reject('Unresolved Eloquent deleted column');
121        }

123        return $state;
