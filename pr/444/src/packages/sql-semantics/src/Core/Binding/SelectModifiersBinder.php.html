22     * @param list<OutputColumn> $outputs
23     * @return list<Ordering>
24     */
25    public function ordering(Node $statement, Scope $scope, array $outputs): array
26    {
27        $nodes = $scope->identifiers->dialect->platform()->query()->orderingNodes($statement);
28        $result = [];
29        foreach ($nodes as $node) {
30            Tree::assertChildren($node, $scope->identifiers->dialect->platform()->syntax()->nodes('orderingChildren'), [',']);
31            $expr = Tree::child($node, $scope->identifiers->dialect->platform()->syntax()->nodes('expression'));
32            if ($expr === null) {
33                Tree::unsupported($node, 'ordering');
34            }
35            $direction = Tree::child($node, $scope->identifiers->dialect->platform()->syntax()->nodes('orderingDirection'));
36            $nulls = Tree::child($node, $scope->identifiers->dialect->platform()->syntax()->nodes('nullsOrder'));
37            $expression = $this->sortExpression($expr, $scope, $outputs);
38            $result[] = new Ordering($expression, $direction !== null && strtoupper(Tree::text($direction)) === 'DESC', $nulls === null ? null : str_contains(strtoupper(Tree::text($nulls)), 'FIRST'));
39        }
40
41        return $result;
42    }

44    /**
45     * @param list<OutputColumn> $outputs
46     * @throws SemanticException
47     */
48    public function sortExpression(Node $node, Scope $scope, array $outputs): Expression
49    {
50        $tokens = $node->tokens();
51        if (count($tokens) === 1 && !in_array($tokens[0]->name, $scope->identifiers->dialect->platform()->syntax()->nodes('stringToken'), true)) {
52            $text = $tokens[0]->text;
53            if (ctype_digit($text)) {
54                $ordinal = (int) $text - 1;
55                if (!isset($outputs[$ordinal])) {
56                    throw new SemanticException('invalid-output-position', 'ORDER BY position is outside the result.', $node);
57                }
58                return $outputs[$ordinal]->expression;
59            }
60            $name = $scope->identifiers->name($tokens[0]);
61            $matches = array_values(array_filter($outputs, static fn (OutputColumn $output): bool => $output->name !== null && $scope->identifiers->equal($output->name, $name)));
62            if (count($matches) > 1) {
63                throw new SemanticException('ambiguous-output', 'ORDER BY alias is ambiguous.', $node);
64            }
65            if ($matches !== []) {
66                return $matches[0]->expression;
67            }
68        }
69
70        return (new ExpressionBinder())->bind($node, $scope);
71    }

73    /**
74     * @return array{?Expression, ?Expression}
75     */
76    public function pagination(Node $statement, Scope $scope): array
77    {
78        $limit = Tree::outer($statement, $scope->identifiers->dialect->platform()->syntax()->nodes('limit'))[0] ?? null;
79        $offset = Tree::outer($statement, $scope->identifiers->dialect->platform()->syntax()->nodes('offset'))[0] ?? null;
80        if ($limit !== null && $limit->tokens() !== [] && strtoupper($limit->tokens()[0]->text) !== 'LIMIT') {
81            Tree::unsupported($limit, 'FETCH pagination');
82        }
83        $expressions = $limit === null ? [] : Tree::outer($limit, $scope->identifiers->dialect->platform()->syntax()->nodes('paginationExpression'));
84        $bound = array_map(static fn (Node $node): Expression => (new ExpressionBinder())->bind($node, $scope), $expressions);
85        if (count($bound) > 2) {
86            Tree::unsupported($statement, 'pagination');
87        }
88        if ($limit !== null && count($bound) === 2 && str_contains(Tree::text($limit), ',')) {
89            return [$bound[1], $bound[0]];
90        }
91        if ($offset !== null) {
92            $node = Tree::outer($offset, $scope->identifiers->dialect->platform()->syntax()->nodes('expression'))[0] ?? null;
93            if ($node === null) {
94                Tree::unsupported($offset, 'offset');
95            }
96            return [$bound[0] ?? null, (new ExpressionBinder())->bind($node, $scope)];
97        }
98        if ($limit !== null && $limit->tokens() !== [] && $bound === []) {
99            Tree::unsupported($limit, 'limit');
100        }
101
102        return [$bound[0] ?? null, $bound[1] ?? null];
103    }
