31    {
32        $nodes = Tree::outer($from, ['table_ref', 'table_reference', 'seltablist']);
33        $result = null;
34        foreach ($nodes as $node) {
35            $right = $this->relation($node);
36            $result = $result === null ? $right : $this->join($result, $right, JoinKind::Cross, null, $from, $this->ids->join());

38        if ($result === null && $from->tokens() !== []) {

78    public function joined(Node $node): BoundRelation
79    {
80        $references = Tree::outer($node, ['table_ref', 'table_reference']);
81        if (count($references) !== 2) {
82            Tree::unsupported($node, 'join operands');
83        }
84        $kindNode = Tree::child($node, ['join_type', 'outer_join_type', 'inner_join_type', 'natural_join_type']);
85        $kind = $kindNode === null ? JoinKind::Inner : $this->kind(Tree::text($kindNode), $node);
86        $qualifier = Tree::child($node, ['join_qual']);
87        if ($qualifier !== null && strtoupper($qualifier->tokens()[0]->text) !== 'ON') {
88            Tree::unsupported($qualifier, 'join qualification');
89        }
90        $condition = $qualifier === null ? Tree::child($node, ['expr']) : (Tree::outer($qualifier, ['a_expr'])[0] ?? null);
91        if ($kindNode === null && in_array('CROSS', array_map(static fn ($token): string => strtoupper($token->text), $node->tokens()), true)) {
92            $kind = JoinKind::Cross;
93        }
94        $id = $this->ids->join();
95        $left = $this->relation($references[0]);
96        $right = $this->relation($references[1]);
97        if ($condition === null && $kind !== JoinKind::Cross) {
98            Tree::unsupported($node, 'join without an ON condition');
99        }
100
101        return $this->join($left, $right, $kind, $condition, $node, $id);

121        $qualifier = Tree::child($node, ['on_using']);
122        if ($prefix === null) {

166    /**
167     * Resolves the logical operation from a join production.
168     */
169    public function kind(string $text, Node $source): JoinKind
170    {
171        $text = strtoupper($text);
172        if (str_contains($text, 'NATURAL') || str_contains($text, 'STRAIGHT')) {
173            Tree::unsupported($source, 'join operation');
174        }
175
176        return match (true) {
177            str_contains($text, 'LEFT') => JoinKind::Left,
178            str_contains($text, 'RIGHT') => JoinKind::Right,
179            str_contains($text, 'FULL') => JoinKind::Full,
180            str_contains($text, 'CROSS'), $text === ',' => JoinKind::Cross,
181            default => JoinKind::Inner,
182        };
183    }

186     * Binds the match predicate before extending the nullable inputs.
187     */
188    public function join(BoundRelation $left, BoundRelation $right, JoinKind $kind, ?Node $condition, Node $source, string $id): BoundRelation
189    {
190        $scope = $left->scope->combine($right->scope, $source);
191        $expression = $condition === null ? null : (new ExpressionBinder())->bind($condition, $scope);
192        if ($expression !== null) {
193            (new ExpressionRules($this->tables->identifiers->dialect))->predicate($expression);
194        }
195        $leftScope = in_array($kind, [JoinKind::Right, JoinKind::Full], true) ? $left->scope->extend($id) : $left->scope;
196        $rightScope = in_array($kind, [JoinKind::Left, JoinKind::Full], true) ? $right->scope->extend($id) : $right->scope;
197
198        return new BoundRelation(new Join($id, $kind, $left->relation, $right->relation, $expression, $source), $leftScope->combine($rightScope, $source));
199    }
