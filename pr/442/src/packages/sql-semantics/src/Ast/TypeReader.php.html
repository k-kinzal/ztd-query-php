26     * Reads the declared built-in type and preserves its modifiers.
27     */
28    public function read(Node $node): TypeDescriptor
29    {
30        $tokens = $node->tokens();
31        $words = [];
32        $modifiers = [];
33        $inModifiers = false;
34        foreach ($tokens as $token) {
35            if ($token->text === '(') {
36                $inModifiers = true;
37            } elseif ($token->text === ')') {
38                $inModifiers = false;
39            } elseif ($token->text !== ',') {
40                if ($inModifiers) {
41                    $modifiers[] = $token->text;
42                } else {
43                    $words[] = strtoupper($token->text);
44                }
45            }
46        }
47        $name = implode(' ', $words);
48        if ($this->dialect === Dialect::Sqlite) {
49            return new TypeDescriptor($this->dialect, strtolower($name), $modifiers, $this->affinity($name));
50        }
51        $canonical = $this->canonical($name);
52        if ($canonical === null) {
53            Tree::unsupported($node, 'type declaration');
54        }
55
56        return new TypeDescriptor($this->dialect, $canonical, $modifiers);
57    }

62    public function canonical(string $name): ?string
63    {
64        return match ($name) {
65            'INT', 'INTEGER', 'INT4' => 'integer',
66            'SMALLINT', 'INT2' => 'smallint',
67            'BIGINT', 'INT8' => 'bigint',
68            'DEC', 'DECIMAL', 'NUMERIC' => 'numeric',
69            'REAL' => $this->dialect === Dialect::MySql ? 'double precision' : 'real',
70            'FLOAT4' => 'real',
71            'DOUBLE', 'DOUBLE PRECISION', 'FLOAT8' => 'double precision',
72            'BOOL', 'BOOLEAN' => $this->dialect === Dialect::MySql ? 'tinyint' : 'boolean',
73            'VARCHAR', 'CHARACTER VARYING', 'CHAR VARYING' => 'varchar',
74            'CHAR', 'CHARACTER' => 'char',
75            'TEXT', 'DATE', 'TIME', 'TIMESTAMP', 'JSON' => strtolower($name),
76            'TINYINT', 'MEDIUMINT', 'DATETIME', 'BLOB' => $this->dialect === Dialect::MySql ? strtolower($name) : null,
77            'UUID', 'BYTEA', 'JSONB', 'TIMESTAMPTZ', 'TIMETZ', 'INTERVAL' => $this->dialect === Dialect::PostgreSql ? strtolower($name) : null,
78            default => null,

85    public function affinity(string $name): string
86    {
87        if (str_contains($name, 'INT')) {

92        }

94            return 'blob';
95        }
96        if (str_contains($name, 'REAL') || str_contains($name, 'FLOA') || str_contains($name, 'DOUB')) {
97            return 'real';
98        }
99
