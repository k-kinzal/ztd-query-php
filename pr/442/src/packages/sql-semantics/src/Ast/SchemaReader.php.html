31     * @param list<Node> $trees
32     * @return list<TableDefinition>
33     * @throws SemanticException
34     */
35    public function read(array $trees): array
36    {
37        $tables = [];
38        foreach ($trees as $tree) {
39            foreach (StatementList::read($tree, $this->identifiers->dialect) as $statement) {
40                $create = Tree::outer($statement, ['CreateStmt', 'create_table_stmt', 'create_table'])[0] ?? null;
41                if ($create === null) {
42                    Tree::unsupported($statement, 'schema statement');
43                }
44                $table = $this->table($this->identifiers->dialect === Dialect::Sqlite ? $statement : $create);
45                $key = $table->schema . "\0" . $table->name;
46                if ($this->identifiers->dialect === Dialect::Sqlite) {
47                    $key = strtolower($key);
48                }
49                if (isset($tables[$key])) {
50                    throw new SemanticException('duplicate-table', 'Duplicate table declaration: ' . $table->name, $create);
51                }
52                $tables[$key] = $table;
53            }
54        }
55
56        return array_values($tables);
57    }

60     * Reads one table and promotes the nullability of primary key columns.
61     */
62    public function table(Node $create): TableDefinition
63    {
64        $header = Tree::outer($create, ['create_table'])[0] ?? $create;
65        $nameNode = Tree::child($header, ['qualified_name', 'table_ident', 'nm']);
66        if ($nameNode === null) {
67            Tree::unsupported($create, 'table name');
68        }
69        $parts = $this->identifiers->parts($nameNode);
70        $sqliteDb = Tree::child($header, ['dbnm']);
71        if ($sqliteDb !== null) {
72            $parts = [$parts[0], ...$this->identifiers->parts($sqliteDb)];
73        }
74        if (count($parts) > 2) {
75            Tree::unsupported($nameNode, 'three-part table name');
76        }
77        $this->validate($create, $header);
78        $columns = [];
79        $constraints = [];
80        foreach ($this->columnNodes($create) as [$column, $attributes]) {
81            [$definition, $localConstraints] = (new ColumnReader($this->identifiers))->read($column, $attributes);
82            $columns[] = $definition;
83            array_push($constraints, ...$localConstraints);
84        }
85        foreach (Tree::outer($create, ['TableConstraint', 'table_constraint_def', 'tcons']) as $node) {
86            $constraint = (new ConstraintReader($this->identifiers))->read($node);
87            if ($constraint === null) {
88                Tree::unsupported($node, 'table constraint');
89            }
90            $constraints[] = $constraint;
91        }
92        if ($columns === []) {
93            Tree::unsupported($create, 'CREATE TABLE without column declarations');
94        }
95        $columns = $this->primaryKeys($columns, $constraints, $create);
96
97        return new TableDefinition(count($parts) === 2 ? $parts[0] : $this->defaultSchema, $parts[count($parts) - 1], $columns, $constraints, $create);
98    }

104    {
105        if ($this->identifiers->dialect === Dialect::Sqlite) {

107            Tree::assertChildren($header, ['createkw', 'nm', 'dbnm'], ['TABLE']);
108            $arguments = Tree::child($source, ['create_table_args']);
109            if ($arguments === null) {

111            }
112            Tree::assertChildren($arguments, ['columnlist', 'conslist_opt'], ['(', ')']);
113            return;
114        }
115        Tree::assertChildren($source, ['qualified_name', 'OptTableElementList', 'table_ident', 'table_element_list'], ['CREATE', 'TABLE', '(', ')']);
116        foreach ($source->find('field_def') as $field) {
117            Tree::assertChildren($field, ['type', 'opt_column_attribute_list'], []);
118        }

125    {
126        $columns = [];
127        if ($this->identifiers->dialect === Dialect::Sqlite) {
128            foreach ($create->find('columnlist') as $list) {
129                $column = Tree::child($list, ['columnname']);
130                if ($column !== null) {
131                    $attributes = Tree::child($list, ['carglist']);
132                    $columns[] = [$column, $attributes === null ? [] : Tree::outer($attributes, ['ccons'])];
133                }
134            }
135            return array_reverse($columns);
136        }
137        foreach (Tree::outer($create, ['columnDef', 'column_def']) as $column) {
138            $columns[] = [$column, Tree::outer($column, ['ColConstraint', 'column_attribute'])];
139        }

146     * @param list<TableConstraint> $constraints
147     * @return list<ColumnDefinition>
148     * @throws SemanticException
149     */
150    public function primaryKeys(array $columns, array $constraints, Node $source): array
151    {
152        $names = array_map(fn (ColumnDefinition $column): string => $this->identifiers->dialect === Dialect::PostgreSql ? $column->name : strtolower($column->name), $columns);
153        if (count(array_unique($names)) !== count($names)) {
154            throw new SemanticException('duplicate-column', 'Duplicate column declaration.', $source);
155        }
156        $primary = [];
157        foreach ($constraints as $constraint) {
158            foreach ($constraint->columns as $name) {
159                if (!in_array($this->identifiers->dialect === Dialect::PostgreSql ? $name : strtolower($name), $names, true)) {
160                    throw new SemanticException('unknown-column', 'Constraint references unknown column: ' . $name, $constraint->source);
161                }
162            }
163            if ($constraint->kind === ConstraintKind::PrimaryKey) {
164                if ($primary !== []) {
165                    throw new SemanticException('duplicate-primary-key', 'More than one primary key.', $constraint->source);
166                }
167                $primary = array_map(fn (string $name): string => $this->identifiers->dialect === Dialect::PostgreSql ? $name : strtolower($name), $constraint->columns);
168            }
169        }
170        $result = [];
171        foreach ($columns as $column) {
172            $notNull = in_array($this->identifiers->dialect === Dialect::PostgreSql ? $column->name : strtolower($column->name), $primary, true) && $this->primaryNotNull($column, $primary, $constraints);
173            $result[] = new ColumnDefinition($column->name, $column->type, $notNull ? Nullability::NotNull : $column->nullability, $column->source, $column->defaultExpression);
174        }
175
176        return $result;
177    }

183    public function primaryNotNull(ColumnDefinition $column, array $primary, array $constraints): bool
184    {
185        if ($this->identifiers->dialect !== Dialect::Sqlite) {
186            return true;
187        }
188        if ($column->type->name !== 'integer' || count($primary) !== 1) {
189            return false;
190        }
191        foreach ($constraints as $constraint) {
192            if ($constraint->kind === ConstraintKind::PrimaryKey && str_contains(strtoupper(Tree::text($constraint->source)), 'DESC')) {
193                return false;
194            }
195        }
196
197        return true;
