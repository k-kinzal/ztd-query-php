57    /**
58     * @param list<Domain> $arguments A modelled static factory, or null for other classes.
59     */
60    public function staticCall(string $className, string $method, array $arguments, Environment $environment, ?Expr\CallLike $call = null, ?FunctionScope $scope = null, ?ExpressionEvaluator $expressions = null): ?Domain
61    {
62        $method = strtolower($method);
63        if (strcasecmp($className, self::FACADE) === 0) {
64            return $this->connectionCall($method, $arguments, new QueryState(['dialect' => Domain::literal($this->dialect)]), $environment);
65        }
66        if (!(new ModelMetadata($this->index))->recognizes($className)) {
67            return null;
68        }
69        $object = $this->allocate(ModelMetadata::BUILDER, (new ModelMetadata($this->index))->state($className, $this->dialect));
70        if ($call !== null && $scope !== null && $expressions !== null) {
71            $callback = $this->callback($call, $object, $method, $arguments, $environment, $scope, $expressions);
72            if ($callback !== null) {
73                return $callback;
74            }
75        }
76
77        return $this->mutate($object, $method, $arguments, $environment);
78    }

81     * @param list<Domain> $arguments A modelled connection or builder method, or null.

83    public function methodCall(Domain $receiver, string $method, array $arguments, Environment $environment, ?Expr\CallLike $call = null, ?FunctionScope $scope = null, ?ExpressionEvaluator $expressions = null): ?Domain
84    {
85        $class = $receiver->type()->soleClassName();
86        $method = strtolower($method);
87        if ($this->isConnection($class)) {
88            $object = $receiver->soleObject();
89            $state = $object?->state === null ? new QueryState(['dialect' => Domain::literal($this->connectionDialect($class))]) : QueryState::from($object);
90
91            return $this->connectionCall($method, $arguments, $state, $environment);
92        }
93        if (!$this->isBuilder($class)) {
94            return null;
95        }
96        $results = [];
97        foreach ($receiver->terms as $term) {
98            $object = $term instanceof ObjectTerm ? $term : new ObjectTerm($class ?? self::QUERY);
99            $callback = $call !== null && $scope !== null && $expressions !== null ? $this->callback($call, $object, $method, $arguments, $environment, $scope, $expressions) : null;
100            $results = array_merge($results, ($callback ?? $this->mutate($object, $method, $arguments, $environment))->terms);
101        }
102
103        $result = Domain::fromTerms($results, $receiver->widened, $receiver->combined);
104        $environment->objects()->rememberValue($result);
105
106        return $result;
107    }

145    /**
146     * @param list<Domain> $arguments Connection factories and raw expressions.
147     */
148    public function connectionCall(string $method, array $arguments, QueryState $state, Environment $environment): ?Domain
149    {
150        if ($method === 'raw' && count($arguments) === 1) {
151            return Domain::of($this->allocate('Illuminate\\Database\\Query\\Expression', new QueryState(['sql' => $arguments[0]])));
152        }
153        if ($method === 'connection' && count($arguments) <= 1) {
154            return Domain::of($this->allocate(self::CONNECTION, $state));
155        }
156        if ($method === 'table' && count($arguments) === 1) {
157            $object = $this->allocate(self::QUERY, $state->with('table', $arguments[0])->with('key', Domain::literal('id')));
158            $environment->objects()->remember($object);
159
160            return Domain::of($object);
161        }
162
163        return null;
164    }

176    /**
177     * @param list<Domain> $arguments Applies one effect, preserving identity and any earlier gaps.
178     */
179    public function mutate(ObjectTerm $object, string $method, array $arguments, Environment $environment): Domain
180    {
181        $state = QueryState::from($object);
182        $grammar = new Grammar($state->string('dialect'));
183        if ($method === 'query' && $arguments === []) {
184            $updated = $state;
185        } elseif (in_array($method, ['withtrashed', 'onlytrashed'], true) && $arguments === [] && $state->get('softDeletes')->soleLiteral()?->value === true) {
186            $updated = $state->with('trashed', Domain::literal($method));
187        } else {
188            $updated = (new Predicates($grammar))->apply($state, $method, $arguments)
189                ?? (new Clauses($grammar))->apply($state, $method, $arguments)
190                ?? $state->reject('Unmodelled Laravel effect: ' . $method);
191        }
192        $object = $updated->object($object);
193        $environment->objects()->remember($object);
194
195        return Domain::of($object);
196    }

198    /**
199     * Retains terminal calls' persistent mutations and identifies their return values.
200     *
201     * @param list<Domain> $arguments
202     */
203    public function execution(Domain $receiver, string $method, array $arguments, Environment $environment): Domain
204    {
205        $snapshots = [];
206        foreach ($receiver->terms as $term) {
207            if (!$term instanceof ObjectTerm) {
208                continue;
209            }
210            $state = QueryState::from($term);
211            if (in_array($method, ['first', 'firstorfail', 'find'], true)) {
212                $state = $state->with('limit', Domain::literal(1));
213                if ($method === 'find') {
214                    $state = (new Predicates(new Grammar($state->string('dialect'))))->basic($state, [$state->get('key'), $arguments[0] ?? Domain::unknown()], 'and');
215                }
216                $snapshots[] = $state->object($term);
217            } elseif (!in_array($method, ['get', 'all', 'pluck', 'count', 'sum', 'avg', 'min', 'max', 'exists', 'doesntexist', 'insert', 'insertorignore', 'update', 'delete'], true)) {
218                $snapshots[] = $state->reject('Unmodelled Laravel execution effects')->object($term);
219            }
220        }
221        $environment->objects()->rememberValue(Domain::fromTerms($snapshots, $receiver->widened, $receiver->combined));
222        $model = $receiver->soleObject() === null ? null : QueryState::from($receiver->soleObject())->string('model');
223        $type = match ($method) {
224            'get', 'all', 'pluck' => 'Illuminate\\Support\\Collection',
225            'first', 'firstorfail', 'find' => $model ?? 'stdClass',
226            'exists', 'doesntexist', 'insert', 'insertorignore' => 'bool',
227            'count', 'update', 'delete', 'insertgetid' => 'int',
228            default => 'mixed',
229        };
230
231        return Domain::opaque(\SqlCatalog\Type\TypeShape::of([$type]), \SqlCatalog\Text\Origin::Call, 'Laravel ' . $method);
232    }

248    /**
249     * A bad overload keeps the builder identity so its eventual execution remains visible.
250     */
251    public function unsupported(Domain $receiver, Environment $environment, string $reason): Domain
252    {
253        $terms = [];
254        foreach ($receiver->terms as $term) {
255            if ($term instanceof ObjectTerm && $term->identity !== null) {
256                $term = QueryState::from($term)->reject($reason)->object($term);
257                $environment->objects()->remember($term);
258            }
259            if ($term instanceof \SqlCatalog\Evaluation\ArrayTerm) {
260                $term = new \SqlCatalog\Evaluation\ArrayTerm(array_map(fn (\SqlCatalog\Evaluation\ArrayEntry $entry): \SqlCatalog\Evaluation\ArrayEntry => new \SqlCatalog\Evaluation\ArrayEntry($entry->key, $this->unsupported($entry->value, $environment, $reason)), $term->entries), $term->complete);
261            }
262            $terms[] = $term;
263        }
264
265        $result = Domain::fromTerms($terms, $receiver->widened, $receiver->combined);
266        $environment->objects()->rememberValue($result);
267
268        return $result;
269    }
