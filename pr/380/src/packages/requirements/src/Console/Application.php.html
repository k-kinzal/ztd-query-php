45    private function execute(Project $project, Options $options): array
46    {
47        if ($options->command === 'lint') {
48            return ['passed' => true, 'message' => count($project->items) . ' items validated.'];
49        }
50        if ($options->command === 'format') {
51            $changed = (new Formatter())->format($project->files, $options->flag('check'));
52            return ['passed' => !$options->flag('check') || $changed === [], 'changed' => $changed];
53        }
54        if (in_array($options->command, ['list', 'spec'], true)) {
55            $items = array_filter($project->items, $options->matches(...));
56            if ($options->command === 'list') {
57                return ['passed' => true, 'items' => array_values(array_map($this->describe(...), $items))];
58            }
59            $results = (new Verifier())->verify($project, $items);
60            $passed = $results !== [];
61            $rows = [];
62            foreach ($results as $id => $result) {
63                $passed = $passed && in_array($result->status, ['passed', 'unsupported'], true);
64                $rows[$id] = ['statement' => $items[$id]->statement, ...$result->toArray()];
65            }
66            return ['passed' => $passed, 'specifications' => $rows, 'errors' => $results === [] ? ['No specifications selected.'] : []];
67        }
68        if (!in_array($options->command, ['check', 'coverage'], true)) {
69            throw new InvalidArgumentException('Unknown command: ' . $options->command);
70        }
71        $analysis = (new Analyzer())->analyze($project, $options->flag('live'));
72        if ($options->command === 'check') {
73            return ['passed' => $analysis->errors === [], 'mode' => $options->flag('live') ? 'live' : 'configured', 'errors' => $analysis->errors, 'evidence' => $analysis->evidence];
74        }
75        $report = (new Coverage())->report($project, $analysis, $options->text('baseline'), $options->flag('allow-removed'), $options->percentage('min-coverage'), $options->percentage('min-diff-coverage'));
76        $report['mode'] = $options->flag('live') ? 'live' : 'configured';
77        $baseline = $options->text('write-baseline');
78        if ($baseline !== null) {
79            if ($analysis->errors !== []) {
80                throw new InvalidArgumentException('Cannot write a baseline with invalid source evidence.');
81            }
82            if (file_put_contents($baseline, json_encode($report, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE | JSON_THROW_ON_ERROR) . "\n") === false) {
83                throw new InvalidArgumentException("Cannot write baseline: $baseline");
84            }
85        }
86        return $report;
87    }
