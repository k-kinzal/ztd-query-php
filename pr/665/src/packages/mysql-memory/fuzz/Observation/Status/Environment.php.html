100    /**
101     * Preserves exact values outside the allowlist and rejects malformed values inside it.
102     */
103    public function value(string $name, string $value): ?string
104    {
105        $domain = $this->domain($name);
106        if ($domain === null) {
107            return $value;
108        }
109        $valid = match ($domain) {
110            'unsigned' => ctype_digit($value) && bccomp($value, '18446744073709551615') <= 0,
111            'cost' => preg_match('/\A[0-9]+(?:\.[0-9]+)?\z/', $value) === 1 && is_finite((float) $value),
112            'verification-mode' => in_array($value, ['0', '1', '2', '3', '4', '5', '6', '7'], true),
113            'public-key' => $value === '' || $this->publicKey($value),
114            'certificate-time' => $value === '' || preg_match('/\A(?:Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec) [ 0-3][0-9] [0-2][0-9]:[0-5][0-9]:[0-5][0-9] [0-9]{4} GMT\z/', $value) === 1,
115            'path' => preg_match('/[\x00-\x1f\x7f]/', $value) === 0,
116            'cipher-list' => $value === '' || preg_match('/\A[A-Za-z0-9_!+@=.-]+(?::[A-Za-z0-9_!+@=.-]+)*\z/', $value) === 1,
117            'tls-versions' => $value === '' || preg_match('/\ATLSv1(?:\.[0-3])?(?:,TLSv1(?:\.[0-3])?)*\z/', $value) === 1,
118            'listen-address' => $value === '' || $value === '*' || filter_var($value, FILTER_VALIDATE_IP) !== false,
119            'tls-library' => $value === '' || preg_match('/\A(?:OpenSSL|LibreSSL|wolfSSL) [0-9]+\.[0-9]+\.[0-9]+[a-z]?(?: [A-Za-z0-9 -]+)?\z/', $value) === 1,
120            'buffer-load' => $value === '' || preg_match('/\ABuffer pool\(s\) load completed at [0-9]{6} [ 0-2][0-9]:[0-5][0-9]:[0-5][0-9]\z/', $value) === 1,
121            default => false,
122        };
123        return $valid ? '<environment:' . $domain . '>' : null;
124    }

126    /**
127     * Checks the PEM envelope and a DER sequence, without trusting arbitrary nonempty strings.
128     */
129    public function publicKey(string $value): bool
130    {
131        if (preg_match('/\A-----BEGIN PUBLIC KEY-----\n([A-Za-z0-9+\/=\n]+)-----END PUBLIC KEY-----\n?\z/', $value, $match) !== 1) {
132            return false;
133        }
134        $der = base64_decode(str_replace("\n", '', $match[1]), true);
135        return is_string($der) && strlen($der) > 32 && $der[0] === "\x30";
136    }
