26    /**
27     * Returns the validated field structure, retaining missing fields as comparison differences.
28     */
29    public function comparable(string $text): ?string
30    {
31        if (preg_match('/\A\n=+\n(?:[0-9]{6}|[0-9]{4}-[0-9]{2}-[0-9]{2}) [ 0-2][0-9]:[0-5][0-9]:[0-5][0-9] (?:0x)?[0-9a-f]+ INNODB MONITOR OUTPUT\n=+\nPer second averages calculated from the last [0-9]+ seconds\n(.*)\n-+\nEND OF INNODB MONITOR OUTPUT\n=+\n\z/s', $text, $report) !== 1) {
32            return null;
33        }
34        $parts = preg_split('/^-+\n([A-Z][A-Z \/]+)\n-+\n/m', $report[1], -1, PREG_SPLIT_DELIM_CAPTURE);
35        if ($parts === false || array_shift($parts) !== '' || count($parts) !== count(self::FIELDS) * 2) {
36            return null;
37        }
38        $structure = [];
39        foreach (array_keys(self::FIELDS) as $index => $section) {
40            if ($parts[$index * 2] !== $section) {
41                return null;
42            }
43            $fields = [];
44            foreach (explode("\n", trim($parts[$index * 2 + 1])) as $line) {
45                $line = preg_replace('/\h+/', ' ', trim($line)) ?? '';
46                $field = $this->field($section, $line);
47                if ($field === null) {
48                    return null;
49                }
50                if ($field !== '') {
51                    $fields[$field] = true;
52                }
53            }
54            ksort($fields);
55            $structure[] = $section . "\n" . implode("\n", array_keys($fields));
56        }
57        return implode("\n", $structure);
58    }

60    /**
61     * Checks every line against a named field or a physical allocation record.
62     */
63    public function field(string $section, string $line): ?string
64    {
65        foreach (self::FIELDS[$section] ?? [] as $template) {
66            $pattern = str_replace(['\{u\}', '\{f\}', '\{state\}'], ['[0-9]{1,20}', '[0-9]+\.[0-9]+', '(?:running|running but idle|stopped|sleeping|waiting for server activity)'], preg_quote($template, '/'));
67            if (preg_match('/\A' . $pattern . '\z/', $line) === 1) {
68                return str_starts_with($template, 'No buffer pool page gets') || str_starts_with($template, 'Buffer pool hit rate') ? '<buffer-pool-hit-rate>' : $template;
69            }
70        }
71        $physical = match ($section) {
72            'TRANSACTIONS' => '/\A(?:---TRANSACTION [0-9]+, not started|[0-9]+ lock struct\(s\), heap size [0-9]+, [0-9]+ row lock\(s\))\z/',
73            'FILE I/O' => '/\A(?:I\/O thread [0-9]+ state: waiting for completed aio requests \((?:insert buffer|log|read|write) thread\)|Pending normal aio reads: \[[0-9, ]*\] , aio writes: \[[0-9, ]*\] ,|ibuf aio reads:)\z/',
74            'INSERT BUFFER AND ADAPTIVE HASH INDEX' => '/\AHash table size [0-9]+, node heap has [0-9]+ buffer\(s\)\z/',
75            default => null,
76        };
77        return $physical !== null && preg_match($physical, $line) === 1 ? '' : null;
78    }
