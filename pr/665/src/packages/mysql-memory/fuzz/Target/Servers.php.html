16 *
17 * MYSQL_VERSION names the release (default 8.4.7). MYSQL_MEMORY_NATIVE_DSN, with
18 * MYSQL_MEMORY_NATIVE_USER and MYSQL_MEMORY_NATIVE_PASSWORD, names a running server to use

29    /**
30     * Starts both servers and answers the differential target over them, and the grammar of the release.
31     *
32     * @return array{Differential, string, Server}
33     */
34    public function start(bool $emulate = true): array
35    {
36        $version = getenv('MYSQL_VERSION') !== false ? (string) getenv('MYSQL_VERSION') : MySqlRelease::DEFAULT;
37        $dsn = getenv('MYSQL_MEMORY_NATIVE_DSN');
38        if ($dsn !== false && $dsn !== '') {
39            $user = $this->environment('MYSQL_MEMORY_NATIVE_USER', 'root');
40            $password = $this->environment('MYSQL_MEMORY_NATIVE_PASSWORD', 'root');
41        } else {
42            $endpoint = Testcontainers::run(MySqlRelease::container($version))->getData(Endpoint::class);
43            $dsn = 'mysql:host=' . $endpoint->host . ';port=' . $endpoint->port;
44            $user = $endpoint->username;
45            $password = $endpoint->password;
46        }
47        $native = new PDO($dsn, $user, $password, [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]);
48        $this->clean($native, $version);
49        $globals = [];
50        foreach ($this->rows($native, str_starts_with($version, '5.6.') ? 'SHOW GLOBAL VARIABLES' : 'SELECT VARIABLE_NAME, VARIABLE_VALUE FROM performance_schema.global_variables') as [$name, $value]) {
51            $globals[strtolower($name)] = $value;
52        }
53        $identity = $native->query('SELECT USER()');
54        $account = $identity === false ? '' : $identity->fetchColumn();
55        $account = is_string($account) ? $account : '';
56        $server = Server::start($version, [], $globals, substr($account, (int) strrpos($account, '@') + 1));
57        $this->guard($native, $version, $password);
58        $this->guard(new PDO($server->dsn(), 'root', '', [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]), $version, $password);
59
60        return [new Differential($dsn, $user, $password, $server->dsn(), $emulate, $version, self::GUARD), 'mysql-' . $version, $server];
61    }

63    /**
64     * Creates the account the differential target repairs the servers through, on one server.
65     *
66     * The account has every privilege, and no generated statement names it, so a statement that
67     * drops or locks the account the statements run as cannot take the repair down with it. It is
68     * created on both servers, so that both list the same accounts.
69     */
70    public function guard(PDO $connection, string $version, string $password): void
71    {
72        $account = "'" . self::GUARD . "'@'%'";
73        $quoted = $connection->quote($password);
74        $statements = str_starts_with($version, '5.6.')
75            ? ["GRANT ALL ON *.* TO {$account} IDENTIFIED BY {$quoted} WITH GRANT OPTION"]
76            : ["CREATE USER IF NOT EXISTS {$account} IDENTIFIED BY {$quoted}", "GRANT ALL ON *.* TO {$account} WITH GRANT OPTION"];
77        foreach ($statements as $statement) {
78            $connection->exec($statement);
79        }
80    }

82    /**
83     * Removes what statements of earlier runs left on the MySQL server outside the fixture database.
84     *
85     * Accounts and roles other than root and the server's own, foreign servers, resource groups
86     * other than the defaults, and the database, table and column grants of root are dropped,
87     * so that the MySQL server starts a run in the state mysql-memory starts in. MySQL 5.6 lacks
88     * DROP USER IF EXISTS; the accounts it drops are those it lists, so it drops them without.
89     */
90    public function clean(PDO $native, string $version = MySqlRelease::DEFAULT): void
91    {
92        $statements = [];
93        foreach ($this->rows($native, 'SELECT User, Host FROM mysql.user') as [$name, $host]) {
94            if (!in_array($name, ['root', self::GUARD, 'healthchecker', 'mysql.infoschema', 'mysql.session', 'mysql.sys'], true)) {
95                $statements[] = (str_starts_with($version, '5.6.') ? 'DROP USER ' : 'DROP USER IF EXISTS ') . $native->quote($name) . '@' . $native->quote($host);
96            }
97        }
98        foreach ($this->rows($native, 'SELECT Server_name FROM mysql.servers') as [$name]) {
99            $statements[] = 'DROP SERVER IF EXISTS `' . str_replace('`', '``', $name) . '`';
100        }
101        foreach ($this->rows($native, 'SELECT RESOURCE_GROUP_NAME FROM information_schema.RESOURCE_GROUPS') as [$name]) {
102            if (!in_array($name, ['SYS_default', 'USR_default'], true)) {
103                $statements[] = 'DROP RESOURCE GROUP `' . str_replace('`', '``', $name) . '` FORCE';
104            }
105        }
106        foreach (['db' => 'Db', 'tables_priv' => 'Db', 'columns_priv' => 'Db', 'procs_priv' => 'Db'] as $table => $column) {
107            if ($this->rows($native, "SELECT 1 FROM mysql.{$table} WHERE User = 'root'") !== []) {
108                $statements[] = "DELETE FROM mysql.{$table} WHERE User = 'root'";
109            }
110        }
111        $statements[] = 'FLUSH PRIVILEGES';
112        foreach ($statements as $statement) {
113            $native->exec($statement);
114        }
115    }

117    /**
118     * Answers the rows of a query as lists of strings, or none when the release lacks the table it reads.
119     *
120     * @return list<list<string>>
121     */
122    public function rows(PDO $native, string $sql): array
123    {
124        try {
125            $statement = $native->query($sql);
126        } catch (PDOException) {
127            return [];
128        }
129        $rows = [];
130        foreach ($statement === false ? [] : $statement->fetchAll(PDO::FETCH_NUM) as $row) {
131            $values = [];
132            foreach (is_array($row) ? $row : [] as $value) {
133                $values[] = is_scalar($value) ? (string) $value : '';
134            }
135            $rows[] = $values;
136        }
137
138        return $rows;
139    }
