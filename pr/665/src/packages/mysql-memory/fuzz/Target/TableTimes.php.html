12/**
13 * Validates the actual-clock update times of the freshly inserted fixture tables.
14 *
15 * SHOW TABLE STATUS reads each server's own commit clock, independently of SET timestamp.
16 * For the plain read on modern releases, only an Update_time inside the interval sampled
17 * around that table's INSERT is interchangeable. NULL, the pinned statement clock, a time
18 * outside that interval, other columns, metadata, warnings and table contents stay exact.
19 * SQL is never changed. This contract does not apply to arbitrary expressions or writes.
20 */
21final class TableTimes
22{
23    /**
24     * The comparison contract recorded after every returned table's timestamp was validated.
25     */
26    public const CONTRACT = 'table-update-time-within-fixture-insert-interval';
27
28    /**
29     * @var array<string, array{string, string}> Each fixture table's insertion interval on this server
30     */
31    public array $intervals = [];
32
33    /**
34     * Selects only a plain inspection of the fresh fixture, whose rows cannot change during the statement.
35     */
36    public static function handles(string $sql, string $version): bool
37    {
38        return !str_starts_with($version, '5.') && preg_match('/\A\s*SHOW\s+TABLE\s+STATUS\s*;?\s*\z/i', $sql) === 1;
39    }
40
41    /**
42     * Reads the server's actual clock in its current time zone without using the pinned statement clock.
43     *
44     * @throws RuntimeException When the server does not return a second-resolution SQL datetime
45     */
46    public function now(PDO $pdo): string
47    {
48        $statement = $pdo->query('SELECT SYSDATE()');
49        $value = $statement === false ? false : $statement->fetchColumn();
50        if (!is_string($value) || preg_match('/\A\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\z/', $value) !== 1) {
51            throw new RuntimeException('Cannot sample the fixture commit clock.');
52        }
53
54        return $value;
55    }
56
57    /**
58     * Executes a fixture statement, sampling immediately before and after each table's INSERT.
59     */
60    public function execute(PDO $pdo, string $sql): void
61    {
62        $table = preg_match('/\AINSERT INTO (t1|t2)\b/', $sql, $match) === 1 ? $match[1] : null;
63        $before = $table === null ? null : $this->now($pdo);
64        (new Fixture())->execute($pdo, $sql);
65        if ($table !== null && $before !== null) {
66            $this->intervals[$table] = [$before, $this->now($pdo)];
67        }
68    }
69
70    /**
71     * @param array<string, mixed> $observation The complete raw observation
72     * @return array<string, mixed> A copy with independently validated commit instants represented by their contract
73     */
74    public function comparable(array $observation): array
75    {
76        $results = $observation['results'] ?? null;
77        if (!is_array($results) || count($results) !== 1 || !is_array($results[0] ?? null)) {
78            return $observation;
79        }
80        $result = $results[0];
81        $rows = $result['rows'] ?? null;
82        $columns = $result['columns'] ?? null;
83        if (!is_array($rows) || count($rows) !== 2 || !is_array($columns) || !is_array($columns[12] ?? null) || array_slice($columns[12], 0, 3) !== ['Update_time', 'TABLES', 'DATETIME']) {
84            return $observation;
85        }
86        $seen = [];
87        foreach ($rows as $index => $row) {
88            if (!is_array($row) || !is_string($row[0] ?? null) || !is_string($row[12] ?? null)) {
89                return $observation;
90            }
91            if (isset($seen[$row[0]]) || !$this->contains($row[0], $row[12])) {
92                return $observation;
93            }
94            $seen[$row[0]] = true;
95            $row[12] = '{' . self::CONTRACT . '}';
96            $rows[$index] = $row;
97        }
98        $result['rows'] = $rows;
99        $observation['results'] = [$result];
100        $observation['contracts'] = [self::CONTRACT];
101
102        return $observation;
103    }
104
105    /**
106     * Tells whether a valid SQL datetime belongs to the sampled insertion interval of a table.
107     */
108    public function contains(string $table, string $value): bool
109    {
110        $interval = $this->intervals[$table] ?? null;
111        if ($interval === null || $interval[0] > $interval[1] || $value < $interval[0] || $value > $interval[1] || preg_match('/\A\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\z/', $value) !== 1) {
112            return false;
113        }
114        $parsed = DateTimeImmutable::createFromFormat('!Y-m-d H:i:s', $value, new DateTimeZone('UTC'));
115
116        return $parsed !== false && $parsed->format('Y-m-d H:i:s') === $value;
117    }
118}
