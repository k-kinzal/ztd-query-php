29/**
30 * Resolves ordinary DISTINCT fields and publishes their copied assignments.
31 *
32 * An unsorted bounded result can emit duplicate-free rows as they arrive.
33 * Full results, sorting and SQL_CALC_FOUND_ROWS read stored assignment fields
34 * after input effects finish. Constant expressions retain expression metadata.
35 * Verified through SQL and PDO on MySQL 8.0, 8.4 and 9.1.
36 *
37 * @visibility MySqlMemory
38 */
39final class DistinctStorage

48    /**
49     * Resolves stored metadata and restores direct assignments as rows are emitted.
50     *
51     * @param list<Evaluable> $expressions The select list and hidden sorting keys
52     */
53    public function apply(QueryPlan $plan, Select $select, ?Scope $outer, array $expressions, bool $sorted): QueryPlan
54    {
55        if (!$this->active($select, $outer)) {
56            return $plan;
57        }
58        $buffered = $sorted || ($select->limit === null && $select->late?->limit === null) || in_array(SelectOption::CalcFoundRows, $select->options, true);
59        $domains = [];
60        $origins = [];
61        $outputs = [];
62        $copied = [];
63        foreach ($this->planner->blocks->fields($select) as $position => $field) {
64            $varying = $field->expression !== null && !$this->planner->compiler->constancy($field->expression, assignmentsVary: false)->constant();
65            $domain = $plan->domains[$position];
66            $origin = $plan->origins[$position] ?? null;
67            if ($varying && $domain->kind === Kind::String) {
68                $domain = Domain::of((new Materialization())->column($domain->resolved(), release: $this->planner->settings->release()), $domain->nullable);
69            }
70            $read = new ColumnRead($domain, $position);
71            $value = $expressions[$position];
72            while ($value instanceof Retyped) {
73                $value = $value->evaluable;
74            }
75            if ($buffered && $varying && $value instanceof Assignment) {
76                $read = new Assignment($value->name, $read, $this->planner->compiler->names->stored($domain), $domain);
77                $origin = new ColumnOrigin('', '<temporary>', '', '', $this->planner->materialized([$domain])[0]->flags ?? 0);
78                $copied[$position] = true;
79            }
80            $outputs[] = $read;
81            $domains[] = $domain;
82            $origins[] = $origin;
83        }
84        $root = (new Emission($this->planner))->bounded($this->constants($plan->root, $buffered), count($domains), false);
85        $root = $copied !== [] ? new Project($this->buffer($root, $expressions, $copied), $outputs) : $root;
86
87        return new QueryPlan($root, $domains, $plan->names, $origins);
88    }

90    /**
91     * Keeps empty, constant, aggregate and window queries on their own field paths.
92     */
93    public function active(Select $select, ?Scope $outer): bool
94    {
95        return !$this->planner->settings->legacy() && in_array(SelectOption::Distinct, $select->options, true)
96            && $select->groupBy === null && (new Grouping($this->planner))->collect($select) === []
97            && (new Windowing($this->planner))->calls($select) === []
98            && !$this->planner->blocks->empty($select, $outer)
99            && (new ConstantTables($this->planner->compiler->facts))->joined($select) > 0;
100    }

102    /**
103     * Completes stored input effects below every output bound before copied reads.
104     * Sorting aliases also publish their stored assignment before any row is skipped.
105     *
106     * @param list<Evaluable> $expressions The select list and hidden keys
107     * @param array<int, true> $positions The stored assignment positions
108     */
109    public function buffer(AccessPath $path, array $expressions, array $positions): AccessPath
110    {
111        if ($path instanceof Project) {
112            return new Project($this->buffer($path->input, $expressions, $positions), $path->expressions, $path->append, $path->retainAt, $path->publish, $path->order, $path->initial);
113        }
114        if ($path instanceof Limit) {
115            return new Limit($this->buffer($path->input, $expressions, $positions), $path->count, $path->offset, $path->countFound, $path->exhaust, $path->initialize);
116        }
117        if ($path instanceof Sort) {
118            $keys = array_intersect_key(array_fill_keys(array_column($path->keys, 0), true), $positions);
119            $input = (new \MySqlMemory\Plan\Window\CopiedAssignments($this->planner))->sorted(new Buffer($path->input), $expressions, $keys);
120
121            return new Sort($input, $path->keys);
122        }
123
124        return new Buffer($path);
125    }

127    /**
128     * Holds DISTINCT input values before deferred constant assignments are emitted.
129     */
130    public function constants(AccessPath $path, bool $buffered): AccessPath
131    {
132        if ($path instanceof Limit) {
133            return new Limit($this->constants($path->input, $buffered), $path->count, $path->offset, $path->countFound, $path->exhaust, $path->initialize);
134        }
135        if ($path instanceof Sort) {
136            return new Sort($this->constants($path->input, $buffered), $path->keys);
137        }
138        if ($buffered && $path instanceof Project) {
139            return new Project(new Buffer($path->input), $path->expressions, $path->append, $path->retainAt, $path->publish, $path->order, $path->initial);
140        }
141
142        return $path;
143    }
