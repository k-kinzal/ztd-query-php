57/**
58 * Plans the FROM clause of a query block: each relation occurrence is placed in the row of the block and read by an access path.
59 *
60 * A comma join and a join without a condition pair every row; USING and NATURAL compare the
61 * columns of one name on both sides. A derived table is computed in a frame of its own, inside
62 * the block's for a LATERAL one.
63 *
64 * @visibility MySqlMemory
65 */

75    /**
76     * Plans a relation and places its occurrences in the scope of the block.
77     *
78     * @throws SqlError When a relation cannot be read
79     */
80    public function plan(Relation $relation, Scope $scope): AccessPath
81    {
82        return match (true) {
83            $relation instanceof TableReference => $this->table($relation, $scope),
84            $relation instanceof DerivedTable => $this->derived($relation, $scope),
85            $relation instanceof JoinedTable => $this->join($relation, $scope),
86            $relation instanceof TableList => $this->list($relation, $scope),
87            $relation instanceof NestedRelation, $relation instanceof OdbcJoin, $relation instanceof EscapedRelation => $this->plan($relation->relation, $scope),
88            $relation instanceof Dual => new SingleRow(),
89            $relation instanceof JsonTable => $this->jsonTable($relation, $scope),
90            default => throw StatementError::NotSupportedYet->error('relation ' . (new ReflectionClass($relation))->getShortName()),
91        };
92    }

94    /**
95     * Plans a table reference: a stored table, a system table with the rows it holds now, or a common table expression.
96     *
97     * @throws SqlError When the table does not exist
98     */
99    public function table(TableReference $reference, Scope $scope): AccessPath
100    {
101        $resolution = $this->planner->compiler->facts->relation($reference)->table;
102        if ($resolution instanceof CommonTable) {
103            $plan = $this->planner->commonTable($resolution->definition);
104            $merged = $resolution->definition instanceof \SqlSemantics\Platform\MySql\Statement\Query\With\CommonTableExpression && $this->mergeable($resolution->definition->query);
105            $placed = $this->shaped($reference, $plan->domains);
106            $scope->place($reference, $placed, $plan->names);
107            $scope->merged[spl_object_id($reference)] = $merged ? $plan->origins : $this->planner->materialized($placed);
108            $scope->derived[spl_object_id($reference)] = $reference->alias->value ?? $reference->name->name->value;
109
110            return new Materialize($plan);
111        }
112        if (!$resolution instanceof DeclaredTable) {
113            throw QueryError::NoSuchTable->error($reference->name->schema->value ?? $this->planner->settings->database, $reference->name->name->value);
114        }
115        $name = $resolution->table->name;
116        $view = $this->planner->dictionary->schema($name->schema->value ?? $this->planner->settings->database)->views[$name->name->value] ?? null;
117        if ($view !== null && $resolution->table === $view->declaration) {
118            return (new Views($this->planner))->plan($reference, $view, $scope, $this);
119        }
120        $system = $this->planner->dictionary->system;
121        $read = $system?->table($resolution->table);
122        $stored = $system !== null && $read !== null ? $system->read($read, $this->planner->compiler->connection) : $this->planner->dictionary->table($name->schema->value ?? $this->planner->settings->database, $name->name->value);
123        if ($stored === null) {
124            throw QueryError::NoSuchTable->error($name->schema->value ?? $this->planner->settings->database, $name->name->value);
125        }
126        $definition = $stored->definition;
127        $selected = null;
128        if ($reference->partitions !== []) {
129            if ($definition->partitioning === null) {
130                throw SchemaError::PartitionClauseOnNonpartitioned->error();
131            }
132            $selected = (new \MySqlMemory\Storage\Partitions($definition, $definition->partitioning, $this->planner->compiler->connection->context))->selected($reference->partitions);
133        }
134        $scope->place($reference, array_map(static fn ($column) => $column->domain, $definition->columns), array_map(static fn ($column): string => $column->name, $definition->columns), $definition);
135        $scan = new TableScan($stored, $selected);
136        $scope->scans[spl_object_id($reference)] = $scan;
137
138        return $scan;
139    }

142     * Plans a derived table.
143     */
144    public function derived(DerivedTable $derived, Scope $scope): AccessPath
145    {
146        $plan = $this->planner->query($derived->query, $derived->lateral ? $scope : $scope->outer);
147        $names = $derived->columns === [] ? $plan->names : array_map(static fn ($name): string => $name->value, $derived->columns);
148        $merged = $this->mergeable($derived->query);
149        $placed = $this->shaped($derived, $plan->domains);
150        $scope->place($derived, $placed, $names);
151        $scope->merged[spl_object_id($derived)] = $merged ? $plan->origins : $this->planner->materialized($placed);
152        $scope->derived[spl_object_id($derived)] = $derived->alias->value ?? '';
153
154        return new Materialize($plan, $derived->lateral);
155    }

157    /**
158     * Plans JSON_TABLE: its document sees the tables before it, and its columns take their declared types.
159     *
160     * A counter is an unsigned BIGINT of ten digits. ON ERROR written before ON EMPTY is deprecated
161     * with a warning. A DEFAULT value of ON EMPTY or ON ERROR is read as
162     * a JSON text when the statement is resolved, and may be an array or an object only for a JSON
163     * column (ER_INVALID_DEFAULT) (verified on a live 8.4 server).
164     * Source: https://dev.mysql.com/doc/refman/8.4/en/json-table-functions.html.
165     *
166     * @throws SqlError When the document, a path or a DEFAULT value is not valid
167     */
168    public function jsonTable(JsonTable $table, Scope $scope): AccessPath
169    {
170        $document = $this->planner->compiler->compile($table->document, $scope);
171        $declared = new Declared($this->planner->settings->connectionCollation, $this->planner->settings->release());
172        $columns = array_map(fn (JsonTableColumn $column): JsonColumn => $this->jsonColumn($column, $declared), $table->columns);
173        [$domains, $names] = self::flattened($columns);
174        $scope->place($table, $domains, $names);
175        $scope->merged[spl_object_id($table)] = $this->planner->materialized($domains);
176        $scope->derived[spl_object_id($table)] = $table->alias->value ?? '';
177
178        return new JsonTableScan($document, self::jsonPath($table->path->value()), $columns);
179    }

181    /**
182     * Answers the types and names of the columns of JSON_TABLE, nested columns flattened in order.
183     *
184     * @param list<JsonColumn> $columns
185     * @return array{list<Domain>, list<string>}
186     */
187    public static function flattened(array $columns): array
188    {
189        $domains = [];
190        $names = [];
191        foreach ($columns as $column) {
192            if ($column->kind === 'nested') {
193                [$innerDomains, $innerNames] = self::flattened($column->columns);
194                array_push($domains, ...$innerDomains);
195                array_push($names, ...$innerNames);
196                continue;
197            }
198            $domains[] = $column->domain;
199            $names[] = $column->name;
200        }
201
202        return [$domains, $names];
203    }

205    /**
206     * Prepares one column of JSON_TABLE.
207     *
208     * @throws SqlError When a path or a DEFAULT value is not valid
209     */
210    public function jsonColumn(JsonTableColumn $column, Declared $declared): JsonColumn
211    {
212        if ($column instanceof OrdinalityColumn) {
213            return new JsonColumn('ordinality', $column->name->value, Domain::integer(Field::LongLong, 10, true)->withNullable(true));
214        }
215        if ($column instanceof NestedColumns) {
216            return new JsonColumn('nested', '', Domain::null(), self::jsonPath($column->path->value()), [JsonResponseKind::Null, null], [JsonResponseKind::Null, null], array_map(fn (JsonTableColumn $inner): JsonColumn => $this->jsonColumn($inner, $declared), $column->columns));
217        }
218        if (!$column instanceof PathColumn) {
219            throw StatementError::NotSupportedYet->error('a JSON_TABLE column');
220        }
221        $domain = Domain::of($declared->tableFunction($column->type), true);
222        $name = $column->name->value;
223        $response = static function (?JsonResponse $response) use ($domain, $name): array {
224            $default = $response?->default;
225            if ($default === null) {
226                return [$response->kind ?? JsonResponseKind::Null, null];
227            }
228            try {
229                $value = Jsons::parse($default instanceof StringLiteral ? $default->value() : '', 1, 'JSON_TABLE');
230            } catch (SqlError $failure) {
231                throw new SqlError($failure->error, $failure->getMessage(), $failure, [[SchemaError::InvalidDefault->value, SchemaError::InvalidDefault->message($name)]]);
232            }
233            if ($domain->kind !== Kind::Json && ($value->type === JsonKind::Array || $value->type === JsonKind::Object)) {
234                throw SchemaError::InvalidDefault->error($name);
235            }
236
237            return [JsonResponseKind::Default, $value];
238        };
239
240        return new JsonColumn($column->exists ? 'exists' : 'path', $name, $domain, self::jsonPath($column->path->value()), $response($column->onEmpty), $response($column->onError));
241    }

274    /**
275     * Tells whether the server merges a derived query into the query that reads it instead of materializing it.
276     *
277     * A single query block that reads tables merges, unless it groups, aggregates, removes
278     * duplicates, limits its rows or has a window; its columns are then the columns and
279     * expressions it reads, with the keys of the columns. MySQL 5.6 materializes every derived
280     * table, as merging came with 5.7, so their columns carry no keys there (verified on a live
281     * 5.6.51 server).
282     * Source: https://dev.mysql.com/doc/refman/8.4/en/derived-table-optimization.html,
283     * https://dev.mysql.com/doc/refman/5.7/en/derived-table-optimization.html.
284     */
285    public function mergeable(\SqlSemantics\Statement\Query $query): bool
286    {
287        return $this->planner->settings->release() !== \SqlSemantics\Contract\GrammarRelease::MySql5651 && (new \SqlSemantics\Platform\MySql\Rules\Typing\Materialization())->mergeable($query);

311    /**
312     * Plans a join.
313     *
314     * @throws SqlError When a condition cannot be compiled
315     */
316    public function join(JoinedTable $join, Scope $scope): AccessPath
317    {
318        $before = array_keys($scope->offsets);
319        $left = $this->plan($join->left, $scope);
320        $middle = array_keys($scope->offsets);
321        $right = $this->plan($join->right, $scope);
322        $after = array_keys($scope->offsets);
323        $kind = $join->operator->keepsRight() ? JoinKind::Right : ($join->operator->keepsLeft() ? JoinKind::Left : JoinKind::Inner);
324        $condition = null;
325        if ($join->on !== null) {
326            $condition = $this->planner->compiler->compile($join->on, $scope);
327        } elseif ($join->using !== [] || $join->operator->natural()) {
328            $leftIds = array_values(array_diff($middle, $before));
329            $rightIds = array_values(array_diff($after, $middle));
330            $names = $join->operator->natural() ? $this->common($scope, $leftIds, $rightIds) : array_map(static fn ($name): string => $name->value, $join->using);
331            $condition = $this->equalities($scope, $leftIds, $rightIds, $names);
332        }
333
334        return new NestedLoopJoin($left, $right, $kind, $condition, $this->lateral($join->right));
335    }

337    /**
338     * Answers the column names both sides of a natural join have, in the order of the left side.
339     *
340     * @param list<int> $left
341     * @param list<int> $right
342     * @return list<string>
343     */
344    public function common(Scope $scope, array $left, array $right): array
345    {
346        $rightNames = [];
347        foreach ($right as $id) {
348            foreach ($this->visible($scope, $id) as $name) {
349                $rightNames[strtolower($name)] = true;
350            }
351        }
352        $names = [];
353        foreach ($left as $id) {
354            foreach ($this->visible($scope, $id) as $name) {
355                if (isset($rightNames[strtolower($name)]) && !in_array(strtolower($name), array_map('strtolower', $names), true)) {
356                    $names[] = $name;
357                }
358            }
359        }
360
361        return $names;
362    }

381    /**
382     * Builds the conjunction of equalities between the columns of each name on both sides.
383     *
384     * @param list<int> $left
385     * @param list<int> $right
386     * @param list<string> $names
387     */
388    public function equalities(Scope $scope, array $left, array $right, array $names): ?Evaluable
389    {
390        $condition = null;
391        foreach ($names as $name) {
392            $leftColumn = $this->find($scope, $left, $name);
393            $rightColumn = $this->find($scope, $right, $name);
394            if ($leftColumn === null || $rightColumn === null) {
395                throw QueryError::BadField->error($name, 'from clause');
396            }
397            $comparator = Comparator::of($leftColumn->domain(), $rightColumn->domain(), '=', $this->planner->settings->connectionCollation, $this->planner->settings->release());
398            $equality = new Compare(ComparisonOperator::Equal, $leftColumn, $rightColumn, $comparator, $this->planner->compiler->operators->truth(true));
399            $condition = $condition === null ? $equality : new Logic(LogicalOperator::And, $condition, $equality, $this->planner->compiler->operators->truth(true));
400        }
401
402        return $condition;
403    }
