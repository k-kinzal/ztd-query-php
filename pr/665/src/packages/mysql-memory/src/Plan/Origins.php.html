17/**
18 * Answers the base column each output column of a query block reads, as the column metadata of its result reports it.
19 *
20 * A select item that is a window function is a column of the temporary table of its window. A
21 * select item of a block WITH ROLLUP that is a grouping expression reads no base column, but in
22 * MySQL 5.6 and 5.7 (verified on live 5.6.51 and 5.7.44 servers); when the rows pass through a
23 * temporary table, for ORDER BY or DISTINCT, it is a column of that table.
24 * The columns of a block with SQL_BUFFER_RESULT, or with DISTINCT over more than one table that is
25 * not constant ({@see ConstantTables}), are of no key, unless the server finds at once that the
26 * block reads no row (verified on a live 8.4 server).
27 *
28 * @visibility MySqlMemory
29 */

39    /**
40     * Answers the base column each output column of a block reads: none for a rolled-up item,
41     * unless the rows pass through a temporary table, and of no key when the result is buffered.
42     *
43     * @param list<Field> $fields The fields of the select list
44     * @param list<bool> $rolled Whether each item is a grouping expression of a block WITH ROLLUP
45     * @param list<Domain> $domains The domains of the select list
46     * @param bool $sorted Whether the block sorts its rows
47     * @return list<ColumnOrigin|null>
48     */
49    public function origins(Select $select, ?Scope $outer, Scope $scope, array $fields, array $rolled, array $domains, bool $sorted): array
50    {
51        $buffered = (in_array(SelectOption::BufferResult, $select->options, true) || (in_array(SelectOption::Distinct, $select->options, true) && (new ConstantTables($this->planner->compiler->facts))->joined($select) > 1)) && !$this->planner->blocks->empty($select, $outer);
52        $materialized = $sorted || in_array(SelectOption::Distinct, $select->options, true);
53        $implicit = $this->planner->compiler->settings->legacy() && $select->groupBy === null && (new Grouping($this->planner))->collect($select) !== [];
54        $supplied = $implicit ? $this->supplied($select->from) : [];
55        $origins = array_map(fn (Field $field, bool $rolled, Domain $domain): ?ColumnOrigin => match (true) {
56            $field->expression !== null && Windowing::windowed($field->expression) => $this->planner->materialized([$domain])[0],
57            $rolled => $materialized ? $this->planner->materialized([$domain])[0] : ($this->planner->compiler->settings->legacy() ? $this->origin($field, $scope) : null),
58            $implicit => $this->kept($field, $scope, $this->origin($field, $scope), $supplied),
59            default => $this->origin($field, $scope),
60        }, $fields, $rolled, $domains);
61
62        return $buffered ? array_map(static fn (?ColumnOrigin $origin): ?ColumnOrigin => $origin?->unkeyed(), $origins) : $origins;
63    }

68     * @param list<int> $supplied The object ids of the relations an outer join supplies NULL for

71    {
72        $resolution = match (true) {

74            $field->expression === null => $field->resolution,
75            default => null,

82        return $domain === null || $domain->nullable ? $origin : new ColumnOrigin($origin->schema, $origin->table, $origin->originalTable, $origin->column, $origin->flags | \MySqlMemory\Result\ColumnFlag::NotNull->value, $origin->exact);

87     *

91    {
92        $supplied = [];
93        foreach ($from === null ? [] : (new Walker())->find($from, \SqlSemantics\Platform\MySql\Statement\Relation\JoinedTable::class, false) as $join) {
94            $sides = [...($join->operator->keepsLeft() && !$join->operator->keepsRight() ? [$join->right] : []), ...($join->operator->keepsRight() && !$join->operator->keepsLeft() ? [$join->left] : [])];
95            foreach ($sides as $side) {
96                foreach ([$side, ...(new Walker())->find($side, \SqlSemantics\Statement\Relation::class, false)] as $relation) {

99            }

105    /**
106     * Answers the base column an output column reads directly, if any; DEFAULT(column) reports the column it reads, a variable of a stored program carries the flags of a column of its type, without BINARY unless it is a string, and a BLOB or TEXT result of a stored function is flagged BLOB (verified on a live 8.4 server).
107     */
108    public function origin(Field $field, Scope $scope): ?ColumnOrigin
109    {
110        $resolution = $field->resolution;
111        if ($field->expression instanceof ColumnUse) {
112            $resolution = $this->planner->compiler->facts->scalar($field->expression)->resolution;
113        }
114        if ($field->expression instanceof \SqlSemantics\Platform\MySql\Statement\Expression\Access\DefaultOfColumn) {
115            $resolution = $this->planner->compiler->facts->scalar($field->expression->column)->resolution;
116        }
117        if ((new ProgramColumns($this->planner))->called($field)) {
118            return (new ProgramColumns($this->planner))->flagged($field, false);
119        }
120        if (!$resolution instanceof ResolvedColumn) {
121            return null;
122        }
123        $id = spl_object_id($resolution->relation);
124        if (isset($scope->derived[$id])) {
125            $position = $this->planner->compiler->names->position($scope, $resolution);
126
127            $inner = $scope->merged[$id][$position] ?? null;
128
129            return new ColumnOrigin($inner->schema ?? '', $scope->derived[$id], $inner->originalTable ?? '', $scope->names[$id][$position] ?? '', $inner->flags ?? 0);
130        }
131        if (!isset($scope->tables[$id])) {
132            return $field->expression instanceof ColumnUse && $this->planner->compiler->connection->program !== null && $scope->locate($resolution->relation) === null ? (new ProgramColumns($this->planner))->flagged($field, true) : null;
133        }
134        $definition = $scope->tables[$id];
135        $position = $this->planner->compiler->names->position($scope, $resolution);
136        $alias = $resolution->relation instanceof \SqlSemantics\Platform\MySql\Statement\Relation\TableReference && $resolution->relation->alias !== null ? $resolution->relation->alias->value : $definition->name;
137        $system = $this->planner->dictionary->system?->origin($definition, $position, $alias);
138        if ($system !== null) {
139            return $system;
140        }
141
142        return new ColumnOrigin($definition->schema, $alias, $definition->name, $definition->columns[$position]->name, $definition->flags($position));
143    }
