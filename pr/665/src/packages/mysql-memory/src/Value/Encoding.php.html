9/**
10 * The bytes of strings in the character sets of the server, and their conversion between sets.
11 *
12 * A string holds the bytes of its character set: utf8mb4 and utf8mb3 hold UTF-8, latin1 holds
13 * Windows-1252 with its five unassigned bytes read as the C1 controls, ucs2, utf16 and utf32 hold
14 * big-endian code units, and binary holds bytes. A character the target set cannot hold converts
15 * to a question mark. A set PHP has no conversion for is read as ASCII.
16 * Source: https://dev.mysql.com/doc/refman/8.4/en/charset-charsets.html,
17 * https://dev.mysql.com/doc/refman/8.4/en/charset-conversion.html.
18 *
19 * @visibility MySqlMemory
20 */

28        'cp866' => 'CP866', 'cp932' => 'CP932', 'eucjpms' => 'eucJP-win', 'euckr' => 'EUC-KR', 'gb18030' => 'GB18030',

30        'koi8u' => 'KOI8-U', 'latin1' => 'Windows-1252', 'latin2' => 'ISO-8859-2', 'latin5' => 'ISO-8859-9',

37     */

48        return $charset->name === 'utf8mb4' || $charset->name === 'utf8mb3';
49    }

51    /**
52     * Converts the bytes of a string from one character set to another.
53     *
54     * A binary string is taken, or given, as its bytes.
55     */
56    public static function convert(string $text, Charset $from, Charset $to): string
57    {
58        $source = self::name($from);
59        $target = self::name($to);
60        if ($source === null || $target === null || $from === $to || ($source === 'UTF-8' && $to->name === 'utf8mb4')) {
61            return $text;
62        }
63        if ($source === 'ASCII' && $from->name !== 'ascii') {
64            $text = (string) preg_replace('/[\x80-\xFF]/', '?', $text);
65        }
66        $converted = (string) mb_convert_encoding($text, $target, $source);
67
68        return $to->name === 'utf8mb3' ? (string) preg_replace('/[\xF0-\xF4][\x80-\xBF]{3}/', '?', $converted) : $converted;
69    }

71    /**
72     * Tells whether the bytes of a string are characters of a character set.
73     */
74    public static function valid(string $text, Charset $charset): bool
75    {
76        $name = self::name($charset);
77        if ($name === null) {
78            return true;
79        }
80        if ($charset->name === 'utf8mb3' && preg_match('/[\xF0-\xF4]/', $text) === 1) {
81            return false;
82        }
83
84        return mb_check_encoding($text, $name);
85    }

91    {

100    /**
101     * Answers the length in bytes of the start of a string whose characters another character set holds.
102     */
103    public static function convertible(string $text, Charset $from, Charset $to): int
104    {
105        $offset = 0;
106        foreach (self::characters($text, $from) as $character) {
107            if (self::convert(self::convert($character, $from, $to), $to, $from) !== $character) {
108                return $offset;
109            }
110            $offset += strlen($character);
111        }
112
113        return $offset;
114    }

117     * Splits a string into the characters of its character set; bytes that are no character split one by one.
118     *
119     * @return list<string>
120     */
121    public static function characters(string $text, Charset $charset): array
122    {
123        $name = self::name($charset);
124        if ($text === '') {
125            return [];
126        }
127        if ($name === null || $charset->maxLength === 1 || !mb_check_encoding($text, $name)) {
128            return str_split($text);
129        }
130
131        return mb_str_split($text, 1, $name);
132    }

137    public static function length(string $text, Charset $charset): int
138    {
139        $name = self::name($charset);
140        if ($name === null || $charset->maxLength === 1 || !mb_check_encoding($text, $name)) {
141            return strlen($text);
142        }
143

145    }

148     * Answers the characters of a string from a position counted from 0, at most a number of them.
149     */
150    public static function slice(string $text, int $start, ?int $length, Charset $charset): string
151    {

157        return mb_substr($text, $start, $length, $name);
