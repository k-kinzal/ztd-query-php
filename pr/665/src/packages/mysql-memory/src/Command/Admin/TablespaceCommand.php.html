38/**
39 * Executes the statements on InnoDB general and undo tablespaces: CREATE, ALTER, RENAME and DROP TABLESPACE, and CREATE, ALTER and DROP UNDO TABLESPACE.
40 *
41 * Each commits the open transaction. An ENGINE option must name an enabled storage engine
42 * (ER_UNKNOWN_STORAGE_ENGINE), and InnoDB (ER_ILLEGAL_HA_CREATE_OPTION). The names mysql,
43 * innodb_system and innodb_temporary, and the names starting with innodb_, are reserved; the
44 * server has those tablespaces, and the undo tablespaces innodb_undo_001 and innodb_undo_002.
45 * Names compare with regard to letter case. A data file name must end with .ibd (.ibu for an
46 * undo tablespace) and may not repeat the file of another tablespace; ENCRYPTION 'Y' needs a
47 * keyring the server lacks, and ALTER TABLESPACE checks the keyring before the value; AUTOEXTEND_SIZE
48 * is 0 or a multiple of 4M from 4M to 4096M. A refused option is followed by the failure of the
49 * statement and an error of the engine. ALTER and
50 * DROP TABLESPACE refuse an ENGINE option on an existing tablespace with a syntax error, and an
51 * active undo tablespace is not empty (verified on a live 8.4 server). The emulator does not
52 * track which tables use a tablespace, so DROP TABLESPACE never finds one in use.
53 * Source: https://dev.mysql.com/doc/refman/8.4/en/create-tablespace.html,
54 * https://dev.mysql.com/doc/refman/8.4/en/alter-tablespace.html,
55 * https://dev.mysql.com/doc/refman/8.4/en/drop-tablespace.html,
56 * https://dev.mysql.com/doc/refman/8.4/en/innodb-undo-tablespaces.html.
57 *
58 * @visibility MySqlMemory
59 */
60final class TablespaceCommand implements Command

80    public function execute(Operation $operation, Session $session, Context $context, Connection $connection): Reply
81    {
82        $statement = $operation->statement;
83        $registry = $session->instance->registry;
84        $session->transaction->commit();
85        match (true) {
86            $statement instanceof CreateTablespace => $this->create($statement, $registry),
87            $statement instanceof CreateUndoTablespace => $this->createUndo($statement, $registry),
88            $statement instanceof RenameTablespace => $this->rename($statement, $registry),
89            $statement instanceof AlterTablespace => $this->alter($statement->name->value, $statement->options, $registry),
90            $statement instanceof AlterTablespaceDatafile => $this->datafile($statement, $registry),
91            $statement instanceof AlterTablespaceAccess => throw StatementError::SyntaxError->error(),
92            $statement instanceof AlterUndoTablespace => $this->activate($statement, $registry),
93            $statement instanceof DropTablespace => $this->drop($statement, $registry),
94            $statement instanceof DropUndoTablespace => $this->dropUndo($statement, $registry),
95            default => throw StatementError::NotSupportedYet->error('this tablespace statement'),
96        };
97
98        return new Completion();
99    }

101    /**
102     * Creates a general tablespace.
103     *
104     * @throws SqlError When the engine, the name, the file or an option is refused
105     */
106    public function create(CreateTablespace $statement, Registry $registry): void
107    {
108        $name = $statement->name->value;
109        $this->engine($statement->options, 'CREATE TABLESPACE');
110        $this->reserved($name);
111        if ($this->exists($name, $registry)) {
112            throw SchemaError::TablespaceExists->error($name);
113        }
114        $failed = 'TABLESPACE ' . $name;
115        $file = $statement->datafile === null ? '' : (new Literals())->bytes($statement->datafile);
116        if ($statement->datafile !== null) {
117            $this->file($file, '.ibd', $failed, $name, $registry);
118        }
119        foreach ($statement->options as $option) {
120            if ($option instanceof SizeOption && $option->kind === SizeOptionKind::FileBlock && !in_array($this->size($option), ['1024', '2048', '4096', '8192', '16384'], true)) {
121                throw new SqlError(SchemaError::IllegalCreateOption, 'InnoDB does not support FILE_BLOCK_SIZE=' . $this->size($option), null, [[SchemaError::CreateFilegroupFailed->value, SchemaError::CreateFilegroupFailed->message($failed)], [SchemaError::IllegalHa->value, SchemaError::IllegalHa->message($name)]]);
122            }
123        }
124        $this->options($statement->options, [SchemaError::CreateFilegroupFailed->value, SchemaError::CreateFilegroupFailed->message($failed)], $name, true);
125        $registry->tablespaces[$name] = new Tablespace($name, false, $file);
126    }

208     * Makes an undo tablespace active or inactive, once its ENGINE option names InnoDB.
209     *
210     * @throws SqlError When the engine is refused, or the tablespace is missing or a general one
211     */
212    public function activate(AlterUndoTablespace $statement, Registry $registry): void
213    {
214        $name = $statement->name->value;
215        $this->engine($statement->options, 'ALTER UNDO TABLESPACE');
216        if (isset(self::SYSTEM[$name]) && self::SYSTEM[$name]) {
217            return;
218        }
219        $tablespace = $this->find($name, $registry);
220        if (!$tablespace->undo) {
221            throw new SqlError(SchemaError::WrongTablespaceName, 'Cannot ALTER UNDO TABLESPACE `' . $name . '` because it is a general tablespace.  Please use ALTER TABLESPACE.', null, [[SchemaError::AlterFilegroupFailed->value, SchemaError::AlterFilegroupFailed->message('UNDO TABLESPACE ' . $name)], [AdministrationError::OperationDisallowed->value, AdministrationError::OperationDisallowed->message('ALTER UNDO TABLEPSPACE', $name)]]);
222        }
223        $tablespace->active = $statement->active;
224    }

274    public function engine(array $options, string $operation): void
275    {
276        $named = null;
277        foreach ($options as $option) {
278            if ($option instanceof EngineOption) {

282        if ($named === null) {
283            return;
284        }

292     * Refuses a reserved tablespace name.
293     *
294     * @throws SqlError When the name is reserved or empty
295     */
296    public function reserved(string $name): void
297    {
298        $reason = match (true) {
299            $name === '' => null,
300            in_array($name, ['innodb_system', 'innodb_temporary', 'mysql'], true) => 'InnoDB: `' . $name . '` is a reserved tablespace name.',
301            str_starts_with($name, 'innodb_') => 'InnoDB: Tablespace names starting with `innodb_` are reserved.',
302            default => false,
303        };
304        if ($reason === false) {
305            return;
306        }
307        $incorrect = SchemaError::WrongTablespaceName->message($name);
308
309        throw new SqlError(SchemaError::WrongTablespaceName, $reason ?? $incorrect, null, [[SchemaError::WrongTablespaceName->value, $incorrect]]);
310    }

330    /**
331     * Checks the name of a data file.
332     *
333     * @throws SqlError When the name lacks the extension, names a directory or repeats the file of another tablespace
334     */
335    public function file(string $file, string $extension, string $failed, string $name, Registry $registry): void
336    {
337        $base = basename($file);
338        $reasons = [];
339        if (str_starts_with($file, '/')) {
340            $reasons[] = 'The DATAFILE location must be in a known directory.';
341        } elseif (str_contains($file, '/')) {
342            $reasons = ['The directory does not exist or is incorrect.', 'The DATAFILE location cannot be under the datadir.'];
343        } elseif (!str_ends_with($base, $extension) || $base === $extension) {
344            $reasons[] = 'The ADD DATAFILE filepath does not have a proper filename.';
345            if (!str_ends_with($base, $extension)) {
346                $reasons[] = "The ADD DATAFILE filepath must end with '" . $extension . "'.";
347            }
348        }
349        if ($reasons !== []) {
350            $following = array_map(static fn (string $reason): array => [SchemaError::WrongFileName->value, $reason], array_slice($reasons, 1));
351            $following[] = [SchemaError::CreateFilegroupFailed->value, SchemaError::CreateFilegroupFailed->message($failed)];
352            $following[] = [SchemaError::WrongFileName->value, SchemaError::WrongFileName->message($file)];
353
354            throw new SqlError(SchemaError::WrongFileName, $reasons[0], null, $following);
355        }
356        foreach ($registry->tablespaces as $tablespace) {
357            if ($tablespace->file === $file) {
358                throw SchemaError::DuplicateTablespaceFile->error($name);
359            }
360        }
361    }

363    /**
364     * Checks the ENCRYPTION and AUTOEXTEND_SIZE options.
365     *
366     * @param list<StorageOption> $options
367     * @param array{int, string} $failed The error that follows a refused option
368     *
369     * @throws SqlError When an option is refused
370     */
371    public function options(array $options, array $failed, string $name, bool $creating): void
372    {
373        foreach ($options as $option) {
374            if ($option instanceof EncryptionOption) {
375                $value = strtoupper((new Literals())->bytes($option->encryption));
376                $engine = $creating ? [AdministrationError::GetErrno->value, AdministrationError::GetErrno->message('138', 'Unsupported extension used for table')] : [AdministrationError::GetErrno->value, AdministrationError::GetErrno->message('168', 'Unknown (generic) error from engine')];
377                if ($creating && !in_array($value, ['Y', 'N'], true)) {
378                    throw new SqlError(SchemaError::InvalidEncryption, SchemaError::InvalidEncryption->message(), null, [$failed, $engine]);
379                }
380                if ($value === 'Y' || !$creating) {
381                    throw new SqlError(AdministrationError::KeyringMissing, AdministrationError::KeyringMissing->message(), null, [$failed, $engine]);
382                }
383            }
384            if ($option instanceof SizeOption && $option->kind === SizeOptionKind::Autoextend) {
385                $size = $this->size($option);
386                $mega = '4194304';
387                $code = match (true) {
388                    $size !== '0' && (bccomp($size, $mega) < 0 || bccomp($size, bcmul($mega, '1024')) > 0) => SchemaError::AutoextendSize,
389                    bcmod($size, $mega) !== '0' => SchemaError::AutoextendMultiple,
390                    default => null,
391                };
392                if ($code !== null) {
393                    $engine = $creating ? [SchemaError::IllegalHa->value, SchemaError::IllegalHa->message($name)] : [AdministrationError::GetErrno->value, AdministrationError::GetErrno->message((string) $code->value, 'Unknown error ' . $code->value)];
394
395                    throw new SqlError($code, $code === SchemaError::AutoextendMultiple ? $code->message('4M') : $code->message(), null, [$failed, $engine]);
396                }
397            }
398        }
399    }
