29/**
30 * Writes the function calls of an expression of a table definition as the server stores them: the calls, casts and temporal arithmetic.
31 *
32 * A function is written in lower case under the name the server keeps for it: SUBSTRING and MID
33 * are substr, UCASE upper, LCASE lower, POWER pow, CHARACTER_LENGTH char_length, ISNULL() a null
34 * test; DATE(), TIME() and TIMESTAMP() of one argument are casts; DATE_ADD, ADDDATE, DATE_SUB and
35 * SUBDATE are additions of an interval; CAST to CHAR names the character set; a JSON path
36 * operator is JSON_EXTRACT (verified on a live 8.4 server).
37 * Source: https://dev.mysql.com/doc/refman/8.4/en/show-create-table.html.
38 *
39 * @visibility MySqlMemory
40 */
41final class CallText

44     * The name the server keeps for a function written under another, by lowercase name.
45     *
46     * @var array<string, string>
47     */
48    public const NAMES = ['substring' => 'substr', 'mid' => 'substr', 'ucase' => 'upper', 'lcase' => 'lower', 'power' => 'pow', 'character_length' => 'char_length', 'schema' => 'database', 'session_user' => 'user', 'system_user' => 'user', 'current_user' => 'current_user'];

57    /**
58     * Writes a call, a cast or temporal arithmetic, or answers null for an expression the writer does not know.
59     */
60    public function scalar(Scalar $scalar): ?string
61    {
62        return match (true) {
63            $scalar instanceof FunctionCall => $this->function($scalar),
64            $scalar instanceof KeywordCall => $this->keyword($scalar),
65            $scalar instanceof ClockCall => strtolower($scalar->clock->value) . '(' . ($scalar->precision->text ?? '') . ')',
66            $scalar instanceof Cast => $this->items->wrap($scalar->operand, 'cast(', ' as ' . $this->target($scalar->target) . ')'),
67            $scalar instanceof BinaryCast => $this->items->wrap($scalar->operand, 'cast(', ' as char charset binary)'),
68            $scalar instanceof CharsetConversion => $this->items->wrap($scalar->operand, 'convert(', ' using ' . strtolower($scalar->charset->name->value ?? 'binary') . ')'),
69            $scalar instanceof JsonExtraction => $this->json($scalar),
70            $scalar instanceof Trim => $this->trim($scalar),
71            $scalar instanceof Extract => $this->items->wrap($scalar->source, 'extract(' . strtolower($scalar->unit->value) . ' from ', ')'),
72            $scalar instanceof Position => $this->call('locate', [$scalar->substring, $scalar->string]),
73            $scalar instanceof CharCall => $this->items->joined(array_map($this->items->scalar(...), $scalar->arguments), 'char(', ',', $scalar->charset === null ? ')' : ' using ' . strtolower($scalar->charset->name->value ?? 'binary') . ')'),
74            $scalar instanceof DefaultOfColumn => $this->call('default', [$scalar->column]),
75            default => $this->arithmetic($scalar),
76        };
77    }

80     * Writes temporal arithmetic, an interval added to or subtracted from a date, or answers null for another expression.
81     */
82    public function arithmetic(Scalar $scalar): ?string
83    {
84        return match (true) {
85            $scalar instanceof IntervalArithmetic => $this->interval($scalar->operand, $scalar->interval->quantity, $scalar->interval->unit, $scalar->subtract),
86            $scalar instanceof IntervalAddition => $this->interval($scalar->operand, $scalar->interval->quantity, $scalar->interval->unit, false),
87            $scalar instanceof DateArithmetic => $this->interval($scalar->date, $scalar->quantity, $scalar->unit, $scalar->subtract),
88            default => null,
89        };
90    }

102    /**
103     * Writes a call of a function named by an identifier.
104     */
105    public function function(FunctionCall $call): ?string
106    {
107        $arguments = array_map(static fn ($argument): Scalar => $argument->expression, $call->arguments);
108        if ($call->schema !== null) {
109            return $this->call('`' . $call->schema->value . '`.`' . $call->name->value . '`', $arguments);
110        }
111        $name = strtolower($call->name->value);
112        if ($name === 'isnull' && count($arguments) === 1) {
113            return $this->items->wrap($arguments[0], '(', ' is null)');
114        }
115        if (in_array($name, ['adddate', 'subdate'], true) && count($arguments) === 2) {
116            return $this->interval($arguments[0], $arguments[1], IntervalUnit::Day, $name === 'subdate');
117        }
118
119        return $this->call(self::NAMES[$name] ?? $name, $arguments);
120    }

122    /**
123     * Writes a function named by a keyword: DATE(), TIME() and TIMESTAMP() of one argument as casts, MOD as %, and the others in lower case.
124     */
125    public function keyword(KeywordCall $call): ?string
126    {
127        $arguments = $call->arguments;
128        $single = count($arguments) === 1;
129
130        return match (true) {
131            $call->function === KeywordFunction::Date && $single => $this->items->wrap($arguments[0], 'cast(', ' as date)'),
132            $call->function === KeywordFunction::Time && $single => $this->items->wrap($arguments[0], 'cast(', ' as time)'),
133            $call->function === KeywordFunction::Timestamp && $single => $this->items->wrap($arguments[0], 'cast(', ' as datetime)'),
134            $call->function === KeywordFunction::Mod && count($arguments) === 2 => $this->items->binary($arguments[0], '%', $arguments[1]),
135            ($call->function === KeywordFunction::AddDate || $call->function === KeywordFunction::SubDate) && count($arguments) === 2 => $this->interval($arguments[0], $arguments[1], IntervalUnit::Day, $call->function === KeywordFunction::SubDate),
136            default => $this->call(self::NAMES[strtolower($call->function->value)] ?? strtolower($call->function->value), $arguments),
137        };
138    }

140    /**
141     * Writes the type a cast converts to; CHAR names its character set, the connection's when none is written.
142     */
143    public function target(CastTarget $target): string
144    {
145        $length = $target->length === null ? '' : '(' . $target->length . ($target->scale === null ? '' : ',' . $target->scale) . ')';
146        if ($target->kind === CastKind::Char || $target->kind === CastKind::NationalChar) {
147            $charset = $target->charset?->binary() === true ? 'binary' : strtolower($target->charset->charset->value ?? ($target->kind === CastKind::NationalChar ? 'utf8mb3' : $this->items->charset));
148
149            return 'char' . $length . ' charset ' . $charset;
150        }
151        if ($target->kind === CastKind::Binary) {
152            return 'char' . $length . ' charset binary';
153        }
154
155        return strtolower($target->kind->value) . $length;
156    }
