39/**
40 * Executes the statements that control replication: START and STOP REPLICA, CHANGE REPLICATION SOURCE and FILTER, RESET BINARY LOGS AND GTIDS and RESET REPLICA, and START and STOP GROUP_REPLICATION.
41 *
42 * The emulated server is no replica: its only channel is the default one, unconfigured, and a
43 * statement naming another channel fails (ER_REPLICA_CHANNEL_DOES_NOT_EXIST). Each statement but
44 * the group replication ones commits the open transaction. START REPLICA cannot start the
45 * receiver thread (ER_REPLICA_CONFIGURATION), noting first that a PASSWORD is sent in plain text;
46 * SQL_THREAD alone starts the applier thread, which STOP REPLICA stops and RESET REPLICA refuses
47 * to run beside (ER_REPLICA_CHANNEL_MUST_STOP). STOP REPLICA notes threads that are already
48 * stopped. CHANGE REPLICATION SOURCE TO is accepted and kept nowhere; CHANGE REPLICATION FILTER
49 * for a channel fails, the replica not being initialized (ER_REPLICA_CONFIGURATION). RESET
50 * BINARY LOGS AND GTIDS deletes every binary log file and starts again from the number TO names,
51 * 1 by default. Group replication is not configured, and its statements refuse an open
52 * transaction (verified on a live 8.4 server). MySQL 5.6 refuses RESET MASTER when log_bin is
53 * off (ER_FLUSH_MASTER_BINLOG_CLOSED; verified on a live 5.6.51 server).
54 * Source: https://dev.mysql.com/doc/refman/8.4/en/start-replica.html,
55 * https://dev.mysql.com/doc/refman/8.4/en/stop-replica.html,
56 * https://dev.mysql.com/doc/refman/8.4/en/change-replication-filter.html,
57 * https://dev.mysql.com/doc/refman/8.4/en/reset-binary-logs-and-gtids.html,
58 * https://dev.mysql.com/doc/refman/8.4/en/reset-replica.html,
59 * https://dev.mysql.com/doc/refman/8.4/en/start-group-replication.html.
60 *
61 * @visibility MySqlMemory
62 */

75     * Controls the replication of the server, as far as the emulated server has any.

77    #[Override]
78    public function execute(Operation $operation, Session $session, Context $context, Connection $connection): Reply
79    {
80        $statement = $operation->statement;
81        $registry = $session->instance->registry;
82        $this->deprecated($statement, $session, $context);
83        (new LegacyReplication())->check($statement, $session);
84        if ($statement instanceof StartGroupReplication || $statement instanceof StopGroupReplication) {
85            throw $session->transaction->open ? StatementError::LockedOrActiveTransaction->error() : AdministrationError::GroupReplicationNotConfigured->error();
86        }
87        $session->transaction->commit();
88        if ($statement instanceof StartReplica) {
89            $this->channel($statement->channel);
90            if ($statement->password !== null) {
91                $session->diagnostics->note(AccountError::InsecurePlainText, AccountError::InsecurePlainText->message());
92            }
93            if ($statement->threads === [] || in_array(ReplicaThread::Receiver, $statement->threads, true)) {
94                throw AdministrationError::ReplicaNotConfigured->error();
95            }
96            $registry->applying = true;
97        } elseif ($statement instanceof StopReplica) {
98            $this->channel($statement->channel);
99            if ($registry->applying && ($statement->threads === [] || in_array(ReplicaThread::Applier, $statement->threads, true))) {
100                $registry->applying = false;
101            } else {
102                $session->diagnostics->note(AdministrationError::ReplicaThreadsStopped, AdministrationError::ReplicaThreadsStopped->message(''));
103            }
104        } elseif ($statement instanceof ChangeReplicationFilter) {
105            if ($statement->channel !== null) {
106                throw AdministrationError::ReplicaNotInitialized->error();
107            }
108        } elseif ($statement instanceof Reset) {
109            foreach ($statement->targets as $target) {
110                if ($target instanceof ResetBinaryLogs && $session->settings()->release() === GrammarRelease::MySql5651 && !\MySqlMemory\Registry\BinaryLog::enabled($session)) {
111                    throw AdministrationError::BinlogClosed->error('RESET MASTER');
112                }
113                $this->reset($target, $registry);
114            }
115        } elseif ($statement instanceof ChangeReplicationSource) {
116            (new SourceChange())->check($statement, $session, $context);
117        }
118
119        return new Completion(0, 0, $session->diagnostics->count());
120    }

122    /**
123     * Warns that a statement written in the legacy vocabulary of replication is deprecated, naming the statement and then each option spelled with MASTER (ER_WARN_DEPRECATED_SYNTAX).
124     *
125     * MySQL 8.0 deprecated START SLAVE, STOP SLAVE, RESET SLAVE, SHOW SLAVE STATUS, SHOW SLAVE
126     * HOSTS and CHANGE MASTER TO with its MASTER_ options, which MySQL 8.4 removed; MySQL 5.6 and
127     * 5.7 know no other vocabulary and do not warn (verified on live 5.7 and 8.0 servers).
128     * Source: https://dev.mysql.com/doc/refman/8.0/en/replication-statements.html,
129     * https://dev.mysql.com/doc/relnotes/mysql/8.4/en/news-8-4-0.html.
130     */
131    public function deprecated(Node $statement, Session $session, Context $context): void
132    {
133        $release = $session->settings()->release();
134        if ($release === GrammarRelease::MySql5651 || $release === GrammarRelease::MySql5744) {
135            return;
136        }
137        $legacy = static fn (Terminology $terminology): bool => $terminology === Terminology::Legacy;
138        $replaced = match (true) {
139            $statement instanceof StartReplica && $legacy($statement->terminology) => ['START SLAVE', 'START REPLICA'],
140            $statement instanceof StopReplica && $legacy($statement->terminology) => ['STOP SLAVE', 'STOP REPLICA'],
141            $statement instanceof ShowReplicaStatus && $legacy($statement->terminology) => ['SHOW SLAVE STATUS', 'SHOW REPLICA STATUS'],
142            $statement instanceof ShowReplicas && $legacy($statement->terminology) => ['SHOW SLAVE HOSTS', 'SHOW REPLICAS'],
143            $statement instanceof ChangeReplicationSource && $statement->synonym => ['CHANGE MASTER', 'CHANGE REPLICATION SOURCE'],
144            $statement instanceof Reset && array_filter($statement->targets, static fn (ResetTarget $target): bool => $target instanceof ResetReplica && $target->synonym) !== [] => ['RESET SLAVE', 'RESET REPLICA'],
145            default => null,
146        };
147        if ($replaced === null) {
148            return;
149        }
150        $context->warning(StatementError::DeprecatedSyntax, ...$replaced);
151        if ($statement instanceof ChangeReplicationSource) {
152            foreach ($statement->options as $option) {
153                if ($option->synonym) {
154                    $context->warning(StatementError::DeprecatedSyntax, $option->kind->keyword(Terminology::Legacy), $option->kind->value);
155                }
156            }
157        }
158    }

160    /**
161     * Resets the binary log or the replica.
162     *
163     * @throws \MySqlMemory\Error\SqlError When the channel does not exist or its applier thread runs
164     */
165    public function reset(ResetTarget $target, Registry $registry): void
166    {
167        if ($target instanceof ResetBinaryLogs) {
168            $registry->binaryLog->reset($target->first === null ? 1 : (int) (new Literals())->number($target->first));
169        }
170        if ($target instanceof ResetReplica) {
171            $this->channel($target->channel);
172            if ($registry->applying) {
173                throw AdministrationError::ReplicaChannelRunning->error('');
174            }
175        }
176    }

178    /**
179     * Refuses a channel other than the default one, naming it in lower case, and a channel name with a line feed.
180     *
181     * @throws \MySqlMemory\Error\SqlError When the channel is not the default one
182     */
183    public function channel(?Text $channel): void
184    {
185        if ($channel === null) {
186            return;
187        }
188        $name = (new Literals())->bytes($channel);
189        if (str_contains($name, "\n")) {
190            throw DataError::WrongValue->error('argument contains not-allowed LF', $name);
191        }
192        if ($name !== '') {
193            throw AdministrationError::ReplicaChannelMissing->error(strtolower($name));
194        }
195    }
