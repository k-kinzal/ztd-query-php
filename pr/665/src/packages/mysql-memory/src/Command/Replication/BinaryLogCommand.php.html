31/**
32 * Executes PURGE BINARY LOGS and BINLOG.
33 *
34 * PURGE BINARY LOGS TO deletes the binary log files before the one it names, which the index
35 * must hold (ER_UNKNOWN_TARGET_BINLOG). PURGE BINARY LOGS BEFORE converts its moment to a
36 * DATETIME, the integer 0 being the zero datetime, and warns that the active file is not purged;
37 * the emulator keeps the other files. The
38 * BINLOG statement decodes its base64 text (ER_BASE64_DECODE_ERROR) and reads the first event:
39 * fewer than 13 bytes, or an event longer than the text, is a syntax error, and only a format
40 * description or row event is allowed (ER_ONLY_FD_AND_RBR_EVENTS_ALLOWED_IN_BINLOG_STATEMENT,
41 * naming the type of the event). The emulator applies no event, so a format description or
42 * row event, which the server would decode, is refused as one it cannot decode (verified on a
43 * live 8.4 server). None of the statements commits the open transaction. A server whose
44 * log_bin is off purges nothing and warns of no file (verified on live 5.6.51 and 5.7.44 servers).
45 * Source: https://dev.mysql.com/doc/refman/8.4/en/purge-binary-logs.html,
46 * https://dev.mysql.com/doc/refman/8.4/en/binlog.html.
47 *
48 * @visibility MySqlMemory
49 */
50final class BinaryLogCommand implements Command
51{
52    /**
53     * The names the server gives the binary log event types it refuses in a BINLOG statement, by type code.
54     */

56        2 => 'Query', 3 => 'Stop', 4 => 'Rotate', 5 => 'Intvar', 9 => 'Append_block', 11 => 'Delete_file', 13 => 'RAND', 14 => 'User var',
57        16 => 'Xid', 17 => 'Begin_load_query', 18 => 'Execute_load_query', 23 => 'Write_rows_v1', 24 => 'Update_rows_v1', 25 => 'Delete_rows_v1',
58        26 => 'Incident', 27 => 'Heartbeat', 28 => 'Ignorable', 33 => 'Gtid', 34 => 'Anonymous_Gtid', 35 => 'Previous_gtids',
59        36 => 'Transaction_context', 37 => 'View_change', 38 => 'XA_prepare', 40 => 'Transaction_payload', 42 => 'Gtid_tagged_log_event',
60    ];

76    /**
77     * Purges the files, or refuses the events.
78     */
79    #[Override]
80    public function execute(Operation $operation, Session $session, Context $context, Connection $connection): Reply
81    {
82        $statement = $operation->statement;
83        $log = $session->instance->registry->binaryLog;
84        $logging = BinaryLog::enabled($session);
85        if ($statement instanceof PurgeLogsTo && $logging) {
86            $number = $log->find((new Literals())->bytes($statement->file)) ?? throw AdministrationError::UnknownTargetBinlog->error();
87            $log->purge($number);
88        } elseif ($statement instanceof PurgeLogsBefore) {
89            $planner = new Planner($statement, $operation->facts, $session->settings(), $connection, $session->instance->dictionary);
90            $moment = $planner->compiler->compile($statement->moment, new Scope());
91            $value = $moment->evaluate(new Frame($context));
92            if ($value !== null && !(is_int($value) && $value === 0)) {
93                (new Moments())->convert($value, $moment->domain(), new Domain(Kind::DateTime, Field::DateTime, 19, 0, false), $context);
94            }
95            if ($logging) {
96                $session->diagnostics->warning(AdministrationError::ActiveLogNotPurged, AdministrationError::ActiveLogNotPurged->message('./' . $log->active()));
97            }
98        } elseif ($statement instanceof BinlogEvent) {
99            $this->event((new Literals())->bytes($statement->events));
100        }
101
102        return new Completion(0, 0, $session->diagnostics->count());
103    }
104
105    /**
106     * Decodes the events of a BINLOG statement and refuses the first.
107     *
108     * @throws \MySqlMemory\Error\SqlError Always: the text is not base64, the event is malformed, or it is not one the statement allows
109     */
110    public function event(string $text): void
111    {
112        $compact = (string) preg_replace('/\s+/', '', $text);
113        $bytes = strlen($compact) % 4 === 0 ? base64_decode($compact, true) : false;
114        if ($bytes === false) {
115            throw DataError::Base64DecodeFailed->error();
116        }
117        if (strlen($bytes) < 13) {
118            throw StatementError::SyntaxError->error();
119        }
120        $length = unpack('V', substr($bytes, 9, 4));
121        $type = ord($bytes[4]);
122        if (!is_array($length) || $length[1] > strlen($bytes) || in_array($type, self::DECODED, true)) {
123            throw StatementError::SyntaxError->error();
124        }
125        if ($type === 29) {
126            throw AdministrationError::NoFormatDescriptionEvent->error('Rows_query');
127        }
128
129        throw AdministrationError::BinlogEventRefused->error(self::EVENTS[$type] ?? 'Unknown');
130    }
