34/**
35 * Prepares a refused HANDLER read from its complete semantic dependencies.
36 *
37 * Opening the handler precedes WHERE; WHERE precedes index lookup and key
38 * operands. No rejected operand is replaced with a fabricated scalar.

56    /**
57     * Resolves handler state and operand scopes before the intrinsic refusal.
58     *
59     * @throws SqlError Always, after earlier dependencies have been prepared
60     * @throws LogicException When a refused structure has lost its non-scalar operand
61     */
62    public function resolve(RefusedRead $read, Session $session): never
63    {
64        $read = new RefusedRead(
65            \SqlSemantics\Platform\MySql\Construction\Occurrences::fresh($read->handler),
66            $read->mode,
67            $read->index,
68            array_map(static fn ($value) => \SqlSemantics\Platform\MySql\Construction\Occurrences::fresh($value), $read->values),
69            $read->where === null ? null : \SqlSemantics\Platform\MySql\Construction\Occurrences::fresh($read->where),
70            $read->limit === null ? null : \SqlSemantics\Platform\MySql\Construction\Occurrences::fresh($read->limit),
71        );
72        $database = $session->variables->database;
73        $context = $session->semantics()->context(\MySqlMemory\Session\Statement\Catalog::declarations($session), true, $database === '' ? null : new SearchPath($database), $session->resolution());
74        $derivation = new Derivation($context);
75        (new HandlerFacts())->read($read->handler, $read->values, $read->where, $read->limit, $derivation);
76        $facts = (new \SqlSemantics\Platform\MySql\Construction\StatementResults())->of($derivation);
77        $this->early($read, $facts, $session);
78        if ($session->preparation->inputPrepared) {
79            throw \MySqlMemory\Error\Family\StatementError::UnsupportedPreparedStatement->error();
80        }
81        $command = new HandlerCommand();
82        $handler = $command->handler($session, $read->handler->name->value);
83        $table = $session->instance->dictionary->table($handler->schema, $handler->table, $session->temporaries);
84        assert($table !== null);
85        $scope = (new \MySqlMemory\Command\Access\HandlerKeys())->scope($read->handler, $table);
86        if ($read->where !== null) {
87            $this->prepare($read->where, $facts, $scope, $session);
88        }
89        if ($read->index !== null) {
90            $key = $command->key($table, $read->index->value, $handler);
91            if (count($read->values) > count($key->columns)) {
92                throw \MySqlMemory\Error\Family\SchemaError::TooManyKeyParts->error(count($key->columns));
93            }
94        }
95        foreach ($read->values as $value) {
96            if (!$value instanceof DefaultRequest) {
97                $this->prepare($value, $facts, $scope, $session);
98            }
99        }
100        throw new LogicException('A refused handler read must retain a non-scalar operand.');
101    }

103    /**
104     * Preserves parsing conditions, stored-call database lookup and LIMIT order.
105     *
106     * @throws SqlError When an input condition precedes opening the handler
107     */
108    public function early(RefusedRead $read, StatementSemantics $facts, Session $session): void
109    {
110        $operands = [...$read->values, ...($read->where === null ? [] : [$read->where])];
111        foreach ($operands as $operand) {
112            (new \MySqlMemory\Session\Problem\Identifiers())->check($operand, $session->settings()->release());
113        }
114        foreach ($facts->diagnostics as $diagnostic) {
115            if (Stages::parsed($diagnostic)) {
116                throw (new Errors())->error($diagnostic, $session);
117            }
118        }
119        if ($session->variables->database === '') {
120            foreach ($operands as $operand) {
121                foreach ([...(new Walker())->find($operand, FunctionCall::class), ...(new Walker())->find($operand, DeferredCall::class)] as $call) {
122                    if ($call->schema === null && !$call->native($session->settings()->release())) {
123                        throw (new Errors())->routine($call, $session);
124                    }
125                }
126            }
127        }
128        (new Reading())->limited($read->limit, $session);
129        foreach ($facts->diagnostics as $diagnostic) {
130            if ($diagnostic instanceof NotSupportedYet && $diagnostic->feature === 'stored functions in HANDLER ... READ') {
131                throw (new Errors())->error($diagnostic, $session);
132            }
133        }
134    }

136    /**
137     * Prepares a predicate or key without evaluating or reinterpreting its names.
138     *
139     * @throws SqlError When a dependency or the operand's scalar role fails
140     */
141    public function prepare(Scalar|RowOperand|DeferredOperand $operand, StatementSemantics $facts, Scope $scope, Session $session): void
142    {
143        $evaluation = new Context($session->modes(), $session->diagnostics, $session->variables, $session->variables->instant());
144        $connection = new Connection($session->variables, $evaluation, $session->user, $session->host, $session->id, program: $session->program, session: WeakReference::create($session));
145        $planner = new Planner($operand, $facts, $session->settings(), $connection, $session->instance->dictionary, $session->temporaries, \MySqlMemory\Session\Statement\Catalog::snapshot($session));
146        if ($operand instanceof Scalar) {
147            $planner->compiler->compile($operand, $scope);
148            return;
149        }
150        (new RowUse())->prepare($operand, $planner->compiler, $scope);
151        throw (new RowUse())->error($operand);
152    }
