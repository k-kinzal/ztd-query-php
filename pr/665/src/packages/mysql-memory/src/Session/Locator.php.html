36/**
37 * Finds where in a statement each column name, select list position, function call and clock call is read, and in which order the server resolves it.
38 *
39 * The server resolves a query block in this order: the index hints of its tables, the derived tables of its FROM clause, the
40 * qualifiers of its `t.*` items, its select list, WHERE, the ON conditions, GROUP BY, HAVING and ORDER BY; it names the clause it
41 * resolves in the message of a name it cannot resolve. A subquery is resolved in the place it is
42 * written. The order of an UPDATE, an INSERT and a DELETE is told by WriteOrder.
43 * Source: https://dev.mysql.com/doc/refman/8.4/en/select.html.
44 *

161    /**
162     * Tells whether one resolution order comes before another: element by element, a prefix first.
163     *
164     * @param list<int> $left
165     * @param list<int> $right
166     */
167    public static function precedes(array $left, array $right): bool
168    {
169        foreach ($left as $index => $value) {
170            if (!isset($right[$index])) {
171                return false;
172            }
173            if ($value !== $right[$index]) {
174                return $value < $right[$index];
175            }
176        }
177
178        return count($left) < count($right);
179    }

211    /**
212     * Locates the column names under a value, read in a clause at an order.
213     *
214     * The nodes under the value are visited depth first in the order of their properties; each
215     * property, and each member of a list, adds its position to the order.
216     *
217     * @param Node|list<Node>|null $value
218     * @param list<int> $order
219     */
220    public function visit(Node|array|null $value, string $clause, array $order): void
221    {
222        $values = [$value];
223        $orders = [$order];
224        while (($at = array_pop($orders)) !== null) {
225            $current = array_pop($values);
226            $children = [];
227            $positions = [];
228            if (is_array($current)) {
229                foreach (array_values($current) as $index => $item) {
230                    $children[] = $item;
231                    $positions[] = [...$at, $index];
232                }
233            } elseif ($current instanceof Select) {
234                $this->select($current, $at);
235            } elseif ($current instanceof QueryExpression) {
236                $this->expression($current, $clause, $at);
237            } elseif ($current instanceof InList) {
238                $children = [$current->operand, $current->elements];
239                $positions = [[...$at, 0], [...$at, 1]];
240            } elseif ($current instanceof InQuery || $current instanceof QuantifiedComparison) {
241                $this->predicates[] = [$current, $clause, $at];
242                $early = self::early($current, $this->release);
243                $this->visit($current->operand, 'IN/ALL/ANY subquery', [...$at, $this->operandFirst ? 0 : ($early ? 2 : 1)]);
244                $children[] = $current->query;
245                $positions[] = [...$at, $this->operandFirst ? 1 : 0];
246            } elseif ($current instanceof WindowSpec) {
247                $this->record($current, $clause, $at);
248                $this->specifications[] = $current;
249            } elseif ($current instanceof Node) {
250                $this->record($current, $clause, $at);
251                $index = 0;
252                foreach (get_object_vars($current) as $property) {
253                    $children[] = $property;
254                    $positions[] = [...$at, $index++];
255                }
256            }
257            array_push($values, ...array_reverse($children));
258            array_push($orders, ...array_reverse($positions));
259        }
260    }

262    /**
263     * Records what a node read in a clause at an order tells, before its children are visited.
264     *
265     * A functional key part marks its cast, ungrouped, as keyed; a cast to an array of a type a
266     * multi-valued index takes, outside a key part, is recorded with its place; a column name,
267     * select list position or function call is located, and so is a clock call; a window function
268     * or aggregate call, and the window names, are recorded as windowed() records them.
269     *
270     * @param list<int> $order
271     */
272    public function record(Node $node, string $clause, array $order): void
273    {
274        if ($node instanceof ExpressionPart) {
275            $keyed = $node->expression;
276            while ($keyed instanceof Grouped) {
277                $keyed = $keyed->operand;
278            }
279            $this->keyed[spl_object_id($keyed)] = true;
280        }
281        if ($node instanceof Cast && $node->array && $node->arrayRefusal() === null && !isset($this->keyed[spl_object_id($node)])) {
282            $this->arrays[] = [$node, $clause, $order];
283        }
284        if ($node instanceof ColumnUse || $node instanceof OutputOrdinal || $node instanceof FunctionCall) {
285            $this->places[spl_object_id($node)] = [$clause, $order];
286            $this->nodes[] = $node;
287        }
288        if ($node instanceof ClockCall) {
289            $this->places[spl_object_id($node)] = [$clause, $order];
290            $this->clocks[] = $node;
291        }
292        if ($node instanceof SetOperation || $node instanceof OrderedSetOperation) {
293            $this->sets[] = [$node, [...$order, (int) array_search('right', array_keys(get_object_vars($node)), true), PHP_INT_MAX]];
294        }
295        $this->windowed($node, $clause, $order);
296    }

299     * Records a window function or aggregate call read in a clause at an order, the window name it uses after OVER, checked where the call is resolved, and the window a specification refines, checked after the ORDER BY of its block.
300     *
301     * @param list<int> $order
302     */
303    public function windowed(Node $node, string $clause, array $order): void
304    {
305        if ($node instanceof WindowFunction || $node instanceof Aggregate || $node instanceof GroupConcat || $node instanceof JsonObjectAggregate) {
306            $this->functions[] = [$node, $this->process && $this->block === [] ? 'process identifier' : $clause, $order];
307        }
308        $over = $node instanceof WindowFunction || $node instanceof Aggregate || $node instanceof GroupConcat || $node instanceof JsonObjectAggregate ? $node->over : null;
309        if ($over instanceof Name) {
310            $this->windows[] = [$over, $order];
311        }
312        if ($node instanceof WindowSpec && $node->base !== null) {
313            $this->windows[] = [$node->base, [...$this->block, 6, PHP_INT_MAX]];
314        }

317    /**
318     * Tells whether the server checks the width of the subquery of IN, ANY or ALL before it resolves the operand.
319     *
320     * The subquery is resolved first. The width is checked before the operand for ALL, and for
321     * ANY with an operator other than `=`; after it for IN, `= ANY` and `<> ALL`. MySQL 5.6 and
322     * 5.7 resolve the subquery first but check its width after resolving the operand, verified
323     * against live 5.6.51 and 5.7.44 servers.
324     */
325    public static function early(InQuery|QuantifiedComparison $predicate, \SqlSemantics\Contract\GrammarRelease $release = \SqlSemantics\Contract\GrammarRelease::MySql847): bool
326    {
327        return !in_array($release, [\SqlSemantics\Contract\GrammarRelease::MySql5651, \SqlSemantics\Contract\GrammarRelease::MySql5744], true)
328            && $predicate instanceof QuantifiedComparison
329            && !($predicate->quantifier === Quantifier::Any && $predicate->operator === ComparisonOperator::Equal)
330            && !($predicate->quantifier === Quantifier::All && $predicate->operator === ComparisonOperator::NotEqual);
331    }

346    /**
347     * Locates the column names of a query block in the order the server resolves its clauses.
348     *
349     * @param list<int> $order
350     */
351    public function select(Select $select, array $order): void
352    {
353        $outer = $this->block;
354        $outerSpecifications = $this->specifications;
355        $this->specifications = [];
356        $this->block = $order;
357        foreach ($select->from === null ? [] : (new \MySqlMemory\Evaluation\Compile\Walker())->find($select->from, \SqlSemantics\Platform\MySql\Statement\Relation\TableReference::class, false) as $reference) {
358            $this->hinted[] = [$reference, $this->release === \SqlSemantics\Contract\GrammarRelease::MySql5651 ? [...$order, 0, PHP_INT_MAX] : [...$order, -1]];
359        }
360        $clause = new FromClause($this);
361        $clause->derived($select->from, $order);
362        foreach ($select->items as $item) {
363            if ($item instanceof TableWildcard) {
364                $this->wildcards[] = [$item, [...$order, 1, -1]];
365            }
366            if ($item instanceof \SqlSemantics\Platform\MySql\Statement\Query\Star && ($select->from === null || $select->from instanceof \SqlSemantics\Platform\MySql\Statement\Relation\Dual)) {
367                $this->stars[] = [$select, [...$order, 1, -2]];
368            }
369        }
370        foreach ($select->items as $index => $item) {
371            $this->visit($item instanceof SelectExpression ? $item->expression : $item, 'field list', [...$order, 1, $index]);
372        }
373        $this->visit($select->where, 'where clause', [...$order, 2]);
374        $clause->conditions($select->from, [...$order, 3]);
375        $this->visit($select->groupBy, 'group statement', [...$order, 4]);
376        $this->visit($select->having, 'having clause', [...$order, 5]);
377        $this->visit($select->orderBy, 'order clause', [...$order, 6]);
378        $this->visit($select->limit, 'field list', [...$order, 7]);
379        $this->specifications($select, $order);
380        $this->specifications = $outerSpecifications;
381        $pending = $select->windows;
382        while (($node = array_pop($pending)) !== null) {
383            if ($node instanceof WindowSpec && $node->base !== null) {
384                $this->windows[] = [$node->base, [...$order, 6, PHP_INT_MAX]];
385            }
386            foreach (get_object_vars($node) as $value) {
387                foreach (is_array($value) ? $value : [$value] as $member) {
388                    if ($member instanceof Node && !$member instanceof Query) {
389                        $pending[] = $member;
390                    }
391                }
392            }
393        }
394        $this->block = $outer;
395    }
