16/**
17 * How a transaction reaches the rows of InnoDB tables: what its reads see, and the row locks its reads and writes take.
18 *
19 * A consistent read of REPEATABLE READ sees the snapshot the first consistent read of the
20 * transaction took, or START TRANSACTION WITH CONSISTENT SNAPSHOT; READ COMMITTED takes a snapshot
21 * for each statement; READ UNCOMMITTED reads the latest rows. Locking reads, and plain reads of
22 * SERIALIZABLE inside a transaction, read the latest committed rows and lock the rows they return.
23 * A transaction that wants a lock another transaction holds waits for it, until the lock is free,
24 * the wait times out, or the wait closes a cycle of waits, a deadlock.
25 * Source: https://dev.mysql.com/doc/refman/8.4/en/innodb-transaction-isolation-levels.html,
26 * https://dev.mysql.com/doc/refman/8.4/en/innodb-consistent-read.html,
27 * https://dev.mysql.com/doc/refman/8.4/en/innodb-locking-reads.html,
28 * https://dev.mysql.com/doc/refman/8.4/en/innodb-locks-set.html,
29 * https://dev.mysql.com/doc/refman/8.4/en/innodb-parameters.html#sysvar_innodb_lock_wait_timeout.
30 *
31 * @visibility MySqlMemory
32 */

47    /**
48     * Answers the rows of a table a read of the statement sees, by row number in row number order.
49     *
50     * A consistent read sees its snapshot, or the latest rows under READ UNCOMMITTED; a locking
51     * read goes through the latest rows. A read of an InnoDB table makes the transaction active.
52     * The rows of a temporary table and of a table of another engine are read as they are.
53     *
54     * @param LockMode|null $locking The lock a locking read takes, or null for a consistent read
55     * @return array<int, list<int|float|string|null>>
56     */
57    public function rows(StoredTable $table, ?LockMode $locking = null): array
58    {
59        $transaction = $this->transaction;
60        if (!Transaction::transactional($table)) {
61            return $table->data->rows;
62        }
63        $transaction->engaged = true;
64        if ($table->definition->temporary) {
65            return $table->data->rows;
66        }
67        if ($locking !== null) {
68            return $transaction->system->latest($table, $transaction);
69        }
70        if ($transaction->isolation === Isolation::ReadUncommitted) {
71            return $table->data->rows;
72        }
73        $transaction->snapshot ??= $transaction->system->open($transaction->id);
74
75        return $transaction->system->visible($table, $transaction, $transaction->snapshot);
76    }

96    /**
97     * Locks a row in a mode, waiting for the transactions that hold it, and answers whether it holds the lock; SKIP LOCKED answers false for a row another transaction holds.
98     *
99     * A wait ends when the lock is free, or after innodb_lock_wait_timeout seconds with
100     * ER_LOCK_WAIT_TIMEOUT, which rolls back the statement. NOWAIT refuses the row at once with
101     * ER_LOCK_NOWAIT.
102     *
103     * @throws SqlError When the lock cannot be taken
104     */
105    public function lock(StoredTable $table, int $number, LockMode $mode, ?LockedRowAction $action = null): bool
106    {
107        if (!Transaction::transactional($table) || $table->definition->temporary) {
108            return true;
109        }
110        $locks = $this->transaction->system->locks;
111        $deadline = null;
112        for (;;) {
113            $blockers = $locks->blockers($table, $number, $this->transaction->id, $mode);
114            if ($blockers === []) {
115                $locks->grant($table, $number, $this->transaction->id, $mode);
116                $this->transaction->engaged = true;
117
118                return true;
119            }
120            if ($action === LockedRowAction::Nowait) {
121                throw TransactionError::LockNowait->error();
122            }
123            if ($action === LockedRowAction::SkipLocked) {
124                return false;
125            }
126            $deadline = $this->wait($blockers, $deadline);
127        }
128    }

130    /**
131     * Waits once for the transactions that hold a lock the transaction wants, and answers when the whole wait times out.
132     *
133     * A wait that would close a cycle of waits is a deadlock: the lighter transaction of the cycle
134     * is rolled back with ER_LOCK_DEADLOCK. Only a statement the listener runs can wait: elsewhere
135     * nothing else runs while it would, so the wait times out at once, and the time it would have
136     * taken passes on the clock of the server.
137     *
138     * @param list<int> $blockers The transactions that hold the lock
139     * @param float|null $deadline When the wait times out, or null when it starts now
140     *
141     * @throws SqlError When the transaction is chosen as the victim of a deadlock, or the wait times out
142     */
143    public function wait(array $blockers, ?float $deadline): float
144    {
145        $transaction = $this->transaction;
146        $locks = $transaction->system->locks;
147        $cycle = $locks->cycle($transaction->id, $blockers);
148        if ($cycle !== []) {
149            $victim = $transaction->system->victim($cycle);
150            if ($victim === $transaction->id) {
151                $this->deadlock();
152            }
153            $locks->victims[$victim] = true;
154        }
155        $timeout = max(1, (int) ($transaction->variables?->read('innodb_lock_wait_timeout') ?? 50));
156        if (!$transaction->system->scheduler->suspendable()) {
157            $transaction->variables?->instance->registry->threads->pass($timeout);
158            throw TransactionError::LockWaitTimeout->error();
159        }
160        $deadline ??= microtime(true) + $timeout;
161        $locks->waits[$transaction->id] = $blockers;
162        try {
163            $transaction->system->scheduler->pause();
164        } finally {
165            unset($locks->waits[$transaction->id]);
166            if ($this->resumed !== null) {
167                ($this->resumed)();
168            }
169        }
170        if (($transaction->variables?->instance->sessions[$transaction->id] ?? null)?->get()?->interrupted === true) {
171            throw \MySqlMemory\Error\Family\StatementError::QueryInterrupted->error();
172        }
173        if (isset($locks->victims[$transaction->id])) {
174            unset($locks->victims[$transaction->id]);
175            $this->deadlock();
176        }
177        if (microtime(true) >= $deadline) {
178            throw TransactionError::LockWaitTimeout->error();
179        }
180
181        return $deadline;
182    }
