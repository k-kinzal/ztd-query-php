44/**
45 * Finds the problems the server reports where it resolves a name, and the clause and resolution order each is found at.
46 *
47 * A name or a select list position that does not resolve, a call of a function the server does
48 * not find, a clock call whose precision is above 6, a cast to an array outside a functional
49 * index, DEFAULT() of a column without a default, and IN, ANY or ALL over a subquery of the wrong
50 * width are each found where the server resolves them (Locator); the first of them is the one the
51 * server reports (verified on a live 8.4 server).
52 * Source: https://dev.mysql.com/doc/refman/8.4/en/select.html.
53 *
54 * @visibility MySqlMemory
55 */
56final class Locations

58    /**
59     * Answers the problems found where names are resolved, by object id, each with its clause and resolution order; and the problems of a column of MATCH, by object id.
60     *
61     * @param list<FunctionCall> $calls The calls of functions the server does not find
62     * @param list<Diagnostic> $diagnostics The problems of the statement the server reports
63     * @return array{array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}>, array<int, true>}
64     */
65    public function located(Operation $operation, array $calls, array $diagnostics, Session $session): array
66    {
67        $locator = (new Locator($session->settings()->release() === GrammarRelease::MySql8044, $session->settings()->release()))->statement($operation->statement);
68        [$located, $matched] = $this->names($operation, $locator, $calls, $diagnostics);
69        foreach ($locator->arrays as [$cast, $clause, $at]) {
70            $refusal = new NotSupportedYet(Cast::ARRAY_OUTSIDE_INDEX);
71            $located[spl_object_id($refusal)] = [$refusal, [$clause, $at]];
72        }
73        $located = $this->defaults($operation, $locator, $session, $located);
74        $located = $this->wildcards($locator, $diagnostics, $located);
75        $located = $this->stars($operation, $locator, $diagnostics, $located);
76        $located = $this->windows($locator, $diagnostics, $located);
77        $located = $this->misplaced($locator, $located);
78        $located = $this->sets($operation, $locator, $diagnostics, $located);
79        $common = (new IndexHints())->common($operation->statement);
80        foreach ($locator->hinted as [$reference, $order]) {
81            $refusal = (new IndexHints())->refusal($reference, $common, $session);
82            if ($refusal !== null) {
83                $located[spl_object_id($refusal)] = [$refusal, ['field list', $order]];
84            }
85        }
86
87        return [$this->widths($operation, $locator, $diagnostics, $located), $matched];
88    }

90    /**
91     * Answers the problems of the located names, positions, calls and clock calls, by object id, each with its place; and those of a column of MATCH, by object id.
92     *
93     * A diagnostic the statement does not report, and a call the server finds, is left out.
94     *
95     * @param list<FunctionCall> $calls The calls of functions the server does not find
96     * @param list<Diagnostic> $diagnostics The problems of the statement the server reports
97     * @return array{array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}>, array<int, true>}
98     */
99    public function names(Operation $operation, Locator $locator, array $calls, array $diagnostics): array
100    {
101        $dropped = array_diff(array_map(spl_object_id(...), $operation->facts->diagnostics), array_map(spl_object_id(...), $diagnostics));
102        $searched = [];
103        foreach ((new Walker())->find($operation->statement, FullTextSearch::class) as $search) {
104            foreach ($search->columns as $column) {
105                $searched[spl_object_id($column)] = true;
106            }
107        }
108        $located = [];
109        $matched = [];
110        foreach ([...$locator->nodes(), ...$locator->clocks()] as $node) {
111            $problem = $operation->facts->covers($node) ? $this->problem($node, $operation) : null;
112            $problem = $problem instanceof FunctionCall && !in_array($problem, $calls, true) ? null : $problem;
113            $problem = $problem instanceof Diagnostic && !in_array($problem, $operation->facts->diagnostics, true) ? null : $problem;
114            $place = $locator->place($node);
115            if ($problem !== null && $place !== null && !in_array(spl_object_id($problem), $dropped, true)) {
116                $located[spl_object_id($problem)] = [$problem, $place];
117                if (isset($searched[spl_object_id($node)])) {
118                    $matched[spl_object_id($problem)] = true;
119                }
120            }
121        }
122
123        return [$located, $matched];
124    }

126    /**
127     * Adds the refusals of DEFAULT() of a column without a default in a query, each placed right after the column it reads.
128     *
129     * @param array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}> $located
130     * @return array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}>
131     */
132    public function defaults(Operation $operation, Locator $locator, Session $session, array $located): array
133    {
134        foreach ($operation->statement instanceof Query ? (new Walker())->find($operation->statement, DefaultOfColumn::class) : [] as $default) {
135            $place = $locator->place($default->column);
136            $name = $place === null ? null : $this->undefaulted($default, $operation, $session);
137            if ($place !== null && $name !== null) {
138                $refusal = DataError::NoDefaultForField->error($name);
139                $located[spl_object_id($refusal)] = [$refusal, [$place[0], [...$place[1], 0]]];
140            }
141        }
142
143        return $located;
144    }

169    /**
170     * Adds the refusal of `*` in a block without tables, placed before the items of the block; EXISTS takes such a block.
171     *
172     * The server expands the star while it resolves the select list of the block, so the
173     * subquery of IN, ANY and ALL, resolved before its operand, reports it first; the query of
174     * EXISTS selects nothing and takes it (verified on live 5.6.51 and 8.4.7 servers).
175     *
176     * @param list<Diagnostic> $diagnostics The problems of the statement the server reports
177     * @param array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}> $located
178     * @return array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}>
179     */
180    public function stars(Operation $operation, Locator $locator, array $diagnostics, array $located): array
181    {
182        $misuses = array_values(array_filter($diagnostics, static fn (Diagnostic $diagnostic): bool => $diagnostic instanceof Misuse && $diagnostic->rule === MisuseRule::StarWithoutTables));
183        if ($misuses === []) {
184            return $located;
185        }
186        $existing = self::existing($operation);
187        foreach ($locator->stars as [$select, $at]) {
188            if (!in_array($select, $existing, true)) {
189                $refusal = array_shift($misuses) ?? QueryError::NoTablesUsed->error();
190                $located[spl_object_id($refusal)] = [$refusal, ['field list', $at]];
191            }
192        }
193
194        return $located;
195    }

237    /**
238     * Adds the refusals of window functions outside the select list and ORDER BY of their block, or in an argument of another window function or of an aggregate, each placed right after its arguments.
239     *
240     * The server refuses such a call where it resolves it, with ER_WINDOW_INVALID_WINDOW_FUNC_USE
241     * (verified on a live 8.4 server).
242     *
243     * @param array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}> $located
244     * @return array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}>
245     */
246    public function misplaced(Locator $locator, array $located): array
247    {
248        $walker = new Walker();
249        $nested = [];
250        foreach ($locator->functions as [$call]) {
251            $window = $call->over instanceof WindowSpec ? array_map(spl_object_id(...), $walker->find($call->over, Scalar::class, false)) : [];
252            foreach ($walker->find($call, Scalar::class, false) as $node) {
253                if ($node !== $call && Windowing::windowed($node) && !in_array(spl_object_id($node), $window, true)) {
254                    $nested[spl_object_id($node)] = true;
255                }
256            }
257        }
258        foreach ($locator->functions as [$call, $clause, $at]) {
259            if (Windowing::windowed($call) && (isset($nested[spl_object_id($call)]) || !in_array($clause, ['field list', 'order clause', 'window partition by', 'window order by'], true))) {
260                $refusal = QueryError::WindowFunctionMisplaced->error(Resolution::named($call));
261                $located[spl_object_id($refusal)] = [$refusal, [$clause, [...$at, PHP_INT_MAX]]];
262            }
263        }
264
265        return $located;
266    }

268    /**
269     * Adds the problems of set operations whose operands have different numbers of columns, each placed once its right operand is resolved; a problem of the statement answers one operation only.
270     *
271     * The server compares the columns of the operands of each set operation after it has
272     * resolved them, before it resolves the operand that follows or the ORDER BY of the
273     * operation (verified on a live 8.4 server).
274     *
275     * @param list<Diagnostic> $diagnostics The problems of the statement the server reports
276     * @param array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}> $located
277     * @return array<int, array{Diagnostic|FunctionCall|ClockCall|SqlError, array{string, list<int>}}>
278     */
279    public function sets(Operation $operation, Locator $locator, array $diagnostics, array $located): array
280    {
281        $claimed = [];
282        foreach ($locator->sets as [$set, $at]) {
283            $left = $set->left instanceof Query && $operation->facts->covers($set->left) ? $operation->facts->query($set->left)->shape : null;
284            $right = $operation->facts->covers($set->right) ? $operation->facts->query($set->right)->shape : null;
285            if ($left === null || $right === null || !$left->complete() || !$right->complete()) {
286                continue;
287            }
288            foreach ($diagnostics as $diagnostic) {
289                if ($diagnostic instanceof CountMismatch && $diagnostic->list === CountedList::SetOperands && $diagnostic->expected === count($left->slots) && $diagnostic->actual === count($right->slots) && !in_array($diagnostic, $claimed, true)) {
290                    $claimed[] = $diagnostic;
291                    $located[spl_object_id($diagnostic)] = [$diagnostic, ['field list', $at]];
292                    break;
293                }
294            }
295        }
296
297        return $located;
298    }

340     * Tells whether a call names a function that is neither native nor declared: a stored function the server does not find.
341     */
342    public static function undeclared(FunctionCall $call, Operation $operation): bool
343    {
344        if (!$operation->facts->covers($call)) {
345            return false;
346        }
347        $type = $operation->facts->scalar($call)->type;
348        if (!$type instanceof Dependent) {
349            return false;
350        }
351        foreach ($type->missing as $missing) {
352            if ($missing instanceof UndeclaredRoutine && $missing->name->name === $call->name) {
353                return true;
354            }
355        }
356
357        return false;
358    }

361     * Answers the problem of a located node: the diagnostic a name or an ordinal resolves to, a call of a function the server does not find, or a clock call whose precision is above 6.
362     */
363    public function problem(ColumnUse|OutputOrdinal|FunctionCall|ClockCall $node, Operation $operation): Diagnostic|FunctionCall|ClockCall|null
364    {
365        if ($node instanceof FunctionCall) {
366            return self::undeclared($node, $operation) ? $node : null;
367        }
368        if ($node instanceof ClockCall) {
369            return $node->decimals() > 6 ? $node : null;
370        }
371        $fact = $operation->facts->scalar($node);
372        if ($fact->resolution instanceof Diagnostic) {
373            return $fact->resolution;
374        }
375
376        return $node instanceof OutputOrdinal && $fact->type instanceof Invalid ? $fact->type->cause : null;

379    /**
380     * Answers the name of the column DEFAULT() reads when the column is of a stored table and has no default, else null.
381     *
382     * The server refuses such a DEFAULT() where it resolves it (ER_NO_DEFAULT_FOR_FIELD), whether
383     * any row is read or not; an AUTO_INCREMENT column reads as 0 (verified on a live 8.4 server).
384     */
385    public function undefaulted(DefaultOfColumn $default, Operation $operation, Session $session): ?string
386    {
387        $resolution = $operation->facts->covers($default->column) ? $operation->facts->scalar($default->column)->resolution : null;
388        $declaration = $resolution instanceof ResolvedColumn ? $resolution->slot->declaration() : null;
389        if ($declaration === null) {
390            return null;
391        }
392        foreach ($session->instance->dictionary->schemas as $schema) {
393            foreach ($schema->tables as $table) {
394                foreach ($table->definition->columns as $column) {
395                    if ($column->declaration === $declaration) {
396                        return $column->default->declared || $column->autoIncrement ? null : $column->name;
397                    }
398                }
399            }
400        }
401
402        return null;
403    }

405    /**
406     * Answers the problem of IN, ANY or ALL over a subquery whose width is not the width of its single-valued operand, and whether the server finds it before it resolves the operand.
407     *
408     * The server checks the width after it resolves the subquery: before it resolves the operand
409     * of ALL, and of ANY with an operator other than `=` (Locator::early()), and after it for IN,
410     * `= ANY` and `<> ALL` (verified on a live 8.4 server). The problem is the one the statement
411     * reports, or a new one when the operand did not resolve.
412     *
413     * @param list<Diagnostic> $diagnostics The problems of the statement
414     * @param list<Diagnostic> $claimed The problems already answered for another predicate
415     * @return array{OperandColumns, bool}|null
416     */
417    public function width(InQuery|QuantifiedComparison $predicate, Operation $operation, array $diagnostics, array $claimed): ?array
418    {
419        $operand = $predicate->operand;
420        while ($operand instanceof Grouped) {
421            $operand = $operand->operand;
422        }
423        if ($operand instanceof Row || !$operation->facts->covers($predicate->query)) {
424            return null;
425        }
426        $shape = $operation->facts->query($predicate->query)->shape;
427        $width = count($shape->slots);
428        if (!$shape->complete() || $width < 2) {
429            return null;
430        }
431        $early = Locator::early($predicate);
432        foreach ($diagnostics as $diagnostic) {
433            if ($diagnostic instanceof OperandColumns && $diagnostic->expected === 1 && $diagnostic->actual === $width && !in_array($diagnostic, $claimed, true)) {
434                return [$diagnostic, $early];
435            }
436        }
437
438        return $early ? [new OperandColumns(1, $width), true] : null;
439    }
