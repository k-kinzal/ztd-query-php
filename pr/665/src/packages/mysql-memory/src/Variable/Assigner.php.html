23/**
24 * Checks and assigns a value to a system variable, as SET does.
25 *
26 * A read-only variable is refused (ER_INCORRECT_GLOBAL_LOCAL_VAR); a global-only one set for
27 * the session and a session-only one set globally are refused. A boolean takes ON, OFF, 1 and
28 * 0; an integer is clipped to its bounds with a warning; NULL is taken by character_set_results,
29 * session_track_system_variables, innodb_tmpdir and innodb_ft_user_stopword_table only, and
30 * refused by any other variable; DEFAULT restores the global value, or the compiled default for
31 * a global assignment. block_encryption_mode takes the name of an AES mode, in any case, or its
32 * number, and holds the name in lower case. A variable MySQL 5.6 or 5.7 deprecates warns as its
33 * assignment is checked (see DeprecatedVariables).
34 * Source: https://dev.mysql.com/doc/refman/8.4/en/set-variable.html.
35 *
36 * @visibility MySqlMemory
37 */

48    /**
49     * Assigns a value, or DEFAULT when the value is null and the domain is null.
50     *
51     * @param string|null $cache The key cache whose parameter is assigned, or null for the variable
52     * @throws \MySqlMemory\Error\SqlError When the variable or the value is refused
53     */
54    public function assign(string $name, Scope $scope, int|float|string|null $value, ?Domain $domain, ?string $cache = null): void
55    {
56        $definition = $this->variables->catalog->find($name);
57        if ($definition === null) {
58            throw AdministrationError::UnknownSystemVariable->error($name);
59        }
60        if ($definition->writability === Writability::ReadOnly) {
61            throw AdministrationError::IncorrectGlobalLocalVariable->error($definition->name, 'read only');
62        }
63        if ($scope === Scope::Session && !$definition->reach->session()) {
64            throw AdministrationError::GlobalVariable->error($definition->name);
65        }
66        if ($scope === Scope::Global && !$definition->reach->global()) {
67            throw AdministrationError::LocalVariable->error($definition->name);
68        }
69        if ($scope === Scope::Session && $definition->writability === Writability::GlobalOnly) {
70            throw AdministrationError::VariableIsReadonly->error('SESSION', $definition->name, 'GLOBAL');
71        }
72        $checked = $domain === null ? ($scope === Scope::Global ? $definition->default : $this->variables->globals->value($definition)) : $this->check($definition, $value, $domain);
73        if ($domain === null && str_starts_with($definition->name, 'innodb_monitor_') && in_array($this->context->modes->release, [\SqlSemantics\Contract\GrammarRelease::MySql5651, \SqlSemantics\Contract\GrammarRelease::MySql5744], true)) {
74            $this->context->diagnostics->warning(1230, 'Default value is not defined for this set option. Please specify correct counter or module name.');
75        }
76        if ($cache !== null) {
77            $this->cache($cache, $definition, $checked);
78
79            return;
80        }
81        if ($definition->name === 'timestamp' && ($domain === null || $checked === null)) {
82            unset($this->variables->session['timestamp']);
83
84            return;
85        }
86        foreach ($this->aliases($definition) as $alias) {
87            if ($scope === Scope::Global) {
88                $this->variables->globals->set($alias, $checked);
89            } else {
90                $this->variables->set($alias, $checked);
91            }
92        }
93    }

135    /**
136     * Checks a value for a variable and answers the value it holds.
137     *
138     * @throws \MySqlMemory\Error\SqlError When the value is refused
139     */
140    public function check(Definition $definition, int|float|string|null $value, Domain $domain): string|int|null
141    {
142        $text = $value === null ? 'NULL' : (string) Convert::toText($value, $domain);
143        $clock = new TimeSettings($this->context);
144        if ($definition->name === 'time_zone') {
145            return $clock->zone($value, $domain);
146        }
147        if ($definition->name === 'lc_time_names') {
148            return $clock->locale($value, $domain);
149        }
150        if ($definition->name === 'timestamp') {
151            return $clock->timestamp($value, $domain);
152        }
153        if ($definition->name === 'transaction_isolation' || $definition->name === 'tx_isolation') {
154            return $this->isolation($definition, $value, $domain, $text);
155        }
156        if ($definition->name === 'sql_mode' && $value !== null && in_array($domain->kind, [Kind::Integer, Kind::Decimal, Kind::Double], true) && in_array($this->context->modes->release, [\SqlSemantics\Contract\GrammarRelease::MySql5651, \SqlSemantics\Contract\GrammarRelease::MySql5744], true)) {
157            return $this->modes($definition, $value, $domain, $text);
158        }
159
160        return match ($definition->shape) {
161            ValueShape::Boolean => $this->boolean($definition, $value, $domain, $text),
162            ValueShape::Integer, ValueShape::Unsigned => $this->integer($definition, $value, $domain, $text),
163            ValueShape::Double, ValueShape::Text => $this->text($definition, $value, $text),
164        };
165    }

184    /**
185     * Checks a boolean value.
186     *
187     * @throws \MySqlMemory\Error\SqlError When the value is not a boolean
188     */
189    public function boolean(Definition $definition, int|float|string|null $value, Domain $domain, string $text): string
190    {
191        if ($value !== null && ($domain->kind === Kind::Integer || $domain->kind === Kind::Decimal) && in_array($text, ['0', '1'], true)) {
192            return $text === '1' ? 'ON' : 'OFF';
193        }
194        if ($value !== null && $domain->kind === Kind::Integer || $domain->kind === Kind::Decimal || $domain->kind === Kind::Double) {
195            if ($domain->kind !== Kind::Integer) {
196                throw AdministrationError::WrongTypeForVariable->error($definition->name);
197            }
198            throw AdministrationError::WrongValueForVariable->error($definition->name, $text);
199        }
200        $word = strtoupper($text);
201        if (in_array($word, ['ON', 'TRUE', '1'], true)) {
202            return 'ON';
203        }
204        if (in_array($word, ['OFF', 'FALSE', '0'], true)) {
205            return 'OFF';
206        }
207
208        throw AdministrationError::WrongValueForVariable->error($definition->name, $text);
209    }

211    /**
212     * Checks an integer value, clipping it to the bounds of the variable; the value and the bounds of an unsigned variable compare as unsigned 64-bit integers, and an unsigned variable without known bounds (the catalogs of MySQL 5.6 and 5.7 have none) takes any unsigned 64-bit integer.
213     *
214     * @throws \MySqlMemory\Error\SqlError When the value is NULL or not an integer
215     */
216    public function integer(Definition $definition, int|float|string|null $value, Domain $domain, string $text): int
217    {
218        if ($value === null) {
219            throw AdministrationError::WrongValueForVariable->error($definition->name, 'NULL');
220        }
221        if ($domain->kind !== Kind::Integer && $domain->kind !== Kind::Year) {
222            throw AdministrationError::WrongTypeForVariable->error($definition->name);
223        }
224        $number = (int) $value;
225        $unsigned = $definition->shape === ValueShape::Unsigned;
226        $minimum = $definition->minimum ?? ($unsigned ? 0 : PHP_INT_MIN);
227        $maximum = $definition->maximum ?? ($unsigned ? -1 : PHP_INT_MAX);
228        if ($unsigned) {
229            $below = (!$domain->unsigned && $number < 0) || \MySqlMemory\Value\Integer::compare($number, true, $minimum, true) < 0;
230            $above = !$below && \MySqlMemory\Value\Integer::compare($number, true, $maximum, true) > 0;
231        } else {
232            $below = $number < $minimum && !($domain->unsigned && $number < 0);
233            $above = $number > $maximum || ($domain->unsigned && $number < 0);
234        }
235        if ($below || $above) {
236            $this->context->warning(DataError::TruncatedWrongValue, $definition->name, $text);
237            $number = $below ? $minimum : $maximum;
238        }
239
240        return $number;
241    }

243    /**
244     * Checks a number assigned to sql_mode in MySQL 5.6 and 5.7: the modes whose bits it sets, each mode the bit of its place in the list of the release; a number that is not an integer is ER_WRONG_TYPE_FOR_VAR, a negative one or one beyond 32 bits ER_WRONG_VALUE_FOR_VAR (verified on live 5.6.51 and 5.7.44 servers).
245     *
246     * @throws \MySqlMemory\Error\SqlError When the number does not stand for modes
247     */
248    public function modes(Definition $definition, int|float|string $value, Domain $domain, string $text): string
249    {
250        if ($domain->kind !== Kind::Integer) {
251            throw AdministrationError::WrongTypeForVariable->error($definition->name);
252        }
253        $number = (int) $value;
254        if ($number < 0 || $number > 4294967295 || $text !== (string) $number) {
255            throw AdministrationError::WrongValueForVariable->error($definition->name, $text);
256        }
257        $names = [];
258        foreach (SqlModes::names($this->context->modes->release) as $bit => $name) {
259            if ((($number >> $bit) & 1) === 1 && !str_starts_with($name, 'NOT_USED')) {
260                $names[] = $name;
261            }
262        }
263        $modes = new SqlModes($names, $this->context->modes->release);
264        $this->deprecated($modes);
265
266        return $modes->toString();
267    }

269    /**
270     * Records the warnings MySQL 5.7 raises for a new sql_mode: that NO_ZERO_DATE, NO_ZERO_IN_DATE and ERROR_FOR_DIVISION_BY_ZERO belong with a strict mode, unless the modes hold all of them with a strict mode or none of them, then that changing NO_AUTO_CREATE_USER from the mode of the session is deprecated (verified on a live 5.7.44 server).
271     */
272    public function deprecated(SqlModes $modes): void
273    {
274        if ($modes->release !== \SqlSemantics\Contract\GrammarRelease::MySql5744) {
275            return;
276        }
277        $held = count(array_filter(['NO_ZERO_DATE', 'NO_ZERO_IN_DATE', 'ERROR_FOR_DIVISION_BY_ZERO'], $modes->has(...)));
278        $strict = $modes->has('STRICT_TRANS_TABLES') || $modes->has('STRICT_ALL_TABLES');
279        if (!($strict && $held === 3) && !(!$strict && $held === 0)) {
280            $this->context->diagnostics->warning(3135, "'NO_ZERO_DATE', 'NO_ZERO_IN_DATE' and 'ERROR_FOR_DIVISION_BY_ZERO' sql modes should be used with strict mode. They will be merged with strict mode in a future release.");
281        }
282        if ($modes->has('NO_AUTO_CREATE_USER') !== $this->context->modes->has('NO_AUTO_CREATE_USER')) {
283            $this->context->diagnostics->warning(3090, "Changing sql mode 'NO_AUTO_CREATE_USER' is deprecated. It will be removed in a future release.");
284        }
285    }

287    /**
288     * Checks a text value; sql_mode and the collation and character set variables are checked by name.
289     *
290     * @throws \MySqlMemory\Error\SqlError When the value is NULL for a variable that does not take it, or not a mode, collation or character set the variable takes
291     */
292    public function text(Definition $definition, int|float|string|null $value, string $text): ?string
293    {
294        if ($value === null) {
295            if (!in_array($definition->name, ['character_set_results', 'session_track_system_variables', 'innodb_tmpdir', 'innodb_ft_user_stopword_table'], true)) {
296                throw AdministrationError::WrongValueForVariable->error($definition->name, 'NULL');
297            }
298
299            return null;
300        }
301        if ($definition->name === 'sql_mode') {
302            $modes = SqlModes::parse($text, $this->context->modes->release);
303            if ($modes === null) {
304                throw AdministrationError::WrongValueForVariable->error($definition->name, $text);
305            }
306            $this->deprecated($modes);
307
308            return $modes->toString();
309        }
310        if ($definition->name === 'block_encryption_mode') {
311            return Ciphers::mode($text) ?? throw AdministrationError::WrongValueForVariable->error($definition->name, $text);
312        }
313        if (str_starts_with($definition->name, 'collation_')) {
314            $collation = Collation::named($text);
315            if ($collation === null) {
316                throw SchemaError::UnknownCollation->error($text);
317            }
318
319            return $collation->name;
320        }
321        if (str_starts_with($definition->name, 'character_set_')) {
322            $charset = Charset::named($text);
323            if ($charset === null) {
324                throw SchemaError::UnknownCharacterSet->error($text);
325            }
326
327            return $charset->name;
328        }
329
330        return $text;
331    }
