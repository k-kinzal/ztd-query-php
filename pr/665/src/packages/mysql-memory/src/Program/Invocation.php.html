20/**
21 * Runs the body of a stored program in the session: in the database of the program, under the sql_mode it was created with.
22 *
23 * The current database and the sql_mode of the session are those of the program while it
24 * runs, and come back when it ends. A stored function is called with the values of its
25 * arguments, each stored into its parameter as into a column of its type; a function cannot
26 * call itself, even through another program (ER_SP_NO_RECURSION). Its statements are part of
27 * the statement that calls it, and the conditions its last statement left are added to the
28 * diagnostics area of that statement, while ROW_COUNT() stays as it was. The value RETURN
29 * answers is stored as into a column of the declared type, named after the call; a function
30 * that ends without RETURN is ER_SP_NORETURNEND (verified on a live 8.4 server).
31 * Source: https://dev.mysql.com/doc/refman/8.4/en/create-procedure.html,
32 * https://dev.mysql.com/doc/refman/8.4/en/stored-routines-syntax.html.
33 *
34 * @visibility MySqlMemory
35 */

45    /**
46     * Runs a body in an activation and answers the jump that ended it, if any.
47     *
48     * @param string $database The database of the program
49     * @param string $mode The sql_mode the program was created with
50     *
51     * @throws SqlError When the body raises an error no handler handles
52     */
53    public function run(Activation $activation, Node $body, string $database, string $mode): ?Jump
54    {
55        $session = $this->session;
56        $variables = $session->variables;
57        $saved = [$session->program, $variables->database, $variables->session];
58        $session->program = $activation;
59        $variables->database = $database;
60        $variables->session['sql_mode'] = $mode;
61        try {
62            return (new Interpreter($session, $activation))->statement($body);
63        } finally {
64            $session->program = $saved[0];
65            $variables->database = $saved[1];
66            if (array_key_exists('sql_mode', $saved[2])) {
67                $variables->session['sql_mode'] = $saved[2]['sql_mode'];
68            } else {
69                unset($variables->session['sql_mode']);
70            }
71        }
72    }

74    /**
75     * Calls a stored function with the values of its arguments and answers the value it returns.
76     *
77     * @param list<array{int|float|string|null, Domain}> $arguments The value and type of each argument
78     * @param string $name The name of the call, which an error converting the value it returns names
79     *
80     * @throws SqlError When the function calls itself, fails, ends without RETURN, or its value is refused
81     */
82    public function function(Routine $routine, array $arguments, string $name): int|float|string|null
83    {
84        $session = $this->session;
85        $caller = $session->program;
86        $qualified = $routine->schema . '.' . $routine->name;
87        if ($caller !== null && $caller->depth('FUNCTION', $qualified) > 0) {
88            throw ProgramError::NoRecursion->error();
89        }
90        $collation = Collation::named($routine->charsets[2]) ?? Collation::known('utf8mb4_0900_ai_ci');
91        $activation = new Activation('FUNCTION', $qualified, true, $collation, $caller);
92
93        return $this->contained(function () use ($activation, $routine, $arguments, $name): int|float|string|null {
94            $activation->scope[] = $this->parameters($routine, $arguments, $routine->mode);
95            $jump = $this->run($activation, \MySqlMemory\Program\Routine\Body::of($routine)->body, $routine->schema, $routine->mode);
96            if ($jump === null || $jump->flow !== Flow::Return) {
97                throw ProgramError::FunctionWithoutReturn->error($routine->name);
98            }
99
100            return (new Store($this->context($routine->mode)))->value($jump->value, $jump->domain ?? Domain::null(), new ColumnDefinition($name, $routine->returned(), Fill::none()));
101        });
102    }

104    /**
105     * Runs a stored function or a trigger inside the statement that invokes it: with a diagnostics area of its own, whose conditions are added to that of the statement once it ends, and leaving ROW_COUNT() as it was.
106     *
107     * @template T
108     *
109     * @param Closure(): T $run
110     * @return T
111     */
112    public function contained(Closure $run): mixed
113    {
114        $session = $this->session;
115        $diagnostics = $session->diagnostics;
116        [$conditions, $signalled, $count] = [$diagnostics->conditions, $diagnostics->signalled, $session->variables->rowCount];
117        $diagnostics->clear();
118        try {
119            return $run();
120        } finally {
121            $own = [$diagnostics->conditions, $diagnostics->signalled];
122            [$diagnostics->conditions, $diagnostics->signalled, $session->variables->rowCount] = [$conditions, $signalled, $count];
123            foreach ($own[0] as $position => [$level, $number, $message]) {
124                if (isset($own[1][$position])) {
125                    $diagnostics->signalled[count($diagnostics->conditions)] = $own[1][$position];
126                }
127                $diagnostics->conditions[] = [$level, $number, $message];
128            }
129        }
130    }

132    /**
133     * Binds the values of the arguments of a routine to its parameters, each stored as into a column of its type.
134     *
135     * @param list<array{int|float|string|null, Domain}> $arguments The value and type of each argument; a missing one leaves its parameter NULL
136     *
137     * @throws SqlError When a value is refused
138     */
139    public function parameters(Routine $routine, array $arguments, string $mode): Row
140    {
141        $collation = Collation::named($routine->charsets[2]) ?? Collation::known('utf8mb4_0900_ai_ci');
142        $context = $this->context($mode);
143        $row = new Row(\MySqlMemory\Program\Routine\Body::of($routine)->parameters, []);
144        foreach (\MySqlMemory\Program\Routine\Body::of($routine)->parameters->parameters as $index => $parameter) {
145            $variable = new Variable($parameter->name->value, VariableDomain::declared($parameter->type, $parameter->collation, $collation));
146            if (isset($arguments[$index])) {
147                $variable->assign($arguments[$index][0], $arguments[$index][1], $context);
148            }
149            $row->variables[] = $variable;
150        }
151
152        return $row;
153    }
