12/**
13 * The accounts and their privileges as the grant tables and the privilege tables of INFORMATION_SCHEMA hold them.
14 *
15 * The grant tables name each static privilege in a column of its own, holding Y or N; a table
16 * or column grant holds a set of privilege names. INFORMATION_SCHEMA names an account as
17 * 'user'@'host' and lists the accounts as the server checks them: those of a host without a
18 * wildcard first, each kind with the installed accounts by user name and then the others in
19 * the order they were created (verified on a live 8.4.7 server). The emulator does not record who granted a
20 * privilege and when, so it reports the grants as made by root@localhost when the server started.
21 * Source: https://dev.mysql.com/doc/refman/8.4/en/grant-tables.html,
22 * https://dev.mysql.com/doc/refman/8.4/en/information-schema-user-privileges-table.html.
23 *
24 * @visibility MySqlMemory
25 */

70    /**
71     * Answers the accounts as the server checks them: those of a host without a wildcard first, each kind with the installed accounts by user name and then the others in the order they were created.
72     *
73     * @return list<Account>
74     */
75    public static function checked(Reading $reading): array
76    {
77        $accounts = array_values($reading->instance->accounts->accounts);
78        $installed = ['mysql.infoschema', 'mysql.session', 'mysql.sys', 'root'];
79        $rank = static fn (Account $account, int $index): array => [strpbrk($account->identity->host, '%_') !== false, !in_array($account->identity->user, $installed, true), in_array($account->identity->user, $installed, true) ? $account->identity->user : '', $index];
80        $ranked = array_map(static fn (Account $account, int $index): array => [$rank($account, $index), $account], $accounts, array_keys($accounts));
81        usort($ranked, static fn (array $left, array $right): int => $left[0] <=> $right[0]);
82
83        return array_map(static fn (array $entry): Account => $entry[1], $ranked);
84    }

118    public static function set(array $held, array $members, bool $grant = false): string
119    {
120        $listed = [];
121        foreach ($members as $name => $written) {
122            if (isset($held[$name]) || ($name === 'GRANT' && $grant)) {
123                $listed[] = $written;
124            }
125        }
126
127        return implode(',', $listed);
