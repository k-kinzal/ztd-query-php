13/**
14 * The functions that write dates and times by a format: DATE_FORMAT and TIME_FORMAT, and GET_FORMAT, which answers formats.
15 *
16 * Each `%` specifier writes a part of the value; `%` before any other character writes that
17 * character, and a format that writes nothing makes the result NULL. The names of days and
18 * months are those of lc_time_names; a zero month has no name and a zero date no day of the
19 * week, which makes the result NULL. TIME_FORMAT writes a time: its hours may pass 23, a negative
20 * time writes a minus sign first, and the specifiers of the date make the result NULL except
21 * those that write numbers, which write zeros. Weeks are counted as WEEK counts them, also for a
22 * date with a zero month or day (verified on a live 8.4 server).
23 * Source: https://dev.mysql.com/doc/refman/8.4/en/date-and-time-functions.html#function_date-format,
24 * https://dev.mysql.com/doc/refman/8.4/en/date-and-time-functions.html#function_get-format.
25 *
26 * @visibility MySqlMemory\Evaluation
27 */
28final class Formatting

61    {

63        if ($parts === null) {
64            return null;
65        }
66        $text = $format->evaluate($frame);

72     * TIME_FORMAT: a time written by a format.
73     */
74    public function time(Frame $frame, Evaluable $value, Evaluable $format): ?string
75    {
76        $time = (new Readings())->time($frame, $value);
77        if ($time === null) {
78            return null;
79        }
80        $text = $format->evaluate($frame);
81        if ($text === null) {
82            return null;
83        }
84        $written = $this->write([0, 0, 0, $time[1], $time[2], $time[3], $time[4]], (string) $text, true, $frame);
85
86        return $written === null || !$time[0] ? $written : '-' . $written;
87    }

102    /**
103     * Writes the parts of a value by a format, or answers null when a specifier has nothing to write or the format writes nothing.
104     *
105     * @param array{int, int, int, int, int, int, int} $parts Year, month, day, hour, minute, second and microsecond
106     * @param bool $time Whether the value is a time, whose hours may pass 23 and whose date is not written
107     */
108    public function write(array $parts, string $format, bool $time, Frame $frame): ?string
109    {
110        $locale = Locale::named((string) $frame->context->variables->read('lc_time_names')) ?? Locale::default();
111        $output = '';
112        for ($index = 0, $size = strlen($format); $index < $size; $index++) {
113            if ($format[$index] !== '%' || $index + 1 >= $size) {
114                $output .= $format[$index];
115                continue;
116            }
117            $specifier = $format[++$index];
118            if ($time && str_contains(self::UNTIMED, $specifier)) {
119                return null;
120            }
121            $written = $this->specifier($specifier, $parts, $locale);
122            if ($written === null) {
123                return null;
124            }
125            $output .= $written;
126        }
127
128        return $output === '' ? null : $output;
129    }

132     * Writes one specifier of the parts of a value, or answers null when it has nothing to write.
133     *
134     * @param array{int, int, int, int, int, int, int} $parts
135     */
136    public function specifier(string $specifier, array $parts, Locale $locale): ?string
137    {
138        return match (true) {
139            str_contains('abMWw', $specifier) => $this->name($specifier, $parts, $locale),
140            str_contains('UuVvXx', $specifier) => $this->week($specifier, $parts),
141            str_contains('fHhIiklprSsT', $specifier) => $this->clock($specifier, $parts),
142            default => $this->calendar($specifier, $parts),
143        };
144    }

146    /**
147     * Writes a specifier of a name or of the day of the week, %a, %b, %M, %W or %w, or answers null when the value has none.
148     *
149     * @param array{int, int, int, int, int, int, int} $parts
150     */
151    public function name(string $specifier, array $parts, Locale $locale): ?string
152    {
153        [$year, $month, $day] = $parts;
154        $undated = $year === 0 && $month === 0;
155        $weekday = Readings::weekday(Readings::dayNumber($year, $month, $day));
156
157        return match ($specifier) {
158            'a' => $undated ? null : $locale->shortDays[$weekday] ?? null,
159            'b' => $month === 0 ? null : $locale->shortMonths[$month - 1] ?? null,
160            'M' => $month === 0 ? null : $locale->months[$month - 1] ?? null,
161            'W' => $undated ? null : $locale->days[$weekday] ?? null,
162            'w' => $undated ? null : (string) (($weekday + 1) % 7),
163            default => $specifier,
164        };
165    }

182            'X' => $this->year($weeks->week($year, $month, $day, 2)[1]),

188    /**
189     * Writes a specifier of the time of day: %f, %H, %h, %I, %i, %k, %l, %p, %r, %S, %s or %T.
190     *
191     * @param array{int, int, int, int, int, int, int} $parts
192     */
193    public function clock(string $specifier, array $parts): string
194    {
195        [, , , $hour, $minute, $second, $micro] = $parts;
196        $twelve = $hour % 12 === 0 ? 12 : $hour % 12;
197        $meridian = $hour % 24 >= 12 ? 'PM' : 'AM';
198
199        return match ($specifier) {
200            'f' => sprintf('%06d', $micro),
201            'H' => sprintf('%02d', $hour),
202            'h', 'I' => sprintf('%02d', $twelve),
203            'i' => sprintf('%02d', $minute),
204            'k' => (string) $hour,
205            'l' => (string) $twelve,
206            'p' => $meridian,
207            'r' => sprintf('%02d:%02d:%02d %s', $hour % 24 % 12 === 0 ? 12 : $hour % 24 % 12, $minute, $second, $meridian),
208            'S', 's' => sprintf('%02d', $second),
209            'T' => sprintf('%02d:%02d:%02d', $hour, $minute, $second),
210            default => $specifier,
211        };
212    }

214    /**
215     * Writes a specifier of the date in numbers, %c, %D, %d, %e, %j, %m, %Y or %y; any other character is written as itself.
216     *
217     * @param array{int, int, int, int, int, int, int} $parts
218     */
219    public function calendar(string $specifier, array $parts): string
220    {
221        [$year, $month, $day] = $parts;
222
223        return match ($specifier) {
224            'c' => (string) $month,
225            'D' => $day . $this->suffix($day),
226            'd' => sprintf('%02d', $day),
227            'e' => (string) $day,
228            'j' => sprintf('%03d', Readings::dayNumber($year, $month, $day) - Readings::dayNumber($year, 1, 1) + 1),
229            'm' => sprintf('%02d', $month),
230            'Y' => sprintf('%04d', $year),
231            'y' => sprintf('%02d', $year % 100),
232            default => $specifier,
233        };
234    }

236    /**
237     * Answers the English suffix of a day of the month: st, nd, rd or th.
238     */
239    public function suffix(int $day): string
240    {
241        $last = $day % 10;
242        if (intdiv($day, 10) === 1 || $last < 1 || $last > 3) {
243            return 'th';
244        }
245
246        return ['st', 'nd', 'rd'][$last - 1];
247    }

252    public function year(int $year): string
