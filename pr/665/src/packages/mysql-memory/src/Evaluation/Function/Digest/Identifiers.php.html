16/**
17 * The identifier functions: UUID, UUID_SHORT, UUID_TO_BIN, BIN_TO_UUID and IS_UUID.
18 *
19 * UUID() is a version 1 UUID: the time in 100-nanosecond intervals since 15 October 1582, a clock
20 * sequence and a node, here random for each server; consecutive calls never repeat it.
21 * UUID_SHORT() starts at `(server_id & 255) << 56 | startup time << 24` and counts up by one for
22 * each call; the emulator takes the startup time when the server first answers UUID_SHORT().
23 * UUID_TO_BIN reads a UUID of 32 hexadecimal digits, with or without the dashes and braces of the
24 * string form; BIN_TO_UUID reads 16 bytes. Both move the time-high part first when the swap flag is
25 * true, and refuse another value with ER_WRONG_VALUE_FOR_TYPE, which quotes at most 128 bytes of it.
26 * IS_UUID tells whether the bytes of a string are a UUID UUID_TO_BIN reads (verified on a live 8.4
27 * server).
28 * Source: https://dev.mysql.com/doc/refman/8.4/en/miscellaneous-functions.html.
29 *
30 * @visibility MySqlMemory
31 */
32final class Identifiers
33{
34    /**
35     * The 100-nanosecond intervals from 15 October 1582 to the Unix epoch.
36     */
37    public const GREGORIAN_OFFSET = 0x01B21DD213814000;

39    /**
40     * The state of the generators of each server, by its global variables: the node, the clock sequence, the last time and the next short UUID.
41     *
42     * @var WeakMap<object, array{string, int, int, int|null}>|null
43     */
44    private static ?WeakMap $generators = null;

46    /**
47     * Answers the functions of the family.
48     *
49     * @return list<Routine>
50     */
51    public function routines(): array
52    {
53        return [
54            new Routine('UUID', 0, 0, $this->uuid(...)),
55            new Routine('UUID_SHORT', 0, 0, $this->uuidShort(...)),
56            new Routine('UUID_TO_BIN', 1, 2, $this->uuidToBin(...)),
57            new Routine('BIN_TO_UUID', 1, 2, $this->binToUuid(...)),
58            new Routine('IS_UUID', 1, 1, fn (Frame $f, array $a): ?int => ($t = Convert::toText($a[0]->evaluate($f), $a[0]->domain())) === null ? null : ($this->bytes($t) === null ? 0 : 1)),
59        ];

62    /**
63     * Answers the state of the generators of the server a frame runs on.
64     *
65     * @return array{string, int, int, int|null}
66     */
67    public function generator(Frame $frame): array
68    {
69        $globals = $frame->context->variables->globals;
70        $state = self::$generators[$globals] ?? null;
71        if ($state === null) {
72            $state = [random_bytes(6), random_int(0, 0x3FFF), 0, null];
73            $this->store($frame, $state);
74        }
75
76        return $state;
77    }

90    /**
91     * UUID(): a version 1 UUID, later than any the server answered before.
92     *
93     * @param list<Evaluable> $arguments
94     */
95    public function uuid(Frame $frame, array $arguments, Domain $result): string
96    {
97        [$node, $sequence, $last, $short] = $this->generator($frame);
98        $time = max((int) (microtime(true) * 10000000) + self::GREGORIAN_OFFSET, $last + 1);
99        $this->store($frame, [$node, $sequence, $time, $short]);
100        $text = sprintf('%08x-%04x-%04x-%04x-%s', $time & 0xFFFFFFFF, ($time >> 32) & 0xFFFF, (($time >> 48) & 0x0FFF) | 0x1000, $sequence | 0x8000, bin2hex($node));
101
102        return (new Hashes())->written($text, $result);
103    }

106     * UUID_SHORT(): the next number of the server's sequence of short UUIDs.
107     *
108     * @param list<Evaluable> $arguments
109     */
110    public function uuidShort(Frame $frame, array $arguments, Domain $result): int
111    {
112        [$node, $sequence, $last, $short] = $this->generator($frame);
113        $short ??= ((((int) $frame->context->variables->read('server_id')) & 255) << 56) + (((int) $frame->context->started) << 24);
114        $this->store($frame, [$node, $sequence, $last, $short + 1]);

116        return $short;

119    /**
120     * Reads the 16 bytes of a UUID written as text, or answers null when the text is not a UUID.
121     */
122    public function bytes(string $text): ?string
123    {
124        if (strlen($text) === 38 && $text[0] === '{' && $text[37] === '}') {
125            $text = substr($text, 1, 36);
126            if (strlen($text) !== 36) {
127                return null;
128            }
129        }
130        if (strlen($text) === 36 && preg_match('/\A[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}\z/', $text) === 1) {
131            $text = str_replace('-', '', $text);
132        }
133
134        return strlen($text) === 32 && ctype_xdigit($text) ? (string) hex2bin($text) : null;
135    }

138     * Reads the swap flag of UUID_TO_BIN and BIN_TO_UUID: true when it is a number other than zero.

144    public function swapped(Frame $frame, array $arguments): bool

146        return isset($arguments[1]) && Convert::toBool($arguments[1]->evaluate($frame), $arguments[1]->domain(), $frame->context) === true;
147    }

149    /**
150     * Quotes a value an error refuses: its text in UTF-8, binary bytes escaped, cut to 128 bytes with each byte of a broken last character as `?`.
151     */
152    public function quoted(string $text, Domain $domain): string
153    {
154        $shown = Convert::shown($text, $domain->kind === Kind::String ? $domain->collation->charset : null);
155        if (strlen($shown) <= 128) {
156            return $shown;
157        }
158        $cut = substr($shown, 0, 128);
159        $whole = $cut;
160        while (!mb_check_encoding($whole, 'UTF-8')) {
161            $whole = substr($whole, 0, -1);
162        }
163
164        return $whole . str_repeat('?', strlen($cut) - strlen($whole));
165    }

167    /**
168     * UUID_TO_BIN(text[, swap]): the 16 bytes of a UUID, the time-high part first when swapped.
169     *
170     * @param list<Evaluable> $arguments
171     *
172     * @throws \MySqlMemory\Error\SqlError When the text is not a UUID
173     */
174    public function uuidToBin(Frame $frame, array $arguments, Domain $result): ?string
175    {
176        $text = Convert::toText($arguments[0]->evaluate($frame), $arguments[0]->domain());
177        if ($text === null) {
178            return null;
179        }
180        $bytes = $this->bytes($text) ?? throw DataError::WrongValueForType->error('string', $this->quoted($text, $arguments[0]->domain()), 'uuid_to_bin');
181
182        return $this->swapped($frame, $arguments) ? substr($bytes, 6, 2) . substr($bytes, 4, 2) . substr($bytes, 0, 4) . substr($bytes, 8) : $bytes;
183    }
184
185    /**
186     * BIN_TO_UUID(bytes[, swap]): the string form of 16 bytes, read with the time-high part first when swapped.
187     *
188     * @param list<Evaluable> $arguments
189     *
190     * @throws \MySqlMemory\Error\SqlError When the value is not 16 bytes
191     */
192    public function binToUuid(Frame $frame, array $arguments, Domain $result): ?string
193    {
194        $bytes = Convert::toText($arguments[0]->evaluate($frame), $arguments[0]->domain());
195        if ($bytes === null) {
196            return null;
197        }
198        if (strlen($bytes) !== 16) {
199            throw DataError::WrongValueForType->error('string', $this->quoted($bytes, $arguments[0]->domain()), 'bin_to_uuid');
200        }
201        if ($this->swapped($frame, $arguments)) {
202            $bytes = substr($bytes, 4, 4) . substr($bytes, 2, 2) . substr($bytes, 0, 2) . substr($bytes, 8);
203        }
204        $hex = bin2hex($bytes);
205
206        return (new Hashes())->written(substr($hex, 0, 8) . '-' . substr($hex, 8, 4) . '-' . substr($hex, 12, 4) . '-' . substr($hex, 16, 4) . '-' . substr($hex, 20), $result);
207    }
