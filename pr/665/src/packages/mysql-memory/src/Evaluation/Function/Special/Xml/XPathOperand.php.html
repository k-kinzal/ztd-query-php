13/**
14 * A value an XPath expression of EXTRACTVALUE computes: a set of nodes, an integer, a double, a string, a truth value or NULL.
15 *
16 * The server computes XPath with its own SQL operations, so a value converts as SQL converts it:
17 * a node set reads as the texts directly in its nodes, joined by spaces; a string reads as a
18 * number with the server's warning; a double keeps the decimals of the literals it was computed
19 * from and is written with them, or as the server writes a double when they are not fixed
20 * (verified on a live 8.4 server).
21 * Source: https://dev.mysql.com/doc/refman/8.4/en/xml-functions.html.
22 *
23 * @visibility MySqlMemory
24 */

105    public function text(XmlDocument $document): ?string
106    {
107        return match ($this->kind) {
108            self::NODES => implode(' ', $document->texts($this->nodes)),
109            self::INTEGER, self::BOOLEAN => (string) $this->value,
110            self::DOUBLE => $this->decimals >= self::NOT_FIXED ? Real::format((float) $this->value) : $this->fixed((float) $this->value),
111            self::STRING => (string) $this->value,

113        };
114    }

120    {
121        $text = sprintf('%.' . $this->decimals . 'F', $value);

130    {

136    }

138    /**

140     */
141    public function holds(XmlDocument $document, Frame $frame, Collation $collation): ?bool
142    {
143        if ($this->kind === self::NODES) {

145        }
