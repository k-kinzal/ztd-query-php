18/**
19 * The numeric functions MOD, ATAN, ATAN2, CRC32, BIT_COUNT and RAND.
20 *
21 * MOD(N, M) is the operator N % M. ATAN(Y, X) and ATAN2(Y, X) are the arc tangent of Y / X in
22 * the quadrant of the point (X, Y). CRC32 checks the bytes of the text of its argument, in the
23 * character set of the argument. BIT_COUNT counts the bits of a binary string, and otherwise of
24 * the 64-bit integer its argument reads as. RAND draws from the generator the server seeds with
25 * the low 32 bits of its argument (Generator).
26 * Source: https://dev.mysql.com/doc/refman/8.4/en/mathematical-functions.html,
27 * https://dev.mysql.com/doc/refman/8.4/en/bit-functions.html.
28 *
29 * @visibility MySqlMemory
30 */

50    /**
51     * ATAN(X), ATAN(Y, X) and ATAN2(Y, X): the arc tangent, in the quadrant of the point (X, Y) with two arguments.
52     *
53     * @param list<Evaluable> $arguments
54     */
55    public function arcTangent(Frame $frame, array $arguments): ?float
56    {
57        $values = [];
58        foreach ($arguments as $argument) {
59            $value = Convert::toDouble($argument->evaluate($frame), $argument->domain(), $frame->context);
60            if ($value === null) {
61                return null;
62            }
63            $values[] = $value;
64        }
65
66        return count($values) === 1 ? atan($values[0]) : atan2($values[0], $values[1] ?? 1.0);
67    }

71     *
72     * @param list<Evaluable> $arguments
73     */
74    public function checksum(Frame $frame, array $arguments): ?int
75    {
76        $text = Convert::toText($arguments[0]->evaluate($frame), $arguments[0]->domain());
77
78        return $text === null ? null : crc32($text);
79    }

81    /**
82     * BIT_COUNT(N): the number of bits set; a binary string counts the bits of its bytes, any other value those of the 64-bit integer it reads as.
83     *
84     * In MySQL 5.6 and 5.7 a binary string is read as an integer too.
85     *
86     * @param list<Evaluable> $arguments
87     */
88    public function bitCount(Frame $frame, array $arguments): ?int
89    {
90        $domain = $arguments[0]->domain();
91        $value = $arguments[0]->evaluate($frame);
92        if ($value === null) {
93            return null;
94        }
95        if ($domain->kind === Kind::String && $domain->collation->bytes() && !in_array($frame->context->modes->release, [GrammarRelease::MySql5651, GrammarRelease::MySql5744], true)) {
96            $count = 0;
97            foreach (count_chars((string) $value, 1) as $byte => $times) {
98                $count += substr_count(decbin($byte), '1') * $times;
99            }
100
101            return $count;
102        }
103        return substr_count(sprintf('%064b', $this->integer($value, $domain, $frame->context)), '1');
104    }

109    public function integer(int|float|string $value, Domain $domain, Context $context): int
110    {

114    /**
115     * RAND() and RAND(N): a random double in [0, 1), or the next of the sequence the seed starts.
116     *
117     * Without a seed the value comes from the random generator of PHP. A seed known for the
118     * statement starts one sequence that each evaluation advances; any other seed starts a
119     * sequence for each evaluation, which answers its first value.
120     *
121     * @param list<Evaluable> $arguments
122     */
123    public function random(Frame $frame, array $arguments): float
124    {
125        $seed = $arguments[0] ?? null;
126        if (!$seed instanceof Generator) {
127            return mt_rand() / (mt_getrandmax() + 1);
128        }
129
130        return $seed->next($frame);
131    }

133    /**
134     * Wraps the seed of RAND in the generator it starts, knowing whether it stays the same for the statement.

138     * @return list<Evaluable>
139     */
140    public function seed(Frame $frame, array $arguments, array $known): array
141    {
142        return $arguments === [] ? [] : [new Generator($arguments[0], $known[0] ?? false)];
