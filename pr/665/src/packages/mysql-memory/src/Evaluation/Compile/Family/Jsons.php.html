51/**
52 * Compiles the JSON operators: `col->'path'`, `col->>'path'` and `value MEMBER OF (array)`, and JSON_VALUE().
53 *
54 * A JSON argument is a JSON value, or a string read as a JSON text after it is converted to
55 * utf8mb4; a value of another type is refused for each row, even NULL (ER_INVALID_TYPE_FOR_JSON),
56 * a binary string is refused (ER_INVALID_JSON_CHARSET, followed by ER_INVALID_TYPE_FOR_JSON), and a text that is no JSON document is
57 * refused with the message of the parser (ER_INVALID_JSON_TEXT_IN_PARAM). `->` is
58 * JSON_EXTRACT() and `->>` JSON_UNQUOTE(JSON_EXTRACT()); the path is read once the document is,
59 * so a NULL document hides a path that is not valid. MEMBER OF is NULL when its value is NULL,
60 * before the array is checked, or when the array is NULL; it tells whether the value equals an
61 * element of the array, or the array itself when it is no array. The value is a JSON value made
62 * from its SQL type: a number keeps its type, a string is a JSON string, a binary string an opaque
63 * value, a temporal value a JSON temporal value, and a predicate a JSON boolean (verified on a
64 * live 8.4 server).
65 * Source: https://dev.mysql.com/doc/refman/8.4/en/json-search-functions.html.
66 *
67 * @visibility MySqlMemory
68 */
69final class Jsons

79     * Compiles `col->'path'` and `col->>'path'`.
80     *
81     * @throws SqlError When the column cannot be compiled
82     */
83    public function extraction(JsonExtraction $node, Scope $scope): Evaluable
84    {
85        $column = $this->compiler->compile($node->column, $scope);
86        $path = $node->path->value;
87        $unquote = $node->unquote;
88        $routine = new Routine($unquote ? '->>' : '->', 1, 1, static function (Frame $f, array $a) use ($path, $unquote): ?string {
89            $document = Jsons::read($a[0], $f, 1, 'json_extract');
90            if ($document === null) {
91                return null;
92            }
93            try {
94                $found = JsonPath::parse($path)->extract($document);
95            } catch (JsonSyntax $failure) {
96                throw DataError::InvalidJsonPath->error($failure->position);
97            }
98            if ($found === null) {
99                return null;
100            }
101
102            return $unquote ? $found->unquoted() : $found->store();
103        });
104
105        return new Call($routine, [$column], $this->compiler->domain($node)->withSource($unquote ? '' : 'json_extract'));
106    }

108    /**
109     * Compiles JSON_VALUE(): the path is read once, and a DEFAULT value is converted to the RETURNING type as CAST converts it.
110     *
111     * @throws SqlError When the path is not valid or an argument cannot be compiled
112     */
113    public function jsonValue(JsonValueCall $node, Scope $scope): Evaluable
114    {
115        $document = $this->compiler->compile($node->document, $scope);
116        try {
117            $path = JsonPath::parse($node->path->value());
118        } catch (JsonSyntax $failure) {
119            throw new SqlError(DataError::InvalidJsonPath, DataError::InvalidJsonPath->message($failure->position), $failure);
120        }
121        $domain = $this->compiler->domain($node);
122        $response = function (?JsonResponse $response) use ($scope, $domain): array {
123            $default = $response?->default;
124
125            return [$response->kind ?? JsonResponseKind::Null, $default === null ? null : new Conversion($this->compiler->compile($default, $scope), $domain, null, 'CHAR')];
126        };
127
128        return new ValueCall($document, $path, $domain, $response($node->onEmpty), $response($node->onError));
129    }

131    /**
132     * Compiles `value MEMBER OF (array)`.
133     *
134     * @throws SqlError When an operand cannot be compiled
135     */
136    public function member(MemberOf $node, Scope $scope): Evaluable
137    {
138        $value = $this->compiler->compile($node->operand, $scope);
139        $array = $this->compiler->compile($node->array, $scope);
140        $boolean = $this->boolean($node->operand);
141        $routine = new Routine('MEMBER OF', 2, 2, static function (Frame $f, array $a) use ($boolean): ?int {
142            $value = $a[0]->evaluate($f);
143            if ($value === null) {
144                return null;
145            }
146            $document = Jsons::read($a[1], $f, 2, 'member of');
147            if ($document === null) {
148                return null;
149            }
150            $member = Jsons::value($value, $a[0]->domain(), $boolean);
151            foreach ($document->type === JsonKind::Array ? $document->children() : [$document] as $element) {
152                if ($member->equals($element)) {
153                    return 1;
154                }
155            }
156
157            return 0;
158        });
159
160        return new Call($routine, [$value, $array], $this->compiler->domain($node));
161    }

177    /**
178     * Reads a JSON argument of a function: a JSON value, or a string holding a JSON text; NULL is null.
179     *
180     * @param int $position The position of the argument, counted from 1
181     * @param string $function The function name the server writes in its messages
182     *
183     * @throws SqlError When the argument is of another type, a binary string, or no JSON text
184     */
185    public static function document(int|float|string|null $value, Domain $domain, int $position, string $function): ?JsonNode
186    {
187        if ($domain->kind !== Kind::String && $domain->kind !== Kind::Json && $domain->kind !== Kind::Null) {
188            throw DataError::InvalidJsonType->error($position, $function);
189        }
190        if ($value === null) {
191            return null;
192        }
193        if ($domain->kind === Kind::Json) {
194            return JsonNode::load((string) $value);
195        }
196        $text = (string) $value;
197        if ($domain->kind === Kind::String) {
198            $charset = $domain->collation->charset;
199            if ($charset === Charset::binary()) {
200                throw new SqlError(DataError::InvalidJsonCharset, DataError::InvalidJsonCharset->message('binary'), null, [[DataError::InvalidJsonType->value, DataError::InvalidJsonType->message($position, $function)]]);
201            }
202            $text = Encoding::convert($text, $charset, Charset::known('utf8mb4'));
203        }
204
205        return self::parse($text, $position, $function);
206    }

209     * Reads a JSON text given to a function.
210     *
211     * @param int $position The position of the argument, counted from 1
212     * @param string $function The function name the server writes in its messages
213     *
214     * @throws SqlError When the text is no JSON document, or nests too deeply
215     */
216    public static function parse(string $text, int $position, string $function): JsonNode
217    {
218        try {
219            return JsonNode::parse($text);
220        } catch (JsonSyntax $failure) {
221            $error = DataError::InvalidJsonTextInParameter->message($position, $function, $failure->reason, $failure->position);
222            if ($failure->deep) {
223                throw new SqlError(DataError::JsonDocumentTooDeep, DataError::JsonDocumentTooDeep->message(), $failure, [[DataError::InvalidJsonTextInParameter->value, $error]]);
224            }
225
226            throw new SqlError(DataError::InvalidJsonTextInParameter, $error, $failure);
227        }
228    }

248    /**
249     * Reads the path an argument of a function holds for the row of a frame; NULL is null.
250     *
251     * A binary string is refused (ER_INVALID_JSON_CHARSET); any other value is read as its text.
252     *
253     * @throws SqlError When the path is not valid, at the position the server names
254     */
255    public static function path(Evaluable $argument, Frame $frame): ?JsonPath
256    {
257        $value = $argument->evaluate($frame);
258        if ($value === null) {
259            return null;
260        }
261        $domain = $argument->domain();
262        $text = (string) Convert::toText($value, $domain);
263        if ($domain->kind === Kind::String) {
264            if ($domain->collation->charset === Charset::binary()) {
265                throw DataError::InvalidJsonCharset->error('binary');
266            }
267            $text = Encoding::convert($text, $domain->collation->charset, Charset::known('utf8mb4'));
268        }
269        try {
270            return JsonPath::parse($text);
271        } catch (JsonSyntax $failure) {
272            throw new SqlError(DataError::InvalidJsonPath, DataError::InvalidJsonPath->message($failure->position), $failure);
273        }
274    }

291    /**
292     * Converts a value that is not NULL to JSON as CAST(value AS JSON) does, and answers it as an SQL expression holds a JSON value.
293     *
294     * A string is read as a JSON text, its errors naming the function cast_as_json, except the
295     * value of an ENUM or SET, which is a JSON string; any other value is made a JSON value from
296     * its type ({@see self::argument()}).
297     * Source: https://dev.mysql.com/doc/refman/8.4/en/json.html ("Converting between JSON and non-JSON values").
298     *
299     * @throws SqlError When a string is no JSON text
300     */
301    public static function cast(int|float|string $value, Evaluable $operand): string
302    {
303        $domain = $operand->domain();
304        if ($domain->kind === Kind::String && $domain->collation->charset !== Charset::binary() && $domain->field !== Field::Enum && $domain->field !== Field::Set) {
305            return self::parse(Encoding::convert((string) $value, $domain->collation->charset, Charset::known('utf8mb4')), 1, 'cast_as_json')->store();
306        }
307
308        return self::value($value, $domain, $operand instanceof Predicate)->store();
309    }

311    /**
312     * Makes the JSON value of an SQL value that is not NULL.
313     *
314     * A number keeps its type, an unsigned integer and a YEAR being unsigned; a string is a JSON
315     * string; a binary string and a BIT value are opaque values, written as `base64:typeN:` and
316     * their bytes in base64, N being the type of the field; a temporal value keeps its type, a
317     * datetime and a time written with six decimals; a JSON value is itself (verified on a live 8.4
318     * server).
319     *
320     * @param bool $boolean Whether the value is that of a predicate
321     */
322    public static function value(int|float|string $value, Domain $domain, bool $boolean): JsonNode
323    {
324        if ($boolean) {
325            return new JsonNode(JsonKind::Boolean, (int) $value !== 0);
326        }
327        $text = (string) Convert::toText($value, $domain);
328
329        return match ($domain->kind) {
330            Kind::Integer => new JsonNode($domain->unsigned ? JsonKind::Unsigned : JsonKind::Integer, $text),
331            Kind::Year => new JsonNode(JsonKind::Unsigned, $text),
332            Kind::Double => new JsonNode(JsonKind::Double, (float) $value),
333            Kind::Decimal => new JsonNode(JsonKind::Decimal, $text),
334            Kind::String => $domain->collation->charset === Charset::binary() ? self::opaque($text, self::stored($domain)) : new JsonNode(JsonKind::String, Encoding::convert($text, $domain->collation->charset, Charset::known('utf8mb4'))),
335            Kind::Date => new JsonNode(JsonKind::Date, $text),
336            Kind::Time => new JsonNode(JsonKind::Time, self::time($text)),
337            Kind::DateTime => new JsonNode($domain->field === Field::Timestamp ? JsonKind::Timestamp : JsonKind::DateTime, self::dateTime($text)),
338            Kind::Json => JsonNode::load((string) $value),
339            Kind::Bit => self::opaque($text, Field::Bit),
340            Kind::Null => new JsonNode(JsonKind::Null),
341        };
342    }

344    /**
345     * Answers the type the server stores a binary string of a domain as: a BLOB by its length, else its field.
346     *
347     * @example A TINYBLOB
348     *     \MySqlMemory\Evaluation\Compile\Family\Jsons::stored(new \MySqlMemory\Typing\Domain(\SqlSemantics\Platform\MySql\Statement\Type\Resolved\Kind::String, \SqlSemantics\Platform\MySql\Statement\Type\Resolved\Field::Blob, 255)) // => \SqlSemantics\Platform\MySql\Statement\Type\Resolved\Field::TinyBlob
349     */
350    public static function stored(Domain $domain): Field
351    {
352        if ($domain->field !== Field::Blob) {
353            return $domain->field;
354        }
355
356        return match (true) {
357            $domain->length <= 255 => Field::TinyBlob,
358            $domain->length <= 65535 => Field::Blob,
359            $domain->length <= 16777215 => Field::MediumBlob,
360            default => Field::LongBlob,
361        };
362    }

364    /**
365     * Makes the opaque value of the bytes of a field: `base64:typeN:` and the bytes in base64 over lines of 76 characters, N the type the server stores the field as.
366     *
367     * @example The bytes of a VARBINARY value
368     *     \MySqlMemory\Evaluation\Compile\Family\Jsons::opaque('ab', \SqlSemantics\Platform\MySql\Statement\Type\Resolved\Field::VarString)->text() // => '"base64:type15:YWI="'
369     */
370    public static function opaque(string $bytes, Field $field): JsonNode
371    {
372        $type = match ($field) {
373            Field::VarString, Field::VarChar => 15,
374            Field::Bit => 16,
375            Field::Decimal, Field::Tiny, Field::Short, Field::Long, Field::Float, Field::Double, Field::Null, Field::Timestamp, Field::LongLong, Field::Int24, Field::Date, Field::Time, Field::DateTime, Field::Year, Field::NewDate,
376            Field::Vector, Field::Json, Field::NewDecimal, Field::Enum, Field::Set, Field::TinyBlob, Field::MediumBlob, Field::LongBlob, Field::Blob, Field::String, Field::Geometry => $field->value,
377        };
378
379        return new JsonNode(JsonKind::Opaque, 'base64:type' . $type . ':' . rtrim(chunk_split(base64_encode($bytes), 76, "\n"), "\n"));
380    }
