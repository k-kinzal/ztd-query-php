208    /**
209     * Compiles an expression in the scope of its query block.
210     *
211     * @throws \MySqlMemory\Error\SqlError When the expression is refused
212     */
213    public function compile(Scalar $node, Scope $scope): Evaluable
214    {
215        $bound = $scope->bound($node);
216        if ($bound !== null) {
217            return $bound[0] === 0 ? $bound[1] : $this->names->outer($bound[1], $bound[0]);
218        }
219
220        $replacement = $this->facts->covers($node) ? $this->facts->scalar($node)->replacement : null;
221
222        return $this->typed($node, $replacement === null ? $this->dispatch($node, $scope) : $this->compile($replacement, $scope));
223    }

225    /**
226     * Answers the type SQL Semantics resolved for a node, with the nullability it derived, or null when it resolved only the class of the type.
227     */
228    public function resolved(Scalar $node): ?Domain
229    {
230        if (!$this->facts->covers($node)) {
231            return null;
232        }
233        $fact = $this->facts->scalar($node);
234        $type = $fact->type;
235        if ($type instanceof NullOnly) {
236            return Domain::null();
237        }
238        if (!$type instanceof Known || !$type->descriptor instanceof Resolved) {
239            return null;
240        }
241
242        return Domain::of($type->descriptor, $fact->nullability !== Nullability::NotNull);
243    }

255    /**
256     * Gives a compiled expression the type SQL Semantics resolved for its node, when it resolved one.
257     */
258    public function typed(Scalar $node, Evaluable $evaluable): Evaluable
259    {
260        if (!$this->facts->covers($node)) {
261            return $evaluable;
262        }
263        $fact = $this->facts->scalar($node);
264        $nullable = match ($fact->nullability) {
265            Nullability::NotNull => false,
266            Nullability::Nullable => true,
267            Nullability::Dependent => $evaluable->domain()->nullable,
268        };
269        $type = $fact->type;
270        $domain = $type instanceof Known && $type->descriptor instanceof Resolved ? Domain::of($type->descriptor, $nullable)->withNumericBytes($evaluable->domain()->numericBytes && $type->descriptor->kind === $evaluable->domain()->kind) : $evaluable->domain()->withNullable($nullable);
271        if ($domain->kind === Kind::String && $this->settings->legacy()) {
272            $domain = ($node instanceof SystemVariable ? $evaluable->domain()->withNullable($nullable) : $domain)->withQuiet(!(new Quietness())->loud($node));
273        } elseif ($domain->kind === Kind::String && $node instanceof FunctionCall && in_array(strtoupper($node->name->value), Calls::QUIET, true)) {
274            $domain = $domain->withQuiet();
275        }
276
277        if ($domain->kind === Kind::Json) {
278            $domain = $domain->withSource($evaluable->domain()->source);
279        }
280
281        return $domain === $evaluable->domain() ? $evaluable : new Retyped($evaluable, $domain);
282    }
