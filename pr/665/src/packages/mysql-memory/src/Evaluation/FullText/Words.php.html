117    /**
118     * @return list<string> Unicode words, including words excluded from the index
119     */
120    public function split(string $text, Charset $charset): array
121    {
122        $text = Encoding::convert($text, $charset, Charset::known('utf8mb4'));
123        preg_match_all("/[\\p{L}\\p{N}_]+(?:'[\\p{L}\\p{N}_]+)*/u", $text, $matches);
124        return $matches[0];
125    }

127    /**
128     * Tests index length and stopword constraints before collation weighting.
129     */
130    public function indexed(string $word): bool
131    {
132        $length = mb_strlen($word, 'UTF-8');
133        return $length >= $this->minimum && $length <= $this->maximum && !in_array(mb_strtolower($word, 'UTF-8'), $this->stopwords, true);
134    }

141        return Ordering::of($this->collation)->key(Encoding::convert($word, Charset::known('utf8mb4'), $this->collation->charset));

144    /**
145     * @param list<string> $words Decoded document words in position order
146     * @return array<string, int> Indexed word frequencies

148    public function frequencies(array $words): array
149    {
150        $frequencies = [];
151        foreach ($words as $word) {
152            if ($this->indexed($word)) {
153                $key = $this->key($word);
154                $frequencies[$key] = ($frequencies[$key] ?? 0) + 1;

156        }
