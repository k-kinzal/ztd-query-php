145     * Every generated structure reaches the executor and leaves its scopes balanced.
146     * This does not establish that a returned SQL error or successful result is correct.

151    #[Test]
152    #[DataProvider('providerStatements')]
153    public function satisfiesGeneratedStatementsUnwindTheirExecutionScopes(Generator $statements, GrammarRelease $release = GrammarRelease::MySql847, ?Generator $contexts = null): void
154    {
155        $this->forAll(new AnalyzedStatementGenerator($release, $statements, $contexts))->withMaxSize(10)->then(static function (Statement $statement) use ($release): void {
156            $session = (new Instance(substr($release->value, 6), databases: ['pbt']))->connect(database: 'pbt');
157            $session->text = 'Semantic input supplied directly, without source SQL';
158            $execution = new Execution($session);
159            try {
160                $reply = $execution->statement($statement);
161                self::assertTrue($reply instanceof ResultSet || $reply instanceof Completion);
162                if ($reply instanceof ResultSet) {
163                    foreach ($reply->rows as $row) {
164                        self::assertCount(count($reply->columns), $row);
165                    }
166                } else {
167                    self::assertGreaterThanOrEqual(0, $reply->affectedRows);
168                }
169                self::assertGreaterThanOrEqual(0, $reply->warnings());
170            } catch (SqlError $error) {
171                $execution->failed($error);
172                self::assertGreaterThan(0, $error->getCode());
173                self::assertMatchesRegularExpression('/^[A-Z0-9]{5}$/D', $error->sqlState());
174                self::assertNotSame('', $error->getMessage());
175                self::assertSame(-1, $session->variables->rowCount);
176                self::assertGreaterThan(0, $session->diagnostics->errors());
177            }
178            self::assertSame([], $session->transaction->statements->running);
179            self::assertSame([], $session->running->using);
180        });
181    }
