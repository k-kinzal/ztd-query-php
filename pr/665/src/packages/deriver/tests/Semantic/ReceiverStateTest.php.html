44     * @param string $symbol Entry callable

49    #[DataProvider('suppliedObjects')]
50    public function testSuppliedObjectsReadTheirDeclaredPropertyTypes(string $symbol, array $arguments, string $expected): void
51    {
52        $session = Analysis::session(self::LAYERS);
53        $entry = $session->derive(new ReturnQuery($symbol, QueryScope::fromEntrypoints([new EntryPoint($symbol, $arguments)])));
54        $symbolic = $session->derive(new ReturnQuery($symbol));
55        foreach ([$entry, $symbolic] as $result) {
56            self::assertEqualsCanonicalizing([$expected], array_values(array_unique(array_map(static fn (Alternative $outcome) => $outcome->values['return']->native(), $result->normalOutcomes), SORT_REGULAR)));
57            self::assertSame([], $result->frontiers);
58            self::assertNotEmpty($result->exceptionalOutcomes);
59            self::assertSame(['Error'], array_values(array_unique(array_map(static fn (Exceptional $outcome) => $outcome->exception->literal, $result->exceptionalOutcomes))));
60        }
61        self::assertCount(count($symbolic->exceptionalOutcomes), $entry->exceptionalOutcomes);
