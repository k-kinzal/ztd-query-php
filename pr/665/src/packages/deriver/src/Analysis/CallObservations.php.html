55    /**
56     * Collects definitions before looking at calls, independent of block allocation order.
57     * @param CallableGraph $callable Captured owner
58     * @param string $selector Requested function or method name, `Class::__construct`, or `*`
59     * @param CallResolution $resolution Captured namespace resolution
60     * @return list<Observation> Matching occurrences
61     */
62    public function within(CallableGraph $callable, string $selector, CallResolution $resolution): array
63    {
64        $constants = [];
65        $sources = [];
66        $calls = [];
67        foreach ($callable->blocks as $block) {
68            foreach ($block->instructions as $instruction) {
69                $sources[$instruction->result] = $instruction->source;
70                if ($instruction->constant !== null) {
71                    $constants[$instruction->result] = $instruction->constant;
72                }
73                if (in_array($instruction->operation, ['invoke', 'invoke-method', 'invoke-static', 'new'], true)) {
74                    $calls[] = $instruction;
75                }
76            }
77        }
78        $result = [];
79        $identity = new CallableIdentity();
80        foreach ($calls as $instruction) {
81            $index = in_array($instruction->operation, ['invoke', 'new'], true) ? 0 : 1;
82            $name = $constants[$instruction->operands[$index] ?? '']->literal ?? null;
83            if (!is_string($name)) {
84                if ($selector === '*' && $instruction->operation !== 'new') {
85                    $result[] = $this->observation($callable->symbol, $instruction, '', $sources);
86                }
87                continue;
88            }
89            $name = match ($instruction->operation) {
90                'invoke' => $resolution->name($name, $instruction),
91                'new' => $this->className($name, $instruction),
92                default => $name,
93            };
94            $selected = $instruction->operation === 'new' ? $name . '::__construct' : $name;
95            if ($selector === '*' || $identity->key($selected) === $identity->key($selector)) {
96                $result[] = $this->observation($callable->symbol, $instruction, $name, $sources);
97            }
98        }
99        return $result;
100    }

104     * @param string $name Created class as compiled
105     * @param Instruction $instruction Creation with its lexical class scope

108    public function className(string $name, Instruction $instruction): string
109    {
110        $scope = $instruction->attributes['scope'] ?? '';
111        if (($instruction->attributes['literal-class'] ?? false) !== true || !is_string($scope) || !in_array(strtolower($name), ['self', 'parent'], true)) {

114        $resolved = (new Dispatch($this->program))->className($name, $scope, '');

120     * @param string $owner Declaring callable

123     * @param array<string, SourceRef> $sources Register source locations

126    public function observation(string $owner, Instruction $instruction, string $name, array $sources): Observation
127    {
128        $arguments = [];
129        foreach ($instruction->arguments as $position => $argument) {
130            $reference = new ExpressionRef($sources[$argument->register] ?? $instruction->source, $owner, $argument->register);
131            $arguments[$position] = $reference;
132            if ($argument->name !== null) {
133                $arguments[$argument->name] = $reference;
134            }
135        }
136        $receiver = in_array($instruction->operation, ['invoke', 'new'], true) ? null : new ExpressionRef($sources[$instruction->operands[0]] ?? $instruction->source, $owner, $instruction->operands[0]);
137        return new Observation($instruction->source, $owner, $instruction->id, $name, $arguments, new ExpressionRef($instruction->source, $owner, $instruction->result), $receiver, $instruction->operation);
