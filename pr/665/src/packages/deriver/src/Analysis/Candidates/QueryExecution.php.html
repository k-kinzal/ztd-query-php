76    {
77        $frames = [];
78        foreach ($query->scope()->entries as $entry) {
79            if (strcasecmp($entry->symbol, $graph->body->symbol) !== 0) {
80                continue;
81            }
82            $bindings = $entry->captures;
83            foreach ($entry->arguments as $name => $value) {
84                $parameter = is_string($name) ? $name : ($graph->body->parameters[$name]->name ?? (string) $name);
85                $bindings[$parameter] = $value;
86            }
87            if ($entry->receiver !== null) {
88                $bindings['this'] = $entry->receiver;
89            }
90            $identity = 'entry:' . hash('sha256', serialize([$entry->symbol, $bindings, $entry->properties]));
91            $frames[] = new Frame($graph, $identity, $bindings, $entry->properties, invocation: !$entry->symbolicArguments);
92        }
93        return $frames === [] ? [new Frame($graph, 'source:' . $graph->body->symbol)] : $frames;
