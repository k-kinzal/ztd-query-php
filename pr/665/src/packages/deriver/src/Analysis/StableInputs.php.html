12 * Proves immutable entry bindings before using declared types in interrupted observations.

17    /**
18     * A parameter type constrains entry only: any write, escape, or unknown symbol-table effect defeats this proof.
19     * @param CallableGraph $body Observation owner, completely indexed within the recovery limit
20     * @param array<string, Instruction> $definitions Complete register definitions
21     * @param string $reason Interruption cause
22     * @return array<string, Term> Typed residuals for bindings that cannot change
23     */
24    public function recover(CallableGraph $body, array $definitions, string $reason): array
25    {
26        if (str_starts_with($body->symbol, 'script:')) {
27            return [];
28        }
29        $types = [];
30        foreach ($body->parameters as $parameter) {
31            if (!$parameter->byReference && preg_match('/\b(self|parent|static)\b/', $parameter->type) !== 1) {
32                $types[$parameter->name] = $this->type($parameter);
33            }
34        }
35        if (!$body->static && $body->className !== '') {
36            $types['this'] = $body->className;
37        }
38        foreach ($body->regions as $region) {
39            foreach ($region->catches as $catch) {
40                unset($types[$catch->variable]);
41            }
42        }
43        foreach ($definitions as $instruction) {
44            if (in_array($instruction->operation, ['dynamic-local', 'symbol-table-boundary', 'unsupported', 'closure'], true) || $this->symbolEffect($instruction, $definitions)) {
45                return [];
46            }
47            foreach ($instruction->operands as $operand) {
48                $local = $definitions[$operand] ?? null;
49                if ($local?->operation === 'local' && !in_array($instruction->operation, ['read', 'read-silent'], true)) {
50                    unset($types[$local->name]);
51                }
52            }
53        }
54        return array_map(static fn (string $type): Term => Term::opaque($reason, $type), $types);
55    }

58     * Keeps implicit nullable defaults in a PHP 8.3 entry bound.
59     * @param \Deriver\ControlFlow\Parameter $parameter Captured signature parameter
60     * @return string Conservative entry type
61     */
62    public function type(\Deriver\ControlFlow\Parameter $parameter): string
63    {
64        if ($parameter->variadic) {
65            return 'array';
66        }
67        $default = $parameter->default->blocks[0]->instructions[0] ?? null;
68        $nullable = $default?->operation === 'constant-fetch' && strtolower($default->name) === 'null';
69        return $nullable && $parameter->type !== 'mixed' && !in_array('null', explode('|', $parameter->type), true) ? $parameter->type . '|null' : $parameter->type;
70    }

72    /**
73     * Dynamic function calls may resolve to extract(), which changes the caller's symbols.
74     * @param Instruction $instruction Candidate invocation
75     * @param array<string, Instruction> $definitions Captured register definitions
76     * @return bool Whether a caller symbol-table effect is possible
77     */
78    public function symbolEffect(Instruction $instruction, array $definitions): bool
79    {
80        if ($instruction->operation !== 'invoke') {
81            return false;
82        }
83        $name = $definitions[$instruction->operands[0] ?? '']->constant->literal ?? null;
84        return !is_string($name) || strtolower(substr($name, (int) strrpos('\\' . $name, '\\'))) === 'extract';
85    }
