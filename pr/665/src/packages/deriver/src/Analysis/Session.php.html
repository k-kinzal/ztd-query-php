187    #[Override]
188    public function derive(Query $query): CandidateCollection
189    {
190        $key = (new QueryEncoding())->key($query);
191        $cached = ($this->cache[$key] ?? null)?->get();
192        if ($cached !== null && $this->configuration->resources->cancellation?->isRequested() !== true) {
193            return $cached;
194        }
195        $result = (new Candidates\QueryExecution($this->candidateIndex, $this->configuration, $this->models, $this->manifest, $this->candidateCache))->derive($query);
196        $this->remember($result);
197        if (!$result->interrupted) {
198            $this->cache[$key] = WeakReference::create($result);
199        }
200        return $result;
201    }

252    /**
253     * Looks up a result's explanation.
254     * @param ResultRef $result Session result reference
255     * @return list<\Deriver\Result\Evidence\Alternative> Candidate derivations
256     * @throws InvalidInputException If the result is absent from this session
257     */
258    #[Override]
259    public function explain(ResultRef $result): array
260    {
261        $derived = ($this->results[$result->id] ?? null)?->get();
262        if ($derived === null) {
263            foreach ($this->liveResults as $live => $_) {
264                if ($live->reference->id === $result->id) {
265                    $derived = $live;
266                    break;
267                }
268            }
269        }
270        $derived ??= throw new InvalidInputException('Unknown or released result reference for this session; retain the CandidateCollection while using explain().');
271        return array_merge(...array_map(static fn ($candidate): array => $candidate->evidence, $derived->candidates));
272    }

310    #[Override]

312    {
313        $result = [];
