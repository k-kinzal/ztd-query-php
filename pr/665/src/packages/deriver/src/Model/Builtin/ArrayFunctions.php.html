101        $mode = $values[1] ?? Term::constant(0);
102        if ($mode->kind !== 'constant') {
103            return Term::opaque('UNSUPPORTED_MODEL_CASE', 'int', $values);
104        }
105        if (!in_array($mode->literal, [0, 1], true)) {

124    {
125        $count = count($array->operands);
126        if ($recursive) {

128                if ($element->kind === 'array' && ($element->attributes['open'] ?? false) === false) {
129                    $nested = $this->count($element, true);
130                    if (!is_int($nested->literal)) {

132                    }
133                    $count += $nested->literal;

143     * Evaluates membership while retaining uncertainty about unknown array parts, including sources merged after a known head.
144     * @param string $name Membership function
145     * @param list<Term> $values Bound arguments
146     * @return Term Boolean predicate
147     */
148    public function membership(string $name, array $values): Term
149    {
150        $needle = $values[0] ?? Term::constant(null);
151        $array = $values[1] ?? Term::constant(null);
152        $head = (new Arrays())->head($array);
153        if ($head !== null) {
154            $found = $this->membership($name, [$needle, new Term('array', operands: $head->operands, attributes: ['open' => true], secret: $array->isSecret()), ...array_slice($values, 2)]);
155            return $found->kind === 'constant' && $found->literal === true ? $found : new Term('intrinsic', $name, $values, ['type' => 'bool']);
156        }
157        if ($array->kind !== 'array') {
158            return new Term('intrinsic', $name, $values, ['type' => 'bool']);
159        }
160        if ($name === 'array_key_exists') {
161            return $this->keyExists($needle, $array);
162        }
163        if (($values[2]->kind ?? 'constant') !== 'constant') {
164            return new Term('intrinsic', $name, $values, ['type' => 'bool']);
165        }
166        $unknown = false;
167        foreach ($array->operands as $element) {
168            $comparison = (new Comparison($this->floatPrecision))->apply(($values[2]->literal ?? false) === true ? '===' : '==', $needle, $element);
169            if ($comparison->kind === 'constant' && $comparison->literal === true) {
170                return Term::constant(true);
171            }
172            $unknown = $unknown || $comparison->kind !== 'constant';
173        }
174        if (!$unknown && ($array->attributes['open'] ?? false) === false) {
175            return Term::constant(false);
176        }
177
178        return new Term('intrinsic', $name, $values, ['type' => 'bool']);
179    }

181    /**
182     * Selects array keys with the target strict or loose comparison semantics.
183     * @param Term $array Closed array shape
184     * @param list<Term> $values Bound filter arguments
185     * @return Term Ordered selected keys
186     */
187    public function keys(Term $array, array $values): Term
188    {
189        $filter = $values[1] ?? new Term('omitted');
190        if ($filter->kind === 'omitted') {
191            return Term::array(array_map(static fn (int|string $key): Term => Term::constant($key), array_keys($array->operands)));
192        }
193        $strict = $values[2] ?? Term::constant(false);
194        if ($strict->kind !== 'constant') {
195            return Term::opaque('UNSUPPORTED_MODEL_CASE', 'array', $values);
196        }
197        $keys = [];
198        foreach ($array->operands as $key => $element) {
199            $equal = (new Comparison($this->floatPrecision))->apply($strict->literal === true ? '===' : '==', $filter, $element);
200            if ($equal->kind !== 'constant') {
201                return Term::opaque('UNSUPPORTED_MODEL_CASE', 'array', $values);
202            }
203            if ($equal->literal === true) {
204                $keys[] = Term::constant($key);
205            }
206        }
207        return Term::array($keys);
208    }

212     * @param Term $needle Key value
213     * @param Term $array Known array shape
214     * @return Term Presence predicate

216    public function keyExists(Term $needle, Term $array): Term
217    {
218        $key = (new Operations())->arrayKey($needle);
219        if ($key->kind === 'throwable') {
220            return $key;
221        }
222        if ($key->kind === 'constant' && (is_int($key->literal) || is_string($key->literal))) {
223            if (array_key_exists($key->literal, $array->operands) && ($array->operands[$key->literal]->attributes['maybeUninitialized'] ?? false) !== true) {
224                return Term::constant(true);
225            }
226            if (($array->attributes['open'] ?? false) === false) {
227                return Term::constant(false);
228            }
229        }
230        return new Term('intrinsic', 'array_key_exists', [$needle, $array], ['type' => 'bool']);
