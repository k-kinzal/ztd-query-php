15 * Reconstructs shared immutable terms from a version 1 report's value graph.
16 *
17 * Redacted records become confidential opaque terms; their missing payload is never
18 * mistaken for a known null. This reader validates value records and references.

50     * Captures and validates every value record, including unreferenced records.
51     * @param string $json UTF-8 JSON report bytes
52     * @param int $maximumNodes Maximum serialized value records
53     * @param int $maximumDepth Maximum value graph depth
54     * @return self Validated shared value graph
55     * @throws JsonException If the document is not valid JSON
56     * @throws InvalidInputException If the version, limits, records, or references are invalid
57     */
58    public static function fromJson(string $json, int $maximumNodes = 20000, int $maximumDepth = 128): self
59    {
60        if ($maximumNodes < 1 || $maximumDepth < 1 || $maximumDepth > 512 || strlen($json) > 33554432) {
61            throw new InvalidInputException('Invalid value-reader limits or report larger than 32 MiB.');
62        }
63        $document = json_decode($json, false, 512, JSON_THROW_ON_ERROR);
64        if (!$document instanceof stdClass || (new Fields())->text($document, 'schemaVersion') !== '1') {
65            throw new InvalidInputException('Unsupported Deriver report schema version.');
66        }
67        $records = (new Fields())->object($document, 'values');
68        if (count(get_object_vars($records)) > $maximumNodes) {
69            throw new InvalidInputException('The serialized value graph exceeds its node limit.');
70        }
71        $reader = new self($records, $maximumDepth);
72        foreach (get_object_vars($records) as $id => $record) {
73            if (!is_string($id) || preg_match('/\Av[0-9]+\z/', $id) !== 1 || !$record instanceof stdClass) {
74                throw new InvalidInputException('Invalid value table identifier or record.');
75            }
76            $reader->read($id);
77        }
78        return $reader;
79    }

81    /**
82     * Returns one interned term and resolves its ordered dependencies.
83     * @param string $id Serialized value identifier
84     * @return Term Shared immutable term
85     * @throws InvalidInputException If the reference is missing, cyclic, or too deep
86     */
87    public function read(string $id): Term
88    {
89        if (isset($this->decoded[$id])) {
90            return $this->decoded[$id];
91        }
92        if (isset($this->active[$id]) || count($this->active) >= $this->maximumDepth) {
93            throw new InvalidInputException('Cyclic or excessively deep value graph.');
94        }
95        $record = (new Fields())->object($this->records, $id);
96        $this->active[$id] = true;
97        $kind = (new JsonText())->decode((new Fields())->text($record, 'kind'));
98        if ($kind === '') {
99            throw new InvalidInputException('Value kinds must not be empty.');
100        }
101        $redacted = (new Fields())->boolean($record, 'redacted');
102        $secret = (new Fields())->boolean($record, 'secret');
103        $operands = $this->operands($record);
104        $attributes = $this->attributes($record);
105        $literal = (new Scalar())->read((new Fields())->object($record, 'literal'));
106        if ($redacted && ($literal !== null || $operands !== [] || $attributes !== [])) {
107            throw new InvalidInputException('A redacted record must not contain a payload.');
108        }
109        $term = $redacted ? new Term('opaque', 'REDACTED', attributes: ['redactedKind' => $kind, 'type' => 'mixed'], secret: true) : new Term($kind, $literal, $operands, $attributes, $secret);
110        $depth = 1;
111        foreach ((new Fields())->objects($record, 'operands') as $operand) {
112            $depth = max($depth, 1 + $this->depths[(new Fields())->text($operand, 'value')]);
113        }
114        if ($depth > $this->maximumDepth) {
115            throw new InvalidInputException('The value graph exceeds its depth limit.');
116        }
117        $this->depths[$id] = $depth;
118        unset($this->active[$id]);
119        return $this->decoded[$id] = $term;
120    }
