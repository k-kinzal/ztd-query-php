48     * Retains a fresh bounded random event and every applicable target exception.
49     * @param Instruction $instruction Random intrinsic
50     * @param State $state Current path
51     * @param list<Term> $values Bound range endpoints
52     * @return list<State> Normal sample, range error, and entropy-source failure
53     */
54    public function random(Instruction $instruction, State $state, array $values): array
55    {
56        $error = $this->rangeError($values, $instruction->name);
57        if ($error !== null) {
58            $state->completion = new Completion('throw', new Term('throwable', $error));
59            return [$state];
60        }
61        $min = $values[0] ?? new Term('omitted');
62        $max = $values[1] ?? new Term('omitted');
63        if ($min->kind === 'constant' && $max->kind === 'constant' && is_int($min->literal) && $min->literal === $max->literal) {
64            $state->registers[$instruction->result] = Term::constant($min->literal, $min->isSecret() || $max->isSecret());
65            return [$state];
66        }
67        $state->registers[$instruction->result] = new Term('external', $instruction->name . ':' . $state->memory->fresh('event'), $values, ['type' => 'int', 'source' => $instruction->name, 'stability' => 'evaluation']);
