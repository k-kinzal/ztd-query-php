32    /**
33     * Transfers an operation on a previously evaluated property address.
34     * @param CallableGraph $caller Executing callable
35     * @param Instruction $instruction Read or mutation
36     * @param State $state Current path
37     * @param PropertySlot $slot Property declaration context
38     * @return list<State> Normal and exceptional alternatives
39     */
40    public function apply(CallableGraph $caller, Instruction $instruction, State $state, PropertySlot $slot): array
41    {
42        $address = $state->addresses[$instruction->operands[0]];
43        $property = $slot->declaration;
44        $boundary = (new PropertyAccessCheck($this->machine))->apply($instruction, $state, $slot, $address);
45        if ($boundary !== null) {
46            return $boundary;
47        }
48        $accessible = (new PropertyLookup($this->machine->context->program))->accessible($property, $slot->scope);
49        if (!$accessible || $property === null) {
50            $magic = (new PropertyMagic($this->machine))->apply($caller, $instruction, $state, $slot);
51            if ($magic !== null) {
52                return $magic;
53            }
54        }
55        if (!$accessible && $instruction->operation === 'read-silent') {
56            $state->registers[$instruction->result] = Term::constant(null);
57            return [$state];
58        }
59        if (!$accessible || $slot->static && ($property === null || !$property->static)) {
60            return [$this->error($state, 'Error')];
61        }
62        if ($slot->static && (!isset($state->memory->cells[$address->root]) || $state->memory->cells[$address->root]->kind === 'array' && ($state->memory->cells[$address->root]->attributes['open'] ?? false) !== true && !isset($state->memory->cells[$address->root]->operands[$address->path[0]]))) {
63            $base = isset($state->offsets[$instruction->operands[0]]) ? (new Path($this->machine->context))->chain($state, $instruction->operands[0])['base'] : $address;
64            return $this->initialize($caller, $instruction, $state, $slot, $base);
65        }
66        return $this->uncertain($caller, $instruction, $state, $slot, $address) ?? $this->transfer($caller, $instruction, $state, $slot, $address);
67    }

69    /**
70     * Applies an operation after resolving access, default initialization, and receiver checks.
71     * @param CallableGraph $caller Executing graph
72     * @param Instruction $instruction Storage operation
73     * @param State $state Accessible path
74     * @param PropertySlot $slot Property context
75     * @param Location $address Property storage
76     * @return list<State> Normal or exceptional alternatives
77     */
78    public function transfer(CallableGraph $caller, Instruction $instruction, State $state, PropertySlot $slot, Location $address): array
79    {
80        if (isset($state->offsets[$instruction->operands[0]])) {
81            return (new Transfer($this->machine))->apply($caller, $instruction, $state);
82        }
83        $property = $slot->declaration;
84        if ($instruction->operation === 'read' && $property !== null && $property->type !== 'mixed' && $state->memory->read($address)->kind === 'uninitialized') {
85            return [$this->error($state, 'Error')];
86        }
87        if ($instruction->operation === 'reference' && $property !== null) {
88            return (new PropertyReference($this->machine))->read($caller, $instruction, $state, $slot, $address);
89        }
90        if ($instruction->operation === 'alias' && $property !== null) {
91            return (new PropertyReference($this->machine))->bind($caller, $instruction, $state, $slot, $address);
92        }
93        if ($instruction->operation === 'increment' && $property !== null) {
94            return (new PropertyReference($this->machine))->increment($caller, $instruction, $state, $slot, $address);
95        }
96        if ($instruction->operation === 'write' && $property !== null) {
97            return $this->write($caller, $instruction, $state, $slot, $address);
98        }
99        $unsetUninitialized = $instruction->operation === 'unset' && $state->memory->read($address)->kind === 'uninitialized' && strcasecmp($slot->scope, $property->className ?? '') === 0;
100        if ($property?->readonly === true && !$unsetUninitialized && in_array($instruction->operation, ['increment', 'reference', 'alias', 'unset'], true)) {
101            return [$this->error($state, 'Error')];
102        }
103        $state->registers[$instruction->result] = (new MemoryStep($this->machine->context))->evaluate($caller, $instruction, $state);
104        return [$state];
105    }

107    /**
108     * Validates a property assignment before making its state change visible.
109     * @param CallableGraph $caller Executing callable
110     * @param Instruction $instruction Assignment
111     * @param State $state Input state
112     * @param PropertySlot $slot Declaration and scope
113     * @param Location $address Property storage
114     * @return list<State> Assignment result and possible type error
115     */
116    public function write(CallableGraph $caller, Instruction $instruction, State $state, PropertySlot $slot, Location $address): array
117    {
118        $property = $slot->declaration;
119        if ($property === null) {
120            return [$this->error($state, 'Error')];
121        }
122        $initialized = $state->memory->read($address)->kind !== 'uninitialized';
123        $cloneWrite = isset($state->memory->cloneWrites[$address->root][$address->path[0]]);
124        if ($property->readonly && ($initialized && !$cloneWrite || strcasecmp($slot->scope, $property->className) !== 0)) {
125            return [$this->error($state, 'Error')];
126        }
127        $value = $state->value($instruction->operands[1]);
128        $type = (new TypeBinding($this->machine->context))->scope($property->type, $property->className, $state->lateStaticClass);
129        $types = (new ReferenceConstraint())->find($state->memory, $address);
130        $check = (new ReferenceAssignment($this->machine->context))->check($value, [$type, ...$types], $caller->strict);
131        (new TypeBinding($this->machine->context))->report($check, $instruction->source, $state);
132        $exception = $state->fork();
133        $exception->completion = new Completion('throw', $check->exception());
134        if ($check->mustFail) {
135            return [$exception];
136        }
137        unset($state->memory->cloneWrites[$address->root][$address->path[0]]);
138        $state->memory->write($address, $check->value);
139        $state->registers[$instruction->result] = $check->value;
140        return $check->mayFail ? [$state, $exception] : [$state];
141    }

143    /**
144     * Evaluates a static property's default once in each memory state.
145     * @param CallableGraph $caller Executing callable
146     * @param Instruction $instruction Pending property access
147     * @param State $state Uninitialized static storage
148     * @param PropertySlot $slot Captured declaration
149     * @param Location $address Static property address
150     * @return list<State> Initialized access results
151     */
152    public function initialize(CallableGraph $caller, Instruction $instruction, State $state, PropertySlot $slot, Location $address): array
153    {
154        $property = $slot->declaration;
155        $initial = $property?->type === 'mixed' ? Term::constant(null) : new Term('uninitialized', attributes: ['type' => $property->type ?? 'mixed']);
156        $state->memory->write($address, $initial);
157        if ($property?->default === null) {
158            return $this->apply($caller, $instruction, $state, $slot);
159        }
160        $paths = [];
161        foreach ($this->machine->run($property->default, new State(clone $state->memory)) as $exit) {
162            $next = $state->fork();
163            $next->memory = $exit->memory;
164            if ($exit->completion->kind === 'throw') {
165                $next->completion = $exit->completion;
166                $paths[] = $next;
167                continue;
168            }
169            $next->memory->write($address, $exit->completion->value ?? $initial);
170            array_push($paths, ...$this->apply($caller, $instruction, $next, $slot));
171        }
172        return $paths;
173    }

187    /**
188     * Retains the possibility that an unknown call unset a declared property.
189     * @param CallableGraph $caller Executing graph
190     * @param Instruction $instruction Property operation
191     * @param State $state Input path
192     * @param PropertySlot $slot Property contract
193     * @param Location $address Property storage
194     * @return list<State>|null Initialized and absent alternatives, or a known presence
195     */
196    public function uncertain(CallableGraph $caller, Instruction $instruction, State $state, PropertySlot $slot, Location $address): ?array
197    {
198        $value = $state->memory->read($address);
199        $unknown = ($value->attributes['maybeUninitialized'] ?? false) === true || $value->kind === 'array-read' && ($state->memory->cells[$address->root]->kind ?? '') === 'opaque';
200        if (!$unknown || $slot->declaration === null) {
201            return null;
202        }
203        $absent = $state->fork();
204        $absent->memory->write($address, new Term('uninitialized'));
205        $attributes = $value->attributes;
206        unset($attributes['maybeUninitialized']);
207        $present = new Term('opaque', is_string($value->literal) ? $value->literal : 'UNKNOWN_PROPERTY', $value->operands, [...$attributes, 'type' => $slot->declaration->type], $value->secret);
208        $state->memory->write($address, $present);
209        return [...$this->transfer($caller, $instruction, $state, $slot, $address), ...$this->transfer($caller, $instruction, $absent, $slot, $address)];
210    }
