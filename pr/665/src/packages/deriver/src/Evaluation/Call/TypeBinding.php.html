29    /**
30     * Validates or coerces an argument under the calling file's strict_types mode.
31     * @param Term $value Actual argument
32     * @param string $type Declared PHP type
33     * @param bool $strict Calling-file scalar coercion mode
34     * @return TypeCheck Valid value and exceptional alternatives
35     */
36    public function check(Term $value, string $type, bool $strict): TypeCheck
37    {
38        if ($type === 'mixed' || $type === '' || $type === 'void') {
39            return new TypeCheck($value);
40        }
41        $types = explode('|', $type);
42        $accepted = $this->matching($value, $types);
43        if ($accepted !== null) {
44            return $accepted;
45        }
46        if (!$strict && in_array('string', $types, true) && $value->kind === 'object' && (new Dispatch($this->context->program))->method((string) ($value->attributes['class'] ?? ''), '__toString') !== null) {
47            return new TypeCheck(Term::opaque('UNSUPPORTED_LANGUAGE_FEATURE', 'string', [$value]), true, coercion: $value);
48        }
49        if (!$strict && $value->kind === 'constant' && $value->literal !== null) {
50            foreach ($this->preference($value, $types) as $scalar) {
51                if (in_array($scalar, $types, true) && $this->coercible($value, $scalar)) {
52                    return $this->coerce($value, $scalar);
53                }
54            }
55        }
56        if (in_array('callable', $types, true) && (new CallableCheck($this->context))->evaluate($value)->kind !== 'constant') {
57            return new TypeCheck($value, true, coercion: $value, operation: 'callable-validation');
58        }
59        if ($value->kind === 'constant' || in_array($value->kind, ['array', 'object', 'closure', 'enum'], true)) {
60            return new TypeCheck($value, true, true);
61        }
62        return new TypeCheck(new Term('type-refinement', $type, [$value], ['type' => $type]), true);
63    }

65    /**
66     * Recognizes values whose runtime type already satisfies one declaration type.
67     * @param Term $value Value
68     * @param string $type One union arm
69     * @return bool Whether acceptance is established
70     */
71    public function accepts(Term $value, string $type): bool
72    {
73        if ($type === 'callable' && (new CallableCheck($this->context))->evaluate($value)->literal === true) {
74            return true;
75        }
76        if ($type === 'object' && in_array($value->kind, ['object', 'closure', 'enum'], true)) {
77            return true;
78        }
79        if (str_contains($type, '&')) {
80            foreach (explode('&', $type) as $part) {
81                if (!$this->accepts($value, trim($part, '()'))) {
82                    return false;
83                }
84            }
85            return true;
86        }
87        if (($value->attributes['type'] ?? '') === $type) {
88            return true;
89        }
90        if ($value->kind === 'constant') {
91            return $this->scalar($value, $type);
92        }
93        if ($value->kind === 'array') {
94            return $type === 'array' || $type === 'iterable';
95        }
96        if ($value->kind === 'closure') {
97            return in_array($type, ['Closure', 'object', 'callable'], true);
98        }
99        $class = $value->attributes['class'] ?? null;
100        return in_array($value->kind, ['object', 'enum'], true) && is_string($class) && (new Dispatch($this->context->program))->subtype($class, $type);
101    }

104     * Checks the subset of weak scalar conversions with verified target behavior.
105     * @param Term $value Concrete scalar
106     * @param string $type Required scalar type
107     * @return bool Whether coercion is legal
108     */
109    public function coercible(Term $value, string $type): bool
110    {
111        if ($type === 'string' || $type === 'bool') {
112            return $value->literal !== null;
113        }
114        if (is_string($value->literal)) {
115            $number = (new NumericString())->parse($value->literal);
116            if ($number === null || is_int($number)) {
117                return $number !== null;
118            }
119        }
120        return $type !== 'int' || (is_finite((float) $value->literal) && (float) $value->literal >= -9223372036854775808.0 && (float) $value->literal < 9223372036854775808.0);
121    }

135     * Resolves relative class names in a union or intersection using the declaration owner.
136     * @param string $type Declared type expression
137     * @param string $class Lexical declaration class
138     * @param string $lateStaticClass Runtime binding for static return types
139     * @return string Normalized resolved type
140     */
141    public function scope(string $type, string $class, string $lateStaticClass): string
142    {
143        $arms = [];
144        foreach (explode('|', $type) as $arm) {
145            $parts = [];
146            foreach (explode('&', $arm) as $part) {
147                $parts[] = (new Dispatch($this->context->program))->className(trim($part, '()'), $class, $lateStaticClass);

149            $arms[] = implode('&', $parts);
150        }
151        return implode('|', $arms);

159    public function scalar(Term $value, string $type): bool
160    {
161        return match ($type) {
162            'int' => is_int($value->literal),
163            'float' => is_float($value->literal) || is_int($value->literal),
164            'string' => is_string($value->literal),
165            'bool' => is_bool($value->literal),
166            'true' => $value->literal === true,
167            'false' => $value->literal === false,
168            'null' => $value->literal === null,
169            default => false,
170        };
171    }

174     * Selects numeric-string union coercion without discarding its float representation.

179    public function preference(Term $value, array $types): array
180    {
181        $floating = is_string($value->literal) && is_float((new NumericString())->parse($value->literal));
182        return $floating && in_array('int', $types, true) && in_array('float', $types, true) ? ['float', 'int', 'string', 'bool'] : ['int', 'float', 'string', 'bool'];

186     * Retains precision-loss diagnostics from weak integer conversion.
187     * @param Term $value Accepted concrete scalar

191    public function coerce(Term $value, string $type): TypeCheck

193        $result = (new Operations($this->context->configuration->target->floatPrecision))->cast($type, $value);

199     * Records a checked coercion's target diagnostic at the operation that requested it.
200     * @param TypeCheck $check Applied type conversion
201     * @param SourceRef $source Argument, return, or property origin
202     * @param State|null $state State affected by an implicit user call
203     */
204    public function report(TypeCheck $check, SourceRef $source, ?State $state = null): void
205    {
206        if ($check->coercion !== null) {
207            $this->context->frontier('UNSUPPORTED_LANGUAGE_FEATURE', $source, $check->operation, [$check->coercion]);
208            if ($state !== null) {
209                (new Havoc())->call($state, [$check->coercion], [], 'UNSUPPORTED_LANGUAGE_FEATURE');
210            }
211        }
212        if ($check->diagnostic) {
213            $this->context->frontier('PHP_WARNING', $source, 'implicit-integer-precision-loss');
214        }
215    }

217    /**
218     * Preserves exact scalar types regardless of the order of union arms.
219     * @param Term $value Actual input
220     * @param list<string> $types Declared union arms
221     * @return TypeCheck|null Accepted value or no matching runtime type
222     */
223    public function matching(Term $value, array $types): ?TypeCheck
224    {
225        if ($value->kind === 'constant' && in_array(get_debug_type($value->literal), $types, true)) {
226            return new TypeCheck($value);
227        }
228        foreach ($types as $part) {
229            if ($this->accepts($value, $part)) {
230                return new TypeCheck($part === 'float' && $value->kind === 'constant' && is_int($value->literal) ? Term::constant((float) $value->literal, $value->isSecret()) : $value);
231            }
232        }
233        return null;
234    }
