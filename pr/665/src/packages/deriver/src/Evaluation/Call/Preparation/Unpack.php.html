32     * @param CallableGraph $caller Executing graph
33     * @param Instruction $instruction Unpacked argument
34     * @param State $state Evaluated array argument
35     * @param Target $target Selected signature
36     * @param int|null $position Next positional index
37     * @return list<State> Prepared arrays or inclusive unresolved unpack paths

39    public function apply(CallableGraph $caller, Instruction $instruction, State $state, Target $target, ?int $position): array
40    {
41        $array = $state->value($instruction->result);
42        if ($array->kind === 'constant' || $array->kind === 'closure' || $array->kind === 'enum') {
43            $state->completion = new Completion('throw', new Term('throwable', 'TypeError'));
44            return [$state];
45        }
46        if ($array->kind !== 'array' || ($array->attributes['open'] ?? false) === true) {
47            return $this->boundary($state, $instruction);
48        }
49        $source = $this->source($instruction, $state, $array);
50        $prepared = $state->memory->allocate(Term::array([]));
51        $state->addresses[$instruction->result] = $prepared;
52        unset($state->properties[$instruction->result]);
53        $paths = [$state];
54        foreach (array_keys($array->operands) as $key) {
55            if (!$this->machine->context->admit($instruction->source)) {
56                return $this->boundary($state, $instruction);
57            }
58            $mode = (new Modes($this->machine))->select($target, $position, is_string($key) ? $key : null);
59            $next = [];
60            foreach ($paths as $path) {
61                foreach ($mode === null ? [false, true] : [$mode] as $reference) {
62                    $next[] = $this->element($path->fork(), $source, $prepared, $key, $reference);
63                }
64            }
65            $paths = (new StateJoin($this->machine->context))->limit($next, $caller);
66            $position = is_int($key) && $position !== null ? $position + 1 : $position;
67        }
68        foreach ($paths as $path) {
69            $path->registers[$instruction->result] = $path->memory->read($prepared);
70        }
71        return $paths;
72    }
