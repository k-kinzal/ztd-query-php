32    /**
33     * Executes one cursor instruction.
34     * @param Instruction $instruction Iterator operation
35     * @param State $state Current path
36     * @return Term Result
37     */
38    public function evaluate(Instruction $instruction, State $state): Term
39    {
40        if ($instruction->operation === 'iterator') {
41            return $this->initialize($instruction, $state);
42        }
43        $iterator = $state->value($instruction->operands[0] ?? '');
44        $id = is_string($iterator->literal) ? $iterator->literal : '';
45        if ($instruction->operation === 'iterator-release') {
46            unset($state->memory->liveArrays[$id], $state->iterators[$id]);
47            return Term::constant(null);
48        }
49        $cursor = $state->iterators[$id] ?? new IteratorCursor(Term::opaque('UNKNOWN_ITERABLE'));
50        $array = $cursor->location === null ? $cursor->array : $state->memory->read($cursor->location);
51        $head = (new Arrays())->head($array);
52        if ($instruction->operation === 'iterate') {
53            return $this->advance($id, new IteratorCursor($array, $cursor->location, $cursor->position + 1), $head, $state);
54        }
55        $key = $this->key($id, $cursor, $array, $head, $state);
56        if ($instruction->operation === 'iterator-address' && $cursor->location !== null) {
57            $location = $key === null ? $this->unknownElement($state, $cursor->location, $array) : new Location($cursor->location->root, [...$cursor->location->path, $key]);
58            $state->addresses[$instruction->result] = $location;
59            return new Term('location', $location->root);
60        }
61        if ($key === null) {
62            return Term::opaque('UNKNOWN_ITERABLE');
63        }
64        if ($instruction->operation === 'iterator-key') {
65            return Term::constant($key, $array->secret);
66        }
67        return $state->memory->element($head ?? $array, $key, $head !== null && $array->isSecret());
68    }

70    /**
71     * Selects known suffix values at their possible iteration positions, retaining an unknown-prefix path.
72     * @param Instruction $instruction Iterator value read
73     * @param State $state Iteration path
74     * @return list<State>|null Refined suffix candidates or ordinary cursor evaluation
75     */
76    public function candidates(Instruction $instruction, State $state): ?array
77    {
78        $iterator = $state->value($instruction->operands[0] ?? '');
79        $cursor = is_string($iterator->literal) ? ($state->iterators[$iterator->literal] ?? null) : null;
80        $split = $cursor === null || $cursor->location !== null ? null : (new Arrays())->tail($cursor->array);
81        if ($split === null) {
82            return null;
83        }
84        [$prefix, $suffix] = $split;
85        $minimum = count((new Arrays())->head($prefix)->operands ?? []);
86        if ($cursor->position < $minimum) {
87            return null;
88        }
89        $results = [];
90        $remaining = $state->fork();
91        $count = new Term('intrinsic', 'count', [$prefix], ['type' => 'int']);
92        foreach (array_values($suffix->operands) as $position => $value) {
93            if ($cursor->position < $position || count($results) >= $this->context->query->budget()->partitions - 1) {
94                break;
95            }
96            if ($cursor->position - $position < $minimum) {
97                continue;
98            }
99            $test = new Term('binary', '===', [$count, Term::constant($cursor->position - $position)], ['type' => 'bool']);
100            $path = $remaining->fork();
101            if ((new Constraints($this->context))->assume($path, $test, true)) {
102                $resolved = $path->memory->dereference($value);
103                $path->registers[$instruction->result] = $cursor->array->isSecret() ? new Term($resolved->kind, $resolved->literal, $resolved->operands, $resolved->attributes, true) : $resolved;
104                $results[] = $path;
105            }
106            if (!(new Constraints($this->context))->assume($remaining, $test, false)) {
107                return $results;
108            }
109        }
110        $remaining->registers[$instruction->result] = Term::opaque('UNKNOWN_ITERABLE', dependencies: [$prefix]);
111        $results[] = $remaining;
112        return $results;
113    }

115    /**
116     * Selects the current key only where its position is certain: tracked buckets of a closed live array, or known leading entries of a snapshot.
117     * @param string $id Iterator identity
118     * @param IteratorCursor $cursor Current cursor
119     * @param Term $array Iterated value
120     * @param Term|null $head Known leading entries of a symbolic merge
121     * @param State $state Current path
122     * @return int|string|null Current key, or null when it is unknown
123     */
124    public function key(string $id, IteratorCursor $cursor, Term $array, ?Term $head, State $state): int|string|null
125    {
126        $closed = $array->kind === 'array' && ($array->attributes['open'] ?? false) === false;
127        if ($cursor->location !== null) {
128            return $closed ? $state->memory->liveArrays[$id]->current ?? null : null;
129        }
130        $entries = $head ?? $array;
131        if ($entries->kind !== 'array' || array_filter($entries->operands, static fn (Term $entry): bool => ($entry->attributes['maybeUninitialized'] ?? false) === true) !== []) {
132            return null;
133        }
134        return array_keys($entries->operands)[$cursor->position] ?? null;
135    }

137    /**
138     * Addresses an unknown entry of by-reference iterated storage, so every write through it invalidates that storage.
139     * @param State $state Current path
140     * @param Location $location Pinned iterated variable
141     * @param Term $subject Iterated value
142     * @return Location Unknown element of the array storage, or of the object's property storage
143     */
144    public function unknownElement(State $state, Location $location, Term $subject): Location
145    {
146        $array = new Location($location->root, $location->path, unknown: true);
147        if ((new TypePredicates())->apply('is_array', $subject)->literal === true) {
148            return $array;
149        }
150        $seen = [];
151        (new Havoc())->reachable($state, $subject, 'UNKNOWN_REFERENCE', $seen);
152        if ($subject->kind === 'object' || (new TypePredicates())->apply('is_object', $subject)->literal === true) {
153            return is_string($subject->literal) ? new Location('object:' . $subject->literal, unknown: true) : $array;
154        }
155        $state->memory->write($array, Term::opaque('UNKNOWN_REFERENCE', dependencies: [$subject]));
156        return is_string($subject->literal) && isset($state->memory->cells['object:' . $subject->literal]) ? new Location('object:' . $subject->literal, unknown: true) : $array;
157    }

159    /**
160     * Moves a cursor by one entry and reports whether that entry exists.
161     * @param string $id Iterator identity
162     * @param IteratorCursor $cursor Cursor at its new position
163     * @param Term|null $head Known leading entries of a symbolic merge
164     * @param State $state Current path
165     * @return Term Entry availability
166     */
167    public function advance(string $id, IteratorCursor $cursor, ?Term $head, State $state): Term
168    {
169        $state->iterators[$id] = $cursor;
170        $array = $cursor->array;
171        if ($head !== null && $cursor->position < count($head->operands)) {
172            return Term::constant(true, $array->secret);
173        }
174        if ($array->kind !== 'array' || ($array->attributes['open'] ?? false) === true) {
175            if (isset($state->memory->liveArrays[$id])) {
176                $state->memory->liveArrays[$id] = new LiveArray($state->memory->liveArrays[$id]->location, []);
177            }
178            return new Term('external', $id . ':has-next:' . $cursor->position, attributes: ['type' => 'bool'], secret: $array->secret);
179        }
180        if (isset($state->memory->liveArrays[$id])) {
181            $state->memory->liveArrays[$id] = $state->memory->liveArrays[$id]->advance();
182            return Term::constant($state->memory->liveArrays[$id]->current !== null, $array->secret);
183        }
184        return Term::constant($cursor->position < count($array->operands), $array->secret);
185    }

187    /**
188     * Pins a live reference array independently of subsequent local variable rebinding.
189     * @param Instruction $instruction Foreach entry
190     * @param State $state Current path
191     * @return Term Iterator identity
192     */
193    public function initialize(Instruction $instruction, State $state): Term
194    {
195        $id = $state->memory->fresh('iterator');
196        $register = $instruction->operands[0] ?? '';
197        $location = ($instruction->attributes['byReference'] ?? false) === true ? ($state->addresses[$register] ?? null) : null;
198        $array = $location === null ? $state->value($register) : $this->referenced($instruction, $state, $location);
199        if ($this->iterable($array) === false) {
200            $this->context->frontier('PHP_WARNING', $instruction->source, 'foreach-non-iterable');
201            $state->iterators[$id] = new IteratorCursor(new Term('array', attributes: ['open' => false], secret: $array->isSecret()));
202            return new Term('iterator', $id);
203        }
204        if ($location !== null) {
205            $location = new Location($state->memory->reference($location));
206            $array = $state->memory->read($location);
207            $state->memory->liveArrays[$id] = new LiveArray($location, $array->kind === 'array' ? array_keys($array->operands) : []);
208        }
209        $state->iterators[$id] = new IteratorCursor($array, $location);
210        return new Term('iterator', $id);
211    }

213    /**
214     * Reads a by-reference subject as PHP's write fetch does, without defining an undefined plain variable.
215     * @param Instruction $instruction Foreach entry
216     * @param State $state Current path
217     * @param Location $location Iterated storage
218     * @return Term Current subject value
219     */
220    public function referenced(Instruction $instruction, State $state, Location $location): Term
221    {
222        $value = $state->memory->read($location);
223        if ($value->kind !== 'uninitialized') {
224            return $value;
225        }
226        if ($location->path === [] && !$location->unknown) {
227            return (new MemoryStep($this->context))->uninitialized($instruction, $location, $state);
228        }
229        $state->memory->reference($location);
230        return $state->memory->read($location);
231    }

233    /**
234     * Decides whether foreach accepts the subject; PHP warns and skips the body for scalars and null.
235     * @param Term $subject Iterated value
236     * @return bool|null Whether the subject is an array or object, or null when its type is not determined
237     */
238    public function iterable(Term $subject): ?bool
239    {
240        if ($subject->kind === 'constant') {
241            return false;
242        }
243        $predicates = new TypePredicates();
244        $array = $predicates->apply('is_array', $subject)->literal;
245        $object = $predicates->apply('is_object', $subject)->literal;
246        if ($array === true || $object === true) {
247            return true;
248        }
249        return $array === false && $object === false ? false : null;
250    }
