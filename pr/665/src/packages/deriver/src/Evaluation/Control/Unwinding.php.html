95        $result = [];
96        while ($state->handlers !== [] && count($state->handlers) > $state->completion->depth) {

102                    $guard = 'catch:' . $handler->region->continuation . ':' . $catch->block;
103                    if ($matched !== null) {

106                        if ($remaining !== null) {

116                    if ($remaining === null) {

118                    }
119                    if ($matched !== null) {

121                    }
122                    $state->completion = new Completion('throw', $remaining);
123                }
124            }
125            if ($handler->phase !== 'finally' && $handler->region->finally !== null) {
126                $state->handlers[] = new Handler($handler->region, 'finally', $state->completion);
