16/**
17 * Joins excess paths into one path that keeps executing, instead of sealing them.
18 *
19 * Paths are joined only when they share their control position, aliases, objects, and handlers, and differ only in
20 * plain values. Differing values are widened, so the joined path contains every input path; guards keep their common
21 * entries and constraints are dropped.
22 * @visibility root
23 */

38    /**
39     * Widens paths that wait at one program point, entered from one predecessor, or that complete with one kind.
40     * @param non-empty-list<State> $states Paths to join
41     * @param bool $completed Whether the paths have left the callable, so their position and registers no longer matter
42     * @param bool $shared Whether the caller already established that the paths share one structure
43     * @return State|null Joined path, or null when the paths differ in more than plain values
44     */
45    public function join(array $states, bool $completed = false, bool $shared = false): ?State
46    {
47        $structure = $shared ? '' : $this->structure($states[0], $completed);
48        foreach ($shared ? [] : $states as $state) {
49            if ($this->structure($state, $completed) !== $structure) {
50                return null;
51            }
52        }
53        $registers = $completed ? [] : $this->values(array_map(static fn (State $state): array => $state->registers, $states), false);
54        $cells = $this->values(array_map(static fn (State $state): array => $state->memory->cells, $states), true, true);
55        $completion = $this->values(array_map(static fn (State $state): array => $state->completion->value === null ? [] : ['value' => $state->completion->value], $states), false);
56        if ($registers === null || $cells === null || $completion === null || isset($completion['value']) !== ($states[0]->completion->value !== null)) {
57            return null;
58        }
59        $result = $states[0]->fork();
60        if (!$completed) {
61            $addresses = $this->locations(array_map(static fn (State $state): array => $state->addresses, $states));
62            $offsets = $this->offsets(array_map(static fn (State $state): array => $state->offsets, $states));
63            $targets = $this->common(array_map(static fn (State $state): array => $state->callTargets, $states));
64            $properties = $this->common(array_map(static fn (State $state): array => $state->properties, $states));
65            if ($addresses === null || $offsets === null || $targets === null || $properties === null) {
66                return null;
67            }
68            [$result->addresses, $result->offsets, $result->callTargets, $result->properties] = [$addresses, $offsets, $targets, $properties];
69        }
70        $result->registers = $registers;
71        $result->memory->cells = $cells;
72        $result->completion = new Completion($result->completion->kind, $completion['value'] ?? null, $result->completion->target, $result->completion->depth);
73        $result->constraints = [];
74        foreach (array_slice($states, 1) as $state) {
75            $this->merge($result, $state);
76        }
77        return $result;
78    }

80    /**
81     * Accumulates bookkeeping that may differ between joined paths.
82     * @param State $result Joined path
83     * @param State $state Another joined path
84     */
85    public function merge(State $result, State $state): void
86    {
87        $result->guard = array_intersect_assoc($result->guard, $state->guard);
88        $result->producers = array_intersect_assoc($result->producers, $state->producers);
89        $result->producers = array_intersect_key($result->producers, $result->registers);
90        $result->evidence = array_values(array_unique([...$result->evidence, ...$state->evidence]));
91        $result->controls = array_values(array_unique([...$result->controls, ...$state->controls]));
92        $result->observed = $result->observed && $state->observed;
93        $result->observedQueries = array_intersect_key($result->observedQueries, $state->observedQueries);
94        foreach ($state->visits as $header => $count) {
95            $result->visits[$header] = max($result->visits[$header] ?? 0, $count);
96        }
97        foreach ($state->memory->versions as $root => $version) {
98            $result->memory->versions[$root] = max($result->memory->versions[$root] ?? 0, $version);
99        }
100        $result->memory->writers = array_intersect_assoc($result->memory->writers, $state->memory->writers);
101        $result->memory->sequence = max($result->memory->sequence, $state->memory->sequence);
102        $result->unknownLocals ??= $state->unknownLocals;
103        $result->memory->unknownShared ??= $state->memory->unknownShared;
104        $this->loops($result, $state);
105    }

107    /**
108     * Keeps loop bookkeeping only where both paths agree, so widening restores no path-specific guard and no loop is
109     * considered stable on behalf of a path that has not converged.
110     * @param State $result Joined path
111     * @param State $state Another joined path
112     */
113    public function loops(State $result, State $state): void
114    {
115        foreach ($result->loopGuards as $header => $guard) {
116            if (isset($state->loopGuards[$header])) {
117                $result->loopGuards[$header] = array_intersect_assoc($guard, $state->loopGuards[$header]);
118            } else {
119                unset($result->loopGuards[$header]);
120            }
121        }
122        foreach ($result->approximations as $header => $cells) {
123            $other = $state->approximations[$header] ?? [];
124            $same = isset($state->approximations[$header]) && array_keys($cells) === array_keys($other) && ($result->loopStructures[$header] ?? null) === ($state->loopStructures[$header] ?? null);
125            foreach ($same ? $cells : [] as $root => $value) {
126                $same = $same && isset($other[$root]) && $this->context->identity->key($value) === $this->context->identity->key($other[$root]);
127            }
128            if (!$same) {
129                unset($result->approximations[$header], $result->loopStructures[$header]);
130            }
131        }
132        if ($result->stableHeader !== $state->stableHeader) {
133            $result->stableHeader = null;
134        }
135    }

137    /**
138     * Fingerprints the parts of a path that joined paths must share exactly: control position, aliases, objects, handlers, and cursors.
139     * The predecessor selects the operands of the block's value merges, so pending paths must share it.
140     * @param State $state Candidate path
141     * @param bool $completed Whether the path has left the callable, so its position, locals, and handlers no longer matter
142     * @return string Structural fingerprint without values and bookkeeping
143     */
144    public function structure(State $state, bool $completed = false): string
145    {
146        $copy = $state->fork();
147        [$copy->registers, $copy->producers, $copy->addresses, $copy->offsets, $copy->callTargets, $copy->properties] = [[], [], [], [], [], []];
148        [$copy->guard, $copy->constraints, $copy->evidence, $copy->controls, $copy->visits] = [[], [], [], [], []];
149        [$copy->loopEntries, $copy->approximations, $copy->loopStructures, $copy->loopGuards, $copy->stableHeader] = [[], [], [], [], null];
150        [$copy->observed, $copy->unknownLocals, $copy->memory->unknownShared] = [false, null, null];
151        [$copy->memory->cells, $copy->memory->versions, $copy->memory->writers, $copy->memory->sequence] = [[], [], [], 0];
152        $copy->observedQueries = [];
153        $copy->completion = new Completion($state->completion->kind, null, $state->completion->target, $state->completion->depth);
154        if ($completed) {
155            [$copy->block, $copy->previous, $copy->locals, $copy->handlers, $copy->iterators] = [0, -1, [], [], []];
156        }
157        return (new StateJoin($this->context))->fingerprint($copy, $this->context->identity);

160    /**
161     * Keeps register metadata defined on every path; registers defined only on some paths are not read after the join.
162     * @template T of object
163     * @param non-empty-list<array<string, T>> $maps One register-indexed map per path
164     * @return array<string, T>|null Shared entries, or null when one register has different metadata
165     */
166    public function common(array $maps): ?array
167    {
168        $join = new StateJoin($this->context);
169        $result = array_intersect_key(...$maps);
170        foreach ($result as $key => $entry) {
171            if ($this->context->resources->reason() !== null) {
172                return null;
173            }
174            $expected = null;
175            foreach ($maps as $map) {
176                if ($map[$key] === $entry) {
177                    continue;
178                }
179                $expected ??= $join->encode($entry, $this->context->identity);
180                if ($join->encode($map[$key], $this->context->identity) !== $expected) {
181                    return null;
182                }
183            }
184        }
185        return $result;
186    }

188    /**
189     * Keeps addresses defined on every path; differing addresses into one storage root become an unknown location in it.
190     * A write through the unknown location invalidates the whole root, so every path's write is included.
191     * @param non-empty-list<array<string, Location>> $maps One register-indexed address map per path
192     * @return array<string, Location>|null Joined addresses, or null when one register addresses different roots
193     */
194    public function locations(array $maps): ?array
195    {
196        $result = array_intersect_key(...$maps);
197        foreach ($result as $key => $location) {
198            if ($this->context->resources->reason() !== null) {
199                return null;
200            }
201            foreach ($maps as $map) {
202                $other = $map[$key];
203                if ([$other->root, $other->path, $other->local, $other->unknown] === [$location->root, $location->path, $location->local, $location->unknown]) {
204                    continue;
205                }
206                if ($other->root !== $location->root) {
207                    return null;
208                }
209                $location = new Location($location->root, local: $location->local === $other->local ? $location->local : '', unknown: true);
210            }
211            $result[$key] = $location;
212        }
213        return $result;
214    }

216    /**
217     * Keeps offsets defined on every path; differing keys below one parent are widened like other plain values.
218     * @param non-empty-list<array<string, Address>> $maps One register-indexed offset map per path
219     * @return array<string, Address>|null Joined offsets, or null when one register has another parent or syntax
220     */
221    public function offsets(array $maps): ?array
222    {
223        $result = array_intersect_key(...$maps);
224        foreach ($result as $register => $address) {
225            if ($this->context->resources->reason() !== null) {
226                return null;
227            }
228            $keys = [];
229            foreach ($maps as $index => $map) {
230                if ($map[$register]->parent !== $address->parent || ($map[$register]->key === null) !== ($address->key === null)) {
231                    return null;
232                }
233                $keys[$index] = $map[$register]->key === null ? [] : ['key' => $map[$register]->key];
234            }
235            $key = $this->values(array_values($keys), false);
236            if ($key === null) {
237                return null;
238            }
239            $result[$register] = new Address($address->parent, $key['key'] ?? null);
240        }
241        return $result;
242    }

244    /**
245     * Widens maps of values key by key.
246     * @param non-empty-list<array<string, Term>> $maps One map per path
247     * @param bool $union Whether keys absent from some paths are joined as uninitialized storage
248     * @param bool $storage Whether keys are storage roots; only variable roots are widened, while object, model, static,
249     *     and constant storage must be identical, because widening their records would lose uninitialized slots
250     * @return array<string, Term>|null Joined values, or null when an identity-bearing value differs
251     */
252    public function values(array $maps, bool $union, bool $storage = false): ?array
253    {
254        $identity = $this->context->identity;
255        $lattice = new Lattice($this->context->models->extensions->domains);
256        $keys = $union ? array_keys(array_replace(...$maps)) : array_keys(array_intersect_key(...$maps));
257        $result = [];
258        foreach ($keys as $key) {
259            if ($this->context->resources->reason() !== null) {
260                return null;
261            }
262            $value = $maps[0][$key] ?? new Term('uninitialized');
263            foreach (array_slice($maps, 1) as $map) {
264                $next = $map[$key] ?? new Term('uninitialized');
265                if ($identity->key($value) === $identity->key($next)) {
266                    continue;
267                }
268                if (!$this->plain($value) || !$this->plain($next) || $storage && !str_starts_with($key, 'cell:') && !str_starts_with($key, 'global:')) {
269                    return null;
270                }
271                $value = $lattice->widen($value, $next);
272            }
273            $result[$key] = $value;
274        }
275        return $result;
276    }

278    /**
279     * Checks that a value carries no storage or object identity.
280     * @param Term $value Candidate value
281     * @return bool Whether widening the value cannot lose aliasing
282     */
283    public function plain(Term $value): bool
284    {
285        /** @var WeakMap<Term, true> $visited */
286        $visited = new WeakMap();
287        $pending = [$value];
288        while ($pending !== []) {
289            $current = array_pop($pending);
290            if (isset($visited[$current]) || ($this->context->plainValues[$current] ?? null) === true) {
291                continue;
292            }
293            if (count($visited) % 256 === 0 && $this->context->resources->reason() !== null || count($visited) >= $this->context->query->budget()->nodes) {
294                return false;
295            }
296            if (($this->context->plainValues[$current] ?? null) === false || in_array($current->kind, self::IDENTITIES, true)) {
297                $this->context->plainValues[$current] = false;
298                return false;
299            }
300            $visited[$current] = true;
301            array_push($pending, ...array_values($current->operands));
302        }
303        foreach ($visited as $checked => $_) {
304            $this->context->plainValues[$checked] = true;
305        }
306        return true;
307    }
