34    public function expand(Frame $frame, Instruction $instruction, string $target, ?Term $callable, int $depth, ?Term $receiver = null, bool $variants = true): Term
35    {
36        $context = $this->engine->context;
37        $reason = $context->boundary($depth);
38        if ($reason !== null) {
39            return (new Enumeration\Suspension())->expression($context, $frame, $instruction, $reason);
40        }
41        $context->referenceExpansions++;
42        $body = (new Invocation\Bodies($this->engine))->select($frame, $instruction, $target, $depth - 1);
43        if ($variants && !$body->replacement && $context->index->program->variants($target) !== []) {
44            return (new Choices())->make(array_map(fn (string $variant): array => [$this->expand($frame, $instruction, $variant, $callable, $depth, $receiver, false), ['declaration:' . $target => $variant]], [$target, ...$context->index->program->variants($target)]));
45        }
46        $graph = $body->implementation;
47        if ($graph instanceof Term) {
48            return $graph;
49        }
50        if ($graph !== null && !$body->replacement) {
51            $error = $this->signature($frame, $instruction, $graph, $depth - 1);
52            if ($error !== null) {
53                return $error;
54            }
55        }
56        if ($instruction->operation === 'new') {

59        }
60        return $this->invoke($frame, $instruction, $target, $callable, $depth, $receiver, $body);

66    public function invoke(Frame $frame, Instruction $instruction, string $target, ?Term $callable, int $depth, ?Term $receiver, Invocation\Body $body): Term
67    {
68        $context = $this->engine->context;
69        $graph = $body->implementation;
70        if (!$graph instanceof Graph) {
71            return $graph ?? $this->missing($frame, $instruction, $target, $receiver, $depth);
72        }
73        if (!(new \Deriver\Evaluation\Call\Member\Access($context->index->program))->allows($graph->body, $frame->graph->body->className)) {
74            return new Term('throwable', 'Error', attributes: ['source' => $instruction->source->path, 'start' => $instruction->source->start]);
75        }
76        if ($graph->body->external || $graph->body->abstract) {
77            return $this->missing($frame, $instruction, $target, $receiver, $depth);
78        }
79        if (($frame->calls[$instruction->id] ?? 0) >= $context->budget->recursion || ($frame->calls[$instruction->id] ?? 0) > 0 && $target === $frame->graph->body->symbol && $this->passThroughRecursion($frame, $instruction)) {
80            return (new Enumeration\Suspension())->expression($context, $frame, $instruction, ($frame->calls[$instruction->id] ?? 0) >= $context->budget->recursion ? 'RECURSION_LIMIT' : 'CYCLE');
81        }
82        $body->enter($context);
83        $bound = $this->bind($frame, $instruction, $graph, $receiver, $depth - 1);
84        if ($callable?->kind === 'closure') {
85            $bound = $this->captures($bound, $callable);
86        }
87        return Evidence\Provenance::wrap($this->engine->returns($bound, $depth - 1), 'call', $instruction->source, ['caller' => $frame->graph->body->symbol, 'callee' => $graph->body->symbol, 'call_site' => $instruction->source->id(), 'parent_context' => $frame->identity, 'context' => $bound->identity, 'called_class' => $bound->calledClass, 'lexical_class' => $graph->body->className, 'receiver' => $receiver === null ? '' : (new \Deriver\Value\Identity())->key($receiver), 'implementation' => $graph->body->source->id()]);
88    }

149    public function passThroughRecursion(Frame $frame, Instruction $call): bool
150    {
151        if ($call->operation !== 'invoke' || count($call->arguments) !== count($frame->graph->body->parameters) || $call->arguments === []) {
152            return false;
153        }
154        if (!$this->stableEnvironment($frame)) {
155            return false;
156        }
157        foreach ($call->arguments as $position => $argument) {
158            $value = $frame->graph->definitions[$argument->register] ?? null;
159            $address = $value?->operation === 'argument' && ($value->attributes['address'] ?? false) === true ? ($frame->graph->definitions[$value->operands[1]] ?? null) : null;
160            if ($address?->operation !== 'local' || $address->name !== ($argument->name ?? $frame->graph->body->parameters[$position]->name ?? null) || (new Storage($this->engine))->modified($frame, $address->name)) {
161                return false;
162            }
163        }
164        return true;

189        if ($call->operation === 'invoke-method' || $receiver !== null) {

191        } elseif ($call->operation === 'new') {

215    /**
216     * Resolves a reference write from the selected implementation.
217     */
218    public function selectedEffect(Frame $caller, Instruction $call, string $address, string $actual, string $target, ?Term $receiver, int $depth): ?Term
219    {
220        $body = (new Invocation\Bodies($this->engine))->select($caller, $call, $target, $depth);
221        $graph = $body->implementation;
222        if ($graph instanceof Term) {
223            return new Term('call-write', $target, [$this->engine->value($caller, $actual, $depth), $graph]);
224        }
225        if ($graph === null) {
226            return $this->unknownWrite($caller, $call, $address, $depth);
227        }
228        foreach ($graph->body->parameters as $position => $parameter) {
229            if (!$parameter->byReference) {
230                continue;
231            }
232            $register = (new Models($this->engine))->actual($call, $parameter->name, $position);
233            $argument = $caller->graph->definitions[$register ?? ''] ?? null;
234            $actual = $argument?->operation === 'argument' ? ($argument->operands[1] ?? '') : ($call->arguments[$position]->location ?? '');
235            if ((new Storage($this->engine))->key($caller, $actual) !== (new Storage($this->engine))->key($caller, $address)) {
236                continue;
237            }
238            if ($graph->body->external) {
239                $previous = $this->engine->value($caller, $register ?? '', $depth);
240                return new Term('call-write', $target . ':$' . $parameter->name, [$previous], ['reason' => 'MISSING_SOURCE', 'source' => $call->source->path, 'start' => $call->source->start, 'end' => $call->source->end]);
241            }
242            $bound = $this->bind($caller, $call, $graph, $receiver);
243            foreach ($graph->definitions as $definition) {
244                if ($definition->operation === 'local' && $definition->name === $parameter->name) {
245                    $body->enter($this->engine->context);
246                    return $this->finalStorage($bound, $definition->result, $depth);
247                }
248            }
249        }
250        return null;
251    }
