29    {
30        $this->context->frames[$frame->identity] = $frame;
31        $instruction = $frame->graph->definitions[$register] ?? null;
32        if ($instruction === null) {
33            return $register === '' ? Term::constant(null) : $this->context->reference($frame, $register, $frame->graph->body->source, reason: 'MISSING_DEFINITION');
34        }
35        if ($instruction->operation === 'constant') {
36            $this->context->record($frame, $instruction);
37            $value = (new Invocation\Rules($this))->apply($frame, $instruction, $depth) ?? $instruction->constant ?? Term::constant(null);
38            $value = Evidence\Provenance::wrap($value, 'source-definition', $instruction->source, ['owner' => $frame->graph->body->symbol, 'definition' => $register, 'operation' => 'constant']);
39            return $this->bounded($frame, $instruction, $value);
40        }
41        $key = $this->context->cacheNamespace . ':' . $frame->identity . ':' . $register . ':depth:' . $depth;
42        $cached = $this->context->values[$key] ?? $this->context->cache->get($key);
43        if ($cached !== null) {
44            $this->context->sharedNodeHits++;
45            $this->context->record($frame, $instruction);
46            return $this->bounded($frame, $instruction, $cached);
47        }
48        if (isset($this->context->active[$key])) {
49            return $this->context->reference($frame, $register, $instruction->source, reason: 'CYCLE', kind: 'recursive');
50        }
51        $this->context->active[$key] = true;
52        $this->context->record($frame, $instruction);
53        $value = $this->instruction($frame, $instruction, $depth);
54        $value = Evidence\Provenance::wrap($value, in_array($instruction->operation, ['read', 'read-silent', 'argument'], true) ? 55        $value = $this->bounded($frame, $instruction, $value);
56        unset($this->context->active[$key]);
57        $this->context->values[$key] = $value;
58        if ($this->context->stopReason === null) {
59            $this->context->cache->put($key, $value);
60        }
61        return $value;

78    public function instruction(Frame $frame, Instruction $instruction, int $depth): Term
79    {
80        $op = $instruction->operation;
81        if (($reason = $this->context->work()) !== null) {
82            return (new Enumeration\Suspension())->expression($this->context, $frame, $instruction, $reason);
83        }
84        $override = (new Invocation\Rules($this))->apply($frame, $instruction, $depth);
85        if ($override !== null) {
86            return $override;
87        }
88        return match ($op) {
89            'closure' => new Term('closure', $instruction->name, attributes: ['context' => $frame->identity, 'creation' => $instruction->result]),
90            'copy', 'write' => $this->value($frame, $instruction->operands[$op === 'write' ? 1 : 0], $depth),
91            'compound', 'increment' => Memory\Mutations::value($this, $frame, $instruction, $depth),
92            'read', 'read-silent' => (new Storage($this))->read($frame, $instruction, $instruction->operands[0], $depth),
93            'argument' => ($instruction->attributes['address'] ?? false) === true ? (new Storage($this))->read($frame, $instruction, $instruction->operands[1], $depth) : $this->value($frame, $instruction->operands[1], $depth),
94            'invoke', 'invoke-method', 'invoke-static', 'new' => (new Calls($this))->value($frame, $instruction, $depth),
95            'phi' => $this->phi($frame, $instruction, $depth),
96            'array-set', 'array-unpack' => (new ArrayConstruction($this))->value($frame, $instruction, $depth),
97            'static-initialized' => (new Memory\Statics())->initialized($this, $frame, $frame->graph->definitions[$instruction->operands[0]], $depth),
98            'constant-fetch' => $this->constant($frame, $instruction, $depth),
99            'class-constant' => (new Language\Expressions($this))->constant($frame, $instruction, $depth),
100            'clone' => (new Language\Expressions($this))->copy($frame, $instruction, $depth),
101            'iterator', 'iterate', 'iterator-value', 'iterator-key', 'iterator-address' => (new Recurrence\Iteration())->value($this, $frame, $instruction, $depth),
102            default => $this->expression($frame, $instruction, $depth),
103        };

138            $value = $frame->graph->body->blocks[$block]->terminator->kind === 'throw' ? new Term('throwable', (string) ($value->attributes['type'] ?? 139            $value = (new Language\Catches())->resolve($this, $frame, $block, $value, $depth);
140            $value = (new Language\Finalizers())->apply($this, $frame, $block, $value, $depth);

149    public function phi(Frame $frame, Instruction $instruction, int $depth): Term
150    {
151        $condition = $this->value($frame, $instruction->operands[2], $depth);
152        $alternatives = [];
153        foreach ((new Choices())->alternatives($condition) as [$test, $guard]) {
154            $truth = (new Operations())->truth($test);
155            foreach ([true, false] as $expected) {
156                if ($truth !== null && $truth !== $expected) {
157                    continue;
158                }
159                $value = $this->value($frame, $instruction->operands[$expected ? 0 : 1], $depth);
160                $value = Evidence\Provenance::operation($value, [$test], 'selection');
161                $value = Evidence\Provenance::wrap($value, 'choice', $instruction->source, ['selector' => $frame->identity . ':condition:' . $instruction->operands[2], 'branch' => $expected]);
162                $guard[$frame->identity . ':condition:' . $instruction->operands[2]] = $expected;
163                $alternatives[] = [$value, $guard];
164            }
165        }
166        return (new Choices())->make($alternatives);
167    }

184    public function evaluate(string $operation, string $name, array $values): Term
185    {
186        foreach ($values as $value) {
187            if ($value->kind === 'throwable') {
188                return $value;
189            }
190        }
191        $operations = new Operations($this->context->configuration->target->floatPrecision);
192        return match ($operation) {
193            'binary' => $operations->binary($name, $values[0], $values[1]),
194            'unary' => $operations->unary($name, $values[0]),
195            'cast' => $operations->cast($name, $values[0]),
196            'not-null' => $values[0]->kind === 'constant' ? Term::constant($values[0]->literal !== null) : new Term('operation', 'not-null', $values),
197            'array-read' => $this->element($values[0], $values[1]),
198            'array-set' => in_array($values[1]->kind, ['constant', 'append'], true) ? (new \Deriver\Value\Arrays())->set($values[0], $values[1]->kind === 'append' ? null : $values[1], $values[2]) : new Term('array-set', operands: $values, attributes: ['type' => 'array']),
199            'intrinsic' => $this->intrinsic($name, $values),
200            'throw' => new Term('throwable', (string) ($values[0]->attributes['type'] ?? 'Throwable'), $values),
201            'external', 'external-body' => new Term('operation', $name !== '' ? $name : $operation, $values, ['reason' => 'EXTERNAL_INPUT']),
202            default => new Term('operation', $operation . ':' . $name, $values, ['reason' => 'UNSUPPORTED_OPERATION']),
203        };
