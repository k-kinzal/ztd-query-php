43    /**
44     * Follows object, cell, array, and capture references without revisiting cycles.
45     * @param State $state Execution state
46     * @param Term $value Reachable value
47     * @param string $reason Root cause
48     * @param array<string, true> $seen Already visited storage roots
49     */
50    public function reachable(State $state, Term $value, string $reason, array &$seen): void
51    {
52        /** @var WeakMap<Term, true> $visited */
53        $visited = new WeakMap();
54        $pending = [$value];
55        while ($pending !== []) {
56            $current = array_pop($pending);
57            if (isset($visited[$current])) {
58                continue;
59            }
60            $visited[$current] = true;
61            $root = null;
62            if (is_string($current->literal) && ($current->kind === 'object' || isset($state->memory->cells['object:' . $current->literal]) || isset($state->memory->cells['model:' . $current->literal]))) {
63                $root = 'object:' . $current->literal;
64                array_push($pending, ...$this->slotValues($state, $current->literal, $reason, $seen));
65            } elseif ($current->kind === 'cell' && is_string($current->literal)) {
66                $root = $current->literal;
67            }
68            if ($root !== null && !isset($seen[$root])) {
69                $seen[$root] = true;
70                $previous = $state->memory->cells[$root] ?? Term::opaque($reason);
71                $pending[] = $previous;
72                if ($previous->kind === 'array') {
73                    $entries = [];
74                    foreach ($previous->operands as $key => $entry) {
75                        $entries[$key] = new Term('opaque', $reason, [$entry], ['type' => $state->memory->propertyTypes[$root][$key] ?? 'mixed', 'maybeUninitialized' => true]);
76                    }
77                    $state->memory->cells[$root] = Term::array($entries, true);
78                } else {
79                    $state->memory->cells[$root] = Term::opaque($reason, dependencies: [$previous]);
80                }
81            }
82            array_push($pending, ...array_values($current->operands));
83        }
84    }

100    /**
101     * Invalidates abstract slots and queues their previous mutable descendants.
102     * @param State $state Execution state
103     * @param string $identity Reachable receiver
104     * @param string $reason Boundary cause
105     * @param array<string, true> $seen Already invalidated storage roots
106     * @return list<Term> Values whose reachable storage must also be invalidated
107     */
108    public function slotValues(State $state, string $identity, string $reason, array &$seen): array
109    {
110        $root = 'model:' . $identity;
111        if (isset($seen[$root]) || !isset($state->memory->cells[$root])) {
112            return [];
113        }
114        $seen[$root] = true;
115        $values = [];
116        foreach ($state->memory->slotContracts as $id => $slot) {
117            if ($slot->invalidation === 'preserve') {
118                continue;
119            }
120            $location = new Location($root, [$id]);
121            $previous = $state->memory->read($location);
122            $values[] = $previous;
123            $state->memory->write($location, Term::opaque($reason, $slot->type, [$previous]));
124        }
125        return $values;
126    }
