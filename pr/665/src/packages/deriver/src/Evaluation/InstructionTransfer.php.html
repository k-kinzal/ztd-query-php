51    /**
52     * Dispatches one instruction to its semantic operation.
53     * @param CallableGraph $callable Current callable
54     * @param Instruction $instruction Instruction
55     * @param State $state Input path
56     * @return list<State> Result paths
57     */
58    public function apply(CallableGraph $callable, Instruction $instruction, State $state): array
59    {
60        $op = $instruction->operation;
61        $prepared = $this->preparation($callable, $instruction, $state);
62        if ($prepared !== null) {
63            return $prepared;
64        }
65        if ($op === 'compound') {
66            return (new CompoundAssignment($this->machine))->apply($callable, $instruction, $state);
67        }
68        if (in_array($op, ['unsupported', 'symbol-table-boundary', 'uncertain-order'], true)) {
69            return $this->boundary($callable, $instruction, $state);
70        }
71        $storage = $this->storage($callable, $instruction, $state);
72        if ($storage !== null) {
73            return $storage;
74        }
75        $conversion = (new Conversions($this->machine))->apply($callable, $instruction, $state);
76        if ($conversion !== null) {
77            return $conversion;
78        }
79        $context = $this->machine->context;
80        if ($op === 'array-read') {
81            return $this->offsetRead($callable, $instruction, $state);
82        }
83        if (in_array($op, ['constant-fetch', 'class-constant'], true)) {
84            return (new ConstantTransfer($this->machine))->apply($callable, $instruction, $state);
85        }
86        if (in_array($op, ['invoke', 'invoke-method', 'invoke-static', 'new', 'clone', 'intrinsic'], true)) {
87            return (new CallExecutor($this->machine))->instruction($callable, $instruction, $state);
88        }
89        if (in_array($op, ['local', 'dynamic-local', 'element-address', 'field-address', 'static-address', 'unsupported-address', 'returned-address', 'read', 'read-silent', 'write', 'alias', 'reference', 'increment', 'unset', 'global', 'static-local', 'static-initialized'], true)) {
90            $value = (new MemoryStep($context))->evaluate($callable, $instruction, $state);
91        } elseif (in_array($op, ['iterator', 'iterate', 'iterator-key', 'iterator-value', 'iterator-address', 'iterator-release'], true)) {
92            $value = (new IterationStep($context))->evaluate($instruction, $state);
93        } elseif ($op === 'array-unpack') {
94            return $this->unpack($instruction, $state);
95        } elseif ($op === 'array-set') {
96            $array = $state->value($instruction->operands[0]);
97            $key = $instruction->operands[1] === '' ? null : (new Reader($context))->key($state->value($instruction->operands[1]), $instruction);
98            $item = $state->registers[$instruction->operands[2]] ?? Term::opaque('UNCOMPUTED_REGISTER');
99            $value = (new Arrays())->set($array, $key, $item);
100        } elseif (in_array($op, ['closure', 'callable', 'callable-method', 'instanceof'], true)) {
101            $value = (new CallableTransfer($context))->evaluate($callable, $instruction, $state);
102        } else {
103            $value = $this->other($callable, $instruction, $state);
104        }
105        $state->registers[$instruction->result] = $value;
106        return (new ScalarErrors($context))->paths($instruction, $state);
107    }

109    /**
110     * Unpacks arrays while preserving invalid-type errors and unknown iterator effects.
111     * @param Instruction $instruction Array construction step
112     * @param State $state Current path
113     * @return list<State> Normal and possible exceptional paths
114     */
115    public function unpack(Instruction $instruction, State $state): array
116    {
117        $array = $state->value($instruction->operands[0]);
118        $item = $state->value($instruction->operands[2]);
119        if ((new TypePredicates())->apply('is_array', $item)->literal === true) {
120            $state->registers[$instruction->result] = (new Arrays())->merge($array, $item);
121            if ($state->registers[$instruction->result]->kind === 'array-merge' && !(new Arrays())->appendable($array)) {
122                $this->machine->context->frontier('WIDENED', $instruction->source, 'symbolic-unpack-append', [$array, $item]);
123                $exception = $state->fork();
124                $exception->completion = new Completion('throw', new Term('throwable', 'Error'));
125                return [$state, $exception];
126            }
127            return [$state];
128        }
129        if ($item->kind === 'constant' || $item->kind === 'closure') {
130            $state->completion = new Completion('throw', new Term('throwable', $item->kind === 'closure' ? 'TypeError' : 'Error'));
131            return [$state];
132        }
133        return (new UnknownCall($this->machine->context))->apply($state, $instruction, [new PassedArgument($item)], null, 'UNSUPPORTED_LANGUAGE_FEATURE');
134    }

136    /**
137     * Resolves call preparation and model state instructions before ordinary value transfer.
138     * @param CallableGraph $callable Executing graph
139     * @param Instruction $instruction Current instruction
140     * @param State $state Current path
141     * @return list<State>|null Specialized result, or null for ordinary instructions
142     */
143    public function preparation(CallableGraph $callable, Instruction $instruction, State $state): ?array
144    {
145        $op = $instruction->operation;
146        if ($op === 'iterator-value') {
147            return (new IterationStep($this->machine->context))->candidates($instruction, $state);
148        }
149        if ($op === 'external-body') {
150            return (new ExternalBody($this->machine->context))->apply($callable, $instruction, $state);
151        }
152        if ($op === 'model-havoc') {
153            return (new Effects($this->machine->context))->apply($instruction, $state);
154        }
155        if ($op === 'model-state-address') {
156            $state->registers[$instruction->result] = (new StateStorage($this->machine->context))->address($instruction, $state);
157            return [$state];
158        }
159        if ($op === 'call-prepare') {
160            return (new Transfer($this->machine))->apply($callable, $instruction, $state);
161        }
162        if ($op === 'callable' || $op === 'callable-method') {
163            return (new Capture($this->machine))->apply($callable, $instruction, $state);
164        }
165        if ($op === 'argument') {
166            return (new Arguments($this->machine))->apply($callable, $instruction, $state);
167        }
168        return null;
169    }

171    /**
172     * Applies property, offset, and shared-reference checks before ordinary storage operations.
173     * @param CallableGraph $callable Executing callable
174     * @param Instruction $instruction Pending operation
175     * @param State $state Input path
176     * @return list<State>|null Checked paths or null when ordinary transfer may proceed
177     */
178    public function storage(CallableGraph $callable, Instruction $instruction, State $state): ?array
179    {
180        $op = $instruction->operation;
181        $slot = (new SlotReference($this->machine->context))->apply($callable, $instruction, $state);
182        if ($slot !== null) {
183            return $slot;
184        }
185        $source = $instruction->operands[1] ?? '';
186        if ($op === 'alias' && ($instruction->attributes['source-prepared'] ?? false) !== true && (isset($state->properties[$source]) || isset($state->offsets[$source]))) {
187            return (new PropertyReference($this->machine))->alias($callable, $instruction, $state);
188        }
189        $property = $state->properties[$instruction->operands[0] ?? ''] ?? null;
190        if ($property !== null && in_array($op, ['read', 'read-silent', 'write', 'increment', 'reference', 'alias', 'unset'], true)) {
191            return (new PropertyTransfer($this->machine))->apply($callable, $instruction, $state, $property);
192        }
193        if (isset($state->offsets[$instruction->operands[0] ?? '']) && in_array($op, ['read', 'read-silent', 'write', 'increment', 'reference', 'alias', 'unset'], true)) {
194            return (new Offset\Transfer($this->machine))->apply($callable, $instruction, $state);
195        }
196        if ($op === 'reference' && ($state->addresses[$instruction->operands[0] ?? '']->unknown ?? true)) {
197            return $this->boundary($callable, new Instruction($instruction->id, 'unsupported', $instruction->source, $instruction->result, name: 'unknown-reference-location'), $state);
198        }
199        $referenceWrite = (new ReferenceAssignment($this->machine->context))->apply($callable, $instruction, $state);
200        if ($referenceWrite !== null) {
201            return $referenceWrite;
202        }
203        return null;
204    }

206    /**
207     * Reads offsets on computed values, including silent tests and ArrayAccess results.
208     * @param CallableGraph $callable Executing graph
209     * @param Instruction $instruction Computed offset read
210     * @param State $state Evaluated container and key
211     * @return list<State> Read results and exceptions
212     */
213    public function offsetRead(CallableGraph $callable, Instruction $instruction, State $state): array
214    {
215        $container = $state->value($instruction->operands[0]);
216        $key = $state->value($instruction->operands[1]);
217        $reader = new Reader($this->machine->context);
218        $silent = ($instruction->attributes['silent'] ?? false) === true;
219        if ($container->kind === 'object' && !$reader->plainObject($container)) {
220            $read = new Instruction($instruction->id, $silent ? 'read-silent' : 'array-read', $instruction->source, $instruction->result, $instruction->operands, attributes: $instruction->attributes);
221            return (new Protocol($this->machine))->apply($callable, $read, $state, new ProtocolAccess($container, $key));
222        }
223        return (new Offset\Transfer($this->machine))->result($state, $instruction, $reader->read($container, $key, $instruction, $state, $silent));
224    }

226    /**
227     * Preserves normal and exceptional exits when arbitrary boundary effects are possible.
228     * @param CallableGraph $callable Executing callable
229     * @param Instruction $instruction Unsupported or unordered operation
230     * @param State $state Input path
231     * @return list<State> Inclusive normal and unknown-throwable paths after conservative effects
232     */
233    public function boundary(CallableGraph $callable, Instruction $instruction, State $state): array
234    {
235        $state->registers[$instruction->result] = $this->other($callable, $instruction, $state);
236        $exception = $state->fork();
237        $exception->completion = new Completion('throw', new Term('throwable', 'Throwable', attributes: ['uncertain' => true]));
238        return [$state, $exception];
239    }

241    /**
242     * Applies exception and boundary instructions before pure evaluation.
243     * @param CallableGraph $callable Current callable
244     * @param Instruction $instruction Instruction
245     * @param State $state Current path
246     * @return Term Result value
247     */
248    public function other(CallableGraph $callable, Instruction $instruction, State $state): Term
249    {
250        $context = $this->machine->context;
251        if ($instruction->operation === 'enter-try') {
252            $region = $instruction->attributes['region'] ?? 0;
253            if (is_int($region) && isset($callable->regions[$region])) {
254                $state->handlers[] = new Handler($callable->regions[$region]);
255            }
256            return Term::constant(null);
257        }
258        if ($instruction->operation === 'throw') {
259            $value = $state->value($instruction->operands[0]);
260            $state->completion = new Completion('throw', $value);
261            return $value;
262        }
263        if (in_array($instruction->operation, ['unsupported', 'symbol-table-boundary', 'uncertain-order'], true)) {
264            $reason = $instruction->operation === 'uncertain-order' ? 'UNSPECIFIED_EVALUATION_ORDER' : ($instruction->operation === 'symbol-table-boundary' ? $instruction->name : 'UNSUPPORTED_LANGUAGE_FEATURE');
265            $values = array_map($state->value(...), $instruction->operands);
266            if (($instruction->attributes['effects'] ?? '') === 'reachable') {
267                (new Havoc())->call($state, $values, [], $reason);
268            } else {
269                (new Havoc())->all($state, $reason);
270            }
271            return $context->frontier($reason, $instruction->source, $instruction->name, $values, (string) ($instruction->attributes['type'] ?? 'mixed'));
272        }
273        return (new PureStep($context))->evaluate($callable, $instruction, $state);
274    }
