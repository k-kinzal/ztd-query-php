16/**
17 * Checks the privileges of a GRANT or REVOKE against the kind of object.
18 *
19 * Rule: PG-PRIVILEGE-CHECK-001. The server knows a fixed list of privilege
20 * names (MAINTAIN from release 17; `rule` is accepted and ignored) and a
21 * fixed set of them for each kind of object (the table of privileges in the
22 * manual). A name outside the list is reported as unrecognized, a known
23 * privilege outside the set of the kind as invalid for that kind. A
24 * relation named in GRANT may be a sequence, so there the sequence
25 * privileges are valid too; for default privileges only the privileges of
26 * the kind itself are. A column list is valid on relations only, limits the
27 * privilege to SELECT, INSERT, UPDATE or REFERENCES, and cannot be written
28 * for default privileges. The right to grant cannot be given to PUBLIC.
29 * The names are compared as decoded, as the server does. Termination: one
30 * pass over a finite list.
31 * Source: https://www.postgresql.org/docs/17/ddl-priv.html#PRIVILEGES-SUMMARY-TABLE, https://www.postgresql.org/docs/16/ddl-priv.html,
32 * https://www.postgresql.org/docs/17/sql-grant.html, https://www.postgresql.org/docs/17/sql-alterdefaultprivileges.html.
33 * Status: Implemented.
34 *
35 * @visibility SqlSemantics\Platform\PostgreSql
36 */
37final class PrivilegeChecks
38{
39    /**
40     * The privilege names the server knows in every supported release.
41     */
42    private const KNOWN = ['insert', 'select', 'update', 'delete', 'truncate', 'references', 'trigger', 'execute', 'usage', 'create', 'temp', 'temporary', 'connect', 'set', 'alter system', 'rule'];
43
44    /**
45     * The privileges that apply to each kind of object, by the name of the kind.
46     */
47    private const RIGHTS = [
48        'Relation' => ['insert', 'select', 'update', 'delete', 'truncate', 'references', 'trigger', 'maintain'],
49        'Sequence' => ['usage', 'select', 'update'],
50        'Database' => ['create', 'temp', 'temporary', 'connect'],
51        'Domain' => ['usage'],
52        'Function' => ['execute'],
53        'Procedure' => ['execute'],
54        'Routine' => ['execute'],
55        'Language' => ['usage'],
56        'LargeObject' => ['select', 'update'],
57        'Parameter' => ['set', 'alter system'],
58        'Schema' => ['usage', 'create'],
59        'Tablespace' => ['create'],
60        'Type' => ['usage'],
61        'ForeignDataWrapper' => ['usage'],
62        'ForeignServer' => ['usage'],
63    ];
64
65    /**
66     * The privileges that can be limited to columns.
67     */
68    private const COLUMN = ['select', 'insert', 'update', 'references'];
69
70    /**
71     * Reports the privileges that are unknown, do not apply to the kind of object, or carry a column list where none is valid.
72     *
73     * @param list<Privilege> $privileges
74     * @param bool $defaults Whether the privileges are default privileges
75     */
76    public function privileges(Derivation $derivation, array $privileges, PrivilegeObjectKind $object, bool $defaults): void
77    {
78        $rights = self::RIGHTS[$object->name];
79        if ($object === PrivilegeObjectKind::Relation && !$defaults) {
80            $rights = [...$rights, ...self::RIGHTS['Sequence']];
81        }
82        foreach ($privileges as $privilege) {
83            if ($privilege->columns !== []) {
84                $this->columns($derivation, $privilege, $object, $defaults);
85                continue;
86            }
87            $this->named($derivation, $privilege->privilege(), $rights, $object->subject());
88        }
89    }
90
91    /**
92     * Reports the problems of a privilege that carries a column list.
93     */
94    public function columns(Derivation $derivation, Privilege $privilege, PrivilegeObjectKind $object, bool $defaults): void
95    {
96        if ($defaults) {
97            $derivation->report(new AccessProblem(AccessProblemRule::DefaultColumns));
98
99            return;
100        }
101        if ($object !== PrivilegeObjectKind::Relation) {
102            $derivation->report(new AccessProblem(AccessProblemRule::ColumnPrivilegesTarget));
103
104            return;
105        }
106        $this->named($derivation, $privilege->privilege(), self::COLUMN, 'column');
107    }
108
109    /**
110     * Reports a privilege name that is unknown or outside the given set; ALL, which has no name, is in every set.
111     *
112     * @param list<string> $rights The privilege names that apply
113     * @param string $subject The words the server names the kind with
114     */
115    public function named(Derivation $derivation, ?string $name, array $rights, string $subject): void
116    {
117        if ($name === null || $name === 'rule') {
118            return;
119        }
120        if (!$this->known($derivation, $name)) {
121            $derivation->report(new AccessProblem(AccessProblemRule::UnrecognizedPrivilege, [$name]));
122        } elseif (!in_array($name, $rights, true)) {
123            $derivation->report(new AccessProblem(AccessProblemRule::InvalidPrivilege, [$name === 'temporary' ? 'TEMP' : strtoupper($name), $subject]));
124        }
125    }
126
127    /**
128     * Tells whether the release knows a privilege name.
129     */
130    public function known(Derivation $derivation, string $name): bool
131    {
132        return in_array($name, self::KNOWN, true) || ($name === 'maintain' && $derivation->context->profile->grammar === GrammarRelease::PostgreSql172);
133    }
134
135    /**
136     * Reports PUBLIC among the grantees of a grant that gives the right to grant.
137     *
138     * @param list<RoleSpec> $grantees
139     */
140    public function grantOption(Derivation $derivation, array $grantees): void
141    {
142        foreach ($grantees as $grantee) {
143            if ($grantee->kind === RoleSpecKind::Everyone) {
144                $derivation->report(new AccessProblem(AccessProblemRule::GrantOptionToPublic));
145            }
146        }
147    }
148}
