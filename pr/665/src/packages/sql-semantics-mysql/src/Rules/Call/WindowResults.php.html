29/**
30 * The result of a function that is only a window function.
31 *
32 * Rule: MYSQL-WINDOW-RESULT-001. ROW_NUMBER, RANK, DENSE_RANK and NTILE are
24 * BIGINT UNSIGNED, CUME_DIST and PERCENT_RANK DOUBLE; none of them is NULL.
33 * BIGINT, CUME_DIST and PERCENT_RANK DOUBLE; none of them is NULL.
34 * LEAD and LAG have the aggregated type of the value and the default
26 * (MYSQL-TYPE-AGGREGATION-001); FIRST_VALUE, LAST_VALUE and NTH_VALUE
27 * the type of the value; each is NULL when the row is outside the frame or
28 * partition. IGNORE NULLS and FROM LAST are rejected by the server
29 * (ER_NOT_SUPPORTED_YET). Terminates: a fixed number of tests on one call.
35 * (MYSQL-TYPE-AGGREGATION-001), JSON for a JSON value and a JSON or NULL
36 * default; FIRST_VALUE, LAST_VALUE and NTH_VALUE
37 * the type of the value; each is read from the temporary table of the
38 * window, which holds an integer as an INT or a BIGINT and a string as a
39 * VARCHAR or a BLOB. FIRST_VALUE, LAST_VALUE and NTH_VALUE are NULL when
40 * the frame has no such row, LEAD and LAG when the row is outside the
41 * partition and the default is NULL or can be. IGNORE NULLS and FROM LAST
42 * are rejected by the server (ER_NOT_SUPPORTED_YET). Terminates: a fixed
43 * number of tests on one call (verified on a live 8.4 server).
44 * Source: https://dev.mysql.com/doc/refman/8.4/en/window-function-descriptions.html.
45 * Status: Implemented.
46 *
47 * @visibility SqlSemantics\Platform\MySql
48 */
49final class WindowResults

51    /**
52     * Answers the facts of a window function call and reports what the server rejects.
53     *
54     * @param list<ScalarFact> $arguments
55     */
56    public function result(WindowFunction $call, array $arguments, Derivation $derivation): ScalarFact
57    {
58        if ($call->nulls === NullTreatment::Ignore) {
59            $derivation->report(new UnsupportedWindowing(WindowingLimit::IgnoreNulls));
60        }
61        if ($call->edge === CountingEdge::Last) {
62            $derivation->report(new UnsupportedWindowing(WindowingLimit::FromLast));
63        }
64
65        return match ($call->kind) {
66            WindowFunctionKind::RowNumber, WindowFunctionKind::Rank, WindowFunctionKind::DenseRank,
53            WindowFunctionKind::Tile => new ScalarFact(new Known(TypeClass::Unsigned->descriptor()), Nullability::NotNull),
67            WindowFunctionKind::Tile => new ScalarFact(new Known(TypeClass::Integer->descriptor()), Nullability::NotNull),
68            WindowFunctionKind::CumulativeDistribution,
69            WindowFunctionKind::PercentRank => new ScalarFact(new Known(TypeClass::Floating->descriptor()), Nullability::NotNull),
70            WindowFunctionKind::Lead, WindowFunctionKind::Lag => new ScalarFact(
71                (new TypeAggregation())->aggregate([$arguments[0]->type, ...(isset($arguments[2]) ? [$arguments[2]->type] : [])]),
58                Nullability::Nullable,
72                (isset($arguments[2]) && $arguments[0]->nullability === Nullability::NotNull && $arguments[2]->nullability === Nullability::NotNull ? Nullability::NotNull : Nullability::Nullable),
73            ),
74            WindowFunctionKind::FirstValue, WindowFunctionKind::LastValue,
75            WindowFunctionKind::NthValue => new ScalarFact($arguments[0]->type ?? new NullOnly(), Nullability::Nullable),
76        };
77    }

79    /**
80     * Resolves the type of a window function call over the resolved types of its arguments, or answers null after reporting collations that conflict.
81     *
82     * @param list<Domain> $arguments
83     */
84    public function resolved(WindowFunction $call, array $arguments, Derivation $derivation): ?Domain
85    {
86        return match ($call->kind) {
87            WindowFunctionKind::RowNumber, WindowFunctionKind::Rank, WindowFunctionKind::DenseRank, WindowFunctionKind::Tile => Domain::integer(Field::LongLong, 21),
88            WindowFunctionKind::CumulativeDistribution, WindowFunctionKind::PercentRank => Domain::double(23),
89            WindowFunctionKind::FirstValue, WindowFunctionKind::LastValue, WindowFunctionKind::NthValue => (new Materialization())->windowed($arguments[0], $derivation->context->profile->grammar),
90            WindowFunctionKind::Lead, WindowFunctionKind::Lag => $this->shifted($call, $arguments, $derivation),
91        };
92    }

94    /**
95     * Resolves the type of LEAD or LAG: the value and the default aggregated, JSON when both are JSON or the default is NULL, as the temporary table of the window holds it.
96     *
97     * MySQL 8.0 reports such a JSON value in the collation of JSON text (verified on a live 8.0 server).
98     *
99     * @param list<Domain> $arguments
100     */
101    public function shifted(WindowFunction $call, array $arguments, Derivation $derivation): ?Domain
102    {
103        $values = [$arguments[0], ...(isset($arguments[2]) ? [$arguments[2]] : [])];
104        $json = $arguments[0]->kind === Kind::Json && array_filter($values, static fn (Domain $value): bool => $value->kind !== Kind::Json && $value->kind !== Kind::Null) === [];
105        $settings = Settings::of($derivation->context);
106        $release = $derivation->context->profile->grammar;
107        if ($json && $release === GrammarRelease::MySql8044) {
108            return new Domain(Kind::Json, Field::Json, 4294967295, 0, false, Collation::known('utf8mb4_bin'));
109        }
110        $aggregated = $json ? $arguments[0] : (new Aggregation(new Collations($settings->connection)))->of($values, strtolower($call->kind->value), $derivation);
111
112        return $aggregated === null ? null : (new Materialization())->windowed($aggregated, $release);
113    }
