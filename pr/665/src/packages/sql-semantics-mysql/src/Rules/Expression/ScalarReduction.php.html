19/**
20 * Resolves a scalar subquery that the server evaluates as its selected expression.
21 *
22 * Rule: MYSQL-SCALAR-REDUCTION-001. A SELECT with one expression and no input table, filter, aggregate or window can be
23 * reduced to that bound expression. In this form MySQL ignores LIMIT and OFFSET, including
24 * LIMIT 0; the expression retains its NULL attribute. The query is still derived before
25 * reduction so its diagnostic and binding checks run. Verified through SQL on MySQL 5.6.51
26 * and 8.4.7; no server implementation is used. A direct column is not reduced in MySQL
27 * 5.x. A tableless ROLLUP is retained through 8.4 and reduced in 9.1, with its nullable
28 * result attribute preserved; verified through the same statement on both releases.
29 * Source: https://dev.mysql.com/doc/refman/8.4/en/scalar-subqueries.html.
30 *
31 * @visibility SqlSemantics\Platform\MySql
32 */
33final class ScalarReduction
34{
35    /**
36     * Answers the existing selected expression when the scalar wrapper can be eliminated.
37     */
38    public function expression(Query $query, GrammarRelease $release = GrammarRelease::MySql847): ?Scalar
39    {
40        while ($query instanceof ParenthesizedQuery || $query instanceof QueryExpression) {
41            $query = $query instanceof ParenthesizedQuery ? $query->query : $query->body;
42        }
43        if (!$query instanceof Select || ($query->from !== null && !$query->from instanceof Dual) || $query->where !== null || $query->having !== null || $query->qualify !== null || $query->into !== null || count($query->items) !== 1 || (new RollupItems())->windowed($query) || ($release !== GrammarRelease::MySql910 && (new RollupItems())->rolls($query))) {
44            return null;
45        }
46        $item = $query->items[0];
47        $legacy = in_array($release, [GrammarRelease::MySql5651, GrammarRelease::MySql5744], true);
48
49        return $item instanceof SelectExpression && !($legacy && $item->expression instanceof ColumnUse) && !(new Aggregation())->aggregates([$item->expression]) ? $item->expression : null;
50    }
51
52    /**
53     * Tells whether a scalar result retains the selected expression's NULL attribute.
54     *
55     * With no table or filter, aggregate and window results retain that attribute too.
56     * A modern LIMIT can make the result empty and nullable; MySQL 5.x retains the
57     * attribute even when LIMIT discards the row. This describes the published result
58     * metadata, including that legacy behavior, rather than a minimum row count.
59     */
60    public function preservesNullability(Query $query, bool $legacy = false): bool
61    {
62        while ($query instanceof ParenthesizedQuery || $query instanceof QueryExpression) {
63            if ($query instanceof QueryExpression && !$legacy && $query->limit !== null) {
64                return false;
65            }
66            $query = $query instanceof ParenthesizedQuery ? $query->query : $query->body;
67        }
68
69        return $query instanceof Select && ($query->from === null || $query->from instanceof Dual) && $query->where === null && $query->having === null && $query->qualify === null && ($legacy || $query->limit === null);
70    }
71}
