30/**
31 * Decides whether an operand keeps its place when an expression is written without grouping parentheses.
32 *
33 * Rule: MYSQL-PRECEDENCE-001. The MySQL grammar stratifies expressions into
34 * expr (OR, XOR, AND, NOT, IS TRUE|FALSE|UNKNOWN), bool_pri (IS NULL, the
35 * comparisons and the quantified comparisons), predicate (IN, BETWEEN, LIKE,
36 * REGEXP, SOUNDS LIKE, MEMBER OF), bit_expr (`|`, `&`, `<<` `>>`, `+` `-`,
37 * `*` `/` `%` DIV MOD, `^`, and `± INTERVAL`) and simple_expr (concatenation
38 * under PIPES_AS_CONCAT, the prefix operators, BINARY, COLLATE and the closed
39 * primaries), and resolves the operators inside a level by the precedence
40 * declarations of `sql_yacc.yy`. Every expression has an opening level, the
41 * weakest level on its left edge, and a closing level, the weakest level on
42 * its right edge, on one scale whose constants this class declares. A prefix
43 * form has the opening level of its nonterminal; a postfix form closes at
44 * its own level; a closed form (parentheses, a call, a literal, a name) is
45 * closed on both edges. Two prefix forms extend their right operand further
46 * than their nonterminal suggests: `INTERVAL e unit + expr` takes everything
47 * above AND, and a user variable assignment `@v := expr` takes everything.
48 * An operator written after an operand takes the whole operand as its left
49 * operand unless it binds into the operand's right edge: the parser, which
50 * stands at the innermost form, binds it into the right operand of the
51 * innermost form on that edge whose right operand bound is weaker than the
52 * operator, provided that right operand can be the operator's left operand. A left operand therefore keeps its place
53 * when its top level admits the operator's left operand and the operator
54 * does not bind into it; for left association an operator binds into a
55 * form of its own level only from the right. A right operand keeps its
56 * place when its opening level is stronger than the operator. Terminates:
57 * each walk follows one edge of a finite tree.
58 * The scale was checked against the parser of every release with
59 * `SELECT <expression>` probes of each pair of forms.
60 * Shared rule: this class is the one precedence model of the package. A
61 * structure class of any family whose grammar slot takes an expression of
62 * a level (the substring of POSITION before IN, the document of JSON_VALUE,
63 * the row number of NTH_VALUE) checks its operand in its constructor with
64 * admits() for the slot level and, when an operator keyword follows the
65 * slot, absorbs() for that keyword's level, using the level constants
66 * declared here; no family keeps its own table of levels.
67 * Source: https://dev.mysql.com/doc/refman/8.4/en/operator-precedence.html,
68 * https://dev.mysql.com/doc/refman/8.4/en/sql-mode.html#sqlmode_high_not_precedence.
69 * Status: Implemented.
70 *
71 * @visibility SqlSemantics\Platform\MySql
72 */

175    /**
176     * Answers the edges of an operator form: its level, the bound of its right operand, the operand on the left edge and the operand on the right edge.
177     *
178     * An operator written after the form binds into its right operand when
179     * the operator is stronger than the bound: the bound is the form's own
180     * level for a binary operator, the level of the keyword for a prefix
181     * form, and the level just below the nonterminal of the right operand
182     * otherwise (the upper bound of BETWEEN is a predicate, the pattern of
183     * LIKE a simple_expr).
184     *
185     * @return array{int, int, Scalar|null, Scalar|null}|null Null for a form that is closed on both edges
186     */
187    public function edges(Scalar $expression): ?array
188    {
189        return match (true) {
190            $expression instanceof Logical => [$expression->operator->level(), $expression->operator->level(), $expression->left, $expression->right],
191            $expression instanceof Arithmetic => [$expression->operator->level(), $expression->operator->level(), $expression->left, $expression->right],
192            $expression instanceof Comparison => [self::BOOL_PRI, self::BOOL_PRI, $expression->left, $expression->right],
193            $expression instanceof Concatenation => [self::SIMPLE_EXPR, self::SIMPLE_EXPR, $expression->left, $expression->right],
194            $expression instanceof Not => [self::NEGATION, self::NEGATION, null, $expression->operand],
195            $expression instanceof Unary => [self::PRIMARY, self::PREFIX, null, $expression->operand],
196            $expression instanceof BinaryCast => [self::PRIMARY, self::PRIMARY, null, $expression->operand],
197            $expression instanceof IntervalAddition => [self::PRIMARY, self::NEGATION, null, $expression->operand],
198            $expression instanceof VariableAssignment => [self::PRIMARY, self::ASSIGNMENT, null, $expression->value],
199            default => $this->postfix($expression),
200        };
201    }

203    /**
204     * Answers the edges of a form whose operator follows its first operand.

209    {
210        return match (true) {
211            $expression instanceof TruthTest => [self::TRUTH, self::TRUTH, $expression->operand, null],
212            $expression instanceof NullTest => [self::BOOL_PRI, self::BOOL_PRI, $expression->operand, null],
213            $expression instanceof QuantifiedComparison => [self::BOOL_PRI, self::BOOL_PRI, $expression->operand, null],
214            $expression instanceof InList, $expression instanceof InQuery, $expression instanceof MemberOf => [self::PREDICATE, self::PREDICATE, $expression->operand, null],
215            $expression instanceof Between => [self::PREDICATE, self::BOOL_PRI, $expression->operand, $expression->high],
216            $expression instanceof Like => [self::PREDICATE, self::BIT_XOR, $expression->operand, $expression->escape ?? $expression->pattern],
217            $expression instanceof Regexp, $expression instanceof SoundsLike => [self::PREDICATE, self::PREDICATE, $expression->operand, $expression->pattern],
218            $expression instanceof IntervalArithmetic => [self::ADDITIVE, self::ADDITIVE, $expression->operand, null],
219            $expression instanceof Collated => [self::COLLATION, self::COLLATION, $expression->operand, null],
220            default => null,
221        };
222    }

224    /**
225     * Answers the weakest level on the left edge of an expression.
226     */
227    public function opening(Scalar $expression): int
228    {
229        $weakest = self::CLOSED;
230        for ($edges = $this->edges($expression); $edges !== null; $edges = $edges[2] === null ? null : $this->edges($edges[2])) {
231            $weakest = min($weakest, $edges[0]);
232        }
233
234        return $weakest;
235    }

249    /**
250     * Answers the level at which an expression stands as an operand: its operator level, the level of its prefix keyword, or closed.
251     *
252     * A trailing interval `x ± INTERVAL n unit` ends with the interval unit and
253     * takes the precedence of INTERVAL, so every bit_expr operator written
254     * after it applies to the whole of it: it stands at the strongest bit_expr level.
255     */
256    public function top(Scalar $expression): int
257    {
258        if ($expression instanceof IntervalArithmetic) {
259            return self::BIT_XOR;
260        }
261
262        return $this->edges($expression)[0] ?? self::CLOSED;
263    }

265    /**
266     * Tells whether an operator written after an expression would bind into the expression's right edge instead of taking the whole expression.
267     *
268     * The operator binds with the given precedence and takes a left operand
269     * of at least the given level.
270     */
271    public function absorbs(Scalar $expression, int $precedence, int $minimum): bool
272    {
273        for ($edges = $this->edges($expression); $edges !== null && $edges[3] !== null; $edges = $this->edges($edges[3])) {
274            if ($precedence > $edges[1] && $this->top($edges[3]) >= $minimum) {
275                return true;
276            }
277        }
278
279        return false;
280    }

282    /**
283     * Tells whether an expression keeps its place as the left operand of an operator of a precedence that takes a left operand of at least a level.
284     */
285    public function fits(Scalar $left, int $precedence, int $minimum): bool
286    {
287        return $this->top($left) >= $minimum && !$this->absorbs($left, $precedence, $minimum);
288    }

290    /**
291     * Tells whether an operand can stand in a slot of a grammar level that ends at a closing token such as `)` or `,`.
292     *
293     * The operand must be of a nonterminal the slot accepts, and nothing on
294     * its left edge may bind weaker than the slot. A slot followed by an
295     * operator keyword additionally needs that the keyword does not bind
296     * into the operand (`absorbs()`).
297     */
298    public function admits(Scalar $operand, int $slot): bool
299    {
300        return $this->opening($operand) >= $slot;
301    }
