38/**
39 * Applies the locking clauses of a query to the tables of the query block they lock and reports the clauses the server refuses.
40 *
41 * Rule: MYSQL-LOCKED-TABLES-001. The locking clauses written after a query
42 * lock the tables of its last query block: the block itself, the block in
43 * parentheses, the last operand of a set operation; the tables of a TABLE
44 * statement are its table, and VALUES has none. The clauses apply in written
45 * order, those written inside parentheses before those outside. A clause
46 * without OF locks the tables of the FROM clause of the block in written
47 * order, leaving out derived tables and common table references; a clause
48 * with OF locks each table it names, found by its correlation name or, when
49 * it has none, its table name, whatever its kind. A table a clause names
50 * that the block does not hold is reported (ER_UNRESOLVED_TABLE_LOCK), and
51 * so is a table a clause locks that is locked already
52 * (ER_DUPLICATE_TABLE_LOCK): by an earlier clause, as a derived table or a
53 * JSON table, which count as locked from the start, or, when the first
54 * query block of the query holds HIGH_PRIORITY, by that option, which
55 * locks every table. A clause without OF names the table by its
56 * correlation name or table name; a clause with OF names it as written.
57 * The first problem ends the check. MySQL 5.6 and 5.7, whose locking clauses
58 * name no tables, check none of this (verified on live 5.6.51 and 5.7.44
59 * servers). Terminates: the walk follows strictly
60 * smaller parts of the query and of the FROM clause. Source:
61 * https://dev.mysql.com/doc/refman/8.4/en/select.html ("Locking Read
62 * Concurrency with NOWAIT and SKIP LOCKED", "OF tbl_name"),
63 * https://dev.mysql.com/doc/refman/8.4/en/innodb-locking-reads.html.
64 * Status: Partial: HIGH_PRIORITY counts for the clauses of a query whose
65 * own first block holds it, not for those of its subqueries, derived
66 * tables, common tables and later set operands.
67 *
68 * @visibility SqlSemantics\Platform\MySql
69 */
70final class LockedTables
71{
72    /**
73     * Applies the clauses of a query block or query statement after those written inside it, and reports the first problem.
74     */
75    public function check(Derivation $derivation, Environment $outer, Select|QueryStatement $query): void
76    {
77        if ($query->locking === [] || in_array($derivation->context->profile->grammar, [GrammarRelease::MySql5651, GrammarRelease::MySql5744], true)) {
78            return;
79        }
80        [$block, $clauses, $common] = $this->chain($query);
81        $entries = $this->entries($derivation, $outer, $block, $common, $this->leading($query)->options ?? []);
82        $earlier = array_slice($clauses, 0, count($clauses) - count($query->locking));
83        foreach ($earlier as $clause) {
84            $entries = $this->apply($derivation, $clause, $entries)[0];
85        }
86        foreach ($query->locking as $clause) {
87            [$entries, $problem] = $this->apply($derivation, $clause, $entries);
88            if ($problem !== null) {
89                $derivation->report($problem);
90
91                return;
92            }
93        }
94    }

96    /**
97     * Applies one clause to the tables of the block and answers their new state with the first problem of the clause.
98     *
99     * @param list<array{VisibleRelation, bool, bool}> $entries Each table with whether a clause without OF locks it and whether it is locked
100     * @return array{list<array{VisibleRelation, bool, bool}>, Misuse|null}
101     */
102    public function apply(Derivation $derivation, LockingClause $clause, array $entries): array
103    {
104        if ($clause->tables === []) {
105            foreach ($entries as $index => [$relation, $listed, $locked]) {
106                if (!$listed) {
107                    continue;
108                }
109                if ($locked) {
110                    return [$entries, new Misuse(MisuseRule::RepeatedLockedTable, $relation->alias ?? $relation->name?->name)];
111                }
112                $entries[$index][2] = true;
113            }
114
115            return [$entries, null];
116        }
117        foreach ($clause->tables as $table) {
118            $found = null;
119            foreach ($entries as $index => [$relation]) {
120                if ($found === null && (new Projection())->admits($derivation, $relation, $table)) {
121                    $found = $index;
122                }
123            }
124            if ($found === null) {
125                return [$entries, new Misuse(MisuseRule::UnknownLockedTable, $table)];
126            }
127            if ($entries[$found][2]) {
128                return [$entries, new Misuse(MisuseRule::RepeatedLockedTable, $table)];
129            }
130            $entries[$found][2] = true;
131        }
132
133        return [$entries, null];
134    }

136    /**
137     * Answers the block a query locks, the clauses that lock it in the order they apply and the common tables the query defines on the way.
138     *
139     * @return array{Query|null, list<LockingClause>, list<Name>}
140     */
141    public function chain(Query $query): array
142    {
143        if ($query instanceof Select) {
144            return [$query, $query->locking, []];
145        }
146        if ($query instanceof QueryStatement) {
147            [$block, $clauses, $common] = $this->chain($query->query);
148
149            return [$block, [...$clauses, ...$query->locking], $common];
150        }
151        if ($query instanceof QueryExpression) {
152            [$block, $clauses, $common] = $this->chain($query->body);
153            $names = $query->with instanceof With ? array_map(static fn ($table): Name => $table->name, $query->with->tables) : [];
154
155            return [$block, $clauses, [...$common, ...$names]];
156        }
157
158        return match (true) {
159            $query instanceof ParenthesizedQuery => $this->chain($query->query),
160            $query instanceof SetOperation, $query instanceof OrderedSetOperation => $this->chain($query->right),
161            default => [$query, [], []],
162        };
163    }

165    /**
166     * Answers the first query block of a query, whose options hold HIGH_PRIORITY when it is given.
167     */
168    public function leading(Query|LeadingUnion $query): ?Select
169    {
170        return match (true) {
171            $query instanceof Select => $query,
172            $query instanceof QueryStatement, $query instanceof ParenthesizedQuery => $this->leading($query->query),
173            $query instanceof QueryExpression => $this->leading($query->body),
174            $query instanceof SetOperation, $query instanceof OrderedSetOperation, $query instanceof LeadingUnion => $this->leading($query->left),
175            default => null,
176        };
177    }

179    /**
180     * Answers the tables of a block in written order, each with whether a clause without OF locks it and whether it starts locked.
181     *
182     * @param list<Name> $common The common tables the query defines around the block
183     * @param list<SelectOption> $options The options of the first block of the query
184     * @return list<array{VisibleRelation, bool, bool}>
185     */
186    public function entries(Derivation $derivation, Environment $outer, ?Query $block, array $common, array $options): array
187    {
188        $priority = in_array(SelectOption::HighPriority, $options, true);
189        $terms = match (true) {
190            $block instanceof Select && $block->from !== null => $this->terms($block->from),
191            $block instanceof ExplicitTable => [$block],
192            default => [],
193        };
194        $entries = [];
195        foreach ($terms as $term) {
196            if ($term instanceof DerivedTable || $term instanceof JsonTable) {
197                $entries[] = [new VisibleRelation($term, new RowShape([]), $term->alias), $term instanceof JsonTable, true];
198                continue;
199            }
200            $name = $term->name();
201            $shared = $name->schema === null && ($outer->commonTable($name->name) !== null || $this->defines($derivation, $common, $name->name));
202            $entries[] = [new VisibleRelation($term, new RowShape([]), $term->alias(), $name), !$shared, $priority];
203        }
204
205        return $entries;
206    }

208    /**
209     * Tells whether a name names one of the common tables the query defines around the block.
210     *
211     * @param list<Name> $common
212     */
213    public function defines(Derivation $derivation, array $common, Name $name): bool
214    {
215        foreach ($common as $defined) {
216            if ($derivation->context->relationNames->equal($defined->value, $name->value)) {
217                return true;
218            }
219        }
220
221        return false;
222    }

224    /**
225     * Answers the tables, derived tables and JSON tables of a FROM clause in written order.
226     *
227     * @return list<TableReference|DerivedTable|JsonTable>
228     */
229    public function terms(Relation $relation): array
230    {
231        return match (true) {
232            $relation instanceof TableReference, $relation instanceof DerivedTable, $relation instanceof JsonTable => [$relation],
233            $relation instanceof TableList => array_merge([], ...array_map($this->terms(...), $relation->members)),
234            $relation instanceof JoinedTable => [...$this->terms($relation->left), ...$this->terms($relation->right)],
235            $relation instanceof NestedRelation, $relation instanceof OdbcJoin, $relation instanceof EscapedRelation => $this->terms($relation->relation),
236            default => [],
237        };
238    }
