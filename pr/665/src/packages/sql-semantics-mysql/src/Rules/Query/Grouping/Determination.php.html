23/**
24 * Finds the columns of a query block that its determined columns determine, the functional dependencies the server derives.
25 *
26 * For the grouping check (MYSQL-ONLY-FULL-GROUP-BY-001), a block determines the columns WHERE or
27 * an inner join compares with a value that reads no column of the block, the columns compared
28 * equal to a determined column, and every column of a table occurrence whose primary key, or
29 * unique key of NOT NULL columns, is determined; an outer join determines only columns of its
30 * inner side. A derived table or a common table determines the columns its query block derives
31 * from determined columns. Terminates: the determined set only grows and is bounded by the
32 * columns of the block, and a derived table is not entered again while its own dependencies are
33 * being found.
34 * Source: https://dev.mysql.com/doc/refman/8.4/en/group-by-handling.html.
35 *
36 * @visibility SqlSemantics\Platform\MySql
37 */
38final class Determination
39{
40    /**
41     * Extends a determined set by the equalities of WHERE and of the joins, by the keys of the tables and by the dependencies inside derived tables, to its fixed point.
42     *
43     * @param array<int, VisibleRelation> $relations
44     * @param array<string, true> $determined
45     * @param array<int, true> $within The query blocks of derived tables whose dependencies are being found, which are not entered again
46     * @return array<string, true>
47     */
48    public function closure(Select $select, array $relations, Facts $facts, array $determined, array $within = []): array
49    {
50        $parts = new BlockParts();
51        $dependencies = [];
52        foreach ($parts->conjuncts($select->where) as $conjunct) {
53            [$determined, $dependencies] = $this->equality($conjunct, $relations, $facts, null, $determined, $dependencies);
54        }
55        foreach ($parts->joins($select->from) as $join) {
56            [$determined, $dependencies] = $this->joined($join, $relations, $facts, $determined, $dependencies);
57        }
58        do {
59            $before = count($determined);
60            foreach ($dependencies as [$from, $to]) {
61                if (isset($determined[$from])) {
62                    $determined[$to] = true;
63                }
64            }
65            $determined = $this->keyed($relations, $facts, $determined, $within);
66        } while (count($determined) > $before);
67
68        return $determined;
69    }

71    /**
72     * Records what an equality determines: a column compared with a value that reads no column of the block, and each of two compared columns by the other; for the condition of an outer join, only columns of its inner side, by a value or by a column of its other side.
73     *
74     * @param array<int, VisibleRelation> $relations
75     * @param array<int, VisibleRelation>|null $inner The occurrences of the inner side of an outer join, or null for WHERE and inner joins
76     * @param array<string, true> $determined
77     * @param list<array{string, string}> $dependencies The pairs of columns where the first determines the second
78     * @return array{array<string, true>, list<array{string, string}>}
79     */
80    public function equality(Scalar $conjunct, array $relations, Facts $facts, ?array $inner, array $determined, array $dependencies): array
81    {
82        if (!$conjunct instanceof Comparison || ($conjunct->operator !== ComparisonOperator::Equal && $conjunct->operator !== ComparisonOperator::NullSafeEqual)) {
83            return [$determined, $dependencies];
84        }
85        $sides = [];
86        foreach ([$conjunct->left, $conjunct->right] as $side) {
87            $columns = (new ColumnReads())->columns($side, $relations, $facts, true);
88            $sides[] = [$columns, (new Matching())->unwrap($side) instanceof ColumnUse && count($columns) === 1 ? $columns[0] : null];
89        }
90        foreach ([[$sides[0], $sides[1]], [$sides[1], $sides[0]]] as [[$read, $from], [, $to]]) {
91            if ($to === null || ($inner !== null && !isset($inner[spl_object_id($to[1]->relation)]))) {
92                continue;
93            }
94            if ($read === []) {
95                $determined[$to[0]] = true;
96            } elseif ($from !== null && ($inner === null || !isset($inner[spl_object_id($from[1]->relation)]))) {
97                $dependencies[] = [$from[0], $to[0]];
98            }
99        }
100
101        return [$determined, $dependencies];
102    }

104    /**
105     * Records what the condition of a join determines: the equalities of ON, and the columns USING or NATURAL compares, each by the other; an outer join determines only columns of its inner side.
106     *
107     * @param array<int, VisibleRelation> $relations
108     * @param array<string, true> $determined
109     * @param list<array{string, string}> $dependencies The pairs of columns where the first determines the second
110     * @return array{array<string, true>, list<array{string, string}>}
111     */
112    public function joined(JoinedTable $join, array $relations, Facts $facts, array $determined, array $dependencies): array
113    {
114        $parts = new BlockParts();
115        $reads = new ColumnReads();
116        $left = $parts->members($join->left, $relations);
117        $right = $parts->members($join->right, $relations);
118        $inner = $join->operator->keepsLeft() ? $right : ($join->operator->keepsRight() ? $left : null);
119        foreach ($parts->conjuncts($join->on) as $conjunct) {
120            [$determined, $dependencies] = $this->equality($conjunct, $relations, $facts, $inner, $determined, $dependencies);
121        }
122        $names = array_map(static fn (Name $name): string => mb_strtolower($name->value), $join->using);
123        if ($join->operator->natural()) {
124            $names = array_keys(array_intersect_key($reads->named($left), $reads->named($right)));
125        }
126        $first = $reads->named($left);
127        $second = $reads->named($right);
128        foreach ($names as $name) {
129            foreach ($first[$name] ?? [] as $one) {
130                foreach ($second[$name] ?? [] as $two) {
131                    if (!$join->operator->keepsRight()) {
132                        $dependencies[] = [$one, $two];
133                    }
134                    if (!$join->operator->keepsLeft()) {
135                        $dependencies[] = [$two, $one];
136                    }
137                }
138            }
139        }
140
141        return [$determined, $dependencies];
142    }

144    /**
145     * Adds every column of a table occurrence whose determining key is determined, and the columns a derived table determines.
146     *
147     * @param array<int, VisibleRelation> $relations
148     * @param array<string, true> $determined
149     * @param array<int, true> $within The query blocks of derived tables whose dependencies are being found
150     * @return array<string, true>
151     */
152    public function keyed(array $relations, Facts $facts, array $determined, array $within = []): array
153    {
154        foreach ($relations as $id => $relation) {
155            $table = $relation->relation instanceof TableReference && $facts->covers($relation->relation) ? $facts->relation($relation->relation)->table : null;
156            if (!$table instanceof DeclaredTable) {
157                $determined = $this->derived($relation, $facts, $determined, $within);
158
159                continue;
160            }
161            foreach ($table->table->keys as $key) {
162                $complete = $key->determines();
163                foreach ($key->columns as $column) {
164                    $complete = $complete && isset($determined[$id . ':' . spl_object_id($column)]);
165                }
166                if ($complete) {
167                    foreach ([...$table->table->columns, ...array_map(static fn ($implicit) => $implicit->column, $table->table->implicit)] as $column) {
168                        $determined[$id . ':' . spl_object_id($column)] = true;
169                    }
170                }
171            }
172        }
173
174        return $determined;
175    }

177    /**
178     * Adds the columns of a derived table or a common table that its determined columns determine inside its query block.
179     *
180     * Inside a block without a grouping modifier, the determined columns
181     * determine the columns of the block they read, closed as the outer
182     * block is; a field is then determined when it reads only determined
183     * columns outside aggregates (a window function, RAND() and a subquery
184     * that reads no column of the block are determined too, verified on a
185     * live 8.4 server), and every field is once the GROUP BY expressions
186     * of the block are.
187     *
188     * @param array<string, true> $determined
189     * @param array<int, true> $within The query blocks of derived tables whose dependencies are being found
190     * @return array<string, true>
191     */
192    public function derived(VisibleRelation $relation, Facts $facts, array $determined, array $within): array
193    {
194        $select = (new BlockParts())->block($relation->relation, $facts);
195        if ($select === null || isset($within[spl_object_id($select)]) || $select->groupBy?->modifier !== null || !$facts->covers($select)) {
196            return $determined;
197        }
198        $fields = $this->fields($relation, $facts->query($select)->projection);
199        if ($fields === null) {
200            return $determined;
201        }
202        $reads = new ColumnReads();
203        $inner = (new BlockParts())->occurrences($select, $facts);
204        $seeds = [];
205        $expressions = [];
206        foreach ($fields as $position => $field) {
207            if (isset($determined[$reads->key(new ResolvedColumn($relation->relation, $relation->shape->slots[$position]))])) {
208                if ($field->expression !== null) {
209                    $expressions[] = $field->expression;
210                }
211                foreach ($this->shown($field, $inner, $facts) as [$key]) {
212                    $seeds[$key] = true;
213                }
214            }
215        }
216        $closed = $this->closure($select, $inner, $facts, $seeds, [...$within, spl_object_id($select) => true]);
217        $grouped = $this->grouped($select, $expressions, $closed, $inner, $facts);
218        foreach ($fields as $position => $field) {
219            if ($grouped || $this->depends($field, $inner, $facts, $closed)) {
220                $determined[$reads->key(new ResolvedColumn($relation->relation, $relation->shape->slots[$position]))] = true;
221            }
222        }
223
224        return $determined;
225    }

227    /**
228     * Answers the fields of a query block that the columns of its derived table show, by position, or null when a column shows no field.
229     *
230     * @param list<Field|OpenStar> $projection The output fields of the block
231     * @return array<int, Field>|null
232     */
233    public function fields(VisibleRelation $relation, array $projection): ?array
234    {
235        $fields = [];
236        foreach (array_keys($relation->shape->slots) as $position) {
237            $field = $projection[$position] ?? null;
238            if (!$field instanceof Field) {
239                return null;
240            }
241            $fields[$position] = $field;
242        }
243
244        return $fields;
245    }

247    /**
248     * Answers the column of the block a field shows: the column a select item names, or that of a star; none for any other expression.
249     *
250     * @param array<int, VisibleRelation> $relations The occurrences of the block
251     * @return list<array{string, ResolvedColumn}>
252     */
253    public function shown(Field $field, array $relations, Facts $facts): array
254    {
255        if ($field->expression !== null) {
256            return (new Matching())->unwrap($field->expression) instanceof ColumnUse ? (new ColumnReads())->columns($field->expression, $relations, $facts, true) : [];
257        }
258
259        return $field->resolution instanceof ResolvedColumn && isset($relations[spl_object_id($field->resolution->relation)]) ? [[(new ColumnReads())->key($field->resolution), $field->resolution]] : [];
260    }

262    /**
263     * Tells whether a block has GROUP BY and every GROUP BY expression is determined: written as one of the determined select items, or reading only determined columns.
264     *
265     * @param list<Scalar> $expressions The expressions of the determined select items
266     * @param array<string, true> $determined
267     * @param array<int, VisibleRelation> $relations The occurrences of the block
268     */
269    public function grouped(Select $select, array $expressions, array $determined, array $relations, Facts $facts): bool
270    {
271        if ($select->groupBy === null) {
272            return false;
273        }
274        $matching = new Matching();
275        foreach ($select->groupBy->items as $item) {
276            $expression = $matching->target($item->expression, $facts);
277            if (!$matching->listed($expression, $expressions, $facts) && $this->undetermined($expression, [], $determined, $relations, $facts) !== null) {
278                return false;
279            }
280        }
281
282        return true;
283    }

285    /**
286     * Tells whether a field of a query block reads only determined columns of the block, and none inside an aggregate.
287     *
288     * @param array<int, VisibleRelation> $relations
289     * @param array<string, true> $determined
290     */
291    public function depends(object $field, array $relations, Facts $facts, array $determined): bool
292    {
293        if (!$field instanceof Field) {
294            return false;
295        }
296        if ($field->expression === null) {
297            return $field->resolution instanceof ResolvedColumn && isset($determined[(new ColumnReads())->key($field->resolution)]);
298        }
299        if ((new Aggregation())->aggregates([$field->expression])) {
300            return false;
301        }
302        foreach ((new ColumnReads())->columns($field->expression, $relations, $facts, true) as [$key]) {
303            if (!isset($determined[$key])) {
304                return false;
305            }
306        }
307
308        return true;
309    }

311    /**
312     * Answers the first column an expression reads outside aggregates that the grouping does not determine, unless it is a GROUP BY expression.
313     *
314     * @param list<Scalar> $groups
315     * @param array<string, true> $determined
316     * @param array<int, VisibleRelation> $relations
317     */
318    public function undetermined(Scalar $expression, array $groups, array $determined, array $relations, Facts $facts): ?ResolvedColumn
319    {
320        if ((new Matching())->listed($expression, $groups, $facts)) {
321            return null;
322        }
323        foreach ((new ColumnReads())->columns($expression, $relations, $facts, false) as [$key, $column]) {
324            if (!isset($determined[$key])) {
325                return $column;
326            }
327        }
328
329        return null;
330    }
