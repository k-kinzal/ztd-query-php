25/**
26 * Derives the output columns of set operations and VALUES statements from the columns they combine.
27 *
28 * Rule: MYSQL-RESULT-SLOTS-001. A set operation is named by the columns of
29 * its first operand. The type of a column is the aggregation of the types
30 * at its position over the operands or rows (MYSQL-TYPE-AGGREGATION, the
31 * rule of the expression family). A UNION column can be NULL when it can in
32 * either operand, an INTERSECT column only when it can in both, an EXCEPT
33 * column when it can in the left operand. A VALUES column is named column_N,
34 * counting from zero, and can be NULL when one of its values can. Operands
35 * or rows of different lengths are reported, and a column one of them lacks
36 * is invalid; while the first operand has columns that are not known the
37 * output is open. Source: https://dev.mysql.com/doc/refman/8.4/en/union.html
38 * ("Result Set Column Names and Data Types"),
39 * https://dev.mysql.com/doc/refman/8.4/en/values.html. Status: Implemented.
40 *
41 * @visibility SqlSemantics\Platform\MySql
42 */
43final class ResultSlots
44{
45    /**
46     * Combines the outputs of the two operands of a set operation.
47     *
48     * @return list<Field|OpenStar>
49     */
50    public function combine(QueryFact $left, QueryFact $right, SetOperator $operator, Derivation $derivation): array
51    {
52        if (!$left->shape->complete()) {
53            return [new OpenStar([...$left->shape->missing, ...$right->shape->missing])];
54        }
55        $problem = null;
56        if ($right->shape->complete() && count($left->shape->slots) !== count($right->shape->slots)) {
57            $problem = new CountMismatch(CountedList::SetOperands, count($left->shape->slots), count($right->shape->slots));
58            $derivation->report($problem);
59        }
60        $fields = [];
61        foreach ($left->shape->slots as $position => $slot) {
62            $other = $right->shape->complete() ? $right->shape->slots[$position] ?? null : null;
63            if ($other !== null) {
64                $type = (new TypeAggregation())->aggregate([$slot->type, $other->type]);
65                $nullability = $this->nullability($operator, $slot->nullability, $other->nullability);
66            } elseif ($problem !== null) {
67                $type = new Invalid($problem);
68                $nullability = Nullability::Dependent;
69            } else {
70                $type = new Dependent($right->shape->missing);
71                $nullability = $operator === SetOperator::Except ? $slot->nullability : Nullability::Dependent;
72            }
73            $fields[] = new Field($position, new OutputSlot($slot->name, $type, $nullability, null, null, $slot->unnamed));
74        }
75
76        return $fields;
77    }
78
79    /**
80     * Answers whether a column of a set operation can be NULL from the facts of both operands.
81     */
82    public function nullability(SetOperator $operator, Nullability $left, Nullability $right): Nullability
83    {
84        if ($operator === SetOperator::Except) {
85            return $left;
86        }
87        if ($operator === SetOperator::Union) {
88            return $left->propagate($right);
89        }
90        if ($left === Nullability::NotNull || $right === Nullability::NotNull) {
91            return Nullability::NotNull;
92        }
93
94        return $left === Nullability::Nullable && $right === Nullability::Nullable ? Nullability::Nullable : Nullability::Dependent;
95    }

97    /**
98     * Derives every value of a VALUES statement and answers its output.
99     */
100    public function values(ValuesQuery $values, Derivation $derivation, Environment $outer): QueryFact
101    {
102        $columns = [];
103        $width = count($values->rows[0]->values);
104        foreach ($values->rows as $row) {
105            if (count($row->values) !== $width) {
106                $derivation->report(new CountMismatch(CountedList::ValueRows, $width, count($row->values)));
107            }
108            foreach ($row->values as $position => $value) {
109                $columns[$position][] = (new Operands())->single($derivation->scalar($value, new Environment($derivation->context, $outer)), $derivation);
110            }
111        }
112        $fields = [];
113        foreach ($columns as $position => $facts) {
114            $nullability = Nullability::NotNull;
115            foreach ($facts as $fact) {
116                $nullability = $nullability->propagate($fact->nullability);
117            }
118            $type = (new TypeAggregation())->aggregate(array_map(static fn (ScalarFact $fact) => $fact->type, $facts));
119            $fields[] = new Field($position, new OutputSlot(new Name('column_' . $position), $type, $nullability));
120        }
121
122        return new QueryFact($fields, $derivation->context->columnNames);
123    }
