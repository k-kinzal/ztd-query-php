21/**
22 * Derives the clauses written after a query: LIMIT, INTO and the locking clauses.
23 *
24 * Rule: MYSQL-TAIL-FACTS-001. The LIMIT operands and the INTO targets see no
25 * column of the query. An INTO variable list whose length differs from the
26 * number of columns of a query whose columns are all known is reported. A
27 * table named by `FOR UPDATE|SHARE OF` that names no table of the query
28 * block is reported; over a set operation the tables are not checked.
29 * Source: https://dev.mysql.com/doc/refman/8.4/en/select-into.html,
30 * https://dev.mysql.com/doc/refman/8.4/en/innodb-locking-reads.html
31 * ("ER_UNRESOLVED_TABLE_LOCK"). Status: Implemented.
32 *
33 * @visibility SqlSemantics\Platform\MySql
34 */
35final class TailFacts
36{
37    /**
38     * Derives the INTO targets and checks the locking clauses.
39     *
40     * @param list<LockingClause> $locking
41     * @param list<VisibleRelation>|null $visible The tables of the query block, or null over another query
42     */
43    public function derive(Derivation $derivation, Environment $outer, QueryFact $fact, ?IntoDestination $into, array $locking, ?array $visible): void
44    {
45        if ($into instanceof IntoVariables) {
46            foreach ($into->targets as $target) {
47                $derivation->scalar($target, new Environment($derivation->context, $outer));
48            }
49            if ($fact->shape->complete() && count($fact->shape->slots) !== count($into->targets)) {
50                $derivation->report(new CountMismatch(CountedList::IntoVariables, count($fact->shape->slots), count($into->targets)));
51            }
52        }
53        if ($visible === null) {
54            return;
55        }
56        foreach ($locking as $clause) {
57            foreach ($clause->tables as $table) {
58                $found = false;
59                foreach ($visible as $relation) {
60                    $found = $found || (new Projection())->admits($derivation, $relation, $table);
61                }
62                if (!$found) {
63                    $derivation->report(new Misuse(MisuseRule::UnknownLockedTable));
64                }
65            }
66        }
67    }
68
69    /**
70     * Derives the operands of a LIMIT clause, which see no column of their query.
71     */
72    public function limit(?Limit $limit, Derivation $derivation, Environment $outer): void
73    {
74        if (!$limit instanceof RowLimit) {
75            return;
76        }
77        $derivation->scalar($limit->count, new Environment($derivation->context, $outer));
78        if ($limit->offset !== null) {
79            $derivation->scalar($limit->offset, new Environment($derivation->context, $outer));
80        }
81    }
82}
