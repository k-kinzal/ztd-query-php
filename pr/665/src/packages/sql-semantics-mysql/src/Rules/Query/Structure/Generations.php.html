14/**
15 * Refuses repeated node occurrences and contradictory language requirements.
16 *
17 * Rule: MYSQL-QUERY-GENERATIONS-001. PROCEDURE ANALYSE, SQL_CACHE and the
18 * legacy trailing-clause forms end before 8.0. WITH, VALUES, TABLE, window
19 * specifications, JSON_TABLE, QUALIFY, INTERSECT and EXCEPT start in 8.x or later. A
20 * graph containing both sets cannot belong to any supported release. This
21 * includes legacy password calls, WEIGHT_STRING levels, table escapes and
22 * generated-column entry points nested inside modern statements. Every MySQL
23 * statement constructor and composite expression checks its entire graph.
24 * This is a constructor invariant; it does not decide whether a statement
25 * executes or whether a particular release accepts every feature.
26 * Source: https://dev.mysql.com/doc/refman/8.4/en/mysql-nutshell.html,
27 * https://dev.mysql.com/doc/refman/8.0/en/with.html,
28 * https://dev.mysql.com/doc/refman/8.4/en/set-operations.html.
29 * Status: Implemented.
30 *

32 * @example Refusing a modern stored-call interpretation inside a legacy query
33 *     $call = new \SqlSemantics\Platform\MySql\Statement\Call\FunctionCall(new \SqlSemantics\Statement\Identifier\Name('X'));

52     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When a node occurs twice or no language profile can contain the combination
53     */
54    public function check(Node $root): void
55    {
56        if ($root instanceof \SqlSemantics\Statement\Statement) {
57            (new \SqlSemantics\Platform\MySql\Rules\Dml\ValueRoles())->check($root);
58        }
59        $pending = [$root];
60        $seen = new SplObjectStorage();
61        $requirements = 0;
62        $releases = ReleaseDomain::ALL;
63        $languages = new ReleaseDomain();
64        $calls = [];
65        $transition = 0;
66        $modes = 0;
67        $spelling = null;
68        $namedParameters = false;
69        $lexical = new LexicalModes();
70        while ($pending !== []) {
71            $value = array_pop($pending);
72            if (is_array($value)) {
73                array_push($pending, ...array_values($value));
74            } elseif (is_object($value)) {
75                if (isset($seen[$value])) {
76                    Check::input(!$value instanceof Node, 'A statement node occurs at one position only.');
77                    continue;
78                }
79                $seen[$value] = true;
80                if ($value instanceof LanguageProfile) {
81                    Check::input($spelling === null || $spelling->compatibleWith($value), 'Observable spellings in one structure require one compatible language profile.');
82                    $spelling = $value;
83                }
84                $namedParameters = $namedParameters || ($value instanceof \SqlSemantics\Platform\MySql\Statement\Literal\Parameter && $value->marker !== '?');
85                if ($value instanceof \SqlSemantics\Platform\MySql\Statement\Call\FunctionCall || $value instanceof \SqlSemantics\Platform\MySql\Statement\Routine\Program\Expression\DeferredCall) {
86                    $calls[] = $value;
87                }
88                $requirements |= $this->requirement($value);
89                $transition |= $this->transition($value);
90                Check::input($transition !== 3, 'A structure cannot combine MySQL 5.6-only forms with forms introduced in MySQL 5.7 or later.');
91                $modes |= $lexical->requirement($value);
92                $lexical->check($modes);
93                Check::input($requirements !== 3, 'A query cannot combine features removed before MySQL 8.0 with features introduced in MySQL 8.0 or later.');
94                $releases &= $languages->of($value) & ($requirements === 1 ? 3 : ($requirements === 2 ? 508 : ReleaseDomain::ALL)) & ($transition === 1 ? 1 : ($transition === 2 ? 510 : ReleaseDomain::ALL));
95                Check::input($releases !== 0, 'The retained operands must share a supported MySQL release.');
96                if ($value instanceof Node) {
97                    array_push($pending, ...array_values(get_object_vars($value)));
98                }
99            }
100        }
101        Check::input((new Signatures())->compatible($calls, $requirements, $transition, $spelling, $releases), 'Native signatures and grammar operands must share a supported language interpretation.');
102        Check::input(!$namedParameters || $spelling === null || $spelling->parameters === ParameterStyle::Named, 'A named parameter cannot accompany a spelling that requires native parameters.');
103    }

108    public function requirement(object $value): int
109    {
110        if ($value instanceof LanguageProfile) {
111            return in_array($value->grammar, [GrammarRelease::MySql5651, GrammarRelease::MySql5744], true) ? 1 : 2;
112        }
113        return ((new LegacyForms())->requires($value) ? 1 : 0) | ((new ModernForms())->requires($value) ? 2 : 0);
114    }
