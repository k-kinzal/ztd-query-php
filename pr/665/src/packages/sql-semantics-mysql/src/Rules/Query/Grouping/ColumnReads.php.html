21/**
22 * Finds the columns of a query block an expression reads, each with a key that identifies one column of one table occurrence.
23 *
24 * The grouping check (MYSQL-ONLY-FULL-GROUP-BY-001) compares columns by these keys: every use
25 * of one column of one occurrence has the same key however it is qualified, and a column of a
26 * derived table or a common table is the column of the occurrence, not the one it selects.
27 * An alias of a select item reads the columns of the item. Aggregates without a window, and the
28 * arguments of GROUPING(), read their columns only when asked, since a grouped block may
29 * aggregate any column; so does the argument of ANY_VALUE(), which the check accepts as it is.
30 * Source: https://dev.mysql.com/doc/refman/8.4/en/group-by-handling.html.
31 *
32 * @visibility SqlSemantics\Platform\MySql
33 */
34final class ColumnReads
35{
36    /**
37     * Answers the columns of the block an expression reads, with their keys, outside aggregates unless asked to look into them.
38     *
39     * @param array<int, VisibleRelation> $relations
40     * @return list<array{string, ResolvedColumn}>
41     */
42    public function columns(Node $value, array $relations, Facts $facts, bool $aggregated): array
43    {
44        if (!$aggregated && ($this->aggregate($value) || $this->anyValue($value))) {
45            return [];
46        }
47        if ($value instanceof ColumnUse && $facts->covers($value)) {
48            $resolution = $facts->scalar($value)->resolution;
49            if ($resolution instanceof AliasTarget) {
50                return $resolution->field->expression === null ? [] : $this->columns($resolution->field->expression, $relations, $facts, $aggregated);
51            }
52
53            return $resolution instanceof ResolvedColumn && isset($relations[spl_object_id($resolution->relation)]) ? [[$this->key($resolution), $resolution]] : [];
54        }
55        $properties = get_object_vars($value);
56        $children = [];
57        array_walk_recursive($properties, static function ($property) use (&$children): void {
58            if ($property instanceof Node) {
59                $children[] = $property;
60            }
61        });
62        $found = [];
63        foreach ($children as $child) {
64            array_push($found, ...$this->columns($child, $relations, $facts, $aggregated));
65        }
66
67        return $found;
68    }
69
70    /**
71     * Tells whether a part of an expression aggregates the rows of a group: an aggregate without a window, or GROUPING().
72     */
73    public function aggregate(Node $value): bool
74    {
75        return $value instanceof Aggregate && $value->over === null
76            || $value instanceof GroupConcat && $value->over === null
77            || $value instanceof JsonObjectAggregate
78            || $value instanceof KeywordCall && $value->function === KeywordFunction::Grouping;
79    }
80
81    /**
82     * Tells whether a part of an expression is a call of ANY_VALUE(), whose argument a grouped block may read whatever columns it reads.
83     */
84    public function anyValue(Node $value): bool
85    {
86        return $value instanceof FunctionCall && $value->schema === null && strcasecmp($value->name->value, 'ANY_VALUE') === 0;
87    }
88
89    /**
90     * Answers the key of the column a resolution names, the same for every use of one column of one occurrence.
91     */
92    public function key(ResolvedColumn $resolution): string
93    {
94        $slot = $resolution->slot;
95        while ($slot->origin instanceof OutputSlot && $slot->column === null) {
96            $slot = $slot->origin;
97        }
98
99        return spl_object_id($resolution->relation) . ':' . spl_object_id($slot->declaration() ?? $slot);
100    }
101
102    /**
103     * Answers the keys of the columns of some occurrences by their lowercase names.
104     *
105     * @param array<int, VisibleRelation> $relations
106     * @return array<string, list<string>>
107     */
108    public function named(array $relations): array
109    {
110        $named = [];
111        foreach ($relations as $relation) {
112            foreach ($relation->shape->slots as $slot) {
113                if ($slot->name !== null) {
114                    $named[mb_strtolower($slot->name->value)][] = $this->key(new ResolvedColumn($relation->relation, $slot));
115                }
116            }
117        }
118
119        return $named;
120    }
