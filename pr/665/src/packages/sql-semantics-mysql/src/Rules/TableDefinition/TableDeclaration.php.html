31/**
32 * Derives the declaration a table or view definition provides.
33 *
34 * Rule: MYSQL-TABLE-DECLARATION-001. A CREATE TABLE declares one column per
35 * column definition, in order, with the type as written and the NULL fact
36 * of MYSQL-COLUMN-FLAGS-001; a column a table-level PRIMARY KEY names is NOT
37 * NULL. A generated column (VIRTUAL or STORED) is an ordinary column for
38 * reading and is declared generated: the server computes its value and
39 * accepts only DEFAULT for it in a write. An INVISIBLE
40 * column is found by name but not by `*` ("not part of SELECT *"): it is an
41 * implicit column of the declaration.
42 *
43 * CREATE TABLE ... SELECT: "columns named only in the CREATE TABLE part come
44 * first. Columns named in both parts or only in the SELECT part come after
45 * that. The data type of SELECT columns can be overridden by also specifying
46 * the column in the CREATE TABLE part." A column named in both parts takes
47 * its name from the SELECT part and the rest from the CREATE TABLE part.
48 * A column of the SELECT part takes the
49 * name, the type and the NULL fact of its output field and is not
50 * generated, also when the field reads a generated column; the column list
51 * stops, and the declaration is incomplete, at the first field that has no
52 * determined name or type or that an unexpanded star leaves open.
53 *
54 * CREATE VIEW: a view (RelationKind::View) with one column per output field,
55 * named by the column list when one is written, with the type and NULL fact
56 * of the field, never generated; incomplete under the same conditions and when the column
57 * list has another length. Every other declaration is a base table.
58 *
59 * CREATE TABLE ... LIKE: the columns of the source table, as new
60 * declarations with the same names, types, NULL facts and generation (the
61 * copy keeps the generation expressions); an undeclared
62 * source leaves the declaration empty and incomplete.
63 * Terminates: one pass over the elements and the fields.
64 * Source: https://dev.mysql.com/doc/refman/8.4/en/create-table.html,
65 * https://dev.mysql.com/doc/refman/8.4/en/create-table-select.html,
66 * https://dev.mysql.com/doc/refman/8.4/en/create-table-like.html,
67 * https://dev.mysql.com/doc/refman/8.4/en/create-view.html,
68 * https://dev.mysql.com/doc/refman/8.4/en/invisible-columns.html. Status: Implemented.
69 *
70 * @visibility SqlSemantics\Platform\MySql
71 */

151     * Answers the keys a definition declares over its columns: the primary key first, then the unique keys in written order.
152     *
153     * A column attribute PRIMARY KEY or UNIQUE, and SERIAL, declare a key of
154     * that one column; a PRIMARY KEY or UNIQUE element a key of its columns. A
155     * key part that names no column, or an expression, makes no key.
156     *
157     * @param list<array{ColumnDefinition|null, Column}> $columns
158     * @return list<Key>
159     */
160    public function keys(CreateTable $definition, array $columns, Comparison $comparison): array
161    {
162        $primary = [];
163        $unique = [];
164        $flags = new ColumnFlags();
165        foreach ($columns as [$element, $column]) {
166            if ($element === null) {
167                continue;
168            }
169            if ($flags->keyword($element->specification, ColumnKeyword::PrimaryKey)) {
170                $primary[] = new Key(true, [$column]);
171            }
172            if ($flags->keyword($element->specification, ColumnKeyword::Unique) || $flags->serial($element->specification)) {
173                $unique[] = new Key(false, [$column]);
174            }
175        }
176        foreach ($definition->elements as $element) {
177            if (!$element instanceof IndexDefinition || ($element->kind !== IndexKind::Primary && $element->kind !== IndexKind::Unique)) {
178                continue;
179            }
180            $parts = [];
181            foreach ($element->parts as $part) {
182                $match = $part instanceof ColumnPart && $part->length === null ? $this->find($part->column, $columns, $comparison) : null;
183                if ($match === null) {
184                    continue 2;
185                }
186                $parts[] = $columns[$match][1];
187            }
188            if ($element->kind === IndexKind::Primary) {
189                $primary[] = new Key(true, $parts);
190            } else {
191                $unique[] = new Key(false, $parts);
192            }
193        }
194
195        return [...$primary, ...$unique];

336     *

338     * not named take the name of their partition followed by sp0, sp1 and so on.
339     *
340     * @return list<Name>

342    public function partitions(CreateTable $definition): array
343    {
344        $partitioning = $definition->partitioning;
345        if (!$partitioning instanceof PartitionClause) {
346            return [];
347        }
348        $partitions = [];
349        foreach ($partitioning->definitions as $partition) {
350            $partitions[] = [$partition->name, array_map(static fn (SubpartitionDefinition $subpartition): Name => $subpartition->name, $partition->subpartitions)];
351        }
352        if ($partitions === []) {
353            $count = $partitioning->partitions === null ? 1 : max(1, (int) $partitioning->partitions->text);
354            for ($index = 0; $index < $count; $index++) {
355                $partitions[] = [new Name('p' . $index), []];
356            }
357        }
358        $subpartitions = $partitioning->subpartitioning?->count === null ? 0 : (int) $partitioning->subpartitioning->count->text;
359        $names = [];
360        foreach ($partitions as [$partition, $named]) {
361            $names[] = $partition;
362            if ($named === [] && $partitioning->subpartitioning !== null) {
363                for ($index = 0; $index < max(1, $subpartitions); $index++) {
364                    $named[] = new Name($partition->value . 'sp' . $index);
