22/**
23 * Raises the warnings of the deprecated parts of a data type, as the server raises them when it reads the type.
24 *
25 * Rule: MYSQL-TYPE-NOTICE-001. A display width of an integer type other than
26 * TINYINT(1) is deprecated, so is ZEROFILL, the digits of FLOAT and DOUBLE,
27 * UNSIGNED on DECIMAL, FLOAT and DOUBLE, YEAR(4), and the character set name
28 * utf8; each raises its warning (MYSQL-DEPRECATION-001) in that order.
29 * Terminates: no recursion. Source:
30 * https://dev.mysql.com/doc/refman/8.4/en/numeric-type-attributes.html.
31 * Status: Implemented.
32 *
33 * @visibility SqlSemantics\Platform\MySql
34 */
35final class TypeNotices
36{
37    /**
38     * Raises the warnings of a data type.
39     */
40    public function type(TypeName $type, Derivation $derivation): void
41    {
42        if ($type instanceof Integral || $type instanceof Decimal || $type instanceof Floating) {
43            $this->numeric($type, $derivation);
44        }
45        if ($type instanceof Temporal && $type->kind === TemporalKind::Year && $type->precision === '4') {
46            Deprecation::raise(Deprecated::YearWidth, $derivation);
47        }
48        if ($type instanceof Character && $type->national) {
49            Deprecation::raise(Deprecated::National, $derivation);
50        }
51        if ($type instanceof Character || $type instanceof Enumeration) {
52            $this->charset($type->charset, $derivation);
53        }
54    }
55
56    /**
57     * Raises the warnings of a numeric type: its display width, ZEROFILL, the digits of FLOAT and DOUBLE, and UNSIGNED on a type with a fraction.
58     */
59    public function numeric(Integral|Decimal|Floating $type, Derivation $derivation): void
60    {
61        $modifiers = $type->modifiers;
62        if ($type instanceof Integral && $type->width !== null && !($type->kind === IntegralKind::TinyInt && $type->width === '1') && !in_array(NumericModifier::Zerofill, $modifiers, true)) {
63            Deprecation::raise(Deprecated::DisplayWidth, $derivation);
64        }
65        if (in_array(NumericModifier::Zerofill, $modifiers, true)) {
66            Deprecation::raise(Deprecated::Zerofill, $derivation);
67        }
68        if ($type instanceof Floating && $type->scale !== null) {
69            Deprecation::raise(Deprecated::FloatingDigits, $derivation);
70        }
71        if (($type instanceof Decimal || $type instanceof Floating) && in_array(NumericModifier::Unsigned, $modifiers, true)) {
72            Deprecation::raise(Deprecated::UnsignedFraction, $derivation);
73        }
74    }
75
76    /**
77     * Raises the warning of the character set name utf8.
78     */
79    public function charset(?CharsetAttribute $charset, Derivation $derivation): void
80    {
81        if ($charset !== null && $charset->charset !== null && strtolower($charset->charset->value) === 'utf8') {
82            Deprecation::raise(Deprecated::Utf8Alias, $derivation);
83        }
84    }
85}
