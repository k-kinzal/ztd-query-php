33/**
34 * Resolves a column name the way MySQL does: the columns of a query first, then its select list aliases.
35 *
36 * In a stored program a name first denotes a parameter, a local variable or a column of the row
37 * of a trigger (MYSQL-PROGRAM-VARIABLE-LOOKUP-001).
38 * Rule: MYSQL-COLUMN-LOOKUP-001. The lookup starts at the innermost query and
39 * moves outwards one query at a time. At one query the relation occurrences
39 * the qualifier admits are searched as in CORE-COLUMN-LOOKUP-001: one known
40 * the qualifier admits (MYSQL-RELATION-QUALIFIER-001 when it writes a
41 * database) are searched as in CORE-COLUMN-LOOKUP-001: one known
42 * slot resolves, several are ambiguous, and an incompletely known occurrence
43 * at that or a nearer query makes the outcome conditional. When no slot has
44 * the name, an unqualified name is searched among the select list aliases
45 * the position may use (GROUP BY, HAVING, ORDER BY): one item, or several
46 * items computing the same expression, resolve to that item; several
47 * different items are ambiguous (ER_NON_UNIQ_ERROR). While an incompletely
48 * known occurrence could still own the name, the alias is not chosen and the
49 * outcome is conditional; so it is while a select list item whose name
50 * depends on missing inputs (OutputSlot::$unnamed) could have the name.
51 * Only a name found neither way continues outwards.
52 * At a HAVING position (MYSQL-HAVING-SCOPE-001) the GROUP BY columns and
53 * the select list are searched first (MYSQL-HAVING-REFERENCE-001); a name
54 * written there outside set functions never sees the columns of the FROM
53 * clause of its own query (ER_BAD_FIELD_ERROR "in 'having clause'"), and a
54 * name of a nested query sees them only when the enclosing block neither
55 * groups, aggregates nor is DISTINCT (Item_ref::fix_fields,
56 * Item_field::fix_outer_field). A select list star over an incompletely
55 * clause of its own query (ER_BAD_FIELD_ERROR "in 'having clause'"). An
56 * aggregate argument, including a nested query in that argument, prefers
57 * the input columns of an enclosing HAVING block. An ordinary nested use
58 * normally sees those inputs only when the block neither groups, aggregates nor is DISTINCT;
59 * eligible merged inputs and MySQL 9.1 also expose them for subsequent grouping checks.
60 * A select list star over an incompletely
61 * known occurrence, and a GROUP BY or select list column of the name that
62 * belongs to such an occurrence, leave a name not found there conditional.
63 * Terminates: the scopes form a finite chain. Source:
64 * https://dev.mysql.com/doc/refman/8.4/en/select.html ("For GROUP BY or
65 * HAVING clauses, it searches the FROM clause before searching in the
66 * select_expr values"), https://dev.mysql.com/doc/refman/8.4/en/problems-with-alias.html.
67 * Status: Implemented.
68 *
69 * @visibility SqlSemantics\Platform\MySql
70 */

74     * Resolves a column name written with an optional relation qualifier.
75     */
76    public function find(Environment $environment, Name $column, ?QualifiedName $qualifier = null): Resolution
77    {
78        $variable = (new ProgramVariables())->find($environment, $column, $qualifier);
79        if ($variable !== null) {
80            return $variable;
81        }
82        $lookup = new ColumnLookup();
83        $open = [];
84        $depth = 0;
85        $argument = false;
86        for ($scope = $environment; $scope !== null; $scope = $scope->outer) {
78            $row = (new HavingScope())->row($scope);
87            $argument = $argument || $scope->aggregateArgument;
88            $row = (new HavingLookup())->row($scope, $column, $qualifier, $depth, $argument);
89            if ($row !== null) {
90                $result = (new ResultReferences())->find($row, $scope, $column, $qualifier, $depth);
91                if ($result instanceof ConditionalColumn) {
92                    return $this->undecided($column, $result->candidates, $open, $result->missing);
93                }
94                if ($result !== null) {
95                    return $open === [] ? $result : $lookup->conditional($column, $result instanceof ResolvedColumn ? [$result] : [], $open);
96                }
97                $open = [...$open, ...$this->unlisted($row, $scope, $column)];
88                if ($depth === 0 || $row->grouped) {
98                if ((new HavingLookup())->blocked($row, $scope, $depth)) {
99                    $depth++;
100                    continue;
101                }
102            }
93            $level = new LookupLevel($scope, $column, $qualifier, $depth);
103            $level = (new HavingLookup())->input($scope, $column, $qualifier, $depth);
104            $found = $level->found();
105            $open = [...$open, ...$level->open()];
106            if ($found !== []) {
107                if ($open !== []) {
108                    return $lookup->conditional($column, $found, $open);
109                }
110
111                return count($found) === 1 ? $found[0] : new AmbiguousColumn($column, $found);
112            }
113            $unnamed = $qualifier === null ? $this->unnamed($scope->aliases) : [];
114            if ($unnamed !== []) {
115                return $this->undecided($column, [], $open, $unnamed);
116            }
117            $aliases = $qualifier === null ? $scope->aliased($column) : [];
118            if ($aliases !== []) {
109                return $open === [] ? $this->alias($column, $aliases) : $lookup->conditional($column, [], $open);
119                return $open === [] ? $this->alias($column, $aliases, $depth) : $lookup->conditional($column, [], $open);
120            }
121            $depth++;
122        }
123
114        return $open === [] ? new MissingColumn($column, $qualifier) : $lookup->conditional($column, [], $open);
124        return $open === [] ? (new HavingLookup())->missing($environment, $column, $qualifier) : $lookup->conditional($column, [], $open);
125    }

127    /**
128     * Answers the facts of a column name: those of the column or select list item it resolves to, else the outcome as the cause.
129     */
130    public function fact(Environment $environment, Name $column, ?QualifiedName $qualifier = null): ScalarFact
131    {
132        $resolution = $this->find($environment, $column, $qualifier);
133        if ($resolution instanceof ResolvedColumn) {
134            return new ScalarFact($resolution->slot->type, $resolution->slot->nullability, $resolution);
135        }
136        if ($resolution instanceof AliasTarget) {
137            return new ScalarFact($resolution->field->type, $resolution->field->nullability, $resolution);
138        }
139        if ($resolution instanceof ConditionalColumn) {
140            return new ScalarFact(new Dependent($resolution->missing), Nullability::Dependent, $resolution);
141        }
142        Check::invariant($resolution instanceof Diagnostic, 'A column lookup resolves, depends on missing inputs, or reports a problem.');
143
144        return new ScalarFact(new Invalid($resolution), Nullability::Dependent, $resolution);
145    }

147    /**
148     * Tells whether a qualifier is NEW or OLD and a trigger body makes that row visible at the position.
149     *
150     * The parser reads `NEW.x` and `OLD.x` in a trigger body as a column of
151     * the row (Item_trigger_field), whatever the position.
152     */
153    public function row(Environment $environment, QualifiedName $qualifier): bool
154    {
155        $word = $qualifier->name->value;
156        if ($qualifier->schema !== null || (strcasecmp($word, 'NEW') !== 0 && strcasecmp($word, 'OLD') !== 0)) {
157            return false;
158        }
159        for ($scope = $environment; $scope !== null; $scope = $scope->outer) {
160            foreach ($scope->relations as $relation) {
161                if ($relation->alias !== null && strcasecmp($relation->alias->value, $word) === 0) {
162                    return true;
163                }
164            }
165        }
166
167        return false;
168    }

170    /**
171     * Answers the incompletely known occurrences that may give a HAVING position a column of the name unseen.
172     *
173     * They are the occurrences a star of the select list may list, and the
174     * occurrences that may own a GROUP BY or select list column of the
175     * name that no known occurrence decides.
176     *
177     * @return list<VisibleRelation>
178     */
179    public function unlisted(GroupedRow $row, Environment $scope, Name $column): array
180    {
181        foreach ($row->selected as $item) {
182            if ($item instanceof OpenStar) {
183                return array_values(array_filter($scope->relations, static fn (VisibleRelation $relation): bool => !$relation->shape->complete()));
184            }
185        }
186        $owners = [];
187        foreach ($row->undecided as $undecided) {
188            if ($scope->context->columnNames->equal($undecided->name->value, $column->value)) {
189                array_push($owners, ...$undecided->relations);
190            }
191        }
192        $open = [];
193        for ($level = $scope; $level !== null; $level = $level->outer) {
194            foreach ($level->relations as $relation) {
195                if (in_array($relation->relation, $owners, true)) {
196                    $open[] = $relation;
197                }
198            }
199        }
200
201        return $open;
202    }

220    /**
221     * Builds the conditional outcome of a name that incompletely known occurrences or undecided item names may own.
222     *
223     * @param list<ResolvedColumn> $candidates
224     * @param list<VisibleRelation> $open
225     * @param list<MissingInput> $unnamed The inputs the names of select list items depend on
226     */
227    public function undecided(Name $column, array $candidates, array $open, array $unnamed): ConditionalColumn
228    {
229        $relations = [];
230        $missing = [];
231        foreach ($open as $relation) {
232            $relations[] = $relation->relation;
233            array_push($missing, ...LookupLevel::undecided($relation));
234        }
235
236        return new ConditionalColumn($column, $candidates, $relations, [...$missing, ...$unnamed]);
237    }

239    /**
240     * Chooses the select list item an alias names: the only one, or the first of items that compute the same expression.
241     *
242     * @param non-empty-list<Field> $fields
243     */
234    public function alias(Name $column, array $fields): AliasTarget|AmbiguousAlias
244    public function alias(Name $column, array $fields, int $depth = 0): AliasTarget|AmbiguousAlias
245    {
246        $first = $fields[0];
247        $equivalence = new Equivalence();
248        foreach ($fields as $field) {
249            if ($field->expression === null || $first->expression === null || $equivalence->difference($first->expression, $field->expression) !== null) {
240                return count($fields) === 1 ? new AliasTarget($first) : new AmbiguousAlias($column, $fields);
250                return count($fields) === 1 ? new AliasTarget($first, $depth) : new AmbiguousAlias($column, $fields);
251            }
252        }
253
244        return new AliasTarget($first);
254        return new AliasTarget($first, $depth);
255    }
