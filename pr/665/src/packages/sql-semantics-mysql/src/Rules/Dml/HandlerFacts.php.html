17/**
18 * Derives the facts of HANDLER ... READ.
19 *
20 * Rule: MYSQL-HANDLER-READ-001. The handler is an open relation under its
21 * name (MYSQL-HANDLER-TABLE-001); WHERE sees it, the key values and LIMIT
22 * see no relation. The statement returns the rows read: every column of the
23 * handler's table, which is session state, so the output is an open star
24 * that depends on that state. Terminates: one pass over the finite parts.
25 * Source: https://dev.mysql.com/doc/refman/8.4/en/handler.html. Status: Implemented.
26 *
27 * @visibility SqlSemantics\Platform\MySql
28 */
29final class HandlerFacts
30{
31    /**
32     * Derives a read of a handler and records its rows as the output.
33     *
34     * @param list<Scalar> $values The key values of an index seek
35     */
36    public function read(OpenHandler $handler, array $values, ?Scalar $where, ?Limit $limit, Derivation $derivation): void
37    {
38        $base = $derivation->environment();
39        $fact = $derivation->relation($handler, $base);
40        foreach ($values as $value) {
41            $derivation->scalar($value, $base);
42        }
43        if ($where !== null) {
44            $derivation->scalar($where, new Environment($derivation->context, $base, [new VisibleRelation($handler, $fact->shape, $handler->name)]));
45        }
46        (new TailFacts())->limit($limit, $derivation, $base);
47        $derivation->output(new QueryFact([new OpenStar($fact->shape->missing)], $derivation->context->columnNames));
48    }
49}
