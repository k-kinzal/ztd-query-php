20/**
21 * Derives the parts that the data manipulation statements share: written columns, values and assignments.
22 *
23 * Rule: MYSQL-DML-SCOPE-001. A written column is a column use resolved
24 * among the tables the statement writes; it becomes a field with the type
24 * among the tables the statement writes, never a variable of a stored
25 * program (MYSQL-PROGRAM-VARIABLE-LOOKUP-001); it becomes a field with the type
26 * and nullability of the column. A value is derived where the statement
27 * says; the value DEFAULT is derived where the only field of the
28 * environment is the column it is assigned to (MYSQL-DML-DEFAULT-001).
29 * Where the server requires distinct written columns (INSERT column lists
30 * and INSERT ... SET), a column resolved twice to the same column, or named
31 * twice when it does not resolve, is reported. Terminates: one pass over
32 * the finite lists. Source: https://dev.mysql.com/doc/refman/8.4/en/insert.html,
33 * https://dev.mysql.com/doc/refman/8.4/en/update.html. Status: Implemented.
34 *
35 * @visibility SqlSemantics\Platform\MySql
36 */
37final class WriteScope
38{
39    /**
40     * Turns the fact of a written column into the field the column stands for.
41     */
42    public function field(int $position, ColumnUse $column, ScalarFact $fact): Field
43    {
44        $origin = $fact->resolution instanceof ResolvedColumn ? $fact->resolution->slot : null;
45
46        return new Field($position, new OutputSlot($column->name, $fact->type, $fact->nullability, null, $origin), null, $fact->resolution);
47    }
48
49    /**
50     * Answers the environment that names the columns a statement writes: the same relations, where a name is never a variable of a stored program.
51     */
52    public function written(Environment $environment): Environment
53    {
54        return new Environment($environment->context, $environment->outer, $environment->relations, $environment->commonTables, $environment->aliases, true);
55    }
56
57    /**
58     * Derives a value assigned to a column.
59     */
60    public function value(Scalar $value, Field $column, Derivation $derivation, Environment $environment): ScalarFact
61    {
62        if ($value instanceof DefaultRequest) {
63            return $derivation->scalar($value, new Environment($derivation->context, $environment, [], [], [$column]));
64        }
65
66        return (new Operands())->single($derivation->scalar($value, $environment), $derivation);
67    }
68
69    /**
70     * Derives assignments: each column where the written tables are visible, each value in its environment.
71     *
72     * @param list<Assignment> $assignments
73     * @param bool $distinct Whether the server requires distinct columns
74     * @return list<Field> The assigned columns in written order
75     */
76    public function assign(array $assignments, Derivation $derivation, Environment $columns, Environment $values, bool $distinct): array
77    {
78        $fields = [];
79        foreach ($assignments as $position => $assignment) {
71            $field = $this->field($position, $assignment->column, $derivation->scalar($assignment->column, $columns));
80            $field = $this->field($position, $assignment->column, $derivation->scalar($assignment->column, $this->written($columns)));
81            $this->value($assignment->value, $field, $derivation, $values);
82            $fields[] = $field;
83        }
84        if ($distinct) {
85            $this->distinct($fields, $derivation);
86        }
87
88        return $fields;
89    }
90
91    /**
92     * Reports each column written again.
93     *
94     * @param list<Field> $fields
95     */
96    public function distinct(array $fields, Derivation $derivation): void
97    {
98        $seen = [];
99        foreach ($fields as $field) {
100            $resolution = $field->resolution;
101            $key = $resolution instanceof ResolvedColumn ? 'slot:' . spl_object_id($resolution->slot) : 'name:' . $derivation->context->columnNames->fold($field->name->value ?? '');
102            if (isset($seen[$key]) && $field->name !== null) {
103                $derivation->report(new DuplicateColumn($field->name));
104            }
105            $seen[$key] = true;
106        }
107    }
108}
