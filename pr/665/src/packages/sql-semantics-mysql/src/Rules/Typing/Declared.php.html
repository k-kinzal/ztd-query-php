43 * Resolves a declared data type into the domain of the column it declares.
44 *
45 * A string type without a character set takes the default collation of its table; the display
46 * lengths are those the server reports for columns of each type.

60    /**
61     * Answers the rules for the columns of a table definition.
62     *
63     * A string column without a character set takes the collation the table options name, else
64     * the default collation of the schema the table is created in.
65     */
66    public static function table(CreateTable $definition, AnalysisContext $context): self
67    {
68        $release = $context->profile->grammar;
69        $collation = Settings::of($context)->schema($definition->name->schema->value ?? $context->searchPath[0]->value);
70        foreach ($definition->options as $option) {
71            if ($option instanceof CharsetOption && $option->charset->name !== null) {
72                $collation = Charset::named($option->charset->name->value)?->defaultCollation($release) ?? $collation;
73            }
74            if ($option instanceof CollationOption && $option->collation->name !== null) {
75                $collation = Collation::named($option->collation->name->value) ?? $collation;
76            }
77        }
78
79        return new self($collation, $release);
80    }

82    /**
83     * Resolves the type of a column: its data type in the collation its COLLATE clause names; a string column has no decimals.
84     */
85    public function column(ColumnSpecification $specification): Domain
86    {
87        $collation = null;
88        $attributes = $specification instanceof OrdinaryColumn || $specification instanceof GeneratedColumn ? $specification->attributes : [];
89        foreach ($attributes as $attribute) {
90            if ($attribute instanceof CollateAttribute && $attribute->collation->name !== null) {
91                $collation = Collation::named($attribute->collation->name->value) ?? $collation;
92            }
93        }
94        if ($specification instanceof GeneratedColumn && $specification->collation?->name !== null) {
95            $collation = Collation::named($specification->collation->name->value) ?? $collation;
96        }
97        $domain = $this->domain($specification->dataType(), $collation);
98        if ($domain->kind !== Kind::String && $domain->kind !== Kind::Json) {
99            return $domain;
100        }
101
102        return new Domain($domain->kind, $domain->field, $domain->length, 0, false, $domain->collation, $domain->members, $domain->coercibility);
103    }

105    /**
106     * Resolves a type into a domain, in the collation a COLLATE clause names when one does; a type without a domain is a long binary string.
107     */
108    public function domain(TypeDescriptor $type, ?Collation $collation = null): Domain
109    {
110        return match (true) {
111            $type instanceof Integral => $this->integral($type),
112            $type instanceof Decimal => Domain::decimal($type->precision === null ? 10 : (int) $type->precision, $type->scale === null ? 0 : (int) $type->scale, in_array(NumericModifier::Unsigned, $type->modifiers, true) || in_array(NumericModifier::Zerofill, $type->modifiers, true)),
113            $type instanceof Floating => $this->floating($type),
114            $type instanceof Character => $this->character($type, $collation),
115            $type instanceof Binary => $this->binary($type),
116            $type instanceof Temporal => $this->temporal($type),
117            $type instanceof Enumeration => $this->enumeration($type, $collation),
118            $type instanceof Elementary => $this->elementary($type),
119            default => new Domain(Kind::String, Field::Blob, 4294967295, Domain::NOT_FIXED, false, Collation::binary()),
120        };
121    }

137     * Resolves an integer type, which reports a declared display width for itself.
138     */
139    public function integral(Integral $type): Domain
140    {
141        $unsigned = in_array(NumericModifier::Unsigned, $type->modifiers, true) || in_array(NumericModifier::Zerofill, $type->modifiers, true);
142        [$field, $length] = match ($type->kind) {
143            IntegralKind::TinyInt => [Field::Tiny, $unsigned ? 3 : 4],
144            IntegralKind::SmallInt => [Field::Short, $unsigned ? 5 : 6],
145            IntegralKind::MediumInt => [Field::Int24, $unsigned ? 8 : 9],
146            IntegralKind::Int => [Field::Long, $unsigned ? 10 : 11],
147            IntegralKind::BigInt => [Field::LongLong, 20],
148        };
149
150        return Domain::column($field, $type->width === null ? $length : (int) $type->width, $unsigned);
151    }

154     * Resolves FLOAT, REAL and DOUBLE, with or without a precision and scale.
155     */
156    public function floating(Floating $type): Domain
157    {
158        $single = $type->kind === FloatingKind::Float && ($type->precision === null || $type->scale !== null || (int) $type->precision <= 24);
159        $unsigned = in_array(NumericModifier::Unsigned, $type->modifiers, true);
160        $decimals = $type->scale === null ? Domain::NOT_FIXED : (int) $type->scale;
161        $length = $type->scale !== null ? (int) $type->precision : ($single ? 12 : 22);
162
163        return new Domain(Kind::Double, $single ? Field::Float : Field::Double, $length, $decimals, $unsigned);
164    }

169    public function character(Character $type, ?Collation $collation): Domain
170    {
171        $collation ??= $this->charset($type->charset, $type->national ? Collation::known('utf8mb3_general_ci') : $this->collation);
172        [$field, $length] = match ($type->kind) {
173            CharacterKind::Char => [Field::String, $type->length === null ? 1 : (int) $type->length],
174            CharacterKind::VarChar, CharacterKind::CharVarying => [Field::VarString, (int) $type->length],
175            CharacterKind::TinyText => [Field::Blob, 255],
176            CharacterKind::Text => [Field::Blob, $type->length === null ? 65535 : $this->textLength((int) $type->length, $collation)],
177            CharacterKind::MediumText, CharacterKind::Long, CharacterKind::LongVarChar, CharacterKind::LongCharVarying => [Field::Blob, 16777215],
178            CharacterKind::LongText => [Field::Blob, 4294967295],
179        };
180
181        return Domain::string($length, $collation, $field);
182    }

187    public function textLength(int $characters, Collation $collation): int
188    {
189        $bytes = $characters * $collation->charset->maxLength;
190
191        return $bytes <= 255 ? 255 : ($bytes <= 65535 ? 65535 : ($bytes <= 16777215 ? 16777215 : 4294967295));

194    /**
195     * Resolves the collation a character set attribute names.
196     */
197    public function charset(?CharsetAttribute $attribute, Collation $collation): Collation
198    {
199        if ($attribute === null) {
200            return $collation;
201        }
202
203        return match ($attribute->form) {
204            CharsetForm::Ascii => Collation::known('ascii_general_ci'),
205            CharsetForm::Unicode => Collation::known('utf8mb4_0900_ai_ci'),
206            CharsetForm::Byte => Collation::binary(),
207            CharsetForm::Binary => Collation::named($collation->charset->name . '_bin') ?? Collation::known('utf8mb4_bin'),
208            CharsetForm::Named, CharsetForm::CharacterSet => $attribute->charset === null ? $collation : (Charset::named($attribute->charset->value)?->defaultCollation($this->release) ?? $collation),
209        };
210    }

218            BinaryKind::Binary => [Field::String, $type->length === null ? 1 : (int) $type->length],
219            BinaryKind::VarBinary => [Field::VarString, (int) $type->length],

230     * Resolves a date, time, datetime, timestamp or year type.
231     */
232    public function temporal(Temporal $type): Domain
233    {
234        $decimals = $type->precision === null ? 0 : (int) $type->precision;
235        $fraction = $decimals > 0 ? $decimals + 1 : 0;
236
237        return match ($type->kind) {
238            TemporalKind::Date => new Domain(Kind::Date, Field::Date, 10),
239            TemporalKind::Time => new Domain(Kind::Time, Field::Time, 10 + $fraction, $decimals),
240            TemporalKind::DateTime => new Domain(Kind::DateTime, Field::DateTime, 19 + $fraction, $decimals),
241            TemporalKind::Timestamp => new Domain(Kind::DateTime, Field::Timestamp, 19 + $fraction, $decimals),
242            TemporalKind::Year => new Domain(Kind::Year, Field::Year, 4, 0, true),
243        };
244    }

246    /**
247     * Resolves ENUM and SET, whose length counts the characters of the members as the statement writes them.
248     */
249    public function enumeration(Enumeration $type, ?Collation $collation): Domain
250    {
251        $collation ??= $this->charset($type->charset, $this->collation);
252        $members = array_map(static fn ($member): string => $member->value, $type->members);
253        $lengths = array_map(static fn (string $member): int => Charset::known('utf8mb4')->length($member), $members);
254        $length = $type->kind === EnumerationKind::Enum ? max([0, ...$lengths]) : array_sum($lengths) + max(0, count($members) - 1);
255
256        return new Domain(Kind::String, $type->kind === EnumerationKind::Enum ? Field::Enum : Field::Set, $length, Domain::NOT_FIXED, false, $collation, $members);
257    }

259    /**
260     * Resolves BOOL, SERIAL, JSON, BIT and VECTOR.
261     */
262    public function elementary(Elementary $type): Domain
263    {
264        return match ($type->kind) {
265            ElementaryKind::Boolean => Domain::integer(Field::Tiny, 1),
266            ElementaryKind::Serial => Domain::integer(Field::LongLong, 20, true),
267            ElementaryKind::Json => new Domain(Kind::Json, Field::Json, 4294967295, Domain::NOT_FIXED, false, Collation::binary()),
268            ElementaryKind::Bit => new Domain(Kind::Bit, Field::Bit, $type->length === null ? 1 : (int) $type->length, 0, true),
269            ElementaryKind::Vector => new Domain(Kind::String, Field::Vector, ($type->length === null ? 2048 : (int) $type->length) * 4, Domain::NOT_FIXED, false, Collation::binary()),
270        };
271    }
