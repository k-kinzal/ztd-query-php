30/**
31 * Resolves the types of the columns of a temporary table: the rows of a set operation and of a derived table the server materializes.
32 *
33 * A derived table merges into the query that reads it when it is a single query block that
34 * reads tables without grouping, aggregating, removing duplicates, limiting or windowing; its
35 * columns keep the types of their expressions. Otherwise it is materialized: a column keeps the
36 * type of a column it copies; a BIGINT narrower than 10 characters becomes an INT, a string loses
37 * its decimals, and NULL becomes an empty binary string. An integer column reports the length of
38 * its expression as its display width, while an expression over it sees the length of its whole
39 * type. Computed strings longer than 512 characters become BLOB columns with a separate
40 * storage capacity, while copied columns retain their type. The rows of a set operation are materialized too, and a TEXT or BLOB column counts its
41 * length in bytes.
42 * Source: https://dev.mysql.com/doc/refman/8.4/en/derived-table-optimization.html,
43 * https://dev.mysql.com/doc/refman/8.4/en/numeric-type-attributes.html.
44 *
45 * @visibility SqlSemantics\Platform\MySql
46 */
47final class Materialization

49    /**
50     * Tells whether a derived query can retain its expression domains in the enclosing block.
51     * MySQL 5.6 always materializes. Later releases also retain a separate block for
52     * assignments or dependent select-list queries. Unresolved subqueries are treated
53     * conservatively when facts are unavailable; normal derivation supplies their bindings.
54     */
55    public function mergeable(Query $query, GrammarRelease $release = GrammarRelease::MySql847, ?\SqlSemantics\Statement\Fact\Facts $facts = null, bool $enabled = true): bool
56    {
57        if (!$enabled || $release === GrammarRelease::MySql5651) {
58            return false;
59        }
60        while ($query instanceof ParenthesizedQuery || ($query instanceof QueryExpression && $query->with === null && $query->orderBy === [] && $query->limit === null)) {
61            $query = $query instanceof ParenthesizedQuery ? $query->query : $query->body;
62        }
63        if (!$query instanceof Select || $query->from === null || $query->from instanceof Dual) {
64            return false;
65        }
66        if ($query->groupBy !== null || $query->having !== null || $query->limit !== null || $query->windows !== [] || in_array(SelectOption::Distinct, $query->options, true)) {
67            return false;
68        }
69        $parts = array_map(static fn (object $item): object => $item instanceof SelectExpression ? $item->expression : $item, $query->items);
70
71        return !(new \SqlSemantics\Platform\MySql\Rules\Query\Grouping\RollupItems())->windowed($query) && !(new Aggregates())->aggregates([...$parts, ...array_map(static fn ($item): object => $item->expression, $query->orderBy)]) && !(new \SqlSemantics\Platform\MySql\Rules\Expression\MergeDependencies())->separate($parts, $release, $facts);
72    }

75     * Tells whether the columns of a materialized query take the narrowest integer type their values fit: those of a single query block or row, not of several VALUES rows or a set operation, whose columns settle on their type first.
76     */
77    public function narrows(Query $query): bool
78    {
79        while ($query instanceof ParenthesizedQuery || $query instanceof QueryExpression) {
80            $query = $query instanceof ParenthesizedQuery ? $query->query : $query->body;
81        }
82
83        return $query instanceof Select || ($query instanceof ValuesQuery && count($query->rows) === 1);
84    }

86    /**
87     * Resolves a column of a materialized derived table, narrowing a short BIGINT to an INT when its query narrows.
88     */
89    public function column(Domain $domain, bool $narrows = true, GrammarRelease $release = GrammarRelease::MySql847): Domain
90    {
91        if ($domain->kind === Kind::Json) {
92            return $this->jsonColumn($domain, release: $release);
93        }
94        if ($domain->kind === Kind::Decimal) {
95            return new Domain($domain->kind, $domain->field, $domain->length, $domain->decimals, $domain->unsigned, $domain->collation, $domain->members, $domain->coercibility, capacity: $domain->capacity, displayDecimals: $domain->displayDecimals, storageField: $domain->storageField);
96        }
97        if ($domain->storageField !== null) {
98            $domain = Domain::integer(Field::LongLong, $domain->length, $domain->unsigned);
99        }
100        if ($domain->field === Field::Bit) {
101            return new Domain(Kind::Bit, Field::Bit, $domain->length, 0, true);
102        }
103        if ($domain->field === Field::VarString && ($domain->kind->temporal() || ($domain->kind === Kind::Year && !$domain->collation->bytes()))) {
104            return $this->text($domain, $domain->length);
105        }
106        $domain = $this->integerColumn($domain, $narrows);
107
108        return $domain->kind === Kind::String ? $this->text($domain, $domain->length) : $this->nothing($domain);
109    }

136    /**
137     * Resolves a materialized projection with the temporary-column rule of its expression.
138     *
139     * GROUP_CONCAT uses the session byte limit divided by the character set's minimum
140     * width. Up to 512 characters the column is VARCHAR; beyond that it is BLOB and
141     * holds the maximum byte bound. This differs from the function's directly reported
142     * domain, including in MySQL 5.6 and 5.7. Verified through SQL with latin1, utf8mb4,
143     * UCS-2 and UTF-16 on MySQL 5.6, 8.0 and 8.4.
144     */
145    public function projected(Domain $domain, Query $query, int $position, Derivation $derivation, bool $narrows = true): Domain
146    {
147        while ($query instanceof ParenthesizedQuery || $query instanceof QueryExpression) {
148            $query = $query instanceof ParenthesizedQuery ? $query->query : $query->body;
149        }
150        if (!$query instanceof Select) {
151            return $this->combined($domain, $narrows, $derivation->context->profile->grammar);
152        }
153        $facts = $derivation->facts();
154        $field = $facts->covers($query) ? ($facts->query($query)->projection[$position] ?? null) : null;
155        $expression = $field instanceof OutputField ? $field->expression : null;
156        while ($expression instanceof Grouped) {
157            $expression = $expression->operand;
158        }
159        $release = $derivation->context->profile->grammar;
160        if ($domain->kind === Kind::Json) {
161            return $this->jsonColumn($domain, $expression, $release);
162        }
163        if ($expression !== null && (new \SqlSemantics\Platform\MySql\Rules\Query\Grouping\RollupItems())->containsWindow([$expression])) {
164            return $this->windowProjection($domain, $query, $expression, $derivation, $narrows);
165        }
166        $stored = (new \SqlSemantics\Platform\MySql\Rules\Expression\SubqueryRows())->stored($expression, $derivation->context->profile->grammar);
167        if ($stored !== null) {
168            return $stored;
169        }
170        $resolution = $expression !== null && $facts->covers($expression) ? $facts->scalar($expression)->resolution : ($field instanceof OutputField ? $field->resolution : null);
171        $text = $domain->kind === Kind::String && in_array($domain->field, [Field::VarChar, Field::VarString, Field::String, Field::TinyBlob, Field::Blob, Field::MediumBlob, Field::LongBlob], true);
172        if (!$text || $resolution instanceof ResolvedColumn || (new \SqlSemantics\Platform\MySql\Rules\Query\Result\OutputFields())->copied($expression, $facts)) {
173            return $this->column($domain, $narrows, $release);
174        }
175
176        return $this->computed($domain, $expression, Settings::of($derivation->context), $derivation->context->profile->grammar);
177    }

179    /**
180     * Stores a derived window output once, retaining fields already stored by aggregation.
181     * Bare BLOB outputs keep their complete byte bound; parent expressions allocate whole
182     * characters. Verified through derived tables and UNION on MySQL 8.0 and 9.1.
183     */
184    public function windowProjection(Domain $domain, Select $query, Scalar $expression, Derivation $derivation, bool $narrows = true): Domain
185    {
186        $facts = $derivation->facts();
187        $release = $derivation->context->profile->grammar;
188        $bare = $expression instanceof \SqlSemantics\Platform\MySql\Statement\Call\SetFunction || $expression instanceof \SqlSemantics\Platform\MySql\Statement\Call\Window\WindowFunction;
189        $logical = $facts->covers($expression) ? (new Precision())->domain($facts->scalar($expression)->type) : null;
190        $reported = $logical === null ? null : (new \SqlSemantics\Platform\MySql\Rules\Query\Result\OutputFields())->domain($query, $expression, $logical, $release, $facts);
191        $domain = $reported !== $logical ? $domain : ($bare ? $this->windowed($domain, $release, $expression) : $this->windowParent($domain, $release, $expression));
192        if ($bare && $domain->field->blob()) {
193            return $this->stored($domain, $domain->metadataLength(), $domain->metadataLength());
194        }
195
196        return $domain->kind === Kind::String ? $this->computed($domain, $expression, Settings::of($derivation->context), $release) : $this->column($domain, $narrows, $release);
197    }

199    /**
200     * Resolves the temporary column of a computed string, retaining its storage capacity separately from its reported width.
201     */
202    public function computed(Domain $domain, ?Scalar $expression, Settings $settings, GrammarRelease $release): Domain
203    {
204        $width = $domain->collation->charset->maxLength;
205        $characters = $expression instanceof GroupConcat ? intdiv($settings->groupConcatMaxLen, $domain->collation->charset->minLength()) : ($domain->field->blob() ? intdiv($domain->metadataLength(), $width) : $domain->length);
206        if ($expression instanceof GroupConcat && $characters <= 512) {
207            return new Domain(Kind::String, Field::VarString, $characters, 0, false, $domain->collation, [], $domain->coercibility);
208        }
209        if ($characters <= 512 && !$domain->field->blob()) {
210            return $this->column($domain);
211        }
212        $bytes = min(4294967295, $characters * $width);
213        $legacy = $release === GrammarRelease::MySql5651;
214
215        return $this->stored($domain, $bytes, $legacy ? $characters : $bytes);
216    }

219     * Resolves a temporary BLOB column with its reported byte bound and storage capacity.
220     * Expressions over the column see the whole storage type; direct projection
221     * retains the original bound. The allocation bound is in characters in 5.6
222     * and bytes in later releases, as observed with materialized casts.
223     */
224    public function stored(Domain $domain, int $bytes, int $allocation): Domain
225    {
226        $capacity = $allocation <= 255 ? 255 : ($allocation <= 65535 ? 65535 : ($allocation <= 16777215 ? 16777215 : 4294967295));
227
228        return new Domain(Kind::String, Field::Blob, $bytes, 0, false, $domain->collation, [], $domain->coercibility, null, $capacity);

243    /**
244     * Resolves a column of the rows of a set operation.
245     *
246     * A column that is NULL in every operand is an empty binary string: before MySQL 8.1, which
247     * simplified how the type of several values is aggregated (Bug #34847836), a CHAR, from 8.1 a
248     * VARCHAR (verified on live 5.6, 5.7, 8.0, 8.4 and 9.1 servers).
249     * Source: https://dev.mysql.com/doc/relnotes/mysql/8.1/en/news-8-1-0.html.
250     */
251    public function set(Domain $domain, GrammarRelease $release = GrammarRelease::MySql847): Domain
252    {
253        if ($domain->kind === Kind::Null && in_array($release, [GrammarRelease::MySql5651, GrammarRelease::MySql5744, GrammarRelease::MySql8044], true)) {
254            return new Domain(Kind::String, Field::String, 0, 0, false, Collation::binary(), [], Coercibility::Ignorable);
255        }
256        if ($domain->kind !== Kind::String) {
257            return $this->nothing($domain);
258        }
259
260        $length = $domain->field === Field::Blob ? min(4294967295, $domain->length * $domain->collation->charset->maxLength) : $domain->length;
261        $legacy = $release === GrammarRelease::MySql5651;
262
263        return $domain->field === Field::Blob && !$legacy ? $this->stored($domain, $length, $length) : $this->text($domain, $length);
264    }

266    /**
267     * Resolves the value a window function reads from the temporary table of its window: MIN, MAX, FIRST_VALUE, LAST_VALUE, NTH_VALUE, LEAD and LAG answer it as the table holds it.
268     *
269     * An integer, a YEAR and a BIT become an INT when shorter than 10 characters and a BIGINT
270     * otherwise, as long as the value; a string a VARCHAR without decimals, a TEXT or BLOB of any
271     * size, or a string longer than 512 characters, a BLOB as long as its bytes; JSON the longest JSON
272     * without decimals; a temporal value takes the length of its type and
273     * fractional digits; NULL an empty binary string, a CHAR before MySQL 8.1 as in a set
274     * operation (verified on live 8.0 and 8.4 servers).
275     * Source: https://dev.mysql.com/doc/refman/8.4/en/window-function-optimization.html.
276     */
277    public function windowed(Domain $domain, GrammarRelease $release = GrammarRelease::MySql847, ?Scalar $expression = null): Domain
278    {
279        $fraction = $domain->decimals > 0 && $domain->decimals <= 6 ? $domain->decimals + 1 : 0;
280
281        return match ($domain->kind) {
282            Kind::Integer, Kind::Year, Kind::Bit => new Domain(Kind::Integer, $domain->length < 10 ? Field::Long : Field::LongLong, $domain->length, 0, $domain->unsigned, null, [], Coercibility::Numeric),
283            Kind::String => ($domain->field->blob() || $domain->length > 512
284                ? new Domain(Kind::String, Field::Blob, min(4294967295, $domain->length * $domain->collation->charset->maxLength), 0, false, $domain->collation, [], $domain->coercibility)
285                : new Domain(Kind::String, Field::VarString, $domain->length, 0, false, $domain->collation, [], $domain->coercibility)),
286            Kind::Date => new Domain(Kind::Date, $domain->field, 10, 0, false, null, [], $domain->coercibility),
287            Kind::DateTime => new Domain(Kind::DateTime, $domain->field, 19 + $fraction, $domain->decimals, false, null, [], $domain->coercibility),
288            Kind::Time => new Domain(Kind::Time, $domain->field, 10 + $fraction, $domain->decimals, false, null, [], $domain->coercibility),
289            Kind::Json => $this->jsonColumn($domain, $expression, $release),
290            Kind::Null => $this->set($domain, $release),
291            Kind::Decimal, Kind::Double => $domain,
292        };
293    }

312     * Resolves JSON temporary-column metadata. MySQL 8.0 rounds the reported byte
313     * bound of control expressions and LEAD/LAG to whole UTF-8 characters.
314     * MySQL 5.7 uses at least 16 MiB, preserving wider source-column and control
315     * bounds. Stored columns use binary
316     * collation independently of that reported bound.
317     * Verified through direct and derived results on 5.7, and windowed results
318     * on 8.0, 8.4 and 9.1.
319     */
320    public function jsonColumn(Domain $domain, ?Scalar $expression = null, GrammarRelease $release = GrammarRelease::MySql847): Domain
321    {
322        while ($expression instanceof Grouped || ($expression instanceof \SqlSemantics\Platform\MySql\Statement\Expression\Operator\Unary && $expression->operator === \SqlSemantics\Platform\MySql\Statement\Expression\Operator\UnaryOperator::Plus) || ($expression instanceof \SqlSemantics\Platform\MySql\Statement\Expression\Access\OdbcEscape && $expression->literal() === null)) {
323            $expression = $expression->operand;
324        }
325        $text = match (true) {
326            $expression instanceof \SqlSemantics\Platform\MySql\Statement\Expression\Branching\CaseExpression => true,
327            $expression instanceof \SqlSemantics\Platform\MySql\Statement\Call\KeywordCall => in_array($expression->function, [\SqlSemantics\Platform\MySql\Statement\Call\KeywordFunction::If, \SqlSemantics\Platform\MySql\Statement\Call\KeywordFunction::Coalesce], true),
328            $expression instanceof \SqlSemantics\Platform\MySql\Statement\Call\FunctionCall => $expression->schema === null && strtoupper($expression->name->value) === 'IFNULL',
