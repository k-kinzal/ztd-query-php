48    /**
49     * Answers the integer an argument writes as a literal, or null when it writes none.
50     *
51     * A string literal counts by the digits it starts with, as the server reads it as a number. An
52     * integer beyond the range of an int counts as the largest int of its sign.
53     */
54    public function constant(int $index): ?int
55    {
56        $node = $this->nodes[$index] ?? null;
57        $negative = false;
58        while ($node instanceof Grouped || ($node instanceof Unary && $node->operator === UnaryOperator::Minus)) {
59            $negative = $negative !== ($node instanceof Unary);
60            $node = $node->operand;
61        }
62        $text = match (true) {
63            $node instanceof NumberLiteral && $node->form === NumberForm::Integer => $node->text,
64            $node instanceof StringLiteral && preg_match('/\A\s*[0-9]+/', $node->value(), $digits) === 1 => trim($digits[0]),
65            default => null,
66        };
67
68        if ($text === null) {
69            return null;
70        }
71        $magnitude = filter_var(ltrim($text, '0') === '' ? '0' : ltrim($text, '0'), FILTER_VALIDATE_INT);
72        $magnitude = is_int($magnitude) ? $magnitude : PHP_INT_MAX;
73
74        return $negative ? -$magnitude : $magnitude;
75    }

77    /**
78     * Answers the length in characters of a value written as text.
79     *
80     * A double without fixed decimals takes 22 characters, a FLOAT its display length (verified on a live 8.4 server);
81     * MySQL 5.6 and 5.7 take the display length of every double: the text of a literal, 23 for a
82     * computed value (verified on live 5.6.51 and 5.7.44 servers).
83     */
84    public function length(Domain $domain): int
85    {
86        $legacy = in_array($this->derivation->context->profile->grammar, [GrammarRelease::MySql5651, GrammarRelease::MySql5744], true);
87
88        return match ($domain->kind) {
89            Kind::Double => $legacy || $domain->decimals < Domain::NOT_FIXED || $domain->field === Field::Float ? $domain->length : 22,
90            Kind::Null => 0,
91            Kind::Integer, Kind::Decimal, Kind::String, Kind::Date, Kind::Time, Kind::DateTime, Kind::Year, Kind::Json, Kind::Bit => $domain->length,
92        };
93    }

95    /**
96     * Resolves a string result of a length in the collation some arguments aggregate to, or null after a conflict.
97     *
98     * A result longer than 65535 bytes is a MEDIUMBLOB or LONGBLOB as long as its bytes
99     * (FormatResults::sized(), verified on live 5.7, 8.0 and 8.4 servers).
100     *
101     * @param list<Domain> $domains The arguments whose collations take part
102     */
103    public function text(array $domains, int $length, string $operation): ?Domain
104    {
105        $settled = $this->collations()->aggregate($domains, $operation, $this->derivation);
106        if ($settled === null) {
107            return null;
108        }
109        [$collation, $coercibility] = $settled;
110
111        return (new FormatResults())->sized($length, $collation, $coercibility);
112    }
