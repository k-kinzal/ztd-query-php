15/**
16 * Resolves the results of the string functions that format, encode, pick or convert: FORMAT, ELT, MAKE_SET, EXPORT_SET, SOUNDEX, QUOTE, ORD, the base conversions and the compression functions.
17 *
18 * A string result is a VARCHAR up to 65535 bytes, a MEDIUMBLOB up to 16777215 and a LONGBLOB
19 * beyond, whose length then counts bytes. The lengths the server reports, verified on a live 8.4
20 * server, are:
21 * - FORMAT: the argument, a separator for each three of its characters, and 32 more for the
22 *   point and the 30 decimals it can take, in the connection collation;
23 * - ELT: its longest string, MAKE_SET: all its strings and a comma between each two, in the
24 *   collation they aggregate to;
25 * - EXPORT_SET: 64 times its longer string and 63 separators, a string converted to binary
26 *   counting its bytes;
27 * - SOUNDEX: its argument, at least 4, in the collation of the argument (binary for NULL);
28 * - QUOTE: twice its argument and the quotes, at least 4; a binary string or NULL takes the
29 *   connection collation, a number or a temporal value latin1_swedish_ci;
30 * - TO_BASE64 and FROM_BASE64: the encoding of the bytes of the argument with a newline after
31 *   each 76 characters, and three bytes for each four;
32 * - BIN, OCT and CONV: 65 characters; HEX twice the bytes of a string and 16 for a number;
33 *   UNHEX half the bytes of its argument;
34 * - COMPRESS: what zlib bounds the compression of the bytes to, with the 4-byte length and a
35 *   byte for a trailing period; UNCOMPRESS and LOAD_FILE 16777216 bytes;
36 * - ORD a BIGINT of 21 digits, UNCOMPRESSED_LENGTH one of 10.
37 * MySQL 5.6 and 5.7 (verified on live 5.6.51 and 5.7.44 servers) add 33 to FORMAT, write BIN,
38 * OCT and CONV in 64 characters, HEX of a number in twice its length, and COMPRESS in a fifth
39 * more than its bytes and 12; their HEX and UNHEX have no decimals.
40 * Source: https://dev.mysql.com/doc/refman/8.4/en/string-functions.html,
41 * https://dev.mysql.com/doc/refman/8.4/en/encryption-functions.html,
42 * https://dev.mysql.com/doc/refman/8.4/en/mathematical-functions.html#function_conv.
43 *
44 * @visibility SqlSemantics\Platform\MySql\Rules\Typing
45 */
46final class FormatResults

48    /**
49     * Answers the rule of each function, by name.
50     *
51     * @return array<string, Closure(Invocation): ?Domain>
52     */
53    public function rules(): array
54    {
55        $radix = fn (Invocation $call): Domain => $this->sized($this->legacy($call) ? 64 : 65, $call->settings->connection, Coercibility::Coercible);
56        $blob = static fn (Invocation $call): Domain => Domain::string(16777216, Collation::binary(), Field::LongBlob, Coercibility::Coercible);
57
58        return [
59            'FORMAT' => fn (Invocation $call): Domain => $this->sized($call->length($call->domain(0)) + intdiv($call->length($call->domain(0)), 3) + ($this->legacy($call) ? 33 : 32), $call->settings->connection, Coercibility::Coercible),
60            'ELT' => fn (Invocation $call): ?Domain => $this->settled($call, array_slice($call->domains, 1), 'elt', static fn (Collation $collation): int => max(0, ...array_map($call->length(...), array_slice($call->domains, 1)))),
61            'MAKE_SET' => fn (Invocation $call): ?Domain => $this->settled($call, array_slice($call->domains, 1), 'make_set', static fn (Collation $collation): int => array_sum(array_map($call->length(...), array_slice($call->domains, 1))) + count($call->domains) - 2),
62            'EXPORT_SET' => fn (Invocation $call): ?Domain => $this->settled($call, array_slice($call->domains, 1, 3), 'export_set', fn (Collation $collation): int => 64 * max($this->width($call, $call->domain(1), $collation), $this->width($call, $call->domain(2), $collation)) + 63 * (count($call->domains) > 3 ? $this->width($call, $call->domain(3), $collation) : 1)),
63            'SOUNDEX' => $this->soundex(...),
64            'QUOTE' => $this->quote(...),
65            'ORD' => static fn (Invocation $call): Domain => Domain::integer(Field::LongLong, 21),
66            'TO_BASE64' => fn (Invocation $call): Domain => $this->sized($this->encoded($this->bytes($call->domain(0))), $call->settings->connection, Coercibility::Coercible),
67            'FROM_BASE64' => fn (Invocation $call): Domain => $this->sized(intdiv($this->bytes($call->domain(0)) * 3, 4), Collation::binary(), Coercibility::Coercible),
68            'BIN' => $radix,
69            'OCT' => $radix,
70            'CONV' => $radix,
71            'HEX' => fn (Invocation $call): Domain => $this->undecimal($call, $this->sized($this->legacy($call) && $call->domain(0)->kind->numeric() ? 2 * $call->length($call->domain(0)) : $this->digitsOf($call->domain(0)), $call->settings->connection, Coercibility::Coercible)),
72            'UNHEX' => fn (Invocation $call): Domain => $this->undecimal($call, $this->sized(intdiv($this->bytes($call->domain(0)) + 1, 2), Collation::binary(), Coercibility::Coercible)),
73            'COMPRESS' => fn (Invocation $call): Domain => $this->sized($this->legacy($call) ? intdiv($this->bytes($call->domain(0)) * 120, 100) + 12 : $this->bound($this->bytes($call->domain(0))), Collation::binary(), Coercibility::Coercible),
74            'UNCOMPRESS' => $blob,
75            'LOAD_FILE' => $blob,
76            'UNCOMPRESSED_LENGTH' => static fn (Invocation $call): Domain => Domain::integer(Field::LongLong, 10),
77        ];
78    }

91     * Answers the result of HEX or UNHEX, which MySQL 5.6 and 5.7 report with no decimals.
92     */
93    public function undecimal(Invocation $call, Domain $domain): Domain
94    {
95        return $this->legacy($call) ? new Domain(Kind::String, $domain->field, $domain->length, 0, false, $domain->collation, [], $domain->coercibility) : $domain;

98    /**
99     * Answers a string of a number of characters: a VARCHAR up to 65535 bytes, else a MEDIUMBLOB or LONGBLOB as long as its bytes.
100     */
101    public function sized(int $characters, Collation $collation, Coercibility $coercibility): Domain
102    {
103        $bytes = $characters * $collation->charset->maxLength;
104        if ($bytes <= 65535) {
105            return Domain::string($characters, $collation, Field::VarString, $coercibility);
106        }
107
108        return Domain::string(min($bytes, 4294967295), $collation, $bytes > 16777215 ? Field::LongBlob : Field::MediumBlob, $coercibility);
109    }

111    /**
112     * Resolves a string in the collation some arguments aggregate to, as long as a length in that collation, or null after a conflict.
113     *
114     * A string made of numbers is coercible.
115     *
116     * @param list<Domain> $domains The arguments whose collations take part
117     * @param Closure(Collation): int $length The length of the result in a collation
118     */
119    public function settled(Invocation $call, array $domains, string $operation, Closure $length): ?Domain
120    {
121        $settled = $call->collations()->aggregate($domains, $operation, $call->derivation);
122        if ($settled === null) {
123            return null;
124        }
125
126        return $this->sized($length($settled[0]), $settled[0], $settled[1] === Coercibility::Numeric ? Coercibility::Coercible : $settled[1]);
127    }

129    /**
130     * Answers the length of an argument in the collation of a result: its bytes when a string of another character set becomes binary.
131     */
132    public function width(Invocation $call, Domain $domain, Collation $collation): int
133    {
134        return $collation->charset->name === 'binary' && $domain->kind === Kind::String && $domain->collation->charset->name !== 'binary' ? $domain->byteLength() : $call->length($domain);
135    }

137    /**
138     * Answers the length of HEX: two digits for each byte of a string, each character of a temporal or JSON value, and 16 for a number.
139     */
140    public function digitsOf(Domain $domain): int
141    {
142        return match ($domain->kind) {
143            Kind::String => $domain->byteLength() * 2,
144            Kind::Date, Kind::Time, Kind::DateTime, Kind::Json => $domain->length * 2,
145            Kind::Null => 0,
146            Kind::Integer, Kind::Decimal, Kind::Double, Kind::Year, Kind::Bit => 16,
147        };
148    }

150    /**
151     * Answers the bytes the text of a value takes: the bytes of a string, the display length of anything else.
152     */
153    public function bytes(Domain $domain): int
154    {
155        return $domain->kind === Kind::Null ? 0 : $domain->byteLength();
156    }

158    /**
159     * Answers the length of the base64 encoding of a number of bytes, with a newline after each 76 characters.
160     */
161    public function encoded(int $bytes): int
162    {
163        $characters = 4 * intdiv($bytes + 2, 3);
164
165        return $characters === 0 ? 0 : $characters + intdiv($characters - 1, 76);
166    }

168    /**
169     * Answers the most bytes COMPRESS answers for a number of bytes: the zlib bound, the 4-byte length and a period.
170     */
171    public function bound(int $bytes): int
172    {
173        return $bytes + ($bytes >> 12) + ($bytes >> 14) + ($bytes >> 25) + 13 + 5;
174    }

176    /**
177     * Resolves INSERT: a string in the collation of its first argument, whatever the collation of the string inserted, as long as both in that collation.
178     *
179     * A first argument that is no string takes the connection collation (verified on a live 8.4 server).
180     */
181    public function inserted(Invocation $call): Domain
182    {
183        $domain = $call->domain(0);
184        [$collation, $coercibility] = $domain->kind === Kind::String ? [$domain->collation, $domain->coercibility] : [$call->settings->connection, Coercibility::Coercible];
185
186        return $this->sized($this->width($call, $domain, $collation) + $this->width($call, $call->domain(3), $collation), $collation, $coercibility);
187    }

189    /**
190     * Resolves SOUNDEX: a string as long as its argument, at least 4, in the collation of the argument.
191     *
192     * NULL answers a binary string; a value that is no string the connection collation.
193     */
194    public function soundex(Invocation $call): Domain
195    {
196        $domain = $call->domain(0);
197        $length = max(4, $call->length($domain));
198
199        return match (true) {
200            $domain->kind === Kind::Null => $this->sized($length, Collation::binary(), Coercibility::Ignorable),
201            $domain->kind === Kind::String => $this->sized($length, $domain->collation, $domain->coercibility),
202            default => $this->sized($length, $call->settings->connection, Coercibility::Coercible),
203        };
204    }
205
206    /**
207     * Resolves QUOTE: twice the argument and two quotes, at least the 4 characters of NULL.
208     *
209     * A string keeps its collation, a binary string takes the connection collation, NULL the
210     * connection collation as ignorable, and a number or temporal value latin1_swedish_ci as numeric.
211     */
212    public function quote(Invocation $call): Domain
213    {
214        $domain = $call->domain(0);
215        $length = max(4, 2 * $call->length($domain) + 2);
216
217        return match (true) {
218            $domain->kind === Kind::Null => $this->sized($length, $call->settings->connection, Coercibility::Ignorable),
219            $domain->kind === Kind::String && $domain->collation->charset->name === 'binary' => $this->sized($length, $call->settings->connection, Coercibility::Coercible),
220            $domain->kind === Kind::String => $this->sized($length, $domain->collation, $domain->coercibility),
221            default => $this->sized($length, Collation::known('latin1_swedish_ci'), Coercibility::Numeric),
222        };
223    }
