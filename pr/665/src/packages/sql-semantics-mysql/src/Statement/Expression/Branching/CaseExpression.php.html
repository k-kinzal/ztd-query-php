23/**
24 * A CASE expression in its simple form, `CASE operand WHEN value THEN result … [ELSE result] END`, or its searched form, `CASE WHEN condition THEN result … [ELSE result] END` (`Item_func_case`).
25 *
26 * The branches are tried in order and the first that matches gives the
27 * result; the order is kept as written.
28 *
29 * Rule: MYSQL-CASE-001. Facts: the type is the aggregate of the results
30 * (MYSQL-TYPE-AGGREGATION-001); the value is NULL when a result can be
31 * NULL or when there is no ELSE. In the simple form the operand and every
32 * value must have the same number of columns (MYSQL-OPERAND-COLUMNS-001);
33 * conditions and results take single values. Terminates: the parts are
34 * strict parts.
35 * Source: https://dev.mysql.com/doc/refman/8.4/en/flow-control-functions.html#operator_case.
36 * Status: Implemented.
37 *
38 * @visibility public
39 * @example Typing a CASE by its results
40 *     $query = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\MySql\Dialect::MySql))->analyze("SELECT a FROM t WHERE CASE a WHEN 1 THEN 'x' ELSE 'y' END");
41 *     [$query->facts->scalar($query->statement->where)->type->descriptor->name(), $query->facts->scalar($query->statement->where)->nullability] // => ['VARCHAR', \SqlSemantics\Statement\Type\Nullability::NotNull]
42 */
43final class CaseExpression implements Scalar
44{
45    use Snapshot;
46
47    /**
48     * @var non-empty-list<CaseBranch> The branches in order
49     */
50    public readonly array $branches;
51
52    /**
53     * @param Scalar|null $operand The compared value of the simple form; null for the searched form
54     * @param list<CaseBranch> $branches The branches in order; at least one
55     * @param Scalar|null $else The result after ELSE, when written
56     */
57    public function __construct(public readonly ?Scalar $operand, array $branches, public readonly ?Scalar $else = null)
58    {
59        $this->branches = Check::listOf($branches, CaseBranch::class, 'A CASE expression has at least one WHEN branch.', 1);
60    }
61
62    /**
63     * Derives the operand, every branch and the ELSE result, and aggregates the result types.
64     */
65    public function deriveScalar(Derivation $derivation, Environment $environment): ScalarFact
66    {
67        $operands = new Operands();
68        $compared = $this->operand === null ? [] : [$derivation->scalar($this->operand, $environment)];
69        $types = [];
70        $nullability = $this->else === null ? Nullability::Nullable : Nullability::NotNull;
71        foreach ($this->branches as $branch) {
72            $condition = $derivation->scalar($branch->condition, $environment);
73            if ($this->operand === null) {
74                $operands->single($condition, $derivation);
75            } else {
76                $compared[] = $condition;
77            }
78            $result = $operands->single($derivation->scalar($branch->result, $environment), $derivation);
79            $types[] = $result->type;
80            $nullability = $nullability->propagate($result->nullability);
81        }
82        $operands->comparable($compared, $derivation);
83        if ($this->else !== null) {
84            $result = $operands->single($derivation->scalar($this->else, $environment), $derivation);
85            $types[] = $result->type;
86            $nullability = $nullability->propagate($result->nullability);
87        }
88
84        return new ScalarFact((new TypeAggregation())->aggregate($types), $nullability);
89        $domains = (new Precision())->all($types);
90        $settings = Settings::of($derivation->context);
91        $domain = $domains === null ? null : (new Aggregation(new Collations($settings->connection)))->of($domains, 'case', $derivation);
92
93        return new ScalarFact($domain === null ? (new TypeAggregation())->aggregate($types) : new Known($domain), $nullability);
94    }
95
96    /**
97     * Writes CASE, the operand, the branches, the ELSE result and END.
98     */
99    public function render(Output $out): void
100    {
101        $out->keyword('CASE')->node($this->operand);
102        foreach ($this->branches as $branch) {
103            $out->node($branch);
104        }
105        if ($this->else !== null) {
106            $out->keyword('ELSE')->node($this->else);
107        }
108        $out->keyword('END');
109    }
