26/**
27 * A system variable read as a value or named as the target of SET: `@@name`, `@@GLOBAL.name`, `@@instance.name`.
28 *
29 * A structured variable is written with the name of its instance or
30 * component before the variable name; `DEFAULT.name` names the instance
31 * `default`. Without a scope the session value is read when the variable has
32 * one and the global value otherwise.
33 *
34 * Rule: MYSQL-SYSTEM-VARIABLE-001. Facts: the type and the NULL fact are
35 * those of the variable in the running server, which no context holds; the
29 * fact names the variable as missing session state. Diagnostics: none.
36 * fact names the variable as missing session state; a parameter of a named
37 * key cache has the type of the variable of the default key cache, as every
38 * parameter is an integer. Diagnostics: a variable
39 * read with an instance name that is neither a key cache variable nor a
40 * variable the release knows by that full name is unknown, named with its
41 * instance (verified on a live 8.4 server); MySQL 5.6 and 5.7 read the name
42 * after the instance as the variable, unknown by that name or, when they
43 * know it, not structured (verified on live 5.6.51 and 5.7.44 servers).
44 * Source: https://dev.mysql.com/doc/refman/8.4/en/using-system-variables.html,
45 * https://dev.mysql.com/doc/refman/8.4/en/structured-system-variables.html. Status: Implemented.
46 *
47 * @visibility public
48 * @example Reading a scoped system variable
49 *     $query = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\MySql\Dialect::MySql))->analyze('SELECT a FROM t WHERE a = @@GLOBAL.sort_buffer_size');
50 *     [$query->statement->where->right->scope, $query->statement->where->right->name->value] // => [\SqlSemantics\Platform\MySql\Statement\Variable\VariableScope::Global, 'sort_buffer_size']

92    /**
93     * Reports a variable written with an instance name that the release does not have by that name: unknown, named with its instance; in MySQL 5.6 and 5.7, which read the name after the instance as the variable, unknown by that name, or not structured when they know it (verified on live 5.6.51, 5.7.44 and 8.4 servers).
94     */
95    public function structured(Derivation $derivation): void
96    {
97        if ($this->instance === null || in_array(strtolower($this->name->value), self::KEY_CACHE, true)) {
98            return;
99        }
100        $grammar = $derivation->context->profile->grammar;
101        $variables = SystemVariables::of($grammar);
102        if ($grammar === GrammarRelease::MySql5651 || $grammar === GrammarRelease::MySql5744) {
103            $derivation->report($variables->find($this->name->value) === null ? new UnknownSystemVariable($this->name->value, $this->assigned, true) : new UnstructuredVariable($this->name->value));
104
105            return;
106        }
107        $name = $this->instance->value . '.' . $this->name->value;
108        if ($variables->find($name) === null) {
109            $derivation->report(new UnknownSystemVariable($name, $this->assigned, true));
110        }
111    }
