27/**
28 * A call of a function that is only a window function: ranking, distribution, and the value functions LEAD, LAG, FIRST_VALUE, LAST_VALUE and NTH_VALUE.
29 *
30 * Rule: MYSQL-WINDOW-FUNCTION-001. The window is required. The count of
31 * NTILE and the offset of LEAD and LAG are an integer, a parameter marker,
32 * a user variable or a routine variable (stable_integer); the row number of
33 * NTH_VALUE is a primary expression. The result follows
34 * MYSQL-WINDOW-RESULT-001. Terminates: the arguments and the window are
35 * strict parts.
36 * Source: https://dev.mysql.com/doc/refman/8.4/en/window-function-descriptions.html.
37 * Status: Implemented.
38 *
39 * @visibility public
40 * @example Typing a ranking function
41 *     $query = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\MySql\Dialect::MySql))->analyze('SELECT RANK() OVER w');
42 *     [$query->field(0)->type->descriptor->name(), $query->field(0)->nullability] // => ['BIGINT', \SqlSemantics\Statement\Type\Nullability::NotNull]
43 * @example Refusing an offset that is not a stable integer
44 *     new \SqlSemantics\Platform\MySql\Statement\Call\Window\WindowFunction(\SqlSemantics\Platform\MySql\Statement\Call\Window\WindowFunctionKind::Tile, [new \SqlSemantics\Platform\MySql\Statement\Literal\NullLiteral()], new \SqlSemantics\Statement\Identifier\Name('w')) // throws \SqlSemantics\Diagnostic\InvalidConstruction
45 */
46final class WindowFunction implements Scalar
47{

55    /**
56     * @param WindowFunctionKind $kind The function
57     * @param list<Scalar> $arguments The arguments in order
58     * @param Name|WindowSpecification $over The window written after OVER
59     * @param NullTreatment|null $nulls The written RESPECT NULLS or IGNORE NULLS
60     * @param CountingEdge|null $edge The written FROM FIRST or FROM LAST of NTH_VALUE
61     */
62    public function __construct(
63        public readonly WindowFunctionKind $kind,
64        array $arguments,
65        public readonly Name|WindowSpecification $over,
66        public readonly ?NullTreatment $nulls = null,
67        public readonly ?CountingEdge $edge = null,
68    ) {
69        $this->arguments = Check::listOf($arguments, Scalar::class, 'Window function arguments are expressions.');
70        [$minimum, $maximum] = $kind->arity();
71        Check::input(count($arguments) >= $minimum && count($arguments) <= $maximum, 'The grammar does not accept this number of arguments for ' . $kind->value . '.');
72        Check::input($nulls === null || $kind->treatsNulls(), $kind->value . ' takes no NULL treatment.');
73        Check::input($edge === null || $kind === WindowFunctionKind::NthValue, 'Only NTH_VALUE counts from an edge.');
74        $counted = $arguments[$kind->counted() ?? -1] ?? null;
75        Check::input($counted === null || $counted instanceof NumberLiteral || $counted instanceof Parameter || $counted instanceof UserVariable || $counted instanceof RoutineVariable, 'The count is an integer, a parameter marker or a variable.');
76        Check::input($kind !== WindowFunctionKind::NthValue || (new Precedence())->admits($arguments[1], Precedence::SIMPLE_EXPR), 'The row number of NTH_VALUE needs a grouping.');
77    }

79    /**
80     * Derives the arguments, the window and the result.
81     */
82    public function deriveScalar(Derivation $derivation, Environment $environment): ScalarFact
83    {
84        $facts = [];
85        foreach ($this->arguments as $argument) {
86            $facts[] = (new Arguments())->one($argument, $derivation, $environment);
87        }
88        (new Windows())->derive($this->over, $derivation, $environment);
89        $fact = (new WindowResults())->result($this, $facts, $derivation);
90        $arguments = (new Precision())->all(array_map(static fn (ScalarFact $argument): TypeFact => $argument->type, $facts));
91        $domain = $arguments === null ? null : (new WindowResults())->resolved($this, $arguments, $derivation);
92
87        return (new WindowResults())->result($this, $facts, $derivation);
93        return $domain === null ? $fact : new ScalarFact(new Known($domain), $fact->nullability);
94    }

97     * Writes the call, its options and its window.
98     */
99    public function render(Output $out): void
100    {
101        $out->keyword($this->kind->value)->glue()->symbol('(')->list($this->arguments)->symbol(')');
102        if ($this->edge !== null) {
103            $out->keyword(...explode(' ', $this->edge->value));
104        }
105        if ($this->nulls !== null) {
106            $out->keyword(...explode(' ', $this->nulls->value));
107        }
108        (new Windows())->render($this->over, $out);
109    }
