9/**
10 * The name of an optimizer hint, and the form of the arguments it takes.
11 *
12 * MySQL 5.7 has the join buffering, multi-range read, condition pushdown,
13 * range, execution time, query block name and subquery hints; MySQL 8.0
14 * adds the join order, derived table, index, index merge, skip scan, hash
15 * join, resource group and SET_VAR hints. MySQL 5.6 has no hints. The names
16 * are keywords of the hint comment and are matched without regard to case.
17 * Source: https://dev.mysql.com/doc/refman/8.4/en/optimizer-hints.html,
18 * https://dev.mysql.com/doc/refman/5.7/en/optimizer-hints.html.
19 *
20 * @visibility public
21 * @example Reading the form of a hint
22 *     \SqlSemantics\Platform\MySql\Statement\Hint\HintName::NoIndex->form() // => \SqlSemantics\Platform\MySql\Statement\Hint\HintForm::Key

67    public function form(): HintForm

86    /**
87     * Tells whether a release has the hint: every hint from 8.0, the hints of 5.7 there, and none in 5.6.
88     *
89     * @example Telling the hints of 5.7 apart
90     *     [\SqlSemantics\Platform\MySql\Statement\Hint\HintName::Bka->available(\SqlSemantics\Contract\GrammarRelease::MySql5744), \SqlSemantics\Platform\MySql\Statement\Hint\HintName::SetVar->available(\SqlSemantics\Contract\GrammarRelease::MySql5744)] // => [true, false]
91     */
92    public function available(GrammarRelease $release): bool
93    {
