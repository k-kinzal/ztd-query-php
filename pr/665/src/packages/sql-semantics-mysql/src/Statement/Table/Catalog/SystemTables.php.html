22/**
23 * The tables of the system databases of one release, read from the catalog generated from a server of the release.
24 *
25 * The information_schema, mysql and performance_schema databases hold them. A statement that
26 * reads one is bound against its declaration, which this catalog supplies with the type each
27 * column has when it is read. The name of information_schema and the names of its tables are
28 * compared without regard to case, whatever lower_case_table_names says; the names of the
29 * other system databases and of their tables follow the setting.
30 * Source: https://dev.mysql.com/doc/refman/8.4/en/information-schema-introduction.html,
31 * https://dev.mysql.com/doc/refman/8.4/en/identifier-case-sensitivity.html.
32 *
33 * @visibility public
34 * @example Finding a table of INFORMATION_SCHEMA by any case
35 *     \SqlSemantics\Platform\MySql\Statement\Table\Catalog\SystemTables::of(\SqlSemantics\Contract\GrammarRelease::MySql847)->find('INFORMATION_SCHEMA', 'schemata')?->name // => 'SCHEMATA'
36 */

66    /**
67     * Answers the system tables of a release: those of the catalog of its series, else of the latest series of its major version.
68     */
69    public static function of(GrammarRelease $release): self
70    {
71        if (!isset(self::$releases[$release->value])) {
72            $directory = dirname(__DIR__, 4) . '/resources/system-tables/';
73            $files = glob($directory . substr($release->value, 0, 9) . '*.php');
74            $major = glob($directory . substr($release->value, 0, 7) . '*.php');
75            $file = is_array($files) && $files !== [] ? $files[0] : (is_array($major) && $major !== [] ? $major[count($major) - 1] : $directory . GrammarRelease::MySql847->value . '.php');
76            /** @var list<array{string, string, string, string|null, int|null, string|null, string|null, string, string, list<array{string, string, string, string, int, int, int, int, string, int, string, string, string|null, string, string, string, string|null}>}> $entries */
77            $entries = require $file;
78            $profile = new LanguageProfile($release);
79            $tables = [];
80            foreach ($entries as [$schema, $name, $type, $engine, $version, $format, $collation, $options, $comment, $columns]) {
81                $read = [];
82                foreach ($columns as [$column, $originalName, $originalTable, $database, $code, $length, $decimals, $flags, $columnCollation, $coercibility, $columnType, $listedNullable, $default, $key, $extra, $columnComment, $listedCollation]) {
83                    $read[] = new SystemColumn($column, self::domain($code, $length, $decimals, $flags, $columnCollation, $coercibility, $columnType), ($flags & 1) === 0, $flags, $originalName, $originalTable, $database, $columnType, $default, $key, $extra, $columnComment, $listedCollation, $listedNullable === 'YES');
84                }
85                $declaration = new Table(
86                    new QualifiedName(new Name($name), new Name($schema)),
87                    $profile,
88                    array_map(static fn (SystemColumn $column): Column => new Column(new Name($column->name), $column->domain, $column->nullable ? Nullability::Nullable : Nullability::NotNull), $read),
89                    [],
90                    true,
91                    $type === 'BASE TABLE' ? RelationKind::BaseTable : RelationKind::View,
92                    [],
93                    [],
94                );
95                $tables[] = new SystemTable($schema, $name, $type, $engine, $version, $format, $collation, $options, $comment, $read, $declaration);
96            }
97            self::$releases[$release->value] = new self($tables);
98        }
99
100        return self::$releases[$release->value];
101    }

111    /**
112     * Builds the type of a read of a column from the metadata a catalog entry records.
113     *
114     * @param int $code The field code a result reports
115     * @param int $length The length in characters, or in bytes for a binary value
116     * @param int $decimals The decimals a result reports
117     * @param int $flags The column definition flags a result reports, which mark an ENUM or a SET
118     * @param string $collation The collation of the value
119     * @param int $coercibility The coercibility of the collation
120     * @param string $columnType The column type, which lists the members of an ENUM or a SET
121     */
122    public static function domain(int $code, int $length, int $decimals, int $flags, string $collation, int $coercibility, string $columnType): Domain
123    {
124        $field = Field::tryFrom($code) ?? Field::VarString;
125        $unsigned = ($flags & 32) !== 0;
126        $text = Collation::named($collation) ?? Collation::binary();
127        $holds = Coercibility::tryFrom($coercibility) ?? Coercibility::Implicit;
128        if (($flags & 256) !== 0 || ($flags & 2048) !== 0) {
129            return new Domain(Kind::String, ($flags & 256) !== 0 ? Field::Enum : Field::Set, $length, $decimals, false, $text, self::members($columnType), $holds);
130        }
131
132        return match ($field) {
133            Field::Tiny, Field::Short, Field::Int24, Field::Long, Field::LongLong => new Domain(Kind::Integer, $field, $length, 0, $unsigned, null, [], Coercibility::Numeric),
134            Field::Decimal, Field::NewDecimal => new Domain(Kind::Decimal, Field::NewDecimal, $length, $decimals, $unsigned, null, [], Coercibility::Numeric),
135            Field::Float, Field::Double => new Domain(Kind::Double, $field, $length, $decimals, $unsigned, null, [], Coercibility::Numeric),
136            Field::Null => Domain::null(),
137            Field::Date, Field::NewDate => new Domain(Kind::Date, Field::Date, $length, $decimals, false, $text, [], $holds),
138            Field::Time => new Domain(Kind::Time, $field, $length, $decimals, false, $text, [], $holds),
139            Field::DateTime, Field::Timestamp => new Domain(Kind::DateTime, $field, $length, $decimals, false, $text, [], $holds),
140            Field::Year => new Domain(Kind::Year, $field, $length, 0, true, null, [], Coercibility::Numeric),
141            Field::Json => new Domain(Kind::Json, $field, $length, $decimals, false, $text->bytes() ? null : $text, [], $holds),
142            Field::Bit => new Domain(Kind::Bit, $field, $length, 0, true, null, [], Coercibility::Numeric),
143            Field::Enum, Field::Set, Field::VarChar, Field::Vector, Field::TinyBlob, Field::MediumBlob, Field::LongBlob, Field::Blob, Field::VarString, Field::String, Field::Geometry => new Domain(Kind::String, $field, $length, $decimals, false, $text, [], $holds),
144        };
145    }

152    public static function members(string $columnType): array
153    {
154        preg_match_all("/'((?:[^']|'')*)'/", $columnType, $matches);
