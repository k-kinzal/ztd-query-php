13/**
14 * The options of CHANGE REPLICATION SOURCE TO and CHANGE MASTER TO: the fields of the server's LEX_SOURCE_INFO.
15 *
16 * Each case holds the keyword of the current vocabulary. An option whose
17 * name begins with SOURCE_ (or GET_SOURCE_) is written MASTER_ (or
18 * GET_MASTER_) in the legacy vocabulary; the other options have one name.
19 * The log file and position options also form the UNTIL condition of START
20 * REPLICA.
21 * Source: https://dev.mysql.com/doc/refman/8.4/en/change-replication-source-to.html,
22 * https://dev.mysql.com/doc/refman/5.7/en/change-master-to.html.
23 *
24 * @visibility public
25 * @example Spelling an option in both vocabularies
26 *     [\SqlSemantics\Platform\MySql\Statement\Replication\Source\SourceOptionKind::Host->keyword(\SqlSemantics\Platform\MySql\Statement\Replication\Terminology::Legacy), \SqlSemantics\Platform\MySql\Statement\Replication\Source\SourceOptionKind::Host->value] // => ['MASTER_HOST', 'SOURCE_HOST']
27 */

68    /**
69     * Answers the keyword of the option in a vocabulary.
70     */
71    public function keyword(Terminology $terminology): string
72    {
73        return $terminology === Terminology::Legacy && $this->renamed() ? str_replace('SOURCE_', 'MASTER_', $this->value) : $this->value;
74    }

76    /**
77     * Tells whether the legacy vocabulary spells the option with MASTER.

79    public function renamed(): bool

81        return str_contains($this->value, 'SOURCE_') && $this !== self::ConnectionAutoFailover;

84    /**
85     * Tells whether a value belongs to the domain of the option.
86     *
87     * A file, host, path, name or algorithm list is a string; a port, count,
88     * switch or level a number; the heartbeat period a number literal that
89     * may have a fraction; IGNORE_SERVER_IDS a parenthesized list;
90     * PRIVILEGE_CHECKS_USER an account or NULL; SOURCE_TLS_CIPHERSUITES a
91     * string or NULL; the other two options their keywords, and for
92     * ASSIGN_GTIDS_TO_ANONYMOUS_TRANSACTIONS also a UUID string.
93     */
94    public function accepts(Text|Numeral|NumberLiteral|ServerIds|AccountName|PrimaryKeyCheck|AnonymousGtids|null $value): bool
95    {
96        return match ($this) {
97            self::Host, self::NetworkNamespace, self::Bind, self::User, self::Password, self::SslCa, self::SslCapath, self::SslCert, self::SslCipher,
98            self::SslKey, self::SslCrl, self::SslCrlpath, self::TlsVersion, self::PublicKeyPath, self::CompressionAlgorithms, self::LogFile,
99            self::RelayLogFile => $value instanceof Text,
100            self::Port, self::ConnectRetry, self::RetryCount, self::Delay, self::Ssl, self::SslVerifyServerCert, self::GetPublicKey,
101            self::ZstdCompressionLevel, self::AutoPosition, self::RequireRowFormat, self::ConnectionAutoFailover, self::GtidOnly, self::LogPosition,
102            self::RelayLogPosition => $value instanceof Numeral,
103            self::HeartbeatPeriod => $value instanceof NumberLiteral,
104            self::IgnoreServerIds => $value instanceof ServerIds,
105            self::PrivilegeChecksUser => $value === null || $value instanceof AccountName,
106            self::TlsCiphersuites => $value === null || $value instanceof Text,
107            self::RequireTablePrimaryKeyCheck => $value instanceof PrimaryKeyCheck,
108            self::AssignGtidsToAnonymousTransactions => $value instanceof AnonymousGtids || $value instanceof Text,
109        };
110    }
