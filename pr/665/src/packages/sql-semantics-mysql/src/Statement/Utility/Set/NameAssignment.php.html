20/**
21 * An assignment to a variable named without `@@`: `SET x = v`, `SET GLOBAL x = v`, `SET a.x = v`.
22 *
23 * Rule: MYSQL-SET-ITEM-001 (names). With a scope keyword the name is a
24 * system variable assigned in that scope. Without one, inside a stored
25 * program the name is a variable the program declares when it declares
26 * one, and `NEW.x` in a trigger is a column of the new row; otherwise it is
27 * a system variable assigned in the scope inherited from an earlier item
28 * (SetVariables::scopeOf()). A qualifier names the component or key cache
29 * instance of a structured system variable; `DEFAULT.x` is the instance
30 * `default`. The value is derived as an expression at a position that sees
31 * no relation, a keyword value has no facts, and a bare name value is its
32 * text for a system variable (MYSQL-SET-WORD-001). Diagnostics: none.
33 * Terminates: the parts are leaves.
34 * Source: https://dev.mysql.com/doc/refman/8.4/en/set-variable.html,
35 * https://dev.mysql.com/doc/refman/8.4/en/set-statement.html,
36 * https://dev.mysql.com/doc/refman/8.4/en/structured-system-variables.html.
37 * Status: Implemented.
38 *
39 * @visibility public
40 * @example Reading a scoped name assignment
41 *     $set = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\MySql\Dialect::MySql))->analyze('SET SESSION sort_buffer_size = DEFAULT');
42 *     [$set->statement->items[0]->scope, $set->statement->items[0]->name->value, $set->statement->items[0]->value] // => [\SqlSemantics\Platform\MySql\Statement\Variable\VariableScope::Session, 'sort_buffer_size', \SqlSemantics\Platform\MySql\Statement\Utility\Set\SetWord::Default]
43 */
44final class NameAssignment implements SetItem
45{
46    use Snapshot;
47
48    /**
49     * @param Name $name The assigned name
50     * @param Scalar|SetWord $value The value: an expression, or a keyword the grammar accepts there
51     * @param VariableScope|null $scope The scope keyword written before the name
52     * @param Name|null $qualifier The component, key cache instance or trigger row written before the name
53     */
54    public function __construct(public readonly Name $name, public readonly Scalar|SetWord $value, public readonly ?VariableScope $scope = null, public readonly ?Name $qualifier = null)
55    {
56        Check::input(!$value instanceof BareName || $value->system === ($scope !== null), 'A bare name value is known to be text exactly when a scope keyword names a system variable.');
57    }
58
59    /**
57     * Derives the value.
60     * Derives the value, and checks that SET can change a variable written with a scope; a name without a scope may be a variable of a stored program.
61     */
62    public function deriveItem(Derivation $derivation): void
63    {
64        if ($this->scope !== null && $this->qualifier === null && !$derivation->inProgram()) {
65            (new VariableAccess())->assign($this->name->value, $this->scope, $derivation);
66        } elseif ($this->qualifier !== null && !$derivation->inProgram() && Settings::of($derivation->context)->row($this->qualifier->value) === null) {
67            (new SystemVariable($this->name, null, $this->qualifier))->structured($derivation);
68        }
69        if ($this->value instanceof Scalar) {
70            (new Operands())->single($derivation->scalar($this->value, $derivation->environment()), $derivation);
71        }
72    }
73
74    /**
75     * Writes the scope, the name, an equals sign and the value.
76     */
77    public function render(Output $out): void
78    {
79        if ($this->scope !== null) {
80            $out->keyword($this->scope->value);
81        }
82        if ($this->qualifier !== null) {
83            $out->name($this->qualifier, NameUse::Label)->symbol('.');
84        }
85        $out->name($this->name, NameUse::Label)->symbol('=');
86        if ($this->value instanceof SetWord) {
87            $out->keyword($this->value->value);
88        } else {
89            $out->node($this->value);
90        }
91    }
92}
