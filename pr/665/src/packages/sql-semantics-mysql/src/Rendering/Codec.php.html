13/**
14 * Spells decoded names as MySQL identifiers of one grammar release.
15 *
16 * Rule: MYSQL-NAME-CODEC-001. A name made of ASCII letters, digits and
17 * underscores that starts with a letter and is no keyword, function name or
18 * introducer of the release (MYSQL-KEYWORDS-001) is written bare. Every other
19 * name is written in backticks with embedded backticks doubled. A backtick
20 * word is an identifier under every sql_mode, at every identifier position,
21 * after `@` and after `@@`, and it decodes (MYSQL-IDENTIFIER-DECODE-001) to
22 * the same name, so the spelling does not depend on ANSI_QUOTES or on the
23 * position. A name position that accepts only a string token is not a name
24 * of this codec: its structure holds a string literal.
25 * Source: https://dev.mysql.com/doc/refman/8.4/en/identifiers.html.
26 * Status: Implemented.
27 *
28 * @visibility SqlSemantics
29 */

73    {
74        if ($this->client === null) {

78        $bytes = $encoding->encode($text, $this->client);
79        if ($encoding->decode($bytes, $this->client) !== $text) {
80            throw new \SqlSemantics\Diagnostic\UnrepresentableName('A name cannot be written in ' . $this->client->name . '.');
81        }

84    }
