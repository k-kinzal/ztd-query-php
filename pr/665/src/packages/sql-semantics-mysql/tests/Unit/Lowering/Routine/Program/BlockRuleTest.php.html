65    #[DataProvider('providerBlockLowersAnUnlabeledBlock')]
66    public function testBlockLowersAnUnlabeledBlock(string $release): void
67    {
68        $semantics = new Semantics(Dialect::MySql, $release);
69        $empty = $semantics->analyze('create procedure p() begin end');
70        self::assertInstanceOf(CreateProcedure::class, $empty->statement);
71        self::assertEquals(new Block(), $empty->statement->body);
72        self::assertSame('CREATE PROCEDURE p() BEGIN END', $empty->toString());
73
74        $full = $semantics->analyze('create procedure p() begin declare x int; set x = 1; set x = 2; end');
75        self::assertInstanceOf(CreateProcedure::class, $full->statement);
76        self::assertEquals(new Block(
77            [new VariableDeclaration([new Name('x')], new Integral(IntegralKind::Int))],
78            [new SetVariables([new NameAssignment(new Name('x'), new NumberLiteral('1'))]), new SetVariables([new NameAssignment(new Name('x'), new NumberLiteral('2'))])],
79        ), $full->statement->body);
80        self::assertSame('CREATE PROCEDURE p() BEGIN DECLARE x INT; SET x = 1; SET x = 2; END', $full->toString());
81    }
