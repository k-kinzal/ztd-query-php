59            . '<tbody>' . $rows . '</tbody></table>'

67     * The counts shown above the table.

69    public function summary(Catalog $catalog): string
70    {
71        $exact = 0;
72        $findings = 0;
73        foreach ($catalog as $entry) {
74            $exact += $entry->isExact() ? 1 : 0;
75            $findings += count($entry->findings);
76        }
77
78        return sprintf(
79            '<p class="summary">%d statement(s) &middot; %d fully resolved &middot; %d finding(s)</p>',
80            $catalog->count(),
81            $exact,
82            $findings,

106    /**
107     * What a reader has to know before trusting the statements listed for a call site.
108     */
109    public function caveats(CatalogEntry $entry): string
110    {
111        $caveats = [];
112        if (!$entry->resolution()->isClosed()) {
113            $caveats[] = 'the search did not close, so these may not be all of them';
114        }
115        if (!$entry->correlated) {
116            $caveats[] = 'assembled from parts that vary independently, so some may be unreachable';
117        }
118        if (count($entry->through) > 1) {
119            $caveats[] = 'read through ' . implode(' &rarr; ', array_map($this->escape(...), $entry->through));
120        }
121
122        return $caveats === [] ? '' : '<p class="caveat">' . implode('; ', $caveats) . '</p>';
123    }

128    public function values(CatalogEntry $entry): string
129    {
130        if ($entry->placeholders === []) {

133        $items = '';
134        foreach ($entry->placeholders as $placeholder) {
135            $items .= '<li><code>' . $this->escape($placeholder->token) . '</code> '
