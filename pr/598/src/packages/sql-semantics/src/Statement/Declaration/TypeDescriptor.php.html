7use InvalidArgumentException;

10/**
11 * A database type: its identity and every declared fact that is independent of the name.
12 *
13 * The identity is a Builtin case or a TypeName. Length, precision, scale,
14 * signedness, character set, array dimensions, interval fields, enumeration
15 * members and storage affinity are separate facts, never part of the name. A
16 * fact a dialect does not have keeps its default. The dialect that read the
17 * declaration only produces the Builtin cases it supports.
18 *
19 * @example Describing a declared type
20 *     $type = new \SqlSemantics\Statement\Declaration\TypeDescriptor(\SqlSemantics\Statement\Declaration\Builtin::Numeric, precision: 10, scale: 2);
21 *     $type->is(\SqlSemantics\Statement\Declaration\Builtin::Numeric) // => true
22 *     $type->scale // => 2
23 *
24 * @visibility public
25 */
26final class TypeDescriptor
27{
28    /**
29     * @param Builtin|TypeName $name Type identity
30     * @param int|null $length Declared length, or display width, in the unit of the type
31     * @param int|null $precision Declared precision, or fractional seconds precision
32     * @param int|null $scale Declared scale; requires a precision
33     * @param bool $unsigned Whether negative values are excluded; only for numeric types
34     * @param bool $zerofill Whether displayed values are padded with zeros; implies unsigned
35     * @param bool $binaryCollation Whether the binary collation of the character set was requested
36     * @param string|null $characterSet Declared character set name; only for character types
37     * @param list<Element> $members Enumeration members as typed literals; exactly the enumeration types have them
38     * @param int $arrayDimensions Number of array dimensions; zero for a scalar
39     * @param IntervalFields|null $intervalFields Fields an interval is restricted to
40     * @param Affinity|null $affinity Storage affinity, in dialects that have one; not a runtime storage-class guarantee
41     * @param NumericSize|null $effectiveNumericSize Enforced decimal size, including dialect defaults; null for unconstrained or affinity-only types
42     * @throws InvalidArgumentException When the facts contradict each other
43     */
44    public function __construct(
45        public readonly Builtin|TypeName $name,
46        public readonly ?int $length = null,
47        public readonly ?int $precision = null,
48        public readonly ?int $scale = null,
49        public readonly bool $unsigned = false,
50        public readonly bool $zerofill = false,
51        public readonly bool $binaryCollation = false,
52        public readonly ?string $characterSet = null,
53        public readonly array $members = [],
54        public readonly int $arrayDimensions = 0,
55        public readonly ?IntervalFields $intervalFields = null,
56        public readonly ?Affinity $affinity = null,
57        public readonly ?NumericSize $effectiveNumericSize = null,
58    ) {
59        Invariant::ensure($length === null || $length >= 0, 'A type length cannot be negative.');
60        Invariant::ensure($precision === null || $precision >= 0, 'A type precision cannot be negative.');
61        Invariant::ensure($scale === null || $precision !== null, 'A scale requires a precision.');
62        Invariant::ensure(!$zerofill || $unsigned, 'Zero padding implies an unsigned type.');
63        Invariant::ensure(!$unsigned || ($name instanceof Builtin && $name->isNumeric()), 'Only numeric types carry a sign fact.');
64        Invariant::ensure((!$binaryCollation && $characterSet === null) || ($name instanceof Builtin && $name->isCharacter()), 'Only character types carry a character set or collation fact.');
65        Invariant::members($members, Element::class);
66        Invariant::elements(...$members);
67        Invariant::ensure(($members !== []) === in_array($name, [Builtin::Enum, Builtin::Set], true), 'Exactly the enumeration types have members.');
68        Invariant::ensure($arrayDimensions >= 0, 'Array dimensions cannot be negative.');
69        Invariant::ensure($intervalFields === null || $name === Builtin::Interval, 'Only intervals carry interval fields.');
70        Invariant::ensure($effectiveNumericSize === null || $name === Builtin::Numeric, 'Only an exact decimal type has an effective decimal size.');
71        Invariant::ensure($effectiveNumericSize === null || $precision === null || $precision === 0 || $effectiveNumericSize->precision === $precision, 'Effective precision must preserve declared precision.');
72        Invariant::ensure($effectiveNumericSize === null || $scale === null || $effectiveNumericSize->scale === $scale, 'Effective scale must preserve declared scale.');
73    }
74
75    /**
76     * Compares the type identity, ignoring every other fact.
77     */
78    public function is(Builtin|TypeName $name): bool
79    {
80        if ($name instanceof Builtin || $this->name instanceof Builtin) {
81            return $this->name === $name;
82        }
83
84        return $this->name->equals($name);
85    }
86
87    /**
88     * Names the type identity for diagnostics.
89     */
90    public function label(): string
91    {
92        return $this->name instanceof Builtin ? $this->name->value : $this->name->qualifiedName();
93    }
94}
