7/**
8 * Where a grammar writes table names, and which of those positions declare, drop, or merely name a table.
9 *
10 * A database package describes its grammar with symbols rather than with
11 * generated classes: the symbols whose values are table names, and the forms
12 * whose names are declarations, drops, common table expressions, or names of
13 * other objects that are not tables. Every other table name is a reference.
14 *
15 * It also describes where common table expressions are visible: the
16 * symbols at which a form writes its WITH clause, the fixed word that makes
17 * the clause recursive, which expressions of its own clause the body of an
18 * expression can name, and the positions, such as the target of a DML
19 * statement, whose names never refer to a common table expression.
20 *
21 * @phpstan-type Site array{rule: string, requires?: list<string>, name?: string, pair?: list<string>, names?: string, list?: array{string, list<string>}, conditional?: string, type?: array{string, list<string>}}
22 * @visibility SqlSemantics
23 */
24final class RelationRules
25{
26    /**
27     * @param list<string> $nameSymbols Grammar symbols whose values are complete table names
28     * @param list<Site> $declarations Forms that declare the table they name
29     * @param list<Site> $drops Forms that drop the tables they name
30     * @param list<Site> $commonTableExpressions Forms that define a common table expression by name
31     * @param list<Site> $ignored Forms whose names are not tables, such as views, or are new names
32     * @param list<Site> $pairs Forms that write a table name as separate values, such as a name and a schema
33     * @param array<string, array<string, list<int>>> $parts The child positions that spell the name, by rule and joined symbols, where not every child does
34     * @param list<string> $withClauses Grammar symbols at which a form writes the WITH clause whose expressions its other values can name
35     * @param string $recursive The fixed word, by its terminal name, that makes a WITH clause recursive
36     * @param WithVisibility $visibility The expressions of a plain WITH clause the body of each of them can name
37     * @param WithVisibility $recursiveVisibility The expressions of a recursive WITH clause the body of each of them can name
38     * @param list<string> $targets Grammar symbols whose names are tables even where a common table expression of that name is visible
39     */
40    public function __construct(
41        public readonly array $nameSymbols,
42        public readonly array $declarations = [],
43        public readonly array $drops = [],
44        public readonly array $commonTableExpressions = [],
45        public readonly array $ignored = [],
46        public readonly array $pairs = [],
47        public readonly array $parts = [],
48        public readonly array $withClauses = [],
49        public readonly string $recursive = 'RECURSIVE',
50        public readonly WithVisibility $visibility = WithVisibility::Preceding,

53    ) {
54    }

56    /**
57     * Answers the sites of a kind that match a form, by its rule and symbols.
58     *
59     * @param list<Site> $sites
60     * @param list<string> $symbols
61     * @return list<Site>
62     */
63    public static function matching(array $sites, string $rule, array $symbols): array
64    {
65        return array_values(array_filter($sites, static fn (array $site): bool => $site['rule'] === $rule && array_diff($site['requires'] ?? [], $symbols) === []));
66    }
