21/**
22 * The operands of a 5.x UNION in written order, the quantifiers between them and the clauses written after the last one.
23 *
24 * Rule: MYSQL-UNION-LEGACY-001. The ORDER BY, LIMIT, INTO and locking
25 * clauses written inside the last unparenthesized SELECT of a union apply
26 * to the whole union, so they are taken out of that block; with the
27 * clauses written after a last parenthesized operand they wrap the union.
28 * In a 5.6 subquery an ORDER BY or LIMIT written after the last SELECT
29 * applies to the union instead, and the last SELECT keeps its own clauses
30 * (the `union_order_or_limit` action makes the fake query block the global
31 * parameters). The server rejects, while parsing, ORDER BY or INTO in an
32 * earlier unparenthesized SELECT, LIMIT there in 5.7 (5.6 keeps it as the
33 * block's own), an ORDER BY or LIMIT a 5.6 subquery writes after an earlier
34 * SELECT other than a LIMIT after the first one, and PROCEDURE ANALYSE in a
35 * union. An earlier SELECT keeps its own locking clauses and, in 5.6, its
36 * own LIMIT; when it is not the first operand the operands up to it form a
37 * leading union (MYSQL-LEADING-UNION-001). The operands are combined
38 * left-deep. A lowering-time value.
39 * Source: https://dev.mysql.com/doc/refman/5.7/en/union.html ("To apply
40 * ORDER BY or LIMIT to an individual SELECT, place the clause inside the
41 * parentheses"; "Only the last SELECT statement can use INTO OUTFILE"),
42 * https://github.com/mysql/mysql-server/blob/mysql-5.6.51/sql/sql_parse.cc
43 * (`mysql_new_select`), https://github.com/mysql/mysql-server/blob/mysql-5.7.44/sql/sql_lex.cc
44 * (`LEX::new_union_query`). Status: Implemented.
45 *
46 * @visibility SqlSemantics\Platform\MySql\Lowering\Query
47 */

60    /**
61     * Combines the operands into one query.
62     *
63     * @throws AnalysisException When the last SELECT writes PROCEDURE ANALYSE (ER_WRONG_USAGE of the 5.6 `procedure_analyse_clause` action, which refuses any block but the first, and of 5.7 `PT_procedure_analyse::contextualize`), or an earlier operand writes a clause the server rejects there
64     * @throws ImplementationGap When clauses cannot follow their query block
65     */
66    public function query(): Query
67    {
68        $operands = $this->operands;
69        $last = count($operands) - 1;
70        [$final, $after] = $operands[$last];
71        $after = $after->then($this->trailer);
72        if ($last === 0) {
73            return $final instanceof Block ? $final->finish($after) : $after->wrap($final);
74        }
75        $query = null;
76        foreach (array_slice($operands, 0, $last) as $index => [$operand, $clauses]) {
77            $operand = $operand instanceof Block ? $this->earlier($operand, $clauses, $index) : $this->enclosed($operand, $clauses, $index);
78            $query = match (true) {
79                $query === null => $operand,
80                $operand instanceof Select && $operand->trailed() => new LeadingUnion($query, $this->quantifiers[$index - 1] ?? null, $operand),
81                default => new SetOperation($query, SetOperator::Union, $this->quantifiers[$index - 1] ?? null, $operand),
82            };
83        }
84        $quantifier = $this->quantifiers[$last - 1] ?? null;
85        if (!$final instanceof Block) {
86            return $after->wrap(new SetOperation($query, SetOperator::Union, $quantifier, $final));
87        }
88        if ($final->trailer->procedure !== null) {
89            throw new AnalysisException('Incorrect usage of PROCEDURE and UNION: PROCEDURE ANALYSE follows a single query block only.');
90        }
91        if (!$after->empty() && !$final->trailer->empty()) {
92            return new OrderedSetOperation($query, SetOperator::Union, $quantifier, $final->select(), $after->orderBy, $after->limit);
93        }
94
95        return $final->trailer->then($after)->wrap(new SetOperation($query, SetOperator::Union, $quantifier, $final->bare()->select()));
96    }

98    /**
99     * Finishes an operand before the last one, rejecting the clauses the server does not accept there.
100     *
101     * @throws AnalysisException When the operand writes such a clause: ER_WRONG_USAGE "UNION and ORDER BY|LIMIT|INTO|PROCEDURE ANALYSE" of `add_select_to_union_list` and `mysql_new_select` (5.6, run by the `union_list` action) and of `LEX::new_union_query` (5.7, run while contextualizing the union), or ER_SYNTAX_ERROR when the clauses written after an earlier SELECT made the fake query block current (`add_select_to_union_list` refuses GLOBAL_OPTIONS_TYPE)
102     */
103    public function earlier(Block $operand, Trailer $clauses, int $index): Query
104    {
105        $trailer = $operand->trailer;
106        if ($trailer->orderBy !== [] || ($trailer->limit !== null && $this->release !== GrammarRelease::MySql5651)) {
107            throw new AnalysisException('Incorrect usage of UNION and ORDER BY or LIMIT: only the last SELECT of a union takes them without parentheses.');
108        }
109        if ($trailer->into !== null || $operand->into !== null) {
110            throw new AnalysisException('Incorrect usage of UNION and INTO: only the last SELECT of a union takes INTO.');
111        }
112        if ($trailer->procedure !== null) {
113            throw new AnalysisException('Incorrect usage of UNION and SELECT ... PROCEDURE ANALYSE(): a union takes no PROCEDURE ANALYSE.');
114        }
115        if ($clauses->orderBy !== [] || ($index > 0 && !$clauses->empty())) {
116            throw new AnalysisException('Syntax error: only a LIMIT follows the first SELECT of a union in a subquery.');
117        }
118
119        return $operand->finish($clauses);
120    }

122    /**
123     * Finishes a parenthesized operand before the last one with the ORDER BY and LIMIT a 5.x subquery writes after it.
124     *
125     * After the first operand the clauses belong to the query block in the
126     * parentheses, unless an ORDER BY follows a block that orders or limits
127     * its rows itself: the `order_clause` action (5.6) and
128     * `PT_order::contextualize` (5.7) then make a fake query block current.
129     * After a later operand the fake query block of the union is current.
130     * The next UNION refuses a current fake query block.
131     *
132     * @throws AnalysisException When the clauses made the fake query block current: ER_SYNTAX_ERROR of `add_select_to_union_list` (5.6, GLOBAL_OPTIONS_TYPE) and ER_WRONG_USAGE "UNION and ORDER BY|LIMIT" of `LEX::new_union_query` (5.7)
133     */
134    public function enclosed(Query $operand, Trailer $clauses, int $index): Query
135    {
136        $inner = $operand;
137        while ($inner instanceof ParenthesizedQuery) {
138            $inner = $inner->query;
139        }
140        $ordered = $inner instanceof Select && ($inner->orderBy !== [] || $inner->limit !== null);
141        if (($index > 0 && !$clauses->empty()) || ($clauses->orderBy !== [] && $ordered)) {
142            throw new AnalysisException('Syntax error: the ORDER BY or LIMIT after this parenthesized SELECT applies to the whole union, so no UNION follows it.');
143        }
144
145        return $clauses->wrap($operand);
146    }
