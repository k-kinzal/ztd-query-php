28/**
29 * Wires the MySQL grammar releases, lowering rules and spelling rules into the analysis.
30 *
31 * Rule: MYSQL-PROFILE-001. A profile is one shipped grammar release, the
32 * five lexical sql_mode settings, and the parameter style. The installed
33 * grammar and keyword artifacts must have the digests the release pins.
34 *
35 * Rule: MYSQL-CONTEXT-001. An unqualified table name is searched in one
36 * database, the current database of the session; a context names it, and
37 * `mysql` is not assumed: without a search path the database is the name
38 * `(current)`, which no qualified name written in SQL can equal by accident
39 * of spelling only if the caller never declares it, so callers that use
40 * qualified names give the real name. Table and database names are compared
41 * exactly, which is the server setting lower_case_table_names=0, the default
42 * on Unix; the setting is server configuration, the language profile has no
43 * field for it, and a server that runs with 1 or 2 needs the core profile to
44 * record it (see the family plan). Column names are compared without regard
45 * to ASCII letter case. The server folds every letter by the simple case
46 * mapping of its system character set, so two column names that differ only
47 * in the case of a letter outside ASCII are one name to the server and two
48 * names here; for names whose letters outside ASCII are written identically
49 * the comparison is exact.
50 * Source: https://dev.mysql.com/doc/refman/8.4/en/identifier-case-sensitivity.html.
51 * Status: Implemented.
52 *
53 * @visibility SqlSemantics
54 */
55final class Platform implements \SqlSemantics\Contract\Platform

62    /**
63     * Fixes the profile of a shipped MySQL release under a session mode.
64     *
65     * The release is named as the parser tags it (`mysql-8.4.7`) or by its number alone (`8.4.7`);
66     * without a release the default of the installed parser is used.
67     *
68     * @throws InvalidArgumentException When the release is not shipped, the mode belongs to another database, or the installed grammar artifacts differ from the pinned ones
69     */
70    public function profile(?string $version, ?\SqlSemantics\Contract\Mode $mode, ParameterStyle $parameters): LanguageProfile
71    {
72        if ($mode !== null && !$mode instanceof Mode) {
73            throw new InvalidArgumentException('MySQL reads SQL under a MySQL session mode; ' . $mode::class . ' given.');
74        }
75        $registry = new VersionRegistry();
76        $name = $version ?? $registry->resolve('mysql')->name;
77        $release = GrammarRelease::tryFrom(str_starts_with($name, 'mysql-') ? $name : 'mysql-' . $name);
78        if ($release === null || $release->database() !== 'mysql' || !in_array($release->value, $registry->names('mysql'), true)) {
79            throw new InvalidArgumentException('No semantic profile exists for the selected grammar release.');
80        }
81        $artifact = $registry->resolve('mysql', $release->value);
82        if (hash_file('sha256', $artifact->tablePath) !== $release->grammarDigest() || hash_file('sha256', $artifact->keywordPath) !== $release->keywordDigest()) {
83            throw new InvalidArgumentException('The installed grammar artifacts do not match the fixed semantic profile.');
84        }
85
86        return new LanguageProfile($release, new LexicalSettings($mode?->toString() ?? ''), $parameters);
87    }

93    {
94        $lexical = $profile->lexical;
95        $mode = new SqlMode($lexical->ansiQuotes, $lexical->pipesAsConcat, $lexical->highNotPrecedence, $lexical->noBackslashEscapes, $lexical->ignoreSpace);
96        $key = $profile->grammar->value . '|' . (new Mode($lexical->ansiQuotes, $lexical->pipesAsConcat, $lexical->highNotPrecedence, $lexical->noBackslashEscapes, $lexical->ignoreSpace))->toString() . '|' . $profile->parameters->value;
97
98        return $this->parsers[$key] ??= new MySqlParser($profile->grammar->value, $mode, new VersionRegistry(), $profile->parameters === ParameterStyle::Named ? ParameterSyntax::Named : ParameterSyntax::Native);
99    }

139    {
