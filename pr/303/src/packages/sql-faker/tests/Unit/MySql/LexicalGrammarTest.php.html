90    public function testGeneratesPublicProviderLexemesThroughDialectGrammar(): void
91    {
92        $faker = Factory::create();
93        $faker->seed(12345);
94        $lexical = new LexicalGrammar($faker, 'mysql-8.4.7');
95        $sql = implode(' ', [
96            $lexical->generateQuotedIdentifier(3, 3),
97            $lexical->generateStringLiteral(3, 3),
98            $lexical->generateNationalStringLiteral(3, 3),
99            $lexical->generateDollarQuotedString(3, 3),
100            $lexical->generateIntegerLiteral(10, 10),
101            $lexical->generateDecimalLiteral(4, 2),
102            $lexical->generateFloatLiteral(4, 2, 2, 2),
103            $lexical->generateHexLiteral(4, 4),
104            $lexical->generateQuotedHexLiteral(2, 2),
105            $lexical->generateBinaryLiteral(4, 4),
106        ]);
107
108        self::assertSame([
109            'IDENT_QUOTED', 'TEXT_STRING', 'NCHAR_STRING', 'DOLLAR_QUOTED_STRING_SYM', 'NUM',
110            'DECIMAL_NUM', 'FLOAT_NUM', 'HEX_NUM', 'HEX_NUM', 'BIN_NUM',
111        ], $lexical->tokenize($sql));
112        self::assertSame('10', $lexical->generateLongIntegerLiteral(10, 10));
113        self::assertMatchesRegularExpression('/^[0-9]{1,20}$/', $lexical->generateUnsignedBigIntLiteral());
114        self::assertSame(
115            ['@', 'LEX_HOSTNAME'],
116            $lexical->tokenize('@' . $lexical->generateHostname(2, 2, 3)),
