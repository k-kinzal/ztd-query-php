11/**
12 * Names the generation plans this dialect's provider is built from.
13 *
14 * A plan is how a caller says which SQL it wants, and the ones a provider
15 * offers are a fixed vocabulary rather than something each caller assembles.
16 * Naming them here keeps that vocabulary in one place and lets the provider
17 * read as a list of what it can generate.
18 *
19 * @visibility root
20 */

23    /**
24     * Directs a walk that never leaves a VALUES list with nothing in it.
25     *
26     * @param non-empty-string|null $startRule Rule the walk begins at, or null for the grammar entry point
27     *
28     * @return GenerationPlan<true> Plan whose row lists always carry a row
29     */
30    public static function withoutEmptyRows(?string $startRule = null): GenerationPlan
31    {
32        $plan = $startRule === null
33            ? GenerationPlan::all()
34            : GenerationPlan::fromRule($startRule);
35
36        return $plan
37            ->withPatternForEveryOccurrence('opt_values', ProductionPattern::nonEmpty())
38            ->requiringNonEmpty();
39    }

154     */
155    public static function foreignKeyConstraint(Grammar $grammar): GenerationPlan
156    {
157        $startRule = isset($grammar->ruleMap['table_constraint_def'])
158            ? 'table_constraint_def'
159            : 'key_def';
160        $constraintNameRule = isset($grammar->ruleMap['opt_constraint_name'])
161            ? 'opt_constraint_name'
162            : 'opt_constraint';
163        $constraintNameSymbol = $constraintNameRule === 'opt_constraint_name'
164            ? 'CONSTRAINT'
165            : 'constraint';
166
167        return GenerationPlan::constrained($startRule, [
168            $startRule => [ProductionPattern::containing('FOREIGN', 'KEY_SYM')],
169            $constraintNameRule => [ProductionPattern::containing($constraintNameSymbol)],
170            'opt_ident' => [ProductionPattern::nonEmpty()],
171            'opt_ref_list' => [ProductionPattern::nonEmpty()],
172        ])->requiringNonEmpty();
173    }
