11/**
12 * Builds a MySQL lexical profile from the server's own lexer source.
13 *
14 * The three files it reads are the keyword table, the scanner and the character
15 * state header, and where the last of those lives moved twice across the
16 * supported releases — so the profile is bound to one exact version and records
17 * the hash of every file it was built from.
18 *
19 */
20final class MySqlProfileBuilder
21{
22    /** @readonly */
23    private LexerSource $source;
24
25    /**
26     * @param LexerSource|null $source Reads the upstream lexer files
27     */
28    public function __construct(?LexerSource $source = null)
29    {
30        $this->source = $source ?? new UpstreamLexerSource();
31    }
32
33    /**
34     * Reports the upstream files a release keeps its lexer in.
35     *
36     * The character-state header moved twice across the supported releases, so
37     * where it lives is decided by the version rather than fixed.
38     *
39     * @param string $version Release tag, e.g. "mysql-8.4.7"
40     *
41     * @return array{table: string, scanner: string, state: string} URLs of the three files
42     */
43    public function sourceUrls(string $version): array
44    {
45        $base = "https://raw.githubusercontent.com/mysql/mysql-server/refs/tags/{$version}";
46        $statePath = match (true) {
47            str_starts_with($version, 'mysql-5.6.') => '/include/m_ctype.h',
48            version_compare(substr($version, strlen('mysql-')), '8.1.0', '<') => '/include/sql_chars.h',
49            default => '/strings/sql_chars.h',
50        };
51
52        return [
53            'table' => $base . '/sql/lex.h',
54            'scanner' => $base . '/sql/sql_lex.cc',
55            'state' => $base . $statePath,
56        ];
57    }
58
59    /**
60     * Fetches registration data and records the reviewed scanner source without inferring its semantics.
61     * @return array<string, mixed>
62     * @throws RuntimeException When an upstream file or registration table is unavailable
63     */
64    public function build(string $version): array
65    {
66        ['table' => $tableUrl, 'scanner' => $scannerUrl, 'state' => $stateUrl] = $this->sourceUrls($version);
67        $table = $this->source->fetch($tableUrl);
68        $scanner = $this->source->fetch($scannerUrl);
69        $state = $this->source->fetch($stateUrl);
70        return [
71            'dialect' => 'mysql',
72            'version' => $version,
73            'sources' => [$tableUrl => hash('sha256', $table), $scannerUrl => hash('sha256', $scanner), $stateUrl => hash('sha256', $state)],
74            'registrations' => (new LexicalProfileCompiler())->registrations($table),
75        ];
76    }
77}
