127    /**
128     * Fixes the common keyword handler to the explicitly reviewed releases.
129     */
130    public function create(string $version, KeywordLexemeGenerator $keywords): LexemeGenerator
131    {
132        return new MatchingLexemeGenerator(
133            static fn (LexemeInput $input): bool => in_array($input->terminal()->name, self::TERMINALS, true),
134            new VersionedLexemeGenerator($version, new VersionCase(
135                ['mysql-5.6.51', 'mysql-5.7.44', 'mysql-8.0.44', 'mysql-8.1.0', 'mysql-8.2.0', 'mysql-8.3.0', 'mysql-8.4.7', 'mysql-9.0.1', 'mysql-9.1.0'],
136                $keywords,
137                'mysql-common-keywords',
138            )),
139        );
140    }
