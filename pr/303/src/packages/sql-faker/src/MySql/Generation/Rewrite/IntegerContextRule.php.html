32        }
33        foreach ($sequence->occurrences('source_def') as $id) {
34            $range = $sequence->range($id);
35            $number = $sequence->child($id, 'real_ulong_num') ?? $sequence->child($id, 'ulong_num');
36            if ($range !== null && $number !== null && in_array($sequence->nameAt($range[0]), ['SOURCE_CONNECTION_AUTO_FAILOVER_SYM', 'GTID_ONLY_SYM', 'REQUIRE_ROW_FORMAT_SYM'], true)) {
37                $sequence = $this->mapped($sequence, $number->id, 'REPLICATION_FLAG_NUMBER', 'mysql.replication-flag');
38            }
39        }
40        foreach ($sequence->occurrences('ternary_option') as $id) {
41            $number = $sequence->child($id, 'ulong_num');
42            if ($number !== null) {
43                $sequence = $this->mapped($sequence, $number->id, 'TERNARY_OPTION_NUMBER', 'sql/sql_yacc.yy:ternary_option');
44            }
45        }
46        $sequence = $this->options($sequence);
47        $sequence = $this->yearWidth($sequence);
48        $sequence = $this->weightStringLength($sequence);
49        foreach ($sequence->occurrences('size_number') as $id) {
50            $identifier = $sequence->child($id, 'IDENT_sys');
51            if ($identifier !== null) {
52                $sequence = $this->mapped($sequence, $identifier->id, 'SIZE_NUMBER', 'sql/sql_yacc.yy:size_number');
53            }
54        }
55        foreach ($sequence->occurrences('source_reset_options') as $id) {
56            $number = $sequence->child($id, 'real_ulonglong_num');
57            if ($number !== null) {
58                $sequence = $this->mapped($sequence, $number->id, 'BINLOG_RESET_INDEX', 'sql/sql_yacc.yy:source_reset_options');
59            }
60        }
61        foreach ($sequence->occurrences('xid') as $id) {
62            $number = $sequence->child($id, 'ulong_num');
63            $range = $number === null ? null : $sequence->range($number->id);
64            if ($number !== null && $range !== null && $sequence->nameAt($range[0]) === 'ULONGLONG_NUM') {
65                $sequence = $this->mapped($sequence, $number->id, 'NUM', 'mysql.xid-format-overflow');
66            }
67        }
68        return $sequence;

86     * Keeps YEAR width checks independent of general integer and table-option domains.
87     */
88    public function yearWidth(TerminalSequence $sequence): TerminalSequence
89    {
90        if (!$this->strictYearWidth) {
91            return $sequence;
92        }
93        foreach ($sequence->occurrences('type') as $id) {
94            $range = $sequence->range($id);
95            $width = $sequence->child($id, 'opt_field_length');
96            $field = $width === null ? null : $sequence->child($width->id, 'field_length');
97            $length = $field === null ? null : $sequence->range($field->id);
98            if ($range !== null && $sequence->nameAt($range[0]) === 'YEAR_SYM' && $length !== null) {
99                $sequence = $sequence->replace($length[0] + 1, 1, [
100                    $sequence->terminals[$length[0] + 1]->replaced('YEAR_WIDTH_NUMBER', 'sql/sql_yacc.yy:YEAR_SYM'),
101                ], 'sql/sql_yacc.yy:YEAR_SYM');
102            }
103        }
104        return $sequence;

112        foreach (['source_def' => ['SOURCE_DELAY_SYM' => 'SOURCE_DELAY_NUMBER'], 'master_def' => ['MASTER_DELAY_SYM' => 'SOURCE_DELAY_NUMBER'], 'create_table_option' => ['STATS_SAMPLE_PAGES_SYM' => 'STATS_SAMPLE_PAGES_NUMBER', 'KEY_BLOCK_SIZE' => 'KEY_BLOCK_SIZE_NUMBER', 'AVG_ROW_LENGTH' => 'AVG_ROW_LENGTH_NUMBER'], 'opt_key_algo' => ['ALGORITHM_SYM' => 'KEY_ALGORITHM_NUMBER'], 'opt_num_parts' => ['PARTITIONS_SYM' => 'PARTITION_COUNT_NUMBER'], 'opt_num_subparts' => ['SUBPARTITIONS_SYM' => 'PARTITION_COUNT_NUMBER']] as $context => $terminals) {
113            foreach ($sequence->occurrences($context) as $id) {
114                $range = $sequence->range($id);
115                $number = $sequence->child($id, 'ulong_num') ?? $sequence->child($id, 'ulonglong_num') ?? $sequence->child($id, 'real_ulong_num');
116                $name = $range === null ? null : $sequence->nameAt($range[0]);
117                $terminal = $terminals[$name ?? ''] ?? null;
118                if ($number !== null && $terminal !== null) {
119                    $sequence = $this->mapped($sequence, $number->id, $terminal, 'sql/sql_yacc.yy:' . $name);
120                }
121            }
122        }
