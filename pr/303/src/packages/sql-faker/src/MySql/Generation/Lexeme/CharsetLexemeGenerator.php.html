27    public function generate(LexemeInput $input): ?LexemeCandidates
28    {
29        $generator = new ValueLexemeGenerator('UNDERSCORE_CHARSET', new WordDomain(['_utf8mb4', '_utf8mb3', '_latin1', '_ascii', '_binary'], true), ['_utf8mb4', '_utf8mb3', '_latin1', '_ascii', '_binary'], 'charset', 'sql/sql_yacc.yy:literal:well-formed-charset');
30        $candidates = $generator->generate($input);
31        if ($candidates === null) {
32            return null;
33        }
34        $bytes = $this->bytes($input->right->parts[0]->lexeme->text ?? '');
35        $valid = [];
36        foreach ($candidates->sequences() as $candidate) {
37            if ($this->accepts($candidate->lexemes[0]->text, $bytes)) {
38                $valid[] = $candidate;
39            }
40        }
41        return LexemeCandidates::of(...$valid);
42    }

44    /**
45     * Decodes complete binary literal forms; text-literal quoting consists only of ASCII bytes.
46     */
47    public function bytes(string $literal): string
48    {
49        $hex = (new RadixDomain('0123456789abcdefABCDEF', '0x', ['X', 'x'], 1, 32))->digits($literal);
50        if ($hex !== null) {
51            return pack('H*', (strlen($hex) % 2 === 1 ? '0' : '') . $hex);
52        }
53        $bits = (new RadixDomain('01', '0b', ['B', 'b'], 1, 64))->digits($literal);
54        if ($bits !== null) {
55            if ($bits === '') {
56                return '';
57            }
58            $bits = str_pad($bits, (int) (ceil(strlen($bits) / 8) * 8), '0', STR_PAD_LEFT);
59            return implode('', array_map(static fn (string $byte): string => chr(intval($byte, 2)), str_split($bits, 8)));
60        }
61        return $literal;
62    }

64    /**
65     * latin1 and binary accept every byte; ASCII and the two UTF-8 widths retain their own contracts.
66     */
67    public function accepts(string $charset, string $bytes): bool
68    {
69        return match (strtolower($charset)) {
70            '_binary', '_latin1' => true,
71            '_ascii' => (new Utf8())->valid($bytes, 1),
72            '_utf8mb3' => (new Utf8())->valid($bytes, 3),
73            '_utf8mb4' => (new Utf8())->valid($bytes, 4),
74            default => false,
75        };
76    }
