10/**
11 * PT_part_definition requires one partition kind for the list; PT_partition checks its declared count.
12 * @see https://github.com/mysql/mysql-server/blob/mysql-8.4.7/sql/parse_tree_partitions.cc
13 * @see https://github.com/mysql/mysql-server/blob/mysql-5.7.44/sql/sql_partition.cc
14 * @see https://github.com/mysql/mysql-server/blob/mysql-8.4.7/sql/sql_partition.cc
15 * @see https://github.com/mysql/mysql-server/blob/mysql-5.6.51/sql/sql_partition.cc
16 */
17final class DefinitionRule implements RewriteRule
18{
19    /**
20     * Uses the explicit partition type, or the first definition for an ADD/REORGANIZE list.
21     */
22    public function rewrite(TerminalSequence $sequence): TerminalSequence
23    {
24        $kinds = [];
25        foreach (array_reverse($sequence->occurrences('part_definition')) as $id) {
26            $range = $sequence->range($id);
27            $values = $sequence->child($id, 'opt_part_values');
28            if ($range === null || $values === null) {
29                continue;
30            }
31            $origin = $sequence->terminals[$range[0]];
32            $list = array_search('part_def_list', $origin->rules, true);
33            if ($list === false) {
34                continue;
35            }
36            $group = $origin->ancestors[$list];
37            $valueRange = $sequence->range($values->id);
38            $current = $valueRange === null ? 'HASH' : ($sequence->nameAt($valueRange[0] + 1) === 'IN_SYM' ? 'LIST' : 'RANGE');
39            if (!isset($kinds[$group])) {
40                $clause = $origin->ancestor('partition_clause');
41                $kinds[$group] = $this->declaredKind($sequence, $clause) ?? $current;
42                $sequence = $this->inferCount($sequence, $clause);
43            }
44            if ($current === $kinds[$group]) {
45                continue;
46            }
47            $sequence = $this->replaceValues($sequence, $id, $values->id, $kinds[$group]);
48        }
49        return $sequence;
50    }
51
52    /**
53     * Returns the enclosing declaration's kind, leaving ALTER lists to infer it from their values.
54     *
55     * @return 'HASH'|'LIST'|'RANGE'|null
56     */
57    public function declaredKind(TerminalSequence $sequence, ?int $clause): ?string
58    {
59        $type = $clause === null ? null : $sequence->child($clause, 'part_type_def');
60        $range = $type === null ? null : $sequence->range($type->id);
61        return $range === null ? null : match ($sequence->nameAt($range[0])) {
62            'RANGE_SYM' => 'RANGE', 'LIST_SYM' => 'LIST', default => 'HASH'
63        };
64    }
65
66    /**
67     * Lets MySQL infer the count from the explicit list instead of checking an independent number.
68     */
69    public function inferCount(TerminalSequence $sequence, ?int $clause): TerminalSequence
70    {
71        $count = $clause === null ? null : $sequence->child($clause, 'opt_num_parts');
72        $range = $count === null ? null : $sequence->range($count->id);
73        return $range === null ? $sequence : $sequence->replace($range[0], $range[1] - $range[0], [], 'sql/parse_tree_partitions.cc:explicit-partition-count');
74    }
75
76    /**
77     * Replaces incompatible values while preserving the definition and its occurrence identity.
78     *
79     * @param 'HASH'|'LIST'|'RANGE' $kind
80     */
81    public function replaceValues(TerminalSequence $sequence, int $definition, int $values, string $kind): TerminalSequence
82    {
83        $range = $sequence->range($values);
84        $identifier = $sequence->child($definition, 'ident');
85        $nameRange = $identifier === null ? null : $sequence->range($identifier->id);
86        $start = $range[0] ?? $nameRange[1] ?? null;
87        if ($start === null) {
88            return $sequence;
89        }
90        $names = match ($kind) {
91            'RANGE' => ['VALUES', 'LESS_SYM', 'THAN_SYM', 'MAX_VALUE_SYM'],
92            'LIST' => ['VALUES', 'IN_SYM', '(', 'NUM', ')'],
93            'HASH' => [],
94        };
95        $replacement = [];
96        foreach ($names as $offset => $name) {
97            $replacement[] = $sequence->insertedFor($name, $values, 'sql/parse_tree_partitions.cc:partition-kind', $offset);
98        }
99        return $sequence->replace($start, $range === null ? 0 : $range[1] - $range[0], $replacement, 'sql/parse_tree_partitions.cc:partition-kind');
100    }
