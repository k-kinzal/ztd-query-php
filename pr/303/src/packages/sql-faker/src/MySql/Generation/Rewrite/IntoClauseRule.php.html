11/**
12 * PT_subquery rejects INTO in subqueries; sql_lex.cc/new_set_operation_query allows it only in the final SELECT.
13 * sql_yacc.yy/view_query_block disables SELECT destinations throughout a view definition.
14 * @see https://github.com/mysql/mysql-server/blob/mysql-8.4.7/sql/parse_tree_nodes.cc
15 * @see https://github.com/mysql/mysql-server/blob/mysql-8.4.7/sql/sql_lex.cc
16 * @see https://github.com/mysql/mysql-server/blob/mysql-8.4.7/sql/sql_yacc.yy
17 */
18final class IntoClauseRule implements RewriteRule
19{
20    /**
21     * Removes view, nested or non-final INTO clauses, preserving ordinary outer output destinations.
22     */
23    #[Override]
24    public function rewrite(TerminalSequence $sequence): TerminalSequence
25    {
26        $destinations = [];
27        foreach ($sequence->occurrences('into_clause') as $id) {
28            $range = $sequence->range($id);
29            if ($range === null) {
30                continue;
31            }
32            $origin = $sequence->terminals[$range[0]];
33            $statement = $origin->ancestor('select_stmt');
34            $source = $origin->ancestor('subquery') === null ? null : 'sql/parse_tree_nodes.cc:PT_subquery:into';
35            if ($origin->within('view_query_block')) {
36                $source = 'sql/sql_yacc.yy:view_query_block:into';
37            }
38            for ($index = $range[1]; $source === null && $index < count($sequence->terminals); ++$index) {
39                $following = $sequence->terminals[$index];
40                $last = count($following->ancestors) - 1;
41                if (in_array($following->name, ['UNION_SYM', 'EXCEPT_SYM', 'INTERSECT_SYM'], true)
42                    && ($following->rules[$last] ?? null) === 'query_expression_body'
43                    && in_array($following->ancestors[$last], $origin->ancestors, true)) {
44                    $source = 'sql/sql_lex.cc:new_set_operation_query:into';
45                }
46            }
47            if ($source === null && $statement !== null) {
48                $source = isset($destinations[$statement]) ? 'sql/parse_tree_nodes.cc:PT_select_stmt:multiple-into' : null;
49                $destinations[$statement] = true;
50            }
51            if ($source !== null) {
52                $sequence = $sequence->replace($range[0], $range[1] - $range[0], [], $source);
53            }
54        }
55        return $sequence;
56    }
