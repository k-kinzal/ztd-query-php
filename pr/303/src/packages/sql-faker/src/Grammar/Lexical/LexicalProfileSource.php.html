10/**
11 * Loads the checked-in lexical profile for one dialect and version.
12 *
13 * A profile is generated from the server's own lexer and committed as a PHP
14 * file, so it is loaded rather than trusted: a file that names a different
15 * dialect or version than the one asked for describes a different server, and
16 * generating against it would produce SQL that only looks right.
17 */
18final class LexicalProfileSource
19{
20    /**
21     * Loads the profile a dialect and version were built from.
22     *
23     * @param string $dialect Dialect key, e.g. "mysql"
24     * @param string $version Profile version, e.g. "mysql-8.4.7"
25     *
26     * @return array<string, mixed> The profile as the file declares it
27     *
28     * @throws RuntimeException When the file is missing or describes another server
29     */
30    public function load(string $dialect, string $version): array
31    {
32        $path = SqlVersion::resolve($dialect, $version)->lexicalPath;
33        if (!file_exists($path)) {
34            throw new RuntimeException("Lexical profile file not found: {$path}");
35        }
36
37        $loaded = require $path;
38        if (!is_array($loaded)) {
39            throw new RuntimeException("Invalid lexical profile file: {$path}");
40        }
41        $profile = array_filter($loaded, static fn (int|string $key): bool => is_string($key), ARRAY_FILTER_USE_KEY);
42
43        if (($profile['dialect'] ?? null) !== $dialect || ($profile['version'] ?? null) !== $version) {
44            throw new RuntimeException(sprintf(
45                'Invalid %s lexical profile: %s',
46                $this->displayName($dialect),
47                $path,
48            ));
49        }
50
51        return $profile;
52    }
53
54    /**
55     * Spells a dialect key the way messages spell it.
56     *
57     * @param string $dialect Dialect key, e.g. "postgresql"
58     *
59     * @return string Display name, e.g. "PostgreSQL"
60     */
61    public function displayName(string $dialect): string
62    {
63        return match ($dialect) {
64            'mysql' => 'MySQL',
65            'postgresql' => 'PostgreSQL',
66            'sqlite' => 'SQLite',
67            default => $dialect,
68        };
69    }
70}
