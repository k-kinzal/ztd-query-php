16/**
17 * Faker Provider for generating syntactically valid PostgreSQL SQL statements.
18 *
19 * This provider uses PostgreSQL's official Bison grammar (gram.y) to generate
20 * SQL that is syntactically valid. Note that generated SQL may not be semantically
21 * valid (tables/columns may not exist).
22 *
23 * maxDepth selects shortest productions once the depth is reached; it is not a SQL length limit.
24 * The basic pipeline resolves lexical spaces without inserting comments. Optional clauses may be empty.
25 * Seed the supplied Faker generator to reproduce a run within the same package version.
26 *
27 * @visibility public
28 *
29 * @example Register the provider with Faker
30 *     $faker = \Faker\Factory::create();
31 *     $faker->addProvider(new \SqlFaker\PostgreSqlProvider($faker));
32 *     $faker->parameterMarker(min: 2, max: 2) // => '$2'
33 */
34final class PostgreSqlProvider extends Base

39     * Register SQL formatters on the supplied Faker generator.
40     * @param Generator $generator Faker generator
41     * @param string|null $version PostgreSQL version tag. Null for default.
42     *
43     * @visibility public
44     * @example Choose a supported database version
45     *     $faker = \Faker\Factory::create();
46     *     $provider = new \SqlFaker\PostgreSqlProvider($faker, 'pg-17.2');
47     *     $provider->integerLiteral(min: 42, max: 42) // => '42'
48     * @example Reject an unsupported database version
49     *     new \SqlFaker\PostgreSqlProvider(\Faker\Factory::create(), 'missing-version') // throws \RuntimeException: Unsupported
50     */
51    public function __construct(Generator $generator, ?string $version = null)
52    {
53        parent::__construct($generator);
54
55        $generator->addProvider($this);
56        $resolvedVersion = PgGrammar::resolveVersion($version);
57        $this->sql = SqlGeneratorFactory::forPostgreSql($generator, PgGrammar::load($resolvedVersion), $resolvedVersion);
58    }
