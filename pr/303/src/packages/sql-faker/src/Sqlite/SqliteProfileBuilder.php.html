12 * Builds a SQLite lexical profile from the release's own tokenizer source.
13 *
14 * SQLite spells its keyword table as C rather than as data, so the profile is
15 * compiled out of the tokenizer and bound to one exact release.
16 */
17final class SqliteProfileBuilder
18{

23     * @param LexerSource|null $source Reads the upstream lexer files
24     */

26    {
27        $this->source = $source ?? new UpstreamLexerSource();
28    }

30    /**
31     * Reports the upstream files a release keeps its tokenizer in.
32     *
33     * SQLite tags its releases by number without the project name, so the
34     * version this package uses has to be spelled back into a tag name.
35     *
36     * @param string $version Release tag, e.g. "sqlite-3.47.2"
37     *
38     * @return array{keywords: string, scanner: string} URLs of the two files
39     */
40    public function sourceUrls(string $version): array
41    {
42        $base = 'https://raw.githubusercontent.com/sqlite/sqlite/refs/tags/version-'
43            . substr($version, strlen('sqlite-'));
44
45        return [
46            'keywords' => $base . '/tool/mkkeywordhash.c',
47            'scanner' => $base . '/src/tokenize.c',
48        ];
49    }
50
51    /**
52     * Fetches registration data and records the reviewed scanner source without inferring its semantics.
53     * @return array<string, mixed>
54     * @throws RuntimeException When an upstream file or registration table is unavailable
55     */
56    public function build(string $version): array
57    {
58        ['keywords' => $keywordUrl, 'scanner' => $scannerUrl] = $this->sourceUrls($version);
59        $keywords = $this->source->fetch($keywordUrl);
60        $scanner = $this->source->fetch($scannerUrl);
61        return [
62            'dialect' => 'sqlite',
63            'version' => $version,
64            'sources' => [$keywordUrl => hash('sha256', $keywords), $scannerUrl => hash('sha256', $scanner)],
65            'keywords' => (new LexicalProfileCompiler())->compile($keywords),
66        ];
67    }
