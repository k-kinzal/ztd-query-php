21    /**
22     * Preserves line count while omitting source inactive in the target parser build.
23     * @throws RuntimeException When conditional directives are unbalanced or malformed
24     */
25    public function process(string $input): string
26    {
27        $active = true;
28        $stack = [];
29        $output = [];
30        foreach (explode("\n", $input) as $line) {
31            if (preg_match('/^%(if|ifdef|ifndef|else|endif)\b(.*)$/', $line, $match) !== 1) {
32                $output[] = $active ? $line : '';
33                continue;
34            }
35            $directive = $match[1];
36            if ($directive === 'else' || $directive === 'endif') {
37                $frame = array_pop($stack);
38                if ($frame === null || ($directive === 'else' && $frame[2])) {
39                    throw new RuntimeException('Unmatched Lemon directive: %' . $directive);
40                }
41                $active = $frame[0];
42                if ($directive === 'else') {
43                    $stack[] = [$frame[0], $frame[1], true];
44                    $active = $frame[0] && !$frame[1];
45                }
46            } else {
47                $selected = $active && $this->condition->evaluate(trim($match[2]));
48                $selected = $directive === 'ifndef' ? !$selected : $selected;
49                $stack[] = [$active, $selected, false];
50                $active = $active && $selected;
51            }
52            $output[] = '';
53        }
54        if ($stack !== []) {
55            throw new RuntimeException('Unterminated Lemon conditional directive.');
56        }
57        return implode("\n", $output);
58    }
