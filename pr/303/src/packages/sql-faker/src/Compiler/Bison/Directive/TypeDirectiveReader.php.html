13/**
14 * Reads `%type`, which gives rules the semantic type their values carry.
15 *
16 * The type tag is what the directive exists for, so a `%type` without one
17 * declares nothing and is reported as absent rather than as an empty list.
18 *
19 * @visibility root

30    {

32    }
33
34    /**

36     *
37     * @param string $directive Directive name including its percent sign
38     *
39     * @return bool True for "%type"
40     */
41    #[Override]
42    public function handles(string $directive): bool
43    {
44        return $directive === '%type';
45    }

47    /**
48     * Consumes the type tag and the rule names it applies to.
49     *
50     * @param BisonTokenStream $stream Stream positioned just after the directive name
51     * @param string $directive Directive name including its percent sign
52     *
53     * @return BisonDeclaration|null The declaration, or null when no type tag follows
54     */
55    #[Override]
56    public function read(BisonTokenStream $stream, string $directive): ?BisonDeclaration
57    {
58        unset($directive);
59
60        $typeTag = $stream->nextIf(BisonLexeme::TypeTag)?->asString();
61        if ($typeTag === null) {
62            return null;
63        }
64
65        /** @var list<string> $symbols */
66        $symbols = [];
67        while ($this->boundary->continuesWith($stream->peek()->type)) {
68            $symbol = $stream->nextIf(BisonLexeme::Identifier)?->asString();
69            if ($symbol === null) {
70                $stream->next();
71                continue;
72            }
73
74            $symbols[] = $symbol;
75        }
76
77        return new BisonTypeDeclaration($typeTag, $symbols);
78    }
