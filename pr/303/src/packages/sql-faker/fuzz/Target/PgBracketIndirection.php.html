7/**
8 * Recognises the syntax error PostgreSQL raises for bracketed indirection.
9 *
10 * PostgreSQL only allows subscripting on an expression it can resolve to an
11 * array, so the grammar can legitimately emit `a[1] b` shapes that the parser
12 * rejects with `syntax error at or near "b"`. That rejection is a property of
13 * the dialect rather than a grammar defect, and it is distinguished from a real
14 * syntax error by checking that the token PostgreSQL complained about is the one
15 * that directly follows a closing bracket in the statement.
16 */
17final class PgBracketIndirection
18{
19    /**
20     * Checks whether a PostgreSQL syntax error is explained by bracketed indirection.
21     *
22     * @param string $sql Statement that PostgreSQL rejected
23     * @param string $message Message PostgreSQL reported for the rejection
24     *
25     * @return bool True when the rejected token directly follows a closing bracket
26     */
27    public function explains(string $sql, string $message): bool
28    {
29        if (!str_contains($sql, '[')) {
30            return false;
31        }
32
33        if (preg_match('/at or near "([^"]+)"/', $message, $matches) !== 1) {
34            return false;
35        }
36
37        return preg_match('/\]\s+' . preg_quote($matches[1], '/') . '\b/', $sql) === 1;
38    }
39}
