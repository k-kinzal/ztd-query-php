40     * Writes a declared table from its definition: the CREATE header, the table elements and the table options.
41     *
42     * @param list<\SqlSemantics\Model\Definition\Table\TemplatePlacement> $templates
43     * @param list<\SqlSemantics\Model\Definition\Relation\Constraint\ExclusionConstraint> $exclusions
44     */
45    public static function declared(\SqlSemantics\Schema\TableDefinition $table, \SqlSemantics\Dialect $dialect, bool $ifNotExists, array $templates = [], array $exclusions = [], ?string $catalog = null): Tree
46    {
47        $columns = [];
48        $columnKey = self::columnKey($table, $dialect);
49        $keyConflict = self::keyConflict($table);
50        $constraints = Definition\Columns::unserial($table->columns, $table->constraints);
51        foreach ($table->columns as $position => $column) {
52            array_push($columns, ...Definition\Table\PostgreSqlTables::templates($templates, $position));
53            $written = Definition\Columns::write($column, $dialect, $keyConflict, Definition\Columns::serialKey($column, $table->constraints) !== null);
54            $columns[] = $columnKey !== null && $columnKey->localColumns() === [$column->name] ? new Tree('column', [$written, Definition\Constraints::column($columnKey, $dialect)]) : $written;
55        }
56        array_push($columns, ...Definition\Table\PostgreSqlTables::templates($templates, count($table->columns)));
57        foreach ($constraints as $constraint) {
58            if ($constraint === $columnKey || $dialect === \SqlSemantics\Dialect::Sqlite && $constraint instanceof PrimaryKey && array_filter($table->columns, static fn ($column): bool => $column->generation instanceof AutoIncrementColumn) !== []) {
59                continue;
60            }
61            $columns[] = Definition\Constraints::write($constraint, $dialect);
62        }
63        foreach ($table->indexes as $index) {
64            $columns[] = Definition\Indexes::inline($index, $dialect);
65        }
66        foreach ($exclusions as $exclusion) {
67            $columns[] = Definition\Relation\ConstraintActions::exclusion($exclusion);
68        }
69        $properties = $table->properties;
70        $modifier = $properties instanceof Table\PostgreSqlProperties ? match ($properties->persistence) {
71            Table\Persistence::Permanent => '', Table\Persistence::Temporary => 'TEMPORARY ', Table\Persistence::Unlogged => 'UNLOGGED ',
72        } : (($properties instanceof Table\MySqlProperties || $properties instanceof Table\SqliteProperties) && $properties->temporary ? 'TEMPORARY ' : '');
73        return new Tree('create-table', [Build::keyword('CREATE ' . $modifier . 'TABLE' . ($ifNotExists ? ' IF NOT EXISTS' : '')), Build::identifier($catalog === null ? self::tableName($table) : [$catalog, $table->schema, $table->name], $dialect), Build::parentheses(Build::separated($columns)), Definition\Storage::table($properties, $dialect)]);

85    /**
86     * Returns the SQLite primary key that must stay a column constraint: a descending key on one INTEGER column that
87     * may hold NULL was declared on the column, where DESC keeps it from aliasing the rowid, while the same key
88     * written as a table constraint would alias the rowid and forbid NULL.
89     */
90    public static function columnKey(\SqlSemantics\Schema\TableDefinition $table, \SqlSemantics\Dialect $dialect): ?PrimaryKey
91    {
92        if ($dialect !== \SqlSemantics\Dialect::Sqlite) {
93            return null;
94        }
95        foreach ($table->constraints as $constraint) {
96            if (!$constraint instanceof PrimaryKey || count($constraint->keys) !== 1 || $constraint->keys[0]->direction !== \SqlSemantics\Schema\Index\Direction::Descending) {
97                continue;
98            }
99            foreach ($table->columns as $column) {
100                if ($constraint->localColumns() === [$column->name] && $column->type->name === 'integer' && $column->nullability !== \SqlSemantics\Type\Nullability::NotNull) {
101                    return $constraint;
102                }
103            }
104        }
105        return null;
106    }
