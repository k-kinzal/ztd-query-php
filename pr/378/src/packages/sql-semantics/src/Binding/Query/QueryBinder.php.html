40     * Resolves one query scope and retains all its nested relational stages.
41     */
42    public function bind(Node $source, ?Scope $parent = null): BoundSelect
43    {
44        $id = $this->context->ids->scope();
45        $context = $this->with($source, $parent);
46        $body = QueryNodes::body($source);
47        $operator = QueryNodes::setOperator($body);
48        if ($operator !== null) {
49            return $this->compound($source, $body, $context, $id, $operator, $parent);
50        }
51        $fromNode = $body->name === 'derived_table_list' ? $body : (QueryNodes::local($body, ['from_clause', 'select_from', 'from'])[0] ?? null);
52        $inputs = new FromBinder($context->tables, $context->ids, $context, $parent, $id);
53        $from = $fromNode === null ? $inputs->explicit($body) : $inputs->bind($fromNode);
54        $scope = $from->scope ?? new Scope($context->tables->identifiers, parent: $parent, queries: $context);
55        $where = $this->expressions($body, ['where_clause', 'opt_where_clause', 'where_opt'], $scope)[0] ?? null;
56        $having = $this->expressions($body, ['having_clause', 'opt_having_clause', 'having_opt'], $scope)[0] ?? null;
57        foreach ([$where, $having] as $predicate) {
58            if ($predicate !== null) {
59                (new ExpressionRules($scope->identifiers->dialect, $scope->diagnostics()))->predicate($predicate);
60            }
61        }
62        $values = (new \SqlSemantics\Binding\Statement\ValuesBinder())->rows($body, $scope);
63        $outputs = $values === [] ? (new ProjectionBinder())->bind($body, $scope) : (new \SqlSemantics\Binding\Statement\ValuesBinder())->outputs($values, $body, $scope);
64        $options = QueryNodes::local($body, ['distinct_clause', 'select_options', 'distinct']);
65        $distinct = $options !== [] && str_contains(strtoupper(Tree::text($options[0])), 'DISTINCT');
66        $tail = new SelectModifiersBinder();
67        [$limit, $offset] = $tail->pagination($source, $scope);
68        $groups = $this->expressions($body, ['group_clause', 'opt_group_clause', 'groupby_opt'], $scope);
69        $clauses = [];
70        foreach (['window_clause', 'opt_window_clause', 'windowdefn_list', 'distinct_clause', 'into_clause', 'locking_clause', 'for_locking_clause'] as $name) {
71            if (QueryNodes::local($body, [$name]) !== []) {
72                $clauses[$name] = $this->expressions($body, [$name], $scope);
73            }
74        }
75        return new BoundSelect($id, $from?->relation, $scope->relations, $outputs, $where, $distinct, $tail->ordering($source, $scope, $outputs), $limit, $offset, $source, $groups, $having, $context->ctes, clauses: $clauses, rows: $values, withTies: str_contains(strtoupper(Tree::text(QueryNodes::local($source, ['limit_clause'])[0] ?? new Node('empty', 0, []))), 'WITH TIES'), syntaxClauses: QueryNodes::clauses($source));
76    }

93    /**
94     * Binds CTEs in declaration order, with recursive names visible to their body.
95     *
96     * @throws SemanticException
97     */
98    public function with(Node $source, ?Scope $parent): QueryContext
99    {
100        $ctes = $this->context->ctes;
101        $with = QueryNodes::local($source, ['with_clause', 'wqlist']);
102        if ($with === []) {
103            $with = Tree::outer($source, ['with_clause', 'wqlist', 'simple_select', 'query_specification', 'oneselect']);
104            $with = array_values(array_filter($with, static fn (Node $node): bool => in_array($node->name, ['with_clause', 'wqlist'], true)));
105        }
106        foreach ($with as $clause) {
107            foreach (Tree::outer($clause, ['common_table_expr', 'wqitem']) as $cte) {
108                $nameNode = Tree::child($cte, ['name', 'ident', 'withnm']);
109                $queryNode = Tree::outer($cte, ['SelectStmt', 'subquery', 'select', 'InsertStmt', 'UpdateStmt', 'DeleteStmt', 'MergeStmt'])[0] ?? null;
110                if ($nameNode === null || $queryNode === null) {
111                    throw new SemanticException('invalid-cte', 'A CTE requires a name and a query.', $cte);
112                }
113                $name = $this->context->tables->identifiers->parts($nameNode)[0];
114                $context = new QueryContext($this->context->tables, $this->context->ids, $ctes);
115                if (in_array($queryNode->name, ['InsertStmt', 'UpdateStmt', 'DeleteStmt', 'MergeStmt'], true)) {
116                    $ctes[$name] = $this->rename((new MutationBinder($context))->bind($queryNode, $queryNode), $cte);
117                    continue;
118                }
119                $body = QueryNodes::body($queryNode);
120                if (QueryNodes::setOperator($body) !== null) {
121                    $branches = $this->branches($body);
122                    $ctes[$name] = $this->rename($context->bind($branches[0], $parent), $cte);
123                    $context = new QueryContext($context->tables, $context->ids, $ctes);
124                }
125                $ctes[$name] = $this->rename($context->bind($queryNode, $parent), $cte);
126            }
127        }
128        return new QueryContext($this->context->tables, $this->context->ids, $ctes);
129    }

134    public function rename(BoundStatement $query, Node $cte): BoundStatement
135    {
136        $aliases = Tree::child($cte, ['opt_name_list', 'opt_derived_column_list', 'eidlist_opt']);
137        if ($aliases === null) {
138            return $query;
139        }
140        $names = array_values(array_filter($this->context->tables->identifiers->parts($aliases), static fn (string $name): bool => !in_array($name, ['(', ')', ','], true)));
141        $outputs = [];
142        foreach ($query->outputs as $index => $output) {
143            $outputs[] = new OutputColumn($index, $names[$index] ?? $output->name, $output->expression);
144        }
145        $class = $query::class;
146        return new $class($query->scopeId, $query->from, $query->relations, $outputs, $query->where, $query->distinct, $query->orderBy, $query->limit, $query->offset, $query->source, $query->groupBy, $query->having, $query->ctes, $query->branches, $query->setOperator, $query->clauses, kind: $query->kind, targets: $query->targets, assignments: $query->assignments, queries: $query->queries, rows: $query->rows, withTies: $query->withTies, syntaxClauses: $query->syntaxClauses, declarations: $query->declarations);
147    }

149    /**
150     * @return non-empty-list<Node>
151     *
152     * @throws SemanticException
153     */
154    public function branches(Node $body): array
155    {
156        $branches = [];
157        foreach ($body->children as $child) {
158            if ($child instanceof Node && in_array($child->name, ['select_clause', 'query_expression_body', 'query_specification', 'selectnowith', 'oneselect', 'select_part2', 'select_init', 'select_paren', 'select_derived_union', 'select_derived'], true)) {
159                $branches[] = $child;
160            }
161        }
162        if ($branches === []) {
163            throw new SemanticException('invalid-set-operation', 'A compound query requires query operands.', $body);
164        }
165        return $branches;

168    /**
169     * Combines output types by ordinal while retaining each branch's dependencies.
170     *
171     * @throws SemanticException
172     */
173    public function compound(Node $source, Node $body, QueryContext $context, string $id, string $operator, ?Scope $parent): BoundSelect
174    {
175        $branches = array_map(static fn (Node $node): BoundSelect => $context->bind($node, $parent), $this->branches($body));
176        $outputs = [];
177        foreach ($branches[0]->outputs as $index => $output) {
178            $operands = [];
179            foreach ($branches as $branch) {
180                if (count($branch->outputs) !== count($branches[0]->outputs)) {
181                    $context->tables->diagnostics->report('set-column-count', 'Compound query operands must have the same width.', $body);
182                }
183                if (!isset($branch->outputs[$index])) {
184                    continue;
185                }
186                $value = $branch->outputs[$index]->expression;
187                $operands[] = $value->kind === ExpressionKind::Cast && $value->symbol === 'implicit' && isset($value->operands[0]) && $value->operands[0]->type->name === 'unknown' ? $value->operands[0] : $value;
188            }
189            $type = (new TypeResolution($context->tables->identifiers->dialect, $context->tables->diagnostics))->common($operands, $body);
190            $nullable = array_filter($operands, static fn (Expression $value): bool => $value->nullability !== Nullability::NotNull) !== [];
191            $outputs[] = new OutputColumn($index, $output->name, new Expression(ExpressionKind::Operator, $type, $nullable ? Nullability::MaybeNull : Nullability::NotNull, $body, $operands, symbol: $operator));
192        }
193        $scope = new Scope($context->tables->identifiers, parent: $parent, queries: $context);
194        $tail = new SelectModifiersBinder();
195        [$limit, $offset] = $tail->pagination($source, $scope);
196        return new BoundSelect($id, null, [], $outputs, null, !str_ends_with($operator, 'ALL'), $tail->ordering($source, $scope, $outputs), $limit, $offset, $source, ctes: $context->ctes, branches: $branches, setOperator: $operator);
197    }
