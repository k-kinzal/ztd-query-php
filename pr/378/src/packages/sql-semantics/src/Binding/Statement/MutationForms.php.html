31     * @param list<TableUse> $targets
32     * @param list<Assignment> $writes
33     * @param list<OutputColumn> $outputs
34     * @param list<Ordering> $orderBy
35     * @throws UnclassifiedSql

38    public static function update(Origin $origin, Node $source, ?BoundRelation $input, array $targets, array $writes, ?Expression $where, array $outputs, QueryContext $context, array $orderBy, ?Expression $limit): UpdateStatement
39    {
40        if ($targets === [] || $writes === []) {
41            throw new UnclassifiedSql('An UPDATE requires a target and assignments.');
42        }
43        $with = (new CteBinder())->clause($origin->source, $context);
44        $words = array_map(static fn ($token): string => strtoupper($token->text), $source->tokens());
45        $prefix = self::modifiers($words, 'UPDATE');
46        $conflict = \SqlSemantics\Ast\Tree::child($source, ['orconf']);
47        $resolution = $conflict === null ? null : \SqlSemantics\Ast\Tree::child($conflict, ['resolvetype']);
48        $response = $resolution === null ? ConstraintResponse::Default : ConstraintResponse::from(strtoupper(\SqlSemantics\Ast\Tree::text($resolution)));
49        if ($input?->relation instanceof Join) {
50            if ($origin->dialect === \SqlSemantics\Dialect::MySql) {
51                if ($orderBy !== [] || $limit !== null) {
52                    throw new \SqlSemantics\InvalidSql(\SqlSemantics\Model\Validation\InputViolation::JoinedMutationPagination, $source);
53                }
54                return new Mutation\UpdateJoinedStatement($origin, $input->relation, $targets, $writes, $where, $outputs, $with, in_array('LOW_PRIORITY', $prefix, true), in_array('IGNORE', $prefix, true));
55            }
56            return new Mutation\UpdateFromStatement($origin, $targets[0], $input->relation->right, $writes, $where, $outputs, $with, $response);
57        }
58        return new Mutation\UpdateTableStatement($origin, $targets[0], $writes, $where, $outputs, $with, $orderBy, $limit, $response, in_array('LOW_PRIORITY', $prefix, true), $origin->dialect === \SqlSemantics\Dialect::MySql && in_array('IGNORE', $prefix, true));
59    }

69    {
70        if ($targets === []) {

72        }
73        $with = (new CteBinder())->clause($origin->source, $context);
74        $words = array_map(static fn ($token): string => strtoupper($token->text), $source->tokens());
75        $prefix = self::modifiers($words, 'DELETE');
76        if ($input !== null && ($input->relation instanceof Join || ($origin->dialect === \SqlSemantics\Dialect::MySql && \SqlSemantics\Binding\Query\QueryNodes::local($source, ['table_alias_ref_list', 'table_wild_list']) !== []))) {
77            if ($origin->dialect === \SqlSemantics\Dialect::MySql) {
78                if ($orderBy !== [] || $limit !== null) {
79                    throw new \SqlSemantics\InvalidSql(\SqlSemantics\Model\Validation\InputViolation::JoinedMutationPagination, $source);
80                }
81                return new Mutation\DeleteJoinedStatement($origin, $input->relation, \SqlSemantics\Binding\Write\DeleteTargets::tables($targets, $source), $where, $outputs, $with, in_array('LOW_PRIORITY', $prefix, true), in_array('IGNORE', $prefix, true), in_array('QUICK', $prefix, true));
82            }
83            return new Mutation\DeleteUsingStatement($origin, $targets[0], $input->relation->right, $where, $outputs, $with);
84        }
85        return new Mutation\DeleteTableStatement($origin, $targets[0], $where, $outputs, $with, $orderBy, $limit, in_array('LOW_PRIORITY', $prefix, true), $origin->dialect === \SqlSemantics\Dialect::MySql && in_array('IGNORE', $prefix, true), in_array('QUICK', $prefix, true));
86    }

88    /**
89     * Returns the MySQL modifiers written right after the verb, so an IGNORE INDEX hint on a table is not read as
90     * the IGNORE modifier.
91     * @param list<string> $words Uppercase statement words
92     * @return list<string>
93     */
94    public static function modifiers(array $words, string $verb): array
95    {
96        $start = array_search($verb, $words, true);
97        $modifiers = [];
98        foreach (array_slice($words, $start === false ? 0 : $start + 1) as $word) {
99            if (!in_array($word, ['LOW_PRIORITY', 'QUICK', 'IGNORE'], true)) {
100                break;
101            }
102            $modifiers[] = $word;
103        }
104        return $modifiers;
105    }
