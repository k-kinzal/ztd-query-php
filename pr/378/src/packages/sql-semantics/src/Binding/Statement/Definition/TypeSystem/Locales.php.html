42    /**
43     * CREATE CONVERSION names two known encodings other than SQL_ASCII and the conversion function.
44     * @throws UnclassifiedSql
45     * @throws InvalidSql
46     */
47    public static function conversion(Origin $origin, Node $source, QueryContext $context): Statement\CreateConversionStatement
48    {
49        $names = Tree::outer($source, ['any_name']);
50        $encodings = array_map(static fn (Node $constant): ?ServerEncoding => self::encoding(TypeDefinitions::label($constant, $context)), Tree::outer($source, ['Sconst']));
51        $from = $encodings[0] ?? null;
52        $to = $encodings[1] ?? null;
53        if ($from === null || $to === null || $from === ServerEncoding::SqlAscii || $to === ServerEncoding::SqlAscii) {
54            throw new InvalidSql(InputViolation::ConversionEncoding, $source);
55        }
56        $name = ObjectAddresses::name($names[0] ?? throw new UnclassifiedSql('A conversion requires its name.'), $context, 2);
57        $function = ObjectAddresses::name($names[1] ?? throw new UnclassifiedSql('A conversion requires its function.'), $context, 2);
58        return new Statement\CreateConversionStatement($origin, $name, $from, $to, $function, Tree::child($source, ['opt_default']) !== null);
59    }

75    public static function collation(Origin $origin, Node $source, QueryContext $context): Statement\CreateCollationStatement|Statement\CopyCollationStatement
76    {
77        $names = Tree::outer($source, ['any_name']);
78        $name = ObjectAddresses::name($names[0] ?? throw new UnclassifiedSql('A collation requires its name.'), $context, 2);
79        $ifNotExists = strtoupper($source->tokens()[2]->text ?? '') === 'IF';
80        if (isset($names[1])) {
81            return new Statement\CopyCollationStatement($origin, $name, ObjectAddresses::name($names[1], $context, 2), $ifNotExists);
82        }
83        $elements = DefinitionElement::named(DefinitionElement::list(Tree::child($source, ['definition']) ?? $source, $context), ['from', 'locale', 'lc_collate', 'lc_ctype', 'provider', 'deterministic', 'rules', 'version'], false);
84        if (isset($elements['from'])) {
85            return count($elements) === 1 ? new Statement\CopyCollationStatement($origin, $name, DefinitionArguments::objectName($elements['from'], $context), $ifNotExists) : throw new InvalidSql(InputViolation::DefinitionRequirement, $source);
86        }
87        $text = static fn (string $key): ?string => isset($elements[$key]) ? DefinitionArguments::text($elements[$key]->argument ?? throw new InvalidSql(InputViolation::DefinitionArgument, $elements[$key]->source), $context) : null;
88        $provider = CollationProvider::tryFrom(strtolower($text('provider') ?? 'libc')) ?? throw new InvalidSql(InputViolation::DefinitionArgument, $elements['provider']->source ?? $source);
89        $deterministic = $elements['deterministic']->argument ?? null;
90        try {
91            return new Statement\CreateCollationStatement($origin, $name, $provider, $text('locale'), $text('lc_collate'), $text('lc_ctype'), $deterministic === null || DefinitionArguments::boolean($deterministic, $context), $text('rules'), $text('version'), $ifNotExists);
92        } catch (InvalidStructure $error) {
93            throw new InvalidSql(InputViolation::DefinitionRequirement, $source, $error);
94        }
95    }
