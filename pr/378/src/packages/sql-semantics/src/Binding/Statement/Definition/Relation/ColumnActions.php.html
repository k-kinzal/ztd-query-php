44    public static function read(Node $command, Scope $scope, QueryContext $context): RelationAction
45    {
46        $words = ObjectAddresses::words($command);
47        $columnNode = Tree::child($command, ['ColId', 'Iconst']) ?? throw new UnclassifiedSql('A column action requires its column.');
48        $column = $context->tables->identifiers->name($columnNode->tokens()[0]);
49        $tail = array_slice($words, (Tree::child($command, ['opt_column']) === null ? 1 : 2) + 1);
50        $identity = Tree::child($command, ['alter_identity_column_option_list']);
51        if ($identity !== null) {
52            return IdentityActions::alter($column, $identity, $scope, $context);
53        }
54        return match ($tail[0] ?? '') {
55            'SET' => self::set($column, $columnNode, $command, $tail, $scope, $context),
56            'DROP' => self::dropProperty($column, $command, $tail),
57            'ADD' => IdentityActions::add($column, $command, $scope, $context),
58            'TYPE' => self::type($column, $command, $scope, $context),
59            'RESET' => new Storage\ResetColumnOptions($column, RelationAlterations::parameterNames($command, $context)),
60            'OPTIONS' => new Storage\SetColumnForeignOptions($column, Collections::nonEmpty(array_map(static fn (Node $change) => WrapperOptions::change($change, $context->tables->identifiers), Tree::outer($command, ['alter_generic_option_elem'])))),
61            default => throw new UnclassifiedSql('Unclassified column action: ' . Tree::text($command)),
62        };

66     * SET selects a default, nullability, expression, statistics, options, storage, compression, or type.
67     * @param list<string> $tail
68     * @throws UnclassifiedSql
69     * @throws InvalidSql

71    public static function set(string $column, Node $columnNode, Node $command, array $tail, Scope $scope, QueryContext $context): RelationAction
72    {
73        $expression = static fn (): \SqlSemantics\Model\Expression => (new ExpressionBinder())->bind(Tree::outer($command, ['a_expr'])[0] ?? throw new UnclassifiedSql('The column action requires its expression.'), $scope);
74        return match ($tail[1] ?? '') {
75            'DEFAULT' => new Column\ColumnDefaultChange($column, $expression()),
76            'NOT' => new Column\SetColumnNullability($column, Nullability::NotNull),
77            'EXPRESSION' => new Column\SetColumnExpression($column, $expression()),
78            'STATISTICS' => self::statistics($column, $columnNode, $command),
79            '(' => new Storage\SetColumnOptions($column, Collections::nonEmpty(StorageParameters::read($command, $scope))),
80            'STORAGE' => new Column\SetColumnStorage($column, Column\ColumnStorageMode::tryFrom(strtoupper($context->tables->identifiers->name($command->tokens()[count($command->tokens()) - 1]))) ?? throw new InvalidSql(InputViolation::ColumnStorage, $command)),

82            'DATA' => self::type($column, $command, $scope, $context),
83            default => throw new UnclassifiedSql('Unclassified column SET action: ' . Tree::text($command)),
84        };

104    /**
105     * A numeric column addresses an index expression by position; DEFAULT restores the system target.
106     * @throws InvalidSql
107     * @throws UnclassifiedSql
108     */
109    public static function statistics(string $column, Node $columnNode, Node $command): Column\SetColumnStatistics
110    {
111        $value = Tree::child($command, ['set_statistics_value']) ?? throw new UnclassifiedSql('SET STATISTICS requires its target.');
112        $text = str_replace([' ', '_'], '', Tree::text($value));
113        if (strtoupper($text) !== 'DEFAULT' && preg_match('/^[+-]?[0-9]{1,9}$/D', $text) !== 1) {
114            throw new InvalidSql(InputViolation::StatisticsTarget, $value);
115        }
116        $position = $columnNode->name === 'Iconst' ? str_replace('_', '', Tree::text($columnNode)) : null;
117        if ($position !== null && (strlen($position) > 9 || (int) $position < 1 || (int) $position > 32767)) {
118            throw new InvalidSql(InputViolation::ColumnPosition, $columnNode);
119        }
120        try {
121            return new Column\SetColumnStatistics($position === null ? $column : (int) $position, strtoupper($text) === 'DEFAULT' ? -1 : (int) $text);
122        } catch (InvalidStructure $error) {
123            throw new InvalidSql(InputViolation::StatisticsTarget, $value, $error);
124        }
125    }
