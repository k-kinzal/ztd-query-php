11/**
12 * Gives unnamed PostgreSQL constraints the names the server chooses when it creates them, so that later DROP
13 * CONSTRAINT statements find them: table_pkey, table_columns_key, table_columns_fkey, table_column_check for a
14 * check on one column and table_check for any other check. A name taken by a constraint of the schema, or for keys
15 * by a relation, is numbered (table_a_check1), and every name is cut to 63 bytes as the server cuts it.
16 *
17 * @visibility SqlSemantics
18 */
19final class PostgreSqlConstraintNames
20{
21    /**
22     * The longest identifier PostgreSQL keeps, in bytes (NAMEDATALEN - 1).
23     */
24    public const MAXIMUM_LENGTH = 63;
25
26    /**
27     * Names the unnamed constraints of every table; a name, once given, stays when the table or its columns are
28     * renamed, as on the server.
29     *
30     * @param list<TableDefinition> $tables
31     * @return list<TableDefinition>
32     */
33    public static function assign(array $tables): array
34    {
35        foreach ($tables as $position => $table) {
36            if (array_filter($table->constraints, static fn (TableConstraint $constraint): bool => $constraint->name === null) === []) {
37                continue;
38            }
39            $tables[$position] = self::table($table, ...self::taken($tables, $table->schema));
40        }
41        return $tables;
42    }

44    /**
45     * Returns the constraint names and the relation names of one namespace, which unnamed constraints avoid.
46     *
47     * @param list<TableDefinition> $tables
48     * @return array{list<string>, list<string>}
49     */
50    public static function taken(array $tables, string $schema): array
51    {
52        $constraints = [];
53        $relations = [];
54        foreach ($tables as $table) {
55            if ($table->schema !== $schema) {
56                continue;
57            }
58            $relations[] = $table->name;
59            foreach ($table->constraints as $constraint) {
60                if ($constraint->name !== null) {
61                    $constraints[] = $constraint->name;
62                }
63            }
64            foreach ($table->indexes as $index) {
65                if ($index->name !== null) {
66                    $relations[] = $index->name;
67                }
68            }
69        }
70        return [$constraints, $relations];
71    }

73    /**
74     * Names the unnamed constraints of one table in the order the server creates them: checks, then the primary
75     * key, then unique keys, then foreign keys, each in declaration order.
76     *
77     * @param list<string> $constraints Constraint names already taken in the namespace
78     * @param list<string> $relations Relation names already taken in the namespace
79     */
80    public static function table(TableDefinition $table, array $constraints, array $relations): TableDefinition
81    {
82        $named = $table->constraints;
83        foreach ([Constraint\Check::class, Constraint\PrimaryKey::class, Constraint\UniqueKey::class, Constraint\ForeignKey::class] as $class) {
84            foreach ($named as $position => $constraint) {
85                if ($constraint->name !== null || $constraint::class !== $class) {
86                    continue;
87                }
88                [$addition, $label] = self::parts($constraint);
89                $taken = $constraint instanceof Constraint\Check || $constraint instanceof Constraint\ForeignKey ? $constraints : [...$constraints, ...$relations];
90                $name = self::choose($table->name, $addition, $label, $taken);
91                $constraints[] = $name;
92                if (!$constraint instanceof Constraint\Check && !$constraint instanceof Constraint\ForeignKey) {
93                    $relations[] = $name;
94                }
95                $named[$position] = self::named($constraint, $name);
96            }
97        }
98        return new TableDefinition($table->schema, $table->name, $table->columns, $named, $table->source, $table->resolved, $table->indexes, $table->properties);
99    }

101    /**
102     * Returns the column part and the label of the name the server derives for a constraint: a check names its
103     * column only when its expression reads exactly one, a key names its key and included columns, a primary key
104     * names none.
105     *
106     * @return array{?string, string}
107     */
108    public static function parts(TableConstraint $constraint): array
109    {
110        if ($constraint instanceof Constraint\Check) {
111            $columns = array_values(array_unique(array_map(static fn (\SqlSemantics\Model\ColumnBinding $binding): string => $binding->column->name, $constraint->predicate->lineage())));
112            return [count($columns) === 1 ? $columns[0] : null, 'check'];
113        }
114        if ($constraint instanceof Constraint\PrimaryKey) {
115            return [null, 'pkey'];
116        }
117        if ($constraint instanceof Constraint\UniqueKey) {
118            return [self::addition(self::indexColumns([...$constraint->localColumns(), ...$constraint->index->include])), 'key'];
119        }
120        return [self::addition($constraint->localColumns()), 'fkey'];
121    }

123    /**
124     * Returns the first of name, name1, name2, ... that no taken name uses.
125     *
126     * @param list<string> $taken
127     */
128    public static function choose(string $table, ?string $addition, string $label, array $taken): string
129    {
130        $pass = 0;
131        do {
132            $name = self::objectName($table, $addition, $pass === 0 ? $label : $label . $pass);
133            $pass++;
134        } while (in_array($name, $taken, true));
135        return $name;
136    }

138    /**
139     * Joins the table name, the column part and the label with underscores, cutting the longer of the first two
140     * until the name fits in 63 bytes, without splitting a character.
141     */
142    public static function objectName(string $table, ?string $addition, string $label): string
143    {
144        $available = self::MAXIMUM_LENGTH - strlen($label) - 1 - ($addition === null ? 0 : 1);
145        $first = strlen($table);
146        $second = $addition === null ? 0 : strlen($addition);
147        while ($first + $second > $available) {
148            if ($first > $second) {
149                $first--;
150            } else {
151                $second--;
152            }
153        }
154        $parts = [mb_strcut($table, 0, $first, 'UTF-8')];
155        if ($addition !== null) {
156            $parts[] = mb_strcut($addition, 0, $second, 'UTF-8');
157        }
158        $parts[] = $label;
159        return implode('_', $parts);
160    }

162    /**
163     * Joins column names with underscores, stopping once the result reaches 64 bytes.
164     *
165     * @param list<string> $columns
166     */
167    public static function addition(array $columns): string
168    {
169        $result = '';
170        foreach ($columns as $column) {
171            $result .= ($result === '' ? '' : '_') . $column;
172            if (strlen($result) > self::MAXIMUM_LENGTH) {
173                break;
174            }
175        }
176        return $result;
177    }

179    /**
180     * Returns the names of the index columns, numbering a name an earlier column already uses (a, a1).
181     *
182     * @param list<string> $columns
183     * @return list<string>
184     */
185    public static function indexColumns(array $columns): array
186    {
187        $result = [];
188        foreach ($columns as $column) {
189            $name = $column;
190            for ($number = 1; in_array($name, $result, true); $number++) {
191                $name = mb_strcut($column, 0, self::MAXIMUM_LENGTH - strlen((string) $number), 'UTF-8') . $number;
192            }
193            $result[] = $name;
194        }
195        return $result;
196    }

198    /**
199     * Returns the constraint with the given name, or unnamed for null, and every other operand unchanged.
200     */
201    public static function named(TableConstraint $constraint, ?string $name): TableConstraint
202    {
203        return match (true) {
204            $constraint instanceof Constraint\Check => new Constraint\Check($constraint->predicate, $constraint->enforced, $constraint->noInherit, $name, $constraint->source, $constraint->onConflict),
205            $constraint instanceof Constraint\PrimaryKey => new Constraint\PrimaryKey($constraint->keys, $constraint->checking, $constraint->nullsDistinct, $name, $constraint->source, $constraint->onConflict, $constraint->index),
206            $constraint instanceof Constraint\UniqueKey => new Constraint\UniqueKey($constraint->keys, $constraint->checking, $constraint->nullsDistinct, $name, $constraint->source, $constraint->onConflict, $constraint->index),
207            $constraint instanceof Constraint\ForeignKey => new Constraint\ForeignKey($constraint->columns, $constraint->referencedTable, $constraint->referencedColumns, $constraint->onDelete, $constraint->onUpdate, $constraint->match, $constraint->checking, $constraint->deleteColumns, $name, $constraint->source, $constraint->indexName),
208            default => $constraint,
209        };
210    }

212    /**
213     * Returns the constraints a table receives from a parent it inherits (its inheritable checks, which keep their
214     * names) or from a LIKE template (its checks with their names and its keys, which the new table names afresh);
215     * foreign keys are never copied.
216     *
217     * @param list<TableConstraint> $constraints
218     * @return list<TableConstraint>
219     */
220    public static function copied(array $constraints, bool $inherited): array
221    {
222        $result = [];
223        foreach ($constraints as $constraint) {
224            if ($constraint instanceof Constraint\Check && !$constraint->noInherit) {
225                $result[] = $constraint;
226            } elseif (!$inherited && ($constraint instanceof Constraint\PrimaryKey || $constraint instanceof Constraint\UniqueKey)) {
227                $result[] = self::named($constraint, null);
228            }
229        }
230        return $result;
231    }
