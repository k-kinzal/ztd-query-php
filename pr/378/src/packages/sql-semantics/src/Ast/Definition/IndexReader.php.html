36        $words = array_map(static fn ($token): string => strtoupper($token->text), $tokens);
37        if (($words[0] ?? '') !== 'CREATE' || !in_array($words[1] ?? '', ['INDEX', 'UNIQUE', 'FULLTEXT', 'SPATIAL'], true)) {
38            return null;
39        }
40        $index = array_search('INDEX', $words, true);
41        $on = array_search('ON', $words, true);
42        if ($index === false || $on === false) {
43            return null;
44        }
45        $header = array_slice($tokens, $index + 1, $on - $index - 1);
46        $names = [];
47        foreach ($header as $token) {
48            if (strtoupper($token->text) === 'USING') {
49                break;
50            }
51            if (!in_array(strtoupper($token->text), ['CONCURRENTLY', 'IF', 'NOT', 'EXISTS', '.'], true)) {
52                $names[] = $this->identifiers->name($token);
53            }
54        }
55        $single = Tree::child($node, $this->identifiers->dialect === \SqlSemantics\Dialect::MySql ? ['ident'] : ['name']);
56        if ($single !== null && Tree::hasTokens($single)) {
57            $names = [$this->identifiers->name($single->tokens()[0])];
58        }
59        $table = $this->target($node, array_slice($tokens, $on + 1));
60        $schema = count($names) > 1 ? $names[0] : (count($table) > 1 ? $table[0] : $this->defaultSchema);
61        return $this->definition($node, $schema, $names === [] ? null : $names[count($names) - 1], count($table) === 1 ? ($schema === '' ? $table : [$schema, $table[0]]) : $table, IndexKeys::read($node, $this->identifiers));
