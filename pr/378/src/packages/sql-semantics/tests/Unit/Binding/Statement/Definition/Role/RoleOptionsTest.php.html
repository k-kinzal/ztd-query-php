38    public function testDefinitionReadsEveryOptionInSourceOrder(): void
39    {
40        $statement = (new Binder((new SchemaBuilder(Dialect::PostgreSql))->build()))->bind("CREATE ROLE r WITH LOGIN SYSID 2 PASSWORD 'x' IN GROUP a, CURRENT_USER CONNECTION LIMIT -1 ROLE b ADMIN d VALID UNTIL 'infinity' NOINHERIT NOSUPERUSER CREATEDB NOCREATEROLE REPLICATION NOBYPASSRLS");
41        self::assertInstanceOf(CreateRoleStatement::class, $statement);
42        self::assertCount(14, $statement->options);
43        self::assertEquals(new RoleAttribute(RoleCapability::Login, true), $statement->options[0]);
44        self::assertEquals(new RoleSystemId(2), $statement->options[1]);
45        $password = $statement->options[2];
46        self::assertInstanceOf(RolePassword::class, $password);
47        self::assertSame("'x'", $password->secret->text);
48        self::assertEquals(new RoleMemberships([new NamedRole('a'), SessionRole::CurrentUser]), $statement->options[3]);
49        self::assertEquals(new ConnectionLimit(-1), $statement->options[4]);
50        self::assertEquals(new RoleMembers([new NamedRole('b')]), $statement->options[5]);
51        self::assertEquals(new RoleAdmins([new NamedRole('d')]), $statement->options[6]);
52        $validity = $statement->options[7];
53        self::assertInstanceOf(RoleValidity::class, $validity);
54        self::assertSame("'infinity'", $validity->until->text);
55        self::assertEquals([new RoleAttribute(RoleCapability::Inherit, false), new RoleAttribute(RoleCapability::Superuser, false), new RoleAttribute(RoleCapability::CreateDb, true), new RoleAttribute(RoleCapability::CreateRole, false), new RoleAttribute(RoleCapability::Replication, true), new RoleAttribute(RoleCapability::BypassRls, false)], array_slice($statement->options, 8));
56    }
