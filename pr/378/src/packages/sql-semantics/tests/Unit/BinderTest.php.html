82    public function testBindSelfJoinPreservesOccurrenceIdentityAndNullProvenance(Dialect $dialect): void
83    {
84        $schema = (new SchemaBuilder($dialect))->build('CREATE TABLE users (id INTEGER PRIMARY KEY, parent_id INTEGER, score INTEGER NOT NULL)');
85        $statement = (new Binder($schema))->bind('SELECT child.id, parent.score AS parent_score, COALESCE(parent.score, 0) AS effective_score FROM users AS child LEFT JOIN users AS parent ON child.parent_id = parent.id');
86        self::assertSame('s0', $statement->scopeId);
87        self::assertSame(['r0', 'r1'], array_column($statement->relations, 'id'));
88        self::assertSame($statement->relations[0]->declaration, $statement->relations[1]->declaration);
89        self::assertSame(['id', 'parent_score', 'effective_score'], array_column($statement->outputs, 'name'));
90        self::assertSame(Nullability::NotNull, $statement->outputs[0]->expression->nullability);
91        self::assertSame(Nullability::MaybeNull, $statement->outputs[1]->expression->nullability);
92        self::assertSame(['j0'], $statement->outputs[1]->expression->nullExtendedBy);
93        self::assertSame('integer', $statement->outputs[1]->expression->type->name);
94        self::assertSame(Nullability::NotNull, $statement->outputs[2]->expression->nullability);
95        self::assertSame([], $statement->outputs[2]->expression->nullExtendedBy);
96        self::assertSame(['j0'], $statement->outputs[2]->expression->operands[0]->nullExtendedBy);
97        self::assertSame('r1', $statement->outputs[2]->expression->lineage()[0]->relationId);
98        self::assertSame(Nullability::NotNull, $statement->relations[1]->declaration->columns[2]->nullability);
99    }
