55    public function testTargetWritesEachClassWithItsKeywordAndStaysFixedAcrossBinding(string $sql, string $expected): void
56    {
57        $binder = new Binder((new SchemaBuilder(Dialect::PostgreSql))->build('CREATE TABLE t(id INTEGER, a INTEGER)'));
58        $statement = $binder->bind($sql);
59        self::assertInstanceOf(GrantPrivilegesStatement::class, $statement);
60        self::assertSame($expected, PrivilegeTargets::target($statement->target)->toString());
61        self::assertStringContainsString(' ON ' . $expected . ' TO ', (new \SqlSemantics\SimpleSerializer())->serialize($statement));
62        $rebound = $binder->bind((new \SqlSemantics\SimpleSerializer())->serialize($statement));
63        self::assertInstanceOf(GrantPrivilegesStatement::class, $rebound);
64        self::assertSame((new \SqlSemantics\SimpleSerializer())->serialize($statement), (new \SqlSemantics\SimpleSerializer())->serialize($rebound));
65        self::assertSame($expected, PrivilegeTargets::target($rebound->target)->toString());
66    }
