20/**
21 * One statement, with everything the analysis established about it.
22 *
23 * The statement comes first and largest, laid out a clause per line, because
24 * it is what the reader came for. Around it are the facts that decide what
25 * to do about it: where it is issued and through what, what it names, what
26 * is bound to it, and what the analyzer could not settle. The other
27 * statements of the same function and on the same table are listed last,
28 * so the reading can continue without going back.
29 *
30 * @visibility root
31 */
32final class StatementPage
33{

72     * The body of one statement's page.
73     */
74    public function render(ReportSite $site, CatalogEntry $entry): string
75    {
76        $page = $site->statementPage($entry->id);
77
78        return '<h1>' . $this->text->chip(strtoupper($entry->kind->value), $this->palette->kind($entry->kind->value))
79            . '<span>' . $this->text->escape($this->title($entry)) . '</span></h1>'
80            . '<p class="lede">' . $this->where($site, $entry) . '</p>'
81            . $this->body($entry)
82            . $this->caveats($entry)
83            . (new SourceCode($this->text))->excerpt($site, $entry)
84            . ($entry->placeholders === []
85                ? $this->facts($site, $entry)
86                : '<div class="split">' . $this->facts($site, $entry) . $this->values($entry) . '</div>')
87            . $this->findings($entry)
88            . $this->related($site, $page, $entry);
89    }

91    /**
92     * What the statement is, in a few words.
93     */
94    public function title(CatalogEntry $entry): string
95    {
96        if ($entry->resolution() === Resolution::NotAnalyzed) {
97            return 'a call nothing was read from';
98        }
99        if ($entry->tables === []) {
100            return 'no table named';
101        }
102        $labels = [];
103        foreach ($entry->tables as $table) {
104            $labels[] = (new TableName($table))->label();
105        }
106
107        return 'on ' . implode(', ', $labels);
108    }

110    /**
111     * Where the statement is issued, as a sentence whose every place is a link.
112     */
113    public function where(ReportSite $site, CatalogEntry $entry): string
114    {
115        $scope = Scope::of($entry->site->function);
116        $sentence = 'Issued at ' . $this->text->link($entry->site->display(), '../' . $site->filePage($entry->site->file), 'mono');
117        if (!$scope->isMain()) {
118            $sentence .= ' in ' . $this->text->link($scope->function(), '../' . $site->functionUrl($entry), 'mono');
119        }
120        $sentence .= ' through ' . $this->text->chip($entry->site->sink, 'chip-sm chip-ghost', 'The database call that was matched');
121        if (count($entry->through) > 1) {
122            $sentence .= ', reached by way of <code>' . $this->text->escape(implode(' → ', $entry->through)) . '</code>';
123        }
124
125        return $sentence . '.';
126    }

128    /**
129     * The statement itself, laid out for reading, or the call it was not read from.
130     *
131     * The copy button takes what the block shows, so what is copied is what
132     * was read: the statement a clause per line, with every gap as `{$}`.
132     * was read: expanded SQL, with named gaps where PHP variables are known.
133     */
134    public function body(CatalogEntry $entry): string
135    {
136        if ($entry->resolution() === Resolution::NotAnalyzed) {
137            $written = $entry->firstGap()?->expression;
138
139            return '<pre class="code code-lead"><span class="tok-com">-- no statement was read from this call</span>'
140                . ($written === null ? '' : "\n" . $this->text->escape($written)) . '</pre>';
141        }
142
143        return '<div class="code-block">'
144            . '<button type="button" class="btn copy" data-dd-copy title="Copy the statement">Copy</button>'
145            . '<pre class="code code-lead">' . $this->sql->render($this->formatter->format($entry->parts())) . '</pre>'
146            . '<details class="as-written"><summary>As written in the source</summary><pre class="code">'
147            . $this->sql->render($entry->parts()) . '</pre></details></div>';
148    }

150    /**
151     * What a reader has to know before trusting the statement.
152     */
153    public function caveats(CatalogEntry $entry): string
154    {
155        $caveats = [];
156        if ($entry->resolution() === Resolution::NotAnalyzed) {
157            $caveats[] = $entry->resolution()->describe();
158        } elseif (!$entry->resolution()->isClosed()) {
159            $caveats[] = $entry->resolution()->describe() . ' The statements listed for this call may not be all of them.';
160        } elseif ($entry->truncated) {
161            $caveats[] = 'A limit on loop passes or on callers cut the search short. The statements listed for this call may not be all of them.';
162        }
163        if (!$entry->correlated) {
164            $caveats[] = 'Assembled from parts that vary independently, so some of the alternatives at this call may be unreachable.';
165        }
166        if ($caveats === []) {
167            return '';
168        }
169        $items = '';
170        foreach ($caveats as $caveat) {
171            $items .= '<li>' . $this->text->escape($caveat) . '</li>';
172        }
173
174        return '<div class="notice tone-warn"><ul>' . $items . '</ul></div>';
175    }

177    /**
178     * The facts about the reading, as a list of labelled values.
179     */
180    public function facts(ReportSite $site, CatalogEntry $entry): string
181    {
182        $resolution = $entry->resolution();
183        $tables = '';
184        foreach ($entry->tables as $table) {
185            $tables .= $this->text->chipLink((new TableName($table))->label(), '../' . $site->tablePage($table), 'chip-ghost') . ' ';
186        }
187        $rows = [
188            ['Resolution', $this->text->chip($resolution->value, $this->palette->resolution($resolution)) . ' <span class="muted">' . $this->text->escape($resolution->describe()) . '</span>'],
189            ['Search', $entry->searchClosed()
190                ? '<span class="muted">closed: every dependency was followed to its end</span>'
191                : '<span class="muted">left open: candidates or dependencies remain unknown</span>'],
192            ['Reachability', '<span class="muted">not assessed: both branches are retained, including constant conditions</span>'],
193            ['Tables', $tables === '' ? '<span class="none">none named</span>' : $tables],
194            ['Kind', $this->text->chip(strtoupper($entry->kind->value), $this->palette->kind($entry->kind->value))],
195            ['Identifier', '<code>' . $this->text->escape($entry->id) . '</code> <span class="muted">stable across runs while the statement is unchanged</span>'],
196        ];
197        $written = '';
198        foreach ($rows as [$label, $value]) {
199            $written .= '<div><dt>' . $this->text->escape($label) . '</dt><dd>' . $value . '</dd></div>';
200        }
201
202        return '<section><h2 id="facts">About this statement</h2><dl class="facts">' . $written . '</dl></section>';
203    }

223    /**
224     * One bind parameter as a row.

226    public function valueRow(Placeholder $placeholder): string
227    {
228        $value = $placeholder->value;
229        $bound = $value === null
230            ? '<span class="none">unbound</span>'
231            : ($value->isResolved()
232                ? '<code>' . $this->text->escape($value->display()) . '</code>'
233                : '<span class="muted">' . $this->text->escape($this->openValue($value->origins)) . '</span>');
234
235        return '<tr><td class="tight"><code>' . $this->text->escape($placeholder->token) . '</code></td>'
236            . '<td class="tight"><code>' . $this->text->escape($value === null ? '?' : $value->type) . '</code></td>'
237            . '<td>' . $bound . '</td></tr>';
238    }

267    /**
268     * What the page is best left from: the places the statement belongs to, and the rest of its function.
269     *
270     * @return list<array{string, list<array{string, string, int|null, bool}>, string|null}>
271     */
272    public function context(ReportSite $site, CatalogEntry $entry): array
273    {
274        $index = $site->index();
275        $scope = Scope::of($entry->site->function);
276        $places = [];
277        if ($site->catalog()->source($entry->site->file) !== null) {
278            $places[] = ['Source code', '#source', null, false];
279        }
280        foreach ($entry->tables as $table) {
281            $places[] = ['Table ' . (new TableName($table))->label(), $site->tablePage($table), count($index->byTable()[$table] ?? []), false];
282        }
283        if ($scope->class !== null) {
284            $places[] = ['Class ' . ($scope->classShort() ?? $scope->class), $site->classPage($scope->class), count($index->byClass()[$scope->class] ?? []), false];
285        }
286        if (!$scope->isMain()) {
287            $places[] = [$scope->display() . '()', $site->functionUrl($entry), count($index->byFunction()[$entry->site->function] ?? []), false];
288        }
289        $places[] = ['File ' . $entry->site->file, $site->filePage($entry->site->file), count($index->byFile()[$entry->site->file] ?? []), false];
290        $siblings = [];
291        foreach ($index->byFunction()[$entry->site->function] ?? [] as $other) {
292            $siblings[] = [$this->text->truncate($other->sql(), self::LABEL), $site->statementPage($other->id), null, $other->id === $entry->id];
292            $siblings[] = [$this->text->truncate($this->sql->plain($other->parts()), self::LABEL), $site->statementPage($other->id), null, $other->id === $entry->id];
293        }
294
295        return [['Belongs to', $places, null], ['Statements of ' . $scope->display(), $siblings, $site->functionUrl($entry)]];
296    }

298    /**
299     * The other statements of the same function, and on the same tables.
300     */
301    public function related(ReportSite $site, string $page, CatalogEntry $entry): string
302    {
303        $index = $site->index();
304        $scope = Scope::of($entry->site->function);
305        $written = '';
306        $siblings = array_values(array_filter($index->byFunction()[$entry->site->function] ?? [], static fn (CatalogEntry $other): bool => $other->id !== $entry->id));
307        if ($siblings !== [] && !$scope->isMain()) {
308            $written .= '<section><h2 id="same-function">Also issued by <code>' . $this->text->escape($scope->display()) . '</code>'
309                . $this->text->count(count($siblings)) . '</h2>'
310                . $this->list->rows($site, $page, array_slice($siblings, 0, self::RELATED), ['function'])
311                . (count($siblings) > self::RELATED ? '<p class="actions">' . $this->text->link('Every statement of this function', '../' . $site->functionUrl($entry)) . '</p>' : '')
312                . '</section>';
313        }
314        foreach (array_slice($entry->tables, 0, 2) as $table) {
315            $others = array_values(array_filter($index->byTable()[$table] ?? [], static fn (CatalogEntry $other): bool => $other->id !== $entry->id));
316            if ($others === []) {
317                continue;
318            }
319            $written .= '<section><h2 id="' . $this->text->escape('same-table-' . $this->text->slug($table)) . '">Also on '
320                . $this->text->chipLink((new TableName($table))->label(), '../' . $site->tablePage($table), 'chip-ghost') . $this->text->count(count($others)) . '</h2>'
321                . $this->list->rows($site, $page, array_slice($others, 0, self::RELATED))
322                . (count($others) > self::RELATED ? '<p class="actions">' . $this->text->link('Every statement on this table', '../' . $site->tablePage($table)) . '</p>' : '')
323                . '</section>';
324        }
325
326        return $written;
327    }
