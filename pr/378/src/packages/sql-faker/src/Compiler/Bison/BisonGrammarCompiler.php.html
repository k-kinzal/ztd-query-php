23/**
24 * Turns a Bison grammar file into the grammar the generators walk.
25 *
26 * The file is read with bison-parser; this class keeps what generation
27 * needs: every alternative of every rule as a sequence of terminals and
28 * nonterminals. A name is a nonterminal when a rule defines it and a
29 * terminal when `%token` declares it; a character literal is a terminal
30 * named by its character; actions, precedence marks, tags, aliases and
31 * string literals are left out. Alternatives of a rule defined in several
32 * places are joined in file order.
33 *
34 * @visibility public
35 *
36 * @example Compiling a small grammar
37 *     $grammar = (new \SqlFaker\Compiler\Bison\BisonGrammarCompiler())->compile("%token NUM\n%start expr\n%%\nexpr: expr '+' term { \$\$ = \$1 + \$3; } | term ;\nterm: NUM ;\n");
38 *     [$grammar->startSymbol, array_keys($grammar->ruleMap), count($grammar->ruleMap['expr']->alternatives)] // => ['expr', ['expr', 'term'], 2]
39 */
40final class BisonGrammarCompiler

49    /**
50     * Compiles a grammar file.
51     *
52     * @param string $source The text of the `.y` or `.yy` file
53     *
54     * @return Grammar The grammar
55     *
56     * @throws SyntaxException When the file is not a grammar Bison accepts, which includes a file without rules
57     * @throws UnknownSymbolException When a rule uses a name that is neither a rule nor a declared token
58     */
59    public function compile(string $source): Grammar
60    {
61        $file = $this->parser->parse($source);
62        $rules = $file->rules();
63        $defined = [];
64        foreach ($rules as $rule) {
65            $defined[$rule->name->value] = true;
66        }
67        $declared = $this->declaredTokens($file);
68        $ruleMap = [];
69        foreach ($rules as $rule) {
70            $productions = [];
71            foreach ($rule->alternatives as $alternative) {
72                $productions[] = new Production($this->symbols($alternative, $defined, $declared));
73            }
74            $name = $rule->name->value;
75            $previous = isset($ruleMap[$name]) ? $ruleMap[$name]->alternatives : [];
76            $ruleMap[$name] = new ProductionRule($name, array_merge($previous, $productions));
77        }
78
79        return new Grammar($this->startSymbol($file), $ruleMap);
80    }
81
82    /**
83     * Collects the names `%token` declares.
84     *
85     * @param GrammarFile $file The tree
86     *
87     * @return array<string, true> The names
88     */
89    public function declaredTokens(GrammarFile $file): array
90    {
91        $declared = [];
92        foreach ($file->allDeclarations() as $declaration) {
93            if (!$declaration instanceof SymbolDeclaration || $declaration->class !== SymbolClass::Token) {
94                continue;
95            }
96            foreach ($declaration->entries as $entry) {
97                if ($entry->symbol->isIdentifier()) {
98                    $declared[$entry->symbol->value] = true;
99                }
100            }
101        }
102
103        return $declared;
104    }

106    /**
107     * Turns the symbols of an alternative into terminals and nonterminals.
108     *
109     * @param Alternative $alternative The alternative
110     * @param array<string, true> $defined Names rules define
111     * @param array<string, true> $declared Names `%token` declares
112     *
113     * @return list<Symbol> The symbols in order, string literals left out
114     *
115     * @throws UnknownSymbolException When a name is neither defined nor declared
116     */
117    public function symbols(Alternative $alternative, array $defined, array $declared): array
118    {
119        $symbols = [];
120        foreach ($alternative->symbols() as $symbol) {
121            if ($symbol->kind === SymbolKind::CharLiteral) {
122                $symbols[] = new Terminal($symbol->value);
123            } elseif ($symbol->kind === SymbolKind::String) {
124                continue;
125            } elseif (isset($defined[$symbol->value])) {
126                $symbols[] = new NonTerminal($symbol->value);
127            } elseif (isset($declared[$symbol->value])) {
128                $symbols[] = new Terminal($symbol->value);
129            } else {
130                throw new UnknownSymbolException($symbol->value);
131            }
132        }
133
134        return $symbols;
135    }

137    /**
138     * Finds the start symbol: what `%start` names, or else the first rule.
139     *
140     * @param GrammarFile $file The tree, with at least one rule
141     *
142     * @return string The start symbol
143     */
144    public function startSymbol(GrammarFile $file): string
145    {
146        foreach ($file->allDeclarations() as $declaration) {
147            if ($declaration instanceof Start) {
148                return $declaration->symbols[0]->value;
149            }
150        }
151
152        return $file->rules()[0]->name->value;
153    }
