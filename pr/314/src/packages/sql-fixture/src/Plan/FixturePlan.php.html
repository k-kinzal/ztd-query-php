9/**
10 * The parsed form of a fixture plan: which tables take part, and how their
11 * columns line up.
12 *
13 * This type is the plan. The DBML relation string is one way to write it down
14 * and `from()` reads it; `toString()` writes it back. Building it up method by
15 * method, or declaring one as a class, produce the same value by other routes:
16 *
17 *     FixturePlan::from('order.id < order_detail.order_id');
18 *
19 *     (new FixturePlan())->withOneToMany('order.id', 'order_detail.order_id');
20 *
21 *     final class OrderWithDetails extends FixturePlan
22 *     {
23 *         public function __construct()
24 *         {
25 *             parent::__construct(
26 *                 Relation::oneToMany('order.id', 'order_detail.order_id'),
27 *                 Relation::manyToOne('order.customer_id', 'customer.id'),
28 *             );
29 *         }
30 *     }
31 *
32 * A subclass names a plan without adding a kind of plan, so the builders below
33 * return a plain FixturePlan rather than the subclass: once a declared plan is
34 * altered it is no longer the plan that class stands for.
35 *
36 * @visibility public
37 * @example Build and print a relational fixture plan
38 *     $plan = \SqlFixture\Plan\FixturePlan::from('users.id < posts.user_id');

215     * whether the two relations agree or not.
216     *
217     * @param list<Relation> $relations
218     */
219    private function rejectColumnsBoundTwice(array $relations): void
220    {
221        $seen = [];
222
223        foreach ($relations as $relation) {
224            $child = $relation->child();
225            $key = $child->toString();
226
227            if (isset($seen[$key])) {
228                throw PlanStructureException::columnsBoundTwice($child, $seen[$key], $relation->parent());
229            }
230
231            $seen[$key] = $relation->parent();
232        }

224    {

274                if ($this->waitsForAny($table, $relations, $pending)) {
275                    $waiting[] = $table;
276                    continue;
277                }

280            }
281
282            if ($ready === []) {

284            }
285
286            $ordered = [...$ordered, ...$ready];
287            $pending = $waiting;
288        }
