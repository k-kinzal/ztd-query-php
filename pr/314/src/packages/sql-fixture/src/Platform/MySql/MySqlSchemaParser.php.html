42    /**
43     * Stop if the parser abandoned part of the definition list.
44     *
45     * On a syntax error the parser gives up on the rest of the definitions,
46     * so a stray keyword takes every column after it. The result is a schema
47     * that looks complete and is not, and the damage only shows up much later
48     * as a column that generates no value.
49     *
50     * Not every error loses something. DEC and FIXED are valid synonyms the
51     * parser reports as unrecognised while still reading the column, so the
52     * test is what came out rather than whether anything was said.
53     */
54    private function assertNothingWasLost(Parser $parser, CreateStatement $stmt, string $sql): void
55    {
56        if ($parser->errors === []) {
57            return;
58        }
59
60        $fields = $stmt->fields;
61        $declared = $this->countDeclaredDefinitions($parser);
62
63        if ($declared !== null && is_array($fields) && count($fields) >= $declared) {
64            return;
65        }
66
67        throw SchemaParseException::invalidSql($sql, $parser->errors[0]->getMessage());
68    }

70    /**
71     * How many definitions the parenthesised body separates, or null when it
72     * is never closed.
73     *
74     * Counting uses the parser's own tokens, which already know that a comma
75     * inside a string or an ENUM is not a separator.
76     */
77    private function countDeclaredDefinitions(Parser $parser): ?int
78    {
79        $list = $parser->list;
80        if ($list === null) {
81            return null;
82        }
83
84        $depth = 0;
85        $definitions = 1;
86
87        foreach ($list->tokens as $token) {
88            if ($token->type !== Token::TYPE_OPERATOR) {
89                continue;
90            }
91
92            if ($token->value === '(') {
93                $depth++;
94                continue;
95            }
96
97            if ($token->value === ')') {
98                $depth--;
99
100                if ($depth === 0) {
101                    return $definitions;
102                }
103
104                continue;
105            }
106
107            if ($depth === 1 && $token->value === ',') {
108                $definitions++;
109            }
110        }
111
112        return null;
113    }

287    {
288        if ($options === null) {

292        if ($options->has('DEFAULT') === false) {
293            return null;
294        }
295
296        $optionsArray = $options->options;
297
298        foreach ($optionsArray as $option) {
299            if (!is_array($option)) {
300                continue;
301            }
302
303            $name = $option['name'] ?? null;
304            if ($name === 'DEFAULT') {
305                $value = $option['value'] ?? null;
306                if ($value === null) {
307                    return null;
308                }
309                if (is_string($value)) {
310                    if (preg_match('/^[\'"](.*)[\'"]\s*$/s', $value, $matches) === 1) {
311                        return $matches[1];
312                    }
313                    if (strtoupper($value) === 'NULL') {
314                        return null;
315                    }
316                    if (strtoupper($value) === 'TRUE') {
317                        return true;
318                    }
319                    if (strtoupper($value) === 'FALSE') {
320                        return false;
321                    }
322                    if (is_numeric($value)) {
323                        if (str_contains($value, '.')) {
324                            return (float) $value;
325                        }
326                        return (int) $value;
327                    }
328                    return $value;
329                }
