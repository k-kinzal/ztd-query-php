12/**
13 * Generates the rows a plan describes, keeping related rows consistent.
14 *
15 * The walk starts at the table the plan is about and follows relations
16 * outwards, once per connected group of tables. Starting anywhere else changes
17 * the answer: a plan written `order.id < order_detail.order_id,
18 * order.customer_id > customer.id` has customer as its only dependency-free
19 * table, and beginning there would make order a collection hanging off it
20 * rather than the one row the plan is about.
21 *
22 * Direction decides count. Walking to a child asks how many, walking to a
23 * parent asks for the one row being referenced. Both carry the collection-ness
24 * of where they came from, so a table reached through a list is a list however
25 * many rows it happens to have.
26 */
27final class PlanGenerator

43    public function generate(FixturePlan $plan, array $overrides = []): FixtureSet
44    {
45        (new PlanSchemaValidator($this->schemas))->validate($plan);
46
48        $run = new GenerationRun($this->specs($overrides));
47        $run = new GenerationRun((new Generation\OverrideSpecs())->specs($overrides));

56            $this->materialize($plan, $table, [], $run->specFor($table)->count ?? 1, false, null, $run);

56        }
57

59    }

163     * Generate the one row this one references and read the linking columns

174        Relation $relation,
175        array $overrides,
176        bool $isList,
177        GenerationRun $run,
178    ): array {
179        if ($this->isAlreadyLinked($relation, $overrides)) {
180            return [];
181        }
182
183        if ($relation->parentIsOptional() && !$run->wasAskedFor($relation->parent()->table)) {
184            return [];
185        }
186
187        $parent = $this->materializeParent($plan, $relation->parent()->table, $isList, $relation, $run);
188
189        return $this->project($parent, $relation);

233    private function project(array $parentRow, Relation $relation): array
234    {
235        $values = [];
236
237        foreach ($relation->columnMap() as $childColumn => $parentColumn) {
238            if (!array_key_exists($parentColumn, $parentRow)) {
239                throw PlanSchemaException::missingValue($childColumn, $relation->parent(), $parentColumn);
240            }

242            $values[$childColumn] = $parentRow[$parentColumn];
243        }

260    /**
261     * Every table joined to this one by relations, however the arrows point.
262     *
263     * @return list<string>
264     */
265    private function connectedTo(FixturePlan $plan, string $table): array
266    {
267        $found = [$table];
268
269        for ($index = 0; $index < count($found); $index++) {
270            foreach ($plan->relations as $relation) {
271                $ends = [$relation->left->table, $relation->right->table];
272
273                foreach ([$ends, array_reverse($ends)] as [$from, $to]) {
274                    if ($from === $found[$index] && !in_array($to, $found, true)) {
275                        $found[] = $to;
276                    }
277                }
278            }
279        }
280
281        return $found;
282    }
