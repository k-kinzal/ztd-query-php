30/**
31 * Allocates each resolved column in its own table occurrence while retaining the parsed association.
32 *
33 * @visibility SqlSemantics\Platform\MySql
34 */
35final class JoinBinding
36{
37    private readonly DerivedBinding|SourceBinding|self $left;
38    private readonly DerivedBinding|SourceBinding|self $right;
39
40    /**
41     * @var non-empty-list<DerivedBinding|SourceBinding>
42     */
43    private readonly array $inputs;
44
45    /**
46     * Binds both operands before any slot is exported.
47     */
48    public function __construct(private readonly BoundStatement $operation, private readonly ParsedJoin $join)
49    {
50        $this->left = self::bindChild($operation, $join->left);
51        $this->right = self::bindChild($operation, $join->right);
52        $left = $this->left instanceof self ? $this->left->inputs : [$this->left];
53        $right = $this->right instanceof self ? $this->right->inputs : [$this->right];
54        $this->inputs = array_merge($left, $right);
55        Check::input(count($this->inputs) <= 61, 'A joined source can have at most 61 input occurrences.');
56    }
57
58    /**
59     * Opens an ordinary source or a joined source, retaining transparent parentheses.
60     */
61    public static function bind(BoundStatement $operation, ?Relation $source): SourceBinding|self
62    {
63        $source = self::ungroup($source);
64        if (!$source instanceof TableList) {
65            return $source instanceof ParsedJoin ? new self($operation, $source) : new SourceBinding($operation, $source);
66        }
67
68        $members = $source->members;
69        Check::input(count($members) <= 61, 'A FROM list can have at most 61 input occurrences.');
70        $input = $members[0];
71        foreach (array_slice($members, 1) as $member) {
72            $input = new ParsedJoin($input, JoinOperator::Cross, $member);
73        }
74        Check::invariant($input instanceof ParsedJoin, 'A FROM list contains at least two inputs.');
75
76        return new self($operation, $input);
77    }
78
79    /**
80     * Opens a joined leaf or subtree in the join's occurrence order.
81     */
82    public static function bindChild(BoundStatement $operation, Relation $source): DerivedBinding|SourceBinding|self
83    {
84        $input = self::ungroup($source);
85
86        return $input instanceof DerivedTable ? new DerivedBinding($operation, $input) : self::bind($operation, $input);
87    }

90     * Resolves transparent relation parentheses without collapsing the join.
91     */
92    public static function ungroup(?Relation $source): ?Relation
93    {
94        while ($source instanceof NestedRelation) {
95            $source = $source->relation;
96        }
97
98        return $source;
99    }

113    /**
114     * Resolves a signed read in the exact owning join input.
115     * @throws InvalidConstruction When the field is outside the signed input domain
116     */
117    public function membershipReference(Scalar $scalar): ColumnReference|IntegralColumnReference
118    {
119        $resolution = $this->operation->facts->scalar($scalar)->resolution;
120        Check::input($resolution instanceof ResolvedColumn && $resolution->depth === 0, 'A membership column must resolve within this join scope.');
121        foreach ($this->inputs as $occurrence => $binding) {
122            if ($binding->relation() === $resolution->relation) {
123                $field = $binding instanceof SourceBinding ? $binding->membershipSlot($resolution) : $binding->resolved($resolution);
124                return match (true) {
125                    $field instanceof ColumnReference => new ColumnReference($field->slot, $occurrence),
126                    $field instanceof IntegralColumnReference => new IntegralColumnReference($field->slot, $occurrence),
127                    $field instanceof TextColumnReference => throw new InvalidConstruction('A membership column requires signed integral storage.'),
128                };
129            }
130        }
131
132        throw new InvalidConstruction('A membership column must resolve within this join scope.');
133    }

159    /**
160     * Resolves the source occurrence and native slot used in a projection.
161     * @throws InvalidConstruction When the field is outside the checked input scope
162     */
163    public function projected(ResolvedColumn $resolution): ColumnReference|TextColumnReference
164    {
165        Check::input($resolution->depth === 0, 'A column must resolve within this join scope.');
166        foreach ($this->inputs as $occurrence => $binding) {
167            if ($binding->relation() === $resolution->relation) {
168                $field = $binding instanceof SourceBinding ? $binding->projected($resolution) : $binding->resolved($resolution);
169
170                return $field instanceof TextColumnReference ? new TextColumnReference($field->slot, $occurrence)
171                    : new ColumnReference($field->slot, $occurrence);
172            }
173        }
174
175        throw new InvalidConstruction('A column must resolve within this join scope.');
176    }

178    /**
179     * Closes ON over this subtree before exporting its table occurrences.
180     * @throws InvalidConstruction when the join has no matching IR operator
181     */
182    public function source(): Join
183    {
184        $on = $this->predicate();
185        $kind = match ($this->join->operator) {
186            JoinOperator::Join, JoinOperator::Cross, JoinOperator::Inner, JoinOperator::Natural, JoinOperator::NaturalInner
187                => $on === null ? JoinKind::Cross : JoinKind::Inner,
188            JoinOperator::Left, JoinOperator::LeftOuter, JoinOperator::NaturalLeft, JoinOperator::NaturalLeftOuter => JoinKind::Left,
189            JoinOperator::Right, JoinOperator::RightOuter, JoinOperator::NaturalRight, JoinOperator::NaturalRightOuter => JoinKind::Right,
190            JoinOperator::StraightJoin => throw new InvalidConstruction('The join operator has no complete MySQL representation.'),
191        };
192        $left = $this->left->source();
193        $right = $this->right->source();
194        Check::invariant($left !== null && $right !== null, 'Both checked join sides have sources.');
195
196        return new Join($kind, $left, $right, $on);
197    }

199    /**
200     * Resolves the explicit or merged condition of this join.
201     */
202    public function predicate(): BooleanLiteral|IntegerComparison|IntegerMembership|IntegerRange|TextComparison|LogicalPredicate|NullLiteral|NullTest|null
203    {
204        if (!$this->join->operator->natural() && $this->join->using === []) {
205            return (new PredicateLowering())->lower($this->join->on, $this);
206        }
207        Check::input($this->left instanceof SourceBinding && $this->right instanceof SourceBinding, 'A merged join requires two ordinary base-table inputs.');
208
209        return (new MergedJoinLowering())->predicate($this->join, $this->left, $this->right);
210    }
