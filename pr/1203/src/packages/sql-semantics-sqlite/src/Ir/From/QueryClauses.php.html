49/**
50 * Converts supported clauses into values with their local references already bound.
51 *
52 * @visibility SqlSemantics\Platform\Sqlite
53 */
54final class QueryClauses

56    /**
57     * Maps each join, collecting its references before exporting the final input slots.
58     *
59     * @return list<Join>
60     */
61    public function joins(?Relation $source, SourceBinding $binding): array
62    {
63        if (!$source instanceof JoinChain) {
64            return [];
65        }
66        $predicates = [];
67        foreach ($source->steps as $step) {
68            Check::input($step->constraint === null || $step->constraint instanceof JoinOn, 'The IR join supports an optional ON predicate.');
69            $predicates[] = $this->predicate($step->constraint?->condition, $binding);
70        }
71        $joins = [];
72        foreach ($source->steps as $index => $step) {
73            $right = $binding->input($index + 1);
74            Check::invariant($right instanceof TableSource || $right instanceof LocalSource, 'Every join step has a bound input occurrence.');
75            $joins[] = new Join($this->kind($step->operator), $right, $predicates[$index]);
76        }
77
78        return $joins;
79    }
80
81    /**
82     * Resolves a join operation without emitting joins that implicitly merge columns.
83     *
84     * @throws InvalidConstruction When the operation is outside the supported join domain
85     */
86    public function kind(JoinOperator $operator): JoinKind
87    {
88        Check::input($operator->valid() && !$operator->natural(), 'The IR join requires an explicit input without NATURAL.');
89        $left = $operator->left();
90        $right = $operator->right();
91
92        return match (true) {
93            $left && $right => JoinKind::Full,
94            $left => JoinKind::Left,
95            $right => JoinKind::Right,
96            in_array(JoinKeyword::Cross, $operator->words, true) => JoinKind::Cross,
97            $operator->words === [] || in_array(JoinKeyword::Inner, $operator->words, true) => JoinKind::Inner,
98            default => throw new InvalidConstruction('The join operator has no complete IR representation.'),
99        };
100    }

102    /**
103     * Converts a filter or ON expression into the closed predicate domain.

106    {
107        return $scalar === null ? null : (new ExpressionLowering())->lower($scalar, $binding);
108    }

110    /**
111     * Binds ordering terms by ordinal and fixes their effective direction and NULL placement.
112     *
113     * @param list<SortTerm> $terms
114     * @return list<Order>
115     */
116    public function ordering(array $terms): array
117    {
118        $orders = [];
119        foreach ($terms as $term) {
120            Check::input($term->expression instanceof OutputOrdinal, 'The IR ordering must reference the projection by ordinal.');
121            $orders[] = new Order(
122                $term->expression->position(),
123                $term->direction === SortDirection::Descending,
124                $term->nulls === null ? null : $term->nulls === NullsOrder::First,
125            );
126        }
127
128        return $orders;
129    }

131    /**
132     * Resolves the supported row-bound operands, retaining their written order.
133     */
134    public function limit(?ParsedLimit $limit, SourceBinding $binding): ?Limit
135    {
136        if ($limit === null) {
137            return null;
138        }
139
140        $expressions = new ExpressionLowering();
141        $count = $expressions->lower($limit->count, $binding);
142        $offset = $limit->offset === null ? null : $expressions->lower($limit->offset, $binding);
143        Check::input($count instanceof IntegerLiteral || $count instanceof Parameter, 'LIMIT requires an integer literal or numbered parameter.');
144        Check::input($offset === null || $offset instanceof IntegerLiteral || $offset instanceof Parameter, 'OFFSET requires an integer literal or numbered parameter.');
145
146        return new Limit($count, $offset, $limit->comma);
147    }
