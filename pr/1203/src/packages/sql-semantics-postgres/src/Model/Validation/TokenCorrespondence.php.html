13 * Compares the significant tokens of the analyzed SQL with those of the rendered SQL.
14 *
15 * Both parse trees are reduced to the ordered comparison keys of their tokens.
16 * Equal sequences establish that rendering the model reproduces every
17 * significant token of the input in order; whitespace, comments, letter case
18 * of keywords, equivalent spellings of one name or literal, and the positions
19 * a database package declares as noise are not significant.
20 *
21 * @visibility SqlSemantics
22 */
23final class TokenCorrespondence
24{
25    /**
26     * Answers the ordered comparison keys of the tokens of a parse tree.
27     *
28     * @return list<string>
29     */
30    public function keys(Node $tree, Productions $productions, LeafKeys $rules): array
31    {
32        $keys = [];
33        $frames = [[$tree, $productions->signature($tree), 0]];
34        while ($frames !== []) {
35            $top = count($frames) - 1;
36            [$node, $signature, $position] = $frames[$top];
37            if ($position >= count($node->children)) {
38                array_pop($frames);
39                continue;
40            }
41            $frames[$top][2] = $position + 1;
42            $child = $node->children[$position];
43            if ($child instanceof Token) {
44                $key = $rules->key($child, $signature, $position);
45                if ($key !== null) {
46                    $keys[] = $key;
47                }
48            } else {
49                $frames[] = [$child, $productions->signature($child), 0];
50            }
51        }
52
53        return $keys;
54    }
55
56    /**
57     * Answers a description of the first difference between two key sequences, or null when they are equal.
58     *
59     * @param list<string> $source
60     * @param list<string> $rendered
61     */
62    public function difference(array $source, array $rendered): ?string
63    {
64        $length = max(count($source), count($rendered));
65        for ($index = 0; $index < $length; $index++) {
66            $expected = $source[$index] ?? '(end)';
67            $actual = $rendered[$index] ?? '(end)';
68            if ($expected !== $actual) {
69                return 'token ' . $index . ': source has ' . $expected . ', rendered SQL has ' . $actual;
70            }
71        }
72
73        return null;
74    }
