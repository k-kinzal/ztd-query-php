10/**
11 * A table name written in a statement and what it resolves to.
12 *
13 * The value is the name as written, so it can be found and rewritten in the
14 * statement by identity. A grammar that writes a qualified name as separate
15 * values of one form, as SQLite's FROM writes a schema and a table, has no
16 * single value for it: the values are all of them, in writing order, and
17 * the value is the first. The name parts are decoded, in writing order, the
18 * schema before the table when the name is qualified.
19 *
20 * @visibility public
21 * @example Reading a resolved reference
22 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite);
23 *     $users = $semantics->analyze('CREATE TABLE users (id INTEGER)', []);
24 *     $reference = $semantics->analyze('SELECT id FROM users', [$users])->resolution?->references[0];
25 *     [$reference?->name, $reference?->kind?->name, $reference?->declaration === $users] // => [['users'], 'Dependency', true]
26 * @example An unresolved reference cannot claim to be a conditional declaration
27 *     $value = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite))->builder()->column('t');
28 *     new \SqlSemantics\Statement\Reference($value, ['t'], \SqlSemantics\Statement\ReferenceKind::Undeclared, conditional: true) // throws \InvalidArgumentException
29 */
30final class Reference
31{
32    /**
33     * Every value that writes the name, in writing order; the value alone for a name written as one value.
34     *
35     * @var non-empty-list<Element>
36     */
37    public readonly array $values;
38
39    /**
40     * @param Element $value The name as written in the statement
41     * @param non-empty-list<string> $name The decoded name parts, in writing order
42     * @param ReferenceKind $kind What the name resolves to
43     * @param Statement|null $declaration The dependency that declares the table, when one does
44     * @param TableDefinition|null $table The declared table, when the name resolves to one
45     * @param bool $conditional Whether the statement only declares or drops the table if that is possible
46     * @param list<Element> $values Every value that writes the name, in writing order, the value first; empty when the value alone writes it
47     *
48     * @throws InvalidArgumentException When the values are not a list of values starting with the value, or an undeclared table or a common table expression has a declaration
49     */
50    public function __construct(
51        public readonly Element $value,
52        public readonly array $name,
53        public readonly ReferenceKind $kind,
54        public readonly ?Statement $declaration = null,
55        public readonly ?TableDefinition $table = null,
56        public readonly bool $conditional = false,
57        array $values = [],
58    ) {
59        Declaration\Invariant::nonEmptyNames($name);
60        Declaration\Invariant::elements($value);
61        Declaration\Invariant::ensure($kind !== ReferenceKind::Undeclared || !$conditional, 'An undeclared reference cannot be conditional.');
62        Declaration\Invariant::ensure(!in_array($kind, [ReferenceKind::Undeclared, ReferenceKind::CommonTableExpression], true) || ($declaration === null && $table === null), 'A reference to an undeclared table or a common table expression has no declaration.');
63        Declaration\Invariant::members($values, Element::class);
64        Declaration\Invariant::ensure($values === [] || $values[0] === $value, 'The values that write a name start with the value of the reference.');
65        Declaration\Invariant::elements(...$values);
66        $this->values = $values === [] ? [$value] : $values;
67    }
