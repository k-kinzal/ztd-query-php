7/**
8 * A structured SQL value that writes its own fields and fixed syntax.
9 *
10 * A value knows the values it is made of, in SQL order, and can rebuild
11 * itself around replacements for them, so any statement can be walked and
12 * rewritten without knowing its concrete classes. Lexical fields such as a
13 * name or a literal spelling, and comments, are not child values.
14 *
15 * @visibility public
16 * @example Accepting a structured SQL value
17 *     $write = static fn (\SqlSemantics\Statement\Element $value): string => \SqlSemantics\Statement\Writer::render($value);
18 *     $write instanceof \Closure // => true
19 * @example Reading the operands of a value
20 *     $value = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite))->analyze('SELECT 1')->command;
21 *     count($value->children()) // => 1
22 */
23interface Element
24{
25    /**
26     * Writes SQL using this value's data, without a parser or a source tree.
27     */
28    public function write(Writer $writer): void;
29
30    /**
31     * Lists the values this value is made of, in the order they are written.
32     *
33     * @return list<Element>
34     */
35    public function children(): array;
36
37    /**
38     * Returns this value rebuilt around what the function answers for each child, keeping every other field.
39     *
40     * A value returns itself when the function answers every child with the
41     * child itself, so a value keeps its identity until something below it
42     * is replaced. The replacement for a child must be a value the child's
43     * position accepts.
44     *
45     * @param callable(Element): Element $replace
46     */
47    public function map(callable $replace): static;
48}
