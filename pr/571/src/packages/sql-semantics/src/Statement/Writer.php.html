7/**
8 * Separates SQL values without changing quoted content or lexical adjacency.
9 *
10 * Values are separated by one space unless the language requires them to touch:
11 * a name and the `.` or `(` after it, and a prefix value such as a variable
12 * marker and what it introduces. Comments are written where the value they
13 * precede is written, and a line comment ends its line.
14 *
15 * @visibility public
16 * @example Rendering an SQL fragment
17 *     $render = static fn (\SqlSemantics\Statement\Element $element): string => \SqlSemantics\Statement\Writer::render($element);
18 *     $render instanceof \Closure // => true
19 * @example Keeping a prefix value attached to what follows it
20 *     $writer = new \SqlSemantics\Statement\Writer();
21 *     $writer->append('@', prefix: true);
22 *     $writer->append('name', true);
23 *     $writer->toString() // => '@name'
24 */
25final class Writer
26{

28    private string $previous = '';
29    private bool $previousIdentifier = false;
30    private bool $previousPrefix = false;
31    private ?bool $previousLineComment = null;

44    /**
45     * Emits one fixed SQL word or an indivisible argument value.
46     *
47     * @param bool $identifier Whether the value is a name, which `.` follows directly and `(` does not
48     * @param bool $prefix Whether the next value must follow this one without a separator
49     */
50    public function append(string $value, bool $identifier = false, bool $prefix = false): void
51    {
52        if ($value === '') {
53            return;
54        }
55        $separator = ' ';
56        if ($this->sql === '') {
57            $separator = '';
58        } elseif ($this->previousLineComment !== null) {
59            $separator = $this->previousLineComment ? "\n" : ' ';
60        } elseif ($this->previousPrefix || $this->previous === '.' || ($value === '(' && !$this->previousIdentifier) || ($value === '.' && $this->previousIdentifier)) {
61            $separator = '';
62        }
63        $this->sql .= $separator . $value;
64        $this->previous = $value;
65        $this->previousIdentifier = $identifier;
66        $this->previousPrefix = $prefix;
67        $this->previousLineComment = null;
68    }

70    /**
71     * Emits the comments written before the symbol at a position of a value.
72     */
73    public function comments(Comments $comments, int $position): void
74    {
75        foreach ($comments->before($position) as $comment) {
76            $this->sql .= ($this->sql === '' ? '' : ($this->previousLineComment === true ? "\n" : ' ')) . $comment;
77            $this->previous = '';
78            $this->previousIdentifier = false;
79            $this->previousPrefix = false;
80            $this->previousLineComment = str_starts_with($comment, '--') || str_starts_with($comment, '#');
81        }
82    }
