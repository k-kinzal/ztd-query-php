21/**
22 * Resolves the table names of a statement against the declarations of its dependencies.
23 *
24 * The dependencies are applied in order: a declaration adds a table, a drop
25 * removes one, and a conditional declaration or drop of a table that is
26 * already there or already gone changes nothing. The statement's own names
27 * then resolve to a common table expression it defines, to a table a
28 * dependency declares, to a table it declares itself, or to a table it
29 * drops; an unqualified name is read in the schemas of the search path, in
30 * order. Any other name is an error, because the dependency that would
31 * declare it was not given, unless the declarations are partial, where it
32 * is an undeclared table. A declaration whose columns come from another
33 * relation, such as CREATE TABLE ... LIKE or ... AS SELECT, declares its
34 * name without a readable table.
35 *
36 * @visibility SqlSemantics
37 */
38final class Resolver
39{
40    private readonly NameSites $sites;
41
42    /**
43     * Resolves statements of the language, reading unqualified names in the schemas of the search path.
44     *
45     * @param non-empty-list<string> $path The schemas an unqualified name is read in, in order
46     */
47    public function __construct(private readonly Language $language, private readonly SchemaReader $reader, private readonly array $path)
48    {
49        $vocabulary = $language->vocabulary();
50        $platform = $language->dialect->platform();
51        $this->sites = new NameSites($vocabulary, $platform->relations(), $platform->names(), new Forms($vocabulary));
52    }

54    /**
55     * Resolves one statement, given its parse tree for the declarations it makes and its dependencies with their resolutions.
56     *
57     * Declarations are resolved before the other names, so a statement can
58     * refer to a table it declares itself.
59     *
60     * @param list<array{Statement, Resolution}> $dependencies
61     * @param Declarations $declarations Whether the dependencies declare every table, or a name no dependency declares is an undeclared table
62     *
63     * @throws SemanticException When a name resolves to nothing under complete declarations, or a declaration conflicts with one in force
64     * @throws LogicException When the relation rules of the dialect do not fit its grammar
65     */
66    public function resolve(Node $tree, Command $command, array $dependencies, Declarations $declarations = Declarations::Complete): Resolution
67    {
68        $relations = new Relations($this->language->dialect->platform()->names(), $this->path, $this->language->dialect->platform()->schema()->implicitSchemas());
69        foreach ($dependencies as [$dependency, $resolution]) {
70            foreach ($resolution->references as $reference) {
71                $relations->apply($reference, $dependency);
72            }
73        }
74        $tables = $this->declarations($tree);
75        $sites = $this->sites->find($command);
76        $references = [];
77        foreach ($sites as $index => $site) {
78            if ($site->kind === ReferenceKind::Declaration) {
79                $references[$index] = $this->declare($site, $relations, $tables, $tree);
80            }
81        }
82        foreach ($sites as $index => $site) {
83            if ($site->kind !== ReferenceKind::Declaration) {
84                $references[$index] = $this->refer($site, $relations, $tree, $declarations);
85            }
86        }
87        ksort($references);
88        $declared = [];
89        foreach ($references as $reference) {
90            if ($reference->kind === ReferenceKind::Declaration && $reference->table !== null && !in_array($reference->table, $declared, true)) {
91                $declared[] = $reference->table;
92            }
93        }
94
95        return new Resolution(array_map(static fn (array $pair): Statement => $pair[0], $dependencies), $declared, array_values($references));
96    }

98    /**
99     * Resolves a declaration site: a new table is put in force, and a conditional declaration of a table in force refers to it instead.
100     *
101     * @param list<TableDefinition> $declarations The readable declarations of the statement
102     *
103     * @throws SemanticException When the table is in force and the declaration is not conditional
104     */
105    public function declare(NameSite $site, Relations $relations, array $declarations, Node $tree): Reference
106    {
107        [$schema, $table] = $relations->qualified($site->name);
108        if (count($site->name) === 1) {
109            foreach ($declarations as $candidate) {
110                if ($relations->same($candidate->schema, $candidate->name, $candidate->schema, $table)) {
111                    $schema = $candidate->schema;
112                    break;
113                }
114            }
115        }
116        $known = $relations->lookup($schema, $table);
117        if ($known !== null && !$site->conditional) {
118            throw new SemanticException('duplicate-table', 'Duplicate table declaration: ' . $table, $tree);
119        }
120        if ($known !== null) {
121            return new Reference($site->value, $site->name, ReferenceKind::Dependency, $known[1], $known[0], true, $site->values);
122        }
123        $definition = $this->declared($declarations, $relations, $schema, $table);
124        $relations->declare($schema, $table, $definition, null);
125
126        return new Reference($site->value, $site->name, ReferenceKind::Declaration, null, $definition, $site->conditional, $site->values);
127    }
128
129    /**
130     * Resolves a site that defines, drops or refers to a table: to a common table expression visible at the site, to a table in force, or, under partial declarations, to an undeclared table unless the dependencies dropped it.
131     *
132     * @throws SemanticException When the name is not in force under complete declarations, and the site is not a conditional drop
133     */
134    public function refer(NameSite $site, Relations $relations, Node $tree, Declarations $declarations = Declarations::Complete): Reference
135    {
136        if ($site->kind === ReferenceKind::CommonTableExpression) {
137            return new Reference($site->value, $site->name, ReferenceKind::CommonTableExpression, values: $site->values);
138        }
139        $known = $relations->find($site->name);
140        $open = $declarations === Declarations::Partial && !$relations->gone($site->name);
141        if ($site->kind === ReferenceKind::Drop) {
142            if ($known === null && !$site->conditional && !$open) {
143                throw new SemanticException('unknown-table', 'Cannot drop an unknown table: ' . implode('.', $site->name), $tree);
144            }
145
146            return new Reference($site->value, $site->name, ReferenceKind::Drop, $known[3] ?? null, $known[2] ?? null, $site->conditional, $site->values);
147        }
148        if ($site->scope->contains($site->name, $this->language->dialect->platform()->names())) {
149            return new Reference($site->value, $site->name, ReferenceKind::CommonTableExpression, values: $site->values);
150        }
151        if ($known === null && $open) {
152            return new Reference($site->value, $site->name, ReferenceKind::Undeclared, values: $site->values);
153        }
154        if ($known === null) {
155            throw new SemanticException('unknown-table', 'No dependency declares the table ' . implode('.', $site->name), $tree);
156        }
157
158        return new Reference($site->value, $site->name, $known[3] === null ? ReferenceKind::Declaration : ReferenceKind::Dependency, $known[3], $known[2], false, $site->values);
159    }

161    /**
162     * Reads the tables the statement declares with their columns, from its parse tree.
163     *
164     * A declaration whose columns come from another relation, such as
165     * CREATE TABLE ... LIKE or ... AS SELECT, is not readable and is left out;
166     * its name still resolves as a declaration without a table.
167     *
168     * @return list<TableDefinition>
169     *
170     * @throws SemanticException When a declaration is readable but invalid
171     */
172    public function declarations(Node $tree): array
173    {
174        $platform = $this->language->dialect->platform();
175        $tables = [];
176        foreach (Tree::outer($tree, [$platform->statementNames()[1], ...$platform->syntax()->nodes('statement')]) as $statement) {
177            $create = Tree::child($statement, $platform->syntax()->nodes('createTable'));
178            if ($create === null) {
179                continue;
180            }
181            try {
182                $tables[] = $this->reader->table($platform->schema()->schemaNode($statement, $create));
183            } catch (SemanticException $error) {
184                if ($error->reason !== 'unsupported-syntax') {
185                    throw $error;
186                }
187            }
188        }
189
190        return $tables;
191    }
192
193    /**
194     * Finds the readable declaration of a table among those of the statement.
195     *
196     * @param list<TableDefinition> $declarations
197     */
198    public function declared(array $declarations, Relations $relations, string $schema, string $name): ?TableDefinition
199    {
200        foreach ($declarations as $table) {
201            if ($relations->same($table->schema, $table->name, $schema, $name)) {
202                return $table;
203            }
204        }
205
206        return null;
207    }
