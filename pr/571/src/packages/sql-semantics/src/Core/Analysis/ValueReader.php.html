51    /**
52     * Lowers a complete parse tree into a statement that keeps its comments, and discards the tree.
53     *
54     * @throws LogicException When parser and model resources disagree
55     */
56    public function statement(Node $root): Statement
57    {
58        $comments = $this->trivia->read($root);
59        $command = $this->lower($root, $comments, true);
60        $this->assertCompleteCommand($command);
61        $around = [];
62        if ($comments->leading !== []) {
63            $around[Statement::BEFORE] = $comments->leading;
64        }
65        if ($comments->trailing !== []) {
66            $around[Statement::AFTER] = $comments->trailing;
67        }
68
69        return new Statement($command, $around === [] ? $this->none : new Comments($around));
70    }

74     *

85    /**
86     * Builds the value of a node, keeping the comments of the symbols it owns.
87     *
88     * @param SourceComments $comments The comments of the tree the node belongs to
89     * @param bool $firstKept Whether an enclosing value or the statement already keeps the comments before the first token
90     *
91     * @throws LogicException When parser and model resources disagree
92     */
93    public function lower(Node $node, SourceComments $comments, bool $firstKept): Element
94    {
95        $recipe = $this->vocabulary->recipe($node->name, $node->ordinal);
96        if ($recipe === null) {
97            throw new LogicException('Parser/model resource mismatch at ' . $node->name . ':' . $node->ordinal);
98        }
99        if (isset($recipe['forward'])) {
100            $child = $node->children[$recipe['forward']] ?? null;
101            if (!$child instanceof Node) {
102                throw new LogicException('Forwarding model requires a structured value');
103            }
104
105            return $this->lower($child, $comments, $firstKept);
106        }
107        if (isset($recipe['constant'])) {
108            $choice = constant($recipe['constant']);
109            if (!$choice instanceof Element) {
110                throw new LogicException('A model choice must implement Element');
111            }
112
113            return $choice;
114        }
115        $positions = [];
116        $begins = [];
117        foreach ($node->children as $index => $child) {
118            $token = $child instanceof Token ? $child : $comments->first($child);
119            if ($token === null) {
120                continue;
121            }
122            $begins[$index] = true;
123            if ($firstKept) {
124                $firstKept = false;
125                continue;
126            }
127            $texts = $comments->before($token);
128            if ($texts !== []) {
129                $positions[$index] = $texts;
130            }
131        }
132        $arguments = [];
133        foreach ($recipe['fields'] as $index) {
134            $child = $node->children[$index] ?? null;
135            if ($child === null) {
136                throw new LogicException('Missing model argument ' . $index . ' in ' . $node->name);
137            }
138            $arguments[] = $child instanceof Token ? $child->text : $this->lower($child, $comments, isset($begins[$index]));
139        }
140
141        return new ($recipe['class'])(...$arguments, comments: $positions === [] ? $this->none : new Comments($positions));
142    }
143}
