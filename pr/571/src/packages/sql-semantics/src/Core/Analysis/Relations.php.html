13/**
14 * The tables in force while a statement is resolved.
15 *
16 * Tables are compared under the dialect's relation name policy. A name
17 * without a schema refers to the table of the first schema of the search
18 * path that has one, and is declared in the first schema of the path. A
19 * table that was dropped is remembered as gone until it is declared again.
20 *
21 * @visibility SqlSemantics
22 */

27     */

36     * Starts with no table in force.
37     *
38     * @param non-empty-list<string> $path The schemas an unqualified name is read in, in order
39     * @param list<string> $implicitSchemas Implicit namespaces searched before the session path for declared tables
40     */
41    public function __construct(private readonly NameRules $names, private readonly array $path, private readonly array $implicitSchemas = [])
42    {
43    }

45    /**
46     * Applies a reference of a dependency: a declaration puts its table in force, a drop takes it out, and any other reference changes nothing.
47     */
48    public function apply(Reference $reference, Statement $dependency): void
49    {
50        if ($reference->kind === ReferenceKind::Declaration) {
51            [$schema, $name] = ($reference->table === null || $reference->table->schema === '') ? $this->qualified($reference->name) : [$reference->table->schema, $reference->table->name];
52            $this->declare($schema, $name, $reference->table, $dependency);
53        } elseif ($reference->kind === ReferenceKind::Drop) {
54            $found = $this->find($reference->name) ?? $this->qualified($reference->name);
55            $this->drop($found[0], $found[1]);
56        }
57    }
58
59    /**
60     * Puts a table in force, with its readable definition and the statement that declares it, when there are any.
61     */
62    public function declare(string $schema, string $name, ?TableDefinition $table, ?Statement $owner): void
63    {
64        $this->tables[] = [$schema, $name, $table, $owner];
65        $this->dropped = array_values(array_filter($this->dropped, fn (array $gone): bool => !$this->same($gone[0], $gone[1], $schema, $name)));
66    }
67
68    /**
69     * Takes a table out of force.
70     */
71    public function drop(string $schema, string $name): void
72    {
73        $this->tables = array_values(array_filter($this->tables, fn (array $known): bool => !$this->same($known[0], $known[1], $schema, $name)));
74        $this->dropped[] = [$schema, $name];
75    }
76
77    /**
78     * Reports whether a name refers to no table because the tables it could refer to were dropped: in its schema, or in every schema of the search path.
79     *
80     * @param non-empty-list<string> $name
81     */
82    public function gone(array $name): bool
83    {
84        $parts = array_slice($name, -2);
85        $table = $parts[count($parts) - 1];
86        foreach (count($parts) === 2 ? [$parts[0]] : $this->path as $schema) {
87            if (array_filter($this->dropped, fn (array $gone): bool => $this->same($gone[0], $gone[1], $schema, $table)) === []) {
88                return false;
89            }
90        }
91
92        return true;
93    }
94
95    /**
96     * Finds a table in force: its readable definition and the statement that declares it, or null when it is not in force.
97     *
98     * @return array{TableDefinition|null, Statement|null}|null
99     */
100    public function lookup(string $schema, string $name): ?array
101    {
102        foreach ($this->tables as [$knownSchema, $knownName, $table, $owner]) {
103            if ($this->same($knownSchema, $knownName, $schema, $name)) {
104                return [$table, $owner];
105            }
106        }
107
108        return null;
109    }

111    /**
112     * Finds the table in force a name refers to: a qualified name in its schema, an unqualified one in the first schema of the search path that has it.
113     *
114     * @param non-empty-list<string> $name
115     * @return array{string, string, TableDefinition|null, Statement|null}|null The schema and name the table is in force under, its definition and its owner
116     */
117    public function find(array $name): ?array
118    {
119        $parts = array_slice($name, -2);
120        $table = $parts[count($parts) - 1];
121        foreach (count($parts) === 2 ? [$parts[0]] : [...$this->implicitSchemas, ...$this->path] as $schema) {
122            foreach ($this->tables as [$knownSchema, $knownName, $definition, $owner]) {
123                if ($this->same($knownSchema, $knownName, $schema, $table)) {
124                    return [$knownSchema, $knownName, $definition, $owner];
125                }
126            }
127        }
128
129        return null;
130    }
131
132    /**
133     * Qualifies a name with the schema an unqualified declaration creates its table in, keeping its last two parts.
134     *
135     * @param non-empty-list<string> $name
136     * @return array{string, string}
137     */
138    public function qualified(array $name): array
139    {
140        $parts = array_slice($name, -2);
141
142        return count($parts) === 2 ? [$parts[0], $parts[1]] : [$this->path[0], $parts[0]];
143    }

145    /**
146     * Compares two qualified table names under the dialect's relation name policy.
147     */
148    public function same(string $schema, string $name, string $otherSchema, string $otherName): bool
149    {
150        return $this->names->relationEqual($schema, $otherSchema) && $this->names->relationEqual($name, $otherName);
151    }
