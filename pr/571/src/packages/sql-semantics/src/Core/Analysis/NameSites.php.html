16/**
17 * Finds every place a statement writes a table name, using the relation rules of its grammar.
18 *
19 * The statement is walked once, with the common table expressions visible
20 * at each value. At each value, the sites the relation rules describe for
21 * its form are read first, so a declaration, drop, definition or ignored
22 * name is not read again as a plain reference; then every remaining name
23 * symbol of the form is a reference.
24 *
25 * @phpstan-import-type Site from RelationRules
26 * @visibility SqlSemantics
27 */
28final class NameSites
29{
30    private const KINDS = ['declarations' => ReferenceKind::Declaration, 'drops' => ReferenceKind::Drop, 'commonTableExpressions' => ReferenceKind::CommonTableExpression, 'ignored' => null, 'pairs' => ReferenceKind::Dependency];

34    /**
35     * Finds name sites in statements of a vocabulary under its relation rules.
36     */
37    public function __construct(private readonly Vocabulary $vocabulary, private readonly RelationRules $rules, private readonly NameRules $names, private readonly Forms $forms)
38    {
39        $this->scopes = new Scopes($vocabulary, $rules, $names, $forms);
40    }

42    /**
43     * Finds the name sites of a command, in writing order.
44     *
45     * A site read at a form is ordered by the value it names, so a target
46     * written after a WITH clause comes after the names inside the clause.
47     *
48     * @return list<NameSite>
49     *
50     * @throws LogicException When the relation rules name a symbol a form does not have
51     */
52    public function find(Command $command): array
53    {
54        $sites = [];
55        $consumed = [];
56        $order = [];
57        foreach ($this->scopes->walk($command) as [$value, $scope]) {
58            $order[spl_object_id($value)] ??= count($order);
59            $shape = $this->vocabulary->shape($value);
60            if ($shape === null || isset($consumed[spl_object_id($value)])) {
61                continue;
62            }
63            $children = $value->children();
64            $scopes = $this->scopes->children($value, $scope);
65            $taken = [];
66            foreach (self::KINDS as $group => $kind) {
67                foreach (RelationRules::matching($this->rules->{$group}, $shape['rule'], $shape['symbols']) as $site) {
68                    if (isset($site['type']) && !in_array(Writer::render($this->forms->child($shape['symbols'], $children, $this->forms->position($shape['symbols'], $site['type'][0]))), $site['type'][1], true)) {
69                        continue;
70                    }
71                    $conditional = isset($site['conditional']) && $this->forms->conditional($shape['symbols'], $children, $site['conditional']);
72                    foreach ($this->named($shape['symbols'], $children, $site, $taken, $consumed) as [$values, $parts, $position]) {
73                        if ($kind !== null) {
74                            $sites[] = new NameSite($values[0], $parts, $kind, $conditional, $scopes[$this->forms->index($shape['symbols'], $position)], $values);
75                        }
76                    }
77                }
78            }
79            foreach ($shape['symbols'] as $position => $symbol) {
80                if (!in_array($symbol, $this->rules->nameSymbols, true) || isset($taken[$position])) {
81                    continue;
82                }
83                $taken[$position] = true;
84                $child = $this->forms->child($shape['symbols'], $children, $position);
85                if ($this->holdsNames($child)) {
86                    continue;
87                }
88                $this->consume($consumed, $child);
89                $parts = $this->parts($child);
90                if ($parts !== []) {
91                    $sites[] = new NameSite($child, $parts, ReferenceKind::Dependency, false, $scopes[$this->forms->index($shape['symbols'], $position)]);
92                }
93            }
94        }
95        usort($sites, static fn (NameSite $left, NameSite $right): int => $order[spl_object_id($left->value)] <=> $order[spl_object_id($right->value)]);
96
97        return $sites;
98    }

100    /**
101     * Answers the names a site reads, each as the values that write it with its decoded parts and the symbol position it is written at, marking the positions it takes and the values it reads.
102     *
103     * A pair writes one name as separate values; the values that write
104     * nothing, such as an absent schema, are left out.
105     *
106     * @param list<string> $symbols
107     * @param list<Element> $children
108     * @param Site $site
109     * @param array<int, true> $taken
110     * @param array<int, true> $consumed
111     * @return list<array{non-empty-list<Element>, non-empty-list<string>, int}>
112     *
113     * @throws LogicException When the site names a symbol the form does not have
114     */
115    public function named(array $symbols, array $children, array $site, array &$taken, array &$consumed): array
116    {
117        if (isset($site['pair'])) {
118            $values = [];
119            $first = null;
120            foreach ($site['pair'] as $symbol) {
121                $position = $this->forms->position($symbols, $symbol, $taken);
122                $first ??= $position;
123                $taken[$position] = true;
124                $values[] = $this->forms->child($symbols, $children, $position);
125                $this->consume($consumed, $values[count($values) - 1]);
126            }
127            $parts = array_merge(...array_map($this->parts(...), $values));
128
129            $written = array_values(array_filter($values, static fn (Element $value): bool => Writer::render($value) !== ''));
130
131            return $parts === [] || $first === null || $written === [] ? [] : [[$written, $parts, $first]];
132        }
133        if (isset($site['name'])) {
134            $position = $this->forms->position($symbols, $site['name'], $taken);
135            $taken[$position] = true;
136            $value = $this->forms->child($symbols, $children, $position);
137            $this->consume($consumed, $value);
138            $parts = $this->parts($value);
139
140            return $parts === [] ? [] : [[[$value], $parts, $position]];
141        }
142        $position = $this->forms->position($symbols, $site['names'] ?? '', $taken);
143        $taken[$position] = true;
144        $list = $this->forms->child($symbols, $children, $position);
145        $this->consume($consumed, $list);
146        $items = isset($site['list']) ? $this->forms->unfold($list, $site['list'][0], $site['list'][1]) : [$list];
147
148        return array_values(array_filter(array_map(fn (Element $item): array => [[$item], $this->parts($item), $position], $items), static fn (array $named): bool => $named[1] !== []));
149    }

151    /**
152     * Marks a value and everything below it as read by a site, so the walk does not read its names again.
153     *
154     * @param array<int, true> $consumed
155     */
156    public function consume(array &$consumed, Element $value): void
157    {
158        foreach (Traversal::walk($value) as $inner) {
159            $consumed[spl_object_id($inner)] = true;
160        }
161    }

163    /**
164     * Reports whether a value is a form that writes a table name somewhere inside it, rather than a name itself.
165     */
166    public function holdsNames(Element $value): bool
167    {
168        $shape = $this->vocabulary->shape($value);
169
170        return $shape !== null && array_intersect($shape['symbols'], $this->rules->nameSymbols) !== [];
171    }
172
173    /**
174     * Decodes the parts of a name value, in writing order.
175     *
176     * @return list<string>
177     */
178    public function parts(Element $name): array
179    {
180        $shape = $this->vocabulary->shape($name);
181        $children = $name->children();
182        if ($shape !== null && isset($this->rules->parts[$shape['rule']][implode(' ', $shape['symbols'])])) {
183            $children = array_values(array_intersect_key($children, array_flip($this->rules->parts[$shape['rule']][implode(' ', $shape['symbols'])])));
184        }
185        if ($children === []) {
186            $text = Writer::render($name);
187
188            return $text === '' ? [] : [$this->names->decode($text)];
189        }
190
191        return array_merge(...array_map($this->parts(...), $children));
192    }
