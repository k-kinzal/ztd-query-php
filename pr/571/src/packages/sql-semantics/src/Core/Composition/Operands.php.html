10 * Decides whether an operand may stand unparenthesized at a position of an expression form.
11 *
12 * The generated contracts of a database package record the binding power of
13 * every expression form, the least power each operand position of a form
14 * accepts, by grammar release, and the grammar rule of each form. A builder
15 * gives them here to parenthesize only the operands the grammar would
16 * otherwise bind differently. Precedence orders the forms of one rule, so an
17 * operand of another rule, reached through the grammar's own reductions, is
18 * not compared.
19 *
20 * @visibility SqlSemantics
21 */
22final class Operands
23{

25     * @param array<class-string<Element>, array<string, int>> $powers Binding power of each form by release
26     * @param array<class-string<Element>, array<string, array<int, int>>> $minimums Least operand power by form, release, and symbol position
27     * @param array<class-string<Element>, string> $rules The grammar rule of each expression form
28     * @param string $version The release the forms are built for
29     */
30    public function __construct(private readonly array $powers, private readonly array $minimums, private readonly array $rules, private readonly string $version)
31    {
32    }
33
34    /**
35     * Answers whether the operand binds strongly enough for a symbol position of a form.
36     *
37     * A value that is not an expression form of the contracts, such as a
38     * literal or a parenthesized expression, binds as strongly as anything.
39     *
40     * @param class-string<Element> $form The form the operand is placed in
41     * @param int $position The symbol position of the operand in the form
42     * @param string $rule The grammar rule of the position
43     */
44    public function fits(Element $operand, string $form, int $position, string $rule): bool
45    {
46        $minimum = $this->minimums[$form][$this->version][$position] ?? null;
47        if ($minimum === null || ($this->rules[$operand::class] ?? $rule) !== $rule) {
48            return true;
49        }
50
51        return ($this->powers[$operand::class][$this->version] ?? PHP_INT_MAX) >= $minimum;
52    }
