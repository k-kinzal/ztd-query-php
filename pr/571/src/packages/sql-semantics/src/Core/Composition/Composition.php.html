16 * The composition every database builder shares: forms by their symbols, leaves by their spelling, and operands by their binding.
17 *
18 * A database package supplies the spelling rules of its names, the roles
19 * its values occupy, and the contracts of its expression forms. Everything
20 * else is read from the vocabulary of the release, so a builder never names
21 * a generated class.

81     * Answers the statement a root envelope holds, when its other children write nothing; otherwise the value itself.
82     */
83    protected function unwrap(Element $value): Element
84    {
85        foreach ($this->envelopes() as [$rule, $symbols]) {
86            $class = $this->classOf($rule, $symbols);
87            if ($class === null || !$value instanceof $class) {
88                continue;
89            }
90            $children = $value->children();
91            foreach (array_slice($children, 1) as $other) {
92                if (Writer::render($other) !== '') {
93                    continue 2;
94                }
95            }
96
97            return $this->unwrap($children[0]);
98        }
99
100        return $value;

159    /**
160     * Spells a name in a role: bare when the release reads it as that name, otherwise quoted.
161     *
162     * @param string $prefix The text the lexer has just read before the name, such as the `.` of a qualified name
163     *
164     * @throws CompositionException When the name is empty or the role admits no quoted name
165     */
166    protected function name(string $role, string $name, string $prefix = ''): Element
167    {
168        if ($name === '') {
169            throw new CompositionException('A name must not be empty.');
170        }
171        $bare = $this->bare($name);
172        if ($bare !== null) {
173            $terminal = $this->leaves->terminal($bare, $prefix);
174            if ($terminal !== null && ($this->admitsBareKeyword() || in_array($terminal, $this->identifierTerminals(), true))) {
175                $leaf = $this->leaves->read($role, $bare, $prefix);
176                if ($leaf !== null) {
177                    return $leaf;
178                }
179            }
180        }
181        $leaf = $this->leaves->read($role, $this->quote($name), $prefix);
182        if ($leaf === null) {
183            throw new CompositionException('The role ' . $role . ' admits no quoted name in ' . $this->language->version);
184        }
185
186        return $leaf;
187    }

223    /**
224     * Folds values into a left-recursive list form, the first value standing alone.
225     *
226     * @param list<string> $symbols The symbols of the recursive form, the list itself first
227     * @param list<Element> $values
228     *
229     * @throws CompositionException When there is no value to fold
230     */
231    protected function fold(string $rule, array $symbols, array $values, int $listPosition = 0, int $itemPosition = 2): Element
232    {
233        $list = array_shift($values);
234        if ($list === null) {
235            throw new CompositionException('A ' . $rule . ' needs at least one value.');
236        }
237        foreach ($values as $value) {
238            $arguments = [];
239            foreach ($symbols as $index => $symbol) {
240                if ($index === $listPosition) {
241                    $arguments[] = $list;
242                } elseif ($index === $itemPosition) {
243                    $arguments[] = $value;
244                }
245            }
246            $list = $this->form($rule, $symbols, $arguments);
247        }
248
249        return $list;
250    }

252    /**
253     * Unfolds a left-recursive list form into its values: the list is its first child and the item its last.
254     *
255     * @param list<string> $symbols The symbols of the recursive form
256     * @return list<Element>
257     */
258    protected function unfold(Element $list, string $rule, array $symbols): array
259    {
260        $class = $this->classOf($rule, $symbols);
261        if ($class === null || !$list instanceof $class) {
262            return [$list];
263        }
264        $children = $list->children();
265
266        return [...$this->unfold($children[0], $rule, $symbols), $children[count($children) - 1]];
267    }

275    {
276        $name = preg_replace('/([a-z0-9])([A-Z])/', '$1 $2', $symbol) ?? $symbol;

298    /**
299     * Spells the magnitude of a finite number so the language reads it as a non-integer that is exactly the number.
300     *
301     * The digits are the fewest that read back as the same double, found
302     * without the `precision` or `serialize_precision` setting, so the
303     * spelling is the same in every environment. The number is written
304     * positionally when that is short, and with an exponent otherwise or
305     * when the language needs the exponent to read an approximate number.
306     *
307     * @param bool $exponent Whether to always write the exponent
308     *
309     * @throws CompositionException When the value is not finite
310     */
311    protected function decimal(float $value, bool $exponent = false): string
312    {
313        if (!is_finite($value)) {
314            throw new CompositionException('A numeric literal must be finite.');
315        }
316        [$digits, $power] = $this->digits(abs($value));
317        if ($exponent || $power < -7 || $power > 20) {
318            return $digits[0] . (strlen($digits) > 1 ? '.' . substr($digits, 1) : '') . 'e' . $power;
319        }
320        if ($power < 0) {
321            return '0.' . str_repeat('0', -$power - 1) . $digits;
322        }
323        $digits = str_pad($digits, $power + 1, '0');
324
325        return substr($digits, 0, $power + 1) . '.' . (strlen($digits) > $power + 1 ? substr($digits, $power + 1) : '0');
326    }
327
328    /**
329     * Answers the fewest significant digits that read back as exactly a finite, non-negative double, and the power of ten of the first.
330     *
331     * @return array{non-empty-string, int}
332     */
333    protected function digits(float $magnitude): array
334    {
335        $precision = 0;
336        do {
337            [$mantissa, $power] = explode('e', sprintf('%.' . $precision . 'e', $magnitude)) + ['', '0'];
338            $digits = preg_replace('/\D/', '', $mantissa) ?? '';
339        } while ((float) (substr($digits, 0, 1) . '.' . substr($digits, 1) . 'e' . $power) !== $magnitude && ++$precision < 17);
340        $digits = rtrim($digits, '0');
341
342        return [$digits === '' ? '0' : $digits, (int) $power];
343    }

353    /**
354     * Spells a parameter marker: the language's own for a position, or `:name` for a name the language reads.
355     *
356     * @param string $positional The language's marker for the position, when the marker is a position
357     *
358     * @throws CompositionException When the position is below one, the name is not a name, or the language reads no named placeholder
359     */
360    protected function marker(int|string $marker, string $positional): string
361    {
362        if (is_int($marker)) {
363            if ($marker < 1) {
364                throw new CompositionException('A parameter position counts from one.');
365            }
366
367            return $positional;
368        }
369        if (preg_match('/^[A-Za-z0-9_]+$/D', $marker) !== 1) {
370            throw new CompositionException('A parameter name holds letters, digits, and underscores.');
371        }
372        if ($this->leaves->terminal(':' . $marker) === null) {
373            throw new CompositionException('The language reads no named placeholder; select the named parameter syntax.');
374        }
375
376        return ':' . $marker;
377    }
