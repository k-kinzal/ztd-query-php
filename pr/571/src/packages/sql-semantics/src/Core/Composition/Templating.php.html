12/**
13 * Composes queries and expressions from templates the release reads, with every slot replaced.
14 *
15 * A template writes the form with bare names where operands and names go:
16 * `slot0`, `slot1`, ... for operands and `slot_0`, `slot_1`, ... for names.
17 * Each operand is placed at the position of its slot, parenthesized where it
18 * binds more weakly than the position needs, and each name is spelled for
19 * the role of its position. A database package supplies how its CAST writes
20 * a type.
21 *
22 * @visibility SqlSemantics
23 */
24trait Templating
25{

64    /**
65     * A searched CASE: the result of the first condition that holds, else the default or NULL.
66     */
67    public function case(array $whens, ?Element $else = null): Element
68    {
69        if ($whens === []) {
70            throw new CompositionException('A CASE needs at least one WHEN.');
71        }
72        $text = 'CASE';
73        $operands = [];
74        foreach ($whens as [$condition, $result]) {
75            $text .= ' WHEN slot' . count($operands) . ' THEN slot' . (count($operands) + 1);
76            array_push($operands, $condition, $result);
77        }
78        if ($else !== null) {
79            $text .= ' ELSE slot' . count($operands);
80            $operands[] = $else;
81        }
82
83        return $this->expression($text . ' END', $operands);
84    }

86    /**
87     * A call of the function a name denotes: the name bare when its characters survive unquoted, quoted otherwise.
88     *
89     * A bare name the release reads as a keyword is the function the grammar
90     * gives that keyword; quoting it would call another function, so a call
91     * the grammar has no form for is an error.
92     */
93    public function call(string $name, array $arguments = []): Element
94    {
95        if ($name === '') {
96            throw new CompositionException('A function name must not be empty.');
97        }
98        $spelled = $this->bare($name) ?? $this->quote($name);
99        $marker = $this->slotMarker($spelled);
100        $slots = '(' . implode(', ', array_map(static fn (int $index): string => $marker . $index, array_keys($arguments))) . ')';
101
102        return $this->expression($spelled . $slots, $arguments, $marker);
103    }

113    /**
114     * A SELECT of columns, each an expression with an optional alias, from an optional table and filtered by an optional condition.
115     *
116     * The table is a table name, such as one answered by table(); it is
117     * written into the template as it is written, so the slots are named
118     * with a word it does not contain.
119     */
120    public function select(array $columns, ?Element $from = null, ?Element $where = null): Element
121    {
122        if ($columns === []) {
123            throw new CompositionException('A SELECT needs at least one column.');
124        }
125        if ($from !== null) {
126            $this->expect($from, $this->tableSymbol(), 'The table of a SELECT');
127        }
128        $table = $from === null ? '' : Writer::render($from);
129        $marker = $this->slotMarker($table);
130        $items = [];
131        $operands = [];
132        $names = [];
133        foreach ($columns as [$expression, $alias]) {
134            $item = $marker . count($operands);
135            $operands[] = $expression;
136            if ($alias !== null) {
137                $item .= ' AS ' . $marker . '_' . count($names);
138                $names[] = $alias;
139            }
140            $items[] = $item;
141        }
142        $text = 'SELECT ' . implode(', ', $items) . ($table === '' ? '' : ' FROM ' . $table);
143        if ($where !== null) {
144            $text .= ' WHERE ' . $marker . count($operands);
145            $operands[] = $where;
146        }
147
148        return $this->substitute($this->unwrap($this->templates()->command($text)), $marker, $operands, $names);
149    }

152     * Composes the expression a template writes as the column of a SELECT, with its slots replaced.

154     * @param list<Element> $operands The operand of each slot, by slot number
155     *
156     * @throws CompositionException When the release reads no such expression
157     */
158    protected function expression(string $text, array $operands, string $marker = 'slot'): Element
159    {
160        $value = $this->templates()->written($this->templates()->command('SELECT ' . $text), $text, $this->roleInterface($this->expressionSymbol()));
161        if ($value === null) {
162            throw new CompositionException('No expression ' . $text . ' in ' . $this->language->version);
163        }

181    /**
182     * Replaces the slots of a template value: an operand at its position, parenthesized where it binds too weakly, and a name spelled for the role of its position.
183     *
184     * A slot is a leaf that writes the marker and a number, `slot0`, for an
185     * operand, or the marker, an underscore, and a number, `slot_0`, for a
186     * name.
187     *
188     * @param string $marker The letters that begin every slot
189     * @param list<Element> $operands The operand of each operand slot, by its number
190     * @param list<string> $names The name of each name slot, by its number
191     *
192     * @throws CompositionException When an operand cannot stand at its position
193     */
194    protected function substitute(Element $value, string $marker, array $operands, array $names = []): Element
195    {
196        $shape = $this->vocabulary->shape($value);
197        if ($shape === null) {
198            return $value;
199        }
200        $positions = array_keys(array_filter($shape['symbols'], $this->vocabulary->isRule(...)));
201        $replacements = [];
202        foreach ($value->children() as $index => $child) {
203            $text = $child->children() === [] ? Writer::render($child) : '';
204            $replacements[] = match (true) {
205                preg_match('/^' . $marker . '(\d+)$/D', $text, $slot) === 1 => $this->placed($operands[(int) $slot[1]] ?? throw new CompositionException('No operand for the slot ' . $text), $shape['rule'], $shape['symbols'], $positions[$index]),
206                preg_match('/^' . $marker . '_(\d+)$/D', $text, $slot) === 1 => $this->name($shape['symbols'][$positions[$index]], $names[(int) $slot[1]] ?? throw new CompositionException('No name for the slot ' . $text)),
207                default => $this->substitute($child, $marker, $operands, $names),
208            };
209        }
210        $index = 0;
211
212        return $value->map(static function () use (&$index, $replacements): Element {
213            return $replacements[$index++];
214        });
215    }

217    /**
218     * Places an operand at a symbol position of a form, parenthesized where it does not belong there.
219     *
220     * @param list<string> $symbols
221     *
222     * @throws CompositionException When the operand cannot stand at the position even parenthesized
223     */
224    protected function placed(Element $operand, string $rule, array $symbols, int $position): Element
225    {
226        $placed = $this->operand($operand, $rule, $symbols, $position);
227        $role = $this->roleInterface($symbols[$position]);
228        if (!$placed instanceof $role) {
229            throw new CompositionException('An operand of ' . $rule . ' must be a ' . $symbols[$position] . ' of ' . $this->language->version . ', ' . $operand::class . ' given.');
230        }
231
232        return $placed;
233    }

242    protected function statesOnly(TypeDescriptor $type, array $facts): void
243    {
244        $stated = [
245            'length' => $type->length !== null,
246            'precision' => $type->precision !== null,
247            'scale' => $type->scale !== null,
248            'unsigned' => $type->unsigned,
249            'zerofill' => $type->zerofill,
250            'binaryCollation' => $type->binaryCollation,
251            'characterSet' => $type->characterSet !== null,
252            'members' => $type->members !== [],
253            'arrayDimensions' => $type->arrayDimensions !== 0,
254            'intervalFields' => $type->intervalFields !== null,
255            'affinity' => $type->affinity !== null,
256        ];
257        foreach ($stated as $fact => $present) {
258            if ($present && !in_array($fact, $facts, true)) {
259                throw new CompositionException('A CAST to ' . $type->label() . ' in ' . $this->language->version . ' cannot state its ' . $fact . '.');
260            }
261        }
262    }
