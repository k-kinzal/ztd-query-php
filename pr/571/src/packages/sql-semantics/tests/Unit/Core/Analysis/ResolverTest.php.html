117    public static function providerScopedStatements(): iterable
118    {
119        $selfReference = 'WITH users AS (SELECT * FROM users WHERE id > 1) SELECT * FROM users';
120        yield 'mysql: a plain expression does not see itself' => [MySqlDialect::MySql, $selfReference, 'users:CTE users:Dependency users:CTE'];
121        yield 'postgresql: a plain expression does not see itself' => [PostgreSqlDialect::PostgreSql, $selfReference, 'users:CTE users:Dependency users:CTE'];
122        yield 'sqlite: an expression always sees itself' => [SqliteDialect::Sqlite, $selfReference, 'users:CTE users:CTE users:CTE'];
123        $forward = "WITH a AS (SELECT * FROM b), b AS (SELECT 'x' AS x) SELECT * FROM a";
124        yield 'mysql: a later expression is not visible' => [MySqlDialect::MySql, $forward, 'a:CTE b:Dependency b:CTE a:CTE'];
125        yield 'postgresql: a later expression is not visible in a plain clause' => [PostgreSqlDialect::PostgreSql, $forward, 'a:CTE b:Dependency b:CTE a:CTE'];
126        yield 'sqlite: a later expression is visible' => [SqliteDialect::Sqlite, $forward, 'a:CTE b:CTE b:CTE a:CTE'];
127        $recursiveForward = "WITH RECURSIVE a AS (SELECT * FROM b), b AS (SELECT 'x' AS x) SELECT * FROM a";
128        yield 'mysql: a later expression is not visible in a recursive clause' => [MySqlDialect::MySql, $recursiveForward, 'a:CTE b:Dependency b:CTE a:CTE'];
129        yield 'postgresql: a later expression is visible in a recursive clause' => [PostgreSqlDialect::PostgreSql, $recursiveForward, 'a:CTE b:CTE b:CTE a:CTE'];
130        yield 'sqlite: a later expression is visible in a recursive clause' => [SqliteDialect::Sqlite, $recursiveForward, 'a:CTE b:CTE b:CTE a:CTE'];
131        $recursive = 'WITH RECURSIVE r AS (SELECT 1 AS n UNION ALL SELECT n + 1 FROM r WHERE n < 3) SELECT * FROM r';
132        $preceding = "WITH b AS (SELECT 'x' AS x), a AS (SELECT * FROM b) SELECT * FROM a";
133        $shadowed = "WITH b AS (SELECT 'outer' AS x) SELECT * FROM (WITH b AS (SELECT 'inner' AS x) SELECT * FROM b) AS s, b";
134        $subquery = "WITH b AS (SELECT 'outer' AS x) SELECT (SELECT x FROM b) AS y";
135        $inner = "SELECT * FROM (WITH b AS (SELECT 'inner' AS x) SELECT * FROM b) AS s WHERE EXISTS (SELECT * FROM b)";
136        $insertQuery = "INSERT INTO users WITH users AS (SELECT 8 AS id, 'y' AS v) SELECT * FROM users";
137        foreach ([MySqlDialect::MySql, PostgreSqlDialect::PostgreSql, SqliteDialect::Sqlite] as $dialect) {
138            yield $dialect->name . ': a recursive expression sees itself' => [$dialect, $recursive, 'r:CTE r:CTE r:CTE'];
139            yield $dialect->name . ': an earlier expression is visible' => [$dialect, $preceding, 'b:CTE a:CTE b:CTE a:CTE'];
140            yield $dialect->name . ': an inner expression shadows an outer one' => [$dialect, $shadowed, 'b:CTE b:CTE b:CTE b:CTE'];
141            yield $dialect->name . ': a subquery sees the clause of its query' => [$dialect, $subquery, 'b:CTE b:CTE'];
142            yield $dialect->name . ': an inner clause is not visible outside its query' => [$dialect, $inner, 'b:CTE b:CTE b:Dependency'];
143            yield $dialect->name . ': a clause of the inserted query does not reach the target' => [$dialect, $insertQuery, 'users:Dependency users:CTE users:CTE'];
144        }
145        $delete = 'WITH users AS (SELECT 9 AS id) DELETE FROM users WHERE id IN (SELECT id FROM users)';
146        $update = "WITH users AS (SELECT 1 AS id) UPDATE users SET v = 'u' WHERE id IN (SELECT id FROM users)";
147        yield 'mysql: the target of DELETE sees the clause' => [MySqlDialect::MySql, $delete, 'users:CTE users:CTE users:CTE'];
148        yield 'mysql: the target of UPDATE sees the clause' => [MySqlDialect::MySql, $update, 'users:CTE users:CTE users:CTE'];
149        foreach ([PostgreSqlDialect::PostgreSql, SqliteDialect::Sqlite] as $dialect) {
150            yield $dialect->name . ': the target of DELETE is a table' => [$dialect, $delete, 'users:CTE users:Dependency users:CTE'];
151            yield $dialect->name . ': the target of UPDATE is a table' => [$dialect, $update, 'users:CTE users:Dependency users:CTE'];
152            yield $dialect->name . ': the target of INSERT is a table' => [$dialect, 'WITH users AS (SELECT 9 AS id, 1 AS v) INSERT INTO users SELECT * FROM users', 'users:CTE users:Dependency users:CTE'];
153        }
154        yield 'postgresql: the target of MERGE is a table and its source sees the clause' => [PostgreSqlDialect::PostgreSql, 'WITH users AS (SELECT 7 AS id) MERGE INTO users USING users AS s ON users.id = s.id WHEN MATCHED THEN DO NOTHING', 'users:CTE users:Dependency users:CTE'];
155        yield 'mysql: an expression name compares as a table name' => [MySqlDialect::MySql, 'WITH B AS (SELECT 1 AS x) SELECT * FROM b', 'B:CTE b:Dependency'];
156        yield 'sqlite: an expression name compares as a table name' => [SqliteDialect::Sqlite, 'WITH B AS (SELECT 1 AS x) SELECT * FROM b', 'B:CTE b:CTE'];
157        yield 'mysql: a qualified name is never an expression' => [MySqlDialect::MySql, 'WITH b AS (SELECT 1 AS x) SELECT * FROM other.b', 'b:CTE other.b:Dependency'];
158    }

217    public function testResolveAppliesDropsAndConditionalDeclarationsInOrder(Dialect $dialect): void
218    {
219        $semantics = new Semantics($dialect);
220        $users = $semantics->analyze('CREATE TABLE users (id INTEGER)', []);
221        $again = $semantics->analyze('CREATE TABLE IF NOT EXISTS users (other INTEGER)', [$users]);
222        self::assertSame(ReferenceKind::Dependency, Resolved::of($again)->references[0]->kind);
223        self::assertTrue(Resolved::of($again)->references[0]->conditional);
224        self::assertSame([], Resolved::of($again)->declarations);
225        $drop = $semantics->analyze('DROP TABLE IF EXISTS users', [$users]);
226        self::assertSame(ReferenceKind::Drop, Resolved::of($drop)->references[0]->kind);
227        self::assertSame($users, Resolved::of($drop)->references[0]->declaration);
228        $missing = $semantics->analyze('DROP TABLE IF EXISTS users', [$users, $drop]);
229        self::assertNull(Resolved::of($missing)->references[0]->declaration);
230        $fresh = $semantics->analyze('CREATE TABLE users (renewed INTEGER)', [$users, $drop]);
231        self::assertSame(ReferenceKind::Declaration, Resolved::of($fresh)->references[0]->kind);
232        self::assertSame('renewed', Resolved::of($fresh)->declarations[0]->columns[0]->name);
233        $this->expectException(SemanticException::class);
234        $this->expectExceptionMessage('No dependency declares the table users');
235        $semantics->analyze('SELECT id FROM users', [$users, $drop]);
236    }
