30/**
31 * Derives the facts of INSERT and REPLACE.
32 *
33 * Rule: MYSQL-INSERT-001. The written table is the one visible relation of
34 * the columns, the rows and the assignments; a value may refer to a column
35 * of the table (insert.html: a value may refer to a column set earlier in
36 * the row). The written columns are the column list, or every column of the
37 * table; they must be distinct. Each row of VALUES has one value per written
38 * column, except an empty row without a column list, which writes the
39 * defaults; rows also agree with each other, an empty row included. A query source is derived where
40 * the table is not visible and must return one column per written column.
41 * ON DUPLICATE KEY UPDATE assigns columns of the written table; its values
42 * see the written table, the row alias (MYSQL-DML-ROW-ALIAS-001), whose
43 * name may not be the table name, and, for a query source that is one query
44 * block without GROUP BY, the relations of its FROM clause. Generated
45 * columns take only DEFAULT (MYSQL-GENERATED-WRITE-001); a row or a query
46 * whose count does not match is reported for the count only, as the
47 * server stops there. A qualified star
48 * in a 5.x column list is reported as the server's unknown column '*'.
49 * Statements return no rows. Terminates: one pass over finite lists; the
50 * source query is derived once. Source:
51 * https://dev.mysql.com/doc/refman/8.4/en/insert.html,
52 * https://dev.mysql.com/doc/refman/8.4/en/insert-select.html,
53 * https://dev.mysql.com/doc/refman/8.4/en/insert-on-duplicate.html,
54 * https://dev.mysql.com/doc/refman/8.4/en/replace.html. Status: Implemented.
55 *
56 * @visibility SqlSemantics\Platform\MySql
57 */

61     * Derives INSERT ... VALUES.
62     */
63    public function rows(InsertRows $insert, Derivation $derivation, Environment $outer): void
64    {
65        [$target, $written] = $this->open($insert->into, $derivation, $outer);
66        $environment = new Environment($derivation->context, $outer, [$target]);
67        $scope = new WriteScope();
68        $generated = new GeneratedWrites();
69        $table = $generated->table($insert->into->table, $derivation);
70        $width = null;
71        foreach ($insert->rows as $index => $row) {
72            $count = count($row->values);
73            $expected = $written === null ? $width : count($written);
74            $mismatch = null;
75            $columns = $count === 0 && $insert->into->columns === null ? null : $expected;
76            $against = $columns !== null && $count !== $columns ? $columns : ($width !== null && $count !== $width ? $width : null);
77            if ($against !== null) {
78                $mismatch = new ValueCountMismatch($against, $count, $index + 1);
79                $derivation->report($mismatch);
80            }
81            $width ??= $count;
82            foreach ($row->values as $position => $value) {
83                $column = $this->column($written, $position, $target, $mismatch);
84                $scope->value($value, $column, $derivation, $environment);
85                $generated->value($column, $value, $mismatch === null ? $table : null, $derivation);
86            }
87        }
88        $this->duplicates($insert->onDuplicate, $insert->alias, $insert->into, $target, $written, [], $derivation, $outer);
89    }

104     * Derives INSERT ... SELECT, TABLE or VALUES.
105     */
106    public function query(InsertQuery $insert, Derivation $derivation, Environment $outer): void
107    {
108        [$target, $written] = $this->open($insert->into, $derivation, $outer);
109        $rows = $derivation->query($insert->source, $outer);
110        if ($written !== null && $rows->shape->complete() && count($rows->shape->slots) !== count($written)) {
111            $derivation->report(new ValueCountMismatch(count($written), count($rows->shape->slots)));
112        } elseif ($written !== null) {
113            $generated = new GeneratedWrites();
114            $generated->query($written, $insert->source, $rows, $generated->table($insert->into->table, $derivation), $derivation);
115        }
116        $sources = $insert->onDuplicate === [] ? [] : (new SourceRelations())->visible($insert->source, $derivation);
117        $this->duplicates($insert->onDuplicate, null, $insert->into, $target, $written, $sources, $derivation, $outer);
118    }

120    /**
121     * Derives the written table and the written columns.
122     *
123     * @return array{VisibleRelation, list<Field>|null} The table as a visible relation, and the written columns, or null when the table is not declared completely and no column list is written
124     */
125    public function open(InsertInto $into, Derivation $derivation, Environment $outer): array
126    {
127        $fact = $derivation->relation($into->table, $outer);
128        $target = new VisibleRelation($into->table, $fact->shape, null, $into->table->name);
129        if ($into->columns === null) {
130            if (!$fact->shape->complete()) {
131                return [$target, null];
132            }
133            $fields = [];
134            foreach ($fact->shape->slots as $position => $slot) {
135                $fields[] = new Field($position, new OutputSlot($slot->name, $slot->type, $slot->nullability, null, $slot));
136            }
137
138            return [$target, $fields];
139        }
140        $environment = new Environment($derivation->context, $outer, [$target]);
141        $scope = new WriteScope();
142        $fields = [];
143        foreach ($into->columns->columns as $position => $column) {
144            if ($column instanceof ColumnUse) {
145                $fields[] = $scope->field($position, $column, $derivation->scalar($column, $environment));
146                continue;
147            }
148            $problem = new WriteMisuse(WriteRule::WildcardColumn);
149            $derivation->report($problem);
150            $fields[] = new Field($position, new OutputSlot(null, new Invalid($problem), Nullability::Dependent));
151        }
152        $scope->distinct($fields, $derivation);
153
154        return [$target, $fields];
155    }

157    /**
158     * Answers the field a value at a position of a row is assigned to.
159     *
160     * @param list<Field>|null $written
161     */
162    public function column(?array $written, int $position, VisibleRelation $target, ?Diagnostic $mismatch): Field
163    {
164        if ($written !== null && isset($written[$position])) {
165            return $written[$position];
166        }
167        if ($written === null) {
168            return new Field($position, new OutputSlot(null, new Dependent($target->shape->missing), Nullability::Dependent));
169        }
170        Check::invariant($mismatch !== null, 'A value beyond the written columns is a reported count mismatch.');
171
172        return new Field($position, new OutputSlot(null, new Invalid($mismatch), Nullability::Dependent));
173    }

175    /**
176     * Derives the row alias and the assignments of ON DUPLICATE KEY UPDATE.
177     *
178     * @param list<Assignment> $assignments
179     * @param list<Field>|null $written
180     * @param list<VisibleRelation> $sources The relations of the FROM clause of a query source
181     */
182    public function duplicates(array $assignments, ?RowAlias $alias, InsertInto $into, VisibleRelation $target, ?array $written, array $sources, Derivation $derivation, Environment $outer): void
183    {
184        $visible = [$target];
185        if ($alias !== null) {
186            if ($derivation->context->relationNames->equal($alias->name->value, $into->table->name->name->value)) {
187                $derivation->report(new Misuse(MisuseRule::DuplicateAlias));
188            }
189            $environment = $written === null ? new Environment($derivation->context, $outer, [$target]) : new Environment($derivation->context, $outer, [], [], $written);
190            $visible[] = new VisibleRelation($alias, $derivation->relation($alias, $environment)->shape, $alias->name);
191        }
192        if ($assignments === []) {
193            return;
194        }
195        $fields = (new WriteScope())->assign($assignments, $derivation, new Environment($derivation->context, $outer, [$target]), new Environment($derivation->context, $outer, [...$visible, ...$sources]), false);
196        (new GeneratedWrites())->assignments($assignments, $fields, $derivation);
197    }
