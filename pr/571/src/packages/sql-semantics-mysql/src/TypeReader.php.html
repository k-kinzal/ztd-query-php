19/**
20 * Reads a type declaration by the lexer's type keyword tokens, so every synonym the lexer knows is covered.
21 *
22 * The keyword token identifies the type; the parenthesized arguments give
23 * its length or precision and scale, or its enumeration members; the
24 * attributes after them give the sign, zero padding, character set and
25 * binary collation facts. A character type declared with the binary
26 * character set is the corresponding octet string type.
27 *
28 * @visibility SqlSemantics

88    /**
89     * Reads one type declaration; SERIAL declares a nonnullable, automatically generated, unique BIGINT UNSIGNED.
90     *
91     * @throws SemanticException When the declaration is outside the lexer's type vocabulary or its arguments are invalid
92     */
93    public function read(Node $node, ValueReader $values): TypeDeclaration
94    {
95        $tokens = $node->tokens();
96        if ($tokens === []) {
97            Tree::unsupported($node, 'type declaration');
98        }
99        if ($tokens[0]->name === 'SERIAL_SYM') {
100            return new TypeDeclaration(new TypeDescriptor(Builtin::BigInt, unsigned: true), autoIncrement: true, notNull: true, unique: true);
101        }
102        $leading = $this->leading($tokens);
103        $kind = $this->kind($leading, $node);
104        $facts = $this->attributes(array_slice($tokens, count($leading)));
105        if (in_array($tokens[0]->name, ['NCHAR_SYM', 'NVARCHAR_SYM', 'NATIONAL_SYM'], true)) {
106            $facts['characterSet'] = 'utf8mb3';
107        }
108        $group = TokenGroups::parentheses($tokens)[0] ?? [];
109        $members = [];
110        if (in_array($kind, [Builtin::Enum, Builtin::Set], true)) {
111            $members = array_map($values->read(...), Tree::outer($node, ['text_string']));
112            $group = [];
113        }
114        [$kind, $length, $precision, $scale] = $this->arguments($kind, array_map(fn (array $argument): int => $this->integer($argument, $node), Numbers::arguments($group)));
115        if (in_array($tokens[0]->name, ['BOOL_SYM', 'BOOLEAN_SYM'], true)) {
116            $length = 1;
117        }
118        if ($facts['octets'] && $kind->isCharacter()) {
119            $kind = self::OCTETS[$kind->value] ?? $kind;
120        }
121        $character = $kind->isCharacter();
122
123        return new TypeDeclaration(new TypeDescriptor($kind, $length, $precision, $scale, $kind->isNumeric() && $facts['unsigned'], $kind->isNumeric() && $facts['zerofill'], $character && $facts['binary'], $character ? $facts['characterSet'] : null, $members, effectiveNumericSize: $this->numericSize($kind, $precision, $scale, $node)));
124    }

126    /**
127     * Identifies the type from the keyword tokens that spell its name.
128     *
129     * @param list<Token> $leading
130     * @throws SemanticException When the first token is not a type keyword
131     */
132    public function kind(array $leading, Node $source): Builtin
133    {
134        $kind = self::KINDS[$leading[0]->name ?? ''] ?? null;
135        if ($kind === null) {
136            Tree::unsupported($source, 'type declaration');
137        }
138        $names = array_map(static fn (Token $token): string => $token->name, $leading);
139        if ($kind === Builtin::Char && array_intersect($names, ['VARYING', 'VARCHAR_SYM', 'VARCHAR']) !== []) {
140            return Builtin::VarChar;
141        }
142        if ($leading[0]->name === 'LONG_SYM' && array_intersect($names, ['VARBINARY_SYM', 'VARBINARY']) !== []) {
143            return Builtin::MediumBlob;
144        }
145
146        return $kind;
147    }

149    /**
150     * Selects the keyword tokens that spell the type name, before any argument or attribute.
151     *
152     * @param list<Token> $tokens
153     * @return list<Token>
154     */
155    public function leading(array $tokens): array
156    {
157        $leading = array_slice($tokens, 0, 1);
158        foreach (array_slice($tokens, 1) as $index => $token) {
159            $next = strtoupper($tokens[$index + 2]->text ?? '');
160            if (!in_array($token->name, self::NAME_TOKENS, true) || ($token->name === 'CHAR_SYM' && $next === 'SET')) {
161                break;
162            }
163            $leading[] = $token;
164        }
165
166        return $leading;
167    }

169    /**
170     * Reads the sign, padding, character set and collation attributes written after the type name.
171     *
172     * @param list<Token> $tokens
173     * @return array{unsigned: bool, zerofill: bool, binary: bool, octets: bool, characterSet: ?string}
174     */
175    public function attributes(array $tokens): array
176    {
177        $facts = ['unsigned' => false, 'zerofill' => false, 'binary' => false, 'octets' => false, 'characterSet' => null];
178        $depth = 0;
179        for ($index = 0; $index < count($tokens); ++$index) {
180            $token = $tokens[$index];
181            $depth += $token->text === '(' ? 1 : ($token->text === ')' ? -1 : 0);
182            if ($depth > 0 || in_array($token->text, ['(', ')'], true)) {
183                continue;
184            }
185            $word = strtoupper($token->text);
186            if ($word === 'CHARSET' || (in_array($word, ['CHARACTER', 'CHAR'], true) && strtoupper($tokens[$index + 1]->text ?? '') === 'SET')) {
187                $index += $word === 'CHARSET' ? 1 : 2;
188                $facts['characterSet'] = strtolower((new NameRules())->name($tokens[$index]));
189                $facts['octets'] = $facts['characterSet'] === 'binary';
190                continue;
191            }
192            match ($word) {
193                'UNSIGNED' => $facts['unsigned'] = true,
194                'ZEROFILL' => $facts['zerofill'] = $facts['unsigned'] = true,
195                'SIGNED' => null,
196                'BINARY' => $facts['binary'] = true,
197                'ASCII' => $facts['characterSet'] = 'latin1',
198                'UNICODE' => $facts['characterSet'] = 'ucs2',
199                'BYTE' => $facts['octets'] = true,
200                default => Tree::unsupported($token, 'type attribute'),
201            };
202        }
203
204        return $facts;
205    }

207    /**
208     * Assigns the numeric arguments to the slots the type gives them; FLOAT with more than 24 bits of precision is a DOUBLE.
209     *
210     * @param list<int> $arguments
211     * @return array{Builtin, ?int, ?int, ?int} Kind, length, precision and scale
212     */
213    public function arguments(Builtin $kind, array $arguments): array
214    {
215        if ($arguments === []) {
216            return [$kind, null, null, null];
217        }
218        if (in_array($kind, [Builtin::Numeric, Builtin::Real, Builtin::DoublePrecision], true)) {
219            if ($kind === Builtin::Real && count($arguments) === 1 && $arguments[0] > 24) {
220                $kind = Builtin::DoublePrecision;
221            }
222            return [$kind, null, $arguments[0], $arguments[1] ?? null];
223        }
224        if ($kind->isTemporal()) {
225            return [$kind, null, $arguments[0], null];
226        }
227
228        return [$kind, $arguments[0], null, null];
229    }

235    public function integer(array $tokens, Node $source): int
236    {
237        $value = Numbers::integer($tokens);
238        if ($value === null || $value < 0) {
239            throw new SemanticException('invalid-type-modifier', 'Invalid type argument: ' . Tree::text($source), $source);
240        }

242        return $value;

245    /**
246     * Resolves decimal defaults and validates the effective size before constructing it.
247     * @throws SemanticException When precision or scale is outside the server's range
248     */
249    public function numericSize(Builtin $kind, ?int $precision, ?int $scale, Node $source): ?NumericSize
250    {
251        if ($kind !== Builtin::Numeric) {
252            return null;
253        }
254        $precision = $precision === null || $precision === 0 ? 10 : $precision;
255        $scale ??= 0;
256        if ($precision < 1 || $precision > 65 || $scale < 0 || $scale > 30 || $scale > $precision) {
257            throw new SemanticException('invalid-type-modifier', 'Invalid decimal precision or scale.', $source);
258        }
259        return new NumericSize($precision, $scale);
260    }
