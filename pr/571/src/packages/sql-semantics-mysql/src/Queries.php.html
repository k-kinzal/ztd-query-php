17/**
18 * Composes MySQL queries: set operations, common table expressions, and the WITH clause of the 8.0 query grammar.
19 *
20 * A query given here may be the command of an analyzed statement; its
21 * terminator envelope is removed. A composed query is a complete query
22 * expression, so it is again a command and a subquery. Releases before 8.0
23 * chain unions as LegacyUnions does and have no common table expressions.
24 *
25 * @visibility SqlSemantics
26 */
27trait Queries
28{
29    /**
30     * `UNION ALL` of two queries, each a complete SELECT of the release.
31     */
32    public function unionAll(Element $left, Element $right): Element
33    {
34        $left = $this->unwrap($left);
35        $right = $this->unwrap($right);
36        if ($this->legacy()) {
37            return $this->legacyUnion($left, $right);
38        }
39        $symbols = ['query_expression_body', 'UNION_SYM', 'union_option', 'query_expression_body'];
40        $body = $this->form('query_expression_body', $symbols, [$this->body($left), $this->form('union_option', ['ALL'], []), $this->body($right)]);
41
42        return $this->form('query_expression', ['query_expression_body', 'opt_order_clause', 'opt_limit_clause'], [$body, $this->form('opt_order_clause', [], []), $this->form('opt_limit_clause', [], [])]);
43    }

45    /**
46     * A common table expression; releases before 8.0 have none.
47     */
48    public function cte(string $name, Element $query, array $columns = []): Element
49    {
50        if ($this->legacy()) {
51            throw new CompositionException('No common table expression in ' . $this->language->version);
52        }
53        $query = $this->unwrap($query);
54        $names = array_map(fn (string $column): IdentForm => $this->identifier($column), $columns);
55        $list = $columns === [] ? $this->form('opt_derived_column_list', [], []) : $this->form('opt_derived_column_list', ['(', 'simple_ident_list', ')'], [$this->fold('simple_ident_list', ['simple_ident_list', ',', 'ident'], $names)]);
56
57        return $this->form('common_table_expr', ['ident', 'opt_derived_column_list', 'AS', 'table_subquery'], [$this->identifier($name), $list, $this->subquery($query)]);
58    }

60    /**
61     * A query preceded by a WITH clause; a query that has one keeps its expressions after the new ones.
62     */
63    public function with(array $ctes, Element $query): Element
64    {
65        if ($this->legacy()) {
66            throw new CompositionException('No common table expression in ' . $this->language->version);
67        }
68        if ($ctes === []) {
69            throw new CompositionException('A WITH clause needs at least one common table expression.');
70        }
71        foreach ($ctes as $cte) {
72            $this->expect($cte, 'common_table_expr', 'A common table expression');
73        }
74        $query = $this->unwrap($query);
75        $locking = $this->classOf('select_stmt', ['query_expression', 'locking_clause_list']);
76        if ($locking !== null && $query instanceof $locking) {
77            return $query->map(fn (Element $child): Element => $child instanceof QueryExpressionForm ? $this->with($ctes, $child) : $child);
78        }
79        $plain = ['query_expression_body', 'opt_order_clause', 'opt_limit_clause'];
80        $withClause = ['with_clause', 'query_expression_body', 'opt_order_clause', 'opt_limit_clause'];
81        $plainClass = $this->classOf('query_expression', $plain);
82        $withClass = $this->classOf('query_expression', $withClause);
83        if ($plainClass !== null && $query instanceof $plainClass) {
84            return $this->form('query_expression', $withClause, [$this->withClause($ctes, null), ...$query->children()]);
85        }
86        if ($withClass !== null && $query instanceof $withClass) {
87            $children = $query->children();
88            $existing = array_shift($children);
89            assert($existing instanceof WithClauseForm);
90
91            return $this->form('query_expression', $withClause, [$this->withClause($ctes, $existing), ...$children]);
92        }
93
94        throw new CompositionException('A WITH clause precedes a query expression of ' . $this->language->version . ', ' . $query::class . ' given.');
95    }

97    /**
98     * Answers a query as a set operand: its body when it has no ORDER BY or LIMIT, otherwise parenthesized.
99     */
100    protected function body(Element $query): Element
101    {
102        if ($query instanceof QueryExpressionBodyForm) {
103            return $query;
104        }
105        $plainClass = $this->classOf('query_expression', ['query_expression_body', 'opt_order_clause', 'opt_limit_clause']);
106        if ($plainClass !== null && $query instanceof $plainClass) {
107            [$body, $order, $limit] = $query->children();
108            if (Writer::render($order) === '' && Writer::render($limit) === '') {
109                return $body;
110            }
111        }
112
113        return $this->subquery($query);
114    }

116    /**
117     * Parenthesizes a complete query, as a subquery or a set operand.
118     */
119    protected function subquery(Element $query): Element
120    {
121        $locking = $this->classOf('select_stmt', ['query_expression', 'locking_clause_list']);
122        if ($locking !== null && $query instanceof $locking) {
123            [$expression, $clauses] = $query->children();
124            $query = $this->form('query_expression_with_opt_locking_clauses', ['query_expression', 'locking_clause_list'], [$expression, $clauses]);
125        }
126
127        return $this->form('query_expression_parens', ['(', 'query_expression_with_opt_locking_clauses', ')'], [$this->expect($query, 'query_expression_with_opt_locking_clauses', 'A subquery')]);
128    }

130    /**
131     * Builds a WITH clause from new expressions and the ones of an existing clause.
132     *
133     * @param list<Element> $ctes
134     *
135     * @throws CompositionException When there is no expression
136     */
137    protected function withClause(array $ctes, ?WithClauseForm $existing): Element
138    {
139        $listSymbols = ['with_list', ',', 'common_table_expr'];
140        $recursive = false;
141        if ($existing !== null) {
142            $recursive = $existing instanceof ($this->classOf('with_clause', ['WITH', 'RECURSIVE_SYM', 'with_list']) ?? '');
143            $ctes = [...$ctes, ...$this->unfold($existing->children()[0], 'with_list', $listSymbols)];
144        }
145        $list = $this->fold('with_list', $listSymbols, $ctes);
146
147        return $recursive ? $this->form('with_clause', ['WITH', 'RECURSIVE_SYM', 'with_list'], [$list]) : $this->form('with_clause', ['WITH', 'with_list'], [$list]);
148    }
