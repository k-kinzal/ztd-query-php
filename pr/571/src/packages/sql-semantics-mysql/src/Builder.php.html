21/**
22 * Composes MySQL values under stable names, for every shipped release and the session's `sql_mode`.
23 *
24 * Names are quoted with backticks when the release reads them as keywords
25 * that cannot be identifiers or when their characters need it; after a `.`,
26 * the lexer reads any word as a name, so a qualifier's parts are spelled
27 * bare more often. Strings are escaped as the mode reads them: backslashes
28 * are doubled unless the mode has `NO_BACKSLASH_ESCAPES`. Under
29 * `HIGH_NOT_PRECEDENCE`, a negation binds as tightly as `!`, so its operand
30 * is parenthesized as the server needs. Common table expressions need
31 * release 8.0 or later.
32 *
33 * @visibility public
34 * @example Composing a condition
35 *     $builder = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\MySql\Dialect::MySql))->builder();
36 *     \SqlSemantics\Statement\Writer::render($builder->and($builder->compare($builder->column('select'), '=', $builder->string("a'b")), $builder->boolean(true))) // => "`select` = 'a''b' AND TRUE"
37 * @example Shadowing a table with a common table expression
38 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\MySql\Dialect::MySql);
39 *     $builder = $semantics->builder();
40 *     $rows = $builder->unionAll($semantics->analyze('SELECT 1 AS id')->command, $semantics->analyze('SELECT 2')->command);
41 *     \SqlSemantics\Statement\Writer::render($builder->with([$builder->cte('users', $rows)], $semantics->analyze('SELECT id FROM users')->command)) // => 'WITH users AS( SELECT 1 AS id UNION ALL SELECT 2 ) SELECT id FROM users'
42 */
43final class Builder extends Composition

168    /**
169     * `NOT`, whose operand is parenthesized as the mode's precedence of NOT needs.
170     */
171    public function not(Element $operand): ExprForm
172    {
173        if ($this->mode()->highNotPrecedence) {
174            $symbols = ['not2', 'simple_expr'];
175            $value = $this->form('simple_expr', $symbols, [$this->form('not2', ['NOT2_SYM'], ['NOT']), $this->operand($operand, 'simple_expr', $symbols, 1)]);
176        } else {
177            $value = $this->form('expr', ['NOT_SYM', 'expr'], [$this->operand($operand, 'expr', ['NOT_SYM', 'expr'], 1)]);
178        }
179        assert($value instanceof ExprForm);
180
181        return $value;
182    }

184    /**
185     * A comparison; `<=>` is the NULL-safe equality of MySQL.
186     */
187    public function compare(Element $left, string $operator, Element $right): ExprForm
188    {
189        $terminal = self::COMPARISONS[$operator] ?? throw new CompositionException('Unknown comparison operator ' . $operator);
190        if (!$this->has('comp_op', [$terminal])) {
191            $symbols = ['bool_pri', $terminal, 'predicate'];
192            $value = $this->form('bool_pri', $symbols, [$this->operand($left, 'bool_pri', $symbols, 0), $this->operand($right, 'bool_pri', $symbols, 2)]);
193            assert($value instanceof ExprForm);
194
195            return $value;
196        }
197        $comparison = $this->form('comp_op', [$terminal], $terminal === 'NE' ? [$operator] : []);
198
199        return $this->infix('bool_pri', ['bool_pri', 'comp_op', 'predicate'], $left, $comparison, $right);
200    }
