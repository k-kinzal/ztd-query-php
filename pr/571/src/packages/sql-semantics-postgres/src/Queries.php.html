17/**
18 * Composes PostgreSQL queries: set operations, common table expressions, and the WITH clause of a SELECT.
19 *
20 * A query given here may be the command of an analyzed statement; its
21 * terminator envelope is removed. A composed query is a complete SELECT
22 * statement, so it is again a command and a preparable statement.
23 *
24 * @visibility SqlSemantics
25 */
26trait Queries
27{
28    /**
29     * `UNION ALL` of two queries; a query with ORDER BY, LIMIT, locking, or WITH is parenthesized.
30     */
31    public function unionAll(Element $left, Element $right): SelectStmtForm
32    {
33        $value = $this->form('simple_select', ['select_clause', 'UNION', 'set_quantifier', 'select_clause'], [$this->setOperand($this->unwrap($left)), $this->form('set_quantifier', ['ALL'], []), $this->setOperand($this->unwrap($right))]);
34        assert($value instanceof SelectStmtForm);
35
36        return $value;
37    }

39    /**
40     * A common table expression over a SELECT, INSERT, UPDATE, DELETE, or MERGE.
41     */
42    public function cte(string $name, Element $query, array $columns = []): CommonTableExprForm
43    {
44        $names = array_map(fn (string $column): ColIdForm => $this->identifier($column), $columns);
45        $list = $columns === [] ? $this->form('opt_name_list', [], []) : $this->form('opt_name_list', ['(', 'name_list', ')'], [$this->fold('name_list', ['name_list', ',', 'name'], $names)]);
46        $value = $this->form('common_table_expr', ['name', 'opt_name_list', 'AS', 'opt_materialized', '(', 'PreparableStmt', ')', 'opt_search_clause', 'opt_cycle_clause'], [
47            $this->identifier($name),
48            $list,
49            $this->form('opt_materialized', [], []),
50            $this->expect($this->unwrap($query), 'PreparableStmt', 'The query of a common table expression'),
51            $this->form('opt_search_clause', [], []),
52            $this->form('opt_cycle_clause', [], []),
53        ]);
54        assert($value instanceof CommonTableExprForm);

59    /**
60     * A SELECT preceded by a WITH clause; a SELECT that has one keeps its expressions after the new ones.
61     */
62    public function with(array $ctes, Element $query): SelectStmtForm
63    {
64        if ($ctes === []) {
65            throw new CompositionException('A WITH clause needs at least one common table expression.');
66        }
67        foreach ($ctes as $cte) {
68            $this->expect($cte, 'common_table_expr', 'A common table expression');
69        }
70        $query = $this->unwrap($query);
71        if ($query instanceof SelectClauseForm) {
72            $value = $this->form('select_no_parens', ['with_clause', 'select_clause'], [$this->withClause($ctes, null), $query]);
73            assert($value instanceof SelectStmtForm);
74
75            return $value;
76        }
77        foreach ($this->selectShapes() as $symbols) {
78            $class = $this->classOf('select_no_parens', $symbols);
79            if ($class === null || !$query instanceof $class) {
80                continue;
81            }
82            $children = $query->children();
83            if ($symbols[0] === 'with_clause') {
84                $existing = array_shift($children);
85                assert($existing instanceof WithClauseForm);
86                $value = $this->form('select_no_parens', $symbols, [$this->withClause($ctes, $existing), ...$children]);
87            } else {
88                $value = $this->form('select_no_parens', ['with_clause', ...$symbols], [$this->withClause($ctes, null), ...$children]);
89            }
90            assert($value instanceof SelectStmtForm);
91
92            return $value;
93        }
94
95        throw new CompositionException('A WITH clause precedes a SELECT of ' . $this->language->version . ', ' . $query::class . ' given.');
96    }

98    /**
99     * Answers a query as a set operand: a plain SELECT as is, any other SELECT parenthesized.
100     */
101    protected function setOperand(Element $query): Element
102    {
103        if ($query instanceof SelectClauseForm) {
104            return $query;
105        }
106
107        return $this->form('select_with_parens', ['(', 'select_no_parens', ')'], [$this->expect($query, 'select_no_parens', 'A set operand')]);
108    }

110    /**
111     * Lists the symbol shapes of select_no_parens that take a WITH clause, the ones that have it first.
112     *
113     * @return list<list<string>>
114     */
115    protected function selectShapes(): array
116    {
117        $shapes = [
118            ['select_clause', 'sort_clause'],
119            ['select_clause', 'opt_sort_clause', 'for_locking_clause', 'opt_select_limit'],
120            ['select_clause', 'opt_sort_clause', 'select_limit', 'opt_for_locking_clause'],
121        ];
122        $withShapes = array_map(static fn (array $shape): array => ['with_clause', ...$shape], [['select_clause'], ...$shapes]);
123
124        return [...$withShapes, ...$shapes];
125    }
126
127    /**
128     * Builds a WITH clause from new expressions and the ones of an existing clause.
129     *
130     * @param list<Element> $ctes
131     *
132     * @throws CompositionException When there is no expression
133     */
134    protected function withClause(array $ctes, ?WithClauseForm $existing): Element
135    {
136        $listSymbols = ['cte_list', ',', 'common_table_expr'];
137        $recursive = false;
138        if ($existing !== null) {
139            $recursive = $existing instanceof ($this->classOf('with_clause', ['WITH', 'RECURSIVE', 'cte_list']) ?? '');
140            $children = $existing->children();
141            $ctes = [...$ctes, ...$this->unfold($children[count($children) - 1], 'cte_list', $listSymbols)];
142        }
143        $list = $this->fold('cte_list', $listSymbols, $ctes);
144
145        return $recursive ? $this->form('with_clause', ['WITH', 'RECURSIVE', 'cte_list'], [$list]) : $this->form('with_clause', ['WITH', 'cte_list'], [$list]);
146    }
