21/**
22 * Composes PostgreSQL values under stable names.
23 *
24 * A name is written bare only when the release reads it back as the same
25 * name: it is quoted when it holds an upper-case letter, which the server
26 * would fold, when its characters need it, or when the grammar reads it as
27 * a keyword that cannot be a column name. Strings are standard-conforming:
28 * a quote is doubled and a backslash is an ordinary character.
29 *
30 * @visibility public
31 * @example Composing a condition
32 *     $builder = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\PostgreSql\Dialect::PostgreSql))->builder();
33 *     \SqlSemantics\Statement\Writer::render($builder->or($builder->compare($builder->column('t', 'User'), '<>', $builder->string('x')), $builder->null())) // => "t.\"User\" <> 'x' OR NULL"
34 * @example Shadowing a table with a common table expression
35 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\PostgreSql\Dialect::PostgreSql);
36 *     $builder = $semantics->builder();
37 *     $rows = $builder->unionAll($semantics->analyze('SELECT 1 AS id')->command, $semantics->analyze('SELECT 2')->command);
38 *     \SqlSemantics\Statement\Writer::render($builder->with([$builder->cte('users', $rows, ['id'])], $semantics->analyze('SELECT id FROM users')->command)) // => 'WITH users ( id ) AS( SELECT 1 AS id UNION ALL SELECT 2 ) SELECT id FROM users'
39 */
