68    /**
69     * Reads one Typename; a set-returning type is not a column type.
70     *
71     * A declaration the database rejects, such as a SETOF column or a type
72     * modifier that is not a constant, is invalid rather than unsupported, so
73     * it is an error and never an unreadable declaration.
74     *
75     * @throws SemanticException When the declaration is outside the modeled surface or invalid
76     */
77    public function read(Node $node): TypeDeclaration
78    {
79        if (strtoupper($node->tokens()[0]->text ?? '') === 'SETOF') {
80            throw new SemanticException('invalid-column-type', 'A column cannot be declared SETOF: ' . Tree::text($node), $node);
81        }
82        $simple = Tree::outer($node, ['SimpleTypename'])[0] ?? null;
83        $inner = $simple === null ? null : (Tree::significant($simple)[0] ?? null);
84        if (!$inner instanceof Node) {
85            Tree::unsupported($node, 'column type');
86        }
87        $facts = ['length' => null, 'precision' => null, 'scale' => null, 'fields' => null, 'autoIncrement' => false];
88        $name = match ($inner->name) {
89            'Numeric' => $this->numeric($inner, $facts),
90            'Bit' => $this->bit($inner, $facts),
91            'Character' => $this->character($inner, $facts),
92            'ConstDatetime' => $this->datetime($inner, $facts),
93            'ConstInterval' => $this->interval($simple, $facts),
94            'JsonType' => Builtin::Json,
95            'GenericType' => $this->generic($inner, $facts),
96            default => Tree::unsupported($node, 'column type'),
97        };
98        $type = new TypeDescriptor($name, $facts['length'], $facts['precision'], $facts['scale'], arrayDimensions: $this->dimensions($node), intervalFields: $facts['fields'], effectiveNumericSize: $this->numericSize($name, $facts['precision'], $facts['scale'], $node));
99
100        return new TypeDeclaration($type, autoIncrement: $facts['autoIncrement'], notNull: $facts['autoIncrement']);
101    }

103    /**
104     * Counts declared array dimensions; the word ARRAY without brackets declares one.
105     */
106    public function dimensions(Node $node): int
107    {
108        $brackets = 0;
109        $word = false;
110        foreach ($node->children as $child) {
111            $brackets += $child instanceof Node && $child->name === 'opt_array_bounds' ? count(array_filter($child->tokens(), static fn (Token $token): bool => $token->text === '[')) : 0;
112            $brackets += $child instanceof Token && $child->text === '[' ? 1 : 0;
113            $word = $word || ($child instanceof Token && strtoupper($child->text) === 'ARRAY');
114        }
115
116        return $brackets > 0 ? $brackets : ($word ? 1 : 0);
117    }

120     * @param array{length: ?int, precision: ?int, scale: ?int, fields: ?IntervalFields, autoIncrement: bool} $facts
121     */
122    public function numeric(Node $node, array &$facts): Builtin
123    {
124        $tokens = $node->tokens();
125        $kind = match ($tokens[0]->name) {
126            'INT_P', 'INTEGER' => Builtin::Integer,
127            'SMALLINT' => Builtin::SmallInt,
128            'BIGINT' => Builtin::BigInt,
129            'REAL' => Builtin::Real,
130            'FLOAT_P' => Builtin::DoublePrecision,
131            'DOUBLE_P' => Builtin::DoublePrecision,
132            'DECIMAL_P', 'DEC', 'NUMERIC' => Builtin::Numeric,
133            'BOOLEAN_P' => Builtin::Boolean,
134            default => Tree::unsupported($node, 'numeric type'),
135        };
136        if ($tokens[0]->name === 'FLOAT_P') {
137            $facts['precision'] = $this->iconst($node);
138            $kind = $facts['precision'] !== null && $facts['precision'] <= 24 ? Builtin::Real : Builtin::DoublePrecision;
139        }
140        if ($kind === Builtin::Numeric) {
141            [$facts['precision'], $facts['scale']] = $this->modifiers($node, 2, $node) + [null, null];
142        }
143
144        return $kind;
145    }

186     * @param array{length: ?int, precision: ?int, scale: ?int, fields: ?IntervalFields, autoIncrement: bool} $facts
187     */
188    public function interval(Node $simple, array &$facts): Builtin
189    {
190        $facts['precision'] = $this->iconst($simple);
191        $words = [];
192        foreach ((Tree::child($simple, ['opt_interval'])?->tokens()) ?? [] as $token) {
193            if (ctype_alpha($token->text) && strtoupper($token->text) !== 'TO') {
194                $words[] = strtolower($token->text);
195            }
196        }
197        $facts['fields'] = $words === [] ? null : IntervalFields::from(implode(' to ', $words));
198
199        return Builtin::Interval;

203     * @param array{length: ?int, precision: ?int, scale: ?int, fields: ?IntervalFields, autoIncrement: bool} $facts
204     */
205    public function generic(Node $node, array &$facts): Builtin|TypeName
206    {
207        $parts = [];
208        foreach ($node->children as $child) {
209            if ($child instanceof Node && in_array($child->name, ['type_function_name', 'attrs'], true)) {
210                foreach ($child->tokens() as $token) {
211                    if ($token->text !== '.') {
212                        $parts[] = (new NameRules())->name($token);
213                    }
214                }
215            }
216        }
217        $catalog = count($parts) === 1 || (count($parts) === 2 && $parts[0] === 'pg_catalog') ? $parts[count($parts) - 1] : null;
218        if ($catalog !== null && isset(self::SERIALS[$catalog])) {
219            $facts['autoIncrement'] = true;
220
221            return self::SERIALS[$catalog];
222        }
223        $kind = $catalog === null ? null : (self::CATALOG[$catalog] ?? null);
224        if ($kind === null) {
225            if ($parts === []) {
226                Tree::unsupported($node, 'type name');
227            }
228
229            return new TypeName($parts);
230        }
231        $this->assign($kind, $this->modifiers($node, $kind === Builtin::Numeric ? 2 : 1, $node), $facts);
232
233        return $kind;
234    }

236    /**
237     * Places catalog type modifiers in the slot the type gives them: precision and scale, seconds precision, or length.
238     *
239     * @param list<int> $modifiers
240     * @param array{length: ?int, precision: ?int, scale: ?int, fields: ?IntervalFields, autoIncrement: bool} $facts
241     */
242    public function assign(Builtin $kind, array $modifiers, array &$facts): void
243    {
244        if ($kind === Builtin::Numeric) {
245            [$facts['precision'], $facts['scale']] = $modifiers + [null, null];
246        } elseif ($kind->isTemporal()) {
247            $facts['precision'] = $modifiers[0] ?? null;
248        } elseif ($kind->isCharacter() || in_array($kind, [Builtin::Bit, Builtin::BitVarying], true)) {
249            $facts['length'] = $modifiers[0] ?? null;
250        }
251    }

277    /**
278     * Reads the integer constants of a modifier list; other modifier forms are not modeled.
279     *
280     * @return list<int>
281     * @throws SemanticException When a modifier is not an integer constant or there are too many
282     */
283    public function modifiers(Node $node, int $limit, Node $source): array
284    {
285        $list = Tree::outer($node, ['expr_list'])[0] ?? null;
286        $expressions = $list === null ? [] : Tree::outer($list, ['a_expr']);
287        if (count($expressions) > $limit) {
288            throw new SemanticException('invalid-type-modifier', 'Too many type modifiers: ' . Tree::text($source), $source);
289        }
290        $values = [];
291        foreach ($expressions as $index => $expression) {
292            $tokens = $expression->tokens();
293            $value = Numbers::integer($tokens);
294            if ($value === null && count($tokens) === 1 && $tokens[0]->name === 'SCONST') {
295                Tree::unsupported($expression, 'type modifier');
296            }
297            if ($value === null) {
298                throw new SemanticException('invalid-type-modifier', 'Type modifiers must be constants: ' . Tree::text($source), $source);
299            }
300            if ($value < 0 && $index === 0) {
301                throw new SemanticException('invalid-type-modifier', 'Invalid type modifier: ' . Tree::text($source), $source);
302            }
303            $values[] = $value;
304        }
305
306        return $values;
307    }

324     * Resolves the default scale while leaving unconstrained numeric types unconstrained.
325     * @throws SemanticException When precision or scale is outside the server's range
326     */
327    public function numericSize(Builtin|TypeName $kind, ?int $precision, ?int $scale, Node $source): ?NumericSize
328    {
329        if ($kind !== Builtin::Numeric || $precision === null) {
330            return null;
331        }
332        $scale ??= 0;
333        if ($precision < 1 || $precision > 1000 || $scale < -1000 || $scale > 1000) {
334            throw new SemanticException('invalid-type-modifier', 'Invalid numeric precision or scale.', $source);
335        }
336        return new NumericSize($precision, $scale);
