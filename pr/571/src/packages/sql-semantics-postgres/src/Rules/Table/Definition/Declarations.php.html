27/**
28 * Derives the declaration a table definition provides.
29 *
30 * Rule: PG-TABLE-DECLARATION-001. The columns of a table defined by a column
31 * list are, in order: the columns of the INHERITS parents (a column of the
32 * same name in several parents, or in the list, is merged into one), then
33 * each column definition and the columns each LIKE clause copies at its
34 * position. A partition has the columns of its parent; a typed table the
35 * attributes of its type. Each column has the type of PG-COLUMN-TYPE-001. A
36 * column is NOT NULL when it is declared NOT NULL, is a serial or identity
37 * column, belongs to a PRIMARY KEY, or is copied from a NOT NULL column of a
38 * parent, a LIKE source or the partitioned table ("Not-null constraints are
39 * always copied to the new table"); every other column can be NULL. A
40 * column of a type the context cannot identify is declared with the type its
41 * name denotes on the search path. The declaration is complete only up to
42 * the first column whose type name is an error or whose parent, LIKE source
43 * or composite type the context does not declare. A column is generated
44 * when its definition has a generation clause, when it is inherited from a
45 * generated column of a parent or of the partitioned table, or when a LIKE
46 * clause copies a generated column and its options include GENERATED
47 * (INCLUDING GENERATED or INCLUDING ALL, applied in order with the EXCLUDING
48 * options; without it the copy is a regular column). An identity column is
49 * not generated in this sense. Source of the inheritance rule: MergeAttributes,
50 * tablecmds.c; https://www.postgresql.org/docs/17/ddl-generated-columns.html. The system
51 * columns are implicit (PG-SYSTEM-COLUMNS-001). A foreign table is declared as one, every other
52 * definition as a base table (PG-RELATION-KIND-001).
53 * Source: https://www.postgresql.org/docs/17/sql-createtable.html, https://www.postgresql.org/docs/17/ddl-inherit.html.
54 * Termination: one pass over the parents and the elements. Status: Implemented.
55 *
56 * @visibility SqlSemantics\Platform\PostgreSql
57 */
58final class Declarations
59{
60    /**
61     * Builds the declaration of a table definition under its name.
62     */
63    public function table(TableForm $form, Derivation $derivation, QualifiedName $name, RelationKind $kind = RelationKind::BaseTable): Table
64    {
65        $set = new ColumnSet();
66        if ($form instanceof ListedColumns) {
67            $this->listed($form, $derivation, $set);
68        } elseif ($form instanceof PartitionOf) {
69            $this->copy($form->parent->name, $derivation, $set, null);
70        } else {
71            $set->close();
72        }
73        foreach ($form->elements() as $element) {
74            $this->constrain($element, $set);
75        }
76
77        return new Table($name, $derivation->context->profile, $set->columns(), (new SystemColumns())->implicit(), !$set->closed(), $kind);
78    }
79
80    /**
81     * Adds the inherited columns, the column definitions and the LIKE columns of a column list.
82     */
83    public function listed(ListedColumns $form, Derivation $derivation, ColumnSet $set): void
84    {
85        foreach ($form->parents as $parent) {
86            $this->copy($parent->name, $derivation, $set, null);
87        }
88        $types = new ColumnTyping();
89        foreach ($form->elements as $element) {
90            if ($element instanceof LikeClause) {
91                $this->copy($element->table, $derivation, $set, $element);
92            } elseif ($element instanceof ColumnDefinition) {
93                $type = $types->descriptor($element->type, $derivation->context);
94                if ($type === null) {
95                    $set->close();
96                }
97                $notNull = $types->serial($element->type) !== null || $this->notNull($element->qualifiers);
98                if ($type !== null && !$set->add($element->name, $type, $notNull, $this->generated($element->qualifiers)) && $notNull) {
99                    $set->require($element->name);
100                }
101            }
102        }
103    }
104
105    /**
106     * Copies the columns of a parent or LIKE source, or closes the set when the context does not declare its columns.
107     *
108     * @param LikeClause|null $like The LIKE clause that copies the columns, or null for a parent
109     */
110    public function copy(QualifiedName $source, Derivation $derivation, ColumnSet $set, ?LikeClause $like): void
111    {
112        $resolution = $derivation->table($source, $derivation->environment());
113        if (!$resolution instanceof DeclaredTable || !$resolution->table->complete) {
114            $set->close();
115
116            return;
117        }
118        $generation = $like !== null && $this->includesGenerated($like);
119        foreach ($resolution->table->columns as $column) {
120            if ($like === null) {
121                $set->inherit($column);
122            } else {
123                $set->add($column->name, $column->type, $column->nullability === \SqlSemantics\Statement\Type\Nullability::NotNull, $generation && $column->generated);
124            }
125        }
126    }
127
128    /**
129     * Tells whether the options of a LIKE clause, applied in order, include the generation expressions.
130     */
131    public function includesGenerated(LikeClause $like): bool
132    {
133        $included = false;
134        foreach ($like->options as $option) {
135            if ($option->kind === LikeOptionKind::Generated || $option->kind === LikeOptionKind::All) {
136                $included = $option->including;
137            }
138        }
139
140        return $included;
141    }
142
143    /**
144     * Tells whether column qualifiers hold a generation clause.
145     *
146     * @param list<Clause> $qualifiers
147     */
148    public function generated(array $qualifiers): bool
149    {
150        foreach ($qualifiers as $qualifier) {
151            if ($qualifier instanceof Generated) {
152                return true;
153            }
154        }
155
156        return false;
157    }
158
159    /**
160     * Applies the NOT NULL facts of column options and primary keys.
161     */
162    public function constrain(Clause $element, ColumnSet $set): void
163    {
164        if ($element instanceof ColumnOptions && $this->notNull($element->qualifiers)) {
165            $set->require($element->name);
166        }
167        if ($element instanceof TablePrimaryKey) {
168            foreach ($element->columns as $column) {
169                $set->require($column);
170            }
171        }
172    }
173
174    /**
175     * Tells whether column qualifiers make the column NOT NULL: NOT NULL, PRIMARY KEY or an identity.
176     *
177     * @param list<Clause> $qualifiers
178     */
179    public function notNull(array $qualifiers): bool
180    {
181        foreach ($qualifiers as $qualifier) {
182            if ($qualifier instanceof NotNull || $qualifier instanceof ColumnPrimaryKey || $qualifier instanceof Identity) {
183                return true;
184            }
185        }
186
187        return false;
188    }
