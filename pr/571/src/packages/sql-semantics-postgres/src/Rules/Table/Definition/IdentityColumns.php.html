15/**
16 * Reports an identity column whose type is not smallint, integer or bigint.
17 *
18 * Rule: PG-IDENTITY-TYPE-001. The server creates the identity sequence with
19 * the column's type, and the sequence accepts only int2, int4 and int8: any
20 * other built-in type, an array, a type with modifiers that is not one of the
21 * three and an interval with fields are reported ("identity column type must
22 * be smallint, integer, or bigint", init_params in sequence.c). A type known
23 * by name only, a domain over an integer type included, depends on the
24 * catalog and is not reported; an invalid type name is reported where it is
25 * written.
26 * Source: https://www.postgresql.org/docs/17/sql-createtable.html,
27 * https://github.com/postgres/postgres/blob/REL_17_2/src/backend/commands/sequence.c. Status: Implemented.
28 *
29 * @visibility SqlSemantics\Platform\PostgreSql
30 */
31final class IdentityColumns
32{
33    /**
34     * Reports the identity type problem of a column of a declared type, or of no known type when null.
35     */
36    public function check(Derivation $derivation, ?TypeDescriptor $type): void
37    {
38        $admitted = $type === null || $type instanceof NamedOnPath || $type instanceof Undetermined
39            || in_array($type, [Builtin::Int2, Builtin::Int4, Builtin::Int8], true);
40        if (!$admitted) {
41            $derivation->report(new DefinitionProblem(DefinitionRule::IdentityType));
42        }
43    }
