23/**
24 * Maps PostgreSQL tokens to comparison keys for the token correspondence check.
25 *
26 * Rule: PG-LEAF-KEY-001. An identifier, and a keyword read through one of
27 * the keyword-category nonterminals, is keyed by its decoded name. A string
28 * constant is keyed by its decoded value, a bit string by its notation and
29 * digits, a number by its exact value, a parameter by its canonical marker,
30 * an operator by its text. Every other token is keyed by its terminal; the
31 * lookahead variants the parser's token filter introduces (`NOT_LA`,
32 * `WITH_LA`, `WITHOUT_LA`, `NULLS_LA`, `FORMAT_LA`) are the same word as
33 * their base keyword and share its key. The noise positions are the union of
34 * the family tables. Status: Implemented.
35 *
36 * @visibility SqlSemantics
37 */
38final class LeafKeys implements \SqlSemantics\Contract\LeafKeys
39{
40    /**
41     * The base keyword of each lookahead terminal.
42     */
43    private const LOOKAHEAD = ['NOT_LA' => 'NOT', 'WITH_LA' => 'WITH', 'WITHOUT_LA' => 'WITHOUT', 'NULLS_LA' => 'NULLS_P', 'FORMAT_LA' => 'FORMAT'];
44
45    /**
46     * @var array<string, list<int>>|null
47     */
48    private static ?array $noise = null;
49
50    /**
51     * Answers the comparison key of a token, or null when the token is declared noise.
52     */
53    public function key(Token $token, string $signature, int $position): ?string
54    {
55        if ($token->text === '' || in_array($position, $this->noise()[$signature] ?? [], true)) {
56            return null;
57        }
58        if (in_array(substr($signature, 0, (int) strpos($signature, ':')), Keywords::CATEGORIES, true)) {
59            return 'name:' . (new Identifiers())->fold($token->text);
60        }
61
62        return match ($token->name) {
63            'IDENT' => 'name:' . (new Identifiers())->decode($token->text),
64            'SCONST' => 'string:' . (new Strings())->decode($token->text),
65            'BCONST' => 'bits:b' . (new Strings())->digits($token->text),
66            'XCONST' => 'bits:x' . (new Strings())->digits($token->text),
67            'ICONST', 'FCONST' => 'number:' . $this->number($token->text),
68            'PARAM' => 'param:' . (str_starts_with($token->text, '$') ? '$' . (new Numerals())->canonical(substr($token->text, 1)) : $token->text),
69            'Op' => 'op:' . $token->text,
70            default => self::LOOKAHEAD[$token->name] ?? $token->name,
71        };
72    }
73
74    /**
75     * Answers the exact value of a numeric constant token as a key.
76     */
77    public function number(string $text): string
78    {
79        $numerals = new Numerals();
80        if ($numerals->integral($text)) {
81            return $numerals->decimal($text);
82        }
83        [$integer, $fraction, $exponent] = $numerals->parts($text);
84
85        return $integer . '.' . $fraction . 'e' . ($exponent ?? '0');
86    }
87
88    /**
89     * Answers the noise positions of every family by production signature.
90     *
91     * @return array<string, list<int>>
92     */
93    public function noise(): array
94    {
95        return self::$noise ??= LeafNoise::positions() + TypesNoise::positions() + ExpressionNoise::positions() + InvocationNoise::positions()
96            + QueryNoise::positions() + ManipulationNoise::positions() + TableNoise::positions() + CatalogNoise::positions()
97            + RoutineNoise::positions() + AccessNoise::positions() + UtilityNoise::positions();
98    }
99
100    /**
101     * Tells whether a written token is the same terminal as the rendered one.
102     */
103    public function synonymous(Token $rendered, Token $written): bool
104    {
105        return $rendered->name === $written->name;
106    }
107}
