64    {
65        $expression = Tree::outer($attribute, ['expr', 'term'])[0] ?? null;
66        $tokens = $attribute->tokens();
67        if ($expression === null) {
68            $text = (new NameRules())->name($tokens[count($tokens) - 1]);
69            return $values->vocabulary->build($values->vocabulary->form('term', ['STRING']) ?? throw new LogicException('Missing string form.'), ["'" . str_replace("'", "''", $text) . "'"]);
70        }
71        $value = $values->read($expression);
72        $sign = $tokens[1]->name ?? '';
73        if (in_array($sign, ['PLUS', 'MINUS'], true)) {
74            return $values->vocabulary->build($values->vocabulary->form('expr', ['PLUS|MINUS', 'expr']) ?? throw new LogicException('Missing unary form.'), [$tokens[1]->text, $value]);

101    /**
102     * A table key accepts a column, optionally parenthesized or collated, never an expression.
103     * @throws SemanticException When a key contains an expression
104     */
105    public function keyColumn(Node $expression): \SqlParser\Lexer\Token
106    {
107        $children = Tree::significant($expression);
108        if (count($children) === 1 && $children[0] instanceof Node && $children[0]->name === 'term') {
109            return $this->keyColumn($children[0]);
110        }
111        if (count($children) === 1 && $children[0] instanceof \SqlParser\Lexer\Token && in_array($children[0]->name, ['ID', 'STRING'], true)) {
112            return $children[0];
113        }
114        if (count($children) === 3 && $children[0] instanceof Node && $children[1] instanceof \SqlParser\Lexer\Token && $children[1]->name === 'COLLATE') {
115            return $this->keyColumn($children[0]);
116        }
117        if (count($children) === 3 && $children[0] instanceof \SqlParser\Lexer\Token && $children[0]->name === 'LP' && $children[1] instanceof Node) {
118            return $this->keyColumn($children[1]);
119        }
120        throw new SemanticException('invalid-key-expression', 'A table key must name columns, not expressions.', $expression);
121    }
