19/**
20 * Decodes literal syntax without evaluating expressions or converting to a column type.
21 * @visibility SqlSemantics
22 */
23final class LiteralDecoder implements LiteralRules
24{
25    /**
26     * @param non-empty-list<Token> $tokens
27     * @throws DecodingException When the value is not a literal
28     */
29    public function decode(array $tokens): Literal
30    {
31        if (count($tokens) !== 1) {
32            throw new DecodingException('The value is not a literal.');
33        }
34        $token = $tokens[0];
35        if ($token->name === 'ID' && in_array(strtoupper($token->text), ['TRUE', 'FALSE'], true)) {
36            return new BooleanLiteral(strtoupper($token->text) === 'TRUE');
37        }
38        return match ($token->name) {
39            'STRING' => new StringLiteral(Quoted::body($token->text)),
40            'BLOB' => new BinaryLiteral(Encoding::hex(substr($token->text, 2, -1))),
41            'NULL' => NullLiteral::Null,
42            'TRUEFALSE' => new BooleanLiteral(strtoupper($token->text) === 'TRUE'),
43            'INTEGER', 'FLOAT', 'QNUMBER' => new NumberLiteral($this->number($token->text)),
44            default => throw new DecodingException('The value is not a literal.'),
45        };
46    }
47
48    /**
49     * Decodes the validated lexical spelling without evaluating SQL.
50     * @throws DecodingException When the encoding is invalid
51     */
52    public function number(string $text): string
53    {
54        $text = str_replace('_', '', $text);
55        if (strtolower(substr($text, 0, 2)) !== '0x') {
56            return $text;
57        }
58        $hex = ltrim(substr($text, 2), '0');
59        if (strlen($hex) > 16) {
60            throw new DecodingException('A hexadecimal integer must fit in 64 bits.');
61        }
62        if (strlen($hex) === 16 && hexdec($hex[0]) >= 8) {
63            $complement = implode('', array_map(static fn (string $digit): string => dechex(15 - (int) hexdec($digit)), str_split($hex)));
64            $magnitude = Encoding::number('0x' . $complement);
65            return '-' . Encoding::successor($magnitude);
66        }
67        return Encoding::number($text);
68    }
