23/**
24 * Composes SQLite values under stable names.
25 *
26 * A name that spells a keyword is always quoted, even one the parser would
27 * fall back to reading as an identifier, because the fallback depends on the
28 * position the name is written in. Double quotes delimit names; strings
29 * double their quotes and know no escapes.
30 *
31 * @visibility public
32 * @example Composing a condition
33 *     $builder = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite))->builder();
34 *     \SqlSemantics\Statement\Writer::render($builder->not($builder->and($builder->compare($builder->column('key'), '>=', $builder->integer(-1)), $builder->parameter()))) // => 'NOT( "key" >= - 1 AND ?1 )'
35 * @example Shadowing a table with a common table expression
36 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite);
37 *     $builder = $semantics->builder();
38 *     $rows = $builder->unionAll($semantics->analyze('SELECT 1 AS id')->command, $semantics->analyze('SELECT 2')->command);
39 *     \SqlSemantics\Statement\Writer::render($builder->with([$builder->cte('users', $rows)], $semantics->analyze('SELECT id FROM users')->command)) // => 'WITH users AS( SELECT 1 AS id UNION ALL SELECT 2 ) SELECT id FROM users'
40 */

63    {

69            default => throw new CompositionException('A column reference has one to three parts.'),

219     * A common table expression over a SELECT.

221    public function cte(string $name, Element $query, array $columns = []): WqitemForm
222    {
223        $entries = array_map(fn (string $column): Element => $this->form('eidlist', ['nm', 'collate', 'sortorder'], [$this->identifier($column), $this->form('collate', [], []), $this->form('sortorder', [], [])]), $columns);
224        $list = $columns === [] ? $this->form('eidlist_opt', [], []) : $this->form('eidlist_opt', ['LP', 'eidlist', 'RP'], [$this->eidlist($entries)]);
225        $value = $this->form('wqitem', ['withnm', 'eidlist_opt', 'wqas', 'LP', 'select', 'RP'], [$this->identifier($name), $list, $this->form('wqas', ['AS'], []), $this->expect($this->unwrap($query), 'select', 'The query of a common table expression')]);
226        assert($value instanceof WqitemForm);

231    /**
232     * A SELECT preceded by a WITH clause; a SELECT that has one keeps its expressions after the new ones.
233     */
234    public function with(array $ctes, Element $query): SelectForm
235    {
236        if ($ctes === []) {
237            throw new CompositionException('A WITH clause needs at least one common table expression.');
238        }
239        foreach ($ctes as $cte) {
240            $this->expect($cte, 'wqitem', 'A common table expression');
241        }
242        $query = $this->unwrap($query);
243        $listSymbols = ['wqlist', 'COMMA', 'wqitem'];
244        $shapes = [['WITH', 'wqlist', 'selectnowith'], ['WITH', 'RECURSIVE', 'wqlist', 'selectnowith']];
245        foreach ($shapes as $symbols) {
246            $class = $this->classOf('select', $symbols);
247            if ($class !== null && $query instanceof $class) {
248                [$list, $body] = $query->children();
249                $value = $this->form('select', $symbols, [$this->fold('wqlist', $listSymbols, [...$ctes, ...$this->unfold($list, 'wqlist', $listSymbols)]), $body]);
250                assert($value instanceof SelectForm);
251
252                return $value;
253            }
254        }
255        $value = $this->form('select', $shapes[0], [$this->fold('wqlist', $listSymbols, $ctes), $this->expect($query, 'selectnowith', 'The query after a WITH clause')]);
256        assert($value instanceof SelectForm);
257
258        return $value;
259    }
