19/**
20 * A conversion of a value to the storage class the affinity of a type name selects.
21 *
22 * Rule: SQLITE-CAST-001. The affinity of the type name decides the result:
23 * INTEGER, TEXT, BLOB and REAL affinity give that storage class; NUMERIC
24 * affinity, which an absent type name also has, gives INTEGER or REAL. A NULL
25 * operand stays NULL, so the NULL fact is that of the operand.
26 * Source: https://sqlite.org/lang_expr.html#castexpr. Status: Implemented.
27 *
28 * @visibility public
29 * @example Reading the type of a CAST
30 *     $query = (new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite))->analyze('SELECT CAST(a AS VARCHAR(10)) FROM t');
31 *     [$query->statement->columns[0]->expression->target->text(), $query->field(0)->type->descriptor] // => ['VARCHAR', \SqlSemantics\Platform\Sqlite\Statement\Type\Storage::Text]
32 */
33final class Cast implements Scalar
34{
35    use Snapshot;
36
37    /**
38     * @param Scalar $operand The converted expression
39     * @param TypeName|null $target The type name; null when none is written
40     */
41    public function __construct(public readonly Scalar $operand, public readonly ?TypeName $target = null)
42    {
43    }
44
45    /**
46     * Derives the storage class from the affinity of the type name.
47     */
48    public function deriveScalar(Derivation $derivation, Environment $environment): ScalarFact
49    {
50        $fact = $derivation->scalar($this->operand, $environment);
51        foreach ($this->target === null ? [] : $this->target->arguments as $argument) {
52            $derivation->scalar($argument->number, $environment);
53        }
54        if ($fact->type instanceof NullOnly) {
55            return new ScalarFact($fact->type, $fact->nullability);
56        }
57        $storages = match ($this->target?->affinity() ?? Affinity::Numeric) {
58            Affinity::Integer => [Storage::Integer],
59            Affinity::Text => [Storage::Text],
60            Affinity::Blob => [Storage::Blob],
61            Affinity::Real => [Storage::Real],
62            Affinity::Numeric => [Storage::Integer, Storage::Real],
63        };
64
65        return new ScalarFact((new Storages())->fact($storages), $fact->nullability);
66    }
67
68    /**
69     * Writes the conversion.
70     */
71    public function render(Output $out): void
72    {
73        $out->keyword('CAST')->glue()->symbol('(')->node($this->operand)->keyword('AS')->node($this->target)->symbol(')');
74    }
