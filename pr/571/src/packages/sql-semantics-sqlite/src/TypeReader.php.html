18 * Reads a declared type name into its conventional identity and its storage affinity.
19 *
20 * The database treats a declared type as a name: the affinity follows the
21 * name text, and the numeric arguments are ignored. Names the documentation
22 * lists become built-in types; other names are kept as named types. Integer
23 * arguments are kept as the length, or precision and scale, the standard
24 * spelling means; other arguments stay only in the typed declaration.

61    /**
62     * Reads a declared type; a column without one holds values of any storage class.
63     */
64    public function read(Node $node, ?Node $table): TypeDeclaration
65    {
66        $tokens = $node->tokens();
67        $names = [];
68        foreach ($tokens as $token) {
69            if ($token->text === '(') {
70                break;
71            }
72            $names[] = (new NameRules())->name($token);
73        }
74        if ($names === []) {
75            return new TypeDeclaration(new TypeDescriptor(Builtin::Dynamic, affinity: Affinity::Blob));
76        }
77        $spelling = strtoupper(implode(' ', $names));
78        $name = self::NAMES[$spelling] ?? new TypeName($names);
79        $affinity = $this->affinity($spelling);
80        if ($name === Builtin::Any && $table !== null && $this->strict($table)) {
81            $affinity = Affinity::Blob;
82        }
83        $arguments = array_map(Numbers::integer(...), Numbers::arguments(TokenGroups::parentheses($tokens)[0] ?? []));
84        [$length, $precision, $scale] = $this->modifiers($name, $arguments);
85
86        return new TypeDeclaration(new TypeDescriptor($name, $length, $precision, $scale, affinity: $affinity));
87    }
88
89    /**
90     * Computes storage affinity from the declared name text in the documented precedence order.
91     */
92    public function affinity(string $spelling): Affinity
93    {
94        if (str_contains($spelling, 'INT')) {
95            return Affinity::Integer;
96        }
97        if (str_contains($spelling, 'CHAR') || str_contains($spelling, 'CLOB') || str_contains($spelling, 'TEXT')) {
98            return Affinity::Text;
99        }
100        if ($spelling === '' || str_contains($spelling, 'BLOB')) {
101            return Affinity::Blob;
102        }
103        if (str_contains($spelling, 'REAL') || str_contains($spelling, 'FLOA') || str_contains($spelling, 'DOUB')) {
104            return Affinity::Real;
105        }
106
107        return Affinity::Numeric;
108    }

110    /**
111     * Reports whether a rowid alias can be declared: the type is spelled exactly INTEGER.
112     */
113    public static function rowidAlias(Node $declaration): bool
114    {
115        $type = Tree::outer($declaration, ['typetoken'])[0] ?? null;
116        $tokens = $type === null ? [] : $type->tokens();
117
118        return count($tokens) === 1 && strcasecmp((new NameRules())->name($tokens[0]), 'INTEGER') === 0;
119    }

121    /**
122     * Keeps integer arguments in the slots the standard spelling gives them; other arguments have no meaning here.
123     *
124     * @param list<int|null> $arguments
125     * @return array{?int, ?int, ?int} Length, precision and scale
126     */
127    public function modifiers(Builtin|TypeName $name, array $arguments): array
128    {
129        if ($name instanceof TypeName || in_array(null, $arguments, true) || $arguments === [] || count($arguments) > 2) {
130            return [null, null, null];
131        }
132        $exact = in_array($name, [Builtin::Numeric, Builtin::Real, Builtin::DoublePrecision], true);
133        if ($exact) {
134            return $arguments[0] < 0 ? [null, null, null] : [null, $arguments[0], $arguments[1] ?? null];
135        }
136        if (count($arguments) !== 1 || $arguments[0] < 0) {
137            return [null, null, null];
138        }
139
140        return $name->isTemporal() ? [null, $arguments[0], null] : [$arguments[0], null, null];
141    }

143    /**
144     * Reports whether the table declares the STRICT option, which stores ANY values as written.
145     */
146    public function strict(Node $table): bool
147    {
148        foreach (Tree::outer($table, ['table_option']) as $option) {
149            if (strtoupper(Tree::text($option)) === 'STRICT') {
150                return true;
151            }
152        }
153
154        return false;
155    }
