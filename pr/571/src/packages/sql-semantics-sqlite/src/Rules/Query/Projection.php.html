27/**
28 * Derives the output fields of a result column list.
29 *
30 * Rule: SQLITE-PROJECTION-001. A result column is named by
31 * SQLITE-RESULT-NAME-001 for the rows a statement returns: by its alias, by
32 * the column a column reference denotes, or by the span of its expression.
33 * `*` contributes every column of every input relation except the columns
34 * merged away by USING and NATURAL joins; `t.*` contributes every column of the relations the
35 * qualifier names. A relation whose columns are not all known contributes
36 * its known columns and an open star that names the missing inputs. A star
37 * without any input relation and a qualifier that names no input relation
38 * are reported, as is a result column that is a row value, which has no
39 * single value. Terminates: one pass over the list.
40 * Source: https://sqlite.org/c3ref/column_name.html,
41 * https://sqlite.org/lang_select.html#generation_of_the_set_of_result_rows.
42 * Status: Implemented.
43 *
44 * @visibility SqlSemantics\Platform\Sqlite
45 */
46final class Projection
47{
48    /**
49     * Derives the result columns where the given environment is visible.
50     *
51     * @param list<ResultColumn|Star|TableStar> $columns
52     * @return list<Field|OpenStar>
53     */
54    public function items(array $columns, Derivation $derivation, Environment $environment): array
55    {
56        $items = [];
57        foreach ($columns as $column) {
58            if ($column instanceof ResultColumn) {
59                $fact = $derivation->scalar($column->expression, $environment);
60                if ($fact->type instanceof Known && $fact->type->descriptor instanceof Vector) {
61                    $derivation->report(new Misuse(MisuseRule::TooManyValueColumns));
62                }
63                $resolution = $column->expression instanceof Grouped ? (new ColumnFacts())->denoted($column->expression, $environment) : $fact->resolution;
64                $origin = $resolution instanceof ResolvedColumn ? $resolution->slot : null;
65                $names = new ResultNames();
66                $items[] = new Field(count($items), new OutputSlot($names->output($column, $resolution), $fact->type, $fact->nullability, null, $origin, $names->unnamed($column, $resolution)), $column->expression, $resolution);
67                continue;
68            }
69            $selected = [];
70            foreach ($environment->relations as $relation) {
71                if ((new ColumnResolver())->admits($environment, $relation, $column instanceof TableStar ? new QualifiedName($column->table) : null)) {
72                    $selected[] = $relation;
73                }
74            }
75            if ($selected === [] && ($column instanceof TableStar || $environment->relations === [])) {
76                $derivation->report($column instanceof TableStar ? new MissingTable(new QualifiedName($column->table)) : new Misuse(MisuseRule::StarWithoutTables));
77            }
78            foreach ($selected as $relation) {
79                $items = $this->expand($items, $relation, $column instanceof Star);
80            }
81        }
82
83        return $items;
84    }
85
86    /**
87     * Appends the columns a star selects from one relation.
88     *
89     * @param list<Field|OpenStar> $items
90     * @return list<Field|OpenStar>
91     */
92    public function expand(array $items, VisibleRelation $relation, bool $merged): array
93    {
94        foreach ($relation->shape->slots as $position => $slot) {
95            if (!$merged || !in_array($position, $relation->hidden, true)) {
96                $items[] = new Field(count($items), new OutputSlot($slot->name, $slot->type, $slot->nullability, null, $slot, $slot->unnamed), null, new ResolvedColumn($relation->relation, $slot));
97            }
98        }
99        if (!$relation->shape->complete()) {
100            $items[] = new OpenStar($relation->shape->missing);
101        }
102
103        return $items;
104    }
105}
