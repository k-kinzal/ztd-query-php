7/**
8 * Reads the exact value of a PostgreSQL numeric constant token without floating point.
9 *
10 * Rule: PG-LEX-NUMBER-001. Scope: the `ICONST` and `FCONST` terminals. A
11 * constant written as an integer, in decimal, hexadecimal (`0x`), octal
12 * (`0o`) or binary (`0b`) digits with optional underscores between digits, is
13 * the integer it denotes whatever its size. The scanner makes it an `ICONST`
14 * holding that integer when it fits 32 bits (`process_integer_literal` in
15 * `scan.l`); otherwise, and for every constant written with a decimal point
16 * or an exponent, it makes an `FCONST` that holds the written text itself
17 * (a `T_Float` node keeps the string), because the parser does not yet know
18 * the type the value is read as. Commands such as `SET`, storage parameters
19 * and trigger arguments receive that text verbatim, so `1e2` and `100.` are
20 * different values there. Source: https://www.postgresql.org/docs/17/sql-syntax-lexical.html#SQL-SYNTAX-CONSTANTS-NUMERIC,
21 * `process_integer_literal` and the `{numeric}`, `{real}` rules of `src/backend/parser/scan.l` of PostgreSQL 17.
22 * Termination: one pass over the digits per conversion step.
23 * Status: Implemented.
24 *
25 * @visibility SqlSemantics
26 */
27final class Numerals
28{
29    /**
30     * Tells whether a numeric token is written as an integer.
31     */
32    public function integral(string $text): bool
33    {
34        return preg_match('/\A(?:0[xXoObB][0-9A-Fa-f_]+|[0-9_]+)\z/', $text) === 1;
35    }
36
37    /**
38     * Answers the canonical decimal digits of a constant written as an integer.
39     */
40    public function decimal(string $text): string
41    {
42        $digits = str_replace('_', '', $text);
43        $base = ['0x' => 16, '0o' => 8, '0b' => 2][strtolower(substr($digits, 0, 2))] ?? 10;
44        if ($base === 10) {
45            return $this->canonical($digits);
46        }
47        $limbs = [0];
48        foreach (str_split(substr($digits, 2)) as $digit) {
49            $carry = (int) hexdec($digit);
50            foreach ($limbs as $position => $limb) {
51                $product = $limb * $base + $carry;
52                $limbs[$position] = $product % 10000000;
53                $carry = intdiv($product, 10000000);
54            }
55            if ($carry > 0) {
56                $limbs[] = $carry;
57            }
58        }
59        $decimal = '';
60        foreach ($limbs as $limb) {
61            $decimal = str_pad((string) $limb, 7, '0', STR_PAD_LEFT) . $decimal;
62        }
63
64        return $this->canonical($decimal);
65    }
66
67    /**
68     * Answers the canonical decimal digits of the integer a constant written as an integer denotes, or null for one with a point or an exponent.
69     */
70    public function integer(string $text): ?string
71    {
72        return $this->integral($text) ? $this->decimal($text) : null;
73    }
74
75    /**
76     * Tells whether a text is one constant the scanner reads as an `FCONST` and keeps as written.
77     *
78     * That is a decimal number with a point, an exponent or both, or an
79     * integer spelling whose value does not fit 32 bits.
80     */
81    public function kept(string $text): bool
82    {
83        $digits = '[0-9](?:_?[0-9])*';
84        if (preg_match('/\A(?:0[xX](?:_?[0-9A-Fa-f])+|0[oO](?:_?[0-7])+|0[bB](?:_?[01])+|' . $digits . ')\z/', $text) === 1) {
85            return !$this->within($this->decimal($text), '2147483647');
86        }
87
88        return preg_match('/\A(?:' . $digits . '(?:\.(?:' . $digits . ')?)?|\.' . $digits . ')(?:[eE][-+]?' . $digits . ')?\z/', $text) === 1;
89    }
90
91    /**
92     * Strips leading zeros from decimal digits; no digits denote zero.
93     */
94    public function canonical(string $digits): string
95    {
96        $stripped = ltrim($digits, '0');
97
98        return $stripped === '' ? '0' : $stripped;
99    }
100
101    /**
102     * Tells whether canonical decimal digits denote a value not above a canonical limit.
103     */
104    public function within(string $digits, string $limit): bool
105    {
106        return strlen($digits) < strlen($limit) || (strlen($digits) === strlen($limit) && strcmp($digits, $limit) <= 0);
107    }
108}
