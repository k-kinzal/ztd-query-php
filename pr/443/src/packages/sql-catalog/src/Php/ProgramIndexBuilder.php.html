42     * The index of every declaration in the given files.
43     *
44     * @param list<ParsedFile> $files
45     */
46    public function build(array $files): ProgramIndex
47    {
48        $classes = [];
49        $functions = [];
50        $constants = [];
51
52        foreach ($files as $file) {
53            foreach ($this->finder->findInstanceOf($file->statements, Node::class) as $node) {
54                if ($node instanceof ClassLike) {
55                    $shape = $this->readClass($node, $file->path);
56                    if ($shape !== null) {
57                        $classes[strtolower($shape->name)] = $shape;
58                    }
59                    continue;
60                }
61                if ($node instanceof Function_) {
62                    $name = $node->namespacedName?->toString() ?? $node->name->toString();
63                    $functions[strtolower($name)] = new FunctionShape(
64                        $name,
65                        $this->readParameters($node->params),
66                        $this->types->read($node->returnType),
67                        $node,
68                        $file->path,
69                    );
70                    continue;
71                }
72                if ($node instanceof Node\Stmt\Const_) {
73                    foreach ($node->consts as $const) {
74                        $constants[$const->namespacedName?->toString() ?? $const->name->toString()] = $const->value;
75                    }
76                }
77            }
78        }
79
80        return new ProgramIndex($classes, $functions, $constants);
81    }

83    /**
84     * The shape of one class-like declaration, or null when it has no resolvable name.
85     */
86    public function readClass(ClassLike $node, string $file = ''): ?ClassShape
87    {
88        $name = $node->namespacedName?->toString() ?? $node->name?->toString();
89        if ($name === null) {
90            return null;
91        }
92
93        $methods = [];
94        $properties = [];
95        foreach ($node->getMethods() as $method) {
96            $methods[strtolower($method->name->toString())] = $this->readMethod($name, $method, $file);
97            $properties += $this->readPromotedProperties($method);
98        }
99        $members = $this->readMembers($node);
100
101        return new ClassShape(
102            $name,
103            $node instanceof Class_ ? $node->extends?->toString() : null,
104            $this->readParentNames($node),
105            $members['traits'],
106            $node instanceof Enum_,
107            $members['constants'],
108            $members['cases'],
109            array_merge($properties, $members['properties']),
110            $methods,
111            $members['defaults'],
112            $members['assigned'],
113            $node->attrGroups !== [],
114        );
115    }

117    /**
118     * Everything a class body declares, read in one pass over it.
119     *
120     * The body is walked once and its declarations are sorted afterwards, which
121     * matters on a source tree with thousands of classes in it.
122     *
123     * @return array{
124     *     traits: list<string>,
125     *     constants: array<string, Node\Expr>,
126     *     cases: array<string, Node\Expr|null>,
127     *     properties: array<string, TypeShape>,
128     *     defaults: array<string, Node\Expr>,
129     *     assigned: array<string, true>
130     * }
131     */
132    public function readMembers(ClassLike $node): array
133    {
134        $members = ['traits' => [], 'constants' => [], 'cases' => [], 'properties' => [], 'defaults' => [], 'assigned' => []];
135        foreach ($this->finder->findInstanceOf([$node], Node::class) as $member) {
136            if ($member instanceof TraitUse) {
137                foreach ($member->traits as $trait) {
138                    $members['traits'][] = $trait->toString();
139                }
140            }
141            if ($member instanceof ClassConst) {
142                foreach ($member->consts as $const) {
143                    $members['constants'][$const->name->toString()] = $const->value;
144                }
145            }
146            if ($member instanceof EnumCase) {
147                $members['cases'][$member->name->toString()] = $member->expr;
148            }
149            if ($member instanceof Property) {
150                foreach ($member->props as $declared) {
151                    $members['properties'][$declared->name->toString()] = $this->types->read($member->type);
152                    if ($declared->default !== null) {
153                        $members['defaults'][$declared->name->toString()] = $declared->default;
154                    }
155                }
156            }
157            $target = $member instanceof Node\Expr ? $this->assignmentTarget($member) : null;
158            if ($target !== null) {
159                $members['assigned'][$target] = true;
160            }
161        }
162
163        return $members;
164    }

184    /**
185     * The property name an expression assigns to, or null when it assigns to none.
186     */
187    public function assignmentTarget(Node\Expr $expression): ?string
188    {
189        $target = match (true) {
190            $expression instanceof Node\Expr\Assign => $expression->var,
191            $expression instanceof Node\Expr\AssignOp => $expression->var,
192            $expression instanceof Node\Expr\AssignRef => $expression->var,
193            default => null,
194        };
195        if (!$target instanceof Node\Expr\PropertyFetch && !$target instanceof Node\Expr\StaticPropertyFetch) {
196            return null;
197        }
198
199        return $target->name instanceof Node\Identifier ? $target->name->toString() : null;
200    }

280     * @return array<string, TypeShape>
281     */
282    public function readPromotedProperties(ClassMethod $method): array
283    {
284        if (strtolower($method->name->toString()) !== '__construct') {
285            return [];
286        }
287
288        $properties = [];
289        foreach ($method->params as $param) {
290            if ($param->flags !== 0 && $param->var instanceof Node\Expr\Variable && is_string($param->var->name)) {
291                $properties[$param->var->name] = $this->types->read($param->type);
292            }
293        }
294
295        return $properties;

320    public function readParameters(array $params): array
321    {
322        $shapes = [];
323        foreach ($params as $param) {
324            if (!$param->var instanceof Node\Expr\Variable || !is_string($param->var->name)) {
325                continue;
326            }
327            $shapes[] = new ParameterShape(
328                $param->var->name,
329                $this->types->read($param->type),

333        }
