8 * A string the analyzer knows in part: resolved runs of text with gaps between them.
9 *
10 * A pattern with no gaps is an exact string. One with gaps describes the shape

12 * when a query is assembled from values that are not statically known.

49    /**
50     * A pattern built from arbitrary segments, coalescing adjacent literals.
51     *
52     * @param list<TextSegment> $segments
53     */
54    public static function fromSegments(array $segments): self
55    {
56        $normalized = [];
57        foreach ($segments as $segment) {
58            $last = $normalized === [] ? null : $normalized[count($normalized) - 1];
59            if ($segment instanceof LiteralText && $segment->text === '') {
60                continue;
61            }
62            if ($segment instanceof LiteralText && $last instanceof LiteralText) {
63                $normalized[count($normalized) - 1] = new LiteralText($last->text . $segment->text);
64                continue;
65            }
66            $normalized[] = $segment;
67        }
68
69        return new self($normalized);
70    }

150    {
151        $parts = [];
152        foreach ($this->segments as $segment) {
153            $parts[] = match ($segment::class) {
154                LiteralText::class => 'text:' . $segment->display(),
155                TextHole::class => 'hole:' . $segment->origin->value,
156                default => 'segment:' . $segment->display(),
157            };
158        }
