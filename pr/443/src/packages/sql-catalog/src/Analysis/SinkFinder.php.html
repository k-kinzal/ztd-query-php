14/**
15 * Finds every call that is written the way a database call is written.
16 *
17 * Reading a statement and finding the call that carries it are separate jobs.
18 * A call whose statement cannot be reconstructed is still a place the program
19 * talks to a database, and a call nothing could be read from is a gap in the
20 * analysis rather than an absence in the program. This pass answers the second
21 * question on its own, so neither can be lost to a failure of the first.
22 *
23 * @visibility root
24 */
25final class SinkFinder
26{
27    private NodeFinder $finder;

29    /**
30     * Builds a finder over the node search.
31     */
32    public function __construct()
33    {
34        $this->finder = new NodeFinder();
35    }

38     * The calls in a file that are written the way a statement-carrying call is written.
39     *
40     * @param list<SinkSpec> $sinks
41     * @return list<Expr\CallLike>
42     */
43    public function find(ParsedFile $file, array $sinks): array
44    {
45        return $this->findIn($file->statements, $this->namesOf($sinks));
46    }

48    /**
49     * The calls in a file written the way any database call is written, including those that only bind values.
50     *
51     * Calls that compose a statement are left out: they hand the statement back
52     * rather than sending it, so they are read as part of whatever call does.
53     *
54     * @param list<SinkSpec> $sinks
55     * @return list<Expr\CallLike>
56     */
57    public function findAll(ParsedFile $file, array $sinks): array
58    {
59        $names = [];
60        foreach ($sinks as $sink) {
61            if ($sink->role !== SinkRole::Compose) {
62                $names[strtolower(ltrim($sink->name, '\\'))] = true;
63            }
64        }
65
66        return $this->findIn($file->statements, $names);
67    }
68
69    /**
70     * The calls among the given nodes that are written the way a statement-carrying call is written.
71     *
72     * @param array<array-key, Node> $nodes
73     * @param array<string, true> $names
74     * @return list<Expr\CallLike>
75     */
76    public function findIn(array $nodes, array $names): array
77    {
78        $found = [];
79        /** @var list<Expr\CallLike> $calls */
80        $calls = $this->finder->findInstanceOf($nodes, Expr\CallLike::class);
81        foreach ($calls as $call) {
82            $written = $this->nameOf($call);
83            if ($written !== null && isset($names[strtolower($written)])) {
84                $found[] = $call;
85            }
86        }
87
88        return $found;
89    }
90
91    /**
92     * The names the statement-carrying calls are written with.
93     *
94     * @param list<SinkSpec> $sinks
95     * @return array<string, true>
96     */
97    public function namesOf(array $sinks): array
98    {
99        $names = [];
100        foreach ($sinks as $sink) {
101            if ($sink->role === SinkRole::Query || $sink->role === SinkRole::Prepare || $sink->role === SinkRole::Modelled) {
102                $names[strtolower(ltrim($sink->name, '\\'))] = true;
103            }
104        }
105
106        return $names;
107    }
108
109    /**
110     * The name a call is written with, or null when it is not written as a name.
111     */
112    public function nameOf(Expr\CallLike $call): ?string
113    {
114        if ($call instanceof Expr\MethodCall || $call instanceof Expr\NullsafeMethodCall) {
115            return $call->name instanceof Node\Identifier ? $call->name->toString() : null;
116        }
117        if ($call instanceof Expr\StaticCall) {
118            return $call->name instanceof Node\Identifier ? $call->name->toString() : null;
119        }
120        if ($call instanceof Expr\FuncCall) {
121            return $call->name instanceof Node\Name ? $call->name->toString() : null;
122        }
123
124        return null;
125    }
126
127    /**
128     * The function body a node is written inside, or null when it is written outside any.
129     */
130    public function enclosingBody(Node $node): ?Node\FunctionLike
131    {
132        $parent = $node->getAttribute('parent');
133        while ($parent instanceof Node) {
134            if ($parent instanceof Node\FunctionLike) {
135                return $parent;
136            }
137            $parent = $parent->getAttribute('parent');
138        }
139
140        return null;
141    }
