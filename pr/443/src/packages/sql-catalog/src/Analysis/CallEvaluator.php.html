26/**
27 * Works out what a call produces.
28 *
29 * A database call produces what the driver hands back: a statement handle
30 * from a preparing call, the statement text itself from a call that composes
31 * one. A call into the analyzed source produces what the callee returns for
32 * these arguments. Recording statements is not done here; it is done at each
33 * database call, from the call itself.
34 *
35 * @visibility root
36 */

73     * The value a call produces.
74     */
75    public function evaluate(
76        Expr\CallLike $node,
77        Environment $environment,
78        FunctionScope $scope,
79        ExpressionEvaluator $expressions,
80    ): Domain {
81        if ($node->isFirstClassCallable()) {
82            return Domain::of(new ObjectTerm('Closure'));
83        }
84        $receiver = $node instanceof Expr\MethodCall || $node instanceof Expr\NullsafeMethodCall ? $expressions->evaluate($node->var, $environment, $scope) : null;
85        $arguments = $this->arguments($node, $environment, $scope, $expressions);
86
87        if ($node instanceof Expr\New_) {
88            $modelled = $this->functions->evaluateCall(new CallContext($node, $arguments, $environment, $scope, $expressions, className: $node->class instanceof Node\Name ? $node->class->toString() : null));
89            if ($modelled !== null) {
90                return $modelled;
91            }
92            $this->applyEffects($node, $environment);
93            $this->invalidateEscapes($node, $environment);
94            return $this->evaluateInstantiation($node, $scope);
95        }
96        if ($node instanceof Expr\MethodCall || $node instanceof Expr\NullsafeMethodCall) {
97            return $this->evaluateMethod($node, $arguments, $environment, $scope, $expressions, $receiver);
98        }
99        if ($node instanceof Expr\StaticCall) {
100            return $this->evaluateStatic($node, $arguments, $scope, $expressions, $environment);
101        }
102        if ($node instanceof Expr\FuncCall) {
103            return $this->evaluateFunction($node, $arguments, $scope, $expressions, $environment);
104        }
105
106        return Domain::opaque(TypeShape::unknown(), Origin::Call, $this->text->render($node));
107    }

109    /**
110     * Opens arguments whose callees may write through references.
111     */
112    public function applyEffects(Expr\CallLike $node, Environment $environment): void
113    {
114        if ($node instanceof Expr\FuncCall && $node->name instanceof Node\Name) {
115            $name = $this->functionName($node->name);
116            if ($this->sinks->matchFunction($name) !== null) {
117                return;
118            }
119            if ($this->index->findFunction($name) === null && $this->functions->supports($name)) {
120                $count = $node->getArgs()[3] ?? null;
121                if ($this->functions->normalize($name) === 'str_replace' && $count !== null) {
122                    (new WriteEffects())->apply((new Derivation\ModifiedNames())->targets($count->value), $environment);
123                }
124                return;
125            }
126        }
127        if ($node instanceof Expr\MethodCall && $node->name instanceof Node\Identifier
128            && $node->var instanceof Expr\Variable && is_string($node->var->name)
129            && $this->sinks->matchMethod($environment->read($node->var->name), $node->name->toString()) !== null) {
130            return;
131        }
132        $effects = new WriteEffects();
133        $written = $effects->own($node, $this->index);
134        $this->references ??= new ReferenceEffects();
135        $effects->apply($written + $this->references->affected($node, $written), $environment);
136    }

140     */
141    public function evaluateInstantiation(Expr\New_ $node, FunctionScope $scope): Domain
142    {
143        if (!$node->class instanceof Node\Name) {
144            return Domain::opaque(TypeShape::of(['object']), Origin::Call, 'new');
145        }
146        $className = $node->class->toString();
147        if (in_array(strtolower($className), ['self', 'static', 'parent'], true)) {
148            $className = $scope->className ?? $className;
149        }
150
151        return Domain::of(new ObjectTerm($className));

173    /**
174     * The value of a method call, which may be a database call.
175     *
176     * @param list<Domain> $arguments
177     */
178    public function evaluateMethod(
179        Expr\MethodCall|Expr\NullsafeMethodCall $node,
180        array $arguments,
181        Environment $environment,
182        FunctionScope $scope,
183        ExpressionEvaluator $expressions,
184        ?Domain $receiver = null,
185    ): Domain {
186        $name = $node->name instanceof Node\Identifier ? $node->name->toString() : null;
187        if ($name === null) {
188            $this->applyEffects($node, $environment);
189            $this->invalidateEscapes($node, $environment);
190            $environment->objects()->invalidate($receiver ?? $expressions->evaluate($node->var, $environment, $scope));
191            return Domain::opaque(TypeShape::unknown(), Origin::Call, $this->text->render($node));
192        }
193
194        $receiver = $environment->refresh($receiver ?? $expressions->evaluate($node->var, $environment, $scope));
195        $sink = $this->sinks->matchMethod($receiver, $name);
196        $modelled = $this->functions->evaluateCall(new CallContext($node, $arguments, $environment, $scope, $expressions, $name, $receiver, sink: $sink));
197        if ($modelled !== null) {
198            return $modelled;
199        }
200        $this->applyEffects($node, $environment);
201        if ($sink !== null) {
202            return $this->applySink($sink, $node, $arguments, $scope);
203        }
204        if (!$this->isOwnReceiver($node) && $this->sinks->models($receiver)) {
205            return Domain::opaque(TypeShape::unknown(), Origin::Call, $this->text->render($node));
206        }
207
208        $this->invalidateEscapes($node, $environment);
209        $className = $receiver->type()->soleClassName();
210        $method = $this->index->findMethod($className, $name);
211        if ($method?->node?->getStmts() === null) {
212            $dispatched = $this->dispatch($className, $name, $arguments, $scope, $expressions);
213            if ($dispatched !== null) {
214                return $dispatched;
215            }
216        }
217
218        return $method === null
219            ? Domain::opaque(TypeShape::unknown(), Origin::Call, $this->text->render($node))
220            : $this->follow($method, $arguments, $scope, $expressions);
221    }

235    /**
236     * The value a call resolves to across the implementations the source declares.
237     *
238     * @param list<Domain> $arguments
239     */
240    public function dispatch(
241        ?string $className,
242        string $method,
243        array $arguments,
244        FunctionScope $scope,
245        ExpressionEvaluator $expressions,
246    ): ?Domain {
247        $result = null;
248        foreach ($this->index->implementationsOf($className, $method) as $implementation) {
249            $value = $this->follow($implementation, $arguments, $scope, $expressions);
250            $result = $result === null ? $value : $result->union($value);
251        }
252
253        return $result;
254    }

261    public function evaluateStatic(
262        Expr\StaticCall $node,
263        array $arguments,
264        FunctionScope $scope,
265        ExpressionEvaluator $expressions,
266        ?Environment $environment = null,
267    ): Domain {
268        $name = $node->name instanceof Node\Identifier ? $node->name->toString() : null;
269        $className = $node->class instanceof Node\Name ? $node->class->toString() : null;
270        if ($name === null || $className === null) {
271            if ($environment !== null) {
272                $this->applyEffects($node, $environment);
273                $this->invalidateEscapes($node, $environment);
274            }
275            return Domain::opaque(TypeShape::unknown(), Origin::Call, $this->text->render($node));
276        }
277        if (in_array(strtolower($className), ['self', 'static', 'parent'], true)) {

279        }

287        $this->applyEffects($node, $environment);
288        if ($sink !== null) {

300    /**
301     * The value of a function call, which may be a database call or a string builder.
302     *
303     * @param list<Domain> $arguments
304     */
305    public function evaluateFunction(
306        Expr\FuncCall $node,
307        array $arguments,
308        FunctionScope $scope,
309        ExpressionEvaluator $expressions,
310        ?Environment $environment = null,
311    ): Domain {
312        if (!$node->name instanceof Node\Name) {
313            if ($environment !== null) {
314                $this->applyEffects($node, $environment);
315                $this->invalidateEscapes($node, $environment);
316            }
317            return Domain::opaque(TypeShape::unknown(), Origin::Call, $this->text->render($node));
318        }
319        $name = $this->functionName($node->name);
320
321        $sink = $this->sinks->matchFunction($name);
322        $environment ??= new Environment();
323        $modelled = $this->functions->evaluateCall(new CallContext($node, $arguments, $environment, $scope, $expressions, $name, sink: $sink));
324        if ($modelled !== null) {
325            return $modelled;
326        }
327        $this->applyEffects($node, $environment);
328        if ($sink !== null) {
329            return $this->applySink($sink, $node, $arguments, $scope);
330        }
331        $modeled = $this->functions->evaluate($name, $arguments);
332        if ($modeled !== null) {
333            return $modeled;
334        }
335        if ($this->external->isFunction($name)) {
336            return Domain::opaque(TypeShape::unknown(), Origin::External, $name . '()');
337        }
338
339        $this->invalidateEscapes($node, $environment);
340
341        $function = $this->index->findFunction($name);
342
343        return $function === null
344            ? Domain::opaque(TypeShape::unknown(), Origin::Call, $this->text->render($node))
345            : $this->follow($function, $arguments, $scope, $expressions);
346    }

374    /**
375     * What a database call hands back.
376     *
377     * A preparing call hands back a handle naming the call it came from, which
378     * is how a later `execute()` finds the statement it binds to. A composing
379     * call hands back the statement it was given.
380     *
381     * @param list<Domain> $arguments
382     */
383    public function applySink(SinkSpec $sink, Expr\CallLike $node, array $arguments, FunctionScope $scope): Domain
384    {
385        return match ($sink->role) {
386            SinkRole::Compose => $arguments[$sink->sqlParameter ?? 0] ?? Domain::unknown(),
387            SinkRole::Prepare => Domain::of(new ObjectTerm(
388                $sink->handleType ?? 'PDOStatement',
389                null,
390                $this->siteKeyOf($node, $scope, $sink->id),
391            )),
392            SinkRole::Query, SinkRole::Modelled => Domain::opaque(TypeShape::unknown(), Origin::Call, $sink->id),
393            SinkRole::Execute, SinkRole::Bind => Domain::opaque(TypeShape::of(['bool']), Origin::Call, $sink->id),
394        };
395    }
