7/**
8 * The work one analysis is allowed to do before it gives up and widens.
9 *
10 * Following calls and re-walking loop bodies is what makes the analyzer
11 * precise, and is also what makes it possible for a pathological input to run
12 * forever. The budget bounds both, so an exhausted budget degrades the result
13 * into gaps rather than hanging the run.
14 *
15 * @visibility root
16 */
17final class EvaluationBudget
18{
19    /**
20     * How many times the search's budget reading what was found may take.
21     */
22    public const READING_ALLOWANCE = 4;
23
24    private int $steps;
25
26    /**
27     * @param int $maxSteps How many expressions may be evaluated while walking one body
28     * @param int $maxDepth How many nested calls may be followed
29     * @param int $maxLoopPasses How many times a loop body is re-walked
30     */
31    public function __construct(
32        public readonly int $maxSteps = 20000,
33        public readonly int $maxDepth = 4,
34        public readonly int $maxLoopPasses = 2,
35    ) {
36        $this->steps = 0;
37    }
38
39    /**
40     * Spends one step, reporting whether the budget still allows work.
41     */
42    public function spend(): bool
43    {
44        $this->steps++;
45
46        return $this->steps <= $this->maxSteps;
47    }
48
49    /**
50     * Whether the step budget is used up.
51     */
52    public function isExhausted(): bool
53    {
54        return $this->steps > $this->maxSteps;
55    }
56
57    /**
58     * Whether even the work of reading what is already in hand has to stop.
59     *
60     * Running out of budget stops the search from going anywhere new, but the
61     * statement it has found so far is still read, so a search cut short keeps
62     * the parts of the statement it did find. Reading has a limit of its own,
63     * several times the search's, so a pathological input still ends.
64     */
65    public function isSpent(): bool
66    {
67        return $this->steps > $this->maxSteps * self::READING_ALLOWANCE;
68    }
69
70    /**
71     * How many steps have been spent.
72     */
73    public function spent(): int
74    {
75        return $this->steps;
76    }
77
78    /**
79     * Refills the budget, at the beginning of a new body.
80     */
81    public function reset(): void
82    {
83        $this->steps = 0;
84    }
85}
