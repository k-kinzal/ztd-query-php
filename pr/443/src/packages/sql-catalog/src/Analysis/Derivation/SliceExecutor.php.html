25/**
26 * Runs the steps a path kept, in order, and reports the values they leave behind.
27 *
28 * Only the assignments the call depends on are run, so the work is what the
29 * statement needs rather than what the body does. When an assignment leaves a
30 * variable with several possible values — a ternary, a call that returns one
31 * of several strings — the run splits there, one run per value, so every later
32 * use of the variable sees the same one. That is what stops `ORDER BY $o, id $o`
33 * from pairing an ascending first column with a descending second.
34 *
35 * Keeping runs apart costs one evaluation of every later step per run, so how
36 * many are kept apart is what the budget can pay for along the path: a short
37 * path keeps a dozen, a body of several hundred assignments under a hundred
38 * conditionals keeps a few, and joins the rest value by value. A joined run
39 * still holds every value; what it gives up is knowing which values go
40 * together.
41 *
42 * @visibility root
43 */
44final class SliceExecutor
45{

83    /**
84     * The environments the steps leave behind, one per run.
85     *
86     * A run that works out something on the way to a statement — what a
87     * callee returns, what a caller passes, what a property holds — stops when
88     * the search's budget is spent. The run that reads the statement itself
89     * goes on until the reading limit, so what was found along the way is not
90     * thrown away for want of the last few steps.
91     *
92     * @param list<SliceStep> $steps The steps, in the order they run
93     * @param bool $closing Whether this run reads the statement itself rather than something on the way to it
94     * @param int|null $limit How many runs to keep apart, or null to size it from the budget
95     * @return list<Environment>
96     */
97    public function run(
98        array $steps,
99        Environment $start,
100        FunctionScope $scope,
101        ExpressionEvaluator $expressions,
102        bool $closing = false,
103        ?int $limit = null,
104    ): array {
105        $limit ??= $this->affordableRuns($steps);
106        /** @var WeakMap<Node, int> $passes */
107        $passes = new WeakMap();
108        /** @var WeakMap<Node, Domain> $iterables */
109        $iterables = new WeakMap();
110        $environments = [$start];
111        foreach ($steps as $index => $step) {
112            if ($closing ? $this->budget->isSpent() : $this->budget->isExhausted()) {
113                return $this->abandon(array_slice($steps, $index), $environments);
114            }
115            $next = [];
116            foreach ($environments as $environment) {
117                $next = array_merge($next, $this->apply($step, $environment, $scope, $expressions, $passes, $iterables, $limit));
118            }
119            $environments = $this->bound($next, $limit);
120        }
121
122        return $environments;
123    }

125    /**
126     * How many runs the budget can keep apart along these steps.
127     *
128     * @param list<SliceStep> $steps
129     */
130    public function affordableRuns(array $steps): int
131    {
132        $size = max(1, $this->size($steps));
133
134        return max(1, min(self::MAX_RUNS, intdiv($this->budget->maxSteps, $size * self::STEP_COST)));
135    }

138     * How many steps there are, counting the steps held as alternatives.
139     *
140     * @param list<SliceStep> $steps
141     */
142    public function size(array $steps): int
143    {
144        $size = 0;
145        foreach ($steps as $step) {
146            $size++;
147            foreach ($step->alternatives as $alternative) {
148                $size += $this->size($alternative);
149            }
150        }

155    /**
156     * The environments with everything the steps not yet run would write left open.
157     *
158     * Stopping partway must not report a variable as holding what it held
159     * before a later assignment the run never got to: that would be a wrong
160     * statement rather than an incomplete one. So every name a remaining step
161     * writes is left open, marked as stopped by the budget.
162     *
163     * @param list<SliceStep> $steps
164     * @param list<Environment> $environments
165     * @return list<Environment>
166     */
167    public function abandon(array $steps, array $environments): array
168    {
169        $names = [];
170        foreach ($steps as $step) {
171            $names += $this->writtenBy($step);
172        }
173        $open = [];
174        foreach ($environments as $environment) {
175            $left = $environment->copy();
176            (new WriteEffects())->apply($names, $left, Origin::Budget);
177            $open[] = $left;
178        }
179
180        return $open;
181    }

186     * @return array<string, true>

188    public function writtenBy(SliceStep $step): array
189    {
190        if ($step->node !== null) {
191            $names = $this->modified->of($step->node);
192            foreach ($step->names as $name) {
193                $names[$name] = true;
194            }
195
196            return $names;
197        }
198        $names = [];
199        foreach ($step->alternatives as $alternative) {
200            foreach ($alternative as $inner) {
201                $names += $this->writtenBy($inner);
202            }
203        }
204
205        return $names;

208    /**
209     * The environments one step leaves behind, starting from one environment.
210     *
211     * @param WeakMap<Node, int> $passes How many passes each loop has started so far
212     * @param WeakMap<Node, Domain> $iterables What each loop went over, as read before its first pass
213     * @return list<Environment>
214     */
215    public function apply(
216        SliceStep $step,
217        Environment $environment,
218        FunctionScope $scope,
219        ExpressionEvaluator $expressions,
220        WeakMap $passes,
221        WeakMap $iterables,
222        int $limit = self::MAX_RUNS,
223    ): array {
224        if ($step->node === null) {
225            $runs = [];
226            foreach ($step->alternatives as $alternative) {
227                $runs = array_merge($runs, $this->run($alternative, $environment->copy(), $scope, $expressions));
228            }
229
230            return $runs;
231        }
232        $next = $environment->copy();
233        $node = $step->node;
234        if ($node instanceof Expr\Closure || $node instanceof Expr\ArrowFunction) {
235            $this->enterClosure($node, $step->names, $next);
236
237            return [$next];
238        }
239        if ($node instanceof Stmt\Global_ || $node instanceof Stmt\Static_ || $node instanceof Stmt\Unset_) {
240            $this->declare($node, $next);
241
242            return [$next];
243        }
244        if ($node instanceof Stmt\Foreach_) {
245            $pass = $passes[$node] ?? 0;
246            $passes[$node] = $pass + 1;
247            $iterable = $iterables[$node] ?? $expressions->evaluate($node->expr, $next, $scope);
248            $iterables[$node] = $iterable;
249            $this->iterate($node, $iterable, $pass, $next, $scope, $expressions);
250
251            return $this->split($next, $this->modified->of($node), $limit);
252        }
253        if ($node instanceof Expr) {
254            $this->assign($node, $next, $scope, $expressions);
255
256            return $this->split($next, $this->modified->of($node), $limit);
257        }
258
259        return [$next];
260    }

262    /**
263     * Binds what a closure defines for itself: its parameters, and the names it reads without defining.
264     *
265     * @param list<string> $names
266     */
267    public function enterClosure(Expr\Closure|Expr\ArrowFunction $closure, array $names, Environment $environment): void
268    {
269        $types = [];
270        foreach ($closure->params as $parameter) {
271            if ($parameter->var instanceof Expr\Variable && is_string($parameter->var->name)) {
272                $types[$parameter->var->name] = $this->types->read($parameter->type);
273            }
274        }
275        foreach ($names as $name) {
276            if (isset($types[$name])) {
277                $environment->write($name, Domain::opaque($types[$name], Origin::Parameter, '$' . $name));
278            } else {
279                $environment->markAbsent($name);
280            }
281        }
282    }

284    /**
285     * Binds what a `global`, `static` or `unset` declaration gives the names it lists.
286     */
287    public function declare(Stmt\Global_|Stmt\Static_|Stmt\Unset_ $declaration, Environment $environment): void
288    {
289        if ($declaration instanceof Stmt\Unset_) {
290            foreach ($declaration->vars as $variable) {
291                if ($variable instanceof Expr\Variable && is_string($variable->name)) {
292                    $environment->markAbsent($variable->name);
293                }
294            }
295
296            return;
297        }
298        foreach ($this->modified->own($declaration) as $name => $_) {
299            $className = $declaration instanceof Stmt\Global_ ? $this->globals->classOf($declaration, $name) : null;
300            $environment->write($name, $className !== null
301                ? Domain::of(new ObjectTerm($className))
302                : Domain::opaque(TypeShape::unknown(), Origin::Unresolved, ($declaration instanceof Stmt\Global_ ? 'global $' : 'static $') . $name));
303        }
304    }

306    /**
307     * Binds a `foreach`'s variables for one pass.
308     *
309     * An array written out in full gives each pass its own element; anything
310     * else gives every pass whatever an element of it can be.
311     */
312    public function iterate(
313        Stmt\Foreach_ $loop,
314        Domain $iterable,
315        int $pass,
316        Environment $environment,
317        FunctionScope $scope,
318        ExpressionEvaluator $expressions,
319    ): void {
320        $array = $iterable->soleArray();
321        $entry = $array === null ? null : ($array->entries[$pass] ?? null);
322        $value = $entry === null ? $this->anyElement($iterable) : $entry->value;
323        $key = $entry === null
324            ? Domain::opaque(TypeShape::of(['int', 'string']), Origin::Loop, 'iterated key')
325            : ($entry->key ?? Domain::literal($pass));
326
327        $expressions->references()->assign($loop->valueVar, $value, $environment, $scope, $expressions);
328        if ($loop->keyVar !== null) {
329            $expressions->references()->assign($loop->keyVar, $key, $environment, $scope, $expressions);
330        }
331    }

333    /**
334     * Whatever an element of an iterable can be.
335     */
336    public function anyElement(Domain $iterable): Domain
337    {
338        $array = $iterable->soleArray();
339        if ($array === null || $array->entries === []) {
340            return Domain::opaque(TypeShape::unknown(), Origin::Loop, 'iterated value');
341        }
342        $values = $array->entries[0]->value;
343        foreach (array_slice($array->entries, 1) as $entry) {
344            $values = $values->union($entry->value);
345        }
346
347        return $array->complete ? $values : $values->union(Domain::opaque(TypeShape::unknown(), Origin::Loop, 'iterated value'));
348    }

350    /**
351     * Runs one assignment, including the kinds the expression evaluator does not write back.
352     */
353    public function assign(Expr $assignment, Environment $environment, FunctionScope $scope, ExpressionEvaluator $expressions): void
354    {
355        $value = $expressions->evaluate($assignment, $environment, $scope);
356        $target = $assignment instanceof Expr\Assign || $assignment instanceof Expr\AssignOp
357            || $assignment instanceof Expr\AssignRef || $assignment instanceof Expr\PreInc
358            || $assignment instanceof Expr\PostInc || $assignment instanceof Expr\PreDec
359            || $assignment instanceof Expr\PostDec ? $assignment->var : null;
360        if ($target === null || $assignment instanceof Expr\Assign || $assignment instanceof Expr\AssignOp\Concat) {
361            return;
362        }
363        $written = $assignment instanceof Expr\AssignRef
364            ? $value
365            : Domain::opaque($value->type(), Origin::Unresolved, $this->text->render($assignment));
366        $expressions->references()->assign($target, $written, $environment, $scope, $expressions);
367    }

369    /**
370     * The environment split into one per value of the names just written, when any has several.
371     *
372     * Bound each name's expansion before splitting the next one. Waiting until
373     * every name has been split materializes their full Cartesian product,
374     * even though only a handful of runs will survive the bound.
375     *
376     * @param array<string, true> $names
377     * @return list<Environment>
378     */
379    public function split(Environment $environment, array $names, int $limit = self::MAX_RUNS): array
380    {
381        if ($limit < 2) {
382            return [$environment];
383        }
384        $environments = [$environment];
385        foreach ($names as $name => $_) {
386            if (!$environment->has($name)) {
387                continue;
388            }
389            $domain = $environment->read($name);
390            if ($domain->widened || count($domain->terms) < 2) {
391                continue;
392            }
393            $split = [];
394            foreach ($environments as $current) {
395                foreach ($domain->terms as $term) {
396                    $one = $current->copy();
397                    $one->narrow($name, Domain::of($term));
398                    $split[] = $one;
399                }
400            }
401            $environments = $this->bound($split, $limit);
402        }
403
404        return $this->bound($environments, $limit);
405    }

407    /**
408     * The environments with repeats dropped and the rest joined once there are too many to keep apart.
409     *
410     * @param list<Environment> $environments
411     * @return list<Environment>
412     */
413    public function bound(array $environments, int $limit = self::MAX_RUNS): array
414    {
415        $unique = [];
416        foreach ($environments as $environment) {
417            $unique[$environment->signature()] ??= $environment;
418        }
419        $unique = array_values($unique);
420        if (count($unique) <= $limit) {
421            return $unique;
422        }
423        $kept = array_slice($unique, 0, $limit - 1);
424        $joined = null;
425        foreach (array_slice($unique, $limit - 1) as $environment) {
426            $joined = $joined === null ? $environment : $joined->join($environment);
427        }
428        if ($joined !== null) {
429            $kept[] = $joined;
430        }
431
432        return $kept;
433    }
