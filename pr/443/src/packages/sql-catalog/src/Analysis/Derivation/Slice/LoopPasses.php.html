14/**
15 * Walks back through a loop one pass at a time.
16 *
17 * A loop that builds part of a statement produces a different statement for
18 * every number of times it runs, so each number of passes is its own path: the
19 * statement as it is before the loop, after one pass, after two, up to the
20 * budget's limit. When a path could still have gone round again at the limit,
21 * it is marked as cut short, because the statements listed are not all of the
22 * ones the loop can produce. A `foreach` over an array written out in full runs
23 * exactly as many times as the array has elements, and only that number is
24 * taken.
25 *
26 * @visibility root
27 */
28final class LoopPasses
29{

32    private FreeNames $names;

41     * Wires the loop walk to the slicer it walks bodies with.
42     */
43    public function __construct(

45        FreeNames $names,
46        ModifiedNames $modified,
47        EvaluationBudget $budget,
48    ) {
49        $this->slicer = $slicer;
50        $this->names = $names;
51        $this->modified = $modified;
52        $this->budget = $budget;
53        $this->steps = new AssignmentSteps($names, $modified);
54    }

56    /**
57     * Whether the node is a loop.
58     */
59    public function isLoop(Node $node): bool
60    {
61        return $node instanceof Stmt\While_ || $node instanceof Stmt\Do_
62            || $node instanceof Stmt\For_ || $node instanceof Stmt\Foreach_;
63    }

65    /**
66     * The runs a loop before the point may have made, as paths that start where the loop finishes.
67     *
68     * @param array<string, true> $needs
69     * @return list<Pending>
70     */
71    public function runs(Stmt $loop, array $needs): array
72    {
73        $exact = $this->writtenPasses($loop);
74        $runs = [];
75        $current = [Pending::needing($needs)];
76        if (!$loop instanceof Stmt\Do_ && ($exact === null || $exact === 0)) {
77            $runs = $this->entering($loop, $current);
78        }
79        for ($pass = 1; $pass <= $this->budget->maxLoopPasses; $pass++) {
80            $current = $this->pass($loop, $current);
81            if ($exact === null || $exact === $pass) {
82                $runs = array_merge($runs, $this->entering($loop, $current));
83            }
84            if (!$this->continues($loop, $current)) {
85                return $runs;
86            }
87        }
88        if ($exact !== null && $exact <= $this->budget->maxLoopPasses) {
89            return $runs;
90        }
91
92        return array_map(static fn (Pending $run): Pending => $run->cut(), $runs);
93    }
94
95    /**
96     * The paths carried out of the start of a loop body the point is written in.
97     *
98     * @param list<Pending> $paths Paths standing at the start of the pass the point is in
99     * @return list<Arrival>
100     */
101    public function leaveBody(Stmt $loop, array $paths): array
102    {
103        $arrivals = [];
104        $current = array_map(fn (Pending $path): Pending => $this->binding($loop, $path), $paths);
105        for ($pass = 0; ; $pass++) {
106            $continues = $this->continues($loop, $current);
107            $last = $pass >= $this->budget->maxLoopPasses || !$continues;
108            $entered = $last && $continues ? array_map(static fn (Pending $path): Pending => $path->cut(), $current) : $current;
109            $arrivals = array_merge($arrivals, $this->slicer->before($loop, $this->entering($loop, $entered)));
110            if ($last) {
111                return $arrivals;
112            }
113            $current = $this->pass($loop, $current);
114        }
115    }

117    /**
118     * The paths walked back over one whole pass, from its end to its start.
119     *
120     * @param list<Pending> $paths
121     * @return list<Pending>
122     */
123    public function pass(Stmt $loop, array $paths): array
124    {
125        if ($loop instanceof Stmt\For_) {
126            $paths = $this->expressions($loop->loop, $paths);
127        }
128        $body = $this->bodyOf($loop);
129        $paths = $this->slicer->walkList($body, count($body), $paths);
130
131        return array_map(fn (Pending $path): Pending => $this->binding($loop, $path), $paths);
132    }

139    public function bodyOf(Stmt $loop): array
140    {
141        if ($loop instanceof Stmt\While_ || $loop instanceof Stmt\Do_
142            || $loop instanceof Stmt\For_ || $loop instanceof Stmt\Foreach_) {
143            return array_values($loop->stmts);
144        }

149    /**
150     * A path walked back over the start of one pass of a `foreach`, where its variables are set.
151     */
152    public function binding(Stmt $loop, Pending $path): Pending
153    {
154        if (!$loop instanceof Stmt\Foreach_ || $path->isDone()) {
155            return $path;
156        }
157        $written = $this->modified->own($loop);
158        if (array_intersect_key($written, $path->needs) === []) {
159            return $path;
160        }
161
162        return $path->through(new SliceStep($loop), array_diff_key($path->needs, $written));
163    }

165    /**
166     * The paths walked back over what runs before a loop's first pass.
167     *
168     * A `foreach` reads the array it goes over once, before the first pass,
169     * whatever the passes then do to the variable holding it; so that is where
170     * the array is looked for, and only on a path that needed the loop's
171     * variables at all.
172     *
173     * @param list<Pending> $paths
174     * @return list<Pending>
175     */
176    public function entering(Stmt $loop, array $paths): array
177    {
178        if ($loop instanceof Stmt\For_) {
179            return $this->expressions($loop->init, $paths);
180        }
181        if (!$loop instanceof Stmt\Foreach_) {
182            return $paths;
183        }
184
185        return array_map(
186            fn (Pending $path): Pending => $this->bound($loop, $path)
187                ? new Pending($path->steps, $path->needs + $this->names->read($loop->expr), $path->truncated, $path->exhausted)
188                : $path,
189            $paths,
190        );
191    }

193    /**
194     * Whether a path took one of a `foreach`'s passes as a step.
195     */
196    public function bound(Stmt\Foreach_ $loop, Pending $path): bool
197    {
198        foreach ($path->steps as $step) {
199            if ($step->node === $loop) {
200                return true;
201            }
202        }
203
204        return false;
205    }

207    /**
208     * The paths walked back over a list of expressions, last first.
209     *
210     * @param array<array-key, Expr> $expressions
211     * @param list<Pending> $paths
212     * @return list<Pending>
213     */
214    public function expressions(array $expressions, array $paths): array
215    {
216        foreach (array_reverse($expressions) as $expression) {
217            $paths = array_map(
218                fn (Pending $path): Pending => $path->isDone() ? $path : $this->steps->over($expression, $path),
219                $paths,
220            );
221        }
222
223        return $paths;
224    }

226    /**
227     * Whether another pass could still change something a path needs.
228     *
229     * @param list<Pending> $paths
230     */
231    public function continues(Stmt $loop, array $paths): bool
232    {
233        foreach ($paths as $path) {
234            if (!$path->isDone() && $this->modified->touches($loop, $path->needs)) {
235                return true;
236            }
237        }
238
239        return false;
240    }

242    /**
243     * How many times a `foreach` over an array written out in full runs, or null when that is not known.
244     */
245    public function writtenPasses(Stmt $loop): ?int
246    {
247        if (!$loop instanceof Stmt\Foreach_ || !$loop->expr instanceof Expr\Array_) {
248            return null;
249        }
250        foreach ($loop->expr->items as $item) {
251            if ($item->unpack) {
252                return null;
253            }
254        }
255
256        return count($loop->expr->items);
257    }
