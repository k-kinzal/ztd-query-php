12/**
13 * Every call in the source, by the name it is written with.
14 *
15 * Going from a parameter to the arguments passed for it means finding the
16 * calls that reach the function. The calls are gathered once, by name, so
17 * that finding them is a lookup; deciding which of them really reach the
18 * function is left to whoever asks, because that can need their receivers.
19 *
20 * @visibility root
21 */
22final class CallerIndex
23{
24    /**
25     * @var array<string, list<Expr\CallLike>>
26     */
27    private array $calls = [];
28
29    /**
30     * @var array<string, list<Expr\New_>>
31     */
32    private array $instantiations = [];
33
34    /**
35     * Gathers the calls of every file.
36     *
37     * @param list<ParsedFile> $files
38     */
39    public function __construct(array $files = [])
40    {
41        $finder = new NodeFinder();
42        foreach ($files as $file) {
43            /** @var list<Expr\CallLike> $calls */
44            $calls = $finder->findInstanceOf($file->statements, Expr\CallLike::class);
45            foreach ($calls as $call) {
46                $this->add($call);
47            }
48        }
49    }
50
51    /**
52     * Files one call under the name it is written with.
53     */
54    public function add(Expr\CallLike $call): void
55    {
56        if ($call instanceof Expr\New_) {
57            if ($call->class instanceof Node\Name) {
58                $this->instantiations[$this->shortName($call->class->toString())][] = $call;
59            }
60
61            return;
62        }
63        $name = $this->nameOf($call);
64        if ($name !== null) {
65            $this->calls[$this->shortName($name)][] = $call;
66        }
67    }
68
69    /**
70     * The calls written with that name, whatever they are called on.
71     *
72     * @return list<Expr\CallLike>
73     */
74    public function named(string $name): array
75    {
76        return $this->calls[$this->shortName($name)] ?? [];
77    }
78
79    /**
80     * The instantiations written with that class name.
81     *
82     * @return list<Expr\New_>
83     */
84    public function instantiating(string $className): array
85    {
86        return $this->instantiations[$this->shortName($className)] ?? [];
87    }
88
89    /**
90     * The name a call is written with, or null when it is not written as a name.
91     */
92    public function nameOf(Expr\CallLike $call): ?string
93    {
94        if ($call instanceof Expr\FuncCall) {
95            return $call->name instanceof Node\Name ? $call->name->toString() : null;
96        }
97        if ($call instanceof Expr\MethodCall || $call instanceof Expr\NullsafeMethodCall || $call instanceof Expr\StaticCall) {
98            return $call->name instanceof Node\Identifier ? $call->name->toString() : null;
99        }
100
101        return null;
102    }
103
104    /**
105     * A name without its namespace, in lower case.
106     */
107    public function shortName(string $name): string
108    {
109        $parts = explode('\\', ltrim($name, '\\'));
110
111        return strtolower($parts[count($parts) - 1]);
112    }
