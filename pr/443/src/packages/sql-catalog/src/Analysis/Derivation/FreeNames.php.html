13/**
14 * The names an expression reads, which are what its value depends on.
15 *
16 * A variable is named as itself; a property of the object a method runs on is
17 * named `this->property`, because an assignment to it earlier in the body is a
18 * definition in the same way an assignment to a variable is; the object itself
19 * is named `this`. Superglobals are not named: nothing in the source defines
20 * them, so there is nothing to look for.
21 *
22 * A database call written inside the expression reads less than it passes.
23 * What a preparing or composing call hands back depends on the statement text
24 * alone, and what a query hands back does not depend on its arguments at all,
25 * so the values it binds are not looked for. Without this, a statement built
26 * from `$wpdb->prepare('(%s, %s)', $option, $value)` would chase every value it
27 * binds through every function that computes one, and none of that can change
28 * the text.
29 *
30 * @visibility root
31 */
32final class FreeNames
33{

48    /**
49     * Builds the reader over the database calls whose arguments it reads selectively.
50     *
51     * @param list<SinkSpec> $sinks
52     */
53    public function __construct(array $sinks = [], ?ExternalInput $external = null, bool $trackObjectEffects = false)
54    {
55        $this->external = $external ?? new ExternalInput();
56        $this->trackObjectEffects = $trackObjectEffects;
57        foreach ($sinks as $sink) {
58            $this->trackObjectEffects = $this->trackObjectEffects || $sink->role === SinkRole::Modelled;
59            $name = strtolower(ltrim($sink->name, '\\'));
60            $positions = $this->sinkArguments[$name] ?? [];
61            if (($sink->role === SinkRole::Compose || $sink->role === SinkRole::Prepare) && $sink->sqlParameter !== null) {
62                $positions[$sink->sqlParameter] = true;
63            }
64            $this->sinkArguments[$name] = $positions;
65        }
66    }

84    /**
85     * The names one node reads.
86     *
87     * @return array<string, true>
88     */
89    public function read(Node $node): array
90    {
91        if ($node instanceof Expr\Variable) {
92            return $this->variable($node);
93        }
94        if ($node instanceof Expr\Assign || $node instanceof Expr\AssignRef) {
95            return $this->read($node->expr) + $this->targetReads($node->var);
96        }
97        $property = $this->propertyName($node);
98        if ($property !== null) {
99            return [$property => true];
100        }
101        if ($node instanceof Expr\Closure) {
102            $names = [];
103            foreach ($node->uses as $use) {
104                $names += $this->variable($use->var);
105            }
106
107            return $names;
108        }
109        if ($node instanceof Expr\ArrowFunction) {
110            return $this->arrowFunction($node);
111        }
112        if ($node instanceof Expr\CallLike) {
113            $positions = $this->sinkArgumentsOf($node);
114            if ($positions !== null) {
115                return $this->sinkCall($node, $positions);
116            }
117        }
118
119        $names = [];
120        foreach (get_object_vars($node) as $sub) {
121            foreach (is_array($sub) ? $sub : [$sub] as $child) {
122                if ($child instanceof Node) {
123                    $names += $this->read($child);
124                }
125            }
126        }
127
128        return $names;
129    }

131    /**
132     * The argument positions whose values decide what a database call hands back, or null when the call is not written as one.
133     *
134     * @return array<int, true>|null
135     */
136    public function sinkArgumentsOf(Expr\CallLike $call): ?array
137    {
138        if ($this->trackObjectEffects) {
139            return null;
140        }
141        $name = null;
142        if ($call instanceof Expr\MethodCall || $call instanceof Expr\NullsafeMethodCall || $call instanceof Expr\StaticCall) {
143            $name = $call->name instanceof Node\Identifier ? $call->name->toString() : null;
144        }
145        if ($call instanceof Expr\FuncCall && $call->name instanceof Node\Name) {
146            $name = $call->name->toString();
147        }
148
149        return $name === null ? null : ($this->sinkArguments[strtolower(ltrim($name, '\\'))] ?? null);
150    }

152    /**
153     * The names a database call reads: what it is called on, and the arguments its result depends on.
154     *
155     * @param array<int, true> $positions
156     * @return array<string, true>
157     */
158    public function sinkCall(Expr\CallLike $call, array $positions): array
159    {
160        $names = $call instanceof Expr\MethodCall || $call instanceof Expr\NullsafeMethodCall ? $this->read($call->var) : [];
161        if ($call->isFirstClassCallable()) {
162            return $names;
163        }
164        foreach ($call->getArgs() as $position => $argument) {
165            if (isset($positions[$position]) || $argument->unpack) {
166                $names += $this->read($argument->value);
167            }
168        }
169
170        return $names;
171    }

173    /**
174     * The names an assignment target reads without being what it writes.
175     *
176     * Writing `$rows[$key]` reads `$key`, and reads `$rows` too, since the
177     * element is added to whatever the array already held. Writing `$sql` reads
178     * nothing: the old value is simply replaced.
179     *
180     * @return array<string, true>
181     */
182    public function targetReads(Node $target): array
183    {
184        if ($this->trackObjectEffects && $target instanceof Expr\PropertyFetch) {
185            return $this->read($target->var);
186        }
187        if ($target instanceof Expr\ArrayDimFetch) {
188            return $this->read($target);
189        }
190        if ($target instanceof Expr\List_ || $target instanceof Expr\Array_) {
191            $names = [];
192            foreach ($target->items as $item) {
193                if ($item !== null) {
194                    $names += ($item->key === null ? [] : $this->read($item->key)) + $this->targetReads($item->value);
195                }
196            }
197
198            return $names;
199        }
200
201        return [];
202    }

204    /**
205     * The name a variable is read under, or nothing when it is external or not a plain name.
206     *
207     * @return array<string, true>
208     */
209    public function variable(Expr\Variable $node): array
210    {
211        if (!is_string($node->name) || $this->external->isVariable($node->name)) {
212            return [];
213        }
214
215        return [$node->name => true];
216    }

218    /**
219     * The tracked name of a property of `$this`, or null when the node is not one.
220     */
221    public function propertyName(Node $node): ?string
222    {
223        if (!$node instanceof Expr\PropertyFetch && !$node instanceof Expr\NullsafePropertyFetch) {
224            return null;
225        }
226        if (!$node->var instanceof Expr\Variable || $node->var->name !== self::THIS) {
227            return null;
228        }
229
230        return $node->name instanceof Node\Identifier ? self::THIS . '->' . $node->name->toString() : null;
231    }
232
233    /**
234     * The names an arrow function captures from the scope it is written in.
235     *
236     * @return array<string, true>
237     */
238    public function arrowFunction(Expr\ArrowFunction $node): array
239    {
240        $names = $this->read($node->expr);
241        foreach ($node->params as $parameter) {
242            if ($parameter->var instanceof Expr\Variable && is_string($parameter->var->name)) {
243                unset($names[$parameter->var->name]);
244            }
245        }
246
247        return $names;
248    }
