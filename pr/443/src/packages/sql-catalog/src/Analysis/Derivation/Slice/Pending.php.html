7/**
8 * One path being walked back from a call: what it passed through, and what it still needs.
9 *
10 * The steps are held nearest-first, in the order the walk finds them, and
11 * turned around only when the path is run. What is still needed is the set of
12 * names no step on the path has defined yet; when it is empty, nothing earlier
13 * in the program can change the argument and the walk along this path is done.
14 *
15 * @visibility root
16 */
17final class Pending
18{
19    /**
20     * @param list<SliceStep> $steps The steps found so far, nearest to the call first
21     * @param array<string, true> $needs The names nothing on the path has defined yet
22     * @param bool $truncated Whether a bound cut the path short of every way it could have gone
23     * @param bool $exhausted Whether the budget ran out before the path was walked to its start
24     */
25    public function __construct(
26        public readonly array $steps,
27        public readonly array $needs,
28        public readonly bool $truncated = false,
29        public readonly bool $exhausted = false,
30    ) {
31    }

35     *

37     */

40        return new self([], $needs);
41    }

103    public function signature(): string
104    {
105        $needs = array_keys($this->needs);
106        sort($needs);
107
108        return implode(',', $needs) . '#' . implode(',', array_map(

114    /**
115     * Several paths gathered into one, when there are more than it pays to keep apart.
116     *
117     * The paths are kept as alternatives of a single step rather than joined
118     * value by value, so running it still runs each of them; what is given up is
119     * only keeping them apart from the paths that come before them.
120     *
121     * @param non-empty-list<self> $paths
122     */
123    public static function gather(array $paths): self
124    {
125        $needs = [];
126        $alternatives = [];
127        $truncated = false;
128        $exhausted = false;
129        foreach ($paths as $path) {
130            $needs += $path->needs;
131            $alternatives[] = $path->forward();
132            $truncated = $truncated || $path->truncated;
133            $exhausted = $exhausted || $path->exhausted;
134        }
135
136        return new self([SliceStep::either($alternatives)], $needs, $truncated, $exhausted);
137    }
