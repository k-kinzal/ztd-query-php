14/**
15 * What a property of an object can hold when a method starts, read from everywhere the class writes it.
16 *
17 * A method that reads `$this->table` without writing it first sees whatever
18 * the object was left holding: the declared default, or the value some
19 * method of the class assigned — most often the constructor, from an
20 * argument the code that creates the object passes. Each of those writes is
21 * worked out at the place it is written, like any other value, so a table
22 * name handed to a repository's constructor resolves where the repository
23 * issues its queries.
24 *
25 * @visibility root
26 */
27final class PropertyWrites
28{

52        $this->modified = $modified ?? new ModifiedNames();

55    /**
56     * Every value the property can hold when a method of the class starts.
57     *
58     * Each method that writes the property is read at the point it finishes,
59     * so a property built up in steps — assigned, then appended to under a
60     * condition — is seen as each of the values it can be left holding rather
61     * than as whichever assignment happens to come last in the source.
62     *
63     * @return list<Domain>
64     */
65    public function valuesOf(string $className, string $property, int $depth, Deriver $deriver): array
66    {
67        $key = strtolower($className) . '->' . $property;
68        if (isset($this->reading[$key])) {
69            return [];
70        }
71        $this->reading[$key] = true;
72        $values = [];
73        $goal = new Expr\PropertyFetch(new Expr\Variable(FreeNames::THIS), $property);
74        foreach ($this->writersOf($className, $property) as $method) {
75            if (!$deriver->spent()) {
76                foreach ($deriver->solveAtExits($method, [$goal], $depth + 1) as $solution) {
77                    $values[] = $solution->values[0];
78                }
79            }
80        }
81        foreach ($this->promotions($className, $property) as $constructor) {
82            $values = array_merge($values, $deriver->binder()->parameterValues($constructor, $property, $depth + 1, $deriver));
83        }
84        $default = $this->defaultOf($className, $property);
85        if ($default !== null) {
86            $values[] = $deriver->binder()->evaluateConstant($default, $className);
87        }
88        unset($this->reading[$key]);
89
90        return $values;
91    }
92
93    /**
94     * The methods of the class and the classes it inherits from that write the property.
95     *
96     * @return list<Stmt\ClassMethod>
97     */
98    public function writersOf(string $className, string $property): array
99    {
100        $key = strtolower($className) . '->' . $property;
101        if (isset($this->writers[$key])) {
102            return $this->writers[$key];
103        }
104        $found = [];
105        foreach ($this->index->lineage($className) as $shape) {
106            foreach ($shape->methods as $method) {
107                $node = $method->node;
108                if ($node instanceof Stmt\ClassMethod && $this->writes($node, $property)) {
109                    $found[] = $node;
110                }
111            }
112        }
113        $this->writers[$key] = $found;
114
115        return $found;
116    }
117
118    /**
119     * Whether a method writes the property of `$this` anywhere in its body.
120     */
121    public function writes(Stmt\ClassMethod $method, string $property): bool
122    {
123        $name = FreeNames::THIS . '->' . $property;
124        foreach ($this->finder->find($method->stmts ?? [], static fn (Node $node): bool => $node instanceof Expr\Assign || $node instanceof Expr\AssignOp || $node instanceof Expr\AssignRef) as $assignment) {
125            if ($assignment instanceof Expr\Assign || $assignment instanceof Expr\AssignOp || $assignment instanceof Expr\AssignRef) {
126                if ($this->modified->baseName($assignment->var) === $name) {
127                    return true;
128                }
129            }
130        }
131
132        return false;
133    }
134
135    /**
136     * The constructors that promote a parameter of that name to the property.
137     *
138     * @return list<Stmt\ClassMethod>
139     */
140    public function promotions(string $className, string $property): array
141    {
142        $constructors = [];
143        foreach ($this->index->lineage($className) as $shape) {
144            $constructor = $shape->methods['__construct'] ?? null;
145            $node = $constructor?->node;
146            if (!$node instanceof Stmt\ClassMethod) {
147                continue;
148            }
149            foreach ($node->params as $parameter) {
150                if ($parameter->flags !== 0 && $parameter->var instanceof Expr\Variable && $parameter->var->name === $property) {
151                    $constructors[] = $node;
152                }
153            }
154        }
155
156        return $constructors;
157    }
158
159    /**
160     * The default the property is declared with, or null when it is declared without one.
161     */
162    public function defaultOf(string $className, string $property): ?Expr
163    {
164        foreach ($this->index->lineage($className) as $shape) {
165            if (isset($shape->propertyDefaults[$property])) {
166                return $shape->propertyDefaults[$property];
167            }
168        }
169
170        return null;
171    }
