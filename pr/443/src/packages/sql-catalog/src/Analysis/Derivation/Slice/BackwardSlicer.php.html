16/**
17 * Walks back from a call to the start of its body, keeping only what its argument depends on.
18 *
19 * The walk starts with the names the argument reads and goes back statement by
20 * statement. An assignment to one of those names is kept as a step, and the
21 * names its right-hand side reads are looked for instead. A branch that assigns
22 * none of them is stepped over whole; one that does splits the path into one
23 * path per arm, which is what keeps the values an arm assigns together. A loop
24 * becomes one path per number of passes, up to the budget's limit. What comes
25 * out at the start of the body is the set of paths, each with the steps that
26 * lead from there to the call and the names still left to find.
27 *
28 * @visibility root
29 */
30final class BackwardSlicer

70     * Every path from the start of the body to the point, with what each still needs.
71     *
72     * @param array<string, true> $needs The names the value at the point reads
73     * @return list<Arrival>
74     */
75    public function sliceFrom(Node $point, array $needs): array
76    {
77        $start = [Pending::needing($needs)];
78        $anchor = $this->tree->anchorOf($point);
79        if ($anchor instanceof FunctionLike) {
80            return $this->leave($anchor, $start);
81        }
82
83        return $anchor === null ? [new Arrival(null, $start[0])] : $this->before($anchor, $start);
84    }

87     * Every path from the start of a body to its end, with what each still needs.
88     *
89     * @param array<string, true> $needs The names read at the end of the body

92    public function sliceFromEnd(FunctionLike $body, array $needs): array
93    {
94        $statements = array_values($body->getStmts() ?? []);
95
96        return $this->leave($body, $this->walkList($statements, count($statements), [Pending::needing($needs)]));
97    }

101     *

103     * @return list<Arrival>

105    public function before(Stmt $statement, array $paths): array
106    {
107        $location = $this->tree->locate($statement);
108        if ($location === null) {
109            return $this->arrive($this->tree->bodyOf($statement), $paths);
110        }
111        [$owner, $list, $index] = $location;
112
113        return $this->leave($owner, $this->walkList($list, $index, $paths));

116    /**
117     * The paths carried out of the start of a statement list, into whatever owns it.
118     *
119     * @param list<Pending> $paths
120     * @return list<Arrival>
121     */
122    public function leave(?Node $owner, array $paths): array
123    {
124        if ($owner === null || $owner instanceof Stmt\Namespace_ || $owner instanceof Stmt\Declare_) {
125            return $this->arrive(null, $paths);
126        }
127        if ($owner instanceof Stmt\ClassMethod || $owner instanceof Stmt\Function_) {
128            return $this->arrive($owner, $paths);
129        }
130        if ($owner instanceof Expr\Closure || $owner instanceof Expr\ArrowFunction) {
131            return $this->leaveClosure($owner, $paths);
132        }
133        if ($owner instanceof Stmt && $this->loops->isLoop($owner)) {
134            return $this->loops->leaveBody($owner, $paths);
135        }
136        if ($owner instanceof Stmt\Finally_) {
137            $try = $owner->getAttribute('parent');
138            if ($try instanceof Stmt\TryCatch) {
139                return $this->before($try, $this->walkList(array_values($try->stmts), count($try->stmts), $paths));
140            }
141        }
142        $statement = $this->owningStatement($owner);
143
144        return $statement === null ? $this->arrive($this->tree->bodyOf($owner), $paths) : $this->before($statement, $paths);
145    }

151    {
152        if ($owner instanceof Stmt\ElseIf_ || $owner instanceof Stmt\Else_ || $owner instanceof Stmt\Case_
153            || $owner instanceof Stmt\Catch_) {
154            $parent = $owner->getAttribute('parent');

162    /**
163     * The paths carried out of a closure into the body it is written in.
164     *
165     * What the closure takes from outside — its `use` list, or for an arrow
166     * function every name that is not one of its parameters — is looked for
167     * where the closure is written. Its parameters, and any name it reads
168     * without defining, are its own: a step says so, and the walk does not look
169     * for them outside.
170     *
171     * @param list<Pending> $paths
172     * @return list<Arrival>
173     */
174    public function leaveClosure(Expr\Closure|Expr\ArrowFunction $closure, array $paths): array
175    {
176        $carried = [];
177        foreach ($paths as $path) {
178            $outside = $this->outside($closure, $path->needs);
179            $own = array_keys(array_diff_key($path->needs, $outside));
180            $carried[] = $own === [] ? $path : $path->through(new SliceStep($closure, [], $own), $outside);
181        }
182        $anchor = $this->tree->anchorOf($closure);
183        if ($anchor instanceof FunctionLike) {
184            return $this->leave($anchor, $carried);
185        }
186
187        return $anchor === null ? $this->arrive(null, $carried) : $this->before($anchor, $carried);
188    }

190    /**
191     * The names among those needed that a closure takes from where it is written.
192     *
193     * @param array<string, true> $needs
194     * @return array<string, true>
195     */
196    public function outside(Expr\Closure|Expr\ArrowFunction $closure, array $needs): array
197    {
198        $parameters = [];
199        foreach ($closure->params as $parameter) {
200            if ($parameter->var instanceof Expr\Variable && is_string($parameter->var->name)) {
201                $parameters[$parameter->var->name] = true;
202            }
203        }
204        if ($closure instanceof Expr\ArrowFunction) {
205            return array_diff_key($needs, $parameters);
206        }
207        $uses = [FreeNames::THIS => true];
208        foreach ($closure->uses as $use) {
209            if (is_string($use->var->name)) {
210                $uses[$use->var->name] = true;
211            }
212        }
213
214        return array_intersect_key($needs, $uses);
215    }

224    {
225        $named = $body;
226        while ($named instanceof Expr\Closure || $named instanceof Expr\ArrowFunction) {
227            $named = $this->tree->bodyOf($named);
228        }

230        return array_map(static fn (Pending $path): Arrival => new Arrival($named, $path), $paths);

233    /**
234     * The paths walked back over the statements of a list, from the given position to its start.
235     *
236     * @param list<Stmt> $list
237     * @param list<Pending> $paths
238     * @return list<Pending>
239     */
240    public function walkList(array $list, int $end, array $paths): array
241    {
242        for ($index = $end - 1; $index >= 0; $index--) {
243            if ($this->allDone($paths)) {
244                break;
245            }
246            $next = [];
247            foreach ($paths as $path) {
248                $next = array_merge($next, $path->isDone() ? [$path] : $this->over($list[$index], $path));
249            }
250            $paths = $this->bound($next);
251        }
252
253        return $paths;
254    }

272    /**
273     * One path walked back over one statement.
274     *
275     * @return list<Pending>
276     */
277    public function over(Stmt $statement, Pending $path): array
278    {
279        if (!$this->budget->spend()) {
280            return [$path->exhaust()];
281        }
282        if (!$this->modified->touches($statement, $path->needs)) {
283            return [$path];
284        }
285        if ($statement instanceof Stmt\Expression) {
286            return [$this->steps->over($statement->expr, $path)];
287        }
288        if ($statement instanceof Stmt\Return_ && $statement->expr !== null && $this->modified->tracksObjects()) {
289            return [$this->steps->over($statement->expr, $path)];
290        }
291        if ($statement instanceof Stmt\Global_ || $statement instanceof Stmt\Static_ || $statement instanceof Stmt\Unset_) {
292            return [$this->steps->declaration($statement, $path)];
293        }
294        if ($statement instanceof Stmt\Block) {
295            return $this->walkList(array_values($statement->stmts), count($statement->stmts), [$path]);
296        }
297        if ($this->loops->isLoop($statement)) {
298            return $this->branch($path, $this->loops->runs($statement, $path->needs));
299        }
300        $arms = $this->arms->of($statement);
301        if ($arms === null) {
302            return [$path];
303        }
304        $paths = $this->branch($path, $this->walkArms($arms, $path->needs));
305        if ($statement instanceof Stmt\If_ || $statement instanceof Stmt\Switch_) {
306            $paths = array_map(fn (Pending $taken): Pending => $this->steps->over($statement->cond, $taken), $paths);
307        }
308
309        return $paths;
310    }

312    /**
313     * Each arm walked back from its end, as paths that start where the branch finishes.
314     *
315     * @param list<list<Stmt>> $arms
316     * @param array<string, true> $needs
317     * @return list<Pending>
318     */
319    public function walkArms(array $arms, array $needs): array
320    {
321        $runs = [];
322        foreach ($arms as $arm) {
323            $runs = array_merge($runs, $this->walkList($arm, count($arm), [Pending::needing($needs)]));
324        }

326        return $runs;

329    /**
330     * A path taken on over the runs a branch may have taken.
331     *
332     * The runs become one step that holds them as alternatives, rather than
333     * one path each. Running the step still runs every arm on its own, so the
334     * values an arm assigns stay together; what is saved is walking the rest of
335     * the body once per arm, which for a body of ten consecutive conditionals
336     * would be a thousand walks.
337     *
338     * @param list<Pending> $runs Paths that start where the branch finishes
339     * @return list<Pending>
340     */
341    public function branch(Pending $path, array $runs): array
342    {
343        if ($runs === []) {
344            return [$path];
345        }
346
347        return [$path->then(count($runs) === 1 ? $runs[0] : Pending::gather($runs))];
348    }

350    /**
351     * The paths with repeats merged and the rest gathered once there are too many to keep apart.
352     *
353     * @param list<Pending> $paths
354     * @return list<Pending>
355     */
356    public function bound(array $paths): array
357    {
358        $unique = [];
359        foreach ($paths as $path) {
360            $unique[$path->signature()] = $path;
361        }
362        $paths = array_values($unique);
363        if (count($paths) <= self::MAX_PATHS) {
364            return $paths;
365        }
366        $kept = array_slice($paths, 0, self::MAX_PATHS - 1);
367        $rest = array_slice($paths, self::MAX_PATHS - 1);
368        if ($rest !== []) {
369            $kept[] = Pending::gather($rest);
370        }
371
372        return $kept;
373    }
