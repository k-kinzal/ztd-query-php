24/**
25 * Binds what a path still needs when it reaches the start of its body.
26 *
27 * A parameter is bound to what each call of the body passes for it, which is
28 * worked out at the call the same way the statement was worked out at the
29 * database call. A property of `$this` is bound to each value the class can
30 * leave it holding. A name read at the top of a file is bound to what an
31 * extension says it holds. Whatever cannot be bound this way is left open,
32 * and says why: nothing calls the body, the budget ran out, or the name is
33 * simply not defined anywhere the analysis can see.
34 *
35 * @visibility root
36 */
37final class EntryBinder

69    /**
70     * The ways in, with those beyond what the budget can run one at a time joined into one.
71     *
72     * Every way in runs the whole path, so a long path with many ways in costs
73     * their product. The ways the budget cannot pay for are joined value by
74     * value into a single way in: every value is still there, and the result
75     * says that its values were paired without knowing they go together.
76     *
77     * @param list<Binding> $bindings
78     * @return list<Binding>
79     */
80    public function affordable(array $bindings, int $limit): array
81    {
82        if (count($bindings) <= $limit) {
83            return $bindings;
84        }
85        $kept = array_slice($bindings, 0, max(0, $limit - 1));
86        $joined = null;
87        $through = [];
88        $truncated = false;
89        foreach (array_slice($bindings, max(0, $limit - 1)) as $binding) {
90            $joined = $joined === null ? $binding->environment : $joined->join($binding->environment);
91            $through = $binding->through;
92            $truncated = $truncated || $binding->truncated;
93        }
94        $kept[] = new Binding($joined ?? new Environment(), $through, $truncated, true);
95
96        return $kept;
97    }

99    /**
100     * The ways into the body a path arrived at the start of, each binding what the path still needs.
101     *
102     * @return list<Binding>
103     */
104    public function bindings(Arrival $arrival, int $depth, Deriver $deriver): array
105    {
106        $bindings = $this->parameterBindings($arrival, $depth, $deriver);
107        $className = $arrival->body instanceof Stmt\ClassMethod && !$arrival->body->isStatic() ? $deriver->classOf($arrival->body) : null;
108        if ($className === null || $this->properties === null || $arrival->path->exhausted) {
109            return $bindings;
110        }
111        $varying = 0;
112        foreach ($arrival->path->needs as $needed => $_) {
113            if (str_starts_with($needed, FreeNames::THIS . '->')) {
114                $values = $this->properties->valuesOf($className, substr($needed, strlen(FreeNames::THIS) + 2), $depth, $deriver);
115                $varying += count($values) > 1 ? 1 : 0;
116                $bindings = $this->withProperty($bindings, $needed, $values, $varying > 1);
117            }
118        }
119
120        return $bindings;
121    }

123    /**
124     * The ways in, one for each value a property can hold on each of them.
125     *
126     * Properties are worked out one at a time, so once more than one of them
127     * can hold several values, the pairings are not known to occur together and
128     * the ways in say so.
129     *
130     * @param list<Binding> $bindings
131     * @param list<Domain> $values
132     * @return list<Binding>
133     */
134    public function withProperty(array $bindings, string $name, array $values, bool $combined = false): array
135    {
136        if ($values === []) {
137            return $bindings;
138        }
139        $split = [];
140        foreach ($bindings as $binding) {
141            foreach ($values as $value) {
142                $environment = $binding->environment->copy();
143                $environment->write($name, $value);
144                $split[] = new Binding($environment, $binding->through, $binding->truncated, $binding->combined || $combined);
145            }
146        }
147        if (count($split) <= Deriver::MAX_SOLUTIONS) {
148            return $split;
149        }
150
151        return array_map(
152            static fn (Binding $binding): Binding => new Binding($binding->environment, $binding->through, true, $binding->combined),
153            array_slice($split, 0, Deriver::MAX_SOLUTIONS),
154        );
155    }

180    /**
181     * The ways into the body, each binding the parameters and other names the path still needs.
182     *
183     * A local with no reaching definition starts absent. Reading it yields null;
184     * its existence remains separate from the values assigned later.
185     * Parameters, declared globals and exhausted paths keep their open values.
186     *
187     * @return list<Binding>
188     */
189    public function parameterBindings(Arrival $arrival, int $depth, Deriver $deriver): array
190    {
191        $body = $arrival->body;
192        $name = $body === null ? FunctionScope::MAIN : $deriver->nameOf($body);
193        $needs = $arrival->path->needs;
194        if ($arrival->path->exhausted) {
195            return [new Binding($this->leaveOpen($needs, Origin::Budget), [$name])];
196        }
197        if ($body === null) {
198            return [new Binding($this->fileScope($needs), [$name])];
199        }
200
201        $parameters = $this->parametersOf($body, $needs);
202        $outside = new Environment();
203        foreach (array_diff_key($needs, $parameters) as $needed => $_) {
204            if ($needed !== FreeNames::THIS && !str_starts_with($needed, FreeNames::THIS . '->')) {
205                $outside->markAbsent($needed);
206            }
207        }
208        if ($parameters === []) {
209            return [new Binding($outside, [$name])];
210        }
211        if ($depth >= $this->budget->maxDepth || $this->budget->isExhausted()) {
212            return [new Binding($this->withParameters($outside, $parameters, Origin::Budget), [$name])];
213        }
214        $callers = $this->callers->of($body, $depth, $deriver);
215        if ($callers->calls === []) {
216            $origin = $callers->partial ? $this->openOrigin() : Origin::Parameter;
217
218            return [new Binding($this->withParameters($outside, $parameters, $origin), [$name], $callers->partial)];
219        }
220
221        $bindings = $this->fromCallers($callers, $parameters, $outside, $name, $depth, $deriver);
222        if ($bindings !== []) {
223            return $bindings;
224        }
225        return [new Binding($this->withParameters($outside, $parameters, $this->openOrigin()), [$name])];
226    }

228    /**
229     * One way in for every way each call can pass the arguments the path needs.
230     *
231     * When not every caller could be asked — there are more than the limit, the
232     * budget ran out partway, or a call might reach the body without that being
233     * certain — every way in found is marked as cut short, not only the ones
234     * found after the search stopped: the set as a whole is what is incomplete.
235     *
236     * @param array<string, array{int, Node\Param}> $parameters
237     * @return list<Binding>
238     */
239    public function fromCallers(CallerSet $callers, array $parameters, Environment $outside, string $name, int $depth, Deriver $deriver): array
240    {
241        $truncated = $callers->partial || count($callers->calls) > Deriver::MAX_CALLERS;
242        $found = [];
243        foreach (array_slice($callers->calls, 0, Deriver::MAX_CALLERS) as $call) {
244            if ($this->budget->isExhausted()) {
245                $truncated = true;
246                break;
247            }
248            $arguments = $this->argumentsFor($call, $parameters);
249            $goals = array_values(array_filter($arguments, static fn (?Expr $argument): bool => $argument !== null));
250            foreach ($deriver->solve($call, $goals, $depth + 1) as $solution) {
251                $environment = $outside->copy();
252                $position = 0;
253                foreach ($arguments as $parameter => $argument) {
254                    $environment->write($parameter, $argument === null
255                        ? $this->defaultOf($parameters[$parameter][1])
256                        : $solution->values[$position++]);
257                }
258                $found[] = [$environment, $solution];
259            }
260        }
261
262        return array_map(
263            static fn (array $one): Binding => new Binding(
264                $one[0],
265                array_merge($one[1]->through, [$name]),
266                $truncated || $one[1]->truncated,
267                $one[1]->combined,
268            ),
269            $found,
270        );
271    }

273    /**
274     * The argument a call passes for each needed parameter, or null where it passes none.
275     *
276     * @param array<string, array{int, Node\Param}> $parameters
277     * @return array<string, Expr|null>
278     */
279    public function argumentsFor(Expr\CallLike $call, array $parameters): array
280    {
281        $arguments = [];
282        $written = $call->isFirstClassCallable() ? [] : $call->getArgs();
283        foreach ($parameters as $name => [$position, $parameter]) {
284            $found = null;
285            foreach ($written as $index => $argument) {
286                if ($argument->name !== null ? $argument->name->toString() === $name : $index === $position) {
287                    $found = $argument->unpack ? null : $argument->value;
288                }
289            }
290            $arguments[$name] = $parameter->variadic ? null : $found;
291        }
292
293        return $arguments;
294    }

297     * What a parameter holds when a call does not pass it.
298     */
299    public function defaultOf(Node\Param $parameter): Domain
300    {
301        $name = $parameter->var instanceof Expr\Variable && is_string($parameter->var->name) ? $parameter->var->name : '';
302        if ($parameter->default !== null && !$parameter->variadic) {
303            return $this->expressions->evaluate($parameter->default, new Environment(), new FunctionScope(''));
304        }
305
306        return Domain::opaque($this->types->read($parameter->type), Origin::Parameter, ($parameter->variadic ? '...$' : '$') . $name);
307    }

323    public function parametersOf(FunctionLike $body, array $needs): array
324    {

327            if ($parameter->var instanceof Expr\Variable && is_string($parameter->var->name)

366    /**
367     * What the names code at the top of a file reads without defining are known to hold.
368     *
369     * Such code runs in the global scope, so a name it reads was set by
370     * whatever ran before it. An extension may say what that is; otherwise it
371     * is left open.
372     *
373     * @param array<string, true> $needs
374     */
375    public function fileScope(array $needs): Environment
376    {
377        $environment = new Environment();
378        foreach ($needs as $name => $_) {
379            $className = $this->globals->declared($name);
380            $environment->write($name, $className !== null
381                ? Domain::of(new ObjectTerm($className))
382                : Domain::opaque(TypeShape::unknown(), Origin::Unresolved, '$' . $name));
383        }
384
385        return $environment;
386    }
