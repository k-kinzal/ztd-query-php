19/**
20 * Works out what expressions can be at a point, starting from the point.
21 *
22 * The walk back from the point gives the paths through the body and what each
23 * still needs at the body's start. A parameter still needed is looked for at
24 * every call of the body, where the same question is asked again about the
25 * arguments; each call that answers is one more way in. Then every way in is
26 * run forward along its path, and the expressions are read at the point. What
27 * is still not found at the top — a parameter nothing calls with, a global, a
28 * name the budget stopped short of — is left open, saying why.
29 *
30 * @visibility root
31 */
32final class Deriver
33{

101     * Every way the expressions can be at the point, each with one value per expression.
102     *
103     * @param list<Expr> $goals
104     * @return list<Solution>
105     */
106    public function solve(Node $point, array $goals, int $depth = 0): array
107    {
108        $key = spl_object_id($point) . ':' . $depth . ':' . implode(',', array_map('spl_object_id', $goals));
109        if (isset($this->remembered[$key])) {
110            return $this->remembered[$key];
111        }
112        $this->remembered[$key] = [];
113        $solutions = $this->run($this->slicer->sliceFrom($point, $this->names->of($goals)), $goals, $this->scopeOf($point), $depth);
114        if ($this->budget->isExhausted()) {
115            unset($this->remembered[$key]);
116        } else {
117            $this->remembered[$key] = $solutions;
118        }
119
120        return $solutions;
121    }

123    /**
124     * Every way the expressions can be when a body finishes, by returning or by running off its end.
125     *
126     * @param list<Expr> $goals Expressions that read nothing but what the body leaves behind, such as a property of `$this`
127     * @return list<Solution>
128     */
129    public function solveAtExits(FunctionLike $body, array $goals, int $depth): array
130    {
131        $key = 'exits:' . spl_object_id($body) . ':' . $depth . ':' . implode(',', array_map('spl_object_id', $goals));
132        if (isset($this->remembered[$key])) {
133            return $this->remembered[$key];
134        }
135        $this->remembered[$key] = [];
136        $needs = $this->names->of($goals);
137        $arrivals = $this->slicer->sliceFromEnd($body, $needs);
138        foreach ((new NodeFinder())->findInstanceOf($body->getStmts() ?? [], Stmt\Return_::class) as $return) {
139            if ($this->tree->bodyOf($return) === $body) {
140                $arrivals = array_merge($arrivals, $this->slicer->sliceFrom($return, $needs));
141            }
142        }
143        $name = $this->nameOf($body);
144        $scope = new FunctionScope($this->tree->fileOf($body), $name, $this->classOf($body), [$name]);
145        $solutions = $this->run($arrivals, $goals, $scope, $depth);
146        if (!$this->budget->isExhausted()) {
147            $this->remembered[$key] = $solutions;
148        }
149
150        return $solutions;
151    }

153    /**
154     * The paths that arrived at a body's start, run forward along every way in, with the expressions read at their end.
155     *
156     * @param list<Arrival> $arrivals
157     * @param list<Expr> $goals
158     * @return list<Solution>
159     */
160    public function run(array $arrivals, array $goals, FunctionScope $scope, int $depth): array
161    {
162        $solutions = [];
163        foreach ($arrivals as $arrival) {
164            $steps = $arrival->path->forward();
165            $affordable = $this->executor->affordableRuns($steps);
166            $bindings = $this->binder->affordable($this->binder->bindings($arrival, $depth, $this), $affordable);
167            $each = max(1, intdiv($affordable, count($bindings)));
168            foreach ($bindings as $binding) {
169                $runs = $this->executor->run($steps, $binding->environment, $scope, $this->expressions, $depth === 0, $each);
170                foreach ($runs as $environment) {
171                    $values = array_map(fn (Expr $goal): Domain => $this->expressions->evaluate($goal, $environment, $scope), $goals);
172                    $solutions[] = new Solution(
173                        array_map($environment->refresh(...), $values),
174                        $binding->through,
175                        $arrival->path->truncated || $binding->truncated,
176                        $binding->combined || $environment->combined,
177                    );
178                }
179            }
180        }
181
182        return $this->bounded($solutions);
183    }
