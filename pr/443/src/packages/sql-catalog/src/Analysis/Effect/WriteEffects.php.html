14/**
15 * Writes whose values cannot be reconstructed, including writes hidden behind calls.
16 *
17 * A wildcard means code can modify the current symbol table. Reference aliases
18 * are treated conservatively, never as evidence that a local is undefined.
19 *
20 * @visibility root
21 */
22final class WriteEffects
23{

25     * Every local may be changed.
26     */
27    public const ALL = '*';

31    /**
32     * The names an operation may change without an ordinary assignment.
33     *
34     * @return array<string, true>
35     */
36    public function own(Node $node, ?ProgramIndex $index = null): array
37    {
38        if ($node instanceof Expr\Include_ || $node instanceof Expr\Eval_) {
39            return [self::ALL => true];
40        }
41        $names = new ModifiedNames();
42        if ($node instanceof Expr\Assign || $node instanceof Expr\AssignOp || $node instanceof Expr\AssignRef) {
43            $dynamic = $this->dynamicTarget($node->var);
44            if ($dynamic !== []) {
45                return $dynamic;
46            }
47        }
48        if ($node instanceof Expr\AssignRef) {
49            return $names->targets($node->var) + $names->targets($node->expr);
50        }
51        if (!$node instanceof Expr\CallLike || $node->isFirstClassCallable()) {
52            return [];
53        }
54        if ($node instanceof Expr\FuncCall && (!$node->name instanceof Node\Name
55            || in_array(strtolower($node->name->getLast()), ['extract', 'parse_str', 'mb_parse_str'], true))) {
56            return [self::ALL => true];
57        }
58
59        return $this->arguments($node, $index);
60    }

62    /**
63     * Invalidates possible aliases before the direct target receives its new value.
64     */
65    public function assignment(Expr $target, Environment $environment): void
66    {
67        $written = (new ModifiedNames())->targets($target);
68        $this->references ??= new ReferenceEffects();
69        $this->apply($this->references->affected($target, $written) + $this->dynamicTarget($target), $environment);

72    /**
73     * Dynamic assignment targets can address any binding in the current scope.
74     *
75     * @return array<string, true>
76     */
77    public function dynamicTarget(Expr $target): array
78    {
79        if ($target instanceof Expr\ArrayDimFetch) {
80            return $this->dynamicTarget($target->var);
81        }
82        if ($target instanceof Expr\Variable && !is_string($target->name)) {
83            return [self::ALL => true];
84        }
85        if ($target instanceof Expr\List_ || $target instanceof Expr\Array_) {
86            foreach ($target->items as $item) {
87                if ($item !== null && $this->dynamicTarget($item->value) !== []) {
88                    return [self::ALL => true];
89                }
90            }
91        }
92
93        return [];
94    }

96    /**
97     * Writable arguments stay open unless a source declaration excludes reference passing.
98     *
99     * @return array<string, true>
100     */
101    public function arguments(Expr\CallLike $call, ?ProgramIndex $index = null): array
102    {
103        $parameters = $call instanceof Expr\FuncCall && $call->name instanceof Node\Name
104            ? $index?->findFunction($call->name->toString())?->node?->getParams() : null;
105        $names = new ModifiedNames();
106        $written = [];
107        foreach ($call->getArgs() as $position => $argument) {
108            $parameter = $argument->name === null ? ($parameters[$position] ?? null) : null;
109            foreach ($parameters ?? [] as $candidate) {
110                if ($argument->name !== null && $candidate->var instanceof Expr\Variable
111                    && $candidate->var->name === $argument->name->toString()) {
112                    $parameter = $candidate;
113                }
114            }
115            if ($parameter !== null && !$parameter->byRef && !$argument->unpack) {
116                continue;
117            }
118            $written += $names->targets($argument->value) + $this->dynamicTarget($argument->value);
119        }
120
121        return $written;
122    }

125     * Opens affected bindings after an operation; later definite assignments may close them again.
126     *
127     * @param array<string, true> $names
128     */
129    public function apply(array $names, Environment $environment, Origin $origin = Origin::Unresolved): void
130    {
131        if (isset($names[self::ALL])) {
132            $names = array_fill_keys($environment->names(), true);
133        }
134        foreach ($names as $name => $_) {
135            $environment->invalidate($name, $origin);
136        }
137    }
