58    /**
59     * Every value the expression can take.
60     */
61    public function evaluate(Expr $node, Environment $environment, FunctionScope $scope): Domain
62    {
63        $this->budget->spend();
64        if ($this->budget->isSpent()) {
65            return Domain::opaque(TypeShape::unknown(), Origin::Budget, 'budget exhausted');
66        }
67
68        $scalar = $this->evaluateScalar($node);
69        if ($scalar !== null) {
70            return $scalar;
71        }
72        $operator = $this->evaluateOperator($node, $environment, $scope);
73        if ($operator !== null) {
74            return $operator;
75        }
76        if ($node instanceof Expr\CallLike) {
77            return $this->calls->evaluate($node, $environment, $scope, $this);
78        }
79        $reference = $this->references->evaluate($node, $environment, $scope, $this);
80        if ($reference !== null) {
81            return $reference;
82        }
83        $this->evaluateOperands($node, $environment, $scope);
84        $effects = new WriteEffects();
85        $effects->apply($effects->own($node), $environment);
86
87        return Domain::opaque(TypeShape::unknown(), Origin::Unresolved, $this->text->render($node));
88    }

90    /**
91     * Evaluates whatever an expression is written over, for what those evaluations find.
92     *
93     * A statement can sit anywhere an expression can, including in an operand
94     * the analyzer has no use for the value of. `$enabled && $pdo->query(…)`
95     * issues a query whatever `$enabled` turns out to be, so finding the call
96     * must not depend on the analyzer caring about the expression around it.
97     * Every call written inside an operand is evaluated even when the value of
98     * the expression holding it is discarded.
99     */
100    public function evaluateOperands(Expr $node, Environment $environment, FunctionScope $scope): void
101    {
102        foreach (get_object_vars($node) as $subNode) {
103            foreach (is_array($subNode) ? $subNode : [$subNode] as $operand) {
104                if ($operand instanceof Expr) {
105                    $this->evaluate($operand, $environment, $scope);
106                }
107            }
108        }
109    }

129    /**
130     * The value of an operator expression, or null when the expression is not one.
131     */
132    public function evaluateOperator(Expr $node, Environment $environment, FunctionScope $scope): ?Domain
133    {
134        if ($node instanceof Expr\Clone_) {
135            return $this->evaluateClone($node, $environment, $scope);
136        }
137        if ($node instanceof Scalar\InterpolatedString) {
138            return $this->evaluateInterpolation($node, $environment, $scope);
139        }
140        if ($node instanceof Expr\BinaryOp\Concat) {
141            return $this->evaluate($node->left, $environment, $scope)
142                ->concat($this->evaluate($node->right, $environment, $scope));
143        }
144        if ($node instanceof Expr\Ternary) {
145            return $this->evaluateTernary($node, $environment, $scope);
146        }
147        if ($node instanceof Expr\Isset_) {
148            return $this->evaluateIsset($node, $environment, $scope);
149        }
150        if ($node instanceof Expr\BinaryOp\Coalesce || $node instanceof Expr\BinaryOp\BooleanAnd
151            || $node instanceof Expr\BinaryOp\BooleanOr || $node instanceof Expr\BinaryOp\LogicalAnd
152            || $node instanceof Expr\BinaryOp\LogicalOr) {
153            return $this->evaluateOptionalRight($node, $environment, $scope);
154        }
155        if ($node instanceof Expr\Match_) {
156            return $this->evaluateMatch($node, $environment, $scope);
157        }
158        if ($node instanceof Expr\Cast) {
159            return $this->evaluateCast($node, $environment, $scope);
160        }
161        if ($node instanceof Expr\Assign) {
162            return $this->evaluateAssign($node, $environment, $scope);
163        }
164        if ($node instanceof Expr\AssignOp\Concat) {
165            return $this->evaluateAppend($node, $environment, $scope);
166        }
167
168        return $this->evaluateResult($node, $environment, $scope);
169    }

171    /**
172     * Clones a tracked object without sharing its allocation with the original.
173     */
174    public function evaluateClone(Expr\Clone_ $node, Environment $environment, FunctionScope $scope): Domain
175    {
176        $value = $this->evaluate($node->expr, $environment, $scope);
177        $terms = [];
178        foreach ($value->terms as $term) {
179            if ($term instanceof ObjectTerm && $term->identity !== null) {
180                $this->clones++;
181                $term = new ObjectTerm($term->className, $term->enumCase, $term->statementId, 'clone:' . $this->clones, $term->state);
182                $environment->objects()->remember($term);
183            }
184            $terms[] = $term;
185        }
186
187        return Domain::fromTerms($terms, $value->widened, $value->combined);
188    }

190    /**
191     * Keeps the effects of executing and skipping a short-circuited operand.
192     */
193    public function evaluateOptionalRight(Expr\BinaryOp $node, Environment $environment, FunctionScope $scope): Domain
194    {
195        $left = $this->evaluate($node->left, $environment, $scope);
196        $taken = $environment->copy();
197        $right = $this->evaluate($node->right, $taken, $scope);
198        $environment->replace($environment->join($taken));
199
200        return $node instanceof Expr\BinaryOp\Coalesce
201            ? $left->union($right)
202            : Domain::opaque(TypeShape::of(['bool']), Origin::Unresolved);

205    /**
206     * The value of a cast, which narrows to a scalar type and, for the numeric
207     * casts, also ends the trail back to whatever the value came from.
208     *
209     * Casting to `int` neutralises a value as a source of injected SQL, so the
210     * result deliberately no longer carries the operand's origin.
211     */
212    public function evaluateCast(Expr\Cast $node, Environment $environment, FunctionScope $scope): Domain
213    {
214        $operand = $this->evaluate($node->expr, $environment, $scope);
215        if ($node instanceof Expr\Cast\String_) {
216            return $operand->concat(Domain::literal(''));
217        }
218        $type = match (true) {
219            $node instanceof Expr\Cast\Int_ => 'int',
220            $node instanceof Expr\Cast\Double => 'float',
221            $node instanceof Expr\Cast\Bool_ => 'bool',
222            $node instanceof Expr\Cast\Array_ => 'array',
223            default => null,
224        };
225        if ($type === null) {
226            return Domain::opaque(TypeShape::of(['object']), Origin::Call, 'cast');
227        }
228        $literal = $operand->soleLiteral();
229        if ($literal !== null && $type !== 'array') {
230            return $this->castLiteral($literal->value, $type);
231        }
232
233        return Domain::opaque(TypeShape::of([$type]), Origin::Call, '(' . $type . ') cast');
234    }

268    public function resultType(Expr $node): ?string
269    {
270        if ($node instanceof Expr\BinaryOp\BooleanAnd || $node instanceof Expr\BinaryOp\BooleanOr) {
271            return 'bool';
272        }
273        if ($node instanceof Expr\BinaryOp\LogicalAnd || $node instanceof Expr\BinaryOp\LogicalOr) {
274            return 'bool';
275        }
276        if ($node instanceof Expr\BooleanNot || $node instanceof Expr\Isset_ || $node instanceof Expr\Empty_) {
277            return 'bool';
278        }
279        if ($node instanceof Expr\Instanceof_) {
280            return 'bool';
281        }
282        if ($node instanceof Expr\PostInc || $node instanceof Expr\PreInc) {
283            return 'int';
284        }
285        if ($node instanceof Expr\PostDec || $node instanceof Expr\PreDec) {
286            return 'int';
287        }

300        $result = Domain::literal('');
301        foreach ($node->parts as $part) {

312    /**
313     * The value of a conditional, which is whichever branch runs.
314     */
315    public function evaluateTernary(Expr\Ternary $node, Environment $environment, FunctionScope $scope): Domain
316    {
317        $condition = $this->evaluate($node->cond, $environment, $scope);
318        $yes = $environment->copy();
319        $no = $environment->copy();
320        $whenTrue = $node->if === null ? $condition : $this->evaluate($node->if, $yes, $scope);
321        $whenFalse = $this->evaluate($node->else, $no, $scope);
322        $environment->replace($yes->join($no));
323
324        return $whenTrue->union($whenFalse);
325    }

341    /**
342     * The value of a match, which is whichever arm runs.
343     */
344    public function evaluateMatch(Expr\Match_ $node, Environment $environment, FunctionScope $scope): Domain
345    {
346        $this->evaluate($node->cond, $environment, $scope);
347        $result = null;
348        $joined = null;
349        $conditions = $environment->copy();
350        foreach ($node->arms as $arm) {
351            foreach ($arm->conds ?? [] as $condition) {
352                $tested = $conditions->copy();
353                $this->evaluate($condition, $tested, $scope);
354                $conditions = $conditions->join($tested);
355            }
356            $branch = $conditions->copy();
357            $value = $this->evaluate($arm->body, $branch, $scope);
358            $result = $result === null ? $value : $result->union($value);
359            $joined = $joined === null ? $branch : $joined->join($branch);
360        }
361
362        $environment->replace($joined ?? $environment);
363
364        return $result ?? Domain::unknown();
365    }
