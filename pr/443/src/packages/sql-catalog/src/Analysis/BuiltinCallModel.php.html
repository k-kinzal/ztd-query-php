19/**
20 * What the PHP functions that shape query text do to the values they are given.
21 *
22 * Queries are assembled with a small, stable set of string functions. Modelling
23 * those keeps a statement resolved that would otherwise become a gap the moment
24 * it passed through `sprintf()` or `implode()`.

26 * @visibility root

44     * Registers every built-in interpretation through the same API as custom models.
45     */
46    public function register(Registry $registry): void
47    {
48        $registry->register('sprintf', fn (array $arguments): Domain => $this->sprintf(array_slice($arguments, 1), $arguments[0] ?? Domain::unknown()));
49        $registry->register('vsprintf', $this->vsprintf(...));
50        $registry->register('implode', $this->implode(...));
51        $registry->register('array_fill', $this->fill(...));
52        $registry->register('join', $this->implode(...));
53        $registry->register('str_repeat', $this->repeat(...));
54        $registry->register('str_replace', $this->replace(...));
55        $registry->register('strval', static fn (array $arguments): Domain => $arguments[0] ?? Domain::unknown());
56        foreach (['strtolower', 'strtoupper', 'ucfirst', 'lcfirst', 'trim', 'ltrim', 'rtrim'] as $function) {
57            $registry->register($function, fn (array $arguments): Domain => $this->transform($function, $arguments));
58        }
59        foreach (['intval', 'count', 'strlen'] as $function) {
60            $registry->register($function, static fn (array $arguments): Domain => Domain::opaque(TypeShape::of(['int']), Origin::Call, $function));
61        }
62        foreach (array_keys(self::STRING_RESULTS) as $function) {
63            $registry->register($function, static fn (array $arguments): Domain => Domain::opaque(TypeShape::of(['string']), Origin::Call, $function));
64        }
65    }

94    /**
95     * The result of `sprintf()`, resolved as far as the format and the arguments allow.
96     *
97     * A format the analyzer knows only in part — `"SELECT * FROM $table WHERE
98     * id IN (%s)"` — still has conversions in the parts it knows, so those are
99     * filled in and the gaps are carried through, rather than the whole result
100     * becoming one gap because one piece of the format was not fixed.
101     *
102     * @param list<Domain> $arguments
103     */
104    public function sprintf(array $arguments, Domain $format): Domain
105    {
106        $result = null;
107        foreach ($format->terms as $term) {
108            $value = $term instanceof LiteralTerm || $term instanceof PatternTerm
109                ? $this->formatPattern($term->toPattern(), $arguments)
110                : Domain::opaque(TypeShape::of(['string']), $this->originOf($format, Origin::Call), 'sprintf');
111            $result = $result === null ? $value : $result->union($value);
112        }
113
114        return $result ?? Domain::opaque(TypeShape::of(['string']), Origin::Call, 'sprintf');
115    }

117    /**
118     * A known or partly known format with its conversions filled in by the arguments.
119     *
120     * @param list<Domain> $arguments
121     */
122    public function formatPattern(TextPattern $format, array $arguments): Domain
123    {
124        $result = Domain::literal('');
125        $index = 0;
126        foreach ($format->segments as $segment) {
127            if (!$segment instanceof LiteralText) {
128                $result = $result->concat(Domain::of(Domain::asTerm(TextPattern::fromSegments([$segment]))));
129                continue;
130            }
131            foreach ($this->splitFormat($segment->text) as $piece) {
132                if (!str_starts_with($piece, '%') || $piece === '%%') {
133                    $result = $result->concat(Domain::literal($piece === '%%' ? '%' : $piece));
134                    continue;
135                }
136                $result = $result->concat(
137                    $arguments[$index] ?? Domain::opaque(TypeShape::unknown(), Origin::Call, 'sprintf'),
138                );
139                $index++;
140            }
141        }
142
143        return $result;
144    }

175    /**
176     * A format string split into literal runs and conversion specifications.
177     *
178     * @return list<string>
179     */
180    public function splitFormat(string $format): array
181    {
182        $pieces = preg_split('/(%%|%[-+ 0\']*\d*(?:\.\d+)?[bcdeEfFgGosuxX])/', $format, -1, PREG_SPLIT_DELIM_CAPTURE);
183        if ($pieces === false) {
184            return [$format];
185        }
186
187        return array_values(array_filter($pieces, static fn (string $piece): bool => $piece !== ''));
188    }

190    /**
191     * The result of `implode()`, resolved when the array is known element by element.
192     *
193     * @param list<Domain> $arguments
194     */
195    public function implode(array $arguments): Domain
196    {
197        $glue = count($arguments) > 1 ? $arguments[0] : Domain::literal('');
198        $array = $this->arrayArgument($arguments);
199        if ($array === null) {
200            $pieces = $arguments[count($arguments) - 1] ?? Domain::unknown();
201
202            return Domain::opaque(TypeShape::of(['string']), $this->originOf($pieces, Origin::Call), 'implode');
203        }
204
205        $result = Domain::literal('');
206        $first = true;
207        foreach ($array->positional() as $element) {
208            $result = $first ? $element : $result->concat($glue)->concat($element);
209            $first = false;
210        }
211
212        return $array->complete ? $result : $result->concat(Domain::opaque(TypeShape::of(['string']), Origin::Call, 'implode'));
213    }

248     * The result of `str_repeat()`, resolved when both arguments are known.
249     *
250     * @param list<Domain> $arguments
251     */
252    public function repeat(array $arguments): Domain
253    {
254        $subject = ($arguments[0] ?? Domain::unknown())->soleLiteral();
255        $times = ($arguments[1] ?? Domain::unknown())->soleLiteral();
256        if ($subject === null || $times === null || !is_int($times->value) || $times->value < 0 || $times->value > 1000) {
257            return Domain::opaque(TypeShape::of(['string']), Origin::Call, 'str_repeat');
258        }
259

263    /**
264     * The result of a one-argument string transformation.
265     *
266     * @param list<Domain> $arguments
267     */
268    public function transform(string $function, array $arguments): Domain
269    {
270        $subject = ($arguments[0] ?? Domain::unknown())->soleLiteral();
271        if ($subject === null) {
272            return Domain::opaque(TypeShape::of(['string']), Origin::Call, $function);
273        }
274        $text = $subject->toText();
275
276        return Domain::literal(match ($function) {
277            'strtolower' => strtolower($text),
278            'strtoupper' => strtoupper($text),
279            'ucfirst' => ucfirst($text),
280            'lcfirst' => lcfirst($text),
281            'ltrim' => ltrim($text),
282            'rtrim' => rtrim($text),
283            default => trim($text),
284        });
285    }

287    /**
288     * The result of `str_replace()`, resolved when every argument is a known string.
289     *
290     * @param list<Domain> $arguments
291     */
292    public function replace(array $arguments): Domain
293    {
294        $search = ($arguments[0] ?? Domain::unknown())->soleLiteral();
295        $replace = ($arguments[1] ?? Domain::unknown())->soleLiteral();
296        $subject = ($arguments[2] ?? Domain::unknown())->soleLiteral();
297        if ($search === null || $replace === null || $subject === null) {
298            return Domain::opaque(TypeShape::of(['string']), Origin::Call, 'str_replace');

300
301        return Domain::literal(str_replace($search->toText(), $replace->toText(), $subject->toText()));
