49     * The value behind the reference, or null when the expression is not one.
50     */
51    public function evaluate(
52        Expr $node,
53        Environment $environment,
54        FunctionScope $scope,
55        ExpressionEvaluator $expressions,
56    ): ?Domain {
57        if ($node instanceof Expr\Variable) {
58            return $this->readVariable($node, $environment, $scope);
59        }
60        if ($node instanceof Expr\Array_) {
61            return $this->readArray($node, $environment, $scope, $expressions);
62        }
63        if ($node instanceof Expr\ArrayDimFetch) {
64            return $this->readElement($node, $environment, $scope, $expressions);
65        }
66        if ($node instanceof Expr\ConstFetch) {
67            return $this->constants->readConstant($node, $scope, $expressions);
68        }
69        if ($node instanceof Expr\ClassConstFetch) {
70            return $this->constants->readClassConstant($node, $scope, $expressions);
71        }
72        if ($node instanceof Expr\PropertyFetch || $node instanceof Expr\NullsafePropertyFetch) {
73            return $this->readProperty($node, $environment, $scope, $expressions);
74        }
75        if ($node instanceof Expr\New_) {
76            return $this->readInstance($node, $scope);
77        }
78        if ($node instanceof Expr\StaticPropertyFetch) {
79            return Domain::opaque(TypeShape::unknown(), Origin::Property, $this->text->render($node));
80        }
81        if ($node instanceof Expr\Closure || $node instanceof Expr\ArrowFunction) {
82            return Domain::of(new ObjectTerm('Closure'));
83        }
84
85        return null;
86    }

89     * The value bound to a variable, or what is known about it when it is external.
90     */
91    public function readVariable(Expr\Variable $node, Environment $environment, FunctionScope $scope): Domain
92    {
93        if (!is_string($node->name)) {
94            return Domain::opaque(TypeShape::unknown(), Origin::Unresolved, 'variable variable');
95        }
96        if ($node->name === 'this') {
97            return $scope->className === null
98                ? Domain::opaque(TypeShape::unknown(), Origin::Unresolved, '$this')
99                : Domain::of(new ObjectTerm($scope->className));
100        }
101        if ($this->external->isVariable($node->name)) {
102            return Domain::opaque(TypeShape::unknown(), Origin::External, '$' . $node->name);
103        }
104
105        return $environment->read($node->name)->withVariable('$' . $node->name);
106    }

111    public function readArray(
112        Expr\Array_ $node,
113        Environment $environment,
114        FunctionScope $scope,
115        ExpressionEvaluator $expressions,
116    ): Domain {
117        $entries = [];
118        $complete = true;
119        foreach ($node->items as $item) {
120            if ($item->unpack) {
121                $complete = false;
122                continue;
123            }
124            $entries[] = new ArrayEntry(
125                $item->key === null ? null : $expressions->evaluate($item->key, $environment, $scope),
126                $expressions->evaluate($item->value, $environment, $scope),
127            );
128        }
129
130        return Domain::of(new ArrayTerm($entries, $complete));
131    }

136    public function readElement(
137        Expr\ArrayDimFetch $node,
138        Environment $environment,
139        FunctionScope $scope,
140        ExpressionEvaluator $expressions,
141    ): Domain {
142        $base = $expressions->evaluate($node->var, $environment, $scope);
143        $array = $base->soleArray();
144        $key = $node->dim === null ? null : $expressions->evaluate($node->dim, $environment, $scope)->soleLiteral();
145        if ($array !== null && $key !== null) {
146            $found = $this->lookup($array, $key->value);
147            if ($found !== null) {

150        }

155    /**
156     * The element stored under a key, or null when the array does not hold one.
157     */
158    public function lookup(ArrayTerm $array, string|int|float|bool|null $key): ?Domain
159    {
160        $wanted = is_bool($key) || $key === null ? null : (string) $key;
161        if ($wanted === null) {
162            return null;
163        }
164        $position = 0;
165        foreach ($array->entries as $entry) {
166            $entryKey = $entry->key === null ? $position++ : $entry->scalarKey();
167            if ($entryKey !== null && (string) $entryKey === $wanted) {
168                return $entry->value;
169            }
170        }
171
172        return null;
173    }

176     * Where the unresolved parts of a domain come from.
177     */
178    public function originOf(Domain $domain): Origin
179    {
180        $origin = Origin::Unresolved;
181        foreach ($domain->terms as $term) {
182            if ($term instanceof OpaqueTerm && $term->origin === Origin::External) {
183                return Origin::External;
184            }
185            if ($term instanceof OpaqueTerm && $origin === Origin::Unresolved) {
186                $origin = $term->origin;
187            }
188        }
189
190        return $origin;
191    }

193    /**
194     * The value of a property read, resolving a settled default when there is one.
195     */
196    public function readProperty(
197        Expr\PropertyFetch|Expr\NullsafePropertyFetch $node,
198        Environment $environment,
199        FunctionScope $scope,
200        ExpressionEvaluator $expressions,
201    ): Domain {
202        $tracked = $this->trackedName($node);
203        if ($tracked !== null && $environment->has($tracked)) {
204            return $environment->read($tracked);
205        }
206        $name = $node->name instanceof Node\Identifier ? $node->name->toString() : null;
207        $receiver = $expressions->evaluate($node->var, $environment, $scope);
208        $owner = $receiver->type()->soleClassName();
209        if ($name === null || $owner === null) {
210            return Domain::opaque(TypeShape::unknown(), Origin::Property, $this->text->render($node));
211        }
212
213        $enum = $this->readEnumProperty($receiver, $owner, $name, $scope, $expressions);
214        if ($enum !== null) {
215            return $enum;
216        }
217
218        foreach ($this->index->lineage($owner) as $shape) {
219            $default = $shape->settledDefault($name);
220            if ($default !== null) {
221                return $expressions->evaluate($default, new Environment(), $scope);
222            }
223        }
224
225        $declared = $this->index->findPropertyType($owner, $name);
226
227        return Domain::opaque($declared ?? TypeShape::unknown(), Origin::Property, $this->text->render($node));
228    }

230    /**
231     * The value behind `->value` or `->name` on an enum.
232     *
233     * A value typed as a backed enum can only ever be one of that enum's cases,
234     * so reading its backing value resolves to the union of them. That is what
235     * turns a parameter typed `Status` into the handful of strings the column
236     * can hold rather than an opaque `string`.
237     */
238    public function readEnumProperty(
239        Domain $receiver,
240        string $owner,
241        string $property,
242        FunctionScope $scope,
243        ExpressionEvaluator $expressions,
244    ): ?Domain {
245        $shape = $this->index->findClass($owner);
246        if ($shape === null || !$shape->enum || ($property !== 'value' && $property !== 'name')) {
247            return null;
248        }
249
250        $selected = $receiver->soleObject()?->enumCase;
251        $result = null;
252        foreach ($shape->enumCases as $case => $backing) {
253            if ($selected !== null && $case !== $selected) {
254                continue;
255            }
256            $value = $property === 'name' || $backing === null
257                ? Domain::literal($case)
258                : $expressions->evaluate($backing, new Environment(), $scope);
259            $result = $result === null ? $value : $result->union($value);
260        }
261
262        return $result;
263    }

277    /**
278     * The name a target is tracked under in the environment, or null when it is not tracked.
279     *
280     * A variable is tracked under its name, and a property of `$this` under
281     * `this->property`, so that assigning it earlier in a body is seen when it
282     * is read later in the same body.
283     */
284    public function trackedName(Expr $target): ?string
285    {
286        if ($target instanceof Expr\Variable) {
287            return is_string($target->name) ? $target->name : null;
288        }
289        if (($target instanceof Expr\PropertyFetch || $target instanceof Expr\NullsafePropertyFetch)
290            && $target->var instanceof Expr\Variable && $target->var->name === 'this'
291            && $target->name instanceof Node\Identifier) {
292            return 'this->' . $target->name->toString();
293        }
294
295        return null;
296    }

298    /**
299     * Writes a value to whatever the target expression names.
300     */
301    public function assign(
302        Expr $target,
303        Domain $value,
304        Environment $environment,
305        FunctionScope $scope,
306        ExpressionEvaluator $expressions,
307    ): void {
308        $this->effects ??= new Effect\WriteEffects();
309        $this->effects->assignment($target, $environment);
310        $name = $this->trackedName($target);
311        if ($name !== null) {
312            $environment->write($name, $value);
313
314            return;
315        }
316        if ($target instanceof Expr\PropertyFetch) {
317            $receiver = $expressions->evaluate($target->var, $environment, $scope);
318            foreach ($receiver->terms as $term) {
319                if ($term instanceof ObjectTerm && $term->identity !== null) {
320                    $environment->objects()->remember(new ObjectTerm($term->className, $term->enumCase, $term->statementId, $term->identity));
321                }
322            }
323
324            return;
325        }
326        if ($target instanceof Expr\ArrayDimFetch) {
327            $this->assignElement($target, $value, $environment, $scope, $expressions);
328
329            return;
330        }
331        if ($target instanceof Expr\List_ || $target instanceof Expr\Array_) {
332            $this->assignList($target, $value, $environment, $scope, $expressions);
333        }
334    }

337     * Writes the elements of an array into the targets a destructuring assignment lists.
338     */
339    public function assignList(
340        Expr\List_|Expr\Array_ $target,
341        Domain $value,
342        Environment $environment,
343        FunctionScope $scope,
344        ExpressionEvaluator $expressions,
345    ): void {
346        $array = $value->soleArray();
347        $position = 0;
348        foreach ($target->items as $item) {
349            if ($item === null) {
350                $position++;
351                continue;
352            }
353            $key = $item->key === null ? $position++ : $expressions->evaluate($item->key, $environment, $scope)->soleLiteral()?->value;
354            $element = $array === null || $key === null ? null : $this->lookup($array, $key);
355            $this->assign(
356                $item->value,
357                $element ?? Domain::opaque(TypeShape::unknown(), $this->originOf($value), $this->text->render($item->value)),
358                $environment,
359                $scope,
360                $expressions,
361            );
362        }
363    }

365    /**
366     * Writes a value into an array held by a tracked name, keeping the key when one is written.
367     *
368     * A write one level down, such as `$parts['where'][] = …`, is not followed
369     * element by element: the array it goes into is marked as no longer known
370     * in full, so reading it back never claims to have seen all of it.
371     */
372    public function assignElement(
373        Expr\ArrayDimFetch $target,
374        Domain $value,
375        Environment $environment,
376        FunctionScope $scope,
377        ExpressionEvaluator $expressions,
378    ): void {
379        $name = $this->trackedName($target->var);
380        if ($name === null) {
381            $base = $target->var;
382            while ($base instanceof Expr\ArrayDimFetch) {
383                $base = $base->var;
384            }
385            if ($base instanceof Expr\PropertyFetch) {
386                $this->assign($base, Domain::unknown(), $environment, $scope, $expressions);
387            }
388            $this->loseTrack($target->var, $environment);
389
390            return;
391        }
392        $key = $target->dim === null ? null : $expressions->evaluate($target->dim, $environment, $scope);
393        $array = $environment->has($name) ? $environment->read($name)->soleArray() : null;
394        $entries = $array === null ? [] : $array->entries;
395        $entries[] = new ArrayEntry($key, $value);
396
397        $environment->write($name, Domain::of(new ArrayTerm($entries, $array === null || $array->complete)));
398    }

400    /**
401     * Marks the array a nested write goes into as no longer known in full.
402     */
403    public function loseTrack(Expr $target, Environment $environment): void
404    {
405        while ($target instanceof Expr\ArrayDimFetch) {
406            $target = $target->var;
407        }
408        $name = $this->trackedName($target);
409        $array = $name === null || !$environment->has($name) ? null : $environment->read($name)->soleArray();
410        if ($name !== null && $array !== null) {
411            $environment->write($name, Domain::of(new ArrayTerm($array->entries, false)));
412        }
413    }
