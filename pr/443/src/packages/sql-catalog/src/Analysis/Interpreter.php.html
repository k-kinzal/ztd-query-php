42/**
43 * Reads the statements a source tree can issue, starting from every call that issues one.
44 *
45 * Every call written the way a database call is written is a starting point.
46 * What it is called on is worked out first, which decides whether it is a
47 * database call at all; then what its arguments can be is worked out from the
48 * call backwards. Each way the arguments can be is one statement at the call,
49 * and a value bound later, by an `execute()` or a `bindValue()`, is attached to
50 * the statement the handle it is called on came from.
51 *
52 * @visibility root
53 */

101    /**
102     * Every statement the files can issue.
103     *
104     * @param list<ParsedFile> $files
105     * @return list<QueryRecord>
106     */
107    public function analyze(array $files): array
108    {
109        $deriver = $this->deriverFor($files);
110        $matcher = new SinkMatcher($this->sinks, $this->index, $this->models);
111        $recorder = new StatementRecorder();
112        $binder = new ValueBinder($recorder);
113        $bindings = [];
114        foreach ($files as $file) {
115            foreach ($this->finder->findAll($file, $this->sinks) as $call) {
116                $this->budget->reset();
117                $bindings = array_merge($bindings, $this->visit($call, $deriver, $matcher, $recorder, $binder));
118            }
119        }
120        foreach ($bindings as [$sink, $solution]) {
121            $values = $solution->values;
122            $receiver = array_shift($values) ?? Domain::unknown();
123            $binder->bindValues($binder->openRecords($receiver), $sink, $values);
124        }
125
126        return $recorder->records();
127    }

139    /**
140     * The deriver wired over the given files.
141     *
142     * @param list<ParsedFile> $files
143     */
144    public function deriverFor(array $files): Deriver
145    {
146        $tree = new SourceTree($files);
147        $effects = $this->modelProviders !== [];
148        $names = new FreeNames($this->sinks, trackObjectEffects: $effects);
149        $modified = new ModifiedNames($names, $effects ? new ObjectEffects($files) : null);
150        $slicer = new BackwardSlicer($tree, $this->budget, $names, $modified);
151        $executor = new SliceExecutor($this->globals, new TypeReader(), $modified, $this->text, $this->budget);
152        $functions = clone $this->functions;
153        $this->models = null;
154        foreach ($this->modelProviders as $provider) {
155            $models = $provider->models(new ModelContext($this->index, new CallbackEffects($slicer, $executor), $this->dialect));
156            $this->models = $this->models?->merge($models) ?? $models;
157        }
158        foreach ($this->models->calls ?? [] as $model) {
159            $functions->registerCall($model);
160        }
161        $external = new ExternalInput();
162        $expressions = new ExpressionEvaluator(
163            new ReferenceEvaluator($this->index, $external, $this->text),
164            new CallEvaluator(
165                $this->index,
166                new SinkMatcher($this->sinks, $this->index, $this->models),
167                $functions,
168                $external,
169                $this->text,
170                new CalleeReturns($slicer, $executor, $this->budget, $names),
171            ),
172            $this->budget,
173            $this->text,
174        );
175
176        return new Deriver(
177            $tree,
178            $slicer,
179            $executor,
180            $expressions,
181            new EntryBinder(
182                new Callers(new CallerIndex($files), $this->index, new SinkMatcher($this->sinks, $this->index, $this->models)),
183                $expressions,
184                $this->budget,
185                $this->globals,
186                new PropertyWrites($this->index, $modified),
187            ),
188            $this->budget,
189            $names,
190        );
191    }

193    /**
194     * Records what one call issues, and hands back the values it binds for later.
195     *
196     * @return list<array{SinkSpec, Solution}>
197     */
198    public function visit(
199        Expr\CallLike $call,
200        Deriver $deriver,
201        SinkMatcher $matcher,
202        StatementRecorder $recorder,
203        ValueBinder $binder,
204    ): array {
205        $scope = $deriver->scopeOf($call);
206        $sink = $this->sinkOf($call, $deriver, $matcher, $scope);
207        if ($sink === null) {
208            if ($this->unidentified($call, $deriver, $matcher)) {
209                $this->recordUnmatched($call, $scope, $recorder);
210            }
211
212            return [];
213        }
214        if ($sink->role === SinkRole::Compose) {
215            return [];
216        }
217        $this->budget->reset();
218        $arguments = $this->argumentsOf($call);
219        if ($sink->role === SinkRole::Modelled) {
220            $this->recordStatements($call, $sink, (new ModelQueries())->solve($call, $this->models->queries[$sink->model ?? ''] ?? null, $deriver), $scope, $recorder, $binder);
221
222            return [];
223        }
224        if ($sink->role === SinkRole::Execute || $sink->role === SinkRole::Bind) {
225            $receiver = $call instanceof Expr\MethodCall || $call instanceof Expr\NullsafeMethodCall ? $call->var : null;
226            if ($receiver === null) {
227                return [];
228            }
229
230            return array_map(
231                static fn (Solution $solution): array => [$sink, $solution],
232                $deriver->solve($call, array_merge([$receiver], $arguments)),
233            );
234        }
235        if ($sink->sqlParameter === null || !isset($arguments[$sink->sqlParameter])) {
236            return [];
237        }
238        $this->recordStatements($call, $sink, $deriver->solve($call, $arguments), $scope, $recorder, $binder);
239
240        return [];
241    }

243    /**
244     * The database call a call is, or null when it is not one the enabled extensions recognise.
245     */
246    public function sinkOf(Expr\CallLike $call, Deriver $deriver, SinkMatcher $matcher, FunctionScope $scope): ?SinkSpec
247    {
248        $name = $this->finder->nameOf($call);
249        if ($name === null) {
250            return null;
251        }
252        if ($call instanceof Expr\FuncCall) {
253            return $matcher->matchFunction($name);
254        }
255        if ($call instanceof Expr\StaticCall) {
256            $className = $call->class instanceof Node\Name ? $call->class->toString() : null;
257            if ($className !== null && in_array(strtolower($className), ['self', 'static', 'parent'], true)) {
258                $className = $scope->className ?? $className;
259            }
260
261            return $className === null ? null : $matcher->matchStatic($className, $name);
262        }
263        if ($call instanceof Expr\MethodCall || $call instanceof Expr\NullsafeMethodCall) {
264            return $matcher->matchMethod($this->receiverOf($call, $deriver), $name);
265        }
266
267        return null;
268    }

270    /**
271     * Everything the receiver of a method call can be.
272     */
273    public function receiverOf(Expr\MethodCall|Expr\NullsafeMethodCall $call, Deriver $deriver): Domain
274    {
275        $receiver = null;
276        foreach ($deriver->solve($call, [$call->var]) as $solution) {
277            $receiver = $receiver === null ? $solution->values[0] : $receiver->union($solution->values[0]);
278        }
279
280        return $receiver ?? Domain::unknown();
281    }

283    /**
284     * Whether a call that carries statement text could not be told apart from a database call.
285     *
286     * A call on something whose class is known, and is not a database class,
287     * is simply not a database call. A call on something whose class could not
288     * be worked out might be one, and that is a gap worth reporting.
289     */
290    public function unidentified(Expr\CallLike $call, Deriver $deriver, SinkMatcher $matcher): bool
291    {
292        if (!$call instanceof Expr\MethodCall && !$call instanceof Expr\NullsafeMethodCall) {
293            return false;
294        }
295        $name = $this->finder->nameOf($call);
296        $carriesText = false;
297        foreach ($matcher->byName(\SqlCatalog\Extension\SinkCallKind::Method, $name ?? '') as $sink) {
298            $carriesText = $carriesText || $sink->role === SinkRole::Query || $sink->role === SinkRole::Prepare || $sink->role === SinkRole::Modelled;
299        }
300
301        return $carriesText && $this->receiverOf($call, $deriver)->type()->classNames() === [];
302    }

318    /**
319     * Records one statement for every way the call's statement text can be.
320     *
321     * @param list<Solution> $solutions
322     */
323    public function recordStatements(
324        Expr\CallLike $call,
325        SinkSpec $sink,
326        array $solutions,
327        FunctionScope $scope,
328        StatementRecorder $recorder,
329        ValueBinder $binder,
330    ): void {
331        $site = new CallSite($scope->file, $call->getStartLine(), $scope->function, $sink->id);
332        $siteKey = $scope->file . ':' . $call->getStartFilePos() . ':' . $sink->id;
333        $all = [];
334        foreach ($solutions as $solution) {
335            $sql = $sink->sqlParameter === null ? null : ($solution->values[$sink->sqlParameter] ?? null);
336            if ($sql === null) {
337                continue;
338            }
339            $records = [];
340            foreach ($sql->patterns() as $pattern) {
341                $records[] = $recorder->record($site, $siteKey, $pattern, $sink->kind, $sql->combined || $solution->combined, $solution->through, $solution->truncated);
342            }
343            $binder->bindValues($records, $sink, $solution->values);
344            $all = array_merge($all, $records);
345        }
346        if ($all === []) {
347            $all[] = $recorder->record($site, $siteKey, $this->unread($call), $sink->kind);
348        }
349        if ($sink->role === SinkRole::Prepare) {
350            $recorder->filePrepared($siteKey, $all);
351        }
352    }
