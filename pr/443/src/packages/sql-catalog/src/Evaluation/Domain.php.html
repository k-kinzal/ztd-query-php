12/**
13 * Every value an expression may take, over-approximated.
14 *
15 * A domain is a bounded set of alternatives. Keeping the set means a query
16 * built from an enum-typed parameter is reported as the handful of statements
17 * it can actually be, instead of one statement with a gap. When the set would
18 * grow past the bound it is generalized into a single shape that still covers
19 * all of it, so the approximation only ever widens.
20 *
21 * @visibility root
22 */
23final class Domain
24{
25    private ?string $signature = null;
26
27    /**
28     * How many alternatives a domain keeps before generalizing them into one shape.
29     */
30    public const MAX_TERMS = 12;

36     */

42    }

76    /**
77     * The domain built from a set of alternatives, deduplicated and bounded.
78     *
79     * @param list<Term> $terms
80     */
81    public static function fromTerms(array $terms, bool $widened = false, bool $combined = false): self
82    {
83        $unique = [];
84        foreach ($terms as $term) {
85            $unique[$term->signature()] = $term;
86        }
87        $kept = array_values($unique);
88
89        if ($kept === []) {
90            return self::unknown();
91        }
92        if (count($kept) <= self::MAX_TERMS) {
93            return new self($kept, $widened, $combined);
94        }
95
96        return new self([self::generalize($kept, Origin::Branch)], true, $combined);
97    }
98
99    /**
100     * The single term that covers every given alternative.
101     *
102     * @param list<Term> $terms
103     */
104    public static function generalize(array $terms, Origin $origin): Term
105    {
106        $patterns = [];
107        foreach ($terms as $term) {
108            $patterns[] = $term->toPattern();
109        }
110
111        return new PatternTerm((new TextGeneralization($origin))->mergeAll($patterns));
112    }

114    /**
115     * The domain admitting every value either domain admits.
116     */
117    public function union(self $other): self
118    {
119        return self::fromTerms(
120            array_merge($this->terms, $other->terms),
121            $this->widened || $other->widened,
122            $this->combined || $other->combined,
123        );
124    }

126    /**
127     * The domain of the strings produced by writing this domain then the other.
128     *
129     * Pairing two sides that each hold several alternatives produces every
130     * combination of them, and nothing here can tell which combinations the
131     * program can actually reach. The result is marked as combined so that a
132     * report can say the alternatives may be wider than the code allows.
133     */
134    public function concat(self $other): self
135    {
136        $terms = [];
137        foreach ($this->terms as $left) {
138            foreach ($other->terms as $right) {
139                $terms[] = self::asTerm($left->toPattern()->concat($right->toPattern()));
140            }
141        }
142        $paired = count($this->terms) > 1 && count($other->terms) > 1;
143
144        return self::fromTerms($terms, $this->widened || $other->widened, $this->combined || $other->combined || $paired);
145    }

161    {

199    /**
200     * Labels opaque alternatives with the PHP variable read at their use site.
201     *
202     * Provenance, resolved fragments, comparison signatures and domain flags
203     * remain unchanged; the name is only an annotation for report readers.
204     */
205    public function withVariable(string $variable): self
206    {
207        if (array_filter($this->terms, static fn (Term $term): bool => $term instanceof OpaqueTerm) === []) {
208            return $this;
209        }
210
211        return new self(
212            array_map(
213                static fn (Term $term): Term => $term instanceof OpaqueTerm
214                    ? new OpaqueTerm($term->type, $term->origin, $term->expression, $variable)
215                    : $term,
216                $this->terms,
217            ),
218            $this->widened,
219            $this->combined,
220        );
221    }

265    /**
266     * A canonical string used to compare domains.
267     */
268    public function signature(): string
269    {
270        if ($this->signature !== null) {
271            return $this->signature;
272        }
273        $parts = [];
274        foreach ($this->terms as $term) {
275            $parts[] = $term->signature();
276        }
277        sort($parts);
278
279        return $this->signature = implode('|', $parts);
280    }
