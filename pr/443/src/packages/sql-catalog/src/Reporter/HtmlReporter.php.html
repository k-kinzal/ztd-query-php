26/**
27 * Writes the catalog as a site of linked HTML pages.
28 *
29 * A reader comes to a catalog to find a statement and decide something about
30 * it, so the site is laid out as the routes to one: by the table it names, by
31 * the namespace and class that issue it, by the file it is written in, by
32 * what the analysis reported on it, or through the whole listing narrowed
33 * down. Every statement has a page of its own, which is where every route
34 * ends.
35 *
36 * @visibility root
37 */

88        return new CatalogArtifacts(

99    public function tablePages(ReportSite $site, PageShell $shell): array
100    {
101        $page = new TablePage();
102        $pages = [];
103        foreach ($site->tables() as $table) {
104            $name = $site->tablePage($table);
105            $label = (new TableName($table))->label();
106            $pages[$name] = $shell->render(
107                $site,
108                $name,
109                $label,
110                [['Overview', ReportSite::INDEX], ['Tables', ReportSite::TABLES], [$label, null]],
111                $page->render($site, $table),
112                $page->context($site, $table),
113            );
114        }
115
116        return $pages;
