20    /**
21     * CLI-shaped defaults, with paths based at the configuration directory.
22     *
23     * @return array<string, list<string>>
24     * @throws InvalidConfigurationException When a setting has an unknown name or invalid value
25     */
26    public static function options(stdClass $data, string $directory): array
27    {
28        $options = [];
29        foreach (get_object_vars($data) as $key => $value) {
30            if ($key === 'function-models') {
31                continue;
32            }
33            if (in_array($key, self::SCALAR_OPTIONS, true)) {
34                if (!is_string($value) || $value === '') {
35                    throw new InvalidConfigurationException(sprintf('Setting "%s" must be a non-empty string.', $key));
36                }
37                $values = [$value];
38            } elseif (in_array($key, self::LIST_OPTIONS, true)) {
39                if (!is_array($value) || !array_is_list($value)) {
40                    throw new InvalidConfigurationException(sprintf('Setting "%s" must be a list of strings.', $key));
41                }
42                $values = [];
43                foreach ($value as $item) {
44                    if (!is_string($item) || $item === '') {
45                        throw new InvalidConfigurationException(sprintf('Setting "%s" must contain non-empty strings.', $key));
46                    }
47                    $values[] = $item;
48                }
49            } else {
50                throw new InvalidConfigurationException(sprintf('Unknown catalog setting "%s".', $key));
51            }
52            if (in_array($key, ['paths', 'output', 'root'], true)) {
53                $values = array_map(static fn (string $path): string => self::absolute($path, $directory), $values);
54            }
55            $options[$key === 'extensions' ? 'extension' : $key] = $values;
56        }
57        $options['root'] ??= [$directory];
58
59        return $options;
60    }

63     * Named function models must be strings identifying autoloadable callables or invokable classes.
64     *
65     * @return array<string, string>
66     * @throws InvalidConfigurationException When the registrations are not a mapping of names to strings
67     */
68    public static function models(stdClass $data): array
69    {
70        $fields = get_object_vars($data);
71        if (!array_key_exists('function-models', $fields)) {
72            return [];
73        }
74        $models = $fields['function-models'];
75        if (!$models instanceof stdClass) {
76            throw new InvalidConfigurationException('Setting "function-models" must be a mapping.');
77        }
78        $registered = [];
79        foreach (get_object_vars($models) as $name => $model) {
80            if ($name === '' || !is_string($model) || $model === '') {
81                throw new InvalidConfigurationException('Function model names and callables must be non-empty strings.');
82            }
83            $registered[$name] = $model;
84        }
85
86        return $registered;
87    }

90     * An absolute path stays unchanged; a relative one starts beside the configuration.
91     */
92    public static function absolute(string $path, string $directory): string
93    {
94        if (str_starts_with($path, '/') || str_starts_with($path, '\\') || preg_match('/^[a-z]:[\\\\\/]/i', $path) === 1) {
95            return $path;
96        }
97
