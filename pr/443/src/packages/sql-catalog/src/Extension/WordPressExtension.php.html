10/**
11 * WordPress's `wpdb`, where nearly all of the SQL an installation issues is written.
12 *
13 * `wpdb::prepare()` interpolates a statement and hands it back for one of the
14 * reading or writing methods to run, so it is recognised as the place the
15 * statement text comes from rather than as a call the analyzer cannot see
16 * through. That is what keeps the familiar
17 * `$wpdb->get_results($wpdb->prepare(…))` readable as the statement it is.
18 *
19 * @visibility root
20 */
21final class WordPressExtension implements ExtensionInterface
22{
23    private const DATABASE = 'wpdb';
24
25    /**
26     * The name the command line selects this extension by.
27     */
28    #[Override]
29    public function name(): string
30    {
31        return 'wordpress';
32    }
33
34    /**
35     * What the extension covers.
36     */
37    #[Override]
38    public function description(): string
39    {
40        return 'WordPress wpdb, including the statements wpdb::prepare() interpolates';
41    }
42
43    /**
44     * The wpdb calls that carry a statement or hand one back.
45     *
46     * @return list<SinkSpec>
47     */
48    #[Override]
49    public function sinks(): array
50    {
51        $sinks = [
52            new SinkSpec(
53                'wordpress.prepare',
54                SinkCallKind::Method,
55                self::DATABASE,
56                'prepare',
57                SinkRole::Compose,
58                sqlParameter: 0,
59                valuesFrom: 1,
60            ),
61        ];
62        foreach ($this->methods() as $method => $kind) {
63            $sinks[] = new SinkSpec(
64                'wordpress.' . $method,
65                SinkCallKind::Method,
66                self::DATABASE,
67                $method,
68                SinkRole::Query,
69                sqlParameter: 0,
70                kind: $kind,
71            );
72        }
73
74        return $sinks;
75    }
76
77    /**
78     * The global the handle is reached through.
79     *
80     * Almost every statement WordPress issues is written after `global $wpdb;`,
81     * which says nothing about what the name holds. Declaring it here is what
82     * lets those calls be read as the database calls they are.
83     *
84     * @return array<string, string>
85     */
86    #[Override]
87    public function globals(): array
88    {
89        return [self::DATABASE => self::DATABASE];
90    }
91
92    /**
93     * The wpdb methods that take a statement, and the kind each implies.
94     *
95     * @return array<string, StatementKind|null>
96     */
97    public function methods(): array
98    {
99        return [
100            'query' => null,
101            'get_results' => StatementKind::Select,
102            'get_row' => StatementKind::Select,
103            'get_col' => StatementKind::Select,
104            'get_var' => StatementKind::Select,
105            'get_col_info' => StatementKind::Select,
106        ];
107    }
108}
