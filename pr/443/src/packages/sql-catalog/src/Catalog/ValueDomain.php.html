12/**
13 * What a bound value can be, reported in the form the catalog stores.
14 *
15 * A value that resolves statically is reported as the literals it can take, so
16 * a parameter typed as a backed enum becomes the union of that enum's values
17 * rather than a bare `string`. A value the analyzer could not follow is
18 * reported as its static type and where it came from.
19 *
20 * @visibility root
21 */
22final class ValueDomain
23{
24    /**
25     * @param string $type The static type, written the way PHP writes it
26     * @param list<string|int|float|bool|null> $values The values it can take, when they are known
27     * @param bool $exhaustive Whether the listed values are all of them
28     * @param list<string> $origins Where unresolved parts of the value come from
29     */
30    public function __construct(
31        public readonly string $type,
32        public readonly array $values,
33        public readonly bool $exhaustive,
34        public readonly array $origins,
35    ) {
36    }
37
38    /**
39     * The report of what an evaluated domain can hold.
40     */
41    public static function fromDomain(Domain $domain): self
42    {
43        $values = [];
44        $origins = [];
45        $exhaustive = !$domain->widened;
46
47        foreach ($domain->terms as $term) {
48            if ($term instanceof LiteralTerm) {
49                $values[] = $term->value;
50                continue;
51            }
52            $exhaustive = false;
53            $origins[] = $term instanceof OpaqueTerm ? $term->origin->value : Origin::Unresolved->value;
54        }
55
56        return new self(
57            $domain->type()->display(),
58            $exhaustive ? $values : [],
59            $exhaustive,
60            array_values(array_unique($origins)),
61        );
62    }
63
64    /**
65     * Whether the value is pinned down to a known set of alternatives.
66     */
67    public function isResolved(): bool
68    {
69        return $this->exhaustive && $this->values !== [];
70    }
71
72    /**
73     * Whether a runtime value is one the domain admits.
74     *
75     * An unresolved domain admits everything, which is what makes it a sound
76     * over-approximation rather than a claim about the value.
77     */
78    public function admits(string|int|float|bool|null $value): bool
79    {
80        if (!$this->isResolved()) {
81            return true;
82        }
83        foreach ($this->values as $candidate) {
84            if ($candidate === $value) {
85                return true;
86            }
87            if (is_scalar($candidate) && is_scalar($value) && (string) $candidate === (string) $value) {
88                return true;
89            }
90        }
91
92        return false;
93    }
94
95    /**
96     * The domain written for a reader.
97     */
98    public function display(): string
99    {
100        if (!$this->isResolved()) {
101            return $this->type;
102        }
103        $written = [];
104        foreach ($this->values as $value) {
105            $written[] = is_string($value) ? "'" . $value . "'" : var_export($value, true);
106        }
107
108        return implode('|', $written);
109    }
