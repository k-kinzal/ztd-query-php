86    public function testSliceFromKeepsWhatTheCallDependsOnInTheOrderItRuns(string $body, array $steps, array $needs): void
87    {
88        $file = (new SourceParser())->parse('t.php', '<?php function f($table, $a) { ' . $body . ' }');
89        $call = (new NodeFinder())->findFirstInstanceOf($file->statements, Expr\FuncCall::class);
90        self::assertInstanceOf(Expr\FuncCall::class, $call);
91        $printer = new Standard();
92
93        $arrivals = (new BackwardSlicer(new SourceTree([$file]), new EvaluationBudget()))
94            ->sliceFrom($call, (new FreeNames())->of([$call->getArgs()[0]->value]));
95
96        self::assertCount(1, $arrivals);
97        self::assertSame($file->statements[0], $arrivals[0]->body);
98        self::assertSame($steps, array_map(
99            static fn (SliceStep $step): string => $step->node instanceof Expr ? $printer->prettyPrintExpr($step->node) : 'either',
100            $arrivals[0]->path->forward(),
101        ));
102        self::assertSame($needs, array_keys($arrivals[0]->path->needs));
103    }
