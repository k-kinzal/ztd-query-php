26/**
27 * The one-time recorder of the facts of a statement under construction.
28 *
29 * Each node derives its own facts and asks the derivation to derive its

73    /**
74     * Derives a nested statement that the database reads with other name-search settings.
75     *
76     * The statement sees the declarations of the same profile under the
77     * search settings of the given context, as the elements of a schema
78     * definition are read with that schema searched first. The context may add
79     * only declarations this statement has already provided, as the elements
80     * of one schema definition see each other.
81     *
82     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When the context holds other declarations or another profile
83     */
84    public function within(AnalysisContext $context, Statement $node): void
85    {
86        $this->admit($context);
87        [$outer, $base] = [$this->context, $this->base];
88        [$this->context, $this->base] = [$context, null];
89        $node->deriveStatement($this);
90        [$this->context, $this->base] = [$outer, $base];
91    }
92
93    /**
94     * Refuses a nested context that differs from this one other than in search settings and own declarations.
95     *
96     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When the context holds other declarations or another profile
97     */
98    public function admit(AnalysisContext $context): void
99    {
100        Check::input($context->profile->compatibleWith($this->context->profile) && $context->complete === $this->context->complete, 'A nested statement is derived under the same profile and completeness.');
101        foreach ($this->context->tables as $table) {
102            Check::input(in_array($table, $context->tables, true), 'A nested statement sees every declaration of the enclosing context.');
103        }
104        foreach ($context->tables as $table) {
105            Check::input(in_array($table, $this->context->tables, true) || in_array($table, $this->declarations, true), 'A nested statement sees only the declarations of the enclosing context and of the statement itself.');
106        }
107    }

140    /**
141     * Derives a statement whose request is only inspected or stored, such as the operand of EXPLAIN or a routine body.
142     *
143     * Every part of the statement receives its facts, and its diagnostics are
144     * kept, but the rows it would return and the declarations it would
145     * provide are discarded: inspecting or storing a statement neither
146     * executes it nor declares anything. An environment, when given, is the
147     * position the statement is read at instead of a statement root, such as
148     * the variables of a routine body or the search settings of EXPLAIN FOR
149     * DATABASE; its context follows the rule of within().
150     *
151     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When the environment holds other declarations or another profile
152     */
153    public function inspected(Statement $node, ?Environment $environment = null): void
154    {
155        [$output, $declarations, $context, $base] = [$this->output, $this->declarations, $this->context, $this->base];
156        if ($environment !== null) {
157            $this->admit($environment->context);
158            [$this->context, $this->base] = [$environment->context, $environment];
159        }
160        $this->output = null;
161        $node->deriveStatement($this);
162        [$this->output, $this->declarations, $this->context, $this->base] = [$output, $declarations, $context, $base];
163    }

211    /**

214    public function table(QualifiedName $name, Environment $environment): TableResolution
215    {
216        if ($name->schema === null) {
217            $common = $environment->commonTable($name->name);
218            if ($common !== null) {

221        }

223        return (new TableLookup())->find($this->context, $name);
224    }
