28/**
29 * Analyzes SQL of one fixed language profile into read-only operations.
30 *
31 * An operation holds the concrete structure of the statement, the facts
32 * derived against an explicit declaration context, and SQL rendered from that
33 * structure. Nothing here changes an operation: a different statement is
34 * analyzed anew or built from explicit inputs.
35 *
36 * @visibility public
37 * @example Resolving a query against the declaration it depends on
38 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite);
39 *     $users = $semantics->analyze('CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT)');
40 *     $query = $semantics->analyze('SELECT name FROM users WHERE id = 1', [$users]);
41 *     $query->field('name')->column() === $users->declarations()[0]->columns[1] // => true
42 * @example Rendering SQL from the analyzed structure
43 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite);
44 *     $semantics->analyze('select   1')->toString() // => 'SELECT 1'
45 * @example Finding the statements of a script
46 *     $semantics = new \SqlSemantics\Facade\Semantics(\SqlSemantics\Platform\Sqlite\Dialect::Sqlite);
47 *     $semantics->split("SELECT 1; SELECT ';'") // => ['SELECT 1;', " SELECT ';'"]
48 */

63     * Fixes the language profile: database, grammar release, session mode and parameter style.
64     *
65     * @param Dialect $dialect The database
66     * @param string|null $grammarVersion A release tag the database package ships, or null for its default
67     * @param Mode|null $mode The session settings SQL is read under, or null for the defaults of the database
68     * @param ParameterStyle $parameters Which parameter markers are read; the named style adds `:name`
69     * @param SearchPath|null $searchPath The schemas an unqualified relation name is searched in, or null for the default of the database
70     *
71     * @throws InvalidArgumentException When the release, mode or search path does not belong to the database
72     */

75        $this->platform = Platforms::of($dialect->database());

90    /**
91     * Creates an immutable declaration context.
92     *
93     * Null declarations give an open context: relations that are not declared
94     * may exist. A list, even an empty one, gives a context that enumerates
95     * every relation unless `$complete` is false. An operation in the list
96     * contributes the declarations it provides; it is not executed, and ALTER,
97     * DROP and writes contribute nothing.
98     *
99     * @param list<Table|Operation>|null $declarations The declarations, or null for an open context without any
100     * @param bool $complete Whether a given list enumerates every relation
101     *
102     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When a declaration belongs to another language profile
103     */
104    public function context(?array $declarations = null, bool $complete = true): AnalysisContext
105    {
106        $tables = [];
107        foreach ($declarations ?? [] as $declaration) {
108            if ($declaration instanceof Operation) {
109                Check::input($this->profile->compatibleWith($declaration->profile()), 'A declaring operation must belong to the selected language profile.');
110                array_push($tables, ...$declaration->declarations());
111            } else {
112                $tables[] = $declaration;
113            }
114        }
115
116        return $this->platform->context($this->profile, $this->searchPath, $tables, $declarations !== null && $complete);
117    }

119    /**
120     * Analyzes one input into an operation against an explicit declaration context.
121     *
122     * Semantic problems of grammatical SQL, such as a missing column, are
123     * facts of the returned operation. The operation is returned only after
124     * its structure was confirmed to hold every operand of the input and its
125     * rendered SQL was confirmed to carry the same significant tokens.
126     *
127     * @param list<Table|Operation>|AnalysisContext|null $context The declarations, a prepared context, or null for an open context
128     *
129     * @throws AnalysisException When the SQL is outside the selected grammar
130     * @throws \SqlSemantics\Diagnostic\InvalidConstruction When the context belongs to another language profile
131     */
132    public function analyze(string $sql, array|AnalysisContext|null $context = null): Operation
133    {
134        $declarations = $context instanceof AnalysisContext ? $context : $this->context($context);
135        Check::input($this->profile->compatibleWith($declarations->profile), 'The context must match the selected language profile.');
136        try {
137            $tree = $this->parser->parse($sql);
138        } catch (SourceException $error) {
139            throw new AnalysisException($error->getMessage(), 0, $error);
140        }
141        $leaves = new Leaves();
142        $statement = (new Publication())->root($this->platform->lower($tree, $this->profile, $leaves));
143        $operation = new Operation($declarations, $statement);
144
145        $graph = new ValueGraph(['SqlSemantics\\Statement\\', 'SqlSemantics\\Contract\\', $this->platform->statementNamespace()]);
146        $dropped = (new LeafEmbedding())->dropped($leaves, $graph->objects($statement));
147        Check::invariant($dropped === null, 'The statement lost an operand of the input: ' . ($dropped === null ? '' : $dropped::class));
148        $tokens = new TokenCorrespondence();
149        $keys = $this->platform->leafKeys($this->profile);
150        $productions = $this->platform->productions($this->profile);
151        $difference = $tokens->difference($tokens->keys($tree, $productions, $keys), $tokens->keys($this->parser->parse($operation->toString()), $productions, $keys));
152        Check::invariant($difference === null, 'The rendered SQL does not carry the tokens of the input: ' . $difference . ' in: ' . $operation->toString());
153
154        return $operation;
155    }

157    /**
158     * Analyzes each statement of a script against the same explicit declaration context.
159     *
160     * No statement is executed for the ones after it.
161     *
162     * @param list<Table|Operation>|AnalysisContext|null $context The declarations, a prepared context, or null for an open context
163     * @return list<Operation>
164     *
165     * @throws AnalysisException When a statement is outside the selected grammar
166     */
167    public function analyzeAll(string $sql, array|AnalysisContext|null $context = null): array
168    {
169        $declarations = $context instanceof AnalysisContext ? $context : $this->context($context);
170        $operations = [];
171        foreach ($this->split($sql) as $statement) {
172            $operations[] = $this->analyze($statement, $declarations);
173        }
174
175        return $operations;
176    }

179     * Finds the statement boundaries of a script, as the database finds them.
180     *
181     * Each text ends with its own terminator; trailing whitespace and comments
182     * stay with the last statement. A semicolon inside a string, a comment, or
183     * a compound statement ends nothing.
184     *

195        }
