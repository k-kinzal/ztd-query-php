16/**
17 * Checks that the layouts of a rendering only re-spelled the rendered tokens.
18 *
19 * Rule: CORE-SPELLING-001 (validation part). The SQL as the structure renders
20 * it and the SQL as the layouts spell it are read by the parser of the
21 * profile. Both must have the same number of tokens, and the comparison keys
22 * of the token correspondence check, which depend on the production and
23 * position of each token, must be equal position by position. A layout
24 * therefore cannot add, drop or change a significant token, its gaps can only
25 * hold whitespace and comments, and a spelling counts as the same token only
26 * where the database reads it as the same request. Status: Implemented.
27 *
28 * @visibility SqlSemantics
29 */
30final class Spellings
31{
32    /**
33     * @param SqlParser $parser The parser of the profile
34     * @param Productions $productions The productions of the grammar release
35     * @param LeafKeys $keys The token rules of the profile
36     */
37    public function __construct(private readonly SqlParser $parser, private readonly Productions $productions, private readonly LeafKeys $keys)
38    {
39    }

41    /**
42     * Refuses a spelled SQL text whose tokens are not those of the rendered SQL text.
43     *
44     * @throws InvariantViolation When a layout changed a token or wrote a token into a gap
45     */
46    public function check(string $rendered, string $spelled): void
47    {
48        if ($rendered === $spelled) {
49            return;
50        }
51        $canonical = $this->tokens($rendered);
52        $written = $this->tokens($spelled);
53        Check::invariant(count($canonical) === count($written), 'The spelled SQL has ' . count($written) . ' tokens where the rendered SQL has ' . count($canonical) . ': ' . $spelled);
54        $correspondence = new TokenCorrespondence();
55        $difference = $correspondence->difference($correspondence->keys($this->tree($rendered), $this->productions, $this->keys), $correspondence->keys($this->tree($spelled), $this->productions, $this->keys));
56        Check::invariant($difference === null, 'The spelled SQL is not the rendered SQL at ' . $difference . ': ' . $spelled);
57    }
58
59    /**
60     * Parses a rendered or spelled text.
61     *
62     * @throws InvariantViolation When the text is outside the grammar
63     */
64    public function tree(string $text): Node
65    {
66        try {
67            return $this->parser->parse($text);
68        } catch (SourceException $error) {
69            throw new InvariantViolation('A rendered or spelled text is outside the grammar: ' . $text, 0, $error);
70        }
71    }

73    /**
74     * Reads the tokens of a text, without the end marker.
75     *
76     * @return list<Token>
77     * @throws InvariantViolation When the text is not lexically valid
78     */
79    public function tokens(string $text): array
80    {
81        try {
82            $tokens = $this->parser->tokenize($text);
83        } catch (SourceException $error) {
84            throw new InvariantViolation('A spelled text is not lexically valid: ' . $text, 0, $error);
85        }
86
87        return array_values(array_filter($tokens, static fn (Token $token): bool => $token->text !== ''));
88    }
