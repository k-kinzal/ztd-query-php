172    private function nested(Node $node): stdClass|array
173    {
174        $object = new stdClass();
175        $list = [];
176        $mapping = null;
177        foreach (Nodes::items($node) as $item) {
178            $paragraph = $item->firstChild();
179            if (!$paragraph instanceof Paragraph) {
180                throw new InvalidArgumentException('Expected a metadata key or value.');
181            }
182            $isKey = $paragraph->firstChild() instanceof Strong;
183            $mapping ??= $isKey;
184            if ($mapping !== $isKey) {
185                throw new InvalidArgumentException('Do not mix mapping keys and sequence values in one metadata list.');
186            }
187            [$key, $text] = $isKey ? Nodes::pair($paragraph) : ['', Nodes::text($paragraph)];
188            $child = $paragraph->next();
189            if ($child !== null) {
190                if (($text !== '' && ($isKey || $text !== '[]')) || $child->next() !== null) {
191                    throw new InvalidArgumentException('Nested metadata must have an empty parent value and one nested list.');
192                }
193                $value = $this->nested($child);
194            } else {
195                $value = $this->scalar($text);
196            }
197            if ($isKey) {
198                if ($key === '' || property_exists($object, $key)) {
199                    throw new InvalidArgumentException('Metadata keys must be nonempty and unique.');
200                }
201                $object->{$key} = $value;
202            } else {
203                $list[] = $value;
204            }
205        }
206        return $mapping === true ? $object : $list;
207    }
