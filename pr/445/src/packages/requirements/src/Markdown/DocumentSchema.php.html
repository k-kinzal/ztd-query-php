72    private function allowed(Node $node, array $schema, string $file): void
73    {
74        Fields::keys($schema, ['type'], 'allBlocks');
75        if ($node instanceof HtmlBlock && $node->parent() instanceof BlockQuote && $node === $node->parent()->firstChild()) {
76            Quotation::annotation($node);
77            return;
78        }
79        if (!$node instanceof ListItem && !in_array($this->type($node), Fields::strings($schema['type'], 'allBlocks.type'), true)) {
80            throw new InvalidArgumentException("$file: document-schema forbids " . $this->type($node) . ' blocks; use quotations, paragraphs and bullet lists.');
81        }
82        if ($node instanceof BlockQuote || $node instanceof ListBlock || $node instanceof ListItem) {
83            foreach ($node->children() as $child) {
84                $this->allowed($child, $schema, $file);
85            }
86        }
87    }

94    {
95        $entries = array_map(static fn (mixed $value): array => Fields::mapping($value, 'block'), Fields::sequence($schema['blocks'], 'blocks'));
96        $counts = array_fill(0, count($entries), 0);
97        $pointer = 0;
98        foreach ($nodes as $node) {
99            $found = false;
100            for ($i = $pointer; $i < count($entries); ++$i) {
101                $entry = $entries[$i];
102                Fields::keys($entry, ['type', 'text', 'minContains', 'maxContains'], 'block');
103                $types = is_string($entry['type']) ? [$entry['type']] : Fields::strings($entry['type'], 'block.type');
104                if (in_array($this->type($node), $types, true) && (!isset($entry['text']) || $this->matches($node instanceof Paragraph ? Nodes::text($node) : '', Fields::mapping($entry['text'], 'block.text')))) {
105                    $pointer = $i;
106                    ++$counts[$i];
107                    $found = true;
108                    break;
109                }
110            }
111            if (!$found && ($schema['additionalBlocks'] ?? true) === false) {
112                throw new InvalidArgumentException("$file: unexpected document-schema block.");
113            }
114        }
115        foreach ($entries as $i => $entry) {
116            $this->occurrences($counts[$i], $entry, "$file: section block " . ($i + 1));
117        }
118    }

121    private function occurrences(int $count, array $schema, string $context): void
122    {
123        $minimum = $schema['minContains'] ?? 1;
124        $maximum = $schema['maxContains'] ?? PHP_INT_MAX;
125        if (!is_int($minimum) || !is_int($maximum) || $count < $minimum || $count > $maximum) {

130    /** @param array<string, mixed> $schema */
131    private function matches(string $text, array $schema): bool
132    {
133        Fields::keys($schema, ['pattern', 'const', 'enum'], 'text constraint');
134        if (isset($schema['pattern']) && preg_match('~' . str_replace('~', '\\~', Fields::text($schema, 'pattern')) . '~u', $text) !== 1) {
135            return false;
136        }
137        return (!isset($schema['const']) || $text === $schema['const']) && (!isset($schema['enum']) || in_array($text, Fields::strings($schema['enum'], 'enum'), true));
138    }
