109    {
110        $workspace = new Workspace();

116        $snapshot = (new Snapshot())->read($file);
117        self::assertSame(['changed' => [], 'removed' => []], (new Snapshot())->compare($analysis, $project, $snapshot));
118        $definition = $loader->document($workspace->directory . '/definition.yaml');
119        $item = Workspace::item();

121        $definition['items'] = [$item];
122        $workspace->write('definition.yaml', $definition);
123        $changed = $loader->load($workspace->directory . '/requirements.yaml');
124        self::assertCount(1, (new Snapshot())->compare((new Analyzer())->analyze($changed), $changed, $snapshot)['changed']);
125        $definition['source'] = ['id' => 'manual', 'uri' => 'source.html', 'format' => 'html', 'selector' => '#a'];
126        $workspace->write('definition.yaml', $definition);
127        $reduced = $loader->load($workspace->directory . '/requirements.yaml');
128        $report = (new Coverage())->report($reduced, (new Analyzer())->analyze($reduced), $file);
