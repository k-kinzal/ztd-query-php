271    public function testCompareDetectsChangedClaimsNewUnitsAndRemovedScope(): void
272    {
273        $workspace = new ProjectDirectory();
274        $loader = new Loader();
275        $project = $loader->load($workspace->directory . '/requirements.yaml');
276        $analysis = (new Analyzer())->analyze($project);
277        $file = $workspace->directory . '/snapshot.json';
278        file_put_contents($file, json_encode((new Snapshot())->create($analysis, $project), JSON_THROW_ON_ERROR));
279        $snapshot = (new Snapshot())->read($file);
280        self::assertSame(['changed' => [], 'removed' => []], (new Snapshot())->compare($analysis, $project, $snapshot));
281        $definition = $loader->document($workspace->directory . '/definition.yaml');
282        $item = ProjectDirectory::item();
283        $item['statement'] = 'The parser shall reject names beginning with digits.';
284        $definition['items'] = [$item];
285        $workspace->write('definition.yaml', $definition);
286        $changed = $loader->load($workspace->directory . '/requirements.yaml');
287        self::assertCount(1, (new Snapshot())->compare((new Analyzer())->analyze($changed), $changed, $snapshot)['changed']);
288        $definition['source'] = ['id' => 'manual', 'uri' => 'source.html', 'format' => 'html', 'selector' => '#a'];
289        $workspace->write('definition.yaml', $definition);
290        $reduced = $loader->load($workspace->directory . '/requirements.yaml');
291        $report = (new Coverage())->report($reduced, (new Analyzer())->analyze($reduced), $file);
292        self::assertFalse($report['passed']);
293        self::assertIsArray($report['removed']);
294        self::assertCount(2, $report['removed']);
295        self::assertTrue((new Coverage())->report($reduced, (new Analyzer())->analyze($reduced), $file, true)['passed']);
296    }
