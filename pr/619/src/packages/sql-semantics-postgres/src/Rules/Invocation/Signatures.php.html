172    /**
173     * Answers the rows of a function name: result, kind, NULL rule and parameter types.
174     *
175     * @return list<array{string, string, string, list<string>}>
176     */
177    public function rows(string $name): array
178    {
179        $rows = [];
180        foreach (self::FUNCTIONS[$name] ?? [] as $row) {
181            $rows[] = [$row[0], $row[1], $row[2], array_slice($row, 3)];
182        }
183        foreach (self::NUMERIC_AGGREGATES[$name] ?? [] as $argument => $result) {
184            $rows[] = [$result, 'a', 'Y', [$argument]];
185        }
186        if ($name === 'min' || $name === 'max') {
187            foreach (self::ORDERED as $argument) {
188                $rows[] = [$argument, 'a', 'Y', [$argument]];
189            }
190        }
191
192        return $rows;
193    }
