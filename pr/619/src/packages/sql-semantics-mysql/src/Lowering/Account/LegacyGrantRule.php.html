20/**
21 * Lowers GRANT and REVOKE of MySQL 5.6 and 5.7.
22 *
23 * Rule: MYSQL-ACCOUNT-GRANT-LEGACY-001. Scope: grant, revoke (5.x
24 * productions), grant_command, revoke_command. The statements are the same
25 * structures as the 8.0 ones; GRANT also carries the account clauses of 5.x
26 * (IDENTIFIED per account, REQUIRE, resource limits in WITH), and 5.6 REVOKE
27 * reads its accounts with the GRANT account list. The marker
28 * clear_privileges holds no operand. Constructs: GrantPrivileges,
29 * GrantProxy, RevokePrivileges, RevokeAll, RevokeProxy. Terminates: every
30 * child is a strict subtree.
31 * Source: https://dev.mysql.com/doc/refman/5.7/en/grant.html,
32 * https://dev.mysql.com/doc/refman/5.7/en/revoke.html. Status: Implemented.
33 *
34 * @visibility SqlSemantics\Platform\MySql
35 */
36final class LegacyGrantRule
37{
38    /**
39     * The object kinds of the grant_command and revoke_command productions that name one.
40     */
41    private const KINDS = [
42        'grant_command: grant_privileges ON opt_table grant_ident TO_SYM grant_list require_clause grant_options' => ObjectKind::Table,
43        'grant_command: grant_privileges ON FUNCTION_SYM grant_ident TO_SYM grant_list require_clause grant_options' => ObjectKind::Function,
44        'grant_command: grant_privileges ON PROCEDURE_SYM grant_ident TO_SYM grant_list require_clause grant_options' => ObjectKind::Procedure,
45        'revoke_command: grant_privileges ON opt_table grant_ident FROM grant_list' => ObjectKind::Table,
46        'revoke_command: grant_privileges ON FUNCTION_SYM grant_ident FROM grant_list' => ObjectKind::Function,
47        'revoke_command: grant_privileges ON PROCEDURE_SYM grant_ident FROM grant_list' => ObjectKind::Procedure,
48        'revoke_command: grant_privileges ON opt_table grant_ident FROM user_list' => ObjectKind::Table,
49        'revoke_command: grant_privileges ON FUNCTION_SYM grant_ident FROM user_list' => ObjectKind::Function,
50        'revoke_command: grant_privileges ON PROCEDURE_SYM grant_ident FROM user_list' => ObjectKind::Procedure,
51    ];

60    /**
61     * Lowers `grant: GRANT clear_privileges grant_command`.
62     *
63     * @throws ImplementationGap When a production has no rule
64     */
65    public function grant(Form $form): Statement
66    {
67        if ($form->signature !== 'grant: GRANT clear_privileges grant_command') {
68            throw ImplementationGap::production($form);
69        }
70        $this->lowering->options->skip($form->node(1));
71        $command = $this->lowering->form($form->node(2));
72        $lists = new UserListRule($this->lowering, new IdentificationRule($this->lowering));
73        $clauses = new ClauseRule($this->lowering);
74        if ($command->signature === 'grant_command: PROXY_SYM ON user TO_SYM grant_list opt_grant_option') {
75            return new GrantProxy($this->lowering->users->account($command->node(2)), $lists->granted($command->node(4)), $clauses->grantOption($command->node(5)));
76        }
77        $kind = self::KINDS[$command->signature] ?? throw ImplementationGap::production($command);
78        $this->table($command);
79        $privileges = new PrivilegeRule($this->lowering);
80
81        return new GrantPrivileges(
82            $privileges->privileges($command->node(0)),
83            $kind,
84            $privileges->level($command->node(3)),
85            $lists->granted($command->node(5)),
86            $clauses->tls($command->node(6)),
87            $clauses->grantOptions($command->node(7)),
88        );
89    }
90
91    /**
92     * Lowers `revoke: REVOKE clear_privileges revoke_command`.
93     *
94     * @throws ImplementationGap When a production has no rule
95     */
96    public function revoke(Form $form): Statement
97    {
98        if ($form->signature !== 'revoke: REVOKE clear_privileges revoke_command') {
99            throw ImplementationGap::production($form);
100        }
101        $this->lowering->options->skip($form->node(1));
102        $command = $this->lowering->form($form->node(2));
103        $privileges = new PrivilegeRule($this->lowering);
104        if ($command->signature === 'revoke_command: ALL opt_privileges , GRANT OPTION FROM grant_list' || $command->signature === 'revoke_command: ALL opt_privileges , GRANT OPTION FROM user_list') {
105            $privileges->words($command->node(1));
106
107            return new RevokeAll(false, $this->users($command->node(6)));
108        }
109        if ($command->signature === 'revoke_command: PROXY_SYM ON user FROM grant_list' || $command->signature === 'revoke_command: PROXY_SYM ON user FROM user_list') {
110            return new RevokeProxy(false, $this->lowering->users->account($command->node(2)), $this->users($command->node(4)));
111        }
112        $kind = self::KINDS[$command->signature] ?? throw ImplementationGap::production($command);
113        $this->table($command);
114
115        return new RevokePrivileges(false, $privileges->privileges($command->node(0)), $kind, $privileges->level($command->node(3)), $this->users($command->node(5)));
116    }

118    /**
119     * Confirms the optional TABLE word of a command that names no routine kind.
120     */
121    public function table(Form $command): void
122    {
123        $kind = $command->node->children[2] ?? null;
124        if ($kind instanceof Node) {
125            $this->lowering->options->present($kind);
126        }
127    }
128
129    /**
130     * Lowers the accounts of a 5.x REVOKE: a grant_list (5.6) or a user_list (5.7).
131     *
132     * @return non-empty-list<UserSpecification>
133     * @throws ImplementationGap When a production has no rule
134     */
135    public function users(Node $list): array
136    {
137        if ($list->name === 'grant_list') {
138            return (new UserListRule($this->lowering, new IdentificationRule($this->lowering)))->granted($list);
139        }
140
141        return (new GrantRule($this->lowering))->specifications($list);
142    }
