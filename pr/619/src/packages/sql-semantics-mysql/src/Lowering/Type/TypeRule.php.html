25/**
26 * Lowers the data type productions of every release into type structure.
27 *
28 * Rule: MYSQL-TYPE-001. Scope: type, int_type, real_type, numeric_type,
29 * opt_PRECISION, spatial_type, and through its collaborators the string
30 * types, the type parts and cast_type. The structure is the type as
31 * requested: its keyword kind, exact length, precision and scale, numeric
32 * attributes in written order and character set attribute. Spellings the
33 * manual defines as synonyms are one kind (TypeNoise lists them); REAL,
34 * `CHAR VARYING` and the LONG forms stay distinct requests. Constructs:
35 * Integral, Decimal, Floating, Elementary, Temporal, Spatial, Character,
36 * Binary, Enumeration, CastTarget. Terminates: every child is a strict
37 * subtree. Source: https://dev.mysql.com/doc/refman/8.4/en/data-types.html.
38 * Status: Implemented.
39 *
40 * @visibility SqlSemantics\Platform\MySql
41 */
42final class TypeRule

164    /**
165     * Lowers a numeric type production, or answers null for another kind of type.
166     *
167     * @throws ImplementationGap When a part has no rule
168     */
169    public function numeric(Form $form): ?TypeName
170    {
171        if ($form->signature === 'type: int_type opt_field_length field_options') {
172            $kind = $this->lowering->productions->form($form->node(0));
173
174            return new Integral(self::INTEGERS[$kind->signature] ?? throw ImplementationGap::production($kind), $this->parts->length($form->node(1)), $this->parts->modifiers($form->node(2)));
175        }
176        if ($form->signature === 'type: real_type opt_precision field_options') {
177            return $this->fraction($this->real($form->node(0)), $form);
178        }
179        if ($form->signature === 'type: numeric_type float_options field_options') {
180            $kind = $this->lowering->productions->form($form->node(0));
181            if (!array_key_exists($kind->signature, self::NUMERIC_KEYWORDS)) {
182                throw ImplementationGap::production($kind);
183            }
184
185            return $this->fraction(self::NUMERIC_KEYWORDS[$kind->signature], $form);
186        }
187
188        return array_key_exists($form->signature, self::NUMERICS) ? $this->fraction(self::NUMERICS[$form->signature], $form) : null;
189    }
190
191    /**
192     * Builds a DECIMAL or approximate type from a production whose second and third symbols are its options and attributes.
193     *
194     * @throws ImplementationGap When a part has no rule
195     */
196    public function fraction(?FloatingKind $kind, Form $form): TypeName
197    {
198        [$precision, $scale] = $this->parts->numbers($form->node(1));
199        $modifiers = $this->parts->modifiers($form->node(2));
200
201        return $kind === null ? new Decimal($precision, $scale, $modifiers) : new Floating($kind, $precision, $scale, $modifiers);
202    }

204    /**
205     * Lowers the REAL or DOUBLE keyword of an approximate type.
206     *
207     * @throws ImplementationGap When the production has no rule
208     */
209    public function real(Node $keyword): FloatingKind
210    {
211        $form = $this->lowering->productions->form($keyword);
212        if ($form->signature === 'real_type: DOUBLE_SYM opt_PRECISION') {
213            $precision = $this->lowering->productions->form($form->node(1));
214            if ($precision->signature !== 'opt_PRECISION:' && $precision->signature !== 'opt_PRECISION: PRECISION') {
215                throw ImplementationGap::production($precision);
216            }
217        }
218
219        return self::REALS[$form->signature] ?? throw ImplementationGap::production($form);
220    }
221
222    /**
223     * Tells whether a REAL or DOUBLE keyword is DOUBLE written with PRECISION.
224     */
225    public function doublePrecision(Node $keyword): bool
226    {
227        $form = $this->lowering->productions->form($keyword);
228        if ($form->signature === 'real_type: DOUBLE_SYM PRECISION') {
229            return true;
230        }
231
232        return $form->signature === 'real_type: DOUBLE_SYM opt_PRECISION' && $this->lowering->productions->form($form->node(1))->signature === 'opt_PRECISION: PRECISION';
233    }

235    /**
236     * Lowers a type written as one keyword, a temporal type or a spatial type, or answers null for another kind of type.
237     *
238     * @throws ImplementationGap When a part has no rule
239     */
240    public function keyword(Form $form): ?TypeName
241    {
242        if (isset(self::ELEMENTARY[$form->signature])) {
243            [$kind, $position] = self::ELEMENTARY[$form->signature];
244
245            return new Elementary($kind, $position === null ? null : $this->parts->length($form->node($position)));
246        }
247        if (isset(self::TEMPORALS[$form->signature])) {
248            [$kind, $position] = self::TEMPORALS[$form->signature];
249
250            return new Temporal($kind, $position === null ? null : $this->parts->length($form->node($position)));
251        }
252        if ($form->signature === 'type: YEAR_SYM opt_field_length field_options') {
253            return new Temporal(TemporalKind::Year, $this->parts->length($form->node(1)), $this->parts->modifiers($form->node(2)));
254        }
255        if ($form->signature !== 'type: spatial_type') {
256            return null;
257        }
258        $kind = $this->lowering->productions->form($form->node(0));
259
260        return new Spatial(self::SPATIALS[$kind->signature] ?? throw ImplementationGap::production($kind));
261    }
