14/**
15 * Lowers one option of CHANGE MASTER TO or CHANGE REPLICATION SOURCE TO, or one log position of UNTIL.
16 *
17 * Rule: MYSQL-SOURCE-OPTION-001. Scope: master_def, master_file_def (5.6,
18 * 5.7), source_def, source_file_def, source_log_file, source_log_pos and
19 * the change_replication_source_* keyword rules (8.0 and later). Each
20 * option production names one field of LEX_SOURCE_INFO (SourceOptionKind);
21 * from 8.0 to 8.3 an option keyword is a rule of its own whose MASTER_ and
22 * SOURCE_ alternatives are synonyms (the parser fills the same field and
23 * adds a deprecation warning for the MASTER_ one), so the option takes the
24 * vocabulary of the statement, which is the one of the release. The value is
25 * lowered by the leaf rules or by MYSQL-SOURCE-VALUE-001. Constructs:
26 * SourceOption. Terminates: unit productions over strict subtrees.
27 * Source: https://dev.mysql.com/doc/refman/8.4/en/change-replication-source-to.html,
28 * https://dev.mysql.com/doc/refman/8.0/en/change-master-to.html.

202    /**
203     * Lowers one option in the vocabulary of its statement.
204     *
205     * @throws ImplementationGap When a production has no rule
206     */
207    public function option(Node $option, Terminology $terminology): SourceOption
208    {
209        $form = $this->lowering->form($option);
210        while (isset(self::FORWARD[$form->signature])) {
211            $form = $this->lowering->form($form->node(0));
212        }
213        if (!array_key_exists($form->signature, self::OPTIONS)) {
214            throw ImplementationGap::production($form);
215        }
216        $kind = self::OPTIONS[$form->signature] ?? $this->keyword($form->node(0));
217
218        return new SourceOption($terminology, $kind, (new SourceValueRule($this->lowering))->value($form));
219    }
220
221    /**
222     * Answers the option an option keyword rule names.
223     *
224     * @throws ImplementationGap When the production has no rule
225     */
226    public function keyword(Node $keyword): SourceOptionKind
227    {
228        $form = $this->lowering->form($keyword);
229
230        return self::KEYWORDS[$form->signature] ?? throw ImplementationGap::production($form);
231    }
