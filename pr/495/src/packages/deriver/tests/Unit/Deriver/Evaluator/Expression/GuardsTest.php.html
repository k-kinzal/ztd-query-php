135    public function testSelectingKeepsStructuredEdgesAndFallThroughEdgesAsAlternatives(): void
136    {
137        $source = new Source('guards.php', '<?php function nested($a,$b){if($a){if($b){$x=1;}}} function grouped($t){switch($t){case 1:case 2:$x=1;break;}}');
138        $tree = (new Resolver())->resolve((new Parser(new ParserConfig('8.3')))->parse($source));
139        $context = new Context(new DefinitionIndex(new Index('guards', [$tree])), new EvaluatorConfig(), new Evaluators(), new Budget());
140        $guards = new Guards(new Evaluator($context));
141        $nested = $context->index->graph('nested');
142        $grouped = $context->index->graph('grouped');
143        self::assertNotNull($nested);
144        self::assertNotNull($grouped);
145        $nested->controls = (new ControlDependencies())->build($nested);
146        $grouped->controls = (new ControlDependencies())->build($grouped);
147        $block = static function (BodyIndex $graph): int {
148            $writes = array_values(array_filter($graph->definitions, static fn (Instruction $instruction): bool => $instruction->operation === 'write' && ($graph->definitions[$instruction->operands[0]]->name ?? '') === 'x'));
149            return $graph->positions[$writes[0]->result][0];
150        };
151        self::assertCount(1, $guards->dependencies(new Frame($nested, 'source:nested'), $block($nested)));
152        self::assertCount(1, $guards->selecting(new Frame($nested, 'source:nested'), $block($nested)));
153        self::assertCount(2, $guards->dependencies(new Frame($grouped, 'source:grouped'), $block($grouped)));
154        self::assertCount(2, $guards->selecting(new Frame($grouped, 'source:grouped'), $block($grouped)));
155    }
