11/**
12 * Finds local file paths without reading or parsing their contents.
13 * Patterns match paths relative to each root: * and ? stay within a component.
14 * The recursive ** wildcard crosses directories and can match zero directories. Exclusions win.
15 * Linked files and directories are skipped. With no roots, searches the current directory;
16 * with no includes, selects every regular file. No directories are excluded implicitly.
17 * @visibility public
18 * @example Selecting only PHP source paths
19 *     $finder = (new \Deriver\Finder())->in(__DIR__)->include('*.php')->exclude('vendor');
20 *     is_array($finder->files()) // => true
21 */
22final class Finder
23{
24    /**
25     * @var list<string>
26     */
27    private array $directories = [];
28    /**
29     * @var list<PathPattern>
30     */
31    private array $includes = [];
32    /**
33     * @var list<PathPattern>
34     */
35    private array $excludes = [];
36
37    /**
38     * Adds local search roots. Results are sorted, unique absolute paths.
39     * @throws InvalidInputException If a root is not a readable local directory
40     */
41    public function in(string $directory, string ...$directories): self
42    {
43        foreach ([$directory, ...$directories] as $root) {
44            if (str_contains($root, '://') || !is_dir($root) || !is_readable($root)) {
45                throw new InvalidInputException('Finder root must be a readable local directory: ' . $root);
46            }
47            $absolute = realpath($root);
48            if ($absolute === false) {
49                throw new InvalidInputException('Cannot resolve Finder root: ' . $root);
50            }
51            $this->directories[] = str_replace('\\', '/', $absolute);
52        }
53        return $this;
54    }
55
56    /**
57     * Includes files matching any supplied root-relative wildcard pattern.
58     */
59    public function include(string $pattern, string ...$patterns): self
60    {
61        foreach ([$pattern, ...$patterns] as $value) {
62            $this->includes[] = new PathPattern($value);
63        }
64        return $this;
65    }
66
67    /**
68     * Excludes matching files and directory subtrees, even when included explicitly.
69     */
70    public function exclude(string $pattern, string ...$patterns): self
71    {
72        foreach ([$pattern, ...$patterns] as $value) {
73            $this->excludes[] = new PathPattern($value);
74        }
75        return $this;
76    }
77
78    /**
79     * Enumerates paths only; file contents are read later by Deriver::prepare().
80     * @return list<string> Sorted, unique absolute paths
81     */
82    public function files(): array
83    {
84        $directories = $this->directories === [] ? (new self())->in('.')->directories : $this->directories;
85        $paths = [];
86        foreach (array_unique($directories) as $directory) {
87            foreach ((new DirectoryScan())->files($directory, $this->includes, $this->excludes) as $path) {
88                $paths[$path] = true;
89            }
90        }
91        $files = array_keys($paths);
92        sort($files, SORT_STRING);
93        return $files;
94    }
95}
