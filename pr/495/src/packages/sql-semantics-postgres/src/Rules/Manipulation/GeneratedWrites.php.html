20/**
21 * Reports a generated column that a data-modifying statement writes with a value other than DEFAULT.
22 *
23 * Rule: PG-GENERATED-WRITE-001. A generated column takes only DEFAULT,
24 * whatever OVERRIDING clause the statement has. An INSERT, the INSERT action
25 * of MERGE included, writes a column of its column list (or, without one, the
26 * column at the position of a value) with a value other than DEFAULT unless
27 * every row of its VALUES gives DEFAULT there; the rows of a query are never
28 * DEFAULT. An UPDATE, the UPDATE action of MERGE and ON CONFLICT DO UPDATE
29 * included, writes a column of its SET list with a value other than DEFAULT
30 * unless the item, or the field of the row constructor of a multiple-column
31 * item, is DEFAULT; a sub-SELECT is never DEFAULT, and assigning a value to
32 * a part of the column (a subscript or a field) writes the column. Parentheses around
33 * DEFAULT do not change it. The server checks the columns in table order
34 * after parse analysis (rewriteTargetListIU, rewriteHandler.c), so the first
35 * generated column so written is reported: `cannot insert a non-DEFAULT
36 * value into column "b"` or `column "b" can only be updated to DEFAULT`,
37 * both with the detail `Column "b" is a generated column.`. A column counts
38 * when its slot is a declared generated column of the target; through a view
39 * the base column is not followed. An action of CREATE RULE is not checked:
40 * the server stores it without rewriting it, and the check applies only when
41 * the rule fires; a statement of a routine body (BEGIN ATOMIC) is checked.
42 * Source: https://www.postgresql.org/docs/17/ddl-generated-columns.html,
43 * https://www.postgresql.org/docs/16/ddl-generated-columns.html,
44 * https://github.com/postgres/postgres/blob/REL_17_2/src/backend/rewrite/rewriteHandler.c. Status: Implemented.
45 *
46 * @visibility SqlSemantics\Platform\PostgreSql
47 */
48final class GeneratedWrites
49{
50    /**
51     * Reports the first generated column an INSERT writes with a value other than DEFAULT.
52     *
53     * @param list<ColumnTarget> $columns The column list; empty to write the columns of the table in order
54     * @param list<list<Scalar>> $rows The VALUES rows; empty when a query supplies the rows
55     * @param int|null $width The number of values per row, or null when it is not known
56     * @param Environment $environment The environment of the statement
57     */
58    public function inserted(VisibleRelation $target, array $columns, array $rows, ?int $width, Derivation $derivation, Environment $environment): void
59    {
60        if (!$this->checked($environment)) {
61            return;
62        }
63        $slots = $columns === [] ? $target->shape->slots : $this->targets($columns, $target, $derivation);
64        $width ??= $columns === [] ? 0 : count($columns);
65        $assignments = new Assignments();
66        $written = [];
67        foreach (array_slice($slots, 0, $width) as $position => $slot) {
68            $defaulted = $rows !== [];
69            foreach ($rows as $row) {
70                $defaulted = $defaulted && $assignments->requested($row[$position] ?? null) !== null;
71            }
72            if ($slot !== null && !$defaulted) {
73                $written[] = $slot;
74            }
75        }
76        $this->report($target, $written, ManipulationMisuseRule::GeneratedInsert, $derivation);
77    }
78
79    /**
80     * Reports the first generated column a SET list writes with a value other than DEFAULT.
81     *
82     * @param list<Assignment|RowAssignment> $assignments
83     * @param Environment $environment The environment of the SET list
84     */
85    public function updated(VisibleRelation $target, array $assignments, Derivation $derivation, Environment $environment): void
86    {
87        if (!$this->checked($environment)) {
88            return;
89        }
90        $rules = new Assignments();
91        $written = [];
92        foreach ($assignments as $assignment) {
93            $columns = $assignment instanceof Assignment ? [$assignment->column] : $assignment->columns;
94            $source = $assignment instanceof RowAssignment ? $rules->bare($assignment->source) : null;
95            foreach ($this->targets($columns, $target, $derivation) as $position => $slot) {
96                $value = $assignment instanceof Assignment ? $assignment->value : ($source instanceof RowConstructor ? $source->fields[$position] ?? null : null);
97                if ($slot !== null && $rules->requested($value) === null) {
98                    $written[] = $slot;
99                }
100            }
101        }
102        $this->report($target, $written, ManipulationMisuseRule::GeneratedUpdate, $derivation);
103    }
104
105    /**
106     * Tells whether the server rewrites the statement where it is written: not inside an action of CREATE RULE, whose scope holds OLD and NEW of the rule.
107     */
108    public function checked(Environment $environment): bool
109    {
110        for ($scope = $environment; $scope !== null; $scope = $scope->outer) {
111            foreach ($scope->relations as $relation) {
112                if ($relation->relation instanceof CreateRule) {
113                    return false;
114                }
115            }
116        }
117
118        return true;
119    }
120
121    /**
122     * Answers the slot of the target each listed column writes, whole or in part, or null for a column the target does not show.
123     *
124     * @param list<ColumnTarget> $columns
125     *
126     * @return list<OutputSlot|null>
127     */
128    public function targets(array $columns, VisibleRelation $target, Derivation $derivation): array
129    {
130        $assignments = new Assignments();
131
132        return array_map(static fn (ColumnTarget $column): ?OutputSlot => $assignments->find($column, $target, $derivation), $columns);
133    }
134
135    /**
136     * Reports the first slot of the target, in table order, that is written and is a generated column.
137     *
138     * @param list<OutputSlot> $written
139     */
140    public function report(VisibleRelation $target, array $written, ManipulationMisuseRule $rule, Derivation $derivation): void
141    {
142        foreach ($target->shape->slots as $slot) {
143            $column = $slot->declaration();
144            if ($column !== null && $column->generated && in_array($slot, $written, true)) {
145                $derivation->report(new ManipulationMisuse($rule, $column->name->value));
146
147                return;
148            }
149        }
150    }
