48/**
49 * Checks the actions of ALTER TABLE against the kind of the altered relation.
50 *
51 * Rule: PG-ALTER-KIND-001. Each action accepts only some kinds of relation
52 * (ATPrepCmd / ATSimplePermissions): for example ADD COLUMN, DROP COLUMN,
53 * SET DATA TYPE, ADD CONSTRAINT and the trigger actions a base table or a
54 * foreign table; ALTER COLUMN ... SET DEFAULT and the identity actions also
55 * a view; SET STATISTICS, column SET/RESET and SET STORAGE also a
56 * materialized view; SET ( storage parameters ) a base table, a view or a
57 * materialized view; SET LOGGED and SET UNLOGGED a base table or a sequence;
58 * the OPTIONS actions only a foreign table; ADD CONSTRAINT ... USING INDEX,
59 * ALTER CONSTRAINT, OF, NOT OF, the row security and rule actions and the
60 * partition actions only a base table; OWNER TO every kind. An action on a
61 * relation of another declared kind is reported as `ALTER action %s cannot
62 * be performed on relation "%s"` with the server's name for the action.
63 * The parent of INHERIT and NO INHERIT must itself be a base table or a
64 * foreign table, and a foreign table takes no primary key, unique,
65 * exclusion or foreign key constraint. Source: https://www.postgresql.org/docs/17/sql-altertable.html,
66 * `tablecmds.c` (ATPrepCmd, alter_table_type_to_string, ATExecAddInherit).
67 * Termination: one check per action. Status: Implemented.
68 *
69 * @visibility SqlSemantics\Platform\PostgreSql
70 */
71final class AlterKinds

83    /**
84     * Reports the actions the declared kind of the relation does not accept.
85     *
86     * @param list<AlterCommand> $commands
87     */
88    public function check(Derivation $derivation, ?RelationKind $kind, Name $relation, array $commands): void
89    {
90        foreach ($kind === null ? [] : $commands as $command) {
91            $accepted = $this->accepted($command);
92            if ($accepted !== null && !in_array($kind, $accepted[1], true)) {
93                $derivation->report(new KindProblem(KindRule::AlterAction, $relation, $accepted[0]));
94            }
95        }
96        foreach ($commands as $command) {
97            if ($command instanceof InheritanceChange && !$command->remove) {
98                $this->parent($derivation, $command);
99            }
100        }
101    }

103    /**
104     * Reports the constraints the added columns and constraints would give a foreign table (PG-RELATION-KIND-001).
105     *
106     * @param list<AlterCommand> $commands
107     */
108    public function foreign(Derivation $derivation, array $commands): void
109    {
110        $elements = [];
111        foreach ($commands as $command) {
112            if ($command instanceof AddColumn) {
113                $elements[] = $command->column;
114            } elseif ($command instanceof AddConstraint) {
115                $elements[] = $command->constraint;
116            }
117        }
118        (new RelationKinds())->foreign($derivation, $elements);
119    }

121    /**
122     * Reports a new parent of INHERIT that is neither a base table nor a foreign table.
123     */
124    public function parent(Derivation $derivation, InheritanceChange $command): void
125    {
126        $kind = (new RelationKinds())->declared($derivation, $command->parent->name);
127        if ($kind !== null && !in_array($kind, self::TABLE_OR_FOREIGN, true)) {
128            $derivation->report(new KindProblem(KindRule::AlterAction, $command->parent->name->name, AlterAction::Inherit));
129        }
130    }

132    /**
133     * Answers the server's name of an action and the kinds of relation it accepts; null for an action every kind accepts.
134     *
135     * @return array{AlterAction, list<RelationKind>}|null
136     */
137    public function accepted(AlterCommand $command): ?array
138    {
139        return match (true) {
140            $command instanceof ColumnAction => $this->column($command->kind),
141            $command instanceof TableAction => $this->table($command->kind),
142            $command instanceof NamedAction => $this->named($command->kind),
143            default => $this->other($command),
144        };
145    }

147    /**
148     * Answers the name and the accepted kinds of the actions that are neither column, table nor named actions.
149     *
150     * @return array{AlterAction, list<RelationKind>}|null
151     */
152    public function other(AlterCommand $command): ?array
153    {
154        return match (true) {
155            $command instanceof AddColumn => [AlterAction::AddColumn, self::TABLE_OR_FOREIGN],
156            $command instanceof DropColumn => [AlterAction::DropColumn, self::TABLE_OR_FOREIGN],
157            $command instanceof TypeChange => [AlterAction::SetDataType, self::TABLE_OR_FOREIGN],
158            $command instanceof DefaultChange => [AlterAction::SetDefault, self::TABLE_VIEW_OR_FOREIGN],
159            $command instanceof AddIdentity => [AlterAction::AddIdentity, self::TABLE_VIEW_OR_FOREIGN],
160            $command instanceof AlterIdentity => [AlterAction::SetColumnAttributes, self::TABLE_VIEW_OR_FOREIGN],
161            $command instanceof SetExpression => [AlterAction::SetExpression, self::TABLE_OR_FOREIGN],
162            $command instanceof ColumnStatistics => [AlterAction::SetStatistics, self::TABLE_MATERIALIZED_OR_FOREIGN],
163            $command instanceof ColumnAttributes => [$command->reset ? AlterAction::ResetOptions : AlterAction::SetColumnAttributes, self::TABLE_MATERIALIZED_OR_FOREIGN],
164            $command instanceof ColumnStorageChange => $command->setting instanceof ColumnStorage ? [AlterAction::SetStorage, self::TABLE_MATERIALIZED_OR_FOREIGN] : [AlterAction::SetCompression, self::TABLE_OR_MATERIALIZED],
165            $command instanceof ColumnForeignOptions => [AlterAction::ColumnOptions, [RelationKind::ForeignTable]],
166            default => $this->relationWide($command),
167        };
168    }

170    /**
171     * Answers the name and the accepted kinds of the actions on the relation as a whole.
172     *
173     * @return array{AlterAction, list<RelationKind>}|null
174     */
175    public function relationWide(AlterCommand $command): ?array
176    {
177        return match (true) {
178            $command instanceof AddConstraint => [AlterAction::AddConstraint, $command->constraint instanceof IndexConstraint ? self::TABLE : self::TABLE_OR_FOREIGN],
179            $command instanceof DropConstraint => [AlterAction::DropConstraint, self::TABLE_OR_FOREIGN],
180            $command instanceof AlterConstraint => [AlterAction::AlterConstraint, self::TABLE],
181            $command instanceof InheritanceChange => [$command->remove ? AlterAction::NoInherit : AlterAction::Inherit, self::TABLE_OR_FOREIGN],
182            $command instanceof OfComposite => [AlterAction::Of, self::TABLE],
183            $command instanceof ReplicaIdentity => [AlterAction::ReplicaIdentity, self::TABLE_OR_MATERIALIZED],
184            $command instanceof RelationOptions => [$command->reset ? AlterAction::ResetRelationOptions : AlterAction::SetRelationOptions, [RelationKind::BaseTable, RelationKind::View, RelationKind::MaterializedView]],
185            $command instanceof SetAccessMethod => [AlterAction::SetAccessMethod, self::TABLE_OR_MATERIALIZED],
186            $command instanceof ForeignOptions => [AlterAction::Options, [RelationKind::ForeignTable]],
187            $command instanceof AttachPartition => [AlterAction::AttachPartition, self::TABLE],
188            $command instanceof AttachIndex => [AlterAction::AttachPartition, []],
189            $command instanceof DetachPartition => [$command->mode === DetachMode::Finalize ? AlterAction::DetachPartitionFinalize : AlterAction::DetachPartition, self::TABLE],
190            default => null,
191        };
192    }
