32/**
33 * Derives CREATE TABLE and CREATE FOREIGN TABLE.
34 *
35 * Rule: PG-TABLE-DEFINITION-001. The new table is named as written; a
36 * temporary table with an unqualified name belongs to the temporary schema
37 * `pg_temp`, and inside CREATE SCHEMA an unqualified table belongs to the
38 * schema being created. The statement provides the declaration of
39 * PG-TABLE-DECLARATION-001 and records it as its own relation fact. The
40 * parents of INHERITS and PARTITION OF are resolved (PG-TABLE-TARGET-001); a
41 * parent of INHERITS must be a table or a foreign table (PG-RELATION-KIND-001).
42 * Every expression of the definition (constraints, generated columns,
43 * partition keys) is derived where the new table is the only visible
44 * relation, its system columns included; defaults and partition bound values
45 * see no column. Problems: a column name written twice, a column named like
46 * a system column, more than one primary key, an array of a serial type, a
47 * primary key, unique, exclusion or foreign key constraint of a foreign table,
48 * and the generation rules of PG-GENERATED-COLUMN-001.
49 * Source: https://www.postgresql.org/docs/17/sql-createtable.html. Termination:
50 * one pass over the elements. Status: Implemented.
51 *
52 * @visibility SqlSemantics\Platform\PostgreSql
53 */
54final class Definitions
55{
56    /**
57     * Provides the declaration and derives the definition; `$schema` is the schema of an enclosing CREATE SCHEMA.
58     */
59    public function derive(CreateTable|CreateForeignTable $create, Derivation $derivation, ?Name $schema): void
60    {
61        $name = $this->name($create, $schema);
62        $table = (new Declarations())->table($create->definition, $derivation, $name, $this->kind($create));
63        $derivation->declare($table);
64        $targets = new Targets();
65        $fact = $derivation->target($create, new RelationFact($targets->shape($table), new DeclaredTable($table)));
66        $scope = $targets->scope($derivation, $create, $name, $fact->shape, $targets->implicit($fact));
67        $this->report($create->definition, $derivation, $name);
68        (new GeneratedColumns())->inherited($derivation, $create->definition);
69        if ($create instanceof CreateTable) {
70            (new CreationSchemas())->check($derivation, $create->name, $create->persistence);
71        } else {
72            (new RelationKinds())->foreign($derivation, $create->definition->elements());
73        }
74        $this->elements($create->definition, $derivation, $scope);
75        if ($create instanceof CreateTable) {
76            $create->partitioning?->deriveClause($derivation, $scope);
77            foreach ($create->options as $option) {
78                $option->deriveClause($derivation, $scope);
79            }
80        }
81    }
82
83    /**
84     * Answers the facts of the new table without providing its declaration.
85     */
86    public function fact(CreateTable|CreateForeignTable $create, Derivation $derivation, ?Name $schema): RelationFact
87    {
88        $table = (new Declarations())->table($create->definition, $derivation, $this->name($create, $schema), $this->kind($create));
89
90        return new RelationFact((new Targets())->shape($table), new DeclaredTable($table));
91    }

93    /**
94     * Answers the kind of relation the definition declares.
95     */
96    public function kind(CreateTable|CreateForeignTable $create): RelationKind
97    {
98        return $create instanceof CreateForeignTable ? RelationKind::ForeignTable : RelationKind::BaseTable;
99    }
100
101    /**
102     * Answers the name the new table is declared under.
103     */
104    public function name(CreateTable|CreateForeignTable $create, ?Name $schema): QualifiedName
105    {
106        $written = $create->name;
107        if ($written->schema !== null) {
108            return $written;
109        }
110        if ($create instanceof CreateTable && $create->persistence->temporary()) {
111            return new QualifiedName($written->name, new Name('pg_temp'));
112        }
113
114        return $schema === null ? $written : new QualifiedName($written->name, $schema);
115    }
116
117    /**
118     * Resolves the parents and derives the elements and the partition bound.
119     */
120    public function elements(TableForm $form, Derivation $derivation, Environment $scope): void
121    {
122        $targets = new Targets();
123        $kinds = new RelationKinds();
124        foreach ($form instanceof ListedColumns ? $form->parents : [] as $parent) {
125            $kind = $kinds->of($derivation->target($parent, $targets->resolve($derivation, $parent->name)));
126            $kinds->require($derivation, $kind, $parent->name->name, [RelationKind::BaseTable, RelationKind::ForeignTable], KindRule::InheritedRelation);
127        }
128        if ($form instanceof PartitionOf) {
129            $derivation->target($form->parent, $targets->resolve($derivation, $form->parent->name));
130            $form->bound->deriveClause($derivation, $scope);
131        }
132        foreach ($form->elements() as $element) {
133            $element->deriveClause($derivation, $scope);
134        }
135    }
136
137    /**
138     * Reports repeated column names (the columns a LIKE clause copies included), system column names, several primary keys and serial arrays.
139     */
140    public function report(TableForm $form, Derivation $derivation, QualifiedName $name): void
141    {
142        $seen = [];
143        $keys = 0;
144        $system = new SystemColumns();
145        foreach ($form->elements() as $element) {
146            $resolution = $element instanceof LikeClause ? $derivation->table($element->table, $derivation->environment()) : null;
147            foreach ($resolution instanceof DeclaredTable && $resolution->table->complete ? $resolution->table->columns : [] as $column) {
148                if (isset($seen[$column->name->value])) {
149                    $derivation->report(new DefinitionProblem(DefinitionRule::DuplicateColumn, $column->name));
150                }
151                $seen[$column->name->value] = true;
152            }
153            if ($element instanceof ColumnDefinition || $element instanceof ColumnOptions) {
154                if (isset($seen[$element->name->value])) {
155                    $derivation->report(new DefinitionProblem(DefinitionRule::DuplicateColumn, $element->name));
156                }
157                $seen[$element->name->value] = true;
158                if ($system->reserved($element->name)) {
159                    $derivation->report(new DefinitionProblem(DefinitionRule::SystemColumnName, $element->name));
160                }
161                foreach ($element->qualifiers as $qualifier) {
162                    $keys += $qualifier instanceof ColumnPrimaryKey ? 1 : 0;
163                }
164            }
165            if ($element instanceof ColumnDefinition && $element->type->array !== null && (new ColumnTyping())->serial($element->type) !== null) {
166                $derivation->report(new DefinitionProblem(DefinitionRule::SerialArray));
167            }
168            $keys += $element instanceof TablePrimaryKey || ($element instanceof IndexConstraint && $element->primary) ? 1 : 0;
169        }
170        if ($keys > 1) {
171            $derivation->report(new DefinitionProblem(DefinitionRule::MultiplePrimaryKeys, $name->name));
172        }
173    }
