308     *

312     * @throws DatabaseException When config is Exception mode and rewrite fails.

314    public function rewrite(string $sql): RewritePlan
315    {
316        try {
317            return $this->rewriter->rewrite($sql);
318        } catch (UnsupportedSqlException $e) {
255            $behavior = $this->config->resolveUnsupportedBehavior($sql);
256
257            if ($behavior === UnsupportedSqlBehavior::Exception) {
258                throw new DatabaseException($e->getMessage(), null, 0, $e);
259            }
260
261            if ($behavior === UnsupportedSqlBehavior::Notice) {
262                trigger_error(
263                    sprintf('[ZTD Notice] Unsupported SQL ignored: %s', $sql),
264                    E_USER_NOTICE
265                );
266            }
267
268            return new RewritePlan($sql, QueryKind::SKIPPED);
319            return $this->refusals->forUnsupported($e, $sql);
320        } catch (UnknownSchemaException $e) {
270            $behavior = $this->config->unknownSchemaBehavior();
271
272            if ($behavior === UnknownSchemaBehavior::Exception) {
273                throw new DatabaseException($e->getMessage(), null, 0, $e);
274            }
275
276            if ($behavior === UnknownSchemaBehavior::Passthrough) {
277                return new RewritePlan($sql, QueryKind::READ);
278            }
279
280            if ($behavior === UnknownSchemaBehavior::Notice) {
281                trigger_error(
282                    sprintf('[ZTD Notice] Unknown table referenced: %s', $e->getIdentifier()),
283                    E_USER_NOTICE
284                );
285            }

287            return new RewritePlan($this->rewriter->emptyResultSelect(), QueryKind::READ);
321            return $this->refusals->forUnknownSchema($e, $sql, $this->rewriter->emptyResultSelect());

400    /**
401     * Apply a mutation without leaking simulation-specific exception types
402     * through the connection adapter boundary.
403     *
404     * @throws DatabaseException When shadow mutation application fails.
405     */
406    private function applyMutation(
407        ShadowMutation $mutation,
408        ResultSet $resultSet,
409        string $sql,
410    ): MutationImpact {
411        $before = $this->shadowStore->get($mutation->tableName());
412        $snapshot = $this->shadowStore->snapshot();
413        try {
414            if ($mutation instanceof ResultSetMutation) {
415                $mutation->applyResultSet($this->shadowStore, $resultSet);
416            } else {
417                $mutation->apply($this->shadowStore, $resultSet->rows);
418            }
419            $this->referentialIntegrity->synchronize(
420                $snapshot,
421                $this->shadowStore,
422                $mutation,
423                $resultSet->rows,
424                $sql,
425            );
426        } catch (SimulationException $e) {
427            $this->shadowStore->restore($snapshot);
428            throw new DatabaseException($e->getMessage(), null, 0, $e);
429        }
430        $impact = new MutationImpact(
431            $mutation,
432            $before,
433            $resultSet->rows,
434            $this->shadowStore->get($mutation->tableName()),
435        );
436        if ($impact->isInsertLike() && $this->rewriter instanceof RewriteStateCommitter) {
437            $this->rewriter->commitRewriteState();
438        }
439        $this->captureLastInsertId($mutation, $impact);
440
441        return $impact;
442    }

444    private function captureLastInsertId(ShadowMutation $mutation, MutationImpact $impact): void
445    {
446        if (!$impact->isInsertLike()) {
447            return;
448        }
449
450        $definition = $this->registry->get($mutation->tableName());
451        $identityColumns = $definition !== null ? array_keys($definition->identityStrategies) : [];
452        if ($identityColumns === [] && $definition !== null && count($definition->primaryKeys) === 1) {
453            $identityColumns = $definition->primaryKeys;
454        }
455        $identityColumn = $identityColumns[0] ?? null;
456        $returningRows = $impact->returningRows();
457        $lastRow = $returningRows[count($returningRows) - 1] ?? null;
458        $identityValue = $identityColumn !== null && $lastRow !== null
459            ? ($lastRow[$identityColumn] ?? null)
460            : null;
461        if (is_int($identityValue) || is_string($identityValue)) {
462            $this->lastInsertId = (string) $identityValue;
463        }
