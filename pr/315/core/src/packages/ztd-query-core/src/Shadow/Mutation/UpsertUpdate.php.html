10/**
11 * What an UPSERT does to a row it found a conflict on.
12 *
13 * Two questions have to be answered about every conflict: whether the
14 * statement goes on to update the row at all, and what the row looks like
15 * once it has. Both depend on where the assignment was worked out — the
16 * database can evaluate an expression against the row as it stood, but not
17 * against a row this same statement changed a moment ago, and then the
18 * expression has to be read here or the statement refused.
19 *
20 * @phpstan-import-type Row from TableDefinition
21 * @phpstan-import-type RowValue from TableDefinition
22 */
23final class UpsertUpdate
24{

26     * @param string $tableName Table being written to
27     * @param array<int, string> $primaryKeys Columns a conflict is found on
28     * @param array<int, string> $updateColumns Columns the statement assigns to
29     * @param array<string, UpsertExpression|null> $updateValues What each column is assigned, as ZTD reads it
30     * @param array<string, string> $updateSqlValues What each column is assigned, as the statement wrote it
31     * @param string|null $updateSqlPredicate Condition on the update, as the statement wrote it
32     * @param UpsertExpression|null $updatePredicate Condition on the update, as ZTD reads it
33     * @param bool $databaseEvaluated Whether the database worked the assignments out already
34     * @param UpsertMutationRow $codec Reads what the database worked out off a row
35     */

63    /**
64     * Reports whether the statement goes on to update the row it conflicted with.
65     *
66     * @param Row $row Row as it arrived, carrying what the database worked out
67     * @param Row $existingRow Row the conflict was found on
68     * @param Row $incomingRow Row the statement was trying to write
69     * @param bool $changedEarlier Whether this same statement already changed the row
70     *
71     * @return bool True when the row is one the statement updates
72     *
73     * @throws UnsupportedSqlException When the condition has to be read here and cannot be
74     */
75    public function applies(array $row, array $existingRow, array $incomingRow, bool $changedEarlier): bool
76    {
77        if (!$this->databaseEvaluated) {
78            return $this->updatePredicate === null
79                || $this->updatePredicate->matches($existingRow, $incomingRow, $this->tableName);
80        }
81        if ($changedEarlier && $this->updateSqlPredicate !== null) {
82            return $this->predicateRead('Sequential UPSERT predicate')
83                ->matches($existingRow, $incomingRow, $this->tableName);
84        }
85        if (array_key_exists($this->codec->predicateColumn(), $row)) {
86            return $this->codec->predicateMatches($row[$this->codec->predicateColumn()]);
87        }
88        if ($this->updateSqlPredicate !== null) {
89            return $this->predicateRead('UPSERT predicate requires local evaluation')
90                ->matches($existingRow, $incomingRow, $this->tableName);
91        }
92
93        return true;
94    }

96    /**
97     * Answers the row as the statement leaves it.
98     *
99     * A statement that names no column to assign carries everything it was
100     * writing over, except what the conflict was found on, which is what
101     * makes it the same row.
102     *
103     * @param Row $row Row as it arrived, carrying what the database worked out
104     * @param Row $existingRow Row the conflict was found on
105     * @param Row $incomingRow Row the statement was trying to write
106     * @param bool $changedEarlier Whether this same statement already changed the row
107     *
108     * @return Row The updated row
109     *
110     * @throws UnsupportedSqlException When an assignment has to be read here and cannot be
111     */
112    public function of(array $row, array $existingRow, array $incomingRow, bool $changedEarlier): array
113    {
114        foreach ($this->updateColumns as $index => $column) {
115            $existingRow = $this->withColumn($existingRow, $column, $index, $row, $incomingRow, $changedEarlier);
116        }
117        if ($this->updateColumns !== []) {
118            return $existingRow;
119        }
120        foreach ($incomingRow as $column => $value) {
121            if (!in_array($column, $this->primaryKeys, true)) {
122                $existingRow[$column] = $value;
123            }
124        }
125
126        return $existingRow;
127    }

129    /**
130     * Answers the row with one column assigned as the statement asked.
131     *
132     * A column the database already worked out is taken from what it worked
133     * out; a column it could not is read here, against the row as this
134     * statement has left it so far, which is what makes assignments written
135     * over one another read in the order they were written.
136     *
137     * @param Row $updatedRow Row as the statement has left it so far
138     * @param string $column Column being assigned
139     * @param int $index Which of the assigned columns this is
140     * @param Row $row Row as it arrived, carrying what the database worked out
141     * @param Row $incomingRow Row the statement was trying to write
142     * @param bool $changedEarlier Whether this same statement already changed the row
143     *
144     * @return Row The row with that column assigned
145     *
146     * @throws UnsupportedSqlException When the assignment has to be read here and cannot be
147     */
148    public function withColumn(
149        array $updatedRow,
150        string $column,
151        int $index,
152        array $row,
153        array $incomingRow,
154        bool $changedEarlier,
155    ): array {
156        if (!$this->databaseEvaluated) {
157            if (isset($this->updateValues[$column])) {
158                $updatedRow[$column] = $this->updateValues[$column]
159                    ->evaluate($updatedRow, $incomingRow, $this->tableName);
160            } elseif (isset($incomingRow[$column])) {
161                $updatedRow[$column] = $incomingRow[$column];
162            }
163
164            return $updatedRow;
165        }
166
167        $worked = $this->codec->valueColumn($index);
168        if ($changedEarlier && isset($this->updateSqlValues[$column])) {
169            $updatedRow[$column] = $this->valueRead($column, 'Sequential UPSERT expression')
170                ->evaluate($updatedRow, $incomingRow, $this->tableName);
171        } elseif (array_key_exists($worked, $row)) {
172            $updatedRow[$column] = $row[$worked];
173        } elseif (isset($this->updateSqlValues[$column])) {
174            $updatedRow[$column] = $this->valueRead($column, 'UPSERT expression requires local evaluation')
175                ->evaluate($updatedRow, $incomingRow, $this->tableName);
176        }
177
178        return $updatedRow;
179    }

200     * Answers what a column is assigned as ZTD reads it, insisting it could be read.
201     *
202     * @param string $column Column being assigned
203     * @param string $reason Why it has to be read here rather than by the database

205     * @return UpsertExpression The assignment

208     */
209    public function valueRead(string $column, string $reason): UpsertExpression
210    {
211        $value = $this->updateValues[$column] ?? null;
212        if ($value === null) {
213            throw new UnsupportedSqlException($this->updateSqlValues[$column] ?? '', $reason);
214        }
215
216        return $value;
217    }
