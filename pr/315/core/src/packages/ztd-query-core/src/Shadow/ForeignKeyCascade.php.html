15/**
16 * Applies to one child table what happened to its parent.
17 *
18 * A parent row that moved only reaches its children where nothing else holds
19 * the key up, so each candidate is checked against the parent table first. An
20 * update carries new values to write; a delete carries none, and the action
21 * decides whether the child goes with it, is set to null, or refuses the
22 * statement outright.
23 *
24 * What comes back is what happened to the child table, in the same shape as
25 * what happened to the parent, which is what lets the caller keep following
26 * the constraints outward without knowing how deep it is.
27 *
28 * @phpstan-import-type Row from TableDefinition
29 * @phpstan-import-type RowValue from TableDefinition
30 */
31final class ForeignKeyCascade
32{

34     * @param ForeignKeyEnds $ends Answers which parent columns a key points at
35     * @param ParentKeyLookup $parents Reports whether a referenced row is still there
36     * @param RowMatch $rows Finds a row among rows
37     */
38    public function __construct(
39        private readonly ForeignKeyEnds $ends,
40        private readonly ParentKeyLookup $parents = new ParentKeyLookup(),
41        private readonly RowMatch $rows = new RowMatch(),
42    ) {
43    }

45    /**
46     * Answers what one constraint makes of what happened to the parent.
47     *
48     * @param ShadowStore $store Shadow the child rows live in, written back in place
49     * @param string $childTable Table holding the key
50     * @param string $constraintName Constraint being followed, for the refusal
51     * @param ForeignKeyDefinition $foreignKey Constraint being followed
52     * @param TableTransition $parent What happened to the parent table
53     * @param string $sql Statement being simulated, for the refusal
54     *
55     * @return TableTransition|null What happened to the child, or null when nothing did
56     *
57     * @throws ForeignKeyViolationException When the action forbids the statement
58     */
59    public function of(
60        ShadowStore $store,
61        string $childTable,
62        string $constraintName,
63        ForeignKeyDefinition $foreignKey,
64        TableTransition $parent,
65        string $sql,
66    ): ?TableTransition {
67        if (!$this->ends->areBalanced($foreignKey)) {
68            return null;
69        }
70
71        $constraint = new FollowedConstraint($childTable, $constraintName, $foreignKey, $sql);
72        $referencedColumns = $this->ends->referencedColumns($foreignKey);
73        $children = new CascadedChildren(array_values($store->get($childTable)));
74        foreach ($parent->updated as $parentChange) {
75            $this->carryUpdate($children, $store, $constraint, $referencedColumns, $parentChange);
76        }
77        foreach ($parent->deleted as $parentRow) {
78            $this->carryDelete($children, $store, $constraint, $referencedColumns, $parentRow);
79        }
80        if ($children->areUnchanged()) {
81            return null;
82        }
83
84        $store->set($childTable, $children->rows());
85
86        return new TableTransition($childTable, $children->deleted(), $children->updated());
87    }

89    /**
90     * Carries a parent row that moved to the children that were holding it.
91     *
92     * A key the parent table still holds up somewhere else has not moved as
93     * far as the child is concerned, so nothing is carried for it.
94     *
95     * @param CascadedChildren $children Child rows as the cascade has left them
96     * @param ShadowStore $store Shadow the parent rows live in
97     * @param FollowedConstraint $constraint Constraint being followed
98     * @param list<string> $referencedColumns Parent columns the key points at
99     * @param RowChange $parentChange What happened to the one parent row
100     *
101     * @throws ForeignKeyViolationException When the action forbids the statement
102     */
103    public function carryUpdate(
104        CascadedChildren $children,
105        ShadowStore $store,
106        FollowedConstraint $constraint,
107        array $referencedColumns,
108        RowChange $parentChange,
109    ): void {
110        $foreignKey = $constraint->foreignKey;
111        $oldValues = $this->rows->valuesOf($parentChange->before, $referencedColumns);
112        $newValues = $this->rows->valuesOf($parentChange->after, $referencedColumns);
113        if ($oldValues === null
114            || $newValues === null
115            || $this->parents->exists($store, $foreignKey->referencedTable, $referencedColumns, $oldValues)
116        ) {
117            return;
118        }
119
120        foreach ($children->rows() as $index => $row) {
121            if (!$this->rows->carries($row, $foreignKey->columns, $oldValues)) {
122                continue;
123            }
124            $children->replace($index, $this->applyAction($row, $newValues, $foreignKey->onUpdate, $constraint));
125        }
126    }

128    /**
129     * Carries a parent row that went to the children that were holding it.
130     *
131     * @param CascadedChildren $children Child rows as the cascade has left them
132     * @param ShadowStore $store Shadow the parent rows live in
133     * @param FollowedConstraint $constraint Constraint being followed
134     * @param list<string> $referencedColumns Parent columns the key points at
135     * @param Row $parentRow The parent row that went
136     *
137     * @throws ForeignKeyViolationException When the action forbids the statement
138     */
139    public function carryDelete(
140        CascadedChildren $children,
141        ShadowStore $store,
142        FollowedConstraint $constraint,
143        array $referencedColumns,
144        array $parentRow,
145    ): void {
146        $foreignKey = $constraint->foreignKey;
147        $oldValues = $this->rows->valuesOf($parentRow, $referencedColumns);
148        if ($oldValues === null
149            || $this->parents->exists($store, $foreignKey->referencedTable, $referencedColumns, $oldValues)
150        ) {
151            return;
152        }
153
154        $gone = [];
155        foreach ($children->rows() as $index => $row) {
156            if (!$this->rows->carries($row, $foreignKey->columns, $oldValues)) {
157                continue;
158            }
159            if ($foreignKey->onDelete === ReferentialAction::Cascade) {
160                $gone[] = $index;
161                continue;
162            }
163            $children->replace($index, $this->applyAction($row, [], $foreignKey->onDelete, $constraint));
164        }
165        $children->remove($gone);
166    }

168    /**
169     * Writes into a child row what the declared action says should be there.
170     *
171     * Only CASCADE and SET NULL let the statement stand. Every other action —
172     * RESTRICT, NO ACTION, SET DEFAULT — means the parent was not free to move,
173     * so the statement is refused rather than quietly rewritten.
174     *
175     * @param Row $row Child row to rewrite
176     * @param list<RowValue> $values Values to carry over, empty when the parent went
177     * @param ReferentialAction $action Action the constraint declares
178     * @param FollowedConstraint $constraint Constraint being followed, for the refusal
179     *
180     * @return Row The child row as it should now be
181     *
182     * @throws ForeignKeyViolationException When the action forbids the statement
183     */
184    public function applyAction(
185        array $row,
186        array $values,
187        ReferentialAction $action,
188        FollowedConstraint $constraint,
189    ): array {
190        if ($action !== ReferentialAction::Cascade && $action !== ReferentialAction::SetNull) {
191            throw ForeignKeyViolationException::of(
192                $constraint->sql,
193                $constraint->childTable,
194                $constraint->constraintName,
195                $constraint->foreignKey->referencedTable,
196                $this->ends->referencedColumns($constraint->foreignKey),
197            );
198        }
199
200        foreach ($constraint->foreignKey->columns as $index => $column) {
201            $row[$column] = $action === ReferentialAction::SetNull ? null : ($values[$index] ?? null);
202        }
203
204        return $row;
205    }
