17/**
18 * Writes one simulated statement into the shadow, all of it or none of it.
19 *
20 * A statement that is refused part-way through must leave the shadow as it
21 * was, or every later statement is simulated against a state the database
22 * would never have been in, so the shadow is taken back to the snapshot
23 * before the refusal is passed on.
24 *
25 * What comes out through here is a driver-shaped failure. A caller went
26 * through an adapter that looks like PDO or mysqli, and a simulation-specific
27 * exception surfacing there would be a type it has no reason to know about.
28 */
29final class ShadowApplication
30{
31    /**
32     * @param ShadowStore $shadowStore Shadow the statement is written into
33     * @param ReferentialIntegrityEnforcer $referentialIntegrity Carries the consequences outward
34     * @param TableDefinitionRegistry $registry Answers what a table declares
35     * @param SqlRewriter $rewriter Rewriter whose own state moves on with a committed insert
36     */
37    public function __construct(
38        private readonly ShadowStore $shadowStore,
39        private readonly ReferentialIntegrityEnforcer $referentialIntegrity,
40        private readonly TableDefinitionRegistry $registry,
41        private readonly SqlRewriter $rewriter,
42    ) {
43    }
44
45    /**
46     * Writes a mutation into the shadow and answers what it came to.
47     *
48     * @param ShadowMutation $mutation Mutation to write
49     * @param ResultSet $resultSet What the rewritten statement read back
50     * @param string $sql Statement being simulated, for the refusal
51     *
52     * @return MutationImpact What the statement came to
53     *
54     * @throws DatabaseException When the shadow refuses the statement
55     */
56    public function apply(ShadowMutation $mutation, ResultSet $resultSet, string $sql): MutationImpact
57    {
58        $before = $this->shadowStore->get($mutation->tableName());
59        $snapshot = $this->shadowStore->snapshot();
60        try {
61            if ($mutation instanceof ResultSetMutation) {
62                $mutation->applyResultSet($this->shadowStore, $resultSet);
63            } else {
64                $mutation->apply($this->shadowStore, $resultSet->rows);
65            }
66            $this->referentialIntegrity->synchronize(
67                $snapshot,
68                $this->shadowStore,
69                $mutation,
70                $resultSet->rows,
71                $sql,
72            );
73        } catch (SimulationException $refusal) {
74            $this->shadowStore->restore($snapshot);
75
76            throw new DatabaseException($refusal->getMessage(), null, 0, $refusal);
77        }
78
79        $impact = new MutationImpact(
80            $mutation,
81            $before,
82            $resultSet->rows,
83            $this->shadowStore->get($mutation->tableName()),
84        );
85        if ($impact->isInsertLike() && $this->rewriter instanceof RewriteStateCommitter) {
86            $this->rewriter->commitRewriteState();
87        }
88
89        return $impact;
90    }

92    /**
93     * Answers the identity a driver would report for a simulated insert.
94     *
95     * A table says which column a database numbers; where it says nothing and
96     * the key is one column, that column is what a database would have
97     * numbered. Only the last row counts, which is what lastInsertId() means.
98     *
99     * @param ShadowMutation $mutation Mutation that was written
100     * @param MutationImpact $impact What it came to
101     *
102     * @return string|null The identity, or null when nothing was inserted or nothing was numbered
103     */
104    public function lastInsertIdOf(ShadowMutation $mutation, MutationImpact $impact): ?string
105    {
106        if (!$impact->isInsertLike()) {
107            return null;
108        }
109
110        $definition = $this->registry->get($mutation->tableName());
111        $identityColumns = $definition !== null ? array_keys($definition->identityStrategies) : [];
112        if ($identityColumns === [] && $definition !== null && count($definition->primaryKeys) === 1) {
113            $identityColumns = $definition->primaryKeys;
114        }
115        $identityColumn = $identityColumns[0] ?? null;
116        $returningRows = $impact->returningRows();
117        $lastRow = $returningRows[count($returningRows) - 1] ?? null;
118        $identityValue = $identityColumn !== null && $lastRow !== null
119            ? ($lastRow[$identityColumn] ?? null)
120            : null;
121
122        return is_int($identityValue) || is_string($identityValue) ? (string) $identityValue : null;
123    }
