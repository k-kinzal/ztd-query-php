319                        break;
320                    }

322                        && substr_compare($sql, $opening, $offset, strlen($opening)) === 0
323                    ) {
324                        $commentDepth++;

327                    }
328                    if (substr_compare($sql, $closing, $offset, strlen($closing)) === 0) {
329                        $commentDepth--;

334                }
335                $tokens[] = self::token($sql, SqlTokenKind::Comment, $start, $offset, $depth, $bracketDepth);

366            }

373            }

381                    if ($profile->isIdentifierPart($sql[$offset])) {

396            }

400                while ($offset < $length && $profile->isIdentifierPart($sql[$offset])) {
401                    $offset++;
402                }
403                $tokens[] = self::token($sql, SqlTokenKind::Word, $start, $offset, $depth, $bracketDepth);

431    private static function scanDelimited(
432        string $sql,
433        int $offset,
434        string $opening,
435        string $closing,
436        bool $backslashEscapes,
437    ): int {
438        $offset += strlen($opening);
439        while (isset($sql[$offset])) {
440            if (substr_compare($sql, $closing, $offset, strlen($closing)) === 0) {
441                if (substr_compare($sql, $closing . $closing, $offset, strlen($closing) * 2) === 0) {
442                    $offset += strlen($closing) * 2;
443                    continue;
444                }
445
446                return $offset + strlen($closing);
447            }
448            if ($backslashEscapes && $sql[$offset] === '\\') {
449                $offset += 2;
450                continue;
451            }
452            $offset++;
453        }
454
455        return strlen($sql);
456    }
