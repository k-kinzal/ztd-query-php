32     * Reads the first parameter's state after the callback's possible effects.
33     *
34     * @param list<Domain> $arguments
35     */
36    public function apply(Node\FunctionLike $callback, array $arguments, Environment $outer, FunctionScope $scope, ExpressionEvaluator $expressions): Domain
37    {
38        if (!$this->supported($callback)) {
39            return Domain::unknown('Callback control flow is not modelled');
40        }
41        $start = $this->bind($callback, $arguments, $outer);
42        $parameter = $callback->getParams()[0]->var ?? null;
43        if (!$parameter instanceof Expr\Variable || !is_string($parameter->name)) {
44            return Domain::unknown('Callback has no named receiver parameter');
45        }
46        $needs = [$parameter->name => true];
47        $identity = $start->read($parameter->name)->soleObject()?->identity;
48        foreach ($start->names() as $name) {
49            if ($identity !== null && $start->read($name)->soleObject()?->identity === $identity) {
50                $needs[$name] = true;
51            }
52        }
53        if ($callback instanceof Expr\ArrowFunction) {
54            $expressions->evaluate($callback->expr, $start, $scope);
55
56            return $start->read($parameter->name);
57        }
58        $statements = array_values($callback->getStmts() ?? []);
59        $terms = [];
60        $widened = false;
61        foreach ($this->slicer->walkList($statements, count($statements), [Pending::needing($needs)]) as $path) {
62            foreach ($this->executor->run($path->forward(), $start->copy(), $scope, $expressions) as $environment) {
63                $value = $environment->read($parameter->name);
64                $terms = array_merge($terms, $value->terms);
65                $widened = $widened || $path->truncated || $value->widened;
66            }
67        }
68
69        return Domain::fromTerms($terms, $widened);
70    }

72    /**
73     * Only terminal returns of the same receiver can be interpreted as mutations.
74     */
75    public function supported(Node\FunctionLike $callback): bool
76    {
77        $parameter = $callback->getParams()[0]->var ?? null;
78        if (!$parameter instanceof Expr\Variable || !is_string($parameter->name)) {
79            return false;
80        }
81        $statements = $callback->getStmts() ?? [];
82        $returns = (new \PhpParser\NodeFinder())->findInstanceOf($statements, Node\Stmt\Return_::class);
83        foreach ($returns as $return) {
84            if ($return !== end($statements)) {
85                return false;
86            }
87            if ($return->expr !== null && (new ObjectEffects())->root($return->expr) !== $parameter->name
88                && !($return->expr instanceof Expr\ConstFetch && strtolower($return->expr->name->toString()) === 'null')) {
89                return false;
90            }
91        }
92
93        return true;
94    }

99     * @param list<Domain> $arguments
100     */
101    public function bind(Node\FunctionLike $callback, array $arguments, Environment $outer): Environment
102    {
103        $environment = $outer->copy();
104        if ($callback instanceof Expr\Closure) {
105            $captured = ['this'];
106            foreach ($callback->uses as $use) {
107                if (is_string($use->var->name)) {
108                    $captured[] = $use->var->name;
109                }
110            }
111            foreach (array_diff($environment->names(), $captured) as $name) {
112                $environment->forget($name);
113            }
114        }
115        foreach ($callback->getParams() as $position => $parameter) {
116            if ($parameter->var instanceof Expr\Variable && is_string($parameter->var->name)) {
117                $environment->write($parameter->var->name, $arguments[$position] ?? Domain::unknown('Callback argument'));
118            }
119        }
120
121        return $environment;
122    }
