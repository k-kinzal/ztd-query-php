17/**
18 * Reads constants: global ones, class ones, enum cases and `::class`.
19 *
20 * A constant is a value the source writes down once and names everywhere
21 * else, which is exactly the kind of value that decides a table or a column
22 * name, so it is resolved to what the declaration says.
23 *
24 * @visibility root
25 */
26final class ConstantReader
27{
28    private ProgramIndex $index;
29
30    private NodeText $text;
31
32    /**
33     * Wires the reader to the declarations it resolves names against.
34     */
35    public function __construct(ProgramIndex $index, NodeText $text)
36    {
37        $this->index = $index;
38        $this->text = $text;
39    }
40
41    /**
42     * The value of a global constant.
43     */
44    public function readConstant(Expr\ConstFetch $node, FunctionScope $scope, ExpressionEvaluator $expressions): Domain
45    {
46        $name = $this->constantName($node);
47        $keyword = strtolower($node->name->toString());
48        if ($keyword === 'true' || $keyword === 'false') {
49            return Domain::literal($keyword === 'true');
50        }
51        if ($keyword === 'null') {
52            return Domain::literal(null);
53        }
54
55        $declared = $this->index->findConstant($name);
56        if ($declared !== null) {
57            return $expressions->evaluate($declared, new Environment(), $scope);
58        }
59
60        return $this->readRuntimeConstant($name);
61    }
62
63    /**
64     * The name a constant reference resolves to.
65     *
66     * An unqualified constant is looked for in its own namespace first and in
67     * the global one second, which is the order PHP itself resolves it in.
68     */
69    public function constantName(Expr\ConstFetch $node): string
70    {
71        $namespaced = $node->name->getAttribute('namespacedName');
72        if ($namespaced instanceof Node\Name && $this->index->findConstant($namespaced->toString()) !== null) {
73            return $namespaced->toString();
74        }
75
76        return $node->name->toString();
77    }
78
79    /**
80     * The value of a constant the running process already defines.
81     */
82    public function readRuntimeConstant(string $name): Domain
83    {
84        $bare = ltrim($name, '\\');
85        if (!str_starts_with($bare, 'PHP_') || !defined($bare)) {
86            return Domain::opaque(TypeShape::unknown(), Origin::Unresolved, $bare);
87        }
88        $value = constant($bare);
89
90        return is_scalar($value) || $value === null ? Domain::literal($value) : Domain::unknown($bare);
91    }
92
93    /**
94     * The value of a class constant, an enum case, or a `::class` reference.
95     */
96    public function readClassConstant(
97        Expr\ClassConstFetch $node,
98        FunctionScope $scope,
99        ExpressionEvaluator $expressions,
100    ): Domain {
101        $className = $this->resolveClassName($node->class, $scope);
102        $constant = $node->name instanceof Node\Identifier ? $node->name->toString() : null;
103        if ($className === null || $constant === null) {
104            return Domain::opaque(TypeShape::unknown(), Origin::Unresolved, $this->text->render($node));
105        }
106        if (strtolower($constant) === 'class') {
107            return Domain::literal($className);
108        }
109
110        $shape = $this->index->findClass($className);
111        if ($shape !== null && $shape->enum && array_key_exists($constant, $shape->enumCases)) {
112            return Domain::of(new ObjectTerm($shape->name, $constant));
113        }
114
115        $declared = $this->index->findClassConstant($className, $constant);
116
117        return $declared === null
118            ? Domain::opaque(TypeShape::unknown(), Origin::Unresolved, $className . '::' . $constant)
119            : $expressions->evaluate($declared, new Environment(), $scope);
120    }
121
122    /**
123     * The class a written class reference names, resolving `self` and `static`.
124     */
125    public function resolveClassName(Node\Name|Expr|Node\Stmt\Class_ $reference, FunctionScope $scope): ?string
126    {
127        if (!$reference instanceof Node\Name) {
128            return null;
129        }
130        $written = $reference->toString();
131
132        return match (strtolower($written)) {
133            'self', 'static', 'parent' => $scope->className,
134            default => $written,
135        };
136    }
137}
