15/**
16 * Formats SQL with the dialect, release, and lexical settings of a supplied parser.
17 *
18 * Compact canonicalizes optional syntax and keywords; other styles preserve tokens.
19 * Output is parsed again to verify the selected style's preservation contract.
20 *
21 * @visibility public
22 * @example Formatting a query
23 *     $formatter = new \SqlFormatter\Formatter(new \SqlParser\Sqlite\SqliteParser());
24 *     $formatter->format('SELECT id,name FROM users') // => "SELECT\n    id,\n    name\nFROM\n    users"
25 * @example Selecting compact output
26 *     $formatter = new \SqlFormatter\Formatter(new \SqlParser\MySql\MySqlParser(), new \SqlFormatter\FormatOptions(\SqlFormatter\Style::Compact));
27 *     $formatter->format("SELECT  id, name\nFROM users") // => 'SELECT id,name FROM users'
28 */
29final class Formatter
30{
31    /**
32     * Reuses a configured parser and an immutable layout preset across calls.
33     */
34    public function __construct(
35        private readonly MySqlParser|PostgreSqlParser|SqliteParser $parser,
36        private readonly FormatOptions $options = new FormatOptions(),
37    ) {
38    }
39
40    /**
41     * @throws SourceException When the input is not accepted by the selected grammar
42     * @throws FormattingException When output verification detects changed syntax
43     */
44    public function format(string $sql): string
45    {
46        $tree = $this->parser->parse($sql);
47        if ($this->options->style === Style::Compact) {
48            return (new Compact\Renderer($this->parser))->render($tree);
49        }
50        $result = (new Renderer(Document::from($tree), $this->options))->render();
51        try {
52            $formatted = $this->parser->parse($result);
53        } catch (SourceException $exception) {
54            throw new FormattingException('Formatted SQL could not be parsed with the original grammar.', 0, $exception);
55        }
56        if (Fingerprint::of($tree) !== Fingerprint::of($formatted)) {
57            throw new FormattingException('Formatting changed the SQL syntax tree.');
58        }
59        return $result;
60    }
