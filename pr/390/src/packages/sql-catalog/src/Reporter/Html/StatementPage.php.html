114    /**
115     * The page numbers a pager shows, with null where it skips a run of them.
116     *

118     */
119    public function windowOf(int $number, int $count): array
120    {
121        $shown = [];
122        for ($page = 1; $page <= $count; $page++) {
123            $near = abs($page - $number) <= self::WINDOW;
124            if ($page === 1 || $page === $count || $near) {
125                $shown[] = $page;
126                continue;
127            }
128            if ($shown !== [] && $shown[count($shown) - 1] !== null) {
129                $shown[] = null;

131        }
132
