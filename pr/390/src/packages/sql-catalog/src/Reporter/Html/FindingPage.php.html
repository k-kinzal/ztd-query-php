40        }
41        $grouped = $this->group($catalog);

43        $sections = '';
44        foreach ($counts as $value => $count) {

46            $sections .= $this->section($site, $rule, $count, $grouped[$value] ?? []);
47        }
48
49        return '<h1>Findings <span class="count">'
50            . $this->text->escape($this->text->plural($stats->findings(), 'finding')) . '</span></h1>'
51            . '<p class="lede">A finding is a judgement about a statement the analyzer read. '
52            . 'What stopped the analysis is reported here too, so a gap in the catalog is never silent.</p>'
53            . $sections;
54    }

56    /**
57     * The statements each rule reported on, worst first.
58     *
59     * @return array<string, list<array{CatalogEntry, Finding}>>
60     */
61    public function group(Catalog $catalog): array
62    {
63        $grouped = [];
64        foreach ($catalog->sorted() as $entry) {
65            foreach ($entry->findings as $finding) {
