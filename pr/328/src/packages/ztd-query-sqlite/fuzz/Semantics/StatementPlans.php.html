93            ->withRule('from', RulePlan::any()->allowing(ProductionPattern::exactly('FROM', 'seltablist')))
94            ->withRule('seltablist', self::identifier('users')->allowing(ProductionPattern::exactly('stl_prefix', 'nm', 'dbnm', 'as', 'on_using')))
95            ->withRule('where_opt', RulePlan::any()->allowing(ProductionPattern::exactly('WHERE', 'expr'))->withRule('expr', $predicate));

163    public static function projection(array $expressions): RulePlan
164    {
165        $prefix = RulePlan::any()->allowing(ProductionPattern::exactly());
166        $projection = RulePlan::any();
167        foreach ($expressions as $expression) {
168            $projection = RulePlan::any()->allowing(ProductionPattern::exactly('sclp', 'scanpt', 'expr', 'scanpt', 'as'))
169                ->withRule('expr', $expression)->withRule('sclp', $prefix);
170            $prefix = RulePlan::any()->allowing(ProductionPattern::exactly('selcollist', 'COMMA'))->withRule('selcollist', $projection);
171        }
172        return $projection;
