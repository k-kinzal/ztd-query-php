12/**
13 * Writes the value of one bold field section of a card.
14 *
15 * Prose is escaped text, metadata a nested list, and list fields bullet lists or "None."
16 * when empty.
17 */
18final class FieldWriter
19{
20    /**
21     * Writes a field value.
22     *
23     * @param string $name The field name
24     * @param mixed $value The value in the shape of the YAML field
25     * @param array<string, string> $links Link targets read for this field by linked ID
26     *
27     * @return string The Markdown section body
28     *
29     * @throws InvalidInputException When the value cannot be written for that field
30     * @throws JsonException When a metadata value cannot be encoded
31     */
32    public function write(string $name, mixed $value, array $links): string
33    {
34        if (is_string($value)) {
35            return Nodes::escape($value);
36        }
37        if ($name === 'metadata') {
38            return rtrim((new MetadataWriter())->write($value));
39        }
40        $rows = [];
41        foreach (Fields::sequence($value, $name) as $entry) {
42            if (in_array($name, ['requirements', 'related'], true) && is_string($entry)) {
43                $rows[] = '- [' . Nodes::escape($entry) . '](' . Nodes::destination($links[$entry] ?? ('#' . Nodes::anchor($entry))) . ')';
44            } else {
45                $record = Record::fields($entry);
46                if ($name === 'tests') {
47                    $row = '- **' . Nodes::escape(Fields::text($record, 'runner')) . ':** ' . Nodes::escape(Fields::text($record, 'target'));
48                    if (isset($record['run'])) {
49                        $row .= "\n  - **run:** " . Nodes::escape(Fields::text($record, 'run'));
50                    }
51                    $rows[] = $row;
52                } elseif ($name === 'design') {
53                    $rows[] = isset($record['url']) ? '- [' . Nodes::escape(Fields::text($record, 'text', Fields::text($record, 'url'))) . '](' . Nodes::destination(Fields::text($record, 'url')) . ')' : '- ' . Nodes::escape(Fields::text($record, 'text'));
54                } else {
55                    throw new InvalidInputException("Cannot render Markdown field '$name'.");
56                }
57            }
58        }
59        return $rows === [] ? 'None.' : implode("\n", $rows);
60    }
61}
