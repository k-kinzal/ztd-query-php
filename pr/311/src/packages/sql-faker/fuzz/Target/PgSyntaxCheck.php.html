86    /**
87     * Verifies that PostgreSQL parses the generated statement.
88     *
89     * @param string $sql Statement produced by the grammar
90     * @param string $input Fuzzer input that produced the statement, so a finding can be replayed
91     *
92     * @throws Error When PostgreSQL rejects the statement for a reason the grammar should not produce
93     */
94    public function verify(string $sql, string $input): void
95    {
96        if ($sql === '') {
97            throw new Error("Statement generation returned an empty string\nInput (hex): " . bin2hex($input));
98        }
99        if (pg_connection_status($this->connection) !== PGSQL_CONNECTION_OK || pg_send_prepare($this->connection, '', $sql) === false) {
100            fwrite(STDERR, 'PostgreSQL connection failed: ' . pg_last_error($this->connection) . "\n");
101            exit(2);
102        }
103        $result = pg_get_result($this->connection);
104        if ($result === false) {
105            fwrite(STDERR, "PostgreSQL Parse returned no result.\n");
106            exit(2);
107        }
108        $status = pg_result_status($result);
109        $state = pg_result_error_field($result, PGSQL_DIAG_SQLSTATE);
110        $error = pg_result_error($result);
111        $message = is_string($error) ? trim($error) : 'No server error text.';
112        pg_free_result($result);
113        while (($extra = pg_get_result($this->connection)) !== false) {
114            pg_free_result($extra);
115        }
116        if ($status === PGSQL_COMMAND_OK) {
117            return;
118        }
119        $state = is_string($state) ? $state : '';
120        if (str_starts_with($state, '08') || in_array($state, ['57P01', '57P02', '57P03'], true)) {
121            fwrite(STDERR, "PostgreSQL connection failed: $message\n");
122            exit(2);
123        }
124        if (isset(self::IGNORED_STATES[$state])) {
125            return;
126        }
127        foreach (self::IGNORED_MESSAGES as [$ignoredState, $pattern, $reason]) {
128            if ($state === $ignoredState && preg_match($pattern, $message) === 1) {
129                return;
130            }
131        }
132        throw new Error(
133            "Unexpected error in generated SQL\n" .
134            'Input (hex): ' . bin2hex($input) . "\n" .
135            "SQL: $sql\n" .
136            "SQLSTATE: $state\n" .
137            "Error: $message"
138        );
139    }
