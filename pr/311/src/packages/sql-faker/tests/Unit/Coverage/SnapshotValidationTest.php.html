236                'stmt',
237                'test-v1',
238            ),
239            'revision-a',
240        );
241        $base = $grammarCoverage->snapshot();
242        $cases = [null, [], ['formatVersion' => '1'], ['formatVersion' => 1]];
243        foreach (['grammarFingerprint', 'generatorRevision', 'root', 'inventoryDigest', 'cumulative', 'checkpoint'] as $field) {
244            $missing = $base;
245            unset($missing[$field]);
246            $cases[] = $missing;
247            $cases[] = array_replace($base, [$field => 7]);
248        }
249        foreach (['reachedIds', 'emittedIds'] as $field) {
250            foreach ([null, [false], ['name' => 'id']] as $invalid) {
251                $cases[] = array_replace($base, ['cumulative' => array_replace($base['cumulative'], [$field => $invalid])]);
252            }
253        }
254        foreach (['savedAt' => 1, 'runId' => false, 'generationsObservedInRun' => '2', 'generationInProgress' => 0] as $field => $invalid) {
255            $cases[] = array_replace($base, ['checkpoint' => array_replace($base['checkpoint'], [$field => $invalid])]);
256        }
257        return array_values(array_map(static fn ($case): array => [json_encode($case, JSON_THROW_ON_ERROR)], $cases));
258    }
