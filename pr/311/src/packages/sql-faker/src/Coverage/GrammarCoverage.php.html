72     * @throws CoverageException When already registered to an incompatible generator

74    public function register(GrammarCoverageInventory $inventory, string $revision): void
75    {
76        if ($this->inventory !== null) {
77            if ($this->inventory->digest !== $inventory->digest || $this->revision !== $revision) {
78                throw new CoverageException('Coverage is already attached to a different generator.');
79            }
80            return;
81        }
82        $this->inventory = $inventory;
83        $this->revision = $revision;
84        try {
85            $this->restore();
86        } catch (CoverageException $failure) {
87            $this->inventory = null;
88            $this->revision = '';
89            $this->store = null;
90            $this->saved = new CoverageSets();
91            throw $failure;
92        }

96     * Restores only after compatibility identity is known; failed setup remains retryable.
97     *
98     * @throws CoverageException When history is corrupt, incompatible or inaccessible
99     * @visibility root
100     */
101    public function restore(): void
102    {
103        if ($this->storageDirectory === null) {
104            return;
105        }
106        $inventory = $this->inventory();
107        $key = hash('sha256', '1:' . $inventory->fingerprint . ':' . $this->revision);
108        $this->store = new CoverageSnapshotStore($this->storageDirectory, $key);
109        $json = $this->store->read();
110        if ($json !== null) {
111            $history = (new SnapshotValidation())->decode($json);
112            $this->merge($history);
113            $this->restoredCheckpoint = $history['checkpoint'];
114            $this->dirty = false;
115        }
116    }

256        $cumulative->include($this->current);
257        return ['formatVersion' => 1, 'grammarFingerprint' => $inventory->fingerprint,
258            'generatorRevision' => $this->revision, 'root' => $inventory->root, 'inventoryDigest' => $inventory->digest,
259            'denominatorIds' => $inventory->denominator, 'inventory' => $inventory->entries,
260            'adaptations' => $inventory->adaptations, 'restoredCheckpoint' => $this->restoredCheckpoint,
261            'current' => $this->current->measurement($inventory->denominator),
262            'cumulative' => $cumulative->measurement($inventory->denominator),
263            'checkpoint' => ['savedAt' => gmdate('c'), 'runId' => $this->runId,

267    /**
268     * Unions compatible cumulative history without counting it as current observation.
269     *
270     * @param History $snapshot
271     * @throws CoverageException When identity, inventory or saved IDs are invalid
272     */
273    public function merge(array $snapshot): void
274    {
275        (new SnapshotValidation())->compatible($snapshot, $this->snapshot(), $this->inventory());
276        $reached = array_fill_keys($snapshot['cumulative']['reachedIds'], true);
277        $emitted = array_fill_keys($snapshot['cumulative']['emittedIds'], true);
278        $this->dirty = $this->dirty || array_diff_key($reached, $this->saved->reached) !== []
279            || array_diff_key($emitted, $this->saved->emitted) !== [];
280        $this->saved->reached += $reached;
281        $this->saved->emitted += $emitted;
282    }

284    /**
285     * Saves dirty cumulative sets without clearing this run's observations.
286     *
287     * @throws CoverageException When a snapshot cannot be encoded
288     */
289    public function flush(): void
290    {
291        if ($this->store === null || !$this->dirty) {
292            return;
293        }
294        try {
295            $snapshot = $this->snapshot();
296            unset($snapshot['current'], $snapshot['inventory'], $snapshot['adaptations'],
297                $snapshot['denominatorIds'], $snapshot['restoredCheckpoint']);
298            $snapshot['cumulative'] = ['reachedIds' => $snapshot['cumulative']['reachedIds'],
299                'emittedIds' => $snapshot['cumulative']['emittedIds']];
300            $json = json_encode($snapshot, JSON_THROW_ON_ERROR | JSON_PRETTY_PRINT);
301        } catch (JsonException $exception) {
302            throw new CoverageException('Cannot encode coverage snapshot.', 0, $exception);
303        }
304        $this->store->write($json);
305        $this->saved->include($this->current);
306        $this->dirty = false;
307    }
