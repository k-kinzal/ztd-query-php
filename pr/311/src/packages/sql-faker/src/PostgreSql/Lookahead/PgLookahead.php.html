7/**
8 * The token substitutions PostgreSQL's parser frontend makes by looking ahead.
9 *
10 * A few words mean different things depending on what follows them: `NOT` before
11 * `NULL` is not the `NOT` of an expression, `WITH` before `TIME` is not the
12 * `WITH` of a CTE. PostgreSQL resolves this between its lexer and its parser by
13 * rewriting the first token once the second is known.
14 *
15 * Generation and tokenizing both have to apply the same rewriting or they would
16 * disagree about what the same text means, and realization has to be able to
17 * walk it backwards to find the keyword a substituted token is spelled as.
18 *
19 * @phpstan-type LookaheadRule array{token: string, followed_by: list<string>}
20 *
21 * @visibility root
22 */
23final class PgLookahead
24{
25    /**
26     * Declares the token substitutions from PostgreSQL 17.2 parser.c/base_yylex.
27     * @return array<string, LookaheadRule>
28     */
29    public static function definitions(): array
30    {
31        return [
32            'FORMAT' => ['token' => 'FORMAT_LA', 'followed_by' => ['JSON']],
33            'NOT' => ['token' => 'NOT_LA', 'followed_by' => ['BETWEEN', 'IN_P', 'LIKE', 'ILIKE', 'SIMILAR']],
34            'NULLS_P' => ['token' => 'NULLS_LA', 'followed_by' => ['FIRST_P', 'LAST_P']],
35            'WITH' => ['token' => 'WITH_LA', 'followed_by' => ['TIME', 'ORDINALITY']],
36            'WITHOUT' => ['token' => 'WITHOUT_LA', 'followed_by' => ['TIME']],
37        ];
38    }

45    }

47    /**
48     * Applies the substitutions to a token sequence.
49     *
50     * @param list<string> $tokens Tokens as read or as generated
51     *
52     * @return list<string> The sequence with each triggered substitution applied
53     */
54    public function applied(array $tokens): array
55    {
56        foreach ($tokens as $index => $token) {
57            $rule = $this->rules[$token] ?? null;
58            if ($rule !== null && in_array($tokens[$index + 1] ?? null, $rule['followed_by'], true)) {
59                $tokens[$index] = $rule['token'];
60            }
61        }
62
63        return $tokens;
64    }

66    /**
67     * Settles each terminal on the spelling its neighbour calls for.
68     *
69     * Unlike applied(), this also walks a substitution backwards: a plan that
70     * asked for the substituted token where the follower does not call for it
71     * gets the base token, because that is what the text would read back as.
72     *
73     * @param list<string> $terminals Terminals a derivation produced
74     *
75     * @return list<string> The terminals with each substitution settled
76     */
77    public function normalized(array $terminals): array
78    {
79        foreach ($terminals as $index => $terminal) {
80            foreach ($this->rules as $base => $rule) {
81                if ($terminal !== $base && $terminal !== $rule['token']) {
82                    continue;
83                }
84
85                $terminals[$index] = in_array($terminals[$index + 1] ?? null, $rule['followed_by'], true)
86                    ? $rule['token']
87                    : $base;
88                break;
89            }
90        }
91
92        return $terminals;
93    }
94
95    /**
96     * Reports the token a substituted one is spelled as.
97     *
98     * A substituted token has no keyword of its own; it borrows the spelling of
99     * the token it replaced, so realization has to ask which that was.
100     *
101     * @param string $terminal Terminal to look up
102     *
103     * @return string|null The token it substitutes for, or null when it substitutes for none
104     */
105    public function baseOf(string $terminal): ?string
106    {
107        foreach ($this->rules as $base => $rule) {
108            if ($rule['token'] === $terminal) {
109                return $base;
110            }
111        }
112
113        return null;
114    }
