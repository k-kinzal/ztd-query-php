19    /**
20     * Inverts a terminal-to-spellings map.
21     *
22     * Spellings are upper-cased because SQL keywords are matched without
23     * regard to case, so the index is keyed the way a tokenizer will ask.
24     *
25     * @param array<string, list<string>> $keywords Spellings by terminal name
26     *
27     * @return array<string, string> Terminal name by upper-cased spelling
28     */
29    public function reversed(array $keywords): array
30    {
31        $index = [];
32        foreach ($keywords as $terminal => $spellings) {
33            foreach ($spellings as $spelling) {
34                $index[strtoupper($spelling)] = $terminal;
35            }
36        }
37
38        return $index;
39    }
