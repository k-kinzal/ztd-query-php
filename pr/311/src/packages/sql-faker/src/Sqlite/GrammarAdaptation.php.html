32     *
33     * Build-time conditionals have already been resolved by LemonParser.
34     * Every selected source alternative remains reachable through its alias.
35     *
36     * @param array<string, ProductionRule> $ruleMap Rules to add the statement rules to
37     * @param ProductionRule $cmd The `cmd` rule every statement is an alternative of
38     *

40     */
41    public function withStatementRules(array $ruleMap, ProductionRule $cmd): array
42    {
43        $groups = $this->statementAlternatives($cmd);
44        foreach (['insert', 'delete', 'update', 'drop_table', 'alter_table'] as $statement) {
45            if ($groups[$statement] !== []) {
46                $ruleMap[$statement] = new ProductionRule($statement, $groups[$statement]);

48        }

53    /**
54     * Sorts the alternatives of `cmd` by the statement each one writes.
55     *
56     * An alternative announces itself either with a keyword or, when it begins
57     * with an optional WITH clause, with the keyword right after it. ALTER and
58     * DROP say what they act on in their second word, and only the ones acting
59     * on a table are wanted here.
60     *
61     * @param ProductionRule $cmd The `cmd` rule every statement is an alternative of
62     *
63     * @return array<string, list<Production>> Alternatives by statement kind
64     */
65    public function statementAlternatives(ProductionRule $cmd): array
66    {
67        $groups = [
68            'insert' => [],
69            'delete' => [],
70            'update' => [],
71            'drop_table' => [],
72            'alter_table' => [],
73        ];
74
75        foreach ($cmd->alternatives as $alternative) {
76            $leading = $alternative->terminalAt(0);
77            if ($leading !== null) {
78                match ($leading->value) {
79                    'DELETE' => $groups['delete'][] = $alternative,
80                    'UPDATE' => $groups['update'][] = $alternative,
81                    'ALTER' => $alternative->terminalAt(1)?->value === 'TABLE'
82                        ? $groups['alter_table'][] = $alternative
83                        : null,
84                    'DROP' => $alternative->terminalAt(1)?->value === 'TABLE'
85                        ? $groups['drop_table'][] = $alternative
86                        : null,
87                    default => null,
88                };
89
90                continue;
91            }
92
93            $second = $alternative->nonTerminalAt(0) === null ? null : $alternative->terminalAt(1);
94            if ($second !== null) {
95                match ($second->value) {
96                    'DELETE' => $groups['delete'][] = $alternative,
97                    'UPDATE' => $groups['update'][] = $alternative,
98                    default => null,
99                };
100
101                continue;
102            }
103
104            if ($alternative->nonTerminalAt(0) !== null && $alternative->nonTerminalAt(1)?->value === 'insert_cmd') {
105                $groups['insert'][] = $alternative;
106            }
107        }
108
109        return $groups;
110    }
