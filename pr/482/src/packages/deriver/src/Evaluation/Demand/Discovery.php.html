29    /**
30     * Closes demanded definitions with a deterministic finite worklist.
31     * @param CallableGraph $callable Graph being evaluated
32     * @return array<string, true> Demanded instruction identifiers
33     */
34    public function instructions(CallableGraph $callable): array
35    {
36        $definitions = [];
37        $pending = [];
38        foreach ($callable->blocks as $block) {
39            $pending[] = $block->terminator->operand;
40            foreach ($block->instructions as $instruction) {
41                $definitions[$instruction->result] = $instruction;
42                if ($this->root($callable, $instruction)) {
43                    $pending[] = $instruction->result;
44                }
45            }
46        }
47        $needed = [];
48        while ($pending !== []) {
49            $register = array_pop($pending);
50            $instruction = $definitions[$register] ?? null;
51            if ($instruction === null || isset($needed[$instruction->id])) {
52                continue;
53            }
54            $needed[$instruction->id] = true;
55            array_push($pending, ...$instruction->operands);
56            foreach ($instruction->arguments as $argument) {
57                $pending[] = $argument->register;
58                $pending[] = $argument->location ?? '';
59            }
60            if (count($needed) > $this->context->query->budget()->nodes) {
61                $this->context->sealed = true;
62                $this->context->frontier('BUDGET_EXCEEDED', $instruction->source, 'dependency-discovery');
63                break;
64            }
65        }
66        return $needed;
67    }

69    /**
70     * Seeds observations, effects, and instructions that can warn or throw.
71     * @param CallableGraph $callable Owning graph
72     * @param Instruction $instruction Candidate definition
73     * @return bool Whether the definition must be evaluated independently of its result
74     */
75    public function root(CallableGraph $callable, Instruction $instruction): bool
76    {
77        if (!in_array($instruction->operation, ['constant', 'copy', 'phi', 'not-null', 'magic-constant'], true)) {
78            return true;
79        }
80        $query = $this->context->query;
81        if ($query instanceof ReturnQuery) {
82            return false;
83        }
84        if ($query instanceof ValueQuery) {
85            return (new CallableIdentity())->key($query->expression->callable) === (new CallableIdentity())->key($callable->symbol) && $query->expression->register === $instruction->result;
86        }
87        if ($query instanceof StateQuery || $query instanceof TupleQuery) {
88            if ((new CallableIdentity())->key($query->point->callable) !== (new CallableIdentity())->key($callable->symbol)) {
89                return false;
90            }
91            if ($query->point->instruction === $instruction->id) {
92                return true;
93            }
94            if ($query instanceof TupleQuery) {
95                foreach ($query->values as $reference) {
96                    if ($reference->register === $instruction->result) {
97                        return true;
98                    }
99                }
100            }
101        }
102        return false;
103    }
