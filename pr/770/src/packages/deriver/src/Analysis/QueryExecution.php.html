51     * Executes one query and builds a complete assessment.
52     * @param Query $query Requested observation
53     * @return DerivationResult Query result
54     * @throws JsonException If captured metadata cannot be encoded
55     */
56    public function derive(Query $query): DerivationResult
57    {
58        $start = microtime(true);
59        $context = new Context($this->program, $query, $this->configuration, $this->models, $this->shared);
60        $symbol = $this->owner($query);
61        $entries = $query->scope()->mode === 'symbolic' ? [new EntryPoint($symbol)] : $query->scope()->entries;
62        foreach ($entries as $entry) {
63            $this->entry($context, $entry, $query->scope()->mode === 'symbolic');
64        }
65        return $this->result($context, $symbol, $start);

69     * Builds one observation result, including bounded recovery after interruption.
70     * @param Context $context Observation context
71     * @param string $symbol Observation owner
72     * @param float $start Execution start
73     * @param string $executionIdentity Optional shared-execution identity
74     * @return DerivationResult Derived observation
75     * @throws JsonException If query metadata cannot be encoded
76     */
77    public function result(Context $context, string $symbol, float $start, string $executionIdentity = ''): DerivationResult
78    {
79        $query = $context->query;
80        if ($context->normal === [] && array_filter($context->frontiers, static fn ($frontier): bool => !in_array($frontier->code, ['WIDENED', 'EXTERNAL_INPUT', 'PHP_WARNING'], true)) !== []) {
81            $interrupted = $context->stopReason ?? (in_array('BUDGET_EXCEEDED', array_column($context->frontiers, 'code'), true) ? 'BUDGET_EXCEEDED' : null);
82            $candidate = $interrupted === null ? null : (new PartialObservation($this->program, $interrupted))->recover($query);
83            $context->normal[] = $candidate ?? new Alternative(['residual' => Term::opaque('INCOMPLETE_DERIVATION')]);
84        }
85        $assessment = (new ResultAssessment())->assess($context);
86        $assumptions = array_values(array_unique($context->assumptions));
87        sort($assumptions);
88        $stopped = $context->stopReason ?? (in_array('STACK_LIMIT', array_column($context->frontiers, 'code'), true) ? 'STACK_LIMIT' : null);
89        $interruption = $stopped === null ? '' : ':' . $stopped . ':' . $context->transfers;

91        $reached = $context->normal !== [] || ($query instanceof ReturnQuery && $context->exceptional !== []);
92        return new DerivationResult(new ResultRef($id), $this->snapshot->id, $query, $context->normal, $context->exceptional, $reached ? 95    /**
96     * Observes one symbolic callable once under a shared resource and logical budget.
97     * @param list<Query> $queries Observations in requested output order
98     * @return \Deriver\Result\ExecutionResultSet Independent observations from the shared execution
99     * @throws InvalidInputException If owners, scopes, or budgets differ
100     * @throws JsonException If query metadata cannot be encoded
101     */
102    public function together(array $queries): \Deriver\Result\ExecutionResultSet
103    {
104        if ($queries === []) {
105            return new \Deriver\Result\ExecutionResultSet([]);
106        }
107        $symbol = $this->owner($queries[0]);
108        $keys = [];
109        foreach ($queries as $query) {
110            if ($query->scope()->mode !== 'symbolic' || (new \Deriver\ControlFlow\CallableIdentity())->key($this->owner($query)) !== (new \Deriver\ControlFlow\CallableIdentity())->key($symbol) || get_object_vars($query->budget()) !== get_object_vars($queries[0]->budget())) {
111                throw new InvalidInputException('Shared execution requires symbolic queries for one callable with identical budgets.');
112            }
113            $keys[] = (new QueryEncoding())->key($query);
114        }
115        $identity = ':batch:' . hash('sha256', implode(':', $keys));
116        $start = microtime(true);
117        $context = new Context($this->program, $queries[0], $this->configuration, $this->models, $this->shared);
118        $batch = new \Deriver\Evaluation\BatchObservations($queries, $context);
119        $context->batch = $batch;
120        $this->entry($context, new EntryPoint($symbol), true);
121        $results = [];
122        foreach ($queries as $index => $_) {
123            $results[] = $this->result($batch->synchronize($index), $symbol, $start, $identity);
124        }
125        $context->batch = null;
126        return new \Deriver\Result\ExecutionResultSet($results);
127    }

129    /**
130     * Executes a symbolic callable or an explicitly initialized entry.
131     * @param Context $context Query context
132     * @param EntryPoint $entry Entry contract
133     * @param bool $symbolic Whether parameters represent all valid inputs
134     * @throws InvalidInputException If supplied receiver properties cannot describe the entry object
135     */
136    public function entry(Context $context, EntryPoint $entry, bool $symbolic): void
137    {
138        $body = $this->program->callable($entry->symbol);
139        if ($body === null) {
140            $context->frontier('INCOMPLETE_SOURCE', new SourceRef($this->snapshot->id, '', 0, 0), $entry->symbol);
141            return;
142        }
143        $machine = new Machine($context);
144        $context->entrySymbol = $body->symbol;
145        $arguments = [];
146        foreach ($entry->arguments as $name => $value) {
147            $arguments[] = new PassedArgument($value, is_string($name) ? $name : null);
148        }
149        $receiver = $body->static ? null : ($entry->receiver ?? ($body->className === '' ? null : Term::parameter('this', $body->className)));
150        if (array_diff(array_keys($entry->captures), array_keys($body->captures)) !== []) {
151            throw new InvalidInputException('Entry captures must name lexical captures declared by the closure.');
152        }
153        $captures = $entry->captures;
154        foreach ($body->captures as $name => $byReference) {
155            $captures[$name] ??= Term::parameter('capture:' . $name);
156        }
157        $initial = $this->initialState();
158        if ($entry->properties !== []) {
159            (new EntryProperties($context))->apply($body, $receiver, $entry->properties, $initial);
160        }
161        $states = (new ArgumentBinding($machine))->bind($body, $initial, $arguments, $receiver, $captures, symbolic: $symbolic || $entry->symbolicArguments);
162        foreach ($states as $state) {
163            if ($state->completion->kind === 'normal') {
164                $machine->run($body, $state);
165            } else {
166                (new ObservationCollector($context))->completion($body, $state);
167            }
168        }
169    }
