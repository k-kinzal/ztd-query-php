29        }
30        $pending = array_map(static fn (Term $term): array => [$term, false], [...array_values($bindings), ...array_values($values)]);
31        foreach (array_keys($memory->cells) as $root) {
32            if (str_starts_with($root, 'global:') || str_starts_with($root, 'static:')) {
33                $pending[] = [new Term('cell', $root), false];
34            }
35        }
36        $cells = [];
37        /** @var WeakMap<Term, bool> $seen */
38        $seen = new WeakMap();
39        while ($pending !== []) {
40            [$term, $confidential] = array_pop($pending);
41            $confidential = $confidential || $term->secret;
42            if (isset($seen[$term]) && ($seen[$term] || !$confidential)) {
43                continue;
44            }
45            $seen[$term] = $confidential;
46            foreach ($term->operands as $operand) {

48            }
49            foreach ($this->roots($term, $memory) as $root) {
50                if (!isset($cells[$root]) || $confidential && !$cells[$root]->secret) {

53                    $pending[] = [$raw, $confidential];
54                }
55            }
56        }
57        ksort($bindings);
58        ksort($cells);
59        return new StorageSnapshot($bindings, $cells);
60    }
