118     * Follows reference cells with cycle detection.
119     * @param Term $value Stored value
120     * @return Term Referenced value
121     */
122    public function dereference(Term $value): Term
123    {
124        $seen = [];
125        $secret = $value->secret;
126        while ($value->kind === 'cell' && is_string($value->literal)) {
127            if (isset($seen[$value->literal])) {
128                return new Term('opaque', 'CYCLIC_REFERENCE', attributes: ['type' => 'mixed', 'dependencyCoverage' => 'partial'], secret: $secret);
129            }
130            $seen[$value->literal] = true;
131            $value = $this->cells[$value->literal] ?? new Term('uninitialized');
132            $secret = $secret || $value->secret;
133        }
134        return $secret && !$value->secret ? new Term($value->kind, $value->literal, $value->operands, $value->attributes, true) : $value;
135    }

200    /**
201     * Materializes a reference cell only when an address escapes by reference.
202     * A global variable's slot then holds only a pointer to the shared cell, so unsetting the name leaves other references intact.
203     * @param Location $location Escaping address
204     * @return string Shared cell identity
205     */
206    public function reference(Location $location): string
207    {
208        if (!$location->unknown && $this->read($location)->kind === 'uninitialized') {
209            $this->write($location, Term::constant(null));
210        }
211        if ($location->path === []) {
212            $root = $location->root;
213            $seen = [];
214            while (($this->cells[$root] ?? null)?->kind === 'cell' && is_string($this->cells[$root]->literal) && !isset($seen[$root])) {
215                $seen[$root] = true;
216                $root = $this->cells[$root]->literal;
217            }
218            if (!$location->unknown && str_starts_with($root, 'global:')) {
219                $cell = $this->allocate($this->cells[$root] ?? Term::constant(null));
220                $this->cells[$root] = new Term('cell', $cell->root);
221                return $cell->root;
222            }
223            return $root;
224        }
225        $raw = $this->raw($location);
226        if ($raw->kind === 'cell' && is_string($raw->literal)) {
227            $value = $this->cells[$raw->literal] ?? new Term('uninitialized');
228            if ($this->read($location)->secret && !$value->secret) {
229                $this->cells[$raw->literal] = new Term($value->kind, $value->literal, $value->operands, $value->attributes, true);
230            }
231            return $raw->literal;
232        }
233        $cell = $this->allocate($this->read($location));
234        $this->write($location, new Term('cell', $cell->root));
235        return $cell->root;
236    }
